## LeetCode 217. 存在重复元素

[在线刷题](https://codefun2000.com/p/P4819)

给定一个整数数组 $nums$，请判断数组中是否存在重复元素。

如果任意一个值在数组中出现了至少两次，输出 $true$；如果数组中所有元素都互不相同，输出 $false$。

**输入** 

```
nums = [1,2,3,1]
```

**输出** 

```
true
```

* $1 <= nums.length <= 10^5$
* $-10^9 <= nums[i] <= 10^9$

**思路：哈希表**

结合[哈希表使用基础](https://codefun2000.com/codenote/hot100/P0018)所学到的语法知识，我们可以想到用一个哈希集合记录已经出现过的数字。

遍历数组时，如果当前数字已经在集合里，说明它之前出现过，直接返回 `true`；如果不在，就把它加入集合。

如果遍历结束都没发现重复元素，就返回 `false`。

```python
class Solution:
    def containsDuplicate(self, nums: List[int]) -> bool:
        # 使用哈希集合记录已经出现过的元素
        seen = set()

        # 遍历数组
        for num in nums:
            # 如果当前元素已经存在，说明有重复
            if num in seen:
                return True

            # 否则加入集合
            seen.add(num)

        # 遍历结束仍无重复
        return False
```

```c
bool containsDuplicate(int* nums, int numsSize) {
```

## Leetcode 1.两数之和

[在线刷题](https://codefun2000.com/p/P4000)

给定一个整数数组 $nums$ 和一个整数目标值 $target$，请你在该数组中找出 和为目标值 $target$ 的那两个整数，并输出它们的数组下标。

你可以假设每种输入只会对应一个答案，并且你不能使用两次相同的元素。

你可以按任意顺序输出答案。

**输入** 
```
4 9
2 7 11 15

```
**输出** 
```
0 1
```

**说明**

因为 $nums[0] + nums[1] == 9$ ，输出$0, 1$

**思路**

单层循环枚举右侧的数$num[j]$ ,这样目标就变成了，在下标$[0,j-1]$这个区域寻找一个数，使得其等于$target-num[j]$。

![](/file/2/kD5eISgpwooP0Ne7zmZtv.png)



从上图不难发现，$j$ 往右枚举的过程中，区域是有**前缀连续性**的。**每一个区域是前一个区域多一个元素,也就是图中每一组中带⭐的元素。**

所以对于要寻找的区间，我们可以用一个集合来维护 数的存在性，每次$j$ 往右挪一格时，就往集合里新增一个元素，查询答案就是查询当前下标$[0,j-1]$所构成的前缀哈希中是否出现过$target - num[j]$。

```python
class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        # 创建一个哈希表来存储元素及其下标
        num_to_index = {}

        # 遍历数组
        for i, num in enumerate(nums):
            # 计算需要的补数
            complement = target - num

            # 如果补数在哈希表中，返回对应的下标
            if complement in num_to_index:
                return [num_to_index[complement], i]

            # 否则，将当前元素及其下标存入哈希表
            num_to_index[num] = i

        # 根据题意，这种情况不会发生
        return []
```

```c
typedef struct {
    int key;
    int value;
    int used;
} HashEntry;

static int hashValue(int key, int capacity) {
    unsigned int x = (unsigned int)key;
    return x % capacity;
}

int* twoSum(int* nums, int numsSize, int target, int* returnSize) {
    // 创建哈希表
    int capacity = numsSize * 2 + 1;
    HashEntry* table = (HashEntry*)calloc(capacity, sizeof(HashEntry));

    // 遍历数组
    for (int i = 0; i < numsSize; i++) {
        int complement = target - nums[i];
        int pos = hashValue(complement, capacity);

        // 在哈希表中查找补数
        while (table[pos].used) {
            if (table[pos].key == complement) {
                int* ans = (int*)malloc(sizeof(int) * 2);
                ans[0] = table[pos].value;
                ans[1] = i;
                *returnSize = 2;
                free(table);
                return ans;
            }
            pos = (pos + 1) % capacity;
        }

        // 将当前元素及其下标加入哈希表
        pos = hashValue(nums[i], capacity);
        while (table[pos].used) {
            pos = (pos + 1) % capacity;
        }

        table[pos].key = nums[i];
        table[pos].value = i;
        table[pos].used = 1;
    }

    // 根据题意，这种情况不会发生
    *returnSize = 0;
    free(table);
    return NULL;
}
```

## Leetcode 49.字母异位词分组

[在线刷题](https://codefun2000.com/p/P4001)

给你一个字符串数组，请你将字母异位词组合在一起。可以按任意顺序输出结果列表。

字母异位词是由重新排列源单词的所有字母得到的一个新单词。

**输入** 

```
6
eat tea tan ate nat bat
```

**输出** 

```
bat
nat tan
ate eat tea
```

**思路1：排序+哈希**

不难想到，两个词为字母异位词，本质上就是两个字符串排序之后的字符串相等。

所以我们可以建立一个哈希表<键:字符串 , 值:排序前的字符串数组>

接着枚举每一个字符串，将它排序作为键归类，然后插入哈希表的值数组即可。
#code-switcher

```python
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        # 哈希表，key 为排序后的字符串，value 为同一组字母异位词
        mp = defaultdict(list)

        # 遍历所有字符串
        for s in strs:
            # 将字符串排序后作为分组依据
            key = "".join(sorted(s))
            mp[key].append(s)

        # 取出所有分组
        ans = list(mp.values())

        # 每组内部按字典序排序，保证输出稳定
        for group in ans:
            group.sort()

        # 分组之间先按长度排序，再按字典序排序，保证输出稳定
        ans.sort(key=lambda group: (len(group), group))

        # 返回分组结果
        return ans
```

```java
class Solution {
    public List<List<String>> groupAnagrams(String[] strs) {
        // 哈希表，key 为排序后的字符串，value 为同一组字母异位词
        Map<String, List<String>> map = new HashMap<>();

        // 遍历所有字符串
        for (String str : strs) {
            // 将字符串转为字符数组并排序
            char[] chars = str.toCharArray();
            Arrays.sort(chars);

            // 排序后的字符串作为分组依据
            String key = new String(chars);

            // 将原字符串加入对应分组
            map.computeIfAbsent(key, k -> new ArrayList<>()).add(str);
        }

        // 取出所有分组
        List<List<String>> ans = new ArrayList<>(map.values());

        // 每组内部按字典序排序，保证输出稳定
        for (List<String> group : ans) {
            Collections.sort(group);
        }

        // 分组之间先按长度排序，再按字典序排序，保证输出稳定
        ans.sort((a, b) -> {
            if (a.size() != b.size()) {
                return a.size() - b.size();
            }

            int len = Math.min(a.size(), b.size());
            for (int i = 0; i < len; i++) {
                int cmp = a.get(i).compareTo(b.get(i));
                if (cmp != 0) {
                    return cmp;
                }
            }

            return 0;
        });

        // 返回分组结果
        return ans;
    }
}
```

```javascript
    const map = new Map();

    // 遍历所有字符串
    for (const str of strs) {

        const key = str.split("").sort().join("");
```

```go
func groupAnagrams(strs []string) [][]string {
	// 哈希表，key 为排序后的字符串，value 为同一组字母异位词
	mp := make(map[string][]string)

	// 遍历所有字符串
	for _, str := range strs {
		// 将字符串排序后作为分组依据
		chars := []rune(str)
		sort.Slice(chars, func(i, j int) bool {
			return chars[i] < chars[j]
		})
		key := string(chars)

		// 将原字符串加入对应分组
		mp[key] = append(mp[key], str)
	}

	// 取出所有分组
	ans := make([][]string, 0, len(mp))
	for _, group := range mp {
		ans = append(ans, group)
	}

	// 每组内部按字典序排序，保证输出稳定
	for _, group := range ans {
		sort.Strings(group)
	}

	// 分组之间先按长度排序，再按字典序排序，保证输出稳定
	sort.Slice(ans, func(i, j int) bool {
		if len(ans[i]) != len(ans[j]) {
			return len(ans[i]) < len(ans[j])
		}

		for k := 0; k < len(ans[i]) && k < len(ans[j]); k++ {
			if ans[i][k] != ans[j][k] {
				return ans[i][k] < ans[j][k]
			}
		}

		return false
	})

	// 返回分组结果
	return ans
}
```

```c
typedef struct {
    char* key;
    char* value;
} Pair;

typedef struct {
    char** items;
    int size;
} Group;

static int cmpChar(const void* a, const void* b) {
    char x = *(const char*)a;
    char y = *(const char*)b;
    return (x > y) - (x < y);
}

static int cmpStringPtr(const void* a, const void* b) {
    const char* s1 = *(const char**)a;
    const char* s2 = *(const char**)b;
    return strcmp(s1, s2);
}

static int cmpPairByKey(const void* a, const void* b) {
    const Pair* p1 = (const Pair*)a;
    const Pair* p2 = (const Pair*)b;
    return strcmp(p1->key, p2->key);
}

static int cmpGroup(const void* a, const void* b) {
    const Group* g1 = (const Group*)a;
    const Group* g2 = (const Group*)b;

    // 先按分组长度排序
    if (g1->size != g2->size) {
        return g1->size - g2->size;
    }

    // 再按字典序排序
    for (int i = 0; i < g1->size && i < g2->size; i++) {
        int cmp = strcmp(g1->items[i], g2->items[i]);
        if (cmp != 0) {
            return cmp;
        }
    }

    return 0;
}

char*** groupAnagrams(
    char** strs,
    int strsSize,
    int* returnSize,
    int** returnColumnSizes
) {
    // 如果输入为空，直接返回空结果
    if (strsSize == 0) {
        *returnSize = 0;
        *returnColumnSizes = NULL;
        return NULL;
    }

    // 将每个字符串转换成排序后的 key
    Pair* pairs = (Pair*)malloc(sizeof(Pair) * strsSize);

    for (int i = 0; i < strsSize; i++) {
        int len = strlen(strs[i]);

        pairs[i].key = (char*)malloc(sizeof(char) * (len + 1));
        strcpy(pairs[i].key, strs[i]);

        // 对 key 的字符排序
        qsort(pairs[i].key, len, sizeof(char), cmpChar);

        // 保存原字符串
        pairs[i].value = strs[i];
    }

    // 按 key 排序，方便分组
    qsort(pairs, strsSize, sizeof(Pair), cmpPairByKey);

    // 临时保存所有分组
    Group* groups = (Group*)malloc(sizeof(Group) * strsSize);
    int groupCount = 0;

    int i = 0;
    while (i < strsSize) {
        int j = i;

        // 找到同一组字母异位词的范围
        while (j < strsSize && strcmp(pairs[i].key, pairs[j].key) == 0) {
            j++;
        }

        int groupSize = j - i;
        groups[groupCount].size = groupSize;
        groups[groupCount].items = (char**)malloc(sizeof(char*) * groupSize);

        // 保存这一组字符串
        for (int k = 0; k < groupSize; k++) {
            groups[groupCount].items[k] = pairs[i + k].value;
        }

        // 每组内部按字典序排序，保证输出稳定
        qsort(groups[groupCount].items, groupSize, sizeof(char*), cmpStringPtr);

        groupCount++;
        i = j;
    }

    // 分组之间先按长度排序，再按字典序排序，保证输出稳定
    qsort(groups, groupCount, sizeof(Group), cmpGroup);
```

### 优化：利用“桶思想”进行信息压缩

上述方法，唯一可以优化的点就是排序归类这一步。

其实除了排序来比较相等+哈希归类，还可以使用"频次表"来比较相等+归类。

举个例子:
 
`aaabbc` 的 频次表为$[3,2,1,0,0,...,0]$ (长度为26)

`cbabaa` 的 频次表为$[3,2,1,0,0,...,0]$ (长度为26)

`cbabzz` 的 频次表为$[1,2,1,0,0,...,2]$ (长度为26)

换句话说，任何一个小写字母构成的字符串，都可以被压缩成一个长度为26的整数数组$counts[] , counts[i] 代表这个字符串中第i个字母出现的次数$。

压缩之后，直接比较出现两个字符串的counts数组是否全等即可。

这样的好处是无论这个数组有多长，我们都可以把它压缩成一个长度固定为26的数组去表征。

这里需要注意的是，哈希表中的键需要经过哈希函数，而counts数组作为键，需要定义它的哈希函数，具体看代码实现！

> 说人话就是，哈希表主体是个数组，它的键$x$的值存储在$hash[f(x)]$ 中。
> 而在各个语言所提供的哈希表实现中，没有给数组定义对应的哈希函数，所以需要做一定的处理。

```c
typedef struct {
    char* key;
    char* value;
} Pair;

static int cmpPair(const void* a, const void* b) {
    const Pair* p1 = (const Pair*)a;
    const Pair* p2 = (const Pair*)b;
    return strcmp(p1->key, p2->key);
}

char*** groupAnagrams(
    char** strs,
    int strsSize,
    int* returnSize,
    int** returnColumnSizes
) {
    // 如果输入为空，直接返回空结果
    if (strsSize == 0) {
        *returnSize = 0;
        *returnColumnSizes = NULL;
        return NULL;
    }

    // 保存每个字符串对应的哈希键
    Pair* pairs = (Pair*)malloc(sizeof(Pair) * strsSize);

    for (int i = 0; i < strsSize; i++) {
        // 定义出现次数表
        int counts[26] = {0};

        // 统计每个字符出现次数
        for (int j = 0; strs[i][j] != '\0'; j++) {
            counts[strs[i][j] - 'a']++;
        }

        // 将出现次数表转换成字符串作为哈希键
        char buffer[512] = "";
        char temp[32];

        for (int j = 0; j < 26; j++) {
            sprintf(temp, "%d#", counts[j]);
            strcat(buffer, temp);
            // 拼接时加入分隔符，避免哈希键冲突
        }

        pairs[i].key = (char*)malloc(strlen(buffer) + 1);
        strcpy(pairs[i].key, buffer);
        pairs[i].value = strs[i];
    }

    // 按哈希键排序，方便后续分组
    qsort(pairs, strsSize, sizeof(Pair), cmpPair);

    // 最多会有 strsSize 个分组
    char*** ans = (char***)malloc(sizeof(char**) * strsSize);
    *returnColumnSizes = (int*)malloc(sizeof(int) * strsSize);

    int groupCount = 0;
    int i = 0;

    while (i < strsSize) {
        int j = i;

        // 找出当前这一组的范围
        while (j < strsSize && strcmp(pairs[i].key, pairs[j].key) == 0) {
            j++;
        }

        int groupSize = j - i;

        // 保存这一组字符串
        ans[groupCount] = (char**)malloc(sizeof(char*) * groupSize);
        for (int k = 0; k < groupSize; k++) {
            ans[groupCount][k] = pairs[i + k].value;
        }

        (*returnColumnSizes)[groupCount] = groupSize;
        groupCount++;

        i = j;
    }

    // 设置返回分组数量
    *returnSize = groupCount;

    // 释放临时哈希键
    for (int k = 0; k < strsSize; k++) {
        free(pairs[k].key);
    }
    free(pairs);

    // 返回所有分组结果
    return ans;
}
```

### 面试问答

**1.这道题的整体思路是什么？**

使用哈希表来对字符串进行分组。可以将每个字符串排序，排序后的字符串用作哈希表的键，而该字符串则作为值存入哈希表中。由于字母异位词的字母组成相同，因此它们经过排序后会得到相同的字符串，如此进行分组。在最终输出时，只需返回哈希表中的所有值，每个值即为一个字母异位词的组合。

时间复杂度为$O(nklogk)$ , 其中$k$是所有字符串中最长字符串的长度

**2.是否有更优的做法？**

这个算法的复杂度瓶颈在于就是字符串排序。观察到字符集大小为26，可以利用“桶”的思想，对字符串进行统计字母出现频次，由于字符集大小为26，这个出现频次表数组的长度固定为26。

如此我们只需要拿频次表作为哈希的键来存储，而不再需要先对它排序，复杂度为$O(nk)$

```c
typedef struct {
    int key;
    int used;
} HashNode;

static unsigned int hashInt(int x) {
    return (unsigned int)x * 2654435761u;
}

static int contains(HashNode* table, int size, int key) {
    unsigned int idx = hashInt(key) % size;

    while (table[idx].used) {
        if (table[idx].key == key) {
            return 1;
        }
        idx = (idx + 1) % size;
    }

    return 0;
}

static void insert(HashNode* table, int size, int key) {
    unsigned int idx = hashInt(key) % size;

    while (table[idx].used) {
        if (table[idx].key == key) {
            return;
        }
        idx = (idx + 1) % size;
    }

    table[idx].used = 1;
    table[idx].key = key;
}

int longestConsecutive(int* nums, int numsSize) {
    if (numsSize == 0) {
        return 0;
    }

    int tableSize = numsSize * 3 + 1;
    HashNode* num_set = (HashNode*)calloc(tableSize, sizeof(HashNode)); // 将数组存入哈希集合

    for (int i = 0; i < numsSize; i++) {
        insert(num_set, tableSize, nums[i]);
    }

    int longest_streak = 0;

    for (int i = 0; i < numsSize; i++) {
        // 枚举当前位置作为起点
        int current_num = nums[i];
        int current_streak = 1;

        while (contains(num_set, tableSize, current_num + 1)) { // 计算序列长度
            current_num += 1;
            current_streak += 1;
        }

        if (current_streak > longest_streak) { // 更新最大长度
            longest_streak = current_streak;
```

$$[x_1,x_2,x_3,...,x_k]$$

中的开头元素：$x_1$ 一定满足 : 数组中不存在$y$ 满足$y = x_1 - 1$

原因很简单，因为如果有这样的$y = x_1 - 1$ , 那么显然最优解可以更大。

因为那些存在$x - 1$ 的 $x$ 作为 起点，一定不会比现在的答案更优。

```c
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int key;
    bool used;
} HashEntry;

static unsigned int hashValue(int key, int capacity) {
    unsigned int x = (unsigned int)key;
    x ^= x >> 16;
    return x % capacity;
}

static bool contains(HashEntry* table, int capacity, int key) {
    unsigned int pos = hashValue(key, capacity);

    while (table[pos].used) {
        if (table[pos].key == key) {
            return true;
        }
        pos = (pos + 1) % capacity;
    }

    return false;
}

static void insert(HashEntry* table, int capacity, int key) {
    unsigned int pos = hashValue(key, capacity);

    while (table[pos].used) {
        if (table[pos].key == key) {
            return;
        }
        pos = (pos + 1) % capacity;
    }

    table[pos].key = key;
    table[pos].used = true;
}

int longestConsecutive(int* nums, int numsSize) {
    // 如果数组为空，直接返回 0
    if (numsSize == 0) {
        return 0;
    }

    // 创建哈希集合
    int capacity = numsSize * 3 + 1;
    HashEntry* table = (HashEntry*)calloc(capacity, sizeof(HashEntry));

    // 将数组元素加入哈希集合
    for (int i = 0; i < numsSize; i++) {
        insert(table, capacity, nums[i]);
    }

    // 记录最长连续序列长度
    int longestStreak = 0;

    // 遍历原数组
    for (int i = 0; i < numsSize; i++) {
        int num = nums[i];

        // 重点：只有当 num - 1 不存在时，num 才可能是连续序列的起点
        if (!contains(table, capacity, num - 1)) {
            int currentNum = num;
            int currentStreak = 1;

            // 向右扩展，计算当前连续序列长度
            while (contains(table, capacity, currentNum + 1)) {
                currentNum++;
                currentStreak++;
            }

            // 更新最大长度
            if (currentStreak > longestStreak) {
                longestStreak = currentStreak;
            }
        }
    }

    free(table);

    // 返回最长连续序列长度
    return longestStreak;
}
```

## Leetcode 560.和为k的子数组

[在线刷题](https://codefun2000.com/p/P4009)

给你一个整数数组$nums$和一个整数$k$，请你统计并输出 该数组中和为 $k$ 的子数组的个数。

子数组是数组中元素的连续非空序列。

**输入** 

```
3 2
1 1 1
```

**输出** 

```
2
```

**提示**

- $1 <= nums.length <= 2 * 10^4$
- $-1000 <= nums[i] <= 1000$
- $-10^7 <= k <= 10^7$

**思路：前缀和 + 哈希**

考虑一个子数组$[a_i,a_{i+1},a_{i+2},...,a_{j}]$的和，

它可以被表示为两段**前缀数组**的差分：

$$a_i+a_{i+1}+a_{i+2}+...+a_{j} =  (a_1+a_{2}+a_{3}+...+a_{j}) - (a_1+a_{2}+a_{3}+...+a_{i-1})$$

如下图所示:

![](/file/2/7DCac19z70Jzndp4M45sC.png =70%x)


那么我们可以先计算出前缀和数组:$suf[i] = a_1+a_2+...+a_{i} = suf[i - 1] + a_{i}$

如此，任意一个子数组$(i,j)$的和都可以通过访问前缀和数组$suf[j] - suf[i - 1]$来$O(1)$计算。

#### 问题转化：**两数之和**

$a_i+a_{i+1}+a_{i+2}+...+a_{j} = k$ 被我们通过前缀和数组转化为:

$suf[j] - suf[i - 1] = k$

此时问题转化为：在前缀和数组$suf$中，选出两个数使得它们的差是$k。

本题和两数之和的变化在于：两数之和只需要知道其中一组解，而本题需要计算解的数量。

所以哈希存储的是<键:前缀和 , 值:该值出现的次数>

```python
class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        n = len(nums)

        # 求解前缀和
        suf = [0] * (n + 1)

        # 为了更好地处理边界情况，值从 1 开始存储
        # suf[0] = 0 表示空子数组的前缀和
        for i in range(1, n + 1):
            suf[i] = suf[i - 1] + nums[i - 1]

        # 记录满足条件的子数组数量
        ans = 0

        # 哈希表，记录前缀和出现的次数
        st = {}
        st[suf[0]] = 1

        for i in range(1, n + 1):
            # 在之前的前缀和中寻找 suf[i] - k
            target = suf[i] - k

            if target in st:
                ans += st[target]

            # 将当前前缀和加入哈希表
            st[suf[i]] = st.get(suf[i], 0) + 1

        return ans
```

```c
typedef struct {
    long long key;
    int value;
    bool used;
} HashEntry;

static unsigned int hashValue(long long key, int capacity) {
    unsigned long long x = (unsigned long long)key;
    x ^= x >> 33;
    x *= 0xff51afd7ed558ccdULL;
    x ^= x >> 33;
    return (unsigned int)(x % capacity);
}

static int getCount(HashEntry* table, int capacity, long long key) {
    unsigned int pos = hashValue(key, capacity);

    while (table[pos].used) {
        if (table[pos].key == key) {
            return table[pos].value;
        }
        pos = (pos + 1) % capacity;
    }

    return 0;
}

static void addCount(HashEntry* table, int capacity, long long key) {
    unsigned int pos = hashValue(key, capacity);

    while (table[pos].used) {
        if (table[pos].key == key) {
            table[pos].value++;
            return;
        }
        pos = (pos + 1) % capacity;
    }

    table[pos].used = true;
    table[pos].key = key;
    table[pos].value = 1;
}

    long long* suf = (long long*)malloc(sizeof(long long) * (n + 1));
    suf[0] = 0;

    // suf[0] = 0 表示空子数组的前缀和

        long long target = suf[i] - k;
```
