## 拓扑排序

拓扑排序是一种**基于BFS**思想的算法，专门用来解决 **有先后依赖关系的任务安排问题**。

### **从一个例子出发**

如下图所示的深度学习课程入门路径图，每个箭头指向的是先修课程👇

![](/file/2/YbCqJNLlpeKSbzn4I4uPd.png =70%x)



拓扑排序算法可以得到一种具体的学习路径序列，例如

这种路径序列又称作**拓扑排序序列**。

### 具体算法

拓扑排序的每一步都在图中寻找一个 **没有先修课程** 的节点，记录下来，并将它和它的连边从图中删除，直到整个图为空。这个过程可以用BFS的思想来完成。

这里我们用一个动画演示拓扑排序的实际计算过程👇

@[video](https://codefun2000.com/p/4099/file/%E6%8B%93%E6%89%91%E6%8E%92%E5%BA%8F%E5%9F%BA%E6%9C%AC%E6%B5%81%E7%A8%8B.mp4)


ps:在图论中，课程的先修关系个数 也就是节点的**出度(out-degree)**。

### **算法限制**

拓扑排序要求输入的是一个有向无环图(DAG)，因为如果课程中有互相依赖的课程集合（成环），则该图不存在拓扑排序序列👇

## LeetCode 207.课程表
[在线刷题](https://codefun2000.com/p/P4068)
你这个学期必须选修 $numCourses$ 门课程，记为 $0$到 $numCourses - 1$ 。

在选修某些课程之前需要一些先修课程。 先修课程按数组 $prerequisites$ 给出，其中 $prerequisites[i] = [ai, bi]$ ，表示如果要学习课程 $a_i$则 必须 先学习课程 $b_i$ 。

- 例如，先修课程对 $[0, 1]$ 表示：想要学习课程 $0$，你需要先完成课程 $1$ 。

请你判断是否可能完成所有课程的学习？如果可以，返回 $true$ ；否则，返回 $false$ 。

**输入描述**

- 第一行输入一个整数 $numCourses$，表示课程总数。
- 第二行输入一个整数 $m$，表示先修课程对的数量。
- 接下来 $m$ 行，每行输入两个整数 $a_i b_i$，表示课程 $a_i$ 需要先修 $b_i$。

**输出描述**


如果可以完成所有课程的学习，输出 true，否则输出 false。

**样例1** 

**输入** 

```
2
1
1 0
```

**输出** 

```
true
```

说明

总共有 $2 $门课程。学习课程 $1$ 之前，你需要完成课程$ 0$ 。这是可能的。

**样例2**

**输入** 

```
2
2
1 0
0 1
```

**输出** 

```
false
```

**说明**

总共有 $2$ 门课程。学习课程 $1 $之前，你需要先完成课程$ 0 $；并且学习课程$ 0 $之前，你还应先完成课程 $1$ 。这是不可能的。

**提示**

- $1 <= numCourses <= 2000$
- $0 <= prerequisites.length <= 5000$
- $prerequisites[i].length == 2$
- $0 <= a_i, b_i < numCourses$
- $prerequisites[i]$中的所有课程对互不相同



本题本质上就是给定一张**有向图**，判断是否存在一个合法的拓扑序列。

在基础知识部分说过了，**图中如果存在环，则不存在合法的拓扑序列。**

但我们在代码实现的过程中无需特别判环，只需要统计已经处理的课程个数，如果其大小不等于$numCourses$ , 则代表图中有环无法被消解，则输出$false$ , 否则输出$true$

**特别注意**，在算法讲解的时候，我们的依赖关系是正的，每次根据后驱寻找前驱。**但是代码实现过程中，通过邻接表找前驱并不好实现，真正好实现的是通过前驱找后驱，所以将关系反着来建边**，这样才能更方便的通过邻接表寻找那些依赖的关系。

```python
class Solution:
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        # edges[x] 存放以 x 为先修课的所有后续课程
        edges = [[] for _ in range(numCourses)]

        # in_degree[i] 表示学习课程 i 之前，还需要完成多少门先修课
        in_degree = [0] * numCourses

        # 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
        for a, b in prerequisites:
            edges[b].append(a)
            in_degree[a] += 1

        # 所有入度为 0 的课程，都可以最开始学习
        q = []
        for i in range(numCourses):
            if in_degree[i] == 0:
                q.append(i)

        # count 记录已经可以完成学习的课程数量
        count = 0

        # 拓扑排序，不断消除入度为 0 的课程
        while q:
            u = q.pop()
            count += 1

            # 学完课程 u 后，它的后续课程少一个先修依赖
            for v in edges[u]:
                in_degree[v] -= 1
                if in_degree[v] == 0:
                    q.append(v)

        # 如果所有课程都被处理，说明不存在环，可以完成所有课程
        return count == numCourses
```

```cpp
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        // edges[x] 存放以 x 为先修课的所有后续课程
        vector<vector<int>> edges(numCourses);

        // inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
        vector<int> inDegree(numCourses, 0);

        // 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
        for (auto& p : prerequisites) {
            int a = p[0];
            int b = p[1];
            edges[b].push_back(a);
            inDegree[a]++;
        }

        // 所有入度为 0 的课程，都可以最开始学习
        vector<int> q;
        for (int i = 0; i < numCourses; i++) {
            if (inDegree[i] == 0) {
                q.push_back(i);
            }
        }

        // count 记录已经可以完成学习的课程数量
        int count = 0;

        // 拓扑排序，不断消除入度为 0 的课程
        while (!q.empty()) {
            int u = q.back();
            q.pop_back();

            for (int v : edges[u]) {

                if (inDegree[v] == 0) {
                    q.push_back(v);
                }
            }
        }

        return count == numCourses;
    }
};
```

```java
class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        // edges[x] 存放以 x 为先修课的所有后续课程
        List<List<Integer>> edges = new ArrayList<>();
        for (int i = 0; i < numCourses; i++) {
            edges.add(new ArrayList<>());
        }

        // inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
        int[] inDegree = new int[numCourses];

        // 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
        for (int[] p : prerequisites) {
            int a = p[0];
            int b = p[1];
            edges.get(b).add(a);
            inDegree[a]++;
        }

        // 所有入度为 0 的课程，都可以最开始学习
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < numCourses; i++) {
            if (inDegree[i] == 0) {
                queue.offer(i);
            }
        }

        // count 记录已经可以完成学习的课程数量
        int count = 0;

        // 拓扑排序，不断消除入度为 0 的课程
        while (!queue.isEmpty()) {
            int u = queue.poll();
            count++;

            // 学完课程 u 后，它的后续课程少一个先修依赖
            for (int v : edges.get(u)) {
                inDegree[v]--;
                if (inDegree[v] == 0) {
                    queue.offer(v);
                }
            }
        }

        // 如果所有课程都被处理，说明不存在环，可以完成所有课程
        return count == numCourses;
    }
}
```

```go
func canFinish(numCourses int, prerequisites [][]int) bool {
	// edges[x] 存放以 x 为先修课的所有后续课程
	edges := make([][]int, numCourses)

	// inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
	inDegree := make([]int, numCourses)

	// 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
	for _, p := range prerequisites {
		a, b := p[0], p[1]
		edges[b] = append(edges[b], a)
		inDegree[a]++
	}

	// 所有入度为 0 的课程，都可以最开始学习
	q := make([]int, 0)
	for i := 0; i < numCourses; i++ {
		if inDegree[i] == 0 {
			q = append(q, i)
		}
	}

	// count 记录已经可以完成学习的课程数量
	count := 0
	head := 0

	// 拓扑排序，不断消除入度为 0 的课程
	for head < len(q) {
		u := q[head]
		head++
		count++

		// 学完课程 u 后，它的后续课程少一个先修依赖
		for _, v := range edges[u] {
			inDegree[v]--
			if inDegree[v] == 0 {
				q = append(q, v)
			}
		}
	}

	// 如果所有课程都被处理，说明不存在环，可以完成所有课程
	return count == numCourses
}
```

```c
bool canFinish(int numCourses, int** prerequisites, int prerequisitesSize, int* prerequisitesColSize) {
    // head[x] 表示以 x 为先修课的第一条边
    int* head = (int*)malloc((size_t)numCourses * sizeof(int));

    // to[i] 表示第 i 条边指向的后续课程，next[i] 表示同一起点的下一条边
    int* to = (int*)malloc((size_t)prerequisitesSize * sizeof(int));
    int* next = (int*)malloc((size_t)prerequisitesSize * sizeof(int));

    // inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
    int* inDegree = (int*)malloc((size_t)numCourses * sizeof(int));

    // 队列中存放所有当前入度为 0 的课程
    int* q = (int*)malloc((size_t)numCourses * sizeof(int));

    for (int i = 0; i < numCourses; i++) {
        head[i] = -1;
        inDegree[i] = 0;
    }

    // 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
    for (int i = 0; i < prerequisitesSize; i++) {
        int a = prerequisites[i][0];
        int b = prerequisites[i][1];

        to[i] = a;
        next[i] = head[b];
        head[b] = i;
        inDegree[a]++;
    }

    // 所有入度为 0 的课程，都可以最开始学习
    int front = 0;
    int rear = 0;
    for (int i = 0; i < numCourses; i++) {
        if (inDegree[i] == 0) {
            q[rear++] = i;
        }
    }

    // count 记录已经可以完成学习的课程数量
    int count = 0;

    // 拓扑排序，不断消除入度为 0 的课程
    while (front < rear) {
        int u = q[front++];
        count++;

        // 学完课程 u 后，它的后续课程少一个先修依赖
        for (int e = head[u]; e != -1; e = next[e]) {
            int v = to[e];
            inDegree[v]--;

            if (inDegree[v] == 0) {
                q[rear++] = v;
            }
        }
    }

    bool ans = count == numCourses;

    free(head);
    free(to);
    free(next);
    free(inDegree);
    free(q);

    // 如果所有课程都被处理，说明不存在环，可以完成所有课程
    return ans;
}
```
