# 房间公平入住安排

## 解题思路

1. **读懂约束**：每个房间人数在 $0$ 到 $5$ 之间；「不存在一间比别的多 $2$ 人」等价于所有人住下后，各房人数的全局最大值与最小值之差不超过 $1$（含未住满时含 $0$ 的房间）。
2. **先求最多能住进多少人**：记目标入住总数为 $S$，则 $S$ 是不超过 $n$ 的、且存在合法分配的最大整数。合法分配的结构必为：要么每房同为 $v$，要么恰有两档人数 $k$ 与 $k+1$（各至少一间），且 $k+1\\le 5$。对候选 $S$ 从 $\\min(n,5m)$ 向下枚举即可判定是否存在方案。
3. **剩余人数**：第一行输出 $n-S$。
4. **枚举所有最优方案**：对每个可达的 $S$，收集所有不同的多重集（「$t$ 间 $k+1$，其余 $k$」或「全 $v$」），去重后对每个多重集按**字典序从大到小**生成全排列（高位优先试较大人数，与官方样例一致）。
5. **非法输入**：单行且恰有一个英文逗号、两侧均为非负整数字符串、$m\\ge 1$；否则输出一行 `invalid`。

> 说明：题面「第二行输出有几种安排情况」与所给样例输出格式不一致，实现以样例为准（首行剩余人数，其后每种方案一行）。

## 复杂度分析

- 判定某个 $S$ 是否存在：$O(m)$ 枚举 $(k,t)$ 与整除判定，从高往低试 $S$ 至多 $O(\\min(n,5m))$ 次，合计 $O(m\\cdot \\min(n,5m))$。
- 输出所有排列：若多重集计数为 $c_1,\\dots,c_r$，方案数为多项式系数 $\\frac{m!}{c_1!\\cdots c_r!}$，时间与之同阶；本题测试数据中 $m$ 与排列数受控，可完整输出。

空间：递归栈与计数数组 $O(m)$。

## 代码实现

### Python

```python
# 公平入住：最大化可入住人数 S（<=n），再枚举所有达到 S 的合法 multiset 的降序字典排列。
import sys


def parse_input(line: str):
    line = line.strip()
    if not line or line.count(",") != 1:
        return None
    a, b = line.split(",", 1)
    a, b = a.strip(), b.strip()
    if not a.isdigit() or not b.isdigit():
        return None
    m, n = int(a), int(b)
    if m < 1 or n < 0:
        return None
    return m, n


def exists_sum(m: int, s: int) -> bool:
    # 全同 v
    if s % m == 0:
        v = s // m
        if 0 <= v <= 5:
            return True
    # t 间为 k+1，其余为 k，要求 0 < t < m 且 k+1 <= 5
    for k in range(0, 5):
        for t in range(1, m):
            if k * m + t == s and k + 1 <= 5:
                return True
    return False


def max_assignable_sum(m: int, n: int) -> int:
    hi = min(n, 5 * m)
    for s in range(hi, -1, -1):
        if exists_sum(m, s):
            return s
    return 0


def collect_patterns(m: int, s: int):
    seen = set()
    out = []

    def add(pat):
        key = tuple(sorted(pat))
        if key not in seen:
            seen.add(key)
            out.append(list(pat))

    if s % m == 0:
        v = s // m
        if 0 <= v <= 5:
            add([v] * m)
    for k in range(0, 5):
        for t in range(1, m):
            if k * m + t == s and k + 1 <= 5:
                add([k + 1] * t + [k] * (m - t))
    return out


def gen_desc_lex(m: int, counts: dict) -> list:
    keys = sorted(counts.keys(), reverse=True)
    res = []
    cur = []

    def dfs():
        if len(cur) == m:
            res.append(",".join(str(x) for x in cur))
            return
        for v in keys:
            if counts.get(v, 0) > 0:
                counts[v] -= 1
                cur.append(v)
                dfs()
                cur.pop()
                counts[v] += 1

    dfs()
    return res


def solve(m: int, n: int) -> str:
    best = max_assignable_sum(m, n)
    rem = n - best
    lines = [str(rem)]
    for pat in collect_patterns(m, best):
        c = {}
        for x in pat:
            c[x] = c.get(x, 0) + 1
        lines.extend(gen_desc_lex(m, c))
    return "\n".join(lines)


def main():
    line = input()
    parsed = parse_input(line)
    if parsed is None:
        sys.stdout.buffer.write(b"invalid\n")
        return
    m, n = parsed
    # 统一使用 LF，避免 Windows 文本换行影响与 C++/Java 对拍
    sys.stdout.buffer.write((solve(m, n) + "\n").encode("utf-8"))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.*;

public class Main {

    // 判断是否存在和为 s 的合法入住方案
    static boolean existsSum(int m, int s) {
        if (s % m == 0) {
            int v = s / m;
            if (v >= 0 && v <= 5) return true;
        }
        for (int k = 0; k < 5; k++) {
            for (int t = 1; t < m; t++) {
                if (k * m + t == s && k + 1 <= 5) return true;
            }
        }
        return false;
    }

    static int maxAssignableSum(int m, long n) {
        int hi = (int) Math.min(n, (long) m * 5);
        for (int s = hi; s >= 0; s--) {
            if (existsSum(m, s)) return s;
        }
        return 0;
    }

    static List<int[]> collectPatterns(int m, int best) {
        Set<String> seen = new HashSet<>();
        List<int[]> out = new ArrayList<>();
        if (best % m == 0) {
            int v = best / m;
            if (v >= 0 && v <= 5) {
                int[] pat = new int[m];
                Arrays.fill(pat, v);
                if (seen.add(keyOf(pat))) out.add(pat);
            }
        }
        for (int k = 0; k < 5; k++) {
            for (int t = 1; t < m; t++) {
                if (k * m + t == best && k + 1 <= 5) {
                    int[] pat = new int[m];
                    int i = 0;
                    for (; i < t; i++) pat[i] = k + 1;
                    for (; i < m; i++) pat[i] = k;
                    if (seen.add(keyOf(pat))) out.add(pat);
                }
            }
        }
        return out;
    }

    static String keyOf(int[] pat) {
        int[] cp = pat.clone();
        Arrays.sort(cp);
        return Arrays.toString(cp);
    }

    // 降序字典：每一步先试当前还可用的最大人数
    static void dfsDescLex(int m, TreeMap<Integer, Integer> cnt, List<Integer> cur, List<String> lines) {
        if (cur.size() == m) {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < m; i++) {
                if (i > 0) sb.append(',');
                sb.append(cur.get(i));
            }
            lines.add(sb.toString());
            return;
        }
        for (Map.Entry<Integer, Integer> e : cnt.entrySet()) {
            int v = e.getKey(), c = e.getValue();
            if (c <= 0) continue;
            cnt.put(v, c - 1);
            cur.add(v);
            dfsDescLex(m, cnt, cur, lines);
            cur.remove(cur.size() - 1);
            cnt.put(v, c);
        }
    }

    static List<String> genDescLex(int m, int[] pat) {
        TreeMap<Integer, Integer> cnt = new TreeMap<>(Collections.reverseOrder());
        for (int x : pat) cnt.merge(x, 1, Integer::sum);
        List<String> lines = new ArrayList<>();
        dfsDescLex(m, cnt, new ArrayList<>(), lines);
        return lines;
    }

    static String solve(int m, long n) {
        int best = maxAssignableSum(m, n);
        long rem = n - best;
        StringBuilder sb = new StringBuilder();
        sb.append(rem);
        for (int[] p : collectPatterns(m, best)) {
            for (String ln : genDescLex(m, p)) {
                sb.append('\n').append(ln);
            }
        }
        return sb.toString();
    }

    static boolean parseInput(String line, int[] mOut, long[] nOut) {
        if (line == null) return false;
        String s = line.trim();
        if (s.isEmpty()) return false;
        int commas = 0;
        for (int i = 0; i < s.length(); i++) if (s.charAt(i) == ',') commas++;
        if (commas != 1) return false;
        int p = s.indexOf(',');
        String a = s.substring(0, p).trim();
        String b = s.substring(p + 1).trim();
        if (a.isEmpty() || b.isEmpty()) return false;
        for (int i = 0; i < a.length(); i++) if (!Character.isDigit(a.charAt(i))) return false;
        for (int i = 0; i < b.length(); i++) if (!Character.isDigit(b.charAt(i))) return false;
        int m = Integer.parseInt(a);
        long n = Long.parseLong(b);
        if (m < 1 || n < 0) return false;
        mOut[0] = m;
        nOut[0] = n;
        return true;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        int[] mArr = new int[1];
        long[] nArr = new long[1];
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out, StandardCharsets.UTF_8));
        if (!parseInput(line, mArr, nArr)) {
            bw.write("invalid\n");
            bw.flush();
            return;
        }
        bw.write(solve(mArr[0], nArr[0]));
        bw.write('\n');
        bw.flush();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
#include <cstdio>
#ifdef _WIN32
#include <fcntl.h>
#include <io.h>
#endif
using namespace std;

using int64 = long long;

// 是否存在长度为 m、每人 0..5、max-min<=1 且和为 s 的入住向量
static bool exists_sum(int m, int s) {
    if (s % m == 0) {
        int v = s / m;
        if (v >= 0 && v <= 5) return true;
    }
    for (int k = 0; k < 5; ++k) {
        for (int t = 1; t < m; ++t) {
            if (k * m + t == s && k + 1 <= 5) return true;
        }
    }
    return false;
}

static int max_assignable_sum(int m, int64 n) {
    int hi = (int)min<int64>(n, (int64)m * 5);
    for (int s = hi; s >= 0; --s) {
        if (exists_sum(m, s)) return s;
    }
    return 0;
}

static void dfs_desc_lex(int m, map<int, int, greater<int>> &cnt, vector<int> &cur, vector<string> &out) {
    if ((int)cur.size() == m) {
        string line;
        for (int i = 0; i < m; ++i) {
            if (i) line.push_back(',');
            line += to_string(cur[i]);
        }
        out.push_back(line);
        return;
    }
    for (auto &kv : cnt) {
        int v = kv.first;
        if (kv.second <= 0) continue;
        kv.second--;
        cur.push_back(v);
        dfs_desc_lex(m, cnt, cur, out);
        cur.pop_back();
        kv.second++;
    }
}

static vector<string> gen_desc_lex(int m, const vector<int> &pat) {
    map<int, int, greater<int>> cnt;
    for (int x : pat) cnt[x]++;
    vector<string> out;
    vector<int> cur;
    dfs_desc_lex(m, cnt, cur, out);
    return out;
}

static string solve_line(int m, int64 n) {
    int best = max_assignable_sum(m, n);
    int64 rem = n - best;
    set<vector<int>> seen;
    vector<vector<int>> patterns;
    auto push_pat = [&](const vector<int> &p) {
        vector<int> key = p;
        sort(key.begin(), key.end());
        if (seen.count(key)) return;
        seen.insert(key);
        patterns.push_back(p);
    };
    if (best % m == 0) {
        int v = best / m;
        if (v >= 0 && v <= 5) push_pat(vector<int>(m, v));
    }
    for (int k = 0; k < 5; ++k) {
        for (int t = 1; t < m; ++t) {
            if (k * m + t == best && k + 1 <= 5) {
                vector<int> p;
                p.insert(p.end(), t, k + 1);
                p.insert(p.end(), m - t, k);
                push_pat(p);
            }
        }
    }
    ostringstream oss;
    oss << rem;
    for (const auto &pat : patterns) {
        for (const auto &ln : gen_desc_lex(m, pat)) oss << '\n' << ln;
    }
    return oss.str();
}

static bool parse_input(const string &line, int &m, int64 &n) {
    string s = line;
    while (!s.empty() && (s.back() == '\r' || s.back() == '\n')) s.pop_back();
    while (!s.empty() && (s.front() == ' ' || s.front() == '\t')) s.erase(s.begin());
    while (!s.empty() && (s.back() == ' ' || s.back() == '\t')) s.pop_back();
    if (s.empty()) return false;
    if (count(s.begin(), s.end(), ',') != 1) return false;
    size_t p = s.find(',');
    string a = s.substr(0, p), b = s.substr(p + 1);
    auto trim = [](string &x) {
        while (!x.empty() && (x.front() == ' ' || x.front() == '\t')) x.erase(x.begin());
        while (!x.empty() && (x.back() == ' ' || x.back() == '\t')) x.pop_back();
    };
    trim(a);
    trim(b);
    if (a.empty() || b.empty()) return false;
    for (char c : a)
        if (!isdigit((unsigned char)c)) return false;
    for (char c : b)
        if (!isdigit((unsigned char)c)) return false;
    m = stoi(a);
    n = stoll(b);
    if (m < 1 || n < 0) return false;
    return true;
}

int main() {
#ifdef _WIN32
    _setmode(1, O_BINARY);
#endif
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string line;
    if (!getline(cin, line)) return 0;
    int m;
    int64 n;
    if (!parse_input(line, m, n)) {
        fwrite("invalid\n", 1, 8, stdout);
        return 0;
    }
    string out = solve_line(m, n);
    out.push_back('\n');
    fwrite(out.data(), 1, out.size(), stdout);
    return 0;
}
```
