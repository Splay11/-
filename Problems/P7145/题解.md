# 删除重复字母

算法标签：栈、贪心、字符串

算法难度：$5 / 10$

## 解题思路

要得到 $s$ 的一个子序列：每个出现过的字母恰好一次，且字典序最小。用单调栈从左到右贪心。

1. 先记下每个字母最后一次出现的下标。后面能不能再「用掉」某个字母，看这个下标。
2. 从左到右扫。若当前字母已经在栈里，再放一次会重复，直接跳过。
3. 若栈顶字母比当前字母大，并且栈顶字母在后面还会出现，就弹出栈顶。这样可以把更小的字母推到更前面，被弹出的字母以后还能再选。
4. 栈顶不能弹（比当前小，或后面再也没有了）时，把当前字母入栈。
5. 常见假解：每个字母只留第一次出现（样例 $1$ 会得到 $bca$）；只留最后一次出现（样例 $2$ 会得到 $adbc$）；当成可以任意重排，输出排序后的去重串（$ba$ 会错成 $ab$）。

## 复杂度分析

- 时间复杂度：$O(|s|)$。每个字符至多入栈、出栈一次。
- 空间复杂度：$O(|\Sigma|)$。栈里字母互不相同，$\Sigma$ 是小写字母表，大小为 $26$。答案字符串长度也至多 $26$。

## 代码实现

### Python

```python
def solve(s):
    """单调栈贪心：每个字母最终只留一次，且整体字典序最小。
    还没进栈的字母，若栈顶比它大、并且栈顶字母后面还会再出现，就可以弹出栈顶，
    把更小的字母尽量往前放。已经在栈里的字母直接跳过。
    """
    last = {}
    for i, c in enumerate(s):
        # 记下每个字母最后一次出现的下标，用来判断后面还能不能再用它
        last[c] = i
    stack = []
    used = set()
    for i, c in enumerate(s):
        if c in used:
            # 这个字母已经选过，相对顺序不能再插一次
            continue
        while stack and stack[-1] > c and last[stack[-1]] > i:
            # 栈顶更大，且后面还能再遇到它，弹掉以换更小的前缀
            used.remove(stack.pop())
        stack.append(c)
        used.add(c)
    return "".join(stack)


def main():
    # 一整行小写字母
    s = input()
    print(solve(s))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

public class Main {
    // 单调栈贪心：每个字母只留一次，字典序尽量小
    static String solve(String s) {
        int[] last = new int[26];
        for (int k = 0; k < 26; k++) {
            last[k] = -1;
        }
        int n = s.length();
        for (int i = 0; i < n; i++) {
            // 记下每个字母最后一次出现的下标
            last[s.charAt(i) - 'a'] = i;
        }
        List<Character> stack = new ArrayList<>();
        boolean[] used = new boolean[26];
        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            int x = c - 'a';
            if (used[x]) {
                // 已经选过，不能再出现
                continue;
            }
            while (!stack.isEmpty()) {
                char top = stack.get(stack.size() - 1);
                // 栈顶更大且后面还会出现，才能弹掉换更小前缀
                if (top > c && last[top - 'a'] > i) {
                    used[top - 'a'] = false;
                    stack.remove(stack.size() - 1);
                } else {
                    break;
                }
            }
            stack.add(c);
            used[x] = true;
        }
        StringBuilder ans = new StringBuilder();
        for (char c : stack) {
            ans.append(c);
        }
        return ans.toString();
    }

    public static void main(String[] args) throws IOException {
        // 一整行小写字母
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        System.out.println(solve(s));
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

// 单调栈贪心：每个字母只留一次，字典序尽量小
string solve(const string& s) {
    int last[26];
    for (int k = 0; k < 26; k++) {
        last[k] = -1;
    }
    int n = (int)s.size();
    for (int i = 0; i < n; i++) {
        // 记下每个字母最后一次出现的下标
        last[s[i] - 'a'] = i;
    }
    vector<char> stack;
    bool used[26] = {};
    for (int i = 0; i < n; i++) {
        int x = s[i] - 'a';
        if (used[x]) {
            // 已经选过，不能再出现
            continue;
        }
        while (!stack.empty()) {
            int top = stack.back() - 'a';
            // 栈顶更大且后面还会出现，才能弹掉换更小前缀
            if (stack.back() > s[i] && last[top] > i) {
                used[top] = false;
                stack.pop_back();
            } else {
                break;
            }
        }
        stack.push_back(s[i]);
        used[x] = true;
    }
    return string(stack.begin(), stack.end());
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // 不含空格，整行就是字符串 s
    string s;
    cin >> s;
    cout << solve(s) << '\n';
    return 0;
}
```
