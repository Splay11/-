## 解题思路

本题模拟能量管理系统的**版本化记录**与**时间段比例加权**查询，核心是：

1. **解析命令**：最后一条为 `QueryNetEnergy`，其余按顺序赋予版本号 $1,2,\ldots$；查询参数中的版本号 `A` 表示全量，否则只统计版本 $\le$ 给定值的记录。
2. **时间交集**：对每条记录的闭区间 $[s,e]$ 与查询闭区间 $[qs,qe]$ 求交；无交集则贡献为 $0$。
3. **比例加权**：交集长度 $L = \min(e,qe) - \max(s,qs)$，记录时长 $T = e - s$，有效量 $= \text{amount} \times L / T$；生产加、消耗减。
4. **四舍五入**：浮点净能量按四舍五入（$0.5$ 远离零）转为 $32$ 位整数。

常见假解：忽略版本截断、部分重叠仍按全额计、用 $T+1$ 算时长、直接截断小数而非四舍五入、把查询命令也当作一条记录。

## 复杂度分析

- 时间复杂度：$O(n)$，$n$ 为命令条数（含查询），每条记录 $O(1)$ 解析与加权。
- 空间复杂度：$O(1)$ 额外空间（不计输入数组本身）。

## 代码实现

### Python

```python
from typing import List


def _weighted(amount: int, start: int, end: int, qs: int, qe: int) -> float:
    if start > qe or end < qs:
        return 0.0
    duration = end - start
    if duration == 0:
        if qs <= start <= qe:
            return float(amount)
        return 0.0
    overlap_start = max(start, qs)
    overlap_end = min(end, qe)
    overlap_len = overlap_end - overlap_start
    # 按交集占比折算有效量
    return amount * overlap_len / duration


def _round_to_i32(x: float) -> int:
    if x >= 0:
        val = int(x + 0.5)
    else:
        val = int(x - 0.5)
    return max(-2147483648, min(2147483647, val))


class Solution:
    def queryNetEnergy(self, commands: List[str]) -> int:
        qparts = commands[-1].split(",")
        version_str = qparts[1].strip()
        qs, qe = int(qparts[2]), int(qparts[3])
        max_version = 10**9 if version_str == "A" else int(version_str)

        total = 0.0
        version = 0
        for cmd in commands[:-1]:
            version += 1
            if version > max_version:
                break
            parts = cmd.split(",")
            if parts[0] == "AddProductionRecord":
                total += _weighted(int(parts[2]), int(parts[3]), int(parts[4]), qs, qe)
            else:
                total -= _weighted(int(parts[1]), int(parts[2]), int(parts[3]), qs, qe)
        return _round_to_i32(total)
```

### Java

```java
public class Solution {
    private static double weighted(long amount, long start, long end, long qs, long qe) {
        if (start > qe || end < qs) return 0.0;
        long duration = end - start;
        if (duration == 0) {
            if (qs <= start && start <= qe) return amount;
            return 0.0;
        }
        long os = Math.max(start, qs);
        long oe = Math.min(end, qe);
        if (os > oe) return 0.0;
        // 交集长度占记录总时长的比例
        return amount * (double) (oe - os) / (double) duration;
    }

    private static int roundToI32(double x) {
        long val = x >= 0 ? (long) Math.floor(x + 0.5) : (long) Math.ceil(x - 0.5);
        if (val > 2147483647L) return 2147483647;
        if (val < -2147483648L) return -2147483648;
        return (int) val;
    }

    public int queryNetEnergy(String[] commands) {
        String[] qparts = commands[commands.length - 1].split(",", -1);
        String versionStr = qparts[1].trim();
        long qs = Long.parseLong(qparts[2]);
        long qe = Long.parseLong(qparts[3]);
        int maxVersion = versionStr.equals("A") ? 1_000_000_000 : Integer.parseInt(versionStr);

        double total = 0.0;
        int version = 0;
        for (int i = 0; i < commands.length - 1; i++) {
            version++;
            if (version > maxVersion) break;  // 超出查询版本上界则停止
            String[] parts = commands[i].split(",", -1);
            if (parts[0].equals("AddProductionRecord")) {
                total += weighted(Long.parseLong(parts[2]), Long.parseLong(parts[3]),
                        Long.parseLong(parts[4]), qs, qe);
            } else if (parts[0].equals("AddConsumptionRecord")) {
                total -= weighted(Long.parseLong(parts[1]), Long.parseLong(parts[2]),
                        Long.parseLong(parts[3]), qs, qe);
            }
        }
        return roundToI32(total);
    }
}
```

### C++

```cpp
#include <cmath>
#include <string>
#include <vector>

using namespace std;

class Solution {
  static double weighted(long long amount, long long start, long long end, long long qs, long long qe) {
    if (start > qe || end < qs) return 0.0;
    long long duration = end - start;
    if (duration == 0) {
      if (qs <= start && start <= qe) return (double)amount;
      return 0.0;
    }
    long long os = max(start, qs);
    long long oe = min(end, qe);
    if (os > oe) return 0.0;
    long long overlap = oe - os;
    return (double)amount * (double)overlap / (double)duration;  // 比例加权
  }

  static int roundToI32(double x) {
    long long val = x >= 0 ? (long long)floor(x + 0.5) : (long long)ceil(x - 0.5);
    if (val > 2147483647LL) return 2147483647;
    if (val < -2147483648LL) return -2147483648;
    return (int)val;
  }

 public:
  int queryNetEnergy(vector<string>& commands) {
    const string& query = commands.back();
    vector<string> qparts;
    size_t start = 0;
    for (size_t i = 0; i <= query.size(); i++) {
      if (i == query.size() || query[i] == ',') {
        qparts.push_back(query.substr(start, i - start));
        start = i + 1;
      }
    }
    bool useAll = (qparts[1] == "A");
    long long qs = stoll(qparts[2]);
    long long qe = stoll(qparts[3]);
    int maxVersion = useAll ? 1000000000 : stoi(qparts[1]);

    double total = 0.0;
    int version = 0;
    for (size_t i = 0; i + 1 < commands.size(); i++) {
      version++;
      if (version > maxVersion) break;
      const string& cmd = commands[i];
      vector<string> parts;
      start = 0;
      for (size_t j = 0; j <= cmd.size(); j++) {
        if (j == cmd.size() || cmd[j] == ',') {
          parts.push_back(cmd.substr(start, j - start));
          start = j + 1;
        }
      }
      if (parts[0] == "AddProductionRecord") {
        total += weighted(stoll(parts[2]), stoll(parts[3]), stoll(parts[4]), qs, qe);
      } else if (parts[0] == "AddConsumptionRecord") {
        total -= weighted(stoll(parts[1]), stoll(parts[2]), stoll(parts[3]), qs, qe);
      }
    }
    return roundToI32(total);
  }
};
```
