# 双班次收益差

## 解题思路

本题是带**基数约束**的划分：把偶数个工单均分成两班，最小化两班收益差。值班侧先选高收益班次，差值为 $\lvert 2S-V\rvert$，与谁拿走哪一班无关，于是只需在「恰好 $\frac{n}{2}$ 个工单」的子集中，让子集和 $S$ 尽量接近总和的一半。

1. $n\le 30$，直接枚举 $C(n,n/2)$ 过大，采用**折半枚举**（meet in the middle）。
2. 把数组切成左右两半，各约 $\frac{n}{2}$ 个。对每一半枚举所有子集，按所选个数把子集和存进列表并排序。
3. 左半选 $k$ 个、右半选 $\frac{n}{2}-k$ 个，在两列有序和上二分，使两部分和尽量接近 $\lfloor V/2\rfloor$。
4. 所有 $k$ 取最小 $\lvert V-2S\rvert$ 即为答案。

常见假解：

- 排序后交替放入两班（对 $1,2,3,6$ 得到差 $4$，正解为 $2$）；
- 做成无个数约束的划分（对 $1,1,1,3$ 会得到 $0$，正解为 $2$）；
- 按中点切开或只考虑连续段。

## 复杂度分析

- 时间复杂度：每一半 $2^{n/2}$ 个子集，排序与二分后单组约为 $O(n\cdot 2^{n/2})$。询问个数不超过 $80$，$n\le 30$ 可过。
- 空间复杂度：$O(2^{n/2})$。

## 代码实现

### Python

```python
import bisect


def enum_sums(arr):
    n = len(arr)
    by_cnt = [[] for _ in range(n + 1)]
    # 枚举半边所有子集，按元素个数归类
    for mask in range(1 << n):
        s = 0
        c = 0
        for i in range(n):
            if mask >> i & 1:
                s += arr[i]
                c += 1
        by_cnt[c].append(s)
    for c in range(n + 1):
        by_cnt[c].sort()
    return by_cnt


def min_diff(vals):
    n = len(vals)
    half = n // 2
    tot = sum(vals)
    sl = enum_sums(vals[:half])
    sr = enum_sums(vals[half:])
    best = tot
    for k in range(half + 1):
        A = sl[k]
        B = sr[half - k]
        if not A or not B:
            continue
        for x in A:
            t = tot // 2 - x
            i = bisect.bisect_left(B, t)
            for j in (i - 1, i, i + 1):
                if 0 <= j < len(B):
                    s = x + B[j]
                    d = tot - 2 * s
                    if d < 0:
                        d = -d
                    if d < best:
                        best = d
    return best


def main():
    q = int(input())
    for _ in range(q):
        input()
        vals = list(map(int, input().split()))
        print(min_diff(vals))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {
    // 枚举半边所有子集，按元素个数归类
    static List<List<Long>> enumSums(long[] arr) {
        int n = arr.length;
        List<List<Long>> byCnt = new ArrayList<List<Long>>();
        for (int i = 0; i <= n; i++) {
            byCnt.add(new ArrayList<Long>());
        }
        for (int mask = 0; mask < (1 << n); mask++) {
            long s = 0;
            int c = 0;
            for (int i = 0; i < n; i++) {
                if (((mask >> i) & 1) == 1) {
                    s += arr[i];
                    c++;
                }
            }
            byCnt.get(c).add(s);
        }
        for (int c = 0; c <= n; c++) {
            Collections.sort(byCnt.get(c));
        }
        return byCnt;
    }

    static long minDiff(long[] vals) {
        int n = vals.length;
        int half = n / 2;
        long tot = 0;
        for (int i = 0; i < n; i++) {
            tot += vals[i];
        }
        long[] left = new long[half];
        long[] right = new long[n - half];
        for (int i = 0; i < half; i++) {
            left[i] = vals[i];
        }
        for (int i = half; i < n; i++) {
            right[i - half] = vals[i];
        }
        List<List<Long>> sl = enumSums(left);
        List<List<Long>> sr = enumSums(right);
        long best = tot;
        // 左半选 k 个，右半选 half-k 个
        for (int k = 0; k <= half; k++) {
            List<Long> A = sl.get(k);
            List<Long> B = sr.get(half - k);
            if (A.isEmpty() || B.isEmpty()) {
                continue;
            }
            for (int p = 0; p < A.size(); p++) {
                long x = A.get(p);
                long t = tot / 2 - x;
                int i = Collections.binarySearch(B, t);
                if (i < 0) {
                    i = -i - 1; // 转为插入位置，等价于 bisect_left
                }
                for (int j = i - 1; j <= i + 1; j++) {
                    if (j >= 0 && j < B.size()) {
                        long s = x + B.get(j);
                        long d = tot - 2 * s;
                        if (d < 0) {
                            d = -d;
                        }
                        if (d < best) {
                            best = d;
                        }
                    }
                }
            }
        }
        return best;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int t = 0; t < q; t++) {
            int n = sc.nextInt();
            long[] vals = new long[n];
            for (int i = 0; i < n; i++) {
                vals[i] = sc.nextLong();
            }
            System.out.println(minDiff(vals));
        }
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 枚举半边所有子集，按元素个数归类
vector<vector<long long>> enum_sums(const vector<long long>& arr) {
    int n = (int)arr.size();
    vector<vector<long long>> by_cnt(n + 1);
    for (int mask = 0; mask < (1 << n); ++mask) {
        long long s = 0;
        int c = 0;
        for (int i = 0; i < n; ++i) {
            if (mask >> i & 1) {
                s += arr[i];
                ++c;
            }
        }
        by_cnt[c].push_back(s);
    }
    for (int c = 0; c <= n; ++c) {
        sort(by_cnt[c].begin(), by_cnt[c].end());
    }
    return by_cnt;
}

long long min_diff(const vector<long long>& vals) {
    int n = (int)vals.size();
    int half = n / 2;
    long long tot = 0;
    for (int i = 0; i < n; ++i) {
        tot += vals[i];
    }
    vector<long long> left(vals.begin(), vals.begin() + half);
    vector<long long> right(vals.begin() + half, vals.end());
    vector<vector<long long>> sl = enum_sums(left);
    vector<vector<long long>> sr = enum_sums(right);
    long long best = tot;
    // 左半选 k 个，右半选 half-k 个
    for (int k = 0; k <= half; ++k) {
        const vector<long long>& A = sl[k];
        const vector<long long>& B = sr[half - k];
        if (A.empty() || B.empty()) {
            continue;
        }
        for (size_t p = 0; p < A.size(); ++p) {
            long long x = A[p];
            long long t = tot / 2 - x;
            int i = (int)(lower_bound(B.begin(), B.end(), t) - B.begin());
            for (int j = i - 1; j <= i + 1; ++j) {
                if (j >= 0 && j < (int)B.size()) {
                    long long s = x + B[j];
                    long long d = tot - 2 * s;
                    if (d < 0) {
                        d = -d;
                    }
                    if (d < best) {
                        best = d;
                    }
                }
            }
        }
    }
    return best;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<long long> vals(n);
        for (int i = 0; i < n; ++i) {
            cin >> vals[i];
        }
        cout << min_diff(vals) << '\n';
    }
    return 0;
}
```
