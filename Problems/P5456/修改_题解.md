# 枚举暴力解法

这道题的正解是最短路加状压 TSP，但是我们用枚举接人顺序的朴素解法也能在考试时拿到一定的分数。

汇合时间看最晚到达的人。小红从 $a$ 驾车去馆子 $b$，可以顺路接一部分同伴；没接到的人按每公里 $10$ 分钟自己走，驾车按每公里 $2$ 分钟。$q$ 比较小时，可以直接枚举「接谁、按什么次序接」。

1. 建图后，对出发、馆子、同伴所在路口做 Dijkstra，得到关键点之间的最短公里数。路和点都可以反复走，所以用最短路即可。
2. 枚举接上车的子集，再枚举这些人的排列作为接人顺序，累加驾车公里，最后乘 $2$。
3. 与未接同伴步行时间的最大值取 $\max$，再对所有方案取最小。

排列数是 $q!$ 量级。$q$ 很小可以通过，但 $q$ 到 $15$ 会超时，所以后面再用状压。

## 枚举解法
#code-switcher
```python
import heapq
import itertools

INF = 10 ** 18


def dijkstra(p, adj, src):
    dist = [INF] * (p + 1)
    dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist[u]:
            continue
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))
    return dist


def min_meet_time(p, edges, a, b, friends):
    adj = [[] for _ in range(p + 1)]
    for x, y, c, z in edges:
        adj[x].append((y, c))
        if z == 1:
            adj[y].append((x, c))
    q = len(friends)
    sp = {}
    seen = set()
    for s in [a, b] + friends:
        if s not in seen:
            seen.add(s)
            sp[s] = dijkstra(p, adj, s)

    def dist(u, v):
        return sp[u][v]

    if q == 0:
        return 2 * dist(a, b)
    walk = [10 * dist(f, b) for f in friends]
    ans = INF
    # 枚举接哪些人
    for mask in range(1 << q):
        seq = [i for i in range(q) if (mask >> i) & 1]
        walkers = 0
        for i in range(q):
            if (mask >> i) & 1 == 0:
                walkers = max(walkers, walk[i])
        if not seq:
            ans = min(ans, max(2 * dist(a, b), walkers))
            continue
        # 再枚举接人顺序
        for perm in itertools.permutations(seq):
            km = dist(a, friends[perm[0]])
            for i in range(1, len(perm)):
                km += dist(friends[perm[i - 1]], friends[perm[i]])
            km += dist(friends[perm[-1]], b)
            ans = min(ans, max(2 * km, walkers))
    return ans


def main():
    p, e, a, b = map(int, input().split())
    edges = []
    for _ in range(e):
        x, y, c, z = map(int, input().split())
        edges.append((x, y, c, z))
    q = int(input())
    friends = [] if q == 0 else list(map(int, input().split()))
    print(min_meet_time(p, edges, a, b, friends))


if __name__ == "__main__":
    main()
```

```cpp
#include <algorithm>
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

const long long INF = (long long)4e18;

vector<long long> dijkstra(int p, const vector<vector<pair<int, int> > >& adj, int src) {
    vector<long long> dist(p + 1, INF);
    dist[src] = 0;
    priority_queue<pair<long long, int>, vector<pair<long long, int> >, greater<pair<long long, int> > > pq;
    pq.push(make_pair(0LL, src));
    while (!pq.empty()) {
        pair<long long, int> cur = pq.top();
        pq.pop();
        long long d = cur.first;
        int u = cur.second;
        if (d != dist[u]) {
            continue;
        }
        for (size_t i = 0; i < adj[u].size(); i++) {
            int v = adj[u][i].first;
            int w = adj[u][i].second;
            long long nd = d + w;
            if (nd < dist[v]) {
                dist[v] = nd;
                pq.push(make_pair(nd, v));
            }
        }
    }
    return dist;
}

long long minMeetTime(int p, const vector<vector<int> >& edges, int a, int b,
                      const vector<int>& friends) {
    vector<vector<pair<int, int> > > adj(p + 1);
    for (size_t i = 0; i < edges.size(); i++) {
        int x = edges[i][0], y = edges[i][1], c = edges[i][2], z = edges[i][3];
        adj[x].push_back(make_pair(y, c));
        if (z == 1) {
            adj[y].push_back(make_pair(x, c));
        }
    }
    int q = (int)friends.size();
    vector<vector<long long> > sp(p + 1);
    vector<char> seen(p + 1, 0);
    vector<int> sources;
    sources.push_back(a);
    sources.push_back(b);
    for (int i = 0; i < q; i++) {
        sources.push_back(friends[i]);
    }
    for (size_t i = 0; i < sources.size(); i++) {
        int s = sources[i];
        if (!seen[s]) {
            seen[s] = 1;
            sp[s] = dijkstra(p, adj, s);
        }
    }
    if (q == 0) {
        return 2 * sp[a][b];
    }
    vector<long long> walk(q);
    for (int i = 0; i < q; i++) {
        walk[i] = 10 * sp[friends[i]][b];
    }
    long long ans = INF;
    int N = 1 << q;
    // 枚举接哪些人
    for (int mask = 0; mask < N; mask++) {
        vector<int> seq;
        long long walkers = 0;
        for (int i = 0; i < q; i++) {
            if ((mask >> i) & 1) {
                seq.push_back(i);
            } else if (walk[i] > walkers) {
                walkers = walk[i];
            }
        }
        if (seq.empty()) {
            long long cur = 2 * sp[a][b];
            if (walkers > cur) {
                cur = walkers;
            }
            if (cur < ans) {
                ans = cur;
            }
            continue;
        }
        sort(seq.begin(), seq.end());
        // 再枚举接人顺序
        do {
            long long km = sp[a][friends[seq[0]]];
            for (size_t i = 1; i < seq.size(); i++) {
                km += sp[friends[seq[i - 1]]][friends[seq[i]]];
            }
            km += sp[friends[seq.back()]][b];
            long long cur = 2 * km;
            if (walkers > cur) {
                cur = walkers;
            }
            if (cur < ans) {
                ans = cur;
            }
        } while (next_permutation(seq.begin(), seq.end()));
    }
    return ans;
}
```

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;
import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    static final long INF = 4_000_000_000_000_000_000L;

    static class Edge {
        int v, w;
        Edge(int v, int w) {
            this.v = v;
            this.w = w;
        }
    }

    static long[] dijkstra(int p, List<List<Edge>> adj, int src) {
        long[] dist = new long[p + 1];
        Arrays.fill(dist, INF);
        dist[src] = 0;
        PriorityQueue<long[]> pq = new PriorityQueue<>(Comparator.comparingLong(x -> x[0]));
        pq.add(new long[] {0, src});
        while (!pq.isEmpty()) {
            long[] cur = pq.poll();
            long d = cur[0];
            int u = (int) cur[1];
            if (d != dist[u]) {
                continue;
            }
            for (Edge e : adj.get(u)) {
                long nd = d + e.w;
                if (nd < dist[e.v]) {
                    dist[e.v] = nd;
                    pq.add(new long[] {nd, e.v});
                }
            }
        }
        return dist;
    }

    static void permute(int[] seq, int l, long[][] sp, int[] friends, int a, int b,
                        long walkers, long[] ans) {
        if (l == seq.length) {
            long km = sp[a][friends[seq[0]]];
            for (int i = 1; i < seq.length; i++) {
                km += sp[friends[seq[i - 1]]][friends[seq[i]]];
            }
            km += sp[friends[seq[seq.length - 1]]][b];
            long cur = Math.max(2 * km, walkers);
            if (cur < ans[0]) {
                ans[0] = cur;
            }
            return;
        }
        for (int i = l; i < seq.length; i++) {
            int tmp = seq[l];
            seq[l] = seq[i];
            seq[i] = tmp;
            permute(seq, l + 1, sp, friends, a, b, walkers, ans);
            tmp = seq[l];
            seq[l] = seq[i];
            seq[i] = tmp;
        }
    }

    static long minMeetTime(int p, int[][] edges, int a, int b, int[] friends) {
        List<List<Edge>> adj = new ArrayList<>();
        for (int i = 0; i <= p; i++) {
            adj.add(new ArrayList<>());
        }
        for (int[] eg : edges) {
            adj.get(eg[0]).add(new Edge(eg[1], eg[2]));
            if (eg[3] == 1) {
                adj.get(eg[1]).add(new Edge(eg[0], eg[2]));
            }
        }
        int q = friends.length;
        long[][] sp = new long[p + 1][];
        boolean[] seen = new boolean[p + 1];
        int[] sources = new int[q + 2];
        sources[0] = a;
        sources[1] = b;
        for (int i = 0; i < q; i++) {
            sources[i + 2] = friends[i];
        }
        for (int s : sources) {
            if (!seen[s]) {
                seen[s] = true;
                sp[s] = dijkstra(p, adj, s);
            }
        }
        if (q == 0) {
            return 2 * sp[a][b];
        }
        long[] walk = new long[q];
        for (int i = 0; i < q; i++) {
            walk[i] = 10 * sp[friends[i]][b];
        }
        long[] ans = new long[] {INF};
        int N = 1 << q;
        for (int mask = 0; mask < N; mask++) {
            int cnt = 0;
            long walkers = 0;
            for (int i = 0; i < q; i++) {
                if (((mask >> i) & 1) != 0) {
                    cnt++;
                } else {
                    walkers = Math.max(walkers, walk[i]);
                }
            }
            if (cnt == 0) {
                ans[0] = Math.min(ans[0], Math.max(2 * sp[a][b], walkers));
                continue;
            }
            int[] seq = new int[cnt];
            int z = 0;
            for (int i = 0; i < q; i++) {
                if (((mask >> i) & 1) != 0) {
                    seq[z++] = i;
                }
            }
            permute(seq, 0, sp, friends, a, b, walkers, ans);
        }
        return ans[0];
    }

        Scanner sc = new Scanner(System.in);
        int p = sc.nextInt();
        int e = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int[][] edges = new int[e][4];
        for (int i = 0; i < e; i++) {
            for (int j = 0; j < 4; j++) {
                edges[i][j] = sc.nextInt();
            }
        }
        int q = sc.nextInt();
        int[] friends = new int[q];
        for (int i = 0; i < q; i++) {
            friends[i] = sc.nextInt();
        }
        sc.close();
        System.out.println(minMeetTime(p, edges, a, b, friends));
    }
}
```

# 正确解法

## 题目分析

$q$ 到 $15$ 时不能再枚举排列，但对「已经接了哪些人、当前停在哪位同伴处」做状压，转移次数是 $q^2\cdot 2^{q}$。

1. 同样先 Dijkstra，得到关键点最短公里数。
2. $dp[mask][i]$：已经接完 $mask$ 里的人，当前停在同伴 $i$ 处的最短驾车公里。再去接一个尚未接到的人 $j$ 即可转移。
3. 枚举最终子集，驾车时间乘 $2$，再与未接同伴步行时间取 $\max$，对所有子集取最小。

## 复杂度分析

- 时间复杂度：$O((q+2)(e+p)\log p + q^2\cdot 2^{q})$。
- 空间复杂度：$O(p + e + q\cdot 2^{q})$。

$p,e\le 10^{3}$，$q\le 15$，可以通过。

## 代码实现

```cpp
    for (int mask = 0; mask < N; mask++) {
        long long car = (mask == 0) ? 2 * sp[a][b] : INF;
        if (mask) {
            for (int i = 0; i < q; i++) {
                if ((mask >> i) & 1) {
                    long long cand = 2 * (dp[mask][i] + sp[friends[i]][b]);
                    if (cand < car) {
                        car = cand;
                    }
                }
            }
        }
        long long walkers = 0;
        for (int i = 0; i < q; i++) {
            if (((mask >> i) & 1) == 0 && walk[i] > walkers) {
                walkers = walk[i];
            }
        }
        long long cur = car > walkers ? car : walkers;
        if (cur < ans) {
            ans = cur;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int p, e, a, b;
    cin >> p >> e >> a >> b;
    vector<vector<int> > edges(e, vector<int>(4));
    for (int i = 0; i < e; i++) {
        cin >> edges[i][0] >> edges[i][1] >> edges[i][2] >> edges[i][3];
    }
    int q;
    cin >> q;
    vector<int> friends(q);
    for (int i = 0; i < q; i++) {
        cin >> friends[i];
    }
    cout << minMeetTime(p, edges, a, b, friends) << '\n';
    return 0;
}
```
