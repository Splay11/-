# 限次罐笼

## 解题思路

普通巷道带非负耗时，罐笼耗时为 $0$，但全程最多坐 $c$ 次，且不能连坐。这是带附加状态的最短路。

1. 若 $s=t$，直接返回 $0$。
2. 状态写成 $(u,used,last)$：当前工位、已用罐笼次数、上一步是否刚坐过罐笼。$used\le c$，$last\in\{0,1\}$，状态数 $O(nc)$。
3. 走巷道 $u\to v$ 耗时 $d$，转移到 $(v,used,0)$。
4. 仅当 $last=0$ 且 $used<c$ 时可坐罐笼，转移到 $(v,used+1,1)$，耗时 $0$。起点初始为 $(s,0,0)$，因此出发就能坐罐笼。
5. 边权非负，用堆优化 Dijkstra。取所有能到达终点的状态的最小距离；都不行则 $-1$。
6. 答案可达 $(n\cdot c)\times 10^9$ 量级，需 $64$ 位整数。

## 复杂度分析

设工位数为 $n$，巷道数为 $q$，次数上限为 $c$。

时间复杂度为 $O((n+q)c\log(nc))$。

空间复杂度为 $O(n c+q)$。

## 代码实现

### Python

```python
import heapq

INF = 10**30


def min_time(n, roads, portals, cap, src, dst):
    # 起点等于终点：不必移动
    if src == dst:
        return 0
    # 普通巷道邻接表
    g = [[] for _ in range(n)]
    for u, v, d in roads:
        g[u].append((v, d))
        g[v].append((u, d))
    # 罐笼邻接表（耗时 0）
    pg = [[] for _ in range(n)]
    for u, v in portals:
        pg[u].append(v)
        pg[v].append(u)
    # dist[工位][已用罐笼次数][上一步是否罐笼]
    dist = [[[INF, INF] for _ in range(cap + 1)] for _ in range(n)]
    dist[src][0][0] = 0
    pq = [(0, src, 0, 0)]
    while pq:
        cur, u, used, last = heapq.heappop(pq)
        if cur != dist[u][used][last]:
            continue
        if u == dst:
            return cur
        # 走普通巷道：结束后“上一步不是罐笼”，次数不变
        for v, d in g[u]:
            nd = cur + d
            if nd < dist[v][used][0]:
                dist[v][used][0] = nd
                heapq.heappush(pq, (nd, v, used, 0))
        # 坐罐笼：不能连坐，且次数还没用完
        if last == 0 and used < cap:
            for v in pg[u]:
                if cur < dist[v][used + 1][1]:
                    dist[v][used + 1][1] = cur
                    heapq.heappush(pq, (cur, v, used + 1, 1))
    return -1


def main():
    n, m = map(int, input().split())
    roads = []
    for _ in range(m):
        u, v, d = map(int, input().split())
        roads.append((u, v, d))
    p = int(input())
    portals = []
    for _ in range(p):
        u, v = map(int, input().split())
        portals.append((u, v))
    cap, src, dst = map(int, input().split())
    print(min_time(n, roads, portals, cap, src, dst))


if __name__ == "__main__":
    main()
```

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {
    static final long INF = 1L << 62;

    static class Edge {
        int to;
        long w;
        Edge(int to, long w) {
            this.to = to;
            this.w = w;
        }
    }

    static long minTime(int n, List<int[]> roads, List<int[]> portals, int cap, int src, int dst) {
        if (src == dst) {
            return 0;
        }
        List<List<Edge>> g = new ArrayList<>();
        List<List<Integer>> pg = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            g.add(new ArrayList<>());
            pg.add(new ArrayList<>());
        }
        for (int[] e : roads) {
            g.get(e[0]).add(new Edge(e[1], e[2]));
            g.get(e[1]).add(new Edge(e[0], e[2]));
        }
        for (int[] e : portals) {
            pg.get(e[0]).add(e[1]);
            pg.get(e[1]).add(e[0]);
        }
        long[][][] dist = new long[n][cap + 1][2];
        for (int i = 0; i < n; i++) {
            for (int u = 0; u <= cap; u++) {
                Arrays.fill(dist[i][u], INF);
            }
        }
        dist[src][0][0] = 0;
        PriorityQueue<long[]> pq = new PriorityQueue<>(Comparator.comparingLong(a -> a[0]));
        pq.add(new long[] {0, src, 0, 0});
        while (!pq.isEmpty()) {
            long[] cur = pq.poll();
            long d = cur[0];
            int u = (int) cur[1];
            int used = (int) cur[2];
            int last = (int) cur[3];
            if (d != dist[u][used][last]) {
                continue;
            }
            if (u == dst) {
                return d;
            }
            for (Edge e : g.get(u)) {
                long nd = d + e.w;
                if (nd < dist[e.to][used][0]) {
                    dist[e.to][used][0] = nd;
                    pq.add(new long[] {nd, e.to, used, 0});
                }
            }
            if (last == 0 && used < cap) {
                for (int v : pg.get(u)) {
                    if (d < dist[v][used + 1][1]) {
                        dist[v][used + 1][1] = d;
                        pq.add(new long[] {d, v, used + 1, 1});
                    }
                }
            }
        }
        return -1;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        List<int[]> roads = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int d = Integer.parseInt(st.nextToken());
            roads.add(new int[] {u, v, d});
        }
        int p = Integer.parseInt(br.readLine().trim());
        List<int[]> portals = new ArrayList<>();
        for (int i = 0; i < p; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            portals.add(new int[] {u, v});
        }
        st = new StringTokenizer(br.readLine());
        int cap = Integer.parseInt(st.nextToken());
        int src = Integer.parseInt(st.nextToken());
        int dst = Integer.parseInt(st.nextToken());
        System.out.println(minTime(n, roads, portals, cap, src, dst));
    }
}
```

```cpp
#include <array>
#include <iostream>
#include <queue>
#include <tuple>
#include <vector>
using namespace std;

const long long INF = (1LL << 62);

struct Road {
    int u, v;
    long long d;
};

struct Edge {
    int to;
    long long w;
    Edge(int to, long long w) : to(to), w(w) {}
};

long long min_time(int n, const vector<Road>& roads,
                   const vector<pair<int, int> >& portals,
                   int cap, int src, int dst) {
    if (src == dst) {
        return 0;
    }
    vector<vector<Edge> > g(n);
    for (size_t i = 0; i < roads.size(); i++) {
        int u = roads[i].u, v = roads[i].v;
        long long d = roads[i].d;
        g[u].push_back(Edge(v, d));
        g[v].push_back(Edge(u, d));
    }
    vector<vector<int> > pg(n);
    for (size_t i = 0; i < portals.size(); i++) {
        int u = portals[i].first, v = portals[i].second;
        pg[u].push_back(v);
        pg[v].push_back(u);
    }
    vector<vector<array<long long, 2> > > dist(
        n, vector<array<long long, 2> >(cap + 1));
    for (int i = 0; i < n; i++) {
        for (int used = 0; used <= cap; used++) {
            dist[i][used][0] = INF;
            dist[i][used][1] = INF;
        }
    }
    dist[src][0][0] = 0;
    typedef tuple<long long, int, int, int> State;
    priority_queue<State, vector<State>, greater<State> > pq;
    pq.push(State(0, src, 0, 0));
    while (!pq.empty()) {
        long long cur;
        int u, used, last;
        tie(cur, u, used, last) = pq.top();
        pq.pop();
        if (cur != dist[u][used][last]) {
            continue;
        }
        if (u == dst) {
            return cur;
        }
        for (size_t i = 0; i < g[u].size(); i++) {
            int v = g[u][i].to;
            long long nd = cur + g[u][i].w;
            if (nd < dist[v][used][0]) {
                dist[v][used][0] = nd;
                pq.push(State(nd, v, used, 0));
            }
        }
        if (last == 0 && used < cap) {
            for (size_t i = 0; i < pg[u].size(); i++) {
                int v = pg[u][i];
                if (cur < dist[v][used + 1][1]) {
                    dist[v][used + 1][1] = cur;
                    pq.push(State(cur, v, used + 1, 1));
                }
            }
        }
    }
    return -1;
}
```

```javascript
const fs = require("fs");
const INF = 1e30;

function minTime(n, roads, portals, cap, src, dst) {
    if (src === dst) return 0;
    const g = Array.from({ length: n }, () => []);
    for (const [u, v, d] of roads) {
        g[u].push([v, d]);
        g[v].push([u, d]);
    }
    const pg = Array.from({ length: n }, () => []);
    for (const [u, v] of portals) {
        pg[u].push(v);
        pg[v].push(u);
    }
    const dist = Array.from({ length: n }, () =>
        Array.from({ length: cap + 1 }, () => [INF, INF])
    );
    dist[src][0][0] = 0;
    const pq = [[0, src, 0, 0]];
    function push(item) {
        pq.push(item);
        let i = pq.length - 1;
        while (i > 0) {
            const p = (i - 1) >> 1;
            if (pq[p][0] <= pq[i][0]) break;
            const tmp = pq[p];
            pq[p] = pq[i];
            pq[i] = tmp;
            i = p;
        }
    }
    function pop() {
        const top = pq[0];
        const last = pq.pop();
        if (pq.length) {
            pq[0] = last;
            let i = 0;
            while (true) {
                let l = i * 2 + 1, r = l + 1, m = i;
                if (l < pq.length && pq[l][0] < pq[m][0]) m = l;
                if (r < pq.length && pq[r][0] < pq[m][0]) m = r;
                if (m === i) break;
                const tmp = pq[m];
                pq[m] = pq[i];
                pq[i] = tmp;
                i = m;
            }
        }
        return top;
    }
    while (pq.length) {
        const [cur, u, used, last] = pop();
        if (cur !== dist[u][used][last]) continue;
        if (u === dst) return cur;
        for (const [v, d] of g[u]) {
            const nd = cur + d;
            if (nd < dist[v][used][0]) {
                dist[v][used][0] = nd;
                push([nd, v, used, 0]);
            }
        }
        if (last === 0 && used < cap) {
            for (const v of pg[u]) {
                if (cur < dist[v][used + 1][1]) {
                    dist[v][used + 1][1] = cur;
                    push([cur, v, used + 1, 1]);
                }
            }
        }
    }
    return -1;
}

function main() {
    const tokens = fs.readFileSync(0, "utf8").trim().split(/\s+/).map(Number);
    let p = 0;
    const n = tokens[p++], m = tokens[p++];
    const roads = [];
    for (let i = 0; i < m; i++) {
        roads.push([tokens[p++], tokens[p++], tokens[p++]]);
    }
    const b = tokens[p++];
    const portals = [];
    for (let i = 0; i < b; i++) {
        portals.push([tokens[p++], tokens[p++]]);
    }
    const cap = tokens[p++], src = tokens[p++], dst = tokens[p++];
    console.log(minTime(n, roads, portals, cap, src, dst));
}

main();
```

### C

```c
#include <stdio.h>
#include <stdlib.h>

#define INF (1LL << 62)

typedef struct Edge {
    int to;
    long long w;
    int next;
} Edge;

typedef struct Node {
    long long d;
    int u, used, last;
} Node;

static Edge *edges;
static int *head, eidx;
static Edge *pedges;
static int *phead, peidx;
static Node *heap;
static int hsz;

void add_edge(int u, int v, long long w) {
    edges[eidx].to = v;
    edges[eidx].w = w;
    edges[eidx].next = head[u];
    head[u] = eidx++;
}

void add_portal(int u, int v) {
    pedges[peidx].to = v;
    pedges[peidx].w = 0;
    pedges[peidx].next = phead[u];
    phead[u] = peidx++;
}

void hpush(Node x) {
    int i = ++hsz;
    heap[i] = x;
    while (i > 1 && heap[i].d < heap[i / 2].d) {
        Node t = heap[i];
        heap[i] = heap[i / 2];
        heap[i / 2] = t;
        i /= 2;
    }
}

Node hpop(void) {
    Node top = heap[1];
    heap[1] = heap[hsz--];
    int i = 1;
    while (1) {
        int l = i * 2, r = l + 1, m = i;
        if (l <= hsz && heap[l].d < heap[m].d) m = l;
        if (r <= hsz && heap[r].d < heap[m].d) m = r;
        if (m == i) break;
        Node t = heap[m];
        heap[m] = heap[i];
        heap[i] = t;
        i = m;
    }
    return top;
}

long long min_time(int n, int cap, int src, int dst) {
    int i, used;
    long long ***dist;
    if (src == dst) return 0;
    dist = (long long ***)malloc((size_t)n * sizeof(long long **));
    for (i = 0; i < n; i++) {
        dist[i] = (long long **)malloc((size_t)(cap + 1) * sizeof(long long *));
        for (used = 0; used <= cap; used++) {
            dist[i][used] = (long long *)malloc(2 * sizeof(long long));
            dist[i][used][0] = INF;
            dist[i][used][1] = INF;
        }
    }
    dist[src][0][0] = 0;
    hsz = 0;
    hpush((Node){0, src, 0, 0});
    while (hsz) {
        Node cur = hpop();
        if (cur.d != dist[cur.u][cur.used][cur.last]) continue;
        if (cur.u == dst) return cur.d;
        for (i = head[cur.u]; i != -1; i = edges[i].next) {
            int v = edges[i].to;
            long long nd = cur.d + edges[i].w;
            if (nd < dist[v][cur.used][0]) {
                dist[v][cur.used][0] = nd;
                hpush((Node){nd, v, cur.used, 0});
            }
        }
        if (cur.last == 0 && cur.used < cap) {
            for (i = phead[cur.u]; i != -1; i = pedges[i].next) {
                int v = pedges[i].to;
                if (cur.d < dist[v][cur.used + 1][1]) {
                    dist[v][cur.used + 1][1] = cur.d;
                    hpush((Node){cur.d, v, cur.used + 1, 1});
                }
            }
        }
    }
    return -1;
}

int main(void) {
    int n, m, p, i, cap, src, dst;
    scanf("%d%d", &n, &m);
    head = (int *)malloc((size_t)n * sizeof(int));
    phead = (int *)malloc((size_t)n * sizeof(int));
    for (i = 0; i < n; i++) {
        head[i] = -1;
        phead[i] = -1;
    }
    edges = (Edge *)malloc((size_t)(2 * m + 5) * sizeof(Edge));
    eidx = 0;
    for (i = 0; i < m; i++) {
        int u, v;
        long long d;
        scanf("%d%d%lld", &u, &v, &d);
        add_edge(u, v, d);
        add_edge(v, u, d);
    }
    scanf("%d", &p);
    pedges = (Edge *)malloc((size_t)(2 * p + 5) * sizeof(Edge));
    peidx = 0;
    for (i = 0; i < p; i++) {
        int u, v;
        scanf("%d%d", &u, &v);
        add_portal(u, v);
        add_portal(v, u);
    }
    scanf("%d%d%d", &cap, &src, &dst);
    heap = (Node *)malloc((size_t)(n * (cap + 1) * 8 + 50) * sizeof(Node));
    printf("%lld\n", min_time(n, cap, src, dst));
    return 0;
}
```
