# 基站干扰指数

## 解题思路

每个基站只需要找到左右两侧「最近的、严格更大」的基站。窗口长度 $t$ 只是距离过滤器：若最近更大元素的距离超过 $t$，窗口内一定没有更大的值，该侧贡献为 $0$。

1. 用单调栈求出 $nge\_left[p]$：下标 $p$ 左侧最近且 $a$ 严格更大的位置；不存在则为 $-1$。从左到右扫描，栈中下标对应的强度保持严格递减，栈顶就是当前的最近更大候选。
2. 再用单调栈求出 $nge\_right[p]$：下标 $p$ 右侧最近且严格更大的位置。从左到右扫描，若当前强度大于栈顶，则当前就是栈顶元素的右侧第一个更大。
3. 对每个 $p$，若 $p-nge\_left[p]\le t$，累加 $a[p]\times(p-nge\_left[p])$；右侧同理。乘积可能达到 $10^{14}$，需要用 $64$ 位整数，并且每步对 $10^9+7$ 取模。
4. 相等强度不算「严格更大」。若改成大于等于，全相等数组会得到非 $0$ 的错误答案。
5. 不能在长度为 $t$ 的窗口里找最大值：最大值不一定是最近的那个更大元素，距离会算错。

样例 $1$ 中 $t=4$，位置 $4$ 左侧最近更大在距离 $4$ 处，刚好落在窗口内。样例 $2$ 中 $t=2$，位置 $0$ 右侧更远处的 $12$ 不能用，贡献为 $0$。

$n\le 10^5$，每个位置向左右各扫最多 $k$ 步会达到 $O(nk)$，在 $k$ 接近 $n$ 时超时。单调栈每个下标至多进栈出栈一次，总时间 $O(n)$。

## 复杂度分析

* 时间复杂度：$O(n)$。左右各扫一遍单调栈，再 $O(n)$ 累加答案。
* 空间复杂度：$O(n)$。需要左右最近更大数组以及单调栈。

$n\le 10^5$，可以通过。

## 代码实现

### Python

```python
MOD = 10 ** 9 + 7


def total_interference(power, k):
    n = len(power)

    # nge_left[i]：i 左侧最近且严格更大的下标；没有则为 -1
    nge_left = [-1] * n
    stack = []
    for i in range(n):
        # 栈里只保留比当前值更大的候选，栈顶就是最近的那个
        while stack and power[stack[-1]] <= power[i]:
            stack.pop()
        if stack:
            nge_left[i] = stack[-1]
        stack.append(i)

    # nge_right[i]：i 右侧最近且严格更大的下标；没有则为 -1
    nge_right = [-1] * n
    stack = []
    for i in range(n):
        # 当前值能作为栈中更小元素的「右侧第一个更大」
        while stack and power[stack[-1]] < power[i]:
            nge_right[stack.pop()] = i
        stack.append(i)

    ans = 0
    for i in range(n):
        left = nge_left[i]
        # 最近更大元素必须落在长度为 k 的搜索窗口内，否则贡献为 0
        if left != -1 and i - left <= k:
            ans = (ans + power[i] * (i - left)) % MOD
        right = nge_right[i]
        if right != -1 and right - i <= k:
            ans = (ans + power[i] * (right - i)) % MOD
    return ans


def main():
    # 第一行：全部基站强度，n 由元素个数得到
    power = list(map(int, input().split()))
    # 第二行：最多向一侧搜索的基站个数
    k = int(input())
    print(total_interference(power, k))


if __name__ == "__main__":
    main()
```

    // 用单调栈求左右最近严格更大元素，距离超过 k 则该侧贡献为 0
    static int totalInterference(int[] power, int k) {

        // ngeLeft[i]：i 左侧最近且严格更大的下标；没有则为 -1
        int[] ngeLeft = new int[n];

// 用单调栈求左右最近严格更大元素，距离超过 k 则该侧贡献为 0
int totalInterference(const vector<int>& power, int k) {
    int n = (int)power.size();

    // ngeLeft[i]：i 左侧最近且严格更大的下标；没有则为 -1
    vector<int> ngeLeft(n, -1);
    vector<int> stack;
    for (int i = 0; i < n; i++) {
        // 栈里只保留比当前值更大的候选，栈顶就是最近的那个
        while (!stack.empty() && power[stack.back()] <= power[i]) {
            stack.pop_back();
        }
        if (!stack.empty()) {
            ngeLeft[i] = stack.back();
        }
        stack.push_back(i);
    }

    // ngeRight[i]：i 右侧最近且严格更大的下标；没有则为 -1
    vector<int> ngeRight(n, -1);
    stack.clear();
    for (int i = 0; i < n; i++) {
        // 当前值能作为栈中更小元素的「右侧第一个更大」
        while (!stack.empty() && power[stack.back()] < power[i]) {
            ngeRight[stack.back()] = i;
            stack.pop_back();
        }
        stack.push_back(i);
    }

    long long ans = 0;
    for (int i = 0; i < n; i++) {
        int left = ngeLeft[i];
        // 最近更大元素必须落在长度为 k 的搜索窗口内
        if (left != -1 && i - left <= k) {
            ans = (ans + 1LL * power[i] * (i - left)) % MOD;
        }
        int right = ngeRight[i];
        if (right != -1 && right - i <= k) {
            ans = (ans + 1LL * power[i] * (right - i)) % MOD;
        }
    }
    return (int)ans;
}

const MOD = 1000000007;

function totalInterference(power, k) {
  const n = power.length;

  // ngeLeft[i]：i 左侧最近且严格更大的下标；没有则为 -1
  const ngeLeft = Array(n).fill(-1);
  const stack = [];
  for (let i = 0; i < n; i++) {
    // 栈里只保留比当前值更大的候选，栈顶就是最近的那个
    while (stack.length && power[stack[stack.length - 1]] <= power[i]) {
      stack.pop();
    }
    if (stack.length) {
      ngeLeft[i] = stack[stack.length - 1];
    }
    stack.push(i);
  }

  // ngeRight[i]：i 右侧最近且严格更大的下标；没有则为 -1
  const ngeRight = Array(n).fill(-1);
  stack.length = 0;
  for (let i = 0; i < n; i++) {
    // 当前值能作为栈中更小元素的「右侧第一个更大」
    while (stack.length && power[stack[stack.length - 1]] < power[i]) {
      ngeRight[stack.pop()] = i;
    }
    stack.push(i);
  }

  let ans = 0;
  for (let i = 0; i < n; i++) {
    const left = ngeLeft[i];
    // 最近更大元素必须落在长度为 k 的搜索窗口内
    if (left !== -1 && i - left <= k) {
      ans = (ans + power[i] * (i - left)) % MOD;
    }
    const right = ngeRight[i];
    if (right !== -1 && right - i <= k) {
      ans = (ans + power[i] * (right - i)) % MOD;
    }
  }
  return ans;
}
