# 查找共用字符

算法标签：哈希表、字符串

算法难度：$3 / 10$

## 解题思路

要找在每个字符串里都出现过的字符。某字符在每个串里最少出现 $t$ 次，答案里就要有 $t$ 个该字符，再按字典序输出。

1. 对 $26$ 个小写字母分别计数。
2. 每个字母的答案次数，等于它在所有字符串中出现次数的最小值。
3. 按 $a$ 到 $z$ 的顺序，把每个字母重复输出对应次数。
4. 若一个都没有，只输出一行 $0$，不要空的第二行。
5. 常见假解：只判断字母是否出现过、忽略出现多次；没按字典序；没有公共字符时仍打印空的第二行。

## 复杂度分析

- 时间复杂度：$O(n\cdot L)$。$L$ 是字符串长度上限。
- 空间复杂度：$O(1)$ 额外计数数组，输出最多 $O(L)$。

## 代码实现

### Python

```python
def solve(words):
    """统计每个小写字母在所有字符串里出现次数的最小值。
    最小值为 k 就把该字母输出 k 次，再按字典序排好。
    """
    min_cnt = [10**9] * 26
    for w in words:
        cnt = [0] * 26
        for ch in w:
            cnt[ord(ch) - 97] += 1
        for i in range(26):
            if cnt[i] < min_cnt[i]:
                min_cnt[i] = cnt[i]
    chars = []
    for i in range(26):
        for _ in range(min_cnt[i]):
            chars.append(chr(97 + i))
    return chars


def main():
    n = int(input())
    words = []
    for _ in range(n):
        words.append(input().strip())
    chars = solve(words)
    print(len(chars))
    if chars:
        print(" ".join(chars))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;

public class Main {
    static ArrayList<Character> solve(String[] words) {
        int[] minCnt = new int[26];
        for (int i = 0; i < 26; i++) {
            minCnt[i] = 1000000000;
        }
        for (int t = 0; t < words.length; t++) {
            int[] cnt = new int[26];
            String w = words[t];
            for (int i = 0; i < w.length(); i++) {
                cnt[w.charAt(i) - 'a']++;
            }
            for (int i = 0; i < 26; i++) {
                if (cnt[i] < minCnt[i]) {
                    minCnt[i] = cnt[i];
                }
            }
        }
        ArrayList<Character> chars = new ArrayList<Character>();
        for (int i = 0; i < 26; i++) {
            for (int k = 0; k < minCnt[i]; k++) {
                chars.add((char) ('a' + i));
            }
        }
        return chars;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        String[] words = new String[n];
        for (int i = 0; i < n; i++) {
            words[i] = br.readLine().trim();
        }
        ArrayList<Character> chars = solve(words);
        System.out.println(chars.size());
        if (!chars.isEmpty()) {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < chars.size(); i++) {
                if (i > 0) {
                    sb.append(' ');
                }
                sb.append(chars.get(i));
            }
            System.out.println(sb.toString());
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

vector<char> solve(const vector<string>& words) {
    int min_cnt[26];
    for (int i = 0; i < 26; i++) {
        min_cnt[i] = 1000000000;
    }
    for (int t = 0; t < (int)words.size(); t++) {
        int cnt[26] = {0};
        const string& w = words[t];
        for (int i = 0; i < (int)w.size(); i++) {
            cnt[w[i] - 'a']++;
        }
        for (int i = 0; i < 26; i++) {
            if (cnt[i] < min_cnt[i]) {
                min_cnt[i] = cnt[i];
            }
        }
    }
    vector<char> chars;
    for (int i = 0; i < 26; i++) {
        for (int k = 0; k < min_cnt[i]; k++) {
            chars.push_back(char('a' + i));
        }
    }
    return chars;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<string> words(n);
    for (int i = 0; i < n; i++) {
        cin >> words[i];
    }
    vector<char> chars = solve(words);
    cout << (int)chars.size() << '\n';
    if (!chars.empty()) {
        for (int i = 0; i < (int)chars.size(); i++) {
            if (i) cout << ' ';
            cout << chars[i];
        }
        cout << '\n';
    }
    return 0;
}
```
