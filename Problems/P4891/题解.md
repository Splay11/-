# 鸟巢场馆安排

## 解题思路

- 本题可抽象为：给定 $n$ 个闭区间 $[s_i,e_i]$，选出尽可能多的区间，使得它们两两不重叠（端点重合也算重叠，与题面「包括起始时间与结束时间」一致）。
- 经典模型是**区间调度 / 活动选择**：将所有区间按**结束时间 $e_i$ 升序**排序（$e_i$ 相同时按 $s_i$ 升序打破并列），然后从左到右贪心：若当前区间起点 $s_i$ 严格大于上一个已选区间的终点 $last$，则选取该区间，并令 $last \leftarrow e_i$。
- 直观理解：在「不重叠」前提下，越早结束的活动越不挡后面的区间，因此每次在「与已安排不冲突」的候选里选结束最早的一定最优；可用交换论证严格证明。
- 实现上读入 $n$，再读一行共 $2n$ 个整数，按顺序两两组成区间即可。

## 复杂度分析

- 排序时间复杂度为 $O(n \log n)$，扫描一遍为 $O(n)$，总时间 $O(n \log n)$。
- 额外使用长度为 $n$ 的区间数组，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
from __future__ import annotations


def solve_interval_scheduling(n: int, intervals: list[tuple[int, int]]) -> int:
    # 按结束时间排序，相等时按开始时间，保证次序稳定、便于复现。
    intervals_sorted = sorted(intervals, key=lambda x: (x[1], x[0]))
    last_end = -(10**30)
    cnt = 0
    for s, e in intervals_sorted:
        # 闭区间端点重合视为冲突，因此用严格大于。
        if s > last_end:
            cnt += 1
            last_end = e
    return cnt


def main() -> None:
    n = int(input())
    parts = input().split()
    nums = list(map(int, parts))
    intervals = [(nums[2 * i], nums[2 * i + 1]) for i in range(n)]
    print(solve_interval_scheduling(n, intervals))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static long solveIntervalScheduling(int n, long[][] intervals) {
        // 按结束时间升序排序，经典区间调度贪心。
        Arrays.sort(intervals, (a, b) -> {
            if (a[1] != b[1]) {
                return Long.compare(a[1], b[1]);
            }
            return Long.compare(a[0], b[0]);
        });
        long lastEnd = Long.MIN_VALUE / 4;
        int cnt = 0;
        for (long[] it : intervals) {
            long s = it[0];
            long e = it[1];
            if (s > lastEnd) {
                cnt++;
                lastEnd = e;
            }
        }
        return cnt;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        String[] parts = br.readLine().trim().split("\\s+");
        long[][] intervals = new long[n][2];
        for (int i = 0; i < n; i++) {
            intervals[i][0] = Long.parseLong(parts[2 * i]);
            intervals[i][1] = Long.parseLong(parts[2 * i + 1]);
        }
        System.out.println(solveIntervalScheduling(n, intervals));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

// 在闭区间意义下，贪心按结束时间最早选取。
int solve_interval_scheduling(int n, const vector<pair<int64, int64>> &intervals) {
    vector<pair<int64, int64>> a = intervals;
    sort(a.begin(), a.end(), [](const pair<int64, int64> &x, const pair<int64, int64> &y) {
        if (x.second != y.second) {
            return x.second < y.second;
        }
        return x.first < y.first;
    });
    int64 last_end = (int64)-4e18;
    int cnt = 0;
    // 依次尝试结束更早的区间，能接上就选。
    for (const auto &pr : a) {
        int64 s = pr.first;
        int64 e = pr.second;
        if (s > last_end) {
            ++cnt;
            last_end = e;
        }
    }
    return cnt;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (!(cin >> n)) {
        return 0;
    }
    vector<int64> flat(2 * n);
    for (int i = 0; i < 2 * n; ++i) {
        cin >> flat[i];
    }
    vector<pair<int64, int64>> intervals;
    intervals.reserve(n);
    for (int i = 0; i < n; ++i) {
        intervals.emplace_back(flat[2 * i], flat[2 * i + 1]);
    }
    cout << solve_interval_scheduling(n, intervals) << '\n';
    return 0;
}
```
