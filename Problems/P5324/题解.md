# 改道后的卸矿等待

## 解题思路

矿车在硐口 $i$ 接通后才改道，下一站直接去卸矿平台，剩下的路程就是该硐口到平台的 $s_i$。值班员同时出发，要走 $t$。两边从同一时刻开始计时，等待时间就是矿石更晚到的那一段。

1. 对每个接通位置 $i$，矿石到达时刻是 $s_i$，值班员到达时刻是 $t$，等待为 $\max(0,\ s_i-t)$。
2. 最后一个硐口就是卸矿平台，$s_k=0$，无论 $t$ 是多少，等待都是 $0$。
3. $s$ 不一定单调，不能把后段路程加起来当到达时间，也不能用前缀最小值之类的额外结构。
4. 从左到右扫一遍即可，不需要排序或预处理。

## 复杂度分析

- 时间复杂度：$O(k)$，每个硐口计算一次。
- 空间复杂度：$O(k)$。

## 代码实现

### Python

```python
def solve(k, t, s):
  # 在硐口 i 接通后，矿石再走 s[i]，值班员走 t
  # 等待就是多出来的时间，不能为负
  ans = []
  for i in range(k):
    wait = s[i] - t
    if wait < 0:
      wait = 0
    ans.append(wait)
  return ans


# 第一行硐口数和步行时间，第二行各硐口到卸矿平台的路程
k, t = map(int, input().split())
s = list(map(int, input().split()))
ans = solve(k, t, s)
print(" ".join(str(x) for x in ans))
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
  static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
  static StringTokenizer st;

  static String next() throws IOException {
    while (st == null || !st.hasMoreTokens()) {
      st = new StringTokenizer(br.readLine());
    }
    return st.nextToken();
  }

  static int nextInt() throws IOException {
    return Integer.parseInt(next());
  }

  static long nextLong() throws IOException {
    return Long.parseLong(next());
  }

  static long[] solve(int k, long t, long[] s) {
    // 在硐口 i 接通后，矿石再走 s[i]，值班员走 t
    // 等待就是多出来的时间，不能为负
    long[] ans = new long[k];
    for (int i = 0; i < k; i++) {
      long wait = s[i] - t;
      if (wait < 0) wait = 0;
      ans[i] = wait;
    }
    return ans;
  }

  public static void main(String[] args) throws IOException {
    // 硐口个数可达 100000，一行很长，Scanner 可能超时
    int k = nextInt();
    long t = nextLong();
    long[] s = new long[k];
    for (int i = 0; i < k; i++) s[i] = nextLong();
    long[] ans = solve(k, t, s);
    StringBuilder sb = new StringBuilder();
    for (int i = 0; i < k; i++) {
      if (i > 0) sb.append(' ');
      sb.append(ans[i]);
    }
    sb.append('\n');
    System.out.print(sb.toString());
  }
}
```

```c
#include <stdio.h>
#include <stdlib.h>

long long *solve(int k, long long t, long long *s) {
  // 在硐口 i 接通后，矿石再走 s[i]，值班员走 t
  long long *ans = (long long *)malloc(sizeof(long long) * k);
  int i;
  for (i = 0; i < k; i++) {
    long long wait = s[i] - t;
    if (wait < 0) wait = 0;
    ans[i] = wait;
  }
  return ans;
}

int main(void) {
  int k, i;
  long long t;
  scanf("%d%lld", &k, &t);
  long long *s = (long long *)malloc(sizeof(long long) * k);
  for (i = 0; i < k; i++) scanf("%lld", &s[i]);
  long long *ans = solve(k, t, s);
  for (i = 0; i < k; i++) {
    if (i) putchar(' ');
    printf("%lld", ans[i]);
  }
  putchar('\n');
  free(s);
  free(ans);
  return 0;
}
```
