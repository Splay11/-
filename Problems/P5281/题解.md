# 场站周转拨付账

## 解题思路

$F_i,R_i,P_i$ 均不超过 $500$，而初始周转金 $s_j$ 可达 $10^9$，不能对每个询问暴力模拟 $K$ 步。

1. **关键观察**：当当前周转金 $>500$ 时，任意 $F_i\le 500$，恒有 $F_i<\text{周转金}$，只会拨付 $P_i$，周转金单调减少。
2. **预处理**：对 $0\le m<1001$ 的初始周转金，倒序 DP 预处理 $f[i][m]$：从第 $i$ 位车主开始、初始周转金为 $m$ 时的最终余额。
3. **单次询问**：
   - $s<1001$：直接查 $f[0][s]$；
   - $s-\sum P_i>500$：全程只拨付，答案为 $s-\sum P_i$；
   - 否则在前缀和 $P$ 上二分，跳到周转金落入小范围的位置，再用 DP 表或后缀处理。

## 复杂度分析

- 时间复杂度：预处理 $O(K\cdot M)$（$M\approx 1001$），每次询问 $O(\log K)$，总计 $O(KM+T\log K)$。
- 空间复杂度：$O(NM)$。
- Python 建议用 PyPy3 提交；C++/Java 可承受更大 $N,Q$。

## 代码实现

### Python

```python
MAXM = 1001
THRESH = 500

n = int(input())
w = [0] * n
a = [0] * n
b = [0] * n
for i in range(n):
  wi, ai, bi = map(int, input().split())
  w[i], a[i], b[i] = wi, ai, bi

prefixB = [0] * (n + 1)
suffixB = [0] * (n + 1)
for i in range(n):
  prefixB[i + 1] = prefixB[i] + b[i]
for i in range(n - 1, -1, -1):
  suffixB[i] = suffixB[i + 1] + b[i]

# f[i][m]：从第 i 人开始、初始银两 m 的最终银两
f = [[0] * MAXM for _ in range(n + 1)]
for m in range(MAXM):
  f[n][m] = m

for i in range(n - 1, -1, -1):
  wi, ai, bi = w[i], a[i], b[i]
  sb = suffixB[i + 1]
  fi1 = f[i + 1]
  row = f[i]
  for m in range(MAXM):
    if wi < m:
      nm = m - bi
      if nm < 0:
        nm = 0
    else:
      nm = m + ai
    if nm < MAXM:
      row[m] = fi1[nm]
    elif nm - sb > THRESH:
      row[m] = nm - sb
    else:
      j, cur = i, nm
      while j < n and cur > THRESH:
        cur -= b[j]
        j += 1
      row[m] = cur if j >= n else f[j][cur]


def apply_from(i, money):
  if i >= n:
    return money
  if money > THRESH and money - suffixB[i] > THRESH:
    return money - suffixB[i]
  if money < MAXM:
    return f[i][money]
  j, cur = i, money
  while j < n and cur > THRESH:
    cur -= b[j]
    j += 1
  return cur if j >= n else f[j][cur]


def answer(x):
  if x < MAXM:
    return f[0][x]
  if x - suffixB[0] > THRESH:
    return x - suffixB[0]
  lo, hi = 0, n
  while lo < hi:
    mid = (lo + hi + 1) // 2
    if x - prefixB[mid] > THRESH:
      lo = mid
    else:
      hi = mid - 1
  i = lo
  money = x - prefixB[i]
  return money if i >= n else apply_from(i, money)


q = int(input())
for _ in range(q):
  print(answer(int(input())))
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
  static final int MAXM = 2000;
  static final int THRESH = 500;

  static int n;
  static int[] w, a, b;
  static long[] prefixB, suffixB;
  static long[][] f;

  static long applyFrom(int i, long money) {
    if (i >= n) return money;
    if (money > THRESH && money - suffixB[i] > THRESH) return money - suffixB[i];
    if (money < MAXM) return f[i][(int) money];
    int j = i;
    long cur = money;
    while (j < n && cur > THRESH) {
      cur -= b[j];
      j++;
    }
    return j >= n ? cur : f[j][(int) cur];
  }

  static long answer(long x) {
    if (x < MAXM) return f[0][(int) x];
    if (x - suffixB[0] > THRESH) return x - suffixB[0];
    int lo = 0, hi = n;
    while (lo < hi) {
      int mid = (lo + hi + 1) / 2;
      if (x - prefixB[mid] > THRESH) lo = mid;
      else hi = mid - 1;
    }
    int i = lo;
    long money = x - prefixB[i];
    return i >= n ? money : applyFrom(i, money);
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    n = Integer.parseInt(br.readLine());
    w = new int[n];
    a = new int[n];
    b = new int[n];
    for (int i = 0; i < n; i++) {
      String[] sp = br.readLine().split(" ");
      w[i] = Integer.parseInt(sp[0]);
      a[i] = Integer.parseInt(sp[1]);
      b[i] = Integer.parseInt(sp[2]);
    }

    prefixB = new long[n + 1];
    suffixB = new long[n + 1];
    for (int i = 0; i < n; i++) prefixB[i + 1] = prefixB[i] + b[i];
    for (int i = n - 1; i >= 0; i--) suffixB[i] = suffixB[i + 1] + b[i];

    f = new long[n + 1][MAXM];
    for (int m = 0; m < MAXM; m++) f[n][m] = m;

    for (int i = n - 1; i >= 0; i--) {
      for (int m = 0; m < MAXM; m++) {
        long nm = w[i] < m ? Math.max(0, m - b[i]) : m + a[i];
        if (nm >= MAXM) {
          if (nm - suffixB[i + 1] > THRESH) f[i][m] = nm - suffixB[i + 1];
          else {
            int j = i;
            long cur = nm;
            while (j < n && cur > THRESH) {
              cur -= b[j];
              j++;
            }
            f[i][m] = j >= n ? cur : f[j][(int) cur];
          }
        } else {
          f[i][m] = f[i + 1][(int) nm];
        }
      }
    }

    int q = Integer.parseInt(br.readLine());
    StringBuilder out = new StringBuilder();
    while (q-- > 0) {
      out.append(answer(Long.parseLong(br.readLine()))).append('\n');
    }
    System.out.print(out);
  }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXM = 2000;
const int THRESH = 500;

int main() {
  int n;
  cin >> n;
  vector<int> w(n), a(n), b(n);
  for (int i = 0; i < n; i++) cin >> w[i] >> a[i] >> b[i];

  vector<long long> prefixB(n + 1, 0), suffixB(n + 1, 0);
  for (int i = 0; i < n; i++) prefixB[i + 1] = prefixB[i] + b[i];
  for (int i = n - 1; i >= 0; i--) suffixB[i] = suffixB[i + 1] + b[i];

  vector<vector<long long>> f(n + 1, vector<long long>(MAXM));
  for (int m = 0; m < MAXM; m++) f[n][m] = m;

  for (int i = n - 1; i >= 0; i--) {
    for (int m = 0; m < MAXM; m++) {
      long long nm = w[i] < m ? max(0LL, m - b[i]) : m + a[i];
      if (nm >= MAXM) {
        if (nm - suffixB[i + 1] > THRESH) f[i][m] = nm - suffixB[i + 1];
        else {
          int j = i;
          long long cur = nm;
          while (j < n && cur > THRESH) {
            cur -= b[j];
            j++;
          }
          f[i][m] = j >= n ? cur : f[j][cur];
        }
      } else {
        f[i][m] = f[i + 1][nm];
      }
    }
  }

  auto applyFrom = [&](int i, long long money) -> long long {
    if (i >= n) return money;
    if (money > THRESH && money - suffixB[i] > THRESH) return money - suffixB[i];
    if (money < MAXM) return f[i][money];
    int j = i;
    long long cur = money;
    while (j < n && cur > THRESH) {
      cur -= b[j];
      j++;
    }
    return j >= n ? cur : f[j][cur];
  };

  auto answer = [&](long long x) -> long long {
    if (x < MAXM) return f[0][x];
    if (x - suffixB[0] > THRESH) return x - suffixB[0];
    int lo = 0, hi = n;
    while (lo < hi) {
      int mid = (lo + hi + 1) / 2;
      if (x - prefixB[mid] > THRESH) lo = mid;
      else hi = mid - 1;
    }
    int i = lo;
    long long money = x - prefixB[i];
    return i >= n ? money : applyFrom(i, money);
  };

  int q;
  cin >> q;
  while (q--) {
    long long x;
    cin >> x;
    cout << answer(x) << "\n";
  }
  return 0;
}
```
