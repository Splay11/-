# 门厅最短通道

## 解题思路

图是无向无权的，从 $1$ 号点走到其他点的最短路就是最少边数，用 BFS 即可。

1. 用邻接表存双向边。重边不影响正确性，自环在第一次访问时已被标记，不会再入队。
2. 从 $1$ 号点出发做 BFS，$dist[1]=0$，第一次走到某点时的层数就是最短距离。
3. 走不到的点保持不可达，不输出。
4. 把所有可达点按距离升序、距离相同按编号升序排序后输出「编号 距离」。

## 复杂度分析

- 时间复杂度：$O(p+e+p\log p)$，BFS 线性，排序至多 $p$ 个点。
- 空间复杂度：$O(p+e)$。

## 代码实现

### Python

```python
from collections import deque


def solve(n, edges):
  # 无向无权图：邻接表存双向边
  g = [[] for _ in range(n + 1)]
  for u, v in edges:
    g[u].append(v)
    g[v].append(u)
  # dist[i] = -1 表示还没走到；1 号距离为 0
  dist = [-1] * (n + 1)
  dist[1] = 0
  q = deque([1])
  while q:
    u = q.popleft()
    for v in g[u]:
      if dist[v] < 0:
        dist[v] = dist[u] + 1
        q.append(v)
  # 只收集可达点，按（距离，编号）排序
  arr = []
  for i in range(1, n + 1):
    if dist[i] >= 0:
      arr.append((dist[i], i))
  arr.sort()
  return arr


n, m = map(int, input().split())
edges = []
for _ in range(m):
  u, v = map(int, input().split())
  edges.append((u, v))
ans = solve(n, edges)
for d, i in ans:
  print(i, d)
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Collections;
import java.util.StringTokenizer;

public class Main {
  static ArrayList<int[]> solve(int n, int[][] edges) {
    // 无向无权图：邻接表存双向边
    ArrayList<Integer>[] g = new ArrayList[n + 1];
    for (int i = 1; i <= n; i++) g[i] = new ArrayList<Integer>();
    for (int[] e : edges) {
      g[e[0]].add(e[1]);
      g[e[1]].add(e[0]);
    }
    // dist[i] = -1 表示还没走到
    int[] dist = new int[n + 1];
    for (int i = 1; i <= n; i++) dist[i] = -1;
    dist[1] = 0;
    ArrayDeque<Integer> q = new ArrayDeque<Integer>();
    q.add(1);
    while (!q.isEmpty()) {
      int u = q.poll();
      for (int v : g[u]) {
        if (dist[v] < 0) {
          dist[v] = dist[u] + 1;
          q.add(v);
        }
      }
    }
    ArrayList<int[]> arr = new ArrayList<int[]>();
    for (int i = 1; i <= n; i++) {
      if (dist[i] >= 0) arr.add(new int[] {dist[i], i});
    }
    Collections.sort(arr, (a, b) -> {
      if (a[0] != b[0]) return a[0] - b[0];
      return a[1] - b[1];
    });
    return arr;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    int n = Integer.parseInt(st.nextToken());
    int m = Integer.parseInt(st.nextToken());
    int[][] edges = new int[m][2];
    for (int i = 0; i < m; i++) {
      st = new StringTokenizer(br.readLine());
      edges[i][0] = Integer.parseInt(st.nextToken());
      edges[i][1] = Integer.parseInt(st.nextToken());
    }
    ArrayList<int[]> ans = solve(n, edges);
    StringBuilder sb = new StringBuilder();
    for (int[] t : ans) {
      sb.append(t[1]).append(' ').append(t[0]).append('\n');
    }
    System.out.print(sb.toString());
  }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<pair<int, int> > solve(int n, const vector<pair<int, int> >& edges) {
  // 无向无权图：邻接表存双向边
  vector<vector<int> > g(n + 1);
  for (size_t i = 0; i < edges.size(); i++) {
    int u = edges[i].first, v = edges[i].second;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  vector<int> dist(n + 1, -1);
  dist[1] = 0;
  queue<int> q;
  q.push(1);
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    for (size_t j = 0; j < g[u].size(); j++) {
      int v = g[u][j];
      if (dist[v] < 0) {
        dist[v] = dist[u] + 1;
        q.push(v);
      }
    }
  }
  vector<pair<int, int> > arr;
  for (int i = 1; i <= n; i++) {
    if (dist[i] >= 0) arr.push_back(make_pair(dist[i], i));
  }
  sort(arr.begin(), arr.end());
  return arr;
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  vector<pair<int, int> > edges;
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    edges.push_back(make_pair(u, v));
  }
  vector<pair<int, int> > ans = solve(n, edges);
  for (size_t i = 0; i < ans.size(); i++) {
    cout << ans[i].second << " " << ans[i].first << "\n";
  }
  return 0;
}
```
