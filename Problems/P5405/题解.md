# 叉车装车优惠

## 解题思路

本题必须按编号 $1\sim n$ 依次决定每件货物装或不装，装上之后占用体积不能超过 $W$。占用体积达到阈值 $T$ 之后，再装的货物按 $\lfloor v_i/2\rfloor$ 计入。目标是最大货值。

下面先给出**记忆化搜索**（最优递归），再改写成标准 DP。两种做法的状态完全一致，复杂度也相同。

1. 设当前要处理第 $i$ 件（下标从 $0$ 开始），叉车上已经占用的体积是 $vol$，优惠是否已经触发记为 $trig$（$0$ 未触发，$1$ 已触发）。这一状态下，后面还能拿到的最大货值记为 $dfs(i,vol,trig)$。
2. 边界：若 $i=n$，没有剩余货物，返回 $0$。一件都不装是合法的，答案至少是 $0$。
3. 转移有两支：
   * 不装第 $i$ 件：直接 $dfs(i+1,vol,trig)$。
   * 装第 $i$ 件：占用 $cost=trig?\lfloor v_i/2\rfloor:v_i$。若 $vol+cost\le W$，新体积 $nvol=vol+cost$，新触发标记 $ntrig=(trig\ \text{或}\ nvol\ge T)$，这一支的货值是 $w_i+dfs(i+1,nvol,ntrig)$。
4. 两支取较大值。同一组 $(i,vol,trig)$ 会被大量重复访问，用数组 $memo[i][vol][trig]$ 记下第一次算出的结果，即记忆化搜索。
5. 把递归改成按货物下标从小到大刷表，就是标准 DP。定义 $dp[j][k]$ 表示「已经处理完当前这件之前的所有货物、占用体积为 $j$、触发标记为 $k$」时的最大货值。不可达记为 $-1$。初始 $dp[0][0]=0$。按件转移时，先把旧层复制到新层（对应不装），再枚举可达状态尝试装入。

样例 $2$ 说明不能贪心先装：第一件体积大、货值低，装上会挤掉后面两件高货值货物，最优是跳过它。

## 复杂度分析

记忆化搜索与标准 DP 的状态数都是 $O(nW)$，每个状态枚举装或不装，转移 $O(1)$。

* 时间复杂度：$O(nW)$。
* 空间复杂度：$O(nW)$（记忆化三维数组）；标准 DP 滚动后可为 $O(W)$。

$n\le 100$，$W\le 5000$，可以通过。纯回溯不记忆化是 $O(2^n)$，只能过小数据。

## 代码实现

### Python

```python
def max_value(n, W, T, v, w):
    # memo[i][vol][trig]：从第 i 件、占用 vol、触发标记 trig 往后的最大货值
    # -1 表示还没算过
    memo = [[[-1, -1] for _ in range(W + 1)] for _ in range(n + 1)]

    def dfs(i, vol, trig):
        # 货物已经决策完
        if i == n:
            return 0
        if memo[i][vol][trig] != -1:
            return memo[i][vol][trig]
        # 不装第 i 件
        best = dfs(i + 1, vol, trig)
        # 装第 i 件：未触发用原体积，已触发用折半
        cost = v[i] // 2 if trig else v[i]
        nvol = vol + cost
        if nvol <= W:
            ntrig = 1 if (trig or nvol >= T) else 0
            take = w[i] + dfs(i + 1, nvol, ntrig)
            if take > best:
                best = take
        memo[i][vol][trig] = best
        return best

    return dfs(0, 0, 0)


def main():
    n, W, T = map(int, input().split())
    v = []
    w = []
    for _ in range(n):
        vi, wi = map(int, input().split())
        v.append(vi)
        w.append(wi)
    print(max_value(n, W, T, v, w))


if __name__ == "__main__":
    main()
```

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static int n, W, T;
    static int[] v, w;
    static int[][][] memo;

    static int dfs(int i, int vol, int trig) {
        // 货物已经决策完
        if (i == n) {
            return 0;
        }
        if (memo[i][vol][trig] != -1) {
            return memo[i][vol][trig];
        }
        // 不装第 i 件
        int best = dfs(i + 1, vol, trig);
        // 装第 i 件：未触发用原体积，已触发用折半
        int cost = trig == 1 ? v[i] / 2 : v[i];
        int nvol = vol + cost;
        if (nvol <= W) {
            int ntrig = (trig == 1 || nvol >= T) ? 1 : 0;
            int take = w[i] + dfs(i + 1, nvol, ntrig);
            if (take > best) {
                best = take;
            }
        }
        memo[i][vol][trig] = best;
        return best;
    }

    static int maxValue(int n_, int W_, int T_, int[] v_, int[] w_) {
        n = n_;
        W = W_;
        T = T_;
        v = v_;
        w = w_;
        memo = new int[n + 1][W + 1][2];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= W; j++) {
                Arrays.fill(memo[i][j], -1);
            }
        }
        return dfs(0, 0, 0);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int W = sc.nextInt();
        int T = sc.nextInt();
        int[] v = new int[n];
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            v[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        System.out.println(maxValue(n, W, T, v, w));
    }
}
```

# 标准 DP 解法

## 解题思路

把上面的递归改成自底向上即可。

1. 用 $dp[j][k]$ 表示当前占用体积为 $j$、触发标记为 $k$ 时的最大货值。不可达为 $-1$。
2. 初始只有 $dp[0][0]=0$（空车、未触发）。
3. 按货物 $i=0\sim n-1$ 依次转移。先把本轮 $new\_dp$ 复制成上一轮的 $dp$，对应「不装这一件」。
4. 再枚举上一轮所有可达状态 $(j,k)$：若装得下，令 $cost=k?\lfloor v_i/2\rfloor:v_i$，$nvol=j+cost$，$ntrig=(k\ \text{或}\ nvol\ge T)$，用 $dp[j][k]+w_i$ 更新 $new\_dp[nvol][ntrig]$。
5. 全部货物处理完后，所有 $dp[j][k]$ 的最大值就是答案。

## 复杂度分析

* 时间复杂度：$O(nW)$。
* 空间复杂度：$O(W)$（滚动两层体积数组）。

## 代码实现

### Python

```python
def max_value(n, W, T, v, w):
    NEG = -1
    # dp[vol][trig]：占用 vol、触发标记 trig 的最大货值
    dp = [[NEG, NEG] for _ in range(W + 1)]
    dp[0][0] = 0
    for i in range(n):
        # 先复制，表示不装第 i 件
        new_dp = [row[:] for row in dp]
        for vol in range(W + 1):
            for trig in (0, 1):
                if dp[vol][trig] < 0:
                    continue
                # 装第 i 件：未触发用原体积，已触发用折半

                if nvol > W:
                    continue
                ntrig = 1 if (trig or nvol >= T) else 0
                val = dp[vol][trig] + w[i]
                if val > new_dp[nvol][ntrig]:
                    new_dp[nvol][ntrig] = val
        dp = new_dp
    ans = 0
    for vol in range(W + 1):
        for trig in (0, 1):
            if dp[vol][trig] > ans:
                ans = dp[vol][trig]
    return ans
```

### Java

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static long maxValue(int n, int W, int T, int[] v, int[] w) {
        final long NEG = -1;
        // dp[vol][trig]：占用 vol、触发标记 trig 的最大货值
        long[][] dp = new long[W + 1][2];
        for (int i = 0; i <= W; i++) {
            Arrays.fill(dp[i], NEG);
        }
        dp[0][0] = 0;
        for (int i = 0; i < n; i++) {
            // 先复制，表示不装第 i 件
            long[][] newDp = new long[W + 1][2];
            for (int vol = 0; vol <= W; vol++) {
                newDp[vol][0] = dp[vol][0];
                newDp[vol][1] = dp[vol][1];
            }
            for (int vol = 0; vol <= W; vol++) {
                for (int trig = 0; trig <= 1; trig++) {
                    if (dp[vol][trig] < 0) {
                        continue;
                    }
                    // 装第 i 件：未触发用原体积，已触发用折半
                    int cost = trig == 1 ? v[i] / 2 : v[i];
                    int nvol = vol + cost;
                    if (nvol > W) {
                        continue;
                    }
                    int ntrig = (trig == 1 || nvol >= T) ? 1 : 0;
                    long val = dp[vol][trig] + w[i];
                    if (val > newDp[nvol][ntrig]) {
                        newDp[nvol][ntrig] = val;
                    }
                }
            }
            dp = newDp;
        }
        long ans = 0;
        for (int vol = 0; vol <= W; vol++) {
            for (int trig = 0; trig <= 1; trig++) {
                if (dp[vol][trig] > ans) {
                    ans = dp[vol][trig];
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int W = sc.nextInt();
        int T = sc.nextInt();
        int[] v = new int[n];
        int[] w = new int[n];

            v[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        System.out.println(maxValue(n, W, T, v, w));
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

long long maxValue(int n, int W, int T, const vector<int>& v, const vector<int>& w) {
    const long long NEG = -1;
    // dp[vol][trig]：占用 vol、触发标记 trig 的最大货值
    vector<vector<long long>> dp(W + 1, vector<long long>(2, NEG));
    dp[0][0] = 0;
    for (int i = 0; i < n; i++) {
        // 先复制，表示不装第 i 件
        vector<vector<long long>> newDp = dp;
        for (int vol = 0; vol <= W; vol++) {
            for (int trig = 0; trig <= 1; trig++) {
                if (dp[vol][trig] < 0) {
                    continue;
                }
                // 装第 i 件：未触发用原体积，已触发用折半
                int cost = trig ? (v[i] / 2) : v[i];
                int nvol = vol + cost;
                if (nvol > W) {
                    continue;
                }
                int ntrig = (trig || nvol >= T) ? 1 : 0;
                long long val = dp[vol][trig] + w[i];
                if (val > newDp[nvol][ntrig]) {
                    newDp[nvol][ntrig] = val;
                }
            }
        }
        dp.swap(newDp);
    }
    long long ans = 0;
    for (int vol = 0; vol <= W; vol++) {
        for (int trig = 0; trig <= 1; trig++) {
            ans = max(ans, dp[vol][trig]);
        }
    }
    return ans;
}

int main() {
    int n, W, T;
    cin >> n >> W >> T;
    vector<int> v(n), w(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i] >> w[i];
    }
    cout << maxValue(n, W, T, v, w) << "\n";
    return 0;
}
```
