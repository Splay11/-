# 回溯暴力解法

这道题的正解是 $动态规划$，但是我们用朴素解法 $回溯$ 也能在考试时拿到一定的分数。

必须按编号 $1\sim n$ 依次决定每件货物装或不装。占用体积达到 $T$ 之后，再装的货物按 $\lfloor v_i/2\rfloor$ 计入，且占用不能超过 $W$。目标是最大货值，允许一件都不装。

朴素做法：从第 $0$ 件开始做「不装 / 装」的回溯。

1. 走到第 $n$ 件时，用当前货值更新答案。
2. 不装：体积和触发标记都不变，去看下一件。
3. 装：未触发用 $v_i$，已触发用 $\lfloor v_i/2\rfloor$。装得下才递归，装上后若体积 $\ge T$ 则之后算已触发。

这样会枚举全部合法装车方案，样例和小 $n$ 能算对。$n$ 较大时是 $O(2^n)$，大数据会超时，所以后面再给出正解。

## 回溯解法

```python
def max_value(n, W, T, v, w):
    ans = 0

    def dfs(i, vol, trig, cur):
        nonlocal ans
        # 全部货物已经决定完
        if i == n:
            if cur > ans:
                ans = cur
            return
        # 不装第 i 件
        dfs(i + 1, vol, trig, cur)
        # 装第 i 件：未触发用原体积，已触发用折半
        cost = v[i] // 2 if trig else v[i]
        nvol = vol + cost
        if nvol <= W:
            ntrig = 1 if (trig or nvol >= T) else 0
            dfs(i + 1, nvol, ntrig, cur + w[i])

    dfs(0, 0, 0, 0)
    return ans


def main():
    n, W, T = map(int, input().split())
    v = []
    w = []
    for _ in range(n):
        vi, wi = map(int, input().split())
        v.append(vi)
        w.append(wi)
    print(max_value(n, W, T, v, w))


if __name__ == "__main__":
    main()
```

```cpp
int nGlobal, WGlobal, TGlobal;
long long ans;
vector<int> v, w;

void dfs(int i, int vol, int trig, long long cur) {
    // 全部货物已经决定完
    if (i == nGlobal) {
        if (cur > ans) {
            ans = cur;
        }
        return;
    }
    // 不装第 i 件
    dfs(i + 1, vol, trig, cur);
    // 装第 i 件：未触发用原体积，已触发用折半
    int cost = trig ? (v[i] / 2) : v[i];

    if (nvol <= WGlobal) {
        int ntrig = (trig || nvol >= TGlobal) ? 1 : 0;
        dfs(i + 1, nvol, ntrig, cur + w[i]);
    }
}
```

# 正确解法

## 题目分析

按编号顺序做 $0/1$ 背包，额外记录优惠有没有触发。

定义 $dp[j][k]$：当前占用体积为 $j$、触发标记为 $k$（$0$ 未触发，$1$ 已触发）时的最大货值。不可达记为 $-1$。

* 初始：$dp[0][0]=0$，其余为 $-1$。
* 不装第 $i$ 件：状态原样保留到下一轮。
* 装第 $i$ 件：从可达的 $(j,k)$ 出发，$cost=k?\lfloor v_i/2\rfloor:v_i$。若 $j+cost\le W$，新体积 $nvol=j+cost$，新标记 $ntrig=(k\ \text{或}\ nvol\ge T)$，用 $dp[j][k]+w_i$ 更新 $new\_dp[nvol][ntrig]$。

全部货物处理完后，所有 $dp[j][k]$ 的最大值就是答案。样例 $2$ 这类「先装低货值会挤掉后面高货值」的情况，DP 会比较跳过与装入两支，不会贪错。

## 复杂度分析

* 时间复杂度：$O(nW)$，每件货物枚举体积与触发标记。
* 空间复杂度：$O(W)$，滚动两层体积数组。

$n\le 100$，$W\le 5000$，可以通过。

## 代码实现

### Python

```python
def max_value(n, W, T, v, w):
    NEG = -1
    # dp[vol][trig]：占用 vol、触发标记 trig 的最大货值
    dp = [[NEG, NEG] for _ in range(W + 1)]
    dp[0][0] = 0
    for i in range(n):
        # 先复制，表示不装第 i 件
        new_dp = [row[:] for row in dp]
        for vol in range(W + 1):
            for trig in (0, 1):
                if dp[vol][trig] < 0:
                    continue
                # 装第 i 件：未触发用原体积，已触发用折半
                cost = v[i] // 2 if trig else v[i]
                nvol = vol + cost
                if nvol > W:
                    continue
                ntrig = 1 if (trig or nvol >= T) else 0
                val = dp[vol][trig] + w[i]
                if val > new_dp[nvol][ntrig]:
                    new_dp[nvol][ntrig] = val
        dp = new_dp
    ans = 0
    for vol in range(W + 1):
        for trig in (0, 1):
            if dp[vol][trig] > ans:
                ans = dp[vol][trig]
    return ans


def main():
    n, W, T = map(int, input().split())
    v = []
    w = []
    for _ in range(n):
        vi, wi = map(int, input().split())
        v.append(vi)
        w.append(wi)
    print(max_value(n, W, T, v, w))


if __name__ == "__main__":
    main()
```

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static long maxValue(int n, int W, int T, int[] v, int[] w) {
        final long NEG = -1;
        // dp[vol][trig]：占用 vol、触发标记 trig 的最大货值
        long[][] dp = new long[W + 1][2];
        for (int i = 0; i <= W; i++) {
            Arrays.fill(dp[i], NEG);
        }
        dp[0][0] = 0;
        for (int i = 0; i < n; i++) {
            // 先复制，表示不装第 i 件
            long[][] newDp = new long[W + 1][2];
            for (int vol = 0; vol <= W; vol++) {
                newDp[vol][0] = dp[vol][0];
                newDp[vol][1] = dp[vol][1];
            }
            for (int vol = 0; vol <= W; vol++) {
                for (int trig = 0; trig <= 1; trig++) {
                    if (dp[vol][trig] < 0) {
                        continue;
                    }
                    // 装第 i 件：未触发用原体积，已触发用折半
                    int cost = trig == 1 ? v[i] / 2 : v[i];
                    int nvol = vol + cost;
                    if (nvol > W) {
                        continue;
                    }
                    int ntrig = (trig == 1 || nvol >= T) ? 1 : 0;
                    long val = dp[vol][trig] + w[i];
                    if (val > newDp[nvol][ntrig]) {
                        newDp[nvol][ntrig] = val;
                    }
                }
            }
            dp = newDp;
        }
        long ans = 0;
        for (int vol = 0; vol <= W; vol++) {
            for (int trig = 0; trig <= 1; trig++) {
                if (dp[vol][trig] > ans) {
                    ans = dp[vol][trig];
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int W = sc.nextInt();
        int T = sc.nextInt();
        int[] v = new int[n];
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            v[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        System.out.println(maxValue(n, W, T, v, w));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

long long maxValue(int n, int W, int T, const vector<int>& v, const vector<int>& w) {
    const long long NEG = -1;
    // dp[vol][trig]：占用 vol、触发标记 trig 的最大货值
    vector<vector<long long>> dp(W + 1, vector<long long>(2, NEG));
    dp[0][0] = 0;
    for (int i = 0; i < n; i++) {
        // 先复制，表示不装第 i 件
        vector<vector<long long>> new_dp = dp;
        for (int vol = 0; vol <= W; vol++) {
            for (int trig = 0; trig <= 1; trig++) {
                if (dp[vol][trig] < 0) {
                    continue;
                }
                // 装第 i 件：未触发用原体积，已触发用折半
                int cost = trig ? (v[i] / 2) : v[i];
                int nvol = vol + cost;
                if (nvol > W) {
                    continue;
                }
                int ntrig = (trig || nvol >= T) ? 1 : 0;
                long long val = dp[vol][trig] + w[i];
                if (val > new_dp[nvol][ntrig]) {
                    new_dp[nvol][ntrig] = val;
                }
            }
        }
        dp.swap(new_dp);
    }
    long long ans = 0;
    for (int vol = 0; vol <= W; vol++) {
        for (int trig = 0; trig <= 1; trig++) {
            ans = max(ans, dp[vol][trig]);
        }
    }
    return ans;
}

int main() {
    int n, W, T;
    cin >> n >> W >> T;
    vector<int> v(n), w(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i] >> w[i];
    }
    cout << maxValue(n, W, T, v, w) << "\n";
    return 0;
}
```
