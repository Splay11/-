# 链表中倒数第 k 个节点

算法标签：链表、双指针

算法难度：$3 / 10$

## 解题思路

要找单向链表的倒数第 $k$ 个节点。题目保证 $1\le k\le n$，所以一定能找到。

1. 用快慢指针：快指针先沿 $next$ 走 $k$ 步，然后快慢一起走。快指针走到空时，慢指针正好停在倒数第 $k$ 个。
2. 快指针先走 $k$ 步之后，它和慢指针之间隔了 $k$ 个节点。两者再同步前进，快指针把剩下的 $n-k$ 步走完，慢指针也就从开头走了 $n-k$ 步，落在下标 $n-k$ 上（从 $0$ 计），即倒数第 $k$ 个。
3. 也可以先数出长度再从头走 $n-k$ 步，但双指针只要一遍扫描，且不依赖提前知道 $n$（本题输入里给了 $n$，两种都行）。
4. 常见假解：把倒数第 $k$ 理解成下标 $k$；快指针只走 $k-1$ 步，答案偏成倒数第 $k-1$ 个。

## 复杂度分析

- 时间复杂度：$O(n)$。建表和双指针各扫一遍。
- 空间复杂度：$O(n)$。存放 $n$ 个链表节点。若只算查找过程，额外指针是 $O(1)$。

题面未写 $n$ 和 $a_i$ 的上界，本套数据按邻近链表题取 $n\le 10^4$、$-10^9\le a_i\le 10^9$。

## 代码实现

### Python

```python
class Node:
    def __init__(self, val):
        self.val = val
        self.next = None


def build_list(vals):
    dummy = Node(0)
    cur = dummy
    for v in vals:
        cur.next = Node(v)
        cur = cur.next
    return dummy.next


def kth_from_end(head, k):
    # 快指针先走 k 步
    fast = head
    for _ in range(k):
        fast = fast.next
    slow = head
    # 一起走，快指针到空时慢指针在倒数第 k 个
    while fast is not None:
        fast = fast.next
        slow = slow.next
    return slow.val


def solve(vals, k):
    head = build_list(vals)
    return kth_from_end(head, k)


def main():
    n, k = map(int, input().split())
    vals = list(map(int, input().split()))
    print(solve(vals, k))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;

public class Main {
    static class Node {
        int val;
        Node next;
        Node(int v) {
            val = v;
        }
    }

    static Node buildList(int[] vals) {
        Node dummy = new Node(0);
        Node cur = dummy;
        for (int v : vals) {
            cur.next = new Node(v);
            cur = cur.next;
        }
        return dummy.next;
    }

    static int kthFromEnd(Node head, int k) {
        Node fast = head;
        for (int i = 0; i < k; i++) {
            fast = fast.next;
        }
        Node slow = head;
        while (fast != null) {
            fast = fast.next;
            slow = slow.next;
        }
        return slow.val;
    }

    static int solve(int[] vals, int k) {
        Node head = buildList(vals);
        return kthFromEnd(head, k);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(br.readLine());
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = Integer.parseInt(st.nextToken());
        }
        System.out.println(solve(vals, k));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

struct Node {
    int val;
    Node* next;
    Node(int v) : val(v), next(nullptr) {}
};

Node* buildList(const vector<int>& vals) {
    Node dummy(0);
    Node* cur = &dummy;
    for (int v : vals) {
        cur->next = new Node(v);
        cur = cur->next;
    }
    return dummy.next;
}

int kthFromEnd(Node* head, int k) {
    Node* fast = head;
    for (int i = 0; i < k; i++) {
        fast = fast->next;
    }
    Node* slow = head;
    while (fast != nullptr) {
        fast = fast->next;
        slow = slow->next;
    }
    return slow->val;
}

int solve(const vector<int>& vals, int k) {
    Node* head = buildList(vals);
    return kthFromEnd(head, k);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k;
    cin >> n >> k;
    vector<int> vals(n);
    for (int i = 0; i < n; i++) {
        cin >> vals[i];
    }
    cout << solve(vals, k) << '\n';
    return 0;
}
```
