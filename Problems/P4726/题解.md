## 解题思路

本题在 $n \times m$ 网格上统计从**唯一全局最小值**格子走到**唯一全局最大值**格子的路径条数。每一步只能走向四邻域中海拔**严格更高**的格子，且高度差不超过 `maxDiff`。路径上每个格子最多访问一次；由于海拔沿步严格递增，图中不存在环路，从任意格子沿合法边走向更高海拔构成有向无环图（DAG），因此可用 **记忆化搜索（DFS + memo）** 自终点（或自起点）统计路径数。

具体做法：

1. 扫描网格得到全局最小值、最大值及其坐标 $(s_x,s_y)$、$(t_x,t_y)$（题面保证各唯一）。
2. 定义 `dfs(i,j)`：从 $(i,j)$ 出发，沿「海拔上升且步长 $\in(0,\text{maxDiff}]$」的四连通边走到终点的路径数。
3. 边界：若 $(i,j)$ 为终点，返回 $1$；否则对四个方向尝试扩展，累加合法邻居的 `dfs` 结果。
4. 用 `memo[i][j]`（或哈希表）记录已算状态，避免重复计算；严格递增保证无环，记忆化安全。

常见假解：把问题当成最短路径只数层；忽略 `maxDiff` 上界；未固定全局最小为起点；在有多条汇入时不用记忆化导致超时（本题 $n,m\le10$ 规模下暴力也可过，但记忆化是标准写法）。

## 复杂度分析

记 $V=nm \le 100$。每个格子至多被 `dfs` 计算一次，每条有向边至多被松弛常数次，故时间复杂度 $O(nm)$，空间 $O(nm)$（网格本身与记忆化表同阶）。

## 代码实现

### Python

```python
from functools import lru_cache
from typing import List


class Solution:
    def countHikingPaths(self, grid: List[List[int]], maxDiff: int) -> int:
        n, m = len(grid), len(grid[0])
        # 找出唯一最小、最大及其坐标
        mn = min(grid[i][j] for i in range(n) for j in range(m))
        mx = max(grid[i][j] for i in range(n) for j in range(m))
        si = sj = ei = ej = -1
        for i in range(n):
            for j in range(m):
                if grid[i][j] == mn:
                    si, sj = i, j
                if grid[i][j] == mx:
                    ei, ej = i, j
        dirs = ((1, 0), (-1, 0), (0, 1), (0, -1))

        @lru_cache(maxsize=None)
        def dfs(i: int, j: int) -> int:
            if i == ei and j == ej:
                return 1
            h = grid[i][j]
            tot = 0
            for di, dj in dirs:
                ni, nj = i + di, j + dj
                if 0 <= ni < n and 0 <= nj < m:
                    nh = grid[ni][nj]
                    d = nh - h
                    # 严格上升且单步高度差不超过 maxDiff
                    if 0 < d <= maxDiff:
                        tot += dfs(ni, nj)
            return tot

        return dfs(si, sj)
```

### Java

```java
import java.util.*;

public class Solution {
    private int n, m, si, sj, ei, ej, maxDiff;
    private int[][] grid;
    private long[][] memo;
    private boolean[][] done;

    public long countHikingPaths(int[][] grid, int maxDiff) {
        this.grid = grid;
        this.maxDiff = maxDiff;
        n = grid.length;
        m = grid[0].length;
        // 预处理全局最小/最大与起终点坐标
        int mn = grid[0][0], mx = grid[0][0];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                mn = Math.min(mn, grid[i][j]);
                mx = Math.max(mx, grid[i][j]);
            }
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == mn) {
                    si = i;
                    sj = j;
                }
                if (grid[i][j] == mx) {
                    ei = i;
                    ej = j;
                }
            }
        memo = new long[n][m];
        done = new boolean[n][m];
        return dfs(si, sj);
    }

    private long dfs(int i, int j) {
        if (i == ei && j == ej) return 1;
        if (done[i][j]) return memo[i][j];
        done[i][j] = true;
        long tot = 0;
        int h = grid[i][j];
        int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        for (int[] d : dirs) {
            int ni = i + d[0], nj = j + d[1];
            if (ni < 0 || ni >= n || nj < 0 || nj >= m) continue;
            int nh = grid[ni][nj];
            int diff = nh - h;
            // 仅统计严格递增且步长在 (0, maxDiff] 的边
            if (diff > 0 && diff <= maxDiff) tot += dfs(ni, nj);
        }
        memo[i][j] = tot;
        return tot;
    }
}
```

### C++

```cpp
#include <algorithm>
#include <functional>
#include <vector>

using namespace std;

class Solution {
   public:
    long long countHikingPaths(vector<vector<int>>& grid, int maxDiff) {
        int n = (int)grid.size(), m = (int)grid[0].size();
        int mn = grid[0][0], mx = grid[0][0];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                mn = min(mn, grid[i][j]);
                mx = max(mx, grid[i][j]);
            }
        int si = -1, sj = -1, ei = -1, ej = -1;
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == mn) si = i, sj = j;
                if (grid[i][j] == mx) ei = i, ej = j;
            }
        static long long memo[12][12];
        static int vis[12][12];
        static int stamp = 1;
        ++stamp;
        const int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        function<long long(int, int)> dfs = [&](int i, int j) -> long long {
            if (i == ei && j == ej) return 1;
            if (vis[i][j] == stamp) return memo[i][j];
            vis[i][j] = stamp;
            long long tot = 0;
            int h = grid[i][j];
            for (int k = 0; k < 4; k++) {
                int ni = i + dirs[k][0], nj = j + dirs[k][1];
                if (ni < 0 || ni >= n || nj < 0 || nj >= m) continue;
                int nh = grid[ni][nj];
                int d = nh - h;
                if (d > 0 && d <= maxDiff) tot += dfs(ni, nj);
            }
            memo[i][j] = tot;
            return tot;
        };
        return dfs(si, sj);
    }
};
```
