# 驻场耗材

## 解题思路

第 $i$ 个课题组获批后要从第 $i$ 天驻到第 $n$ 天，共 $n-i+1$ 天，总消耗为 $v_i\cdot(n-i+1)$。库存上限是 $b$，目标是批准的个数最多。

1. 每批准一个课题组，人数加 $1$，与是谁无关，只与消耗有关。
2. 因此把所有人的总消耗算出来，从小到大排序，依次批准，直到下一个人会超出 $b$。
3. 这就是价值全为 $1$ 的背包：按重量升序贪心最优。
4. $n\le 100$，排序即可。

## 复杂度分析

时间复杂度为 $O(n\log n)$。

空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
def max_accept(n, b, v):
    # 第 i 人从当天驻场到第 n 天，共 (n-i) 天（下标从 0 计），每天耗 v[i]
    costs = []
    for i in range(n):
        costs.append(v[i] * (n - i))
    # 每人贡献都是 1，要人数最多就优先批准花费更小的
    costs.sort()
    used = 0
    ans = 0
    for c in costs:
        if used + c <= b:
            used += c
            ans += 1
        else:
            break
    return ans


def main():
    n, b = map(int, input().split())
    v = list(map(int, input().split()))
    print(max_accept(n, b, v))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static int maxAccept(int n, long b, int[] v) {
        long[] costs = new long[n];
        for (int i = 0; i < n; i++) {
            // 下标从 0 计：从当天到第 n 天共 (n-i) 天
            costs[i] = (long) v[i] * (n - i);
        }
        Arrays.sort(costs);
        long used = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (used + costs[i] <= b) {
                used += costs[i];
                ans++;
            } else {
                break;
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long b = sc.nextLong();
        int[] v = new int[n];
        for (int i = 0; i < n; i++) {
            v[i] = sc.nextInt();
        }
        System.out.println(maxAccept(n, b, v));
        sc.close();
    }
}
```

### C++

```cpp
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int max_accept(int n, long long b, const vector<int>& v) {
    vector<long long> costs(n);
    for (int i = 0; i < n; i++) {
        // 下标从 0 计：从当天到第 n 天共 (n-i) 天
        costs[i] = 1LL * v[i] * (n - i);
    }
    sort(costs.begin(), costs.end());
    long long used = 0;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (used + costs[i] <= b) {
            used += costs[i];
            ans++;
        } else {
            break;
        }
    }
    return ans;
}

int main() {
    int n;
    long long b;
    cin >> n >> b;
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }
    cout << max_accept(n, b, v) << '\n';
    return 0;
}
```

### C

```c
#include <stdio.h>
#include <stdlib.h>

int cmp_ll(const void *a, const void *b) {
    long long x = *(const long long *)a;
    long long y = *(const long long *)b;
    if (x < y) return -1;
    if (x > y) return 1;
    return 0;
}

int max_accept(int n, long long b, int *v) {
    long long *costs = (long long *)malloc((size_t)n * sizeof(long long));
    long long used = 0;
    int i, ans = 0;
    for (i = 0; i < n; i++) {
        /* 下标从 0 计：从当天到第 n 天共 (n-i) 天 */
        costs[i] = (long long)v[i] * (n - i);
    }
    qsort(costs, (size_t)n, sizeof(long long), cmp_ll);
    for (i = 0; i < n; i++) {
        if (used + costs[i] <= b) {
            used += costs[i];
            ans++;
        } else {
            break;
        }
    }
    free(costs);
    return ans;
}

int main(void) {
    int n, i;
    long long b;
    int *v;
    scanf("%d%lld", &n, &b);
    v = (int *)malloc((size_t)n * sizeof(int));
    for (i = 0; i < n; i++) {
        scanf("%d", &v[i]);
    }
    printf("%d\n", max_accept(n, b, v));
    free(v);
    return 0;
}
```
