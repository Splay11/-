# 货位好坏线性判别

## 解题思路

这是写死规则的线性分类：补偏置、标签映成正负一、全零起步，固定十轮、学习率一，按原顺序遇错就更新。

1. 每条训练特征前面加 `1`，标签 `0`/`1` 变成 `-1`/`+1`。
2. 权向量 $h$ 从全 `0` 开始。一轮里按输入顺序扫样本，用 $h^\top \bar{v} \ge 0$ 判成 $+1$ 否则 $-1$；判错就把 $t\cdot\bar{v}$ 加到 $h$ 上。
3. 这样连走 `10` 轮。测试时用同一套符号函数，再把 $+1$ 映回 `1`、$-1` 映回 `0`。
4. 不要改成别的优化器或提前停，输出顺序必须与 `test` 一致。

## 复杂度分析

- 时间复杂度：$O(\textit{epochs}\cdot n\cdot d)$，本题轮数固定为 `10`。
- 空间复杂度：$O(nd)$。

## 代码实现

### Python

```python
import json
import numpy as np


def solve(train, test):
  # 标签 0/1 映到 -1/+1，特征前面补 1 当作偏置
  X = []
  y = []
  for feat, lab in train:
    X.append([1.0] + [float(v) for v in feat])
    y.append(-1.0 if lab == 0 else 1.0)
  X = np.array(X, dtype=float)
  y = np.array(y, dtype=float)
  # 权向量从全 0 开始，学习率 1，固定走 10 轮
  h = np.zeros(X.shape[1], dtype=float)
  for _ in range(10):
    for i in range(len(X)):
      score = float(np.dot(h, X[i]))
      pred = 1.0 if score >= 0.0 else -1.0
      # 分错才更新：权向量沿正确方向加一刀
      if pred != y[i]:
        h = h + y[i] * X[i]
  ans = []
  for feat in test:
    xb = np.array([1.0] + [float(v) for v in feat], dtype=float)
    score = float(np.dot(h, xb))
    pred = 1.0 if score >= 0.0 else -1.0
    ans.append(1 if pred == 1.0 else 0)
  return ans


raw = input()
more = []
try:
  while True:
    more.append(input())
except EOFError:
  pass
if more:
  raw = raw + "\n" + "\n".join(more)
obj = json.loads(raw)
print(json.dumps(solve(obj["train"], obj["test"])))
```

import java.io.BufferedReader;
import java.io.InputStreamReader;

public class Main {
  static String src;
  static int pos;

  static void skip() {
    while (pos < src.length()) {
      char c = src.charAt(pos);
      if (c != ' ' && c != '\n' && c != '\r' && c != '\t') break;
      pos++;
    }
  }

  static void eat(char c) {
    skip();
    pos++;
  }

  static double readNum() {
    skip();
    int j = pos;
    if (src.charAt(j) == '-') j++;
    while (j < src.length()) {
      char c = src.charAt(j);
      if ((c >= '0' && c <= '9') || c == '.' || c == 'e' || c == 'E' || c == '+' || c == '-') j++;
      else break;
    }
    double v = Double.parseDouble(src.substring(pos, j));
    pos = j;
    return v;
  }

  static double[] readVec() {
    eat('[');
    skip();
    if (src.charAt(pos) == ']') {
      pos++;
      return new double[0];
    }
    double[] tmp = new double[64];
    int n = 0;
    while (true) {
      if (n == tmp.length) {
        double[] b = new double[n * 2];
        System.arraycopy(tmp, 0, b, 0, n);
        tmp = b;
      }
      tmp[n++] = readNum();
      skip();
      if (src.charAt(pos) == ',') {
        pos++;
        continue;
      }
      eat(']');
      break;
    }
    double[] a = new double[n];
    System.arraycopy(tmp, 0, a, 0, n);
    return a;
  }

  static int[] solve(double[][] feats, double[] labs, double[][] test) {
    int n = feats.length;
    int d = feats[0].length + 1;
    double[][] X = new double[n][d];
    double[] t = new double[n];
    for (int i = 0; i < n; i++) {
      X[i][0] = 1.0;
      for (int j = 0; j < feats[i].length; j++) X[i][j + 1] = feats[i][j];
      t[i] = labs[i] == 0 ? -1.0 : 1.0;
    }
    // 权向量从 0 起，固定 10 轮，分错才加一刀
    double[] h = new double[d];
    for (int ep = 0; ep < 10; ep++) {
      for (int i = 0; i < n; i++) {
        double score = 0;
        for (int j = 0; j < d; j++) score += h[j] * X[i][j];
        double pred = score >= 0 ? 1.0 : -1.0;
        if (pred != t[i]) {
          for (int j = 0; j < d; j++) h[j] += t[i] * X[i][j];
        }
      }
    }
    int[] ans = new int[test.length];
    for (int i = 0; i < test.length; i++) {
      double score = h[0];
      for (int j = 0; j < test[i].length; j++) score += h[j + 1] * test[i][j];
      ans[i] = score >= 0 ? 1 : 0;
    }
    return ans;
  }

    pos = 0;
    // 只解析 train / test 两个键
    double[][] feats = new double[0][];
    double[] labs = new double[0];

    while (true) {

      if (src.charAt(pos) == '}') break;

      int keyStart = pos;
      while (src.charAt(pos) != '"') pos++;
      String key = src.substring(keyStart, pos);
      pos++;
      eat(':');
      if (key.equals("train")) {
        eat('[');
        double[][] tf = new double[16][];
        double[] tl = new double[16];

        if (src.charAt(pos) != ']') {

            eat(']');
            if (tn == tf.length) {
              double[][] nf = new double[tn * 2][];
              double[] nl = new double[tn * 2];

              tl = nl;

          }

        System.arraycopy(tf, 0, feats, 0, tn);
        System.arraycopy(tl, 0, labs, 0, tn);
      } else if (key.equals("test")) {

```javascript
function solve(train, test) {
  var X = [], y = [];
  for (var i = 0; i < train.length; i++) {
    var feat = train[i][0], lab = train[i][1];
    X.push([1].concat(feat.map(Number)));
    y.push(lab === 0 ? -1 : 1);
  }
  var d = X[0].length;
  var h = [];
  for (var j = 0; j < d; j++) h.push(0);
  for (var ep = 0; ep < 10; ep++) {
    for (var i = 0; i < X.length; i++) {
      var score = 0;
      for (var j = 0; j < d; j++) score += h[j] * X[i][j];
      var pred = score >= 0 ? 1 : -1;
      if (pred !== y[i]) {
        for (var j = 0; j < d; j++) h[j] += y[i] * X[i][j];
      }
    }
  }
  var ans = [];
  for (var i = 0; i < test.length; i++) {
    var score = h[0];
    for (var j = 0; j < test[i].length; j++) score += h[j + 1] * test[i][j];
    ans.push(score >= 0 ? 1 : 0);
  }
  return ans;
}
```

```c
#include <stdio.h>
#include <stdlib.h>

int *solve(double **X, double *t, int n, int d, double **test, int m, int *outn) {
  double *h = (double *)calloc(d, sizeof(double));
  int ep, i, j;
  for (ep = 0; ep < 10; ep++) {
    for (i = 0; i < n; i++) {
      double score = 0;
      for (j = 0; j < d; j++) score += h[j] * X[i][j];
      double pred = score >= 0 ? 1.0 : -1.0;
      if (pred != t[i]) {
        for (j = 0; j < d; j++) h[j] += t[i] * X[i][j];
      }
    }
  }
  int *ans = (int *)malloc(sizeof(int) * m);
  for (i = 0; i < m; i++) {
    double score = h[0];
    for (j = 1; j < d; j++) score += h[j] * test[i][j - 1];
    ans[i] = score >= 0 ? 1 : 0;
  }
  free(h);
  *outn = m;
  return ans;
}
```
