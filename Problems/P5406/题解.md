# 应急协议

## 解题思路

边权为正，求最短路，但全程至多把一条已有通道的费用改成 $0$。直接把某条边改成 $0$ 再跑 Dijkstra，要对每条边做一次，最大数据过不了。

关键观察：最优路线要么不用协议，要么恰好免费走一条通道 $(u,v)$。后一种情况，路线被拆成三段：$s$ 走到 $u$（全部按原价）、免费穿过这条通道、$v$ 走到 $t$（全部按原价）。两头都是普通最短路。

1. 在原图上从 $s$ 做一次堆优化 Dijkstra，得到 $ds[u]$。
2. 再从 $t$ 做一次，得到 $dt[u]$。无向图不用建反图。
3. 不用协议时，答案是 $ds[t]$。
4. 枚举每条通道 $(u,v)$：免费走 $u\to v$ 的路程是 $ds[u]+dt[v]$，免费走 $v\to u$ 是 $ds[v]+dt[u]$。取所有候选的最小值。
5. 仍是无穷大则输出 $-1$。$s=t$ 时答案为 $0$。

不能只在原来的最短路上挑一条边免费。样例 $1$ 里该免费的是一条看起来很长、根本不在最短路上的通道。路程可能超过 $2^{31}$，用 $64$ 位整数。

## 复杂度分析

- 时间复杂度：$O((n+m)\log n)$，两次 Dijkstra，再 $O(m)$ 枚举通道。
- 空间复杂度：$O(n+m)$。

$n\le 8000$，$m\le 2\times 10^4$，可以通过。对每条边改权再跑最短路是 $O(m(n+m)\log n)$，只能过小数据。

## 代码实现

### Python

```python
import heapq

INF = 10 ** 18


def dijkstra(n, src, adj):
    dist = [INF] * (n + 1)
    dist[src] = 0
    heap = [(0, src)]
    while heap:
        d, u = heapq.heappop(heap)
        if d != dist[u]:
            continue
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(heap, (nd, v))
    return dist


def min_with_one_free(n, s, t, edges, adj):
    # 入口即出口，不必移动
    if s == t:
        return 0
    ds = dijkstra(n, s, adj)
    dt = dijkstra(n, t, adj)
    # 不用协议，整段都按原长度走
    ans = ds[t]
    for u, v, w in edges:
        # 免费走 u -> v：s 到 u 的最短，加上 v 到 t 的最短
        if ds[u] < INF and dt[v] < INF:
            nd = ds[u] + dt[v]
            if nd < ans:
                ans = nd
        # 通道是双向的，另一方向同样可以免费
        if ds[v] < INF and dt[u] < INF:
            nd = ds[v] + dt[u]
            if nd < ans:
                ans = nd
    return -1 if ans >= INF else ans


def main():
    n, m, s, t = map(int, input().split())
    edges = []
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))
    print(min_with_one_free(n, s, t, edges, adj))


if __name__ == "__main__":
    main()
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {
    static final long INF = 1L << 62;

    static long[] dijkstra(int n, int src, ArrayList<int[]>[] adj) {
        long[] dist = new long[n + 1];
        for (int i = 0; i <= n; i++) {
            dist[i] = INF;
        }
        dist[src] = 0;
        PriorityQueue<long[]> heap = new PriorityQueue<long[]>((a, b) -> Long.compare(a[0], b[0]));
        heap.add(new long[] {0, src});
        while (!heap.isEmpty()) {
            long[] cur = heap.poll();
            long d = cur[0];
            int u = (int) cur[1];
            if (d != dist[u]) {
                continue;
            }
            for (int i = 0; i < adj[u].size(); i++) {
                int v = adj[u].get(i)[0];
                int w = adj[u].get(i)[1];
                long nd = d + w;
                if (nd < dist[v]) {
                    dist[v] = nd;
                    heap.add(new long[] {nd, v});
                }
            }
        }
        return dist;
    }

    static long minWithOneFree(int n, int s, int t, int[][] edges, ArrayList<int[]>[] adj) {
        // 入口即出口，不必移动
        if (s == t) {
            return 0;
        }
        long[] ds = dijkstra(n, s, adj);
        long[] dt = dijkstra(n, t, adj);
        long ans = ds[t];
        for (int i = 0; i < edges.length; i++) {
            int u = edges[i][0];
            int v = edges[i][1];
            // 免费走 u -> v
            if (ds[u] < INF / 2 && dt[v] < INF / 2) {
                long nd = ds[u] + dt[v];
                if (nd < ans) {
                    ans = nd;
                }
            }
            // 通道是双向的，另一方向同样可以免费
            if (ds[v] < INF / 2 && dt[u] < INF / 2) {
                long nd = ds[v] + dt[u];
                if (nd < ans) {
                    ans = nd;
                }
            }
        }
        return ans >= INF / 2 ? -1 : ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        int m = Integer.parseInt(st.nextToken());
        int s = Integer.parseInt(st.nextToken());

        ArrayList<int[]>[] adj = new ArrayList[n + 1];
```

```cpp
vector<long long> dijkstra(int n, int src, const vector<vector<pair<int, int> > >& adj) {
    vector<long long> dist(n + 1, INF);
    dist[src] = 0;
    priority_queue<pair<long long, int>, vector<pair<long long, int> >, greater<pair<long long, int> > > heap;
    heap.push(make_pair(0LL, src));
    while (!heap.empty()) {
        long long d = heap.top().first;
        int u = heap.top().second;
        heap.pop();
        if (d != dist[u]) {
            continue;
        }
        for (size_t i = 0; i < adj[u].size(); i++) {
            int v = adj[u][i].first;
            int w = adj[u][i].second;
            long long nd = d + w;
            if (nd < dist[v]) {
                dist[v] = nd;
                heap.push(make_pair(nd, v));
            }
        }
    }
    return dist;
}
```

```javascript
const INF = 1e18;

function dijkstra(n, src, adj) {
    const dist = Array(n + 1).fill(INF);
    dist[src] = 0;
    const heap = [[0, src]];
    const push = (d, u) => {
        heap.push([d, u]);
        let i = heap.length - 1;
        while (i > 0) {
            const p = (i - 1) >> 1;
            if (heap[p][0] <= heap[i][0]) break;
            const tmp = heap[p];
            heap[p] = heap[i];
            heap[i] = tmp;
            i = p;
        }
    };
    const pop = () => {
        const top = heap[0];
        const last = heap.pop();
        if (heap.length === 0) return top;
        heap[0] = last;
        let i = 0;
        while (true) {
            let l = i * 2 + 1;
            let r = l + 1;
            let sm = i;
            if (l < heap.length && heap[l][0] < heap[sm][0]) sm = l;
            if (r < heap.length && heap[r][0] < heap[sm][0]) sm = r;
            if (sm === i) break;
            const tmp = heap[i];
            heap[i] = heap[sm];
            heap[sm] = tmp;
            i = sm;
        }
        return top;
    };
    while (heap.length) {
        const cur = pop();
        const d = cur[0];
        const u = cur[1];
        if (d !== dist[u]) continue;
        for (let i = 0; i < adj[u].length; i++) {
            const v = adj[u][i][0];
            const w = adj[u][i][1];
            const nd = d + w;
            if (nd < dist[v]) {
                dist[v] = nd;
                push(nd, v);
            }
        }
    }
    return dist;
}

function minWithOneFree(n, s, t, edges, adj) {
    // 入口即出口，不必移动
    if (s === t) {
        return 0;
    }
    const ds = dijkstra(n, s, adj);
    const dt = dijkstra(n, t, adj);
    let ans = ds[t];
    for (let i = 0; i < edges.length; i++) {
        const u = edges[i][0];
        const v = edges[i][1];
        // 免费走 u -> v
        if (ds[u] < INF && dt[v] < INF) {
            const nd = ds[u] + dt[v];
            if (nd < ans) ans = nd;
        }
        // 通道是双向的，另一方向同样可以免费
        if (ds[v] < INF && dt[u] < INF) {
            const nd = ds[v] + dt[u];
            if (nd < ans) ans = nd;
        }
    }
    return ans >= INF ? -1 : ans;
}

function main(input) {
    const lines = input.trim().split(/\n/);
    const head = lines[0].trim().split(/\s+/).map(Number);
    const n = head[0], m = head[1], s = head[2], t = head[3];
    const edges = [];
    const adj = Array.from({ length: n + 1 }, () => []);
    for (let i = 0; i < m; i++) {
        const a = lines[1 + i].trim().split(/\s+/).map(Number);
        edges.push(a);
        adj[a[0]].push([a[1], a[2]]);
        adj[a[1]].push([a[0], a[2]]);
    }
    console.log(minWithOneFree(n, s, t, edges, adj));
}
```

### C

```c
#include <stdio.h>
#include <string.h>

#define INF 4000000000000000000LL
#define MAXN 8010
#define MAXE 50010

typedef struct {
    int to, w, next;
} Edge;

Edge edges[MAXE];
int headArr[MAXN];
int ecnt;
int eu[20005], ev[20005], ew[20005];
long long heapD[1000005];
int heapU[1000005];
int hsz;

void add(int u, int v, int w) {
    edges[ecnt].to = v;
    edges[ecnt].w = w;
    edges[ecnt].next = headArr[u];
    headArr[u] = ecnt++;
}

void heapPush(long long d, int u) {
    int i = ++hsz;
    heapD[i] = d;
    heapU[i] = u;
    while (i > 1) {
        int p = i / 2;
        if (heapD[p] <= heapD[i]) break;
        long long td = heapD[p];
        int tu = heapU[p];
        heapD[p] = heapD[i];
        heapU[p] = heapU[i];
        heapD[i] = td;
        heapU[i] = tu;
        i = p;
    }
}

void heapPop(long long *d, int *u) {
    *d = heapD[1];
    *u = heapU[1];
    heapD[1] = heapD[hsz];
    heapU[1] = heapU[hsz];
    hsz--;
    int i = 1;
    while (1) {
        int l = i * 2, r = l + 1, sm = i;
        if (l <= hsz && heapD[l] < heapD[sm]) sm = l;
        if (r <= hsz && heapD[r] < heapD[sm]) sm = r;
        if (sm == i) break;
        long long td = heapD[i];
        int tu = heapU[i];
        heapD[i] = heapD[sm];
        heapU[i] = heapU[sm];
        heapD[sm] = td;
        heapU[sm] = tu;
        i = sm;
    }
}

void dijkstra(int n, int src, long long dist[]) {
    int i;
    for (i = 1; i <= n; i++) dist[i] = INF;
    dist[src] = 0;
    hsz = 0;
    heapPush(0, src);
    while (hsz > 0) {
        long long d;
        int u;
        heapPop(&d, &u);
        if (d != dist[u]) continue;
        int e;
        for (e = headArr[u]; e != -1; e = edges[e].next) {
            int v = edges[e].to;
            int w = edges[e].w;
            long long nd = d + w;
            if (nd < dist[v]) {
                dist[v] = nd;
                heapPush(nd, v);
            }
        }
    }
}

long long minWithOneFree(int n, int s, int t, int m) {
    static long long ds[MAXN], dt[MAXN];
    int i;
    // 入口即出口，不必移动
    if (s == t) {
        return 0;
    }
    dijkstra(n, s, ds);
    dijkstra(n, t, dt);
    long long ans = ds[t];
    for (i = 0; i < m; i++) {
        int u = eu[i], v = ev[i];
        // 免费走 u -> v
        if (ds[u] < INF / 2 && dt[v] < INF / 2) {
            long long nd = ds[u] + dt[v];
            if (nd < ans) ans = nd;
        }
        // 通道是双向的，另一方向同样可以免费
        if (ds[v] < INF / 2 && dt[u] < INF / 2) {
            long long nd = ds[v] + dt[u];
            if (nd < ans) ans = nd;
        }
    }
    return ans >= INF / 2 ? -1 : ans;
}

int main(void) {
    int n, m, s, t, i;
    scanf("%d%d%d%d", &n, &m, &s, &t);
    memset(headArr, -1, sizeof(headArr));
    ecnt = 0;
    for (i = 0; i < m; i++) {
        scanf("%d%d%d", &eu[i], &ev[i], &ew[i]);
        add(eu[i], ev[i], ew[i]);
        add(ev[i], eu[i], ew[i]);
    }
    printf("%lld\n", minWithOneFree(n, s, t, m));
    return 0;
}
```
