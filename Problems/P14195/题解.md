# P14195 所有模块的构建顺序 — 题解

## 解题思路

题意给出模块集合与依赖列表，每条依赖 **`[x, y]` 表示 $x$ 依赖 $y$**，即合法构建顺序中 **$y$ 必须出现在 $x$ 的左侧**。在图论上，将每条依赖建成有向边 **$y \rightarrow x$**（先构建 $y$，再构建 $x$），问题转化为求 **所有拓扑排序**。

- **判环**：若图存在有向环，则不存在能包含全部顶点的拓扑序，应返回空数组。可用 **Kahn 算法**（不断删除入度为 $0$ 的顶点并删边）统计能否删完 $N$ 个顶点；若不能，说明有环。
- **枚举所有拓扑序**：在确认无环后，用 **DFS + 回溯** 维护当前路径 `path` 与动态入度表 `indeg`：
  - 每一步在「尚未放入路径且当前入度为 $0$」的模块中，按 **模块名字典序** 依次尝试（保证搜索顺序稳定，但最终仍需对生成的整行字符串列表排序，与题意「多条结果按字典序」一致）。
  - 选中模块 $m$ 后，将其加入路径，并对所有 $m \rightarrow v$ 的边执行 `indeg[v]--`；回溯时恢复。
  - 当路径长度达到 $N$ 时，将 `" ".join(path)` 加入答案。
- **输出格式**：将所有完整顺序字符串收集后 **`sorted`** 一次，保证与题面「按字典序排序后输出」一致。

**区分度 / 易错点**：依赖方向不要反（应建 $y \rightarrow x$）；有环时不能只靠 DFS「走不满 $N$」就当作无输出——必须先做全局可拓扑性判定；多解时必须对**结果字符串**排序，而不是只对每一步候选排序。

**测试设计简述**：样例与官方一致；另含链、菱形、无依赖全排列、并行汇合、单点/双边等小数据；在题面保证「合法拓扑序数量 $<10!$」前提下控制规模，避免指数爆炸超出评测资源。

## 复杂度分析

设 $N$ 为模块数，$M$ 为依赖条数，$K$ 为合法拓扑序条数（题面保证 $K < 10!$）。

- **建图与判环（Kahn）**：时间 $O(N+M)$，空间 $O(N+M)$。
- **枚举所有拓扑序**：搜索树规模与 $K$ 及 DAG 结构相关；在题面保证下可接受。实现上每步需扫描入度为 $0$ 的候选，粗略上界可写为 $O(K \cdot N^2)$ 的保守估计；数据保证 $K$ 不大，实际可过。
- **空间**：除邻接表外，递归栈深度 $O(N)$，路径数组 $O(N)$。

## 代码实现

### Python

```python
from collections import defaultdict, deque
from typing import Dict, List


class Solution:
    def allBuildOrders(self, modules: List[str], dependencies: List[List[str]]) -> List[str]:
        if not modules:
            return []
        # indeg[x]：在当前 DAG 中，还有多少条来自「必须先于 x」的边未满足
        indeg: Dict[str, int] = {m: 0 for m in modules}
        # g[u]：所有「依赖 u」的模块 v，即边 u -> v（u 必须先构建）
        g: Dict[str, List[str]] = defaultdict(list)
        for a, b in dependencies:
            if a not in indeg or b not in indeg:
                continue
            g[b].append(a)
            indeg[a] += 1

        # Kahn：若能消掉全部 N 个点则无环，否则直接返回 []
        q = deque([m for m in modules if indeg[m] == 0])
        tmp = indeg.copy()
        seen = 0
        while q:
            u = q.popleft()
            seen += 1
            for v in g[u]:
                tmp[v] -= 1
                if tmp[v] == 0:
                    q.append(v)
        if seen != len(modules):
            return []

        indeg_mut = indeg.copy()
        path: List[str] = []
        out: List[str] = []

        def dfs() -> None:
            if len(path) == len(modules):
                # 一条完整构建顺序：模块名用单个空格连接
                out.append(" ".join(path))
                return
            # 仅允许当前入度为 0 且尚未加入路径的模块作为下一步
            cand = sorted([m for m in modules if m not in path and indeg_mut[m] == 0])
            for m in cand:
                path.append(m)
                for nei in g[m]:
                    indeg_mut[nei] -= 1
                dfs()
                for nei in g[m]:
                    indeg_mut[nei] += 1
                path.pop()

        dfs()
        return sorted(out)
```

### Java

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Deque;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

class Solution {
    public List<String> allBuildOrders(List<String> modules, List<List<String>> dependencies) {
        if (modules.isEmpty()) {
            return new ArrayList<>();
        }
        Map<String, Integer> indeg = new HashMap<>();
        for (String m : modules) {
            indeg.put(m, 0);
        }
        Map<String, List<String>> g = new HashMap<>();
        for (List<String> e : dependencies) {
            String a = e.get(0), b = e.get(1);
            if (!indeg.containsKey(a) || !indeg.containsKey(b)) {
                continue;
            }
            g.computeIfAbsent(b, k -> new ArrayList<>()).add(a);
            indeg.put(a, indeg.get(a) + 1);
        }

        // Kahn 判环：复制入度，避免污染后续 DFS 使用的初始入度
        Map<String, Integer> tmp = new HashMap<>(indeg);
        Deque<String> q = new ArrayDeque<>();
        for (String m : modules) {
            if (tmp.get(m) == 0) {
                q.addLast(m);
            }
        }
        int seen = 0;
        while (!q.isEmpty()) {
            String u = q.removeFirst();
            seen++;
            for (String v : g.getOrDefault(u, new ArrayList<>())) {
                tmp.put(v, tmp.get(v) - 1);
                if (tmp.get(v) == 0) {
                    q.addLast(v);
                }
            }
        }
        if (seen != modules.size()) {
            return new ArrayList<>();
        }

        Map<String, Integer> indegMut = new HashMap<>(indeg);
        List<String> path = new ArrayList<>();
        List<String> out = new ArrayList<>();

        dfsAll(modules, g, indegMut, path, out);
        Collections.sort(out);
        return out;
    }

    private void dfsAll(
            List<String> modules,
            Map<String, List<String>> g,
            Map<String, Integer> indegMut,
            List<String> path,
            List<String> out) {
        if (path.size() == modules.size()) {
            out.add(String.join(" ", path));
            return;
        }
        Set<String> inPath = new HashSet<>(path);
        List<String> cand = new ArrayList<>();
        for (String m : modules) {
            if (!inPath.contains(m) && indegMut.get(m) == 0) {
                cand.add(m);
            }
        }
        Collections.sort(cand);
        for (String m : cand) {
            path.add(m);
            for (String nei : g.getOrDefault(m, new ArrayList<>())) {
                indegMut.put(nei, indegMut.get(nei) - 1);
            }
            dfsAll(modules, g, indegMut, path, out);
            for (String nei : g.getOrDefault(m, new ArrayList<>())) {
                indegMut.put(nei, indegMut.get(nei) + 1);
            }
            path.remove(path.size() - 1);
        }
    }
}
```

### C++

```cpp
#include <algorithm>
#include <queue>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <vector>
using namespace std;

class Solution {
public:
    vector<string> allBuildOrders(vector<string>& modules, vector<vector<string>>& dependencies) {
        if (modules.empty()) {
            return {};
        }
        unordered_map<string, int> indeg;
        unordered_map<string, vector<string>> g;
        for (auto& m : modules) {
            indeg[m] = 0;
        }
        for (auto& e : dependencies) {
            const string& a = e[0];
            const string& b = e[1];
            if (!indeg.count(a) || !indeg.count(b)) {
                continue;
            }
            g[b].push_back(a);
            indeg[a]++;
        }

        // Kahn：判断是否能完成全部模块的拓扑剥离
        auto tmp = indeg;
        queue<string> q;
        for (auto& m : modules) {
            if (tmp[m] == 0) {
                q.push(m);
            }
        }
        int seen = 0;
        while (!q.empty()) {
            string u = q.front();
            q.pop();
            seen++;
            for (auto& v : g[u]) {
                if (--tmp[v] == 0) {
                    q.push(v);
                }
            }
        }
        if (seen != (int)modules.size()) {
            return {};
        }

        unordered_map<string, int> indegMut = indeg;
        vector<string> path;
        vector<string> out;
        dfsAll(modules, g, indegMut, path, out);
        sort(out.begin(), out.end());
        return out;
    }

private:
    void dfsAll(
            vector<string>& modules,
            unordered_map<string, vector<string>>& g,
            unordered_map<string, int>& indegMut,
            vector<string>& path,
            vector<string>& out) {
        if ((int)path.size() == (int)modules.size()) {
            string line;
            for (size_t i = 0; i < path.size(); i++) {
                if (i) {
                    line.push_back(' ');
                }
                line += path[i];
            }
            out.push_back(line);
            return;
        }
        unordered_set<string> inPath(path.begin(), path.end());
        vector<string> cand;
        for (auto& m : modules) {
            if (!inPath.count(m) && indegMut[m] == 0) {
                cand.push_back(m);
            }
        }
        sort(cand.begin(), cand.end());
        for (auto& m : cand) {
            path.push_back(m);
            for (auto& nei : g[m]) {
                indegMut[nei]--;
            }
            dfsAll(modules, g, indegMut, path, out);
            for (auto& nei : g[m]) {
                indegMut[nei]++;
            }
            path.pop_back();
        }
    }
};
```
