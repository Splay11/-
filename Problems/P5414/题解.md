# 挑选四件礼物

## 解题思路

要从 $m$ 件礼品里恰好挑 $4$ 件，总价不超过 $10000$，总价尽量大；总价相同则编号升序后取字典序最小的一组。凑不齐就输出 $0$。

1. 先按编号从小到大排序。后面按这个顺序决定选不选，才能直接得到字典序最小的方案。
2. 用 $0$-$1$ 背包记录「恰好选了 $k$ 件时，哪些总价能凑出来」。$k$ 最多是 $4$，总价上限是 $10000$，用位集压缩 $s$ 这一维。

定义 $dp[i][k][s]$：只用下标 $i,i+1,\ldots,n-1$ 这些礼品（已按编号升序），恰好选 $k$ 件、总价恰好为 $s$ 是否可行。

边界：空后缀只能凑出 $0$ 件、$0$ 元。

$$
dp[n][0][0]=\mathrm{true},\quad \text{其余 }dp[n][k][s]=\mathrm{false}
$$

转移：第 $i$ 件选或不选。

$$
dp[i][k][s]=dp[i+1][k][s]\ \lor\ \bigl(k\ge 1\land s\ge w[i]\land dp[i+1][k-1][s-w[i]]\bigr)
$$

位集把 $s$ 压进二进制，左移 $w[i]$ 位就是「选第 $i$ 件、总价整体加上 $w[i]$」：

$$
dp[i][k]=dp[i+1][k]\ \lor\ \bigl(dp[i+1][k-1]\ll w[i]\bigr)
$$

超过 $10000$ 的位丢掉。最大总价

$$
S=\max\{\,s\mid dp[0][4][s]=\mathrm{true},\ 0\le s\le 10000\,\}
$$

若不存在这样的 $s$，输出 $0$。从左到右能选第 $i$ 件，当且仅当 $dp[i+1][remain\_k-1][remain\_s-w[i]]$ 为真。

3. 从「全部礼品、恰好 $4$ 件」的位集里取出 $S$。若一位都没有，答案就是 $0$。
4. 再预处理后缀：从每一个下标 $i$ 开始往后，恰好选 $k$ 件能凑出哪些总价。这样判断「当前这件选不选」只需看剩下的人能不能补齐。
5. 从左到右扫编号。设还要选 $remain\_k$ 件、还要凑 $remain\_s$ 元。若选下标 $i$ 之后，后缀仍能凑出 $remain\_k-1$ 件、$remain\_s-w[i]$ 元，就选它。能选则选，得到的就是字典序最小的那一组。

四重循环枚举在 $m=2000$ 时大约 $2\times 10^{11}$ 次，会超时。位集每次移位合并，总时间大约是 $O(m\times 4\times 10000/w)$，可以通过。

## 复杂度分析

* 时间复杂度：$O(m\cdot 4\cdot 10000/w)$，其中 $w$ 是机器字长或位集实现里一次能处理的位数。
* 空间复杂度：$O(m\cdot 4\cdot 10000/w)$，要存每一层后缀位集。

$m\le 2000$，可以通过。

## 代码实现

### Python

```python
CAP = 10000
NEED = 4


def pick_four(b, w):
    n = len(b)
    if n < NEED:
        return None
    order = sorted(range(n), key=lambda i: b[i])
    b = [b[i] for i in order]
    w = [w[i] for i in order]
    mask = (1 << (CAP + 1)) - 1
    # suffix[i][k]：从下标 i 到末尾恰好选 k 件，能凑出的总价集合
    cur = [0] * (NEED + 1)
    cur[0] = 1
    suffix = [None] * (n + 1)
    suffix[n] = cur[:]
    for i in range(n - 1, -1, -1):
        nxt = cur[:]
        ww = w[i]
        for k in range(NEED - 1, -1, -1):
            # dp[i][k][s]：从 i 往后恰好选 k 件，总价能否为 s
            # 不选第 i 件：nxt 已拷贝 cur，即继承 dp[i+1][k]
            # 选第 i 件：左移 w[i] 位等于总价整体加上售价，或进 dp[i][k+1]
            # k 从大到小，保证每件最多用一次；掩码丢掉超过 10000 的总价
            nxt[k + 1] = (nxt[k + 1] | (cur[k] << ww)) & mask
        cur = nxt
        suffix[i] = cur[:]
    bits = suffix[0][NEED]
    if bits == 0:
        return None
    remain_s = bits.bit_length() - 1
    remain_k = NEED
    ans = []
    for i in range(n):
        if remain_k == 0:
            break
        ns = remain_s - w[i]
        nk = remain_k - 1
        # 判断 dp[i+1][nk][ns] 是否为真：选完当前件后后缀还能不能凑齐
        # 编号已从小到大排好，能选就选，得到字典序最小的一组
        if ns >= 0 and ((suffix[i + 1][nk] >> ns) & 1):
            ans.append(b[i])
            remain_s = ns
            remain_k = nk
    return ans


def main():
    m = int(input())
    b = []
    w = []
    for _ in range(m):
        bi, wi = map(int, input().split())
        b.append(bi)
        w.append(wi)
    ans = pick_four(b, w)
    if ans is None:
        print(0)
    else:
        print(" ".join(str(x) for x in ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.BitSet;
import java.util.Scanner;

public class Main {
    static final int CAP = 10000;
    static final int NEED = 4;

    static ArrayList<Integer> pickFour(int[] b, int[] w) {
        int n = b.length;
        if (n < NEED) {
            return null;
        }
        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) {
            order[i] = i;
        }
        Arrays.sort(order, (i, j) -> Integer.compare(b[i], b[j]));
        int[] bb = new int[n];
        int[] ww = new int[n];
        for (int i = 0; i < n; i++) {
            bb[i] = b[order[i]];
            ww[i] = w[order[i]];
        }

        BitSet[][] suffix = new BitSet[n + 1][NEED + 1];
        for (int i = 0; i <= n; i++) {
            for (int k = 0; k <= NEED; k++) {
                suffix[i][k] = new BitSet(CAP + 1);
            }
        }
        suffix[n][0].set(0);
        for (int i = n - 1; i >= 0; i--) {
            for (int k = 0; k <= NEED; k++) {
                suffix[i][k] = (BitSet) suffix[i + 1][k].clone();
            }
            for (int k = NEED - 1; k >= 0; k--) {
                // dp[i][k][s]：从 i 往后恰好选 k 件，总价能否为 s
                // clone 后缀是不选第 i 件，继承 dp[i+1][k]
                // 左移 w[i] 再或上去，是选第 i 件、总价加上售价
                // k 从大到小，保证每件最多用一次
                suffix[i][k + 1].or(shiftLeft(suffix[i + 1][k], ww[i]));
            }
        }
        if (suffix[0][NEED].isEmpty()) {
            return null;
        }
        int remainS = suffix[0][NEED].previousSetBit(CAP);
        int remainK = NEED;
        ArrayList<Integer> ans = new ArrayList<>();
        for (int i = 0; i < n && remainK > 0; i++) {
            int ns = remainS - ww[i];
            int nk = remainK - 1;
            // 判断 dp[i+1][nk][ns] 是否可行：选完当前件后后缀还能不能凑齐
            // 编号已升序，能选就选，得到字典序最小方案
            if (ns >= 0 && suffix[i + 1][nk].get(ns)) {
                ans.add(bb[i]);
                remainS = ns;
                remainK = nk;
            }
        }
        return ans;
    }

    static BitSet shiftLeft(BitSet src, int sh) {
        BitSet dst = new BitSet(CAP + 1);
        for (int s = src.nextSetBit(0); s >= 0; s = src.nextSetBit(s + 1)) {
            if (s + sh <= CAP) {
                dst.set(s + sh);
            }
        }
        return dst;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int[] b = new int[m];
        int[] w = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        ArrayList<Integer> ans = pickFour(b, w);
        if (ans == null) {
            System.out.println(0);
        } else {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < ans.size(); i++) {
                if (i > 0) {
                    sb.append(' ');
                }
                sb.append(ans.get(i));
            }
            System.out.println(sb.toString());
        }
        sc.close();
    }
}
```

```cpp
#include <bitset>
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int CAP = 10000;
const int NEED = 4;
const int MAXN = 2005;

static bitset<CAP + 1> suffix[MAXN][NEED + 1];

vector<int> pickFour(vector<int> b, vector<int> w) {
    int n = (int)b.size();
    if (n < NEED) {
        return {};
    }
    vector<int> order(n);
    for (int i = 0; i < n; i++) {
        order[i] = i;
    }
    sort(order.begin(), order.end(), [&](int i, int j) {
        return b[i] < b[j];
    });
    vector<int> bb(n), ww(n);
    for (int i = 0; i < n; i++) {
        bb[i] = b[order[i]];
        ww[i] = w[order[i]];
    }

    for (int k = 0; k <= NEED; k++) {
        suffix[n][k].reset();
    }
    suffix[n][0].set(0);
    for (int i = n - 1; i >= 0; i--) {
        for (int k = 0; k <= NEED; k++) {
            suffix[i][k] = suffix[i + 1][k];
        }
        for (int k = NEED - 1; k >= 0; k--) {
            // dp[i][k][s]：从 i 往后恰好选 k 件，总价能否为 s
            // 先拷贝 suffix[i+1]，对应不选第 i 件
            // 再或上左移 w[i] 位，对应选第 i 件后总价整体加上售价
            // k 从大到小，保证每件最多用一次
            suffix[i][k + 1] |= (suffix[i + 1][k] << ww[i]);
        }
    }
    if (suffix[0][NEED].none()) {
        return {};
    }
    int remainS = CAP;
    while (remainS >= 0 && !suffix[0][NEED].test(remainS)) {
        remainS--;
    }
    int remainK = NEED;
    vector<int> ans;
    for (int i = 0; i < n && remainK > 0; i++) {
        int ns = remainS - ww[i];
        int nk = remainK - 1;
        // 判断 dp[i+1][nk][ns] 是否可行：选完当前件后后缀还能不能凑齐
        // 编号已升序，能选就选，得到字典序最小方案
        if (ns >= 0 && suffix[i + 1][nk].test(ns)) {
            ans.push_back(bb[i]);
            remainS = ns;
            remainK = nk;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int m;
    cin >> m;
    vector<int> b(m), w(m);
    for (int i = 0; i < m; i++) {
        cin >> b[i] >> w[i];
    }
    vector<int> ans = pickFour(b, w);
    if (ans.empty()) {
        cout << 0 << '\n';
    } else {
        for (int i = 0; i < (int)ans.size(); i++) {
            if (i) cout << ' ';
            cout << ans[i];
        }
        cout << '\n';
    }
    return 0;
}
```

```javascript
const CAP = 10000;
const NEED = 4;

function pickFour(b, w) {
  const n = b.length;
  if (n < NEED) {
    return null;
  }
  const order = [...Array(n).keys()].sort((i, j) => b[i] - b[j]);
  b = order.map((i) => b[i]);
  w = order.map((i) => w[i]);
  const mask = (1n << BigInt(CAP + 1)) - 1n;
  let cur = Array(NEED + 1).fill(0n);
  cur[0] = 1n;
  const suffix = Array(n + 1);
  suffix[n] = cur.slice();
  for (let i = n - 1; i >= 0; i--) {
    const nxt = cur.slice();
    const ww = BigInt(w[i]);
    for (let k = NEED - 1; k >= 0; k--) {
      // dp[i][k][s]：从 i 往后恰好选 k 件，总价能否为 s
      // nxt 先拷贝 cur，对应不选第 i 件
      // 左移 w[i] 再或上去，对应选第 i 件、总价加上售价
      // k 从大到小，保证每件最多用一次；掩码丢掉超过 10000 的位
      nxt[k + 1] = (nxt[k + 1] | (cur[k] << ww)) & mask;
    }
    cur = nxt;
    suffix[i] = cur.slice();
  }
  const bits = suffix[0][NEED];
  if (bits === 0n) {
    return null;
  }
  let remainS = bits.toString(2).length - 1;
  let remainK = NEED;
  const ans = [];
  for (let i = 0; i < n && remainK > 0; i++) {
    const ns = remainS - w[i];
    const nk = remainK - 1;
    // 判断 dp[i+1][nk][ns] 是否可行：选完当前件后后缀还能不能凑齐
    // 编号已升序，能选就选，得到字典序最小方案
    if (ns >= 0 && ((suffix[i + 1][nk] >> BigInt(ns)) & 1n) === 1n) {
      ans.push(b[i]);
      remainS = ns;
      remainK = nk;
    }
  }
  return ans;
}
```

### C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define CAP 10000
#define NEED 4
#define MAXN 2005
#define WORDS 157

unsigned long long suffix[MAXN][NEED + 1][WORDS];
int b[MAXN], w[MAXN], order[MAXN];

int cmp_order(const void *x, const void *y) {
    int i = *(const int *)x;
    int j = *(const int *)y;
    return b[i] - b[j];
}

void bits_copy(unsigned long long *dst, unsigned long long *src) {
    memcpy(dst, src, sizeof(unsigned long long) * WORDS);
}

void bits_or_shift(unsigned long long *dst, unsigned long long *src, int sh) {
    int sw = sh / 64, sb = sh % 64;
    int i;
    for (i = WORDS - 1; i >= 0; i--) {
        unsigned long long v = 0;
        int from = i - sw;
        if (from >= 0) {
            v = src[from] << sb;
            if (sb && from - 1 >= 0) {
                v |= src[from - 1] >> (64 - sb);
            }
        }
        dst[i] |= v;
    }
    dst[WORDS - 1] &= (1ULL << ((CAP + 1) % 64)) - 1ULL;
}

int bits_test(unsigned long long *src, int pos) {
    return (int)((src[pos / 64] >> (pos % 64)) & 1ULL);
}

int bits_none(unsigned long long *src) {
    int i;
    for (i = 0; i < WORDS; i++) {
        if (src[i]) {
            return 0;
        }
    }
    return 1;
}

int main(void) {
    int m, i, k, n;
    int bb[MAXN], ww[MAXN];
    int remain_s, remain_k, ns, nk;
    int ans[4], ac;

    scanf("%d", &m);
    n = m;
    for (i = 0; i < n; i++) {
        scanf("%d %d", &b[i], &w[i]);
        order[i] = i;
    }
    if (n < NEED) {
        printf("0\n");
        return 0;
    }
    qsort(order, n, sizeof(int), cmp_order);
    for (i = 0; i < n; i++) {
        bb[i] = b[order[i]];
        ww[i] = w[order[i]];
    }
    memset(suffix, 0, sizeof(suffix));
    suffix[n][0][0] = 1ULL;
    for (i = n - 1; i >= 0; i--) {
        for (k = 0; k <= NEED; k++) {
            bits_copy(suffix[i][k], suffix[i + 1][k]);
        }
        for (k = NEED - 1; k >= 0; k--) {
            /* dp[i][k][s]：从 i 往后恰好选 k 件，总价能否为 s */
            /* bits_copy 已继承不选第 i 件的 dp[i+1][k] */
            /* 左移 w[i] 再或上，对应选第 i 件、总价加上售价 */
            /* k 从大到小，保证每件最多用一次 */
            bits_or_shift(suffix[i][k + 1], suffix[i + 1][k], ww[i]);
        }
    }
    if (bits_none(suffix[0][NEED])) {
        printf("0\n");
        return 0;
    }
    remain_s = CAP;
    while (remain_s >= 0 && !bits_test(suffix[0][NEED], remain_s)) {
        remain_s--;
    }
    remain_k = NEED;
    ac = 0;
    for (i = 0; i < n && remain_k > 0; i++) {
        ns = remain_s - ww[i];
        nk = remain_k - 1;
        /* 判断 dp[i+1][nk][ns] 是否可行：选完当前件后后缀还能不能凑齐 */
        /* 编号已升序，能选就选，得到字典序最小方案 */
        if (ns >= 0 && bits_test(suffix[i + 1][nk], ns)) {
            ans[ac++] = bb[i];
            remain_s = ns;
            remain_k = nk;
        }
    }
    for (i = 0; i < ac; i++) {
        if (i) {
            printf(" ");
        }
        printf("%d", ans[i]);
    }
    printf("\n");
    return 0;
}
```
