# 旅行商巡格子

## 解题思路

网格没有障碍，两点之间最短路就是曼哈顿距离。把所有非 $0$ 格子记下来，从中心出发把它们都走到，最后再走回中心，求总步数最小值。

1. 扫一遍格子，把非 $0$ 的坐标放进 $points$，共 $k$ 个点。起点是中心 $(s_x,s_y)=(\lfloor n/2\rfloor,\lfloor m/2\rfloor)$。
2. 用 DFS 枚举访问顺序：当前在 $(x,y)$，已经走过的距离是 $step$，用集合 $vis$ 记下已经访问过的必访点。
3. 若 $vis$ 里已经有 $k$ 个点，说明必访点都走完了，再补上走回中心的曼哈顿距离，用它更新答案 $ans$。
4. 若当前 $step$ 已经不小于目前的 $ans$，这条路不可能更优，直接返回。
5. 否则枚举还没访问的点 $p$，走进去并累加曼哈顿距离，递归结束后再从 $vis$ 里拿掉（回溯）。

## 复杂度分析

时间复杂度为 $O(k!)$，会枚举必访点的访问顺序，最优剪枝只能减掉一部分。

空间复杂度为 $O(k)$，用于递归栈和访问集合。

## 代码实现

### Python

```python
def main():
    n, m = map(int, input().split())
    arr = []
    for _ in range(n):
        arr.append(list(map(int, input().split())))

    # 收集所有非 0 格子，这些点必须走到
    points = []
    for i in range(n):
        for j in range(m):
            if arr[i][j] != 0:
                points.append((i, j))
    k = len(points)
    s_x = n // 2
    s_y = m // 2
    vis = set()
    ans = 10 ** 9

    def dfs(x, y, step):
        nonlocal ans
        # 必访点都走完了，补上回到中心的距离
        if len(vis) == k:
            ans = min(ans, step + abs(x - s_x) + abs(y - s_y))
            return
        # 当前路程已经不优于已知答案，剪枝
        if step >= ans:
            return
        for p in points:
            if p in vis:
                continue
            vis.add(p)
            dfs(p[0], p[1], step + abs(x - p[0]) + abs(y - p[1]))
            vis.remove(p)

    dfs(s_x, s_y, 0)
    print(ans)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;
import java.util.Set;

public class Main {
    static List<int[]> points = new ArrayList<int[]>();
    static Set<Integer> vis = new HashSet<Integer>();
    static int k, sX, sY, ans;

    static int id(int[] p) {
        return p[0] * 100 + p[1];
    }

    static void dfs(int x, int y, int step) {
        // 必访点都走完了，补上回到中心的距离
        if (vis.size() == k) {
            int tot = step + Math.abs(x - sX) + Math.abs(y - sY);
            if (tot < ans) {
                ans = tot;
            }
            return;
        }
        // 当前路程已经不优于已知答案，剪枝
        if (step >= ans) {
            return;
        }
        for (int[] p : points) {
            int pid = id(p);
            if (vis.contains(pid)) {
                continue;
            }
            vis.add(pid);
            dfs(p[0], p[1], step + Math.abs(x - p[0]) + Math.abs(y - p[1]));
            vis.remove(pid);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (arr[i][j] != 0) {
                    points.add(new int[] {i, j});
                }
            }
        }
        k = points.size();
        sX = n / 2;
        sY = m / 2;
        ans = 1000000000;
        dfs(sX, sY, 0);
        System.out.println(ans);
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <set>
#include <cstdlib>
using namespace std;

vector<pair<int, int> > points;
set<pair<int, int> > vis;
int k, s_x, s_y, ans;

void dfs(int x, int y, int step) {
    // 必访点都走完了，补上回到中心的距离
    if ((int)vis.size() == k) {
        int tot = step + abs(x - s_x) + abs(y - s_y);
        if (tot < ans) {
            ans = tot;
        }
        return;
    }
    // 当前路程已经不优于已知答案，剪枝
    if (step >= ans) {
        return;
    }
    for (int i = 0; i < k; i++) {
        pair<int, int> p = points[i];
        if (vis.count(p)) {
            continue;
        }
        vis.insert(p);
        dfs(p.first, p.second, step + abs(x - p.first) + abs(y - p.second));
        vis.erase(p);
    }
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int> > arr(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> arr[i][j];
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (arr[i][j] != 0) {
                points.push_back(make_pair(i, j));
            }
        }
    }

    s_x = n / 2;
    s_y = m / 2;
```

```javascript
const fs = require("fs");

function main() {
    const tokens = fs.readFileSync(0, "utf8").trim().split(/\s+/).map(Number);
    let p = 0;
    const n = tokens[p++];
    const m = tokens[p++];
    const arr = [];
    for (let i = 0; i < n; i++) {
        const row = [];
        for (let j = 0; j < m; j++) {
            row.push(tokens[p++]);
        }
        arr.push(row);
    }

    const points = [];
    for (let i = 0; i < n; i++) {
        for (let j = 0; j < m; j++) {
            if (arr[i][j] !== 0) {
                points.push([i, j]);
            }
        }
    }
    const k = points.length;
    const sX = Math.floor(n / 2);
    const sY = Math.floor(m / 2);
    const vis = new Set();
    let ans = 1e9;

    function key(a, b) {
        return a + "," + b;
    }

    function dfs(x, y, step) {
        // 必访点都走完了，补上回到中心的距离
        if (vis.size === k) {
            ans = Math.min(ans, step + Math.abs(x - sX) + Math.abs(y - sY));
            return;
        }
        // 当前路程已经不优于已知答案，剪枝
        if (step >= ans) {
            return;
        }
        for (let t = 0; t < points.length; t++) {
            const pt = points[t];
            const kid = key(pt[0], pt[1]);
            if (vis.has(kid)) {
                continue;
            }
            vis.add(kid);
            dfs(pt[0], pt[1], step + Math.abs(x - pt[0]) + Math.abs(y - pt[1]));
            vis.delete(kid);
        }
    }

    dfs(sX, sY, 0);
    console.log(String(ans));
}

main();
```

### C

```c
#include <stdio.h>
#include <stdlib.h>

int points[25][2];
int vis[25];
int k, s_x, s_y, ans;

void dfs(int x, int y, int step) {
    int i, tot, used;
    used = 0;
    for (i = 0; i < k; i++) {
        if (vis[i]) {
            used++;
        }
    }
    /* 必访点都走完了，补上回到中心的距离 */
    if (used == k) {
        tot = step + abs(x - s_x) + abs(y - s_y);
        if (tot < ans) {
            ans = tot;
        }
        return;
    }
    /* 当前路程已经不优于已知答案，剪枝 */
    if (step >= ans) {
        return;
    }
    for (i = 0; i < k; i++) {
        if (vis[i]) {
            continue;
        }
        vis[i] = 1;
        dfs(points[i][0], points[i][1], step + abs(x - points[i][0]) + abs(y - points[i][1]));
        vis[i] = 0;
    }
}

int main(void) {
    int n, m, i, j, v;
    scanf("%d %d", &n, &m);
    k = 0;
    for (i = 0; i < n; i++) {
        for (j = 0; j < m; j++) {
            scanf("%d", &v);
            if (v != 0) {
                points[k][0] = i;
                points[k][1] = j;
                k++;
            }
        }
    }
    s_x = n / 2;
    s_y = m / 2;
    ans = 1000000000;
    dfs(s_x, s_y, 0);
    printf("%d\n", ans);
    return 0;
}
```
