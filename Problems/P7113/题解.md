# 首尾删除的最小总代价

## 解题思路

每次只能删当前序列的左端或右端，剩下的一定还是原数组的一段连续区间。用 DFS 搜索。

设 $dfs(l,r)$ 表示把 $a[l..r]$ 全部删完的最小总代价。

1. 边界：若 $l>r$，区间已空，代价为 $0$。
2. 当前长度 $len=r-l+1$。只能先删左端或先删右端：
   - 删左端：代价为 $len\cdot a[l]+dfs(l+1,r)$
   - 删右端：代价为 $len\cdot a[r]+dfs(l,r-1)$
3. 取较小者：
   $$
   dfs(l,r)=\min\bigl(len\cdot a[l]+dfs(l+1,r),\ len\cdot a[r]+dfs(l,r-1)\bigr)
   $$
4. 同一区间会被重复搜到，用字典把 $(l,r)$ 的结果记下来。答案为 $dfs(0,n-1)$。
5. 代价可能超过 $32$ 位整数，需使用 $64$ 位整数。

## 复杂度分析

区间状态有 $O(n^2)$ 个，每个状态 $O(1)$ 转移。

加上记忆化后时间复杂度为 $O(n^2)$。

空间复杂度为 $O(n^2)$，用于字典和递归栈。

在 $n\le 1000$ 时记忆化可以通过；不记忆化的普通递归是指数级，只能通过前面的小数据。

## 代码实现

### Python

```python
import sys

sys.setrecursionlimit(1000000)


def main():
    n = int(input())
    a = list(map(int, input().split()))
    d = {}

    # dfs(l,r) 表示把区间 [l,r] 全部删完的最小代价
    def dfs(l, r):
        if l > r:
            return 0
        # 算过就直接用，避免重复递归
        if (l, r) in d:
            return d[(l, r)]
        length = r - l + 1
        # 删除左端
        left = length * a[l] + dfs(l + 1, r)
        # 删除右端
        right = length * a[r] + dfs(l, r - 1)
        d[(l, r)] = min(left, right)
        return d[(l, r)]

    print(dfs(0, n - 1))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.HashMap;
import java.util.Scanner;

public class Main {
    static int[] a;
    static HashMap<Long, Long> d = new HashMap<Long, Long>();

    static long key(int l, int r) {
        return ((long) l << 32) ^ r;
    }

    // dfs(l,r) 表示把区间 [l,r] 全部删完的最小代价
    static long dfs(int l, int r) {
        if (l > r) {
            return 0;
        }
        // 算过就直接用，避免重复递归
        long k = key(l, r);
        if (d.containsKey(k)) {
            return d.get(k);
        }
        int length = r - l + 1;
        // 删除左端
        long left = (long) length * a[l] + dfs(l + 1, r);
        // 删除右端
        long right = (long) length * a[r] + dfs(l, r - 1);
        long ans = left < right ? left : right;
        d.put(k, ans);
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        System.out.println(dfs(0, n - 1));
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <map>
#include <vector>
using namespace std;

vector<int> a;
map<pair<int, int>, long long> d;

// dfs(l,r) 表示把区间 [l,r] 全部删完的最小代价
long long dfs(int l, int r) {
    if (l > r) {
        return 0;
    }
    // 算过就直接用，避免重复递归
    pair<int, int> k = make_pair(l, r);
    if (d.count(k)) {
        return d[k];
    }
    int length = r - l + 1;
    // 删除左端
    long long leftv = 1LL * length * a[l] + dfs(l + 1, r);
    // 删除右端
    long long rightv = 1LL * length * a[r] + dfs(l, r - 1);
    d[k] = leftv < rightv ? leftv : rightv;
    return d[k];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    a.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    cout << dfs(0, n - 1) << "\n";
    return 0;
}
```

### C

```c
#include <stdio.h>

#define N 2005

int a[N];
long long d[N][N];
int vis[N][N];

// dfs(l,r) 表示把区间 [l,r] 全部删完的最小代价
long long dfs(int l, int r) {
    if (l > r) {
        return 0;
    }
    // 算过就直接用，避免重复递归
    if (vis[l][r]) {
        return d[l][r];
    }
    vis[l][r] = 1;
    int length = r - l + 1;
    // 删除左端
    long long leftv = (long long)length * a[l] + dfs(l + 1, r);
    // 删除右端
    long long rightv = (long long)length * a[r] + dfs(l, r - 1);
    d[l][r] = leftv < rightv ? leftv : rightv;
    return d[l][r];
}

int main(void) {
    int n, i;
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }
    printf("%lld\n", dfs(0, n - 1));
    return 0;
}
```
