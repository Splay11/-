# 加速窗口收益

一句话总结：用三段 Kadane 式 DP 记录「还没加速 / 正在加速 / 加速已经结束」，扫一遍取最大。

算法标签：动态规划、最大子段和、Kadane

算法难度：$5 / 10$

## 解题思路

最多把一段连续作业的收益乘 $2$，再在最终序列上求非空最大子段和。不开窗口时就是普通 Kadane；开了窗口时，最终选中的那段必然盖住加速区间。

1. 从左到右扫，维护三个值：
   - $d_0$：以当前项结尾，整段都没加速；
   - $d_1$：以当前项结尾，当前项处在加速窗口内；
   - $d_2$：以当前项结尾，加速窗口已经在前面结束，现在按原收益接上。
2. 转移：
   - $d_0=\max(v_p,\,d_0+v_p)$，普通 Kadane；
   - $d_1=\max(2v_p,\,d_0+2v_p,\,d_1+2v_p)$，从当前项新开窗口，或接在未加速段后面开始加倍，或继续加倍；
   - $d_2=\max(d_1+v_p,\,d_2+v_p)$，窗口刚结束或已经结束，只能加原值。
3. 每个位置更新全局答案。全是负数时加倍会更亏，$d_0$ 会落到最大的一项，这是对的。
4. 和可能到 $2\times 2\times 10^{5}\times 10^{9}$，必须用 $64$ 位整数。常见假解：只做 Kadane、把答案写成两倍 Kadane（漏掉「中间加倍、两边仍按原值延伸」）、用 `int` 溢出。

## 复杂度分析

- 时间复杂度：$O(m)$。每组只扫一遍。
- 空间复杂度：$O(m)$。存收益数组；DP 只需常数个变量。

所有组 $m$ 之和不超过 $200000$。评测时限 $3$ 秒、内存 $512$ MB。

## 代码实现

### Python

```python
NEG = -10**30


def max_gain(v):
    # 三段 DP：尚未加速 / 正在加速 / 加速已结束
    d0 = v[0]
    d1 = 2 * v[0]
    d2 = NEG
    ans = d0 if d0 > d1 else d1
    for i in range(1, len(v)):
        x = v[i]
        # 不加速：新开一段或接上一段
        nd0 = x if x > d0 + x else d0 + x
        t = 2 * x
        # 正在加速：只取当前项、从未加速段转入、或继续加速
        nd1 = t
        if d0 + t > nd1:
            nd1 = d0 + t
        if d1 + t > nd1:
            nd1 = d1 + t
        # 加速已结束：只能加原值
        nd2 = d1 + x
        if d2 + x > nd2:
            nd2 = d2 + x
        d0, d1, d2 = nd0, nd1, nd2
        if d0 > ans:
            ans = d0
        if d1 > ans:
            ans = d1
        if d2 > ans:
            ans = d2
    return ans


def solve_all(groups):
    return [max_gain(v) for v in groups]


def main():
    # 第一行组数；每行先 m 再跟 m 个收益
    k = int(input())
    groups = []
    for _ in range(k):
        parts = list(map(int, input().split()))
        m = parts[0]
        v = parts[1:]
        groups.append(v[:m])
    # 一行输出 k 个答案，空格隔开
    ans = solve_all(groups)
    print(" ".join(str(x) for x in ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    static final long NEG = -(1L << 60);

    // 三段 DP：尚未加速 / 正在加速 / 加速已结束
    static long maxGain(long[] v) {
        long d0 = v[0];
        long d1 = 2 * v[0];
        long d2 = NEG;
        long ans = d0 > d1 ? d0 : d1;
        for (int i = 1; i < v.length; i++) {
            long x = v[i];
            long t = 2 * x;
            // 不加速
            long nd0 = x > d0 + x ? x : d0 + x;
            // 正在加速
            long nd1 = t;
            if (d0 + t > nd1) {
                nd1 = d0 + t;
            }
            if (d1 + t > nd1) {
                nd1 = d1 + t;
            }
            // 加速已结束
            long nd2 = d1 + x;
            if (d2 + x > nd2) {
                nd2 = d2 + x;
            }
            d0 = nd0;
            d1 = nd1;
            d2 = nd2;
            if (d0 > ans) {
                ans = d0;
            }
            if (d1 > ans) {
                ans = d1;
            }
            if (d2 > ans) {
                ans = d2;
            }
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        // 单行最长约 2e5 个数，用 BufferedReader 读一整行
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int k = Integer.parseInt(br.readLine().trim());
        StringBuilder sb = new StringBuilder();
        for (int t = 0; t < k; t++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int m = Integer.parseInt(st.nextToken());
            long[] v = new long[m];
            for (int i = 0; i < m; i++) {
                v[i] = Long.parseLong(st.nextToken());
            }
            if (t > 0) {
                sb.append(' ');
            }
            sb.append(maxGain(v));
        }
        // 一行输出 k 个答案，空格隔开
        System.out.println(sb.toString());
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

const long long NEG = -(1LL << 60);

// 三段 DP：尚未加速 / 正在加速 / 加速已结束
long long maxGain(const vector<long long>& v) {
    long long d0 = v[0];
    long long d1 = 2 * v[0];
    long long d2 = NEG;
    long long ans = d0 > d1 ? d0 : d1;
    int m = (int)v.size();
    for (int i = 1; i < m; i++) {
        long long x = v[i];
        long long t = 2 * x;
        // 不加速：新开一段或接到旧段后面
        long long nd0 = x > d0 + x ? x : d0 + x;
        // 正在加速：新开窗口 / 转入 / 继续
        long long nd1 = t;
        if (d0 + t > nd1) {
            nd1 = d0 + t;
        }
        if (d1 + t > nd1) {
            nd1 = d1 + t;
        }
        long long nd2 = d1 + x;
        if (d2 + x > nd2) {
            nd2 = d2 + x;
        }
        d0 = nd0;
        d1 = nd1;
        d2 = nd2;
        if (d0 > ans) {
            ans = d0;
        }
        if (d1 > ans) {
            ans = d1;
        }
        if (d2 > ans) {
            ans = d2;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    // 第一行组数；每组先读 m 再读 m 个收益
    int k;
    cin >> k;
    vector<long long> ans;
    ans.reserve(k);
    for (int t = 0; t < k; t++) {
        int m;
        cin >> m;
        vector<long long> v(m);
        for (int i = 0; i < m; i++) {
            cin >> v[i];
        }
        ans.push_back(maxGain(v));
    }
    // 一行输出 k 个答案，空格隔开
    for (int i = 0; i < k; i++) {
        if (i) {
            cout << ' ';
        }
        cout << ans[i];
    }
    cout << "\n";
    return 0;
}
```
