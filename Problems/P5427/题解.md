# 质因数跳跃

一句话总结：把每个下标看成图上的点，用当前值的质因数作为左右跳步长，从 $0$ 做 BFS 判断能否到达末位。

算法标签：BFS、质因数分解、隐式图、可达性

算法难度：$5 / 10$

## 解题思路

本题本质是隐式图上的可达性。

1. 序列长度为 $m$，下标 $0,1,\dots,m-1$ 各是一个点。
2. 从点 $p$ 出发，把 $seq_p$ 分解成不同质因数 $d$，则可以走向 $p+d$ 与 $p-d$（须落在区间 $[0,m-1]$ 内）。值为 $1$ 时没有质因数，该点没有出边。
3. 问从 $0$ 能否走到 $m-1$。$m=1$ 时起点即终点，答案为 `true`。
4. 每个点最多访问一次。值域 $\le 10^3$，每个点的质因数个数很少，出边是常数级，用队列 BFS 即可。
5. 常见假解：只向右跳、把合数因子（如 $4,8,9$）也当步长、把 $1$ 当成可以跳、或 $m=1$ 且值为 $1$ 时误判不能到达。

## 复杂度分析

- 时间复杂度：$O(m \log A)$。每个下标进队至多一次，质因数分解对 $A \le 10^3$ 是对数级。
- 空间复杂度：$O(m)$，访问标记与队列。

## 代码实现

### Python

```python
def prime_factors(x):
    # 分解出全部不同质因数；1 没有质因数
    if x <= 1:
        return []
    factors = []
    if x % 2 == 0:
        factors.append(2)
        while x % 2 == 0:
            x //= 2
    d = 3
    while d * d <= x:
        if x % d == 0:
            factors.append(d)
            while x % d == 0:
                x //= d
        d += 2
    if x > 1:
        factors.append(x)
    return factors


def can_reach(seq):
    # 从下标 0 按 ±质因数跳，看能否到达最后一个下标
    m = len(seq)
    if m == 1:
        return True
    vis = [False] * m
    q = [0]
    vis[0] = True
    head = 0
    while head < len(q):
        p = q[head]
        head += 1
        for d in prime_factors(seq[p]):
            for nxt in (p + d, p - d):
                if 0 <= nxt < m and not vis[nxt]:
                    if nxt == m - 1:
                        return True
                    vis[nxt] = True
                    q.append(nxt)
    return False


def main():
    # 一行空格分隔的整个序列
    seq = list(map(int, input().split()))
    print("true" if can_reach(seq) else "false")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;
import java.util.Scanner;

public class Main {
    // 分解出全部不同质因数；1 没有质因数
    static List<Integer> primeFactors(int x) {
        List<Integer> factors = new ArrayList<>();
        if (x <= 1) {
            return factors;
        }
        if (x % 2 == 0) {
            factors.add(2);
            while (x % 2 == 0) {
                x /= 2;
            }
        }
        for (int d = 3; (long) d * d <= x; d += 2) {
            if (x % d == 0) {
                factors.add(d);
                while (x % d == 0) {
                    x /= d;
                }
            }
        }
        if (x > 1) {
            factors.add(x);
        }
        return factors;
    }

    // 从下标 0 做 BFS，边为 ±质因数且不越界
    static boolean canReach(int[] seq) {
        int m = seq.length;
        if (m == 1) {
            return true;
        }
        boolean[] vis = new boolean[m];
        Deque<Integer> q = new ArrayDeque<>();
        vis[0] = true;
        q.addLast(0);
        while (!q.isEmpty()) {
            int p = q.removeFirst();
            for (int d : primeFactors(seq[p])) {
                int[] cand = {p + d, p - d};
                for (int nxt : cand) {
                    if (nxt < 0 || nxt >= m || vis[nxt]) {
                        continue;
                    }
                    if (nxt == m - 1) {
                        return true;
                    }
                    vis[nxt] = true;
                    q.addLast(nxt);
                }
            }
        }
        return false;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] parts = sc.nextLine().trim().split("\\s+");
        int[] seq = new int[parts.length];
        for (int i = 0; i < parts.length; i++) {
            seq[i] = Integer.parseInt(parts[i]);
        }
        sc.close();
        System.out.println(canReach(seq) ? "true" : "false");
    }
}
```

### C++

```cpp
#include <iostream>
#include <queue>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

// 分解出全部不同质因数；1 没有质因数
vector<int> primeFactors(int x) {
    vector<int> factors;
    if (x <= 1) {
        return factors;
    }
    if (x % 2 == 0) {
        factors.push_back(2);
        while (x % 2 == 0) {
            x /= 2;
        }
    }
    for (int d = 3; 1LL * d * d <= x; d += 2) {
        if (x % d == 0) {
            factors.push_back(d);
            while (x % d == 0) {
                x /= d;
            }
        }
    }
    if (x > 1) {
        factors.push_back(x);
    }
    return factors;
}

// 从下标 0 做 BFS，边为 ±质因数且不越界
bool canReach(const vector<int>& seq) {
    int m = (int)seq.size();
    if (m == 1) {
        return true;
    }
    vector<char> vis(m, 0);
    queue<int> q;
    vis[0] = 1;
    q.push(0);
    while (!q.empty()) {
        int p = q.front();
        q.pop();
        vector<int> factors = primeFactors(seq[p]);
        for (int d : factors) {
            int cand[2] = {p + d, p - d};
            for (int nxt : cand) {
                if (nxt < 0 || nxt >= m || vis[nxt]) {
                    continue;
                }
                if (nxt == m - 1) {
                    return true;
                }
                vis[nxt] = 1;
                q.push(nxt);
            }
        }
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string line;
    getline(cin, line);
    stringstream ss(line);
    vector<int> seq;
    int x;
    while (ss >> x) {
        seq.push_back(x);
    }
    cout << (canReach(seq) ? "true" : "false") << '\n';
    return 0;
}
```
