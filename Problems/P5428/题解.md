# 作业限放接单

一句话总结：把「连续放掉不超过 $lim$」转成相邻两次接单的下标间隔不超过 $lim+1$，再对「刚好接 $t$ 项」做最小和 DP，用滑动窗口维护上一层最小值。

算法标签：动态规划、滑动窗口、序列选取

算法难度：$6 / 10$

## 解题思路

本题是带空隙约束的子序列选取。

1. 设队列长度为 $n$。空队列（输入首行单独一个 $0$）按题意直接输出 $0$。
2. 若 $t=0$，相当于一项都不接，整段都算放掉，当且仅当 $n \le lim$ 时答案为 $0$，否则为 $-1$。
3. 接 $t$ 项会留下 $n-t$ 项被放掉，这些放掉的项分到 $t+1$ 个空隙（队首、相邻接单之间、队尾），每个空隙长度不能大于 $lim$。因此若 $n-t > (t+1)\cdot lim$，直接判定无解。
4. 合法方案等价于：选出下标 $p_1 < p_2 < \dots < p_t$，满足 $p_1 \le lim$、$p_{j+1}-p_j \le lim+1$、$n-1-p_t \le lim$，并使 $arr_{p_1}+\cdots+arr_{p_t}$ 最小。
5. 令 $dp[j][i]$ 表示已经接了 $j$ 项、且最后一项下标为 $i$ 的最小耗时。转移为
   $$dp[j][i] = arr_i + \min_{i-lim-1 \le k \le i-1} dp[j-1][k].$$
   $i$ 从左到右扫时，合法的 $k$ 是一个滑动窗口，用单调队列维护最小值，把转移做到 $O(1)$。
6. 初值：只接 $1$ 项时，队首空隙为 $i$，故仅当 $i \le lim$ 时 $dp[1][i]=arr_i$。答案取所有满足 $n-1-i \le lim$ 的 $dp[t][i]$ 的最小值；若均为无穷则输出 $-1$。
7. 常见假解：全局挑 $t$ 个最小却不管空隙；接满 $t$ 项就停、不管队尾；把间隔写成 $lim$ 而不是 $lim+1$；$t=0$ 一律输出 $0$。

## 复杂度分析

- 时间复杂度：$O(nt)$。共 $t$ 层，每层每个下标进单调队列至多一次。
- 空间复杂度：$O(n)$，滚动保存上一层 $dp$。

## 代码实现

### Python

```python
def is_empty(arr):
    # 题面约定：空队列用一行 0 表示，直接输出 0
    return (not arr) or arr == [0]


def min_cost(arr, t, lim):
    # 在连续放掉不超过 lim 的前提下，刚好接 t 项的最小耗时；做不到返回 -1
    n = len(arr)
    if n == 0:
        return 0
    if t == 0:
        # 一项都不接，整段都算放掉，长度必须 <= lim
        return 0 if n <= lim else -1
    if t > n:
        return -1
    # t+1 个空隙各最多放掉 lim 项
    if n - t > (t + 1) * lim:
        return -1

    inf = 10**18
    # dp[i]：当前已经接了若干项、且最后一项是下标 i 的最小耗时
    dp = [inf] * n
    for i in range(n):
        if i <= lim:
            dp[i] = arr[i]

    # 再接第 2..t 项：相邻两次接单的下标差最多 lim+1
    for _ in range(2, t + 1):
        ndp = [inf] * n
        dq = []
        head = 0
        for i in range(n):
            prev = i - 1
            if prev >= 0 and dp[prev] < inf:
                while head < len(dq) and dp[dq[-1]] >= dp[prev]:
                    dq.pop()
                dq.append(prev)
            lo = i - lim - 1
            while head < len(dq) and dq[head] < lo:
                head += 1
            if head < len(dq):
                ndp[i] = dp[dq[head]] + arr[i]
        dp = ndp

    ans = inf
    for i in range(n):
        # 队尾放掉的项数也不能超过 lim
        if n - 1 - i <= lim and dp[i] < ans:
            ans = dp[i]
    return -1 if ans >= inf else ans


def main():
    arr = list(map(int, input().split()))
    if is_empty(arr):
        print(0)
        return
    t = int(input())
    lim = int(input())
    print(min_cost(arr, t, lim))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Main {
    static final long INF = 4_000_000_000_000_000_000L;

    // 空队列用一行单独的 0 表示
    static boolean isEmpty(int[] arr) {
        return arr.length == 0 || (arr.length == 1 && arr[0] == 0);
    }

    // 连续放掉不超过 lim，刚好接 t 项的最小耗时
    static long minCost(int[] arr, int t, int lim) {
        int n = arr.length;
        if (n == 0) {
            return 0;
        }
        if (t == 0) {
            return n <= lim ? 0 : -1;
        }
        if (t > n) {
            return -1;
        }
        if (n - t > (long) (t + 1) * lim) {
            return -1;
        }

        long[] dp = new long[n];
        for (int i = 0; i < n; i++) {
            dp[i] = (i <= lim) ? arr[i] : INF;
        }

        // 第 2..t 次接单：上一层下标落在 [i-lim-1, i-1] 内取最小
        for (int k = 2; k <= t; k++) {
            long[] ndp = new long[n];
            for (int i = 0; i < n; i++) {
                ndp[i] = INF;
            }
            Deque<Integer> dq = new ArrayDeque<>();
            for (int i = 0; i < n; i++) {
                int prev = i - 1;
                if (prev >= 0 && dp[prev] < INF) {
                    while (!dq.isEmpty() && dp[dq.peekLast()] >= dp[prev]) {
                        dq.pollLast();
                    }
                    dq.addLast(prev);
                }
                int lo = i - lim - 1;
                while (!dq.isEmpty() && dq.peekFirst() < lo) {
                    dq.pollFirst();
                }
                if (!dq.isEmpty()) {
                    ndp[i] = dp[dq.peekFirst()] + arr[i];
                }
            }
            dp = ndp;
        }

        long ans = INF;
        for (int i = 0; i < n; i++) {
            if (n - 1 - i <= lim && dp[i] < ans) {
                ans = dp[i];
            }
        }
        return ans >= INF ? -1 : ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String first = sc.nextLine().trim();
        if (first.isEmpty()) {
            System.out.println(0);
            sc.close();
            return;
        }
        String[] parts = first.split("\\s+");
        int[] arr = new int[parts.length];
        for (int i = 0; i < parts.length; i++) {
            arr[i] = Integer.parseInt(parts[i]);
        }
        if (isEmpty(arr)) {
            System.out.println(0);
            sc.close();
            return;
        }
        int t = Integer.parseInt(sc.nextLine().trim());
        int lim = Integer.parseInt(sc.nextLine().trim());
        sc.close();
        System.out.println(minCost(arr, t, lim));
    }
}
```

### C++

```cpp
#include <deque>
#include <iostream>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

const long long INF = (long long)4e18;

// 空队列用一行单独的 0 表示
bool isEmpty(const vector<int>& arr) {
    return arr.empty() || (arr.size() == 1 && arr[0] == 0);
}

// 连续放掉不超过 lim，刚好接 t 项的最小耗时
long long minCost(const vector<int>& arr, int t, int lim) {
    int n = (int)arr.size();
    if (n == 0) {
        return 0;
    }
    if (t == 0) {
        return n <= lim ? 0 : -1;
    }
    if (t > n) {
        return -1;
    }
    if (n - t > (t + 1) * 1LL * lim) {
        return -1;
    }

    vector<long long> dp(n, INF);
    for (int i = 0; i < n; i++) {
        if (i <= lim) {
            dp[i] = arr[i];
        }
    }

    // 第 2..t 次接单：上一层下标落在 [i-lim-1, i-1] 内取最小
    for (int k = 2; k <= t; k++) {
        vector<long long> ndp(n, INF);
        deque<int> dq;
        for (int i = 0; i < n; i++) {
            int prev = i - 1;
            if (prev >= 0 && dp[prev] < INF) {
                while (!dq.empty() && dp[dq.back()] >= dp[prev]) {
                    dq.pop_back();
                }
                dq.push_back(prev);
            }
            int lo = i - lim - 1;
            while (!dq.empty() && dq.front() < lo) {
                dq.pop_front();
            }
            if (!dq.empty()) {
                ndp[i] = dp[dq.front()] + arr[i];
            }
        }
        dp.swap(ndp);
    }

    long long ans = INF;
    for (int i = 0; i < n; i++) {
        if (n - 1 - i <= lim && dp[i] < ans) {
            ans = dp[i];
        }
    }
    return ans >= INF ? -1 : ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string line;
    getline(cin, line);
    stringstream ss(line);
    vector<int> arr;
    int x;
    while (ss >> x) {
        arr.push_back(x);
    }
    if (isEmpty(arr)) {
        cout << 0 << '\n';
        return 0;
    }
    int t, lim;
    cin >> t >> lim;
    cout << minCost(arr, t, lim) << '\n';
    return 0;
}
```
