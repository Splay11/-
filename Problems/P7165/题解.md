# 24点游戏

算法标签：回溯、搜索

算法难度：$6 / 10$

## 解题思路

四个 $1$ 到 $9$ 的数，用加减乘除和括号凑出 $24$。每张牌必须用一次，除法是实数除法，不能把数字拼接，也不能把减号当一元负号。

1. 把四个数放进列表。每次从里面取出两个数 $x,y$，算出 $x+y$、$x-y$、$x\times y$，以及 $y\ne 0$ 时的 $x/y$，把结果塞回去，对剩下的数继续搜。
2. 枚举有序对，减法和除法的两个方向都会覆盖到。
3. 只剩一个数时，若与 $24$ 的差小于 $10^{-6}$ 就成功。实数除法会有误差，不能直接用整数相等。
4. 常见假解：用整数除法（$8/(3-8/3)$ 这类会错）；没枚举括号结构；除以 $0$ 没跳过。

四个数的搜索状态很少，回溯足够。

## 复杂度分析

- 时间复杂度：状态数是常数级（四个数的全排列与表达式树）。
- 空间复杂度：$O(1)$ 递归深度最多 $4$。

## 代码实现

### Python

```python
def solve(cards):
    """四个数做 24 点：每次取出两个数做加减乘除，把结果放回去继续。
    除法是实数除法；除数接近 0 时跳过。最后只剩一个数且与 24 足够接近就算成功。
    """
    eps = 1e-6

    def dfs(a):
        if len(a) == 1:
            return abs(a[0] - 24.0) < eps
        n = len(a)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                rest = []
                for k in range(n):
                    if k != i and k != j:
                        rest.append(a[k])
                x = a[i]
                y = a[j]
                cands = [x + y, x - y, x * y]
                if abs(y) > eps:
                    cands.append(x / y)
                for v in cands:
                    if dfs(rest + [v]):
                        return True
        return False

    return dfs([float(x) for x in cards])


def main():
    cards = list(map(int, input().split()))
    print("true" if solve(cards) else "false")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;
import java.util.StringTokenizer;

public class Main {
    static final double EPS = 1e-6;

    static boolean dfs(ArrayList<Double> a) {
        if (a.size() == 1) {
            return Math.abs(a.get(0) - 24.0) < EPS;
        }
        int n = a.size();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                ArrayList<Double> rest = new ArrayList<Double>();
                for (int k = 0; k < n; k++) {
                    if (k != i && k != j) rest.add(a.get(k));
                }
                double x = a.get(i), y = a.get(j);
                double[] cands = new double[4];
                int m = 3;
                cands[0] = x + y; cands[1] = x - y; cands[2] = x * y;
                if (Math.abs(y) > EPS) { cands[3] = x / y; m = 4; }
                for (int t = 0; t < m; t++) {
                    ArrayList<Double> nxt = new ArrayList<Double>(rest);
                    nxt.add(cands[t]);
                    if (dfs(nxt)) return true;
                }
            }
        }
        return false;
    }

    static boolean solve(int[] cards) {
        ArrayList<Double> a = new ArrayList<Double>();
        for (int i = 0; i < 4; i++) a.add((double) cards[i]);
        return dfs(a);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int[] cards = new int[4];
        for (int i = 0; i < 4; i++) cards[i] = Integer.parseInt(st.nextToken());
        System.out.println(solve(cards) ? "true" : "false");
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

const double EPS = 1e-6;

bool dfs(vector<double> a) {
    if ((int)a.size() == 1) return fabs(a[0] - 24.0) < EPS;
    int n = (int)a.size();
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            vector<double> rest;
            for (int k = 0; k < n; k++) if (k != i && k != j) rest.push_back(a[k]);
            double x = a[i], y = a[j];
            double cands[4] = {x + y, x - y, x * y};
            int m = 3;
            if (fabs(y) > EPS) { cands[3] = x / y; m = 4; }
            for (int t = 0; t < m; t++) {
                vector<double> nxt = rest;
                nxt.push_back(cands[t]);
                if (dfs(nxt)) return true;
            }
        }
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    vector<int> cards(4);
    for (int i = 0; i < 4; i++) cin >> cards[i];
    vector<double> a;
    for (int i = 0; i < 4; i++) a.push_back(cards[i]);
    cout << (dfs(a) ? "true" : "false") << '\n';
    return 0;
}
```
