# 断面调度后的浊度

## 解题思路

每个断面的下界是覆盖它的回灌泵里最大的那个，上界是覆盖它的稀释闸里最小的那个，再按公式夹到 $[lo_i,hi_i]$ 里。区间修改次数很多，用扫描线把区间拆成端点事件，从左扫到右维护当前生效的上界、下界。

1. 稀释闸 $[L,R]$ 上界 $c$：在位置 $L$ 加入 $c$，在 $R+1$ 删除 $c$。扫到 $i$ 时，当前集合的最小值就是 $hi_i$；集合为空则没有上界。
2. 回灌泵 $[L,R]$ 下界 $d$：同样在 $L$ 加入、$R+1$ 删除。当前集合的最大值就是 $lo_i$；集合为空则 $lo_i=0$。
3. 然后 $e_i=\min(\max(v_i,lo_i),hi_i)$。下界比上界更大时，结果会被上界卡住，这是公式本身的效果。
4. 用可重集合（或带删除标记的堆）维护当前值，保证能正确处理重复的上界或下界。

## 复杂度分析

- 时间复杂度：$O((w+x+y)\log(x+y))$，每个端点对集合插删一次。
- 空间复杂度：$O(w+x+y)$。

```python
import heapq
from collections import defaultdict


class LazyHeap:
  def __init__(self, want_max):
    self.h = []
    self.cnt = defaultdict(int)
    self.want_max = want_max

  def add(self, x):
    # 大根堆用取负来模拟
    if self.want_max:
      x = -x
    heapq.heappush(self.h, x)
    self.cnt[x] += 1

  def remove(self, x):
    # 懒删除：只改计数，真正弹出放到取堆顶时
    if self.want_max:
      x = -x
    self.cnt[x] -= 1

  def top(self):
    while self.h and self.cnt[self.h[0]] == 0:
      heapq.heappop(self.h)
    if not self.h:
      return None
    x = self.h[0]
    if self.want_max:
      return -x
    return x


def solve(w, v, caps, floors):
  # 稀释闸 [L,R] 上界 c：L 加入，R+1 删除
  add_c = [[] for _ in range(w + 2)]
  del_c = [[] for _ in range(w + 2)]
  for L, R, c in caps:
    add_c[L].append(c)
    del_c[R + 1].append(c)
  # 回灌泵 [L,R] 下界 d：同样拆成端点事件
  add_d = [[] for _ in range(w + 2)]
  del_d = [[] for _ in range(w + 2)]
  for L, R, d in floors:
    add_d[L].append(d)
    del_d[R + 1].append(d)
  # 稀释取最小上界，回灌取最大下界
  hc = LazyHeap(False)
  hd = LazyHeap(True)
  inf = 10 ** 18
  e = [0] * w
  for i in range(1, w + 1):
    for x in del_c[i]:
      hc.remove(x)
    for x in add_c[i]:
      hc.add(x)
    for x in del_d[i]:
      hd.remove(x)
    for x in add_d[i]:
      hd.add(x)
    hi = hc.top()
    if hi is None:
      hi = inf
    lo = hd.top()
    if lo is None:
      lo = 0
    val = v[i - 1]
    if val < lo:
      val = lo
    if val > hi:
      val = hi
    e[i - 1] = val
  return e


w, x, y = map(int, input().split())
v = list(map(int, input().split()))
caps = []
for _ in range(x):
  L, R, c = map(int, input().split())
  caps.append((L, R, c))
floors = []
for _ in range(y):
  L, R, d = map(int, input().split())
  floors.append((L, R, d))
e = solve(w, v, caps, floors)
print(" ".join(str(z) for z in e))
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;
import java.util.TreeMap;

public class Main {
  static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
  static StringTokenizer st;

  static String next() throws IOException {
    while (st == null || !st.hasMoreTokens()) {
      st = new StringTokenizer(br.readLine());
    }
    return st.nextToken();
  }

  static int nextInt() throws IOException {
    return Integer.parseInt(next());
  }

  // 动态 int 列表，避免 ArrayList<Integer> 装箱
  static class IL {
    int[] a = new int[2];
    int n;
    void add(int x) {
      if (n == a.length) {
        int[] b = new int[n * 2];
        for (int i = 0; i < n; i++) b[i] = a[i];
        a = b;
      }
      a[n++] = x;
    }
  }

  static void add(TreeMap<Integer, Integer> mp, int x) {
    Integer c = mp.get(x);
    mp.put(x, c == null ? 1 : c + 1);
  }

  static void remove(TreeMap<Integer, Integer> mp, int x) {
    int c = mp.get(x);
    if (c == 1) mp.remove(x);
    else mp.put(x, c - 1);
  }

  static int[] solve(int w, int[] v, IL[] addC, IL[] delC, IL[] addD, IL[] delD) {
    // hc 维护当前稀释上界，取最小；hd 维护当前回灌下界，取最大
    TreeMap<Integer, Integer> hc = new TreeMap<Integer, Integer>();
    TreeMap<Integer, Integer> hd = new TreeMap<Integer, Integer>();
    int inf = 2000000000;
    int[] e = new int[w];
    for (int i = 1; i <= w; i++) {
      for (int j = 0; j < delC[i].n; j++) remove(hc, delC[i].a[j]);
      for (int j = 0; j < addC[i].n; j++) add(hc, addC[i].a[j]);
      for (int j = 0; j < delD[i].n; j++) remove(hd, delD[i].a[j]);
      for (int j = 0; j < addD[i].n; j++) add(hd, addD[i].a[j]);
      int hi = hc.isEmpty() ? inf : hc.firstKey();
      int lo = hd.isEmpty() ? 0 : hd.lastKey();
      int val = v[i - 1];
      if (val < lo) val = lo;
      if (val > hi) val = hi;
      e[i - 1] = val;
    }
    return e;
  }

  public static void main(String[] args) throws IOException {
    // 断面个数可达 200000，Scanner 会超时，改用缓冲读入
    int w = nextInt();
    int x = nextInt();
    int y = nextInt();
    int[] v = new int[w];
    for (int i = 0; i < w; i++) v[i] = nextInt();
    IL[] addC = new IL[w + 2];
    IL[] delC = new IL[w + 2];
    IL[] addD = new IL[w + 2];
    IL[] delD = new IL[w + 2];
    for (int i = 0; i <= w + 1; i++) {
      addC[i] = new IL();
      delC[i] = new IL();
      addD[i] = new IL();
      delD[i] = new IL();
    }
    // 稀释闸 [L,R] 上界 c：L 加入，R+1 删除
    for (int i = 0; i < x; i++) {
      int L = nextInt(), R = nextInt(), c = nextInt();
      addC[L].add(c);
      delC[R + 1].add(c);
    }
    // 回灌泵 [L,R] 下界 d：同样拆成端点事件
    for (int i = 0; i < y; i++) {
      int L = nextInt(), R = nextInt(), d = nextInt();
      addD[L].add(d);
      delD[R + 1].add(d);
    }
    int[] e = solve(w, v, addC, delC, addD, delD);
    StringBuilder sb = new StringBuilder();
    for (int i = 0; i < w; i++) {
      if (i > 0) sb.append(' ');
      sb.append(e[i]);
    }
    sb.append('\n');
    System.out.print(sb.toString());
  }
}
```

```cpp
vector<long long> solve(int w, vector<long long> v, vector<vector<long long> > caps,
                        vector<vector<long long> > floors) {
  // 稀释闸 [L,R] 上界 c：L 加入，R+1 删除
  vector<vector<long long> > add_c(w + 2), del_c(w + 2), add_d(w + 2), del_d(w + 2);
  for (int i = 0; i < (int)caps.size(); i++) {
    add_c[(int)caps[i][0]].push_back(caps[i][2]);
    del_c[(int)caps[i][1] + 1].push_back(caps[i][2]);
  }
  for (int i = 0; i < (int)floors.size(); i++) {
    add_d[(int)floors[i][0]].push_back(floors[i][2]);
    del_d[(int)floors[i][1] + 1].push_back(floors[i][2]);
  }
  // 可重集合维护当前上界最小值、下界最大值
  multiset<long long> hc, hd;
  const long long inf = 1000000000000000000LL;
  vector<long long> e(w);
  for (int i = 1; i <= w; i++) {
    for (int j = 0; j < (int)del_c[i].size(); j++) hc.erase(hc.find(del_c[i][j]));
    for (int j = 0; j < (int)add_c[i].size(); j++) hc.insert(add_c[i][j]);
    for (int j = 0; j < (int)del_d[i].size(); j++) hd.erase(hd.find(del_d[i][j]));
    for (int j = 0; j < (int)add_d[i].size(); j++) hd.insert(add_d[i][j]);
    long long hi = hc.empty() ? inf : *hc.begin();
    long long lo = hd.empty() ? 0 : *hd.rbegin();
    long long val = v[i - 1];
    if (val < lo) val = lo;
    if (val > hi) val = hi;
    e[i - 1] = val;
  }
  return e;
}
```

```javascript
function solve(w, v, caps, floors) {
  var addC = [], delC = [], addD = [], delD = [];
  for (var i = 0; i <= w + 1; i++) {
    addC[i] = []; delC[i] = []; addD[i] = []; delD[i] = [];
  }
  for (var i = 0; i < caps.length; i++) {
    addC[caps[i][0]].push(caps[i][2]);
    delC[caps[i][1] + 1].push(caps[i][2]);
  }
  for (var i = 0; i < floors.length; i++) {
    addD[floors[i][0]].push(floors[i][2]);
    delD[floors[i][1] + 1].push(floors[i][2]);
  }
  var hc = {};
  var hd = {};
  function add(mp, x) { mp[x] = (mp[x] || 0) + 1; }
  function rem(mp, x) {
    mp[x]--;
    if (mp[x] === 0) delete mp[x];
  }
  function mini(mp) {
    var ans = null;
    for (var k in mp) {
      var x = Number(k);
      if (ans === null || x < ans) ans = x;
    }
    return ans;
  }
  function maxi(mp) {
    var ans = null;
    for (var k in mp) {
      var x = Number(k);
      if (ans === null || x > ans) ans = x;
    }
    return ans;
  }
  var inf = 1e18;
  var e = [];
  for (var i = 1; i <= w; i++) {
    for (var j = 0; j < delC[i].length; j++) rem(hc, delC[i][j]);
    for (var j = 0; j < addC[i].length; j++) add(hc, addC[i][j]);
    for (var j = 0; j < delD[i].length; j++) rem(hd, delD[i][j]);
    for (var j = 0; j < addD[i].length; j++) add(hd, addD[i][j]);
    var hi = mini(hc); if (hi === null) hi = inf;
    var lo = maxi(hd); if (lo === null) lo = 0;
    var val = v[i - 1];
    if (val < lo) val = lo;
    if (val > hi) val = hi;
    e.push(val);
  }
  return e;
}
```

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
  long long *a;
  int n, cap;
} Vec;

void vec_init(Vec *v) {
  v->n = 0;
  v->cap = 4;
  v->a = (long long *)malloc(sizeof(long long) * v->cap);
}

void vec_push(Vec *v, long long x) {
  if (v->n == v->cap) {
    v->cap *= 2;
    v->a = (long long *)realloc(v->a, sizeof(long long) * v->cap);
  }
  v->a[v->n++] = x;
}

typedef struct {
  long long *a;
  int n, cap;
} Multi;

void multi_init(Multi *m) {
  m->n = 0;
  m->cap = 8;
  m->a = (long long *)malloc(sizeof(long long) * m->cap);
}

void multi_add(Multi *m, long long x) {
  if (m->n == m->cap) {
    m->cap *= 2;
    m->a = (long long *)realloc(m->a, sizeof(long long) * m->cap);
  }
  int i = m->n - 1;
  while (i >= 0 && m->a[i] > x) {
    m->a[i + 1] = m->a[i];
    i--;
  }
  m->a[i + 1] = x;
  m->n++;
}

void multi_del(Multi *m, long long x) {
  int i;
  for (i = 0; i < m->n; i++) {
    if (m->a[i] == x) {
      int j;
      for (j = i; j + 1 < m->n; j++) m->a[j] = m->a[j + 1];
      m->n--;
      return;
    }
  }
}

long long *solve(int w, long long *v, int x, long long cap[][3], int y, long long fl[][3]) {
  Vec *add_c = (Vec *)malloc(sizeof(Vec) * (w + 2));
  Vec *del_c = (Vec *)malloc(sizeof(Vec) * (w + 2));
  Vec *add_d = (Vec *)malloc(sizeof(Vec) * (w + 2));
  Vec *del_d = (Vec *)malloc(sizeof(Vec) * (w + 2));
  int i, j;
  for (i = 0; i <= w + 1; i++) {
    vec_init(&add_c[i]); vec_init(&del_c[i]);
    vec_init(&add_d[i]); vec_init(&del_d[i]);
  }
  for (i = 0; i < x; i++) {
    vec_push(&add_c[(int)cap[i][0]], cap[i][2]);
    vec_push(&del_c[(int)cap[i][1] + 1], cap[i][2]);
  }
  for (i = 0; i < y; i++) {
    vec_push(&add_d[(int)fl[i][0]], fl[i][2]);
    vec_push(&del_d[(int)fl[i][1] + 1], fl[i][2]);
  }
  Multi hc, hd;
  multi_init(&hc);
  multi_init(&hd);
  long long inf = 1000000000000000000LL;
  long long *e = (long long *)malloc(sizeof(long long) * w);
  for (i = 1; i <= w; i++) {
    for (j = 0; j < del_c[i].n; j++) multi_del(&hc, del_c[i].a[j]);
    for (j = 0; j < add_c[i].n; j++) multi_add(&hc, add_c[i].a[j]);
    for (j = 0; j < del_d[i].n; j++) multi_del(&hd, del_d[i].a[j]);
    for (j = 0; j < add_d[i].n; j++) multi_add(&hd, add_d[i].a[j]);
    long long hi = hc.n ? hc.a[0] : inf;
    long long lo = hd.n ? hd.a[hd.n - 1] : 0;
    long long val = v[i - 1];
    if (val < lo) val = lo;
    if (val > hi) val = hi;
    e[i - 1] = val;
  }
  for (i = 0; i <= w + 1; i++) {
    free(add_c[i].a); free(del_c[i].a); free(add_d[i].a); free(del_d[i].a);
  }
  free(add_c); free(del_c); free(add_d); free(del_d);
  free(hc.a); free(hd.a);
  return e;
}

int main(void) {
  int w, x, y, i;
  scanf("%d%d%d", &w, &x, &y);
  long long *v = (long long *)malloc(sizeof(long long) * w);
```
