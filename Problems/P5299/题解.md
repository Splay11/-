# 三级档位前缀翻盘

## 解题思路

串长 $L\le 13$，状态只有 $3^L$ 个。每次把某个前缀反转，要把 `A`、`B`、`C` 排成非降序。

1. 把串看成三进制整数，便于当数组下标。
2. 所有已经排好的串（先 `A` 再 `B` 再 `C`）距离为 $0$，作为多源起点。
3. 前缀反转做两次等于没做，所以从有序串向外 BFS，得到每个串的最少次数。
4. 对每个询问直接查表。

## 复杂度分析

- 时间复杂度：$O(3^L\cdot L+T\cdot L)$。
- 空间复杂度：$O(3^L)$。

## 代码实现

### Python

```python
from collections import deque


def encode(s):
  # A=0,B=1,C=2，左边是高位
  x = 0
  for ch in s:
    x = x * 3 + (ord(ch) - 65)
  return x


def rev_prefix(x, n, p):
  # 拆出各位，把前 p 个倒过来再拼回去
  digs = [0] * n
  t = x
  for i in range(n - 1, -1, -1):
    digs[i] = t % 3
    t //= 3
  i, j = 0, p - 1
  while i < j:
    digs[i], digs[j] = digs[j], digs[i]
    i += 1
    j -= 1
  y = 0
  for d in digs:
    y = y * 3 + d
  return y


def build_dist(n):
  tot = 3 ** n
  dist = [-1] * tot
  q = deque()
  # 所有「若干 A + 若干 B + 若干 C」距离为 0
  for na in range(n + 1):
    for nb in range(n - na + 1):
      nc = n - na - nb
      x = 0
      for _ in range(na):
        x = x * 3
      for _ in range(nb):
        x = x * 3 + 1
      for _ in range(nc):
        x = x * 3 + 2
      dist[x] = 0
      q.append(x)
  while q:
    x = q.popleft()
    d = dist[x]
    # 枚举长度至少为 2 的前缀
    for p in range(2, n + 1):
      y = rev_prefix(x, n, p)
      if dist[y] < 0:
        dist[y] = d + 1
        q.append(y)
  return dist


n, qn = map(int, input().split())
dist = build_dist(n)
out = []
for _ in range(qn):
  s = input().strip()
  out.append(str(dist[encode(s)]))
print("\n".join(out))
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
  static int n;
  static int[] dist;

  static int revPrefix(int x, int p) {
    int[] d = new int[n];
    int t = x;
    for (int i = n - 1; i >= 0; i--) {
      d[i] = t % 3;
      t /= 3;
    }
    int i = 0, j = p - 1;
    while (i < j) {
      int tmp = d[i];
      d[i] = d[j];
      d[j] = tmp;
      i++;
      j--;
    }
    int y = 0;
    for (int k = 0; k < n; k++) y = y * 3 + d[k];
    return y;
  }

  static void build() {
    int tot = 1;
    for (int i = 0; i < n; i++) tot *= 3;
    dist = new int[tot];
    Arrays.fill(dist, -1);
    ArrayDeque<Integer> q = new ArrayDeque<>();
    for (int na = 0; na <= n; na++) {
      for (int nb = 0; nb <= n - na; nb++) {
        int nc = n - na - nb;
        int x = 0;
        for (int i = 0; i < na; i++) x = x * 3;
        for (int i = 0; i < nb; i++) x = x * 3 + 1;
        for (int i = 0; i < nc; i++) x = x * 3 + 2;
        dist[x] = 0;
        q.add(x);
      }
    }
    while (!q.isEmpty()) {
      int x = q.poll();
      int d0 = dist[x];
      for (int p = 2; p <= n; p++) {
        int y = revPrefix(x, p);
        if (dist[y] < 0) {
          dist[y] = d0 + 1;
          q.add(y);
        }
      }
    }
  }

  static int encode(String s) {
    int x = 0;
    for (int i = 0; i < s.length(); i++) x = x * 3 + (s.charAt(i) - 'A');
    return x;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] nq = br.readLine().trim().split("\\s+");
    n = Integer.parseInt(nq[0]);
    int qn = Integer.parseInt(nq[1]);
    build();
    StringBuilder out = new StringBuilder();
    for (int i = 0; i < qn; i++) {
      String s = br.readLine().trim();
      out.append(dist[encode(s)]).append('\n');
    }
    System.out.print(out);
  }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> distv;

int rev_prefix(int x, int p) {
  int d[13];
  int t = x;
  for (int i = n - 1; i >= 0; i--) {
    d[i] = t % 3;
    t /= 3;
  }
  int i = 0, j = p - 1;
  while (i < j) {
    swap(d[i], d[j]);
    i++;
    j--;
  }
  int y = 0;
  for (int k = 0; k < n; k++) y = y * 3 + d[k];
  return y;
}

void build() {
  int tot = 1;
  for (int i = 0; i < n; i++) tot *= 3;
  distv.assign(tot, -1);
  queue<int> q;
  for (int na = 0; na <= n; na++) {
    for (int nb = 0; nb <= n - na; nb++) {
      int nc = n - na - nb;
      int x = 0;
      for (int i = 0; i < na; i++) x = x * 3 + 0;
      for (int i = 0; i < nb; i++) x = x * 3 + 1;
      for (int i = 0; i < nc; i++) x = x * 3 + 2;
      distv[x] = 0;
      q.push(x);
    }
  }
  while (!q.empty()) {
    int x = q.front();
    q.pop();
    int d0 = distv[x];
    for (int p = 2; p <= n; p++) {
      int y = rev_prefix(x, p);
      if (distv[y] < 0) {
        distv[y] = d0 + 1;
        q.push(y);
      }
    }
  }
}

int encode(const string& s) {
  int x = 0;
  for (char c : s) x = x * 3 + (c - 'A');
  return x;
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int qn;
  cin >> n >> qn;
  build();
  while (qn--) {
    string s;
    cin >> s;
    cout << distv[encode(s)] << "\n";
  }
  return 0;
}
```
