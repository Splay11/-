## 解题思路

本题要求把长度为 $L$ 的序列 $w$ 划成恰好 $k$ 段连续非空子段，使各段元素和的最大值（峰值负载 $peak$）最小。

1. 设 $pref[i]$ 为前 $i$ 层（下标 $0..i-1$）的计算量之和，则一段 $w_l..w_r$ 的和为 $pref[r+1]-pref[l]$。
2. 定义 $dp[i][j]$：前 $i+1$ 层（下标 $0..i$）划成恰好 $j$ 个计算节点时，各段和的最大值的最小值。
3. 转移：最后一段从 $p+1$ 到 $i$，则
   $$dp[i][j] = \min_{p} \max\bigl(dp[p][j-1],\ pref[i+1]-pref[p+1]\bigr)$$
   其中 $p$ 至少为 $j-2$（保证前 $j-1$ 段非空）。
4. 初始：$dp[i][1] = pref[i+1]$（整段给 $1$ 个节点）。答案为 $dp[L-1][k]$。

## 复杂度分析

- 时间复杂度：$O(L^2 k)$，其中 $L \le 128$，$k \le L$。
- 空间复杂度：$O(Lk)$。

## 代码实现

### Python

```python
def min_peak(w, k):
    """动态规划求最小峰值负载 peak。"""
    L = len(w)

    # pref[i] = w[0] + w[1] + ... + w[i-1]，pref[0]=0
    # 有了前缀和，任意连续段 w[l..r] 的和可 O(1) 求出
    pref = [0]
    for x in w:
        pref.append(pref[-1] + x)

    def seg(l, r):
        # 段 [l, r]（闭区间）的元素和，即该段的负载
        return pref[r + 1] - pref[l]

    INF = 10 ** 18
    # dp[i][j]：考虑前 i+1 层（下标 0..i），划成恰好 j 段时，各段和的最大值的最小值
    dp = [[INF] * (k + 1) for _ in range(L)]

    # 边界：只分 1 段时，整段 [0..i] 的和就是峰值
    for i in range(L):
        dp[i][1] = seg(0, i)

    # j 表示当前要划成几段；i 表示考虑到第 i 层
    for j in range(2, k + 1):
        # 至少要有 j 层才能划成 j 段（每段非空），所以 i 从 j-1 开始
        for i in range(j - 1, L):
            best = INF
            # p 是「倒数第二段」的结尾下标，最后一段为 [p+1, i]
            # p 至少 j-2：前面 j-1 段至少各占 1 层
            for p in range(j - 2, i):
                # 前 p 层划 j-1 段的最优峰值 dp[p][j-1]
                # 与最后一段 seg(p+1,i) 取 max，才是 j 段划分下的峰值
                # 枚举所有合法 p，取峰值的最小值
                best = min(best, max(dp[p][j - 1], seg(p + 1, i)))
            dp[i][j] = best

    # 全部 L 层划成 k 段的最小峰值
    return dp[L - 1][k]


# 读入：层数 L、各层计算量 w、节点数 k
L = int(input())
w = list(map(int, input().split()))
k = int(input())
print(min_peak(w, k))
```

### Java

```java
import java.util.Scanner;

public class Main {
    /**
     * 动态规划：dp[i][j] 表示前 i+1 层划成 j 段的最小峰值。
     */
    static long minPeak(int[] w, int k) {
        int L = w.length;

        // 前缀和，pref[i] = w[0]+...+w[i-1]
        long[] pref = new long[L + 1];
        for (int i = 0; i < L; i++) {
            pref[i + 1] = pref[i] + w[i];
        }

        long INF = (long) 1e18;
        long[][] dp = new long[L][k + 1];

        // 初始化 dp 为无穷大，再填 j=1 的边界
        for (int i = 0; i < L; i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = INF;
            }
            // 只分 1 段：峰值 = 前缀 [0..i] 的总和
            dp[i][1] = pref[i + 1];
        }

        // 枚举段数 j 与当前结尾层 i
        for (int j = 2; j <= k; j++) {
            for (int i = j - 1; i < L; i++) {
                long best = INF;
                // p：上一段结尾；最后一段 [p+1, i]
                for (int p = j - 2; p < i; p++) {
                    long lastSeg = pref[i + 1] - pref[p + 1];  // 最后一段负载
                    long peak = Math.max(dp[p][j - 1], lastSeg);  // 该划分方案的峰值
                    best = Math.min(best, peak);  // 取所有 p 中最小的峰值
                }
                dp[i][j] = best;
            }
        }
        return dp[L - 1][k];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int L = sc.nextInt();
        int[] w = new int[L];
        for (int i = 0; i < L; i++) {
            w[i] = sc.nextInt();
        }
        int k = sc.nextInt();
        System.out.println(minPeak(w, k));
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 动态规划求最小峰值 peak
long long minPeak(const vector<int>& w, int k) {
    int L = (int)w.size();

    // pref[i] = w[0..i-1] 之和，用于 O(1) 算区间段负载
    vector<long long> pref(L + 1, 0);
    for (int i = 0; i < L; i++) {
        pref[i + 1] = pref[i] + w[i];
    }

    const long long INF = (long long)1e18;
    // dp[i][j]：前 i+1 层划 j 段的最小峰值
    vector<vector<long long>> dp(L, vector<long long>(k + 1, INF));

    for (int i = 0; i < L; i++) {
        dp[i][1] = pref[i + 1];  // j=1 的边界
    }

    for (int j = 2; j <= k; j++) {
        for (int i = j - 1; i < L; i++) {
            long long best = INF;
            for (int p = j - 2; p < i; p++) {
                long long last = pref[i + 1] - pref[p + 1];  // 最后一段 [p+1,i]
                // 峰值 = max(前 p 层 j-1 段最优, 最后一段和)，再对所有 p 取 min
                best = min(best, max(dp[p][j - 1], last));
            }
            dp[i][j] = best;
        }
    }
    return dp[L - 1][k];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int L;
    cin >> L;
    vector<int> w(L);
    for (int i = 0; i < L; i++) {
        cin >> w[i];
    }
    int k;
    cin >> k;
    cout << minPeak(w, k) << '\n';
    return 0;
}
```
