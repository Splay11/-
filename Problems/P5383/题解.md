# 并箱放行

## 解题思路

两件货能免检，当且仅当 $(w_i+w_j)$ 是 $t$ 的倍数。只由余数决定：记 $r_i=w_i\bmod t$，则要 $r_i+r_j\equiv 0\pmod t$。

1. 用长度为 $t$ 的数组统计每个余数出现次数 $cnt[r]$。
2. 余数 $0$ 只能和余数 $0$ 配对，方案数为 $C(cnt[0],2)$。
3. 若 $t$ 为偶数，余数 $t/2$ 只能和自己配对，方案数为 $C(cnt[t/2],2)$。
4. 其余余数 $r$ 只能和 $t-r$ 配对，方案数为 $cnt[r]\times cnt[t-r]$。枚举时只走 $r<t-r$，避免算重。
5. $n$ 最大为 $100000$，答案可能到 $C(n,2)$，用 $64$ 位整数。

## 复杂度分析

时间复杂度为 $O(n+t)$。

空间复杂度为 $O(t)$。

## 代码实现

### Python

```python
def count_pairs(n, t, w):
    # 只关心重量对装载模数 t 的余数
    # w[i]+w[j] 是 t 的倍数，当且仅当 (w[i]%t + w[j]%t) % t == 0
    cnt = [0] * t
    for x in w:
        cnt[x % t] += 1
    ans = 0
    # 余数 0 的货物只能和同类配对
    ans += cnt[0] * (cnt[0] - 1) // 2
    # t 为偶数时，余数 t/2 也只能和同类配对
    if t % 2 == 0:
        half = t // 2
        ans += cnt[half] * (cnt[half] - 1) // 2
    # 其余余数 r 只能和 t-r 配对，每对余数类只计一次
    for r in range(1, (t + 1) // 2):
        ans += cnt[r] * cnt[t - r]
    return ans


def main():
    # 第一行件数与装载模数，第二行各件重量
    n, t = map(int, input().split())
    w = list(map(int, input().split()))
    print(count_pairs(n, t, w))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    // 统计无序对 (i,j) 使得 (w[i]+w[j]) % t == 0
    static long countPairs(int n, int t, long[] w) {
        long[] cnt = new long[t];
        for (int i = 0; i < n; i++) {
            cnt[(int) (w[i] % t)]++;  // 按余数分桶
        }
        long ans = 0;
        // 余数 0 只能和同类配对
        ans += cnt[0] * (cnt[0] - 1) / 2;
        // t 为偶数时，余数 t/2 也只能和同类配对
        if (t % 2 == 0) {
            long half = cnt[t / 2];
            ans += half * (half - 1) / 2;
        }
        // r 与 t-r 互补，每对余数类只乘一次，避免算重
        for (int r = 1; r < (t + 1) / 2; r++) {
            ans += cnt[r] * cnt[t - r];
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();  // 件数
        int t = sc.nextInt();  // 装载模数
        long[] w = new long[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextLong();  // 第 i 件重量
        }
        System.out.println(countPairs(n, t, w));
        sc.close();
    }
}
```

```c
#include <stdio.h>

long long count_pairs(int n, int t, long long *w) {
    /* 按余数分桶；两件能配对当且仅当余数之和是 t 的倍数 */
    long long *cnt = (long long *)calloc((size_t)t, sizeof(long long));
    long long ans = 0;
    int i, r;
    for (i = 0; i < n; i++) {
        cnt[w[i] % t]++;
    }
    /* 余数 0 只能和同类配对 */
    ans += cnt[0] * (cnt[0] - 1) / 2;
    /* t 为偶数时，余数 t/2 也只能和同类配对 */
    if (t % 2 == 0) {
        long long half = cnt[t / 2];
        ans += half * (half - 1) / 2;
    }
    /* r 与 t-r 互补，每对余数类只乘一次 */
    for (r = 1; r < (t + 1) / 2; r++) {
        ans += cnt[r] * cnt[t - r];
    }
    free(cnt);
    return ans;
}
```
