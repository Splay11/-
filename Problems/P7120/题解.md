# 找出3位偶数

## 解题思路

本题用到的是「计数（哈希表思想）+ 枚举」。要点在于：数组里同一个数字可能出现多次，它决定了某个数字在拼数时最多能用几次；而最终答案只要互不相同的整数，重复的数字不该产生重复答案。

1. 先统计每个数字 $0 \sim 9$ 在 $digits$ 中出现了几次，记在长度为 $10$ 的计数数组 $cnt$ 里。用计数而不是直接排序，是因为「能不能再取一个数字 $d$」只需要看 $cnt[d]$ 是否大于 $0$。
2. 按「百位 $a$、十位 $b$、个位 $c$」三层循环枚举候选整数，三个数位的取值范围分别是：
   - 百位 $a$ 从 $1$ 到 $9$：从 $1$ 开始就天然排除了前导零；
   - 十位 $b$ 从 $0$ 到 $9$；
   - 个位 $c$ 只取 $0,2,4,6,8$：个位是偶数才能保证整个数是偶数。
3. 枚举时用「先扣后用、用完再还」的方式检查可用次数：进入某层时把对应数字的次数减 $1$，若发现 $cnt[d] = 0$ 说明这个数字已经用完，这个分支直接跳过；出这一层时再把次数加回来，保证不影响其他分支。
4. 三个数位都取到合法数字后，$a \times 100 + b \times 10 + c$ 就是一个满足条件的整数，直接收集。
5. 因为百位、十位、个位都是从小到大枚举的，收集顺序天然就是递增顺序；又因为数位确定则整数唯一，同一个整数不可能被收集两次，所以既不需要排序，也不需要额外去重。
6. 输出时按题面要求分两种情形：设答案为 $k$ 个，第一行输出 $k$；若 $k > 0$，第二行输出这 $k$ 个数；若 $k = 0$，只输出一行 $0$，不打第二行。

关于规模：三层循环的总枚举量是 $9 \times 10 \times 5 = 450$ 次，与 $n$ 无关。因此即使不利用计数数组、直接暴力枚举三个不同下标（$n \le 100$ 时约 $10^6$ 种排列）也能通过，但「计数 + 枚举候选数位」的写法更短、更不容易在去重和多重集次数上出错。

## 复杂度分析

时间复杂度为 $O(n)$：统计 $cnt$ 需要扫一遍数组 $O(n)$，之后枚举候选数位是固定的 $450$ 次（常数），因此总量是 $O(n + 450) = O(n)$。

空间复杂度为 $O(k)$，其中 $k$ 是答案个数：除存放答案的数组外，只用了一个长度为 $10$ 的计数数组，是常数空间；$k$ 最多为 $450$。

## 代码实现

### Python

```python
class Solution:
    def threeDigitEvens(self, digits):
        # cnt[d] 表示数字 d 在数组里出现了几次
        cnt = [0] * 10
        for d in digits:
            cnt[d] += 1

        res = []
        # 百位从 1 开始枚举，天然排除了前导零
        for a in range(1, 10):
            if cnt[a] == 0:
                continue
            # 用掉一个 a（先扣掉，后面还要还回来）
            cnt[a] -= 1
            for b in range(10):
                if cnt[b] == 0:
                    continue
                cnt[b] -= 1
                # 个位只能是偶数，才能保证整个数是偶数
                for c in (0, 2, 4, 6, 8):
                    if cnt[c] == 0:
                        continue
                    # 三个数位都够用，组成一个合法的三位偶数
                    res.append(a * 100 + b * 10 + c)
                # 还回 b，继续试下一个十位
                cnt[b] += 1
            # 还回 a，继续试下一个百位
            cnt[a] += 1

        # 百位、十位、个位都是从小到大枚举的，所以 res 已经是递增顺序，
        # 且每个整数只被枚举一次（数位确定则整数唯一），天然互不相同
        return res


def main():
    # 第一行：数组长度 n
    n = int(input())
    # 第二行：n 个 0~9 的数字
    digits = list(map(int, input().split()))
    ans = Solution().threeDigitEvens(digits)

    # 第一行输出个数 k
    print(len(ans))
    # 只有 k>0 时才输出第二行；k=0 时题面要求只输出一行 0
    if ans:
        print(" ".join(str(x) for x in ans))


if __name__ == "__main__":
    main()```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static List<Integer> threeDigitEvens(int[] digits) {
        // cnt[d] 表示数字 d 在数组里出现了几次
        int[] cnt = new int[10];
        for (int d : digits) {
            cnt[d]++;
        }

        List<Integer> res = new ArrayList<>();
        // 百位从 1 开始枚举，天然排除了前导零
        for (int a = 1; a <= 9; a++) {
            if (cnt[a] == 0) continue;
            cnt[a]--; // 用掉一个 a
            for (int b = 0; b <= 9; b++) {
                if (cnt[b] == 0) continue;
                cnt[b]--; // 用掉一个 b
                // 个位只能是偶数，才能保证整个数是偶数
                for (int c = 0; c <= 8; c += 2) {
                    if (cnt[c] == 0) continue;
                    // 三个数位都够用，组成一个合法的三位偶数
                    res.add(a * 100 + b * 10 + c);
                }
                cnt[b]++; // 还回 b，继续试下一个十位
            }
            cnt[a]++; // 还回 a，继续试下一个百位
        }

        // 三个数位都是从小到大枚举的，所以 res 已经是递增顺序；
        // 数位确定则整数唯一，因此不会出现重复
        return res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 第一行：数组长度 n
        int n = sc.nextInt();
        // 第二行：n 个 0~9 的数字
        int[] digits = new int[n];
        for (int i = 0; i < n; i++) {
            digits[i] = sc.nextInt();
        }

        List<Integer> ans = threeDigitEvens(digits);

        // 第一行输出个数 k
        System.out.println(ans.size());
        // 只有 k>0 时才输出第二行；k=0 时题面要求只输出一行 0
        if (!ans.isEmpty()) {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < ans.size(); i++) {
                if (i > 0) sb.append(' ');
                sb.append(ans.get(i));
            }
            System.out.println(sb.toString());
        }
        sc.close();
    }
}```

```cpp
#include <iostream>
#include <vector>
using namespace std;

vector<int> threeDigitEvens(const vector<int>& digits) {
    // cnt[d] 表示数字 d 在数组里出现了几次
    int cnt[10] = {0};
    for (int d : digits) {
        cnt[d]++;
    }

    vector<int> res;
    // 百位从 1 开始枚举，天然排除了前导零
    for (int a = 1; a <= 9; a++) {
        if (cnt[a] == 0) continue;
        cnt[a]--;  // 用掉一个 a
        for (int b = 0; b <= 9; b++) {
            if (cnt[b] == 0) continue;
            cnt[b]--;  // 用掉一个 b
            // 个位只能是偶数，才能保证整个数是偶数
            for (int c = 0; c <= 8; c += 2) {
                if (cnt[c] == 0) continue;
                // 三个数位都够用，组成一个合法的三位偶数
                res.push_back(a * 100 + b * 10 + c);
            }
            cnt[b]++;  // 还回 b，继续试下一个十位
        }
        cnt[a]++;  // 还回 a，继续试下一个百位
    }

    // 三个数位都是从小到大枚举的，所以 res 已经是递增顺序；
    // 数位确定则整数唯一，因此不会出现重复
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    // 第一行：数组长度 n
    int n;
    cin >> n;
    // 第二行：n 个 0~9 的数字
    vector<int> digits(n);
    for (int i = 0; i < n; i++) {
        cin >> digits[i];
    }

    vector<int> ans = threeDigitEvens(digits);

    // 第一行输出个数 k
    cout << ans.size() << "\n";
    // 只有 k>0 时才输出第二行；k=0 时题面要求只输出一行 0
    if (!ans.empty()) {
        for (size_t i = 0; i < ans.size(); i++) {
            if (i) cout << " ";
            cout << ans[i];
        }
        cout << "\n";
    }
    return 0;
}```
