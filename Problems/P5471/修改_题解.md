# DFS暴力解法

这道题的正解是状态压缩动态规划，但是我们用朴素解法深度优先搜索也能在考试时拿到一定的分数。

要从仓站 $0$ 出发，把客户 $1\dots m-1$ 各走恰好一次，再回到仓站。到达站点 $p$ 的时刻必须落在 $[lo_p,hi_p]$ 里：早到就等到 $lo_p$，等候算进总时长；迟到则这条路作废。目标是回到仓站的时刻尽量小，没有合法走法就输出 $-1$。

1. 用深度优先搜索枚举还没走过的客户。当前停在 $u$、时刻为 $now$，对每个未访问的客户 $v$ 算出到达时刻 $t=now+dur[u][v]$，必要时等到 $lo_v$，超过 $hi_v$ 就剪掉。
2. 所有客户都走过之后，再加上 $dur[u][0]$ 回到仓站，同样卡 $[lo_0,hi_0]$，用合法时刻更新答案。
3. 客户有 $m-1$ 个，搜索量大约是 $(m-1)!$。样例和较小的 $m$ 可以算对；$m$ 接近 $15$ 时会超时，所以后面再用状压。

## DFS解法

```python
def min_tour(m, lo, hi, dur):
    # DFS 枚举客户访问顺序，ans 记下回到仓站的最早时刻
    inf = 10 ** 9
    ans = inf
    used = [False] * m

    def dfs(cnt, u, now):
        nonlocal ans
        # cnt 个客户已经送达，当前停在 u，到达时刻为 now
        if cnt == m - 1:
            t = now + dur[u][0]
            if t < lo[0]:
                t = lo[0]
            if t <= hi[0] and t < ans:
                ans = t
            return
        for v in range(1, m):
            if used[v]:
                continue
            t = now + dur[u][v]
            if t < lo[v]:
                t = lo[v]
            if t > hi[v]:
                continue
            used[v] = True
            dfs(cnt + 1, v, t)
            used[v] = False

    dfs(0, 0, 0)
    return -1 if ans >= inf else ans


def main():
    m = int(input())
    lo = []
    hi = []
    for _ in range(m):
        a, b = map(int, input().split())
        lo.append(a)
        hi.append(b)
    dur = []
    for _ in range(m):
        dur.append(list(map(int, input().split())))
    print(min_tour(m, lo, hi, dur))


if __name__ == "__main__":
    main()
```

# 正确解法

## 题目分析

客户至多 $14$ 个，可以用二进制 $mask$ 记下已经送达的集合。第 $v-1$ 位为 $1$ 表示客户 $v$ 已送达。仓站不进 $mask$。

1. 设 $dp[mask][u]$ 为：已送达集合恰好是 $mask$，当前停在站点 $u$ 时，最早的到达时刻。初始 $dp[0][0]=0$。
2. 从 $(mask,u)$ 枚举尚未送达的客户 $v$，到达时刻 $t=dp[mask][u]+dur[u][v]$。若 $t<lo_v$ 则改成 $lo_v$；若 $t>hi_v$ 则丢弃。合法时用 $t$ 更新 $dp[mask\cup\{v\}][v]$。
3. 全集 $full$ 时，对每个最后一站 $u$ 计算 $t=dp[full][u]+dur[u][0]$，再卡仓站窗口 $[lo_0,hi_0]$，取最小 $t$。没有合法走法输出 $-1$。

## 复杂度分析

- 时间复杂度：$O(2^{m-1}\cdot m^{2})$。每个子集、每个当前位置，枚举下一个客户。
- 空间复杂度：$O(2^{m-1}\cdot m)$。状态数组。

$m\le 15$ 时约 $3.6\times 10^{6}$ 次转移。评测时限 $1$ 秒、内存 $512$ MB。

## 代码实现

### Python

```python
def min_tour(m, lo, hi, dur):
    # 客户编号 1..m-1，仓站为 0。mask 的第 (v-1) 位表示客户 v 是否已送达
    k = m - 1
    full = (1 << k) - 1
    inf = 10 ** 9
    # dp[mask][u]：已送达集合为 mask、当前停在 u 时的最早到达时刻
    dp = [[inf] * m for _ in range(1 << k)]
    dp[0][0] = 0
    for mask in range(1 << k):
        for u in range(m):
            now = dp[mask][u]
            if now >= inf:
                continue
            for v in range(1, m):
                bit = 1 << (v - 1)
                if mask & bit:
                    continue
                t = now + dur[u][v]
                if t < lo[v]:
                    t = lo[v]
                if t > hi[v]:
                    continue
                nmask = mask | bit
                if t < dp[nmask][v]:
                    dp[nmask][v] = t
    ans = inf
    for u in range(1, m):
        if dp[full][u] >= inf:
            continue
        t = dp[full][u] + dur[u][0]
        if t < lo[0]:
            t = lo[0]
        if t > hi[0]:
            continue
        if t < ans:
            ans = t
    return -1 if ans >= inf else ans


def main():

    lo = []
    hi = []
    for _ in range(m):
        a, b = map(int, input().split())
        lo.append(a)
        hi.append(b)
    dur = []
    for _ in range(m):
        dur.append(list(map(int, input().split())))
```

```java
import java.util.Scanner;

public class Main {
    static int minTour(int m, int[] lo, int[] hi, int[][] dur) {
        int k = m - 1;
        int full = (1 << k) - 1;
        int inf = 1000000000;
        // dp[mask][u]：已送达集合为 mask、当前停在 u 时的最早到达时刻
        int[][] dp = new int[1 << k][m];
        for (int mask = 0; mask <= full; mask++) {
            for (int u = 0; u < m; u++) {
                dp[mask][u] = inf;
            }
        }
        dp[0][0] = 0;
        for (int mask = 0; mask <= full; mask++) {
            for (int u = 0; u < m; u++) {
                int now = dp[mask][u];
                if (now >= inf) {
                    continue;
                }
                for (int v = 1; v < m; v++) {
                    int bit = 1 << (v - 1);
                    if ((mask & bit) != 0) {
                        continue;
                    }
                    int t = now + dur[u][v];
                    if (t < lo[v]) {
                        t = lo[v];
                    }
                    if (t > hi[v]) {
                        continue;
                    }
                    int nmask = mask | bit;
                    if (t < dp[nmask][v]) {
                        dp[nmask][v] = t;
                    }
                }
            }
        }
        int ans = inf;
        for (int u = 1; u < m; u++) {
            if (dp[full][u] >= inf) {
                continue;
            }
            int t = dp[full][u] + dur[u][0];
            if (t < lo[0]) {
                t = lo[0];
            }
            if (t > hi[0]) {
                continue;
            }
            if (t < ans) {
                ans = t;
            }
        }
        return ans >= inf ? -1 : ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int[] lo = new int[m];
        int[] hi = new int[m];
        for (int p = 0; p < m; p++) {
            lo[p] = sc.nextInt();
            hi[p] = sc.nextInt();
        }
        int[][] dur = new int[m][m];
        for (int p = 0; p < m; p++) {
            for (int q = 0; q < m; q++) {
                dur[p][q] = sc.nextInt();
            }
        }
        System.out.println(minTour(m, lo, hi, dur));
        sc.close();
    }
}
```

```cpp
#include <iostream>

#include <algorithm>
using namespace std;

int minTour(int m, const vector<int>& lo, const vector<int>& hi, const vector<vector<int>>& dur) {
    int k = m - 1;
    int full = (1 << k) - 1;
    const int INF = 1000000000;
    // dp[mask][u]：已送达集合为 mask、当前停在 u 时的最早到达时刻
    vector<vector<int>> dp(1 << k, vector<int>(m, INF));
    dp[0][0] = 0;
    for (int mask = 0; mask <= full; mask++) {
        for (int u = 0; u < m; u++) {
            int now = dp[mask][u];
            if (now >= INF) {
                continue;
            }
            for (int v = 1; v < m; v++) {
                int bit = 1 << (v - 1);
                if (mask & bit) {
                    continue;
                }
                int t = now + dur[u][v];
                if (t < lo[v]) {
                    t = lo[v];
                }
                if (t > hi[v]) {
                    continue;
                }
                int nmask = mask | bit;
                if (t < dp[nmask][v]) {
                    dp[nmask][v] = t;
                }
            }
        }
    }
    int ans = INF;
    for (int u = 1; u < m; u++) {
        if (dp[full][u] >= INF) {
            continue;
        }
        int t = dp[full][u] + dur[u][0];
        if (t < lo[0]) {
            t = lo[0];
        }
        if (t > hi[0]) {
            continue;
        }
        ans = min(ans, t);
    }
    return ans >= INF ? -1 : ans;
}

        }
    }
    cout << minTour(m, lo, hi, dur) << "\n";
    return 0;
}
```
