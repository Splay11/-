# 技能链压测过关

## 解题思路

初始负载 $10^{e}$，用四类技能把负载打到小于 $1$。

关键规则（与样例一致）：
- 技能甲提供「下一次」增幅系数，且不能连续放甲；
- **只有技能乙的固定伤害会乘上增幅**；技能丙、丁按 $\lfloor r\cdot\alpha/q\rfloor$、$\lfloor\sqrt{r}\rfloor$ 结算，若接在甲后仅消耗增幅；
- 丙、丁各至多一次；乙的每个 $h_j$、甲的每个 $w_k$ 至多一次。

状态可用 DFS + 记忆化搜索：
- 已用增幅集合（$2^{c}$，$c\le 10$）
- 已用乙集合（$2^{d}$，$d\le 17$）
- 丙/丁是否用过
- 当前挂起的增幅系数（$1$ 表示无挂起）

同一状态只保留到达时的最小负载：更低负载更优，重复以更高负载到达可剪枝。

## 复杂度分析

- 时间复杂度：状态数约 $O(2^{c}\cdot 2^{d}\cdot 4\cdot c)$ 量级，实际访问远少于此；单询问在约束内可过。
- 空间复杂度：与访问状态数同阶。

## 代码实现

### Python

```python
import math
import sys


def solve(c, d, alpha, q, e, w, h):
    hp0 = 10**e
    memo = {}

    def dfs(hp, mw, mh, u3, u4, pend):
        if hp < 1:
            return True
        key = (mw, mh, u3, u4, pend)
        prev = memo.get(key)
        if prev is not None and prev <= hp:
            return False
        memo[key] = hp

        # 技能甲：挂起增幅
        if pend == 1:
            for i in range(c):
                if (mw >> i) & 1:
                    continue
                if dfs(hp, mw | (1 << i), mh, u3, u4, w[i]):
                    return True

        mult = pend
        # 技能乙：乘增幅
        for j in range(d):
            if (mh >> j) & 1:
                continue
            if dfs(hp - h[j] * mult, mw, mh | (1 << j), u3, u4, 1):
                return True
        # 技能丙：不乘增幅
        if not u3:
            dmg = hp * alpha // q
            if dfs(hp - dmg, mw, mh, True, u4, 1):
                return True
        # 技能丁：不乘增幅
        if not u4:
            dmg = int(math.isqrt(hp)) if hp >= 0 else 0
            if dfs(hp - dmg, mw, mh, u3, True, 1):
                return True
        return False

    return dfs(hp0, 0, 0, False, False, 1)


data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)
t = next(it)
out = []
for _ in range(t):
    c = next(it); d = next(it); alpha = next(it); q = next(it); e = next(it)
    w = [next(it) for _ in range(c)]
    h = [next(it) for _ in range(d)]
    out.append("Yes" if solve(c, d, alpha, q, e, w, h) else "No")
sys.stdout.write("\n".join(out) + "\n")
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static int c, d, alpha, q, e;
    static long[] w, h;
    static HashMap<String, Long> memo;

    static long isqrt(long x) {
        if (x <= 0) return 0;
        long r = (long) Math.sqrt(x);
        while (r > 0 && r > x / r) r--;
        while (r + 1 > 0 && (r + 1) <= x / (r + 1)) r++;
        return r;
    }

    static boolean dfs(long hp, int mw, int mh, int u3, int u4, long pend) {
        if (hp < 1) return true;
        String key = mw + "," + mh + "," + u3 + "," + u4 + "," + pend;
        Long prev = memo.get(key);
        if (prev != null && prev <= hp) return false;
        memo.put(key, hp);

        if (pend == 1) {
            for (int i = 0; i < c; i++) {
                if (((mw >> i) & 1) != 0) continue;
                if (dfs(hp, mw | (1 << i), mh, u3, u4, w[i])) return true;
            }
        }
        long mult = pend;
        for (int j = 0; j < d; j++) {
            if (((mh >> j) & 1) != 0) continue;
            long nhp = (mult != 0 && h[j] > (hp - 1) / mult) ? 0 : hp - h[j] * mult;
            if (dfs(nhp, mw, mh | (1 << j), u3, u4, 1)) return true;
        }
        if (u3 == 0) {
            long dmg = hp * (long) alpha / q;
            if (dfs(hp - dmg, mw, mh, 1, u4, 1)) return true;
        }
        if (u4 == 0) {
            if (dfs(hp - isqrt(hp), mw, mh, u3, 1, 1)) return true;
        }
        return false;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder raw = new StringBuilder();
        for (String line; (line = br.readLine()) != null; ) raw.append(line).append(' ');
        StringTokenizer st = new StringTokenizer(raw.toString());
        int t = Integer.parseInt(st.nextToken());
        StringBuilder sb = new StringBuilder();
        while (t-- > 0) {
            c = Integer.parseInt(st.nextToken());
            d = Integer.parseInt(st.nextToken());
            alpha = Integer.parseInt(st.nextToken());
            q = Integer.parseInt(st.nextToken());
            e = Integer.parseInt(st.nextToken());
            w = new long[c];
            h = new long[d];
            for (int i = 0; i < c; i++) w[i] = Long.parseLong(st.nextToken());
            for (int i = 0; i < d; i++) h[i] = Long.parseLong(st.nextToken());
            memo = new HashMap<>();
            long hp = 1;
            for (int i = 0; i < e; i++) hp *= 10;
            sb.append(dfs(hp, 0, 0, 0, 0, 1) ? "Yes" : "No").append('\n');
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <cmath>
#include <iostream>
#include <map>
#include <tuple>
#include <vector>
using namespace std;

int c, d, alpha, q, e;
vector<long long> w, h;
map<tuple<int, int, int, int, long long>, long long> memo;

long long isqrt_ll(long long x) {
    if (x <= 0) return 0;
    long long r = (long long)sqrt((long double)x);
    while (r > 0 && r > x / r) r--;
    while (r + 1 > 0 && r + 1 <= x / (r + 1)) r++;
    return r;
}

bool dfs(long long hp, int mw, int mh, int u3, int u4, long long pend) {
    if (hp < 1) return true;
    auto key = make_tuple(mw, mh, u3, u4, pend);
    auto it = memo.find(key);
    if (it != memo.end() && it->second <= hp) return false;
    memo[key] = hp;

    if (pend == 1) {
        for (int i = 0; i < c; i++) {
            if ((mw >> i) & 1) continue;
            if (dfs(hp, mw | (1 << i), mh, u3, u4, w[i])) return true;
        }
    }
    long long mult = pend;
    for (int j = 0; j < d; j++) {
        if ((mh >> j) & 1) continue;
        long long nhp = (mult != 0 && h[j] > (hp - 1) / mult) ? 0 : hp - h[j] * mult;
        if (dfs(nhp, mw, mh | (1 << j), u3, u4, 1)) return true;
    }
    if (!u3) {
        if (dfs(hp - hp * alpha / q, mw, mh, 1, u4, 1)) return true;
    }
    if (!u4) {
        if (dfs(hp - isqrt_ll(hp), mw, mh, u3, 1, 1)) return true;
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        cin >> c >> d >> alpha >> q >> e;
        w.assign(c, 0);
        h.assign(d, 0);
        for (int i = 0; i < c; i++) cin >> w[i];
        for (int i = 0; i < d; i++) cin >> h[i];
        memo.clear();
        long long hp = 1;
        for (int i = 0; i < e; i++) hp *= 10;
        cout << (dfs(hp, 0, 0, 0, 0, 1) ? "Yes" : "No") << '\n';
    }
    return 0;
}
```
