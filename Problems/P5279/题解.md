# 最长匀差脉冲

## 解题思路

一段匀差脉冲就是下标连续的一段，相邻差对模 $K$ 都相等。从左到右扫一遍，把相同变化量的极长段找出来即可。

1. 只有一个读数时，长度是 $1$，变化量记 $0$。
2. 从位置 $i$ 算出 $d=(x_{i+1}-x_i)\bmod K$，尽量向右延伸到差不再等于 $d$。
3. 用当前段长度更新答案；同样长时不更新，从而保留更靠左的起点。
4. 下一段从当前段的最后一个位置接着比，避免漏掉两段交界处。
5. 取模时差可能为负，C++ / Java 要先把余数调到 $[0,K)$。

## 复杂度分析

- 时间复杂度：$O(p)$，每个位置最多作为段首、段尾各处理常数次。
- 空间复杂度：$O(p)$。

## 代码实现

### Python

```python
def solve(n, m, a):
  # 默认：单点段长度为 1，变化量记 0，起点 1
  best_len, best_l, best_d = 1, 1, 0
  if n == 1:
    return best_len, best_l, best_d
  i = 0
  while i < n - 1:
    d = (a[i + 1] - a[i]) % m
    j = i
    while j + 1 < n and (a[j + 1] - a[j]) % m == d:
      j += 1
    cur = j - i + 1
    if cur > best_len:
      best_len, best_l, best_d = cur, i + 1, d
    i = j
  return best_len, best_l, best_d


n, m = map(int, input().split())
a = list(map(int, input().split()))
L, p, d = solve(n, m, a)
print(L, p, d)
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
  static long[] solve(int n, long m, long[] a) {
    long bestLen = 1, bestL = 1, bestD = 0;
    if (n == 1) return new long[] {bestLen, bestL, bestD};
    int i = 0;
    while (i < n - 1) {
      long d = (a[i + 1] - a[i]) % m;
      if (d < 0) d += m;
      int j = i;
      while (j + 1 < n) {
        long cur = (a[j + 1] - a[j]) % m;
        if (cur < 0) cur += m;
        if (cur != d) break;
        j++;
      }
      long curLen = j - i + 1;
      if (curLen > bestLen) {
        bestLen = curLen;
        bestL = i + 1;
        bestD = d;
      }
      i = j;
    }
    return new long[] {bestLen, bestL, bestD};
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    int n = Integer.parseInt(st.nextToken());
    long m = Long.parseLong(st.nextToken());
    st = new StringTokenizer(br.readLine());
    long[] a = new long[n];
    for (int i = 0; i < n; i++) a[i] = Long.parseLong(st.nextToken());
    long[] ans = solve(n, m, a);
    System.out.println(ans[0] + " " + ans[1] + " " + ans[2]);
  }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<long long> solve(int n, long long m, const vector<long long>& a) {
  long long best_len = 1, best_l = 1, best_d = 0;
  if (n == 1) {
    vector<long long> one(3);
    one[0] = 1;
    one[1] = 1;
    one[2] = 0;
    return one;
  }
  int i = 0;
  while (i < n - 1) {
    long long d = (a[i + 1] - a[i]) % m;
    if (d < 0) d += m;
    int j = i;
    while (j + 1 < n) {
      long long cur = (a[j + 1] - a[j]) % m;
      if (cur < 0) cur += m;
      if (cur != d) break;
      j++;
    }
    long long cur_len = j - i + 1;
    if (cur_len > best_len) {
      best_len = cur_len;
      best_l = i + 1;
      best_d = d;
    }
    i = j;
  }
  vector<long long> ans(3);
  ans[0] = best_len;
  ans[1] = best_l;
  ans[2] = best_d;
  return ans;
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  long long m;
  cin >> n >> m;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<long long> ans = solve(n, m, a);
  cout << ans[0] << " " << ans[1] << " " << ans[2] << "\n";
  return 0;
}
```
