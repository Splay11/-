## 解题思路

本题分两步：

1. **作文批改（空格规范化）**：去掉首尾空格；扫描中间，把连续多个空格压成一个空格（单词间最多一个空格）。
2. **最长无重复子串**：在规范化后的串上，用**滑动窗口**维护当前窗口内「按忽略大小写」不重复的字符集合；右指针扩展，若新字符与窗口内已有字符冲突（大小写视为相同），则左指针跳到冲突字符上次出现位置的下一位。

整体与 [LeetCode 3. 无重复字符的最长子串](https://leetcode.cn/problems/longest-substring-without-repeating-characters/) 相同，只是多了预处理；重复判重键为 `tolower(ch)`。

## 复杂度分析

- 设 $n = \texttt{story.length}$，规范化与滑动窗口各扫一遍，**时间复杂度** $O(n)$。
- 哈希表记录每个字符（小写键）最近下标，**空间复杂度** $O(\min(n, \Sigma))$，$\Sigma$ 为字符集大小（本题 ASCII）。

## 代码实现

### Python

```python
class Solution:
    def lengthOfLongestSubstring(self, story: str) -> int:
        # 批改：去首尾空格，中间连续空格压成一个
        s = story.strip(" ")
        if not s:
            return 0
        parts = []
        i = 0
        while i < len(s):
            if s[i] == " ":
                if parts:
                    parts.append(" ")
                while i < len(s) and s[i] == " ":
                    i += 1
            else:
                parts.append(s[i])
                i += 1
        t = "".join(parts).rstrip(" ")

        # 滑动窗口：键为小写字符，记录最近下标
        last = {}
        left = 0
        best = 0
        for right, ch in enumerate(t):
            key = ch.lower()
            if key in last and last[key] >= left:
                left = last[key] + 1
            last[key] = right
            best = max(best, right - left + 1)
        return best
```

### Java

```java
public class Solution {
    private String normalize(String story) {
        int a = 0, b = story.length();
        while (a < b && story.charAt(a) == ' ') a++;
        while (b > a && story.charAt(b - 1) == ' ') b--;
        String s = story.substring(a, b);
        if (s.isEmpty()) return "";
        StringBuilder out = new StringBuilder();
        for (int i = 0; i < s.length();) {
            if (s.charAt(i) == ' ') {
                if (out.length() > 0) out.append(' ');
                while (i < s.length() && s.charAt(i) == ' ') i++;
            } else {
                out.append(s.charAt(i++));
            }
        }
        while (out.length() > 0 && out.charAt(out.length() - 1) == ' ')
            out.setLength(out.length() - 1);
        return out.toString();
    }

    public int lengthOfLongestSubstring(String story) {
        String t = normalize(story);
        if (t.isEmpty()) return 0;
        java.util.Map<Character, Integer> last = new java.util.HashMap<>();
        int left = 0, best = 0;
        for (int right = 0; right < t.length(); right++) {
            char key = Character.toLowerCase(t.charAt(right));
            Integer prev = last.get(key);
            // 窗口内已有同字符（忽略大小写），收缩左边界
            if (prev != null && prev >= left) left = prev + 1;
            last.put(key, right);
            best = Math.max(best, right - left + 1);
        }
        return best;
    }
}
```

### C++

```cpp
#include <algorithm>
#include <cctype>
#include <string>
#include <unordered_map>

using namespace std;

class Solution {
public:
    string normalize(const string& story) {
        size_t a = 0, b = story.size();
        while (a < b && story[a] == ' ') a++;
        while (b > a && story[b - 1] == ' ') b--;
        string s = story.substr(a, b - a);
        if (s.empty()) return "";
        string out;
        for (size_t i = 0; i < s.size();) {
            if (s[i] == ' ') {
                if (!out.empty()) out.push_back(' ');
                while (i < s.size() && s[i] == ' ') i++;
            } else {
                out.push_back(s[i++]);
            }
        }
        while (!out.empty() && out.back() == ' ') out.pop_back();
        return out;
    }

    int lengthOfLongestSubstring(string story) {
        string t = normalize(story);
        if (t.empty()) return 0;
        unordered_map<char, int> last;  // 小写键 -> 该字符最近下标
        int left = 0, best = 0;
        for (int right = 0; right < (int)t.size(); right++) {
            char key = (char)tolower((unsigned char)t[right]);
            unordered_map<char, int>::iterator it = last.find(key);
            // 窗口内已有同字符（忽略大小写），左边界跳到冲突位置之后
            if (it != last.end() && it->second >= left)
                left = it->second + 1;
            last[key] = right;
            best = max(best, right - left + 1);
        }
        return best;
    }
};
```
