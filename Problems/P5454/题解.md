# 雷达等距计数

一句话总结：枚举每个探测原点，用平方距离把其余点分桶，同一距离有 $c$ 个点则贡献 $c\cdot(c-1)$ 组有序对。

算法标签：哈希表、枚举、计数、计算几何

算法难度：$3 / 10$

## 解题思路

有序三元 $(p,q,r)$ 要求 $p$ 到 $q$、$p$ 到 $r$ 的欧氏距离相等，且三点互异。关键是固定原点 $p$ 之后，看其余点按距离如何扎堆。

1. 枚举原点 $p$。对每个 $p$，扫描其余 $m-1$ 个点，按到 $p$ 的距离分桶。比较平方距离 $dx^2+dy^2$，既避开开方的浮点误差，又与原距离相等性完全一致。
2. 某一距离上有 $c$ 个点时，从中挑两个做成有序对 $(q,r)$，方案数是 $P(c,2)=c\cdot(c-1)$，不是组合数 $C(c,2)$。题面明确 $(p,q,r)$ 与 $(p,r,q)$ 各计一次。
3. 把所有原点、所有桶的贡献加起来即为答案。点数互异，不会出现距离为 $0$ 的桶。
4. 坐标到 $\pm 10^5$，平方距离最大约 $8\times 10^{10}$，必须用 $64$ 位整数存 $dx^2+dy^2$；答案最大约 $m(m-1)(m-2)$，同样要用 $64$ 位。常见假解：用 `int` 乘平方溢出、用浮点比距离、只计组合不计排列、三重循环 $O(m^3)$。

## 复杂度分析

- 时间复杂度：$O(m^2)$。每个原点扫一遍其余点，哈希表均摊 $O(1)$ 插入。
- 空间复杂度：$O(m)$。单个原点的分桶表最多 $m-1$ 个键。

$m\le 2\times 10^3$，约 $4\times 10^6$ 次运算。评测时限 $3$ 秒、内存 $512$ MB，给 Python 哈希表和 Java 虚拟机留余量；$O(m^3)$ 仍会超时。

## 代码实现

### Python

```python
def count_equidistant(loc):
    # 统计等心探测组：有序三元 (p, q, r) 且 p 到 q、r 的欧氏距离相等
    m = len(loc)
    ans = 0
    # 每个点都当一次探测原点
    for p in range(m):
        ux, uy = loc[p]
        buckets = {}
        for q in range(m):
            if p == q:
                continue
            dx = loc[q][0] - ux
            dy = loc[q][1] - uy
            # 用平方距离比较，避免开方误差；Python 整数不会溢出
            d2 = dx * dx + dy * dy
            buckets[d2] = buckets.get(d2, 0) + 1
        # 同一距离 c 个点，有序对有 c*(c-1) 种
        for c in buckets.values():
            ans += c * (c - 1)
    return ans


def main():
    m = int(input())
    loc = []
    for _ in range(m):
        u, v = map(int, input().split())
        loc.append((u, v))
    print(count_equidistant(loc))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    // 统计等心探测组：有序三元 (p, q, r) 且 p 到 q、r 的欧氏距离相等
    static long countEquidistant(int[][] loc) {
        int m = loc.length;
        long ans = 0;
        // 复用同一张表，避免每轮 new HashMap 把内存顶满
        Map<Long, Integer> buckets = new HashMap<>();
        // 每个点都当一次探测原点
        for (int p = 0; p < m; p++) {
            buckets.clear();
            int ux = loc[p][0];
            int uy = loc[p][1];
            for (int q = 0; q < m; q++) {
                if (p == q) {
                    continue;
                }
                // 先转 long 再乘，否则 dx*dx 会在 int 里溢出
                long dx = (long) loc[q][0] - ux;
                long dy = (long) loc[q][1] - uy;
                long d2 = dx * dx + dy * dy;
                buckets.put(d2, buckets.getOrDefault(d2, 0) + 1);
            }
            // 同一距离 c 个点，有序对有 c*(c-1) 种
            for (int c : buckets.values()) {
                ans += (long) c * (c - 1);
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int[][] loc = new int[m][2];
        // 读入 m 座雷达站的坐标
        for (int i = 0; i < m; i++) {
            loc[i][0] = sc.nextInt();
            loc[i][1] = sc.nextInt();
        }
        sc.close();
        System.out.println(countEquidistant(loc));
    }
}
```

### C++

```cpp
#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;

// 统计等心探测组：有序三元 (p, q, r) 且 p 到 q、r 的欧氏距离相等
long long countEquidistant(const vector<pair<int, int>>& loc) {
    int m = (int)loc.size();
    long long ans = 0;
    // 每个点都当一次探测原点
    for (int p = 0; p < m; p++) {
        unordered_map<long long, int> buckets;
        buckets.reserve(m);
        int ux = loc[p].first;
        int uy = loc[p].second;
        for (int q = 0; q < m; q++) {
            if (p == q) {
                continue;
            }
            // 先转 long long 再乘，否则 dx*dx 会在 int 里溢出
            long long dx = (long long)loc[q].first - ux;
            long long dy = (long long)loc[q].second - uy;
            long long d2 = dx * dx + dy * dy;
            buckets[d2]++;
        }
        // 同一距离 c 个点，有序对有 c*(c-1) 种
        for (auto& kv : buckets) {
            long long c = kv.second;
            ans += c * (c - 1);
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int m;
    cin >> m;
    vector<pair<int, int>> loc(m);
    // 读入 m 座雷达站的坐标
    for (int i = 0; i < m; i++) {
        cin >> loc[i].first >> loc[i].second;
    }
    cout << countEquidistant(loc) << '\n';
    return 0;
}
```
