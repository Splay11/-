# 模拟暴力解法

这道题的正解是**线段树维护可合并区间状态**，但我们用朴素模拟合并也能在考试时拿到一定的分数。每次区间查询从 $l$ 到 $r$ 依次合并，单点更新直接改数组；$B$、$Q$ 较小时完全够用，大数据会超时。

## 模拟解法

```python
import math


def merge(a, b):
    """两块状态合并，a 在左、b 在右。"""
    mx_a, wd_a, sv_a = a
    mx_b, wd_b, sv_b = b
    mx = max(mx_a, mx_b)
    ea = math.exp(mx_a - mx)
    eb = math.exp(mx_b - mx)
    return mx, wd_a * ea + wd_b * eb, sv_a * ea + sv_b * eb


def query_range(blocks, l, r):
    """
    朴素区间查询：从第 l 块到第 r 块（1-indexed）顺序合并。
    时间 O(r-l+1)，大数据会 TLE。
    """
    mx, wd, sv = blocks[l - 1]  # 先取左端点块
    for i in range(l + 1, r + 1):
        mx, wd, sv = merge((mx, wd, sv), blocks[i - 1])
    return sv / wd  # attn = 加权和 / 权重和


B, Q = map(int, input().split())
blocks = []
for _ in range(B):
    mx, wd, sv = map(float, input().split())
    blocks.append((mx, wd, sv))

for _ in range(Q):
    parts = input().split()
    if parts[0] == "1":
        i = int(parts[1])
        blocks[i - 1] = float(parts[2]), float(parts[3]), float(parts[4])
    else:
        l, r = int(parts[1]), int(parts[2])
        print(f"{query_range(blocks, l, r):.6f}")
```

```cpp
struct State {
    double mx, wd, sv;
};

State mergeState(const State& a, const State& b) {
    State c;
    c.mx = max(a.mx, b.mx);
    double ea = exp(a.mx - c.mx);
    double eb = exp(b.mx - c.mx);
    c.wd = a.wd * ea + b.wd * eb;
    c.sv = a.sv * ea + b.sv * eb;
    return c;
}

// 朴素：O(区间长度) 逐块合并
double queryRange(const vector<State>& blocks, int l, int r) {
    State cur = blocks[l - 1];
    for (int i = l + 1; i <= r; i++)
        cur = mergeState(cur, blocks[i - 1]);
    return cur.sv / cur.wd;
}
```

# 正确解法

## 题目分析

合并运算满足结合律，可用线段树：每个结点存区间合并后的 $(mx, wd, sv)$，`merge` 按题面公式合并左右子区间；单点更新改叶子后向上 pushup，区间查询 $O(\log B)$。

## 复杂度分析

- 时间复杂度：$O(Q \log B)$。
- 空间复杂度：$O(B)$。

## 代码实现

### Python

```python
import math


def merge(a, b):
    """
    合并左块 a、右块 b，返回 (mx, wd, sv)。
    mx = max(mx_a, mx_b)，wd/sv 按题面指数重缩放后相加。
    """
    mx_a, wd_a, sv_a = a
    mx_b, wd_b, sv_b = b
    mx = max(mx_a, mx_b)
    ea = math.exp(mx_a - mx)
    eb = math.exp(mx_b - mx)
    return mx, wd_a * ea + wd_b * eb, sv_a * ea + sv_b * eb


class SegTree:
    def __init__(self, blocks):
        self.n = len(blocks)
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.tree = [(0.0, 0.0, 0.0)] * (2 * self.size)
        # 叶子：每个块对应一个结点
        for i, st in enumerate(blocks):
            self.tree[self.size + i] = st
        # 内部结点：左右儿子 merge
        for i in range(self.size - 1, 0, -1):
            self.tree[i] = merge(self.tree[i * 2], self.tree[i * 2 + 1])

    def update(self, pos, val):
        i = self.size + pos
        self.tree[i] = val
        i //= 2
        while i:
            self.tree[i] = merge(self.tree[i * 2], self.tree[i * 2 + 1])
            i //= 2

    def query(self, l, r):
        l += self.size
        r += self.size
        left = right = None
        while l <= r:
            if l & 1:
                left = self.tree[l] if left is None else merge(left, self.tree[l])
                l += 1
            if not (r & 1):
                right = self.tree[r] if right is None else merge(self.tree[r], right)
                r -= 1
            l //= 2
            r //= 2
        if left is None:
            return right
        if right is None:
            return left
        return merge(left, right)


B, Q = map(int, input().split())
blocks = []
for _ in range(B):
    mx, wd, sv = map(float, input().split())
    blocks.append((mx, wd, sv))
seg = SegTree(blocks)

for _ in range(Q):
    parts = input().split()
    if parts[0] == "1":
        i = int(parts[1]) - 1
        st = float(parts[2]), float(parts[3]), float(parts[4])
        seg.update(i, st)
    else:
        l = int(parts[1]) - 1
        r = int(parts[2]) - 1
        mx, wd, sv = seg.query(l, r)
        print(f"{sv / wd:.6f}")
```

### Java

```java
import java.util.Scanner;

public class Main {
    /** 块状态三元组 */
    static class State {
        double mx, wd, sv;
        State(double mx, double wd, double sv) {
            this.mx = mx; this.wd = wd; this.sv = sv;
        }
    }

    /** 左块 a、右块 b 按题面公式 merge */
    static State merge(State a, State b) {
        double mx = Math.max(a.mx, b.mx);
        double ea = Math.exp(a.mx - mx);  // 数值稳定：先减再 exp
        double eb = Math.exp(b.mx - mx);
        return new State(mx, a.wd * ea + b.wd * eb, a.sv * ea + b.sv * eb);
    }

    static class SegTree {
        int n, size;
        State[] tree;

        SegTree(State[] blocks) {
            n = blocks.length;
            size = 1;
            while (size < n) size <<= 1;  // 扩展到 2 的幂
            tree = new State[2 * size];
            for (int i = 0; i < 2 * size; i++) tree[i] = new State(0, 0, 0);
            for (int i = 0; i < n; i++) tree[size + i] = blocks[i];  // 叶子
            for (int i = size - 1; i >= 1; i--)  // 建树
                tree[i] = merge(tree[i * 2], tree[i * 2 + 1]);
        }

        void update(int pos, State val) {
            int i = size + pos;
            tree[i] = val;
            for (i /= 2; i >= 1; i /= 2)  // 向上 pushup
                tree[i] = merge(tree[i * 2], tree[i * 2 + 1]);
        }

        State query(int l, int r) {
            l += size; r += size;
            boolean hasL = false, hasR = false;
            State left = null, right = null;
            while (l <= r) {
                if ((l & 1) == 1) {  // 右端点，归左侧
                    left = hasL ? merge(left, tree[l]) : tree[l];
                    hasL = true; l++;
                }
                if ((r & 1) == 0) {  // 左端点，归右侧
                    right = hasR ? merge(tree[r], right) : tree[r];
                    hasR = true; r--;
                }
                l /= 2; r /= 2;
            }
            if (!hasL) return right;
            if (!hasR) return left;
            return merge(left, right);  // 左 + 右
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int B = sc.nextInt();
        int Q = sc.nextInt();
        State[] blocks = new State[B];
        for (int i = 0; i < B; i++)
            blocks[i] = new State(sc.nextDouble(), sc.nextDouble(), sc.nextDouble());
        SegTree seg = new SegTree(blocks);
        for (int t = 0; t < Q; t++) {
            int op = sc.nextInt();
            if (op == 1) {
                int i = sc.nextInt() - 1;
                seg.update(i, new State(sc.nextDouble(), sc.nextDouble(), sc.nextDouble()));
            } else {
                int l = sc.nextInt() - 1;
                int r = sc.nextInt() - 1;
                State res = seg.query(l, r);
                System.out.printf("%.6f%n", res.sv / res.wd);
            }
        }
        sc.close();
    }
}
```

```cpp
#include <cmath>
#include <iostream>
#include <vector>
using namespace std;

struct State { double mx, wd, sv; };

State mergeState(const State& a, const State& b) {
    State c;
    c.mx = max(a.mx, b.mx);
    double ea = exp(a.mx - c.mx);
    double eb = exp(b.mx - c.mx);
    c.wd = a.wd * ea + b.wd * eb;
    c.sv = a.sv * ea + b.sv * eb;
    return c;
}

struct SegTree {
    int n, size;
    vector<State> tree;

    SegTree(const vector<State>& blocks) : n((int)blocks.size()) {
        size = 1;
        while (size < n) size <<= 1;
        tree.assign(2 * size, {0, 0, 0});
        for (int i = 0; i < n; i++) tree[size + i] = blocks[i];
        for (int i = size - 1; i >= 1; i--)
            tree[i] = mergeState(tree[i * 2], tree[i * 2 + 1]);
    }

    void update(int pos, const State& val) {
        int i = size + pos;
        tree[i] = val;
        for (i /= 2; i >= 1; i /= 2)
            tree[i] = mergeState(tree[i * 2], tree[i * 2 + 1]);
    }

    State query(int l, int r) {
        l += size; r += size;
        bool hasL = false, hasR = false;
        State left{}, right{};
        while (l <= r) {
            if (l & 1) {
                left = hasL ? mergeState(left, tree[l]) : tree[l];
                hasL = true; l++;
            }
            if (!(r & 1)) {
                right = hasR ? mergeState(tree[r], right) : tree[r];
                hasR = true; r--;
            }
            l /= 2; r /= 2;
        }
        if (!hasL) return right;
        if (!hasR) return left;
        return mergeState(left, right);
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int B, Q;
    cin >> B >> Q;
    vector<State> blocks(B);
    for (int i = 0; i < B; i++) cin >> blocks[i].mx >> blocks[i].wd >> blocks[i].sv;
    SegTree seg(blocks);
    cout.setf(ios::fixed);
    cout.precision(6);
    for (int t = 0; t < Q; t++) {
        int op;
        cin >> op;
        if (op == 1) {
            int i;
            State st;
            cin >> i >> st.mx >> st.wd >> st.sv;
            seg.update(i - 1, st);  // 单点更新
        } else {
            int l, r;
            cin >> l >> r;
            State res = seg.query(l - 1, r - 1);
            cout << res.sv / res.wd << '\n';  // attn
        }
    }
    return 0;
}
```
