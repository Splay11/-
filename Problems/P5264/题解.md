## 解题思路

本题维护 $B$ 个块，每块状态为 $(mx, wd, sv)$，合并运算满足结合律。需要支持单点更新与区间合并查询。

1. 定义合并函数 `merge(A, B)`，按题面公式计算 $mx_C, wd_C, sv_C$；查询结果为 $sv / wd$。
2. 合并满足结合律，可用**线段树**维护区间合并结果：叶子存单块状态，内部结点存子区间合并状态。
3. 单点更新：修改叶子并向上 pushup；区间查询：标准线段树区间合并，时间 $O(\log B)$。

## 复杂度分析

- 时间复杂度：每次更新或查询 $O(\log B)$，总 $O(Q \log B)$。
- 空间复杂度：$O(B)$。

## 代码实现

### Python

```python
import math


def merge(a, b):
    """
    按题面公式合并两块状态 A、B（A 在左，B 在右）。
    返回合并后的 (mx, wd, sv)。
    """
    mx_a, wd_a, sv_a = a
    mx_b, wd_b, sv_b = b
    mx = max(mx_a, mx_b)  # 合并后的 logit 最大值
    # 以 mx 为基准做数值稳定的指数重缩放
    ea = math.exp(mx_a - mx)
    eb = math.exp(mx_b - mx)
    wd = wd_a * ea + wd_b * eb   # 重缩放后的权重和
    sv = sv_a * ea + sv_b * eb   # 重缩放后的加权和
    return mx, wd, sv


class SegTree:
    """线段树：每个结点存区间合并后的 (mx, wd, sv)。"""

    def __init__(self, blocks):
        self.n = len(blocks)
        # size 取 >= n 的最小 2 的幂，方便用数组模拟完全二叉树
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.tree = [(0.0, 0.0, 0.0)] * (2 * self.size)

        # 叶子 [size, size+n) 存各块初始状态
        for i, st in enumerate(blocks):
            self.tree[self.size + i] = st
        # 自底向上：父结点 = merge(左儿子, 右儿子)
        for i in range(self.size - 1, 0, -1):
            self.tree[i] = merge(self.tree[i * 2], self.tree[i * 2 + 1])

    def update(self, pos, val):
        """单点更新：第 pos 块（0-indexed）改为 val，并向上 pushup。"""
        i = self.size + pos
        self.tree[i] = val
        i //= 2
        while i:
            self.tree[i] = merge(self.tree[i * 2], self.tree[i * 2 + 1])
            i //= 2

    def query(self, l, r):
        """区间查询 [l, r]（0-indexed 闭区间）合并后的状态。"""
        l += self.size
        r += self.size
        left = right = None  # 分别维护「靠左」「靠右」两部分的合并结果
        while l <= r:
            if l & 1:  # l 是右儿子，整块纳入左侧答案
                left = self.tree[l] if left is None else merge(left, self.tree[l])
                l += 1
            if not (r & 1):  # r 是左儿子，整块纳入右侧答案
                right = self.tree[r] if right is None else merge(self.tree[r], right)
                r -= 1
            l //= 2
            r //= 2
        # 最终：merge(左侧, 右侧)，保证顺序 A 在左、B 在右
        if left is None:
            return right
        if right is None:
            return left
        return merge(left, right)


B, Q = map(int, input().split())
blocks = []
for _ in range(B):
    mx, wd, sv = map(float, input().split())
    blocks.append((mx, wd, sv))
seg = SegTree(blocks)

for _ in range(Q):
    parts = input().split()
    if parts[0] == "1":
        # 操作 1：单点更新第 i 块
        i = int(parts[1]) - 1
        st = float(parts[2]), float(parts[3]), float(parts[4])
        seg.update(i, st)
    else:
        # 操作 2：查询 [l,r] 合并后的 attn = sv/wd
        l = int(parts[1]) - 1
        r = int(parts[2]) - 1
        mx, wd, sv = seg.query(l, r)
        print(f"{sv / wd:.6f}")
```

### Java

```java
import java.util.Scanner;

public class Main {
    /** 块状态：mx 最大值，wd 权重和，sv 加权和 */
    static class State {
        double mx, wd, sv;

        State(double mx, double wd, double sv) {
            this.mx = mx;
            this.wd = wd;
            this.sv = sv;
        }
    }

    /** 左块 a、右块 b 按题面公式合并 */
    static State merge(State a, State b) {
        double mx = Math.max(a.mx, b.mx);
        double ea = Math.exp(a.mx - mx);
        double eb = Math.exp(b.mx - mx);
        return new State(mx, a.wd * ea + b.wd * eb, a.sv * ea + b.sv * eb);
    }

    static class SegTree {
        int n, size;
        State[] tree;

        SegTree(State[] blocks) {
            n = blocks.length;
            size = 1;
            while (size < n) size <<= 1;
            tree = new State[2 * size];
            for (int i = 0; i < 2 * size; i++) tree[i] = new State(0, 0, 0);
            // 叶子赋值
            for (int i = 0; i < n; i++) tree[size + i] = blocks[i];
            // 建树
            for (int i = size - 1; i >= 1; i--)
                tree[i] = merge(tree[i * 2], tree[i * 2 + 1]);
        }

        void update(int pos, State val) {
            int i = size + pos;
            tree[i] = val;
            // 向上更新祖先
            for (i /= 2; i >= 1; i /= 2)
                tree[i] = merge(tree[i * 2], tree[i * 2 + 1]);
        }

        State query(int l, int r) {
            l += size;
            r += size;
            boolean hasL = false, hasR = false;
            State left = null, right = null;
            while (l <= r) {
                if ((l & 1) == 1) {
                    left = hasL ? merge(left, tree[l]) : tree[l];
                    hasL = true;
                    l++;
                }
                if ((r & 1) == 0) {
                    right = hasR ? merge(tree[r], right) : tree[r];
                    hasR = true;
                    r--;
                }
                l /= 2;
                r /= 2;
            }
            if (!hasL) return right;
            if (!hasR) return left;
            return merge(left, right);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int B = sc.nextInt();
        int Q = sc.nextInt();
        State[] blocks = new State[B];
        for (int i = 0; i < B; i++)
            blocks[i] = new State(sc.nextDouble(), sc.nextDouble(), sc.nextDouble());
        SegTree seg = new SegTree(blocks);
        for (int t = 0; t < Q; t++) {
            int op = sc.nextInt();
            if (op == 1) {
                int i = sc.nextInt() - 1;
                seg.update(i, new State(sc.nextDouble(), sc.nextDouble(), sc.nextDouble()));
            } else {
                int l = sc.nextInt() - 1;
                int r = sc.nextInt() - 1;
                State res = seg.query(l, r);
                System.out.printf("%.6f%n", res.sv / res.wd);
            }
        }
        sc.close();
    }
}
```

### C++

```cpp
#include <cmath>
#include <iostream>
#include <vector>
using namespace std;

struct State {
    double mx, wd, sv;  // logit 最大值、权重和、加权和
};

// 合并相邻两块（a 在左，b 在右）
State mergeState(const State& a, const State& b) {
    State c;
    c.mx = max(a.mx, b.mx);
    double ea = exp(a.mx - c.mx);  // exp(mx_a - mx_c)
    double eb = exp(b.mx - c.mx);
    c.wd = a.wd * ea + b.wd * eb;
    c.sv = a.sv * ea + b.sv * eb;
    return c;
}

struct SegTree {
    int n, size;
    vector<State> tree;

    SegTree(const vector<State>& blocks) : n((int)blocks.size()) {
        size = 1;
        while (size < n) size <<= 1;
        tree.assign(2 * size, {0, 0, 0});
        for (int i = 0; i < n; i++) tree[size + i] = blocks[i];
        for (int i = size - 1; i >= 1; i--)
            tree[i] = mergeState(tree[i * 2], tree[i * 2 + 1]);
    }

    void update(int pos, const State& val) {
        int i = size + pos;
        tree[i] = val;
        for (i /= 2; i >= 1; i /= 2)
            tree[i] = mergeState(tree[i * 2], tree[i * 2 + 1]);
    }

    State query(int l, int r) {
        l += size;
        r += size;
        bool hasL = false, hasR = false;
        State left{}, right{};
        while (l <= r) {
            if (l & 1) {
                left = hasL ? mergeState(left, tree[l]) : tree[l];
                hasL = true;
                l++;
            }
            if (!(r & 1)) {
                right = hasR ? mergeState(tree[r], right) : tree[r];
                hasR = true;
                r--;
            }
            l /= 2;
            r /= 2;
        }
        if (!hasL) return right;
        if (!hasR) return left;
        return mergeState(left, right);
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int B, Q;
    cin >> B >> Q;
    vector<State> blocks(B);
    for (int i = 0; i < B; i++) cin >> blocks[i].mx >> blocks[i].wd >> blocks[i].sv;
    SegTree seg(blocks);
    cout.setf(ios::fixed);
    cout.precision(6);
    for (int t = 0; t < Q; t++) {
        int op;
        cin >> op;
        if (op == 1) {
            int i;
            State st;
            cin >> i >> st.mx >> st.wd >> st.sv;
            seg.update(i - 1, st);
        } else {
            int l, r;
            cin >> l >> r;
            State res = seg.query(l - 1, r - 1);
            cout << res.sv / res.wd << '\n';  // 输出 attn
        }
    }
    return 0;
}
```
