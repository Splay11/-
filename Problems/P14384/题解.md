## 解题思路

本题是**固定规则**下的离散事件模拟，不是求最优调度。

1. 按时间维护事件队列：**车辆到达**（优先级低）与**充电桩释放**（优先级高）；同一时刻先处理释放再处理到达。
2. 用**小根堆**维护 $N$ 个桩的最早空闲时刻；用**队列**维护按到达先后（FCFS）等待的车辆。
3. 在时刻 $t$ 尝试分配：若队首车辆满足 $t-AT \le WT$，则占用最早可用的桩，开始充电并在 $t+CT$ 时刻产生释放事件；否则队首失败出队。
4. 统计失败出队次数即为答案。

**区分度**：

- 忽略「同刻先释放后到达」会错样例 2 类边界。
- 等待队列不按到达顺序会错多车同到场景。
- 把 $WT$ 当成充电时长或把失败车辆仍占桩都会错。

## 复杂度分析

- 时间：$O((M + K)\log M)$，$K$ 为成功充电次数（每辆车至多一次成功分配），事件堆与桩堆均为对数级。
- 空间：$O(M + N)$。

## 代码实现

### Python

```python
import heapq
from typing import List


class Solution:
    def countFailedCharging(self, n: int, cars: List[List[int]]) -> int:
        events = []
        for i, (at, _ct, _wt) in enumerate(cars):
            heapq.heappush(events, (at, 1, "A", i))  # 到达事件，同刻优先级 1

        piles = [0] * n
        heapq.heapify(piles)  # 桩空闲时刻小根堆
        waiting = []
        failed = 0

        while events:
            t = events[0][0]
            batch = []
            while events and events[0][0] == t:
                batch.append(heapq.heappop(events))
            batch.sort(key=lambda x: x[1])  # 先处理释放(0)再到达(1)

            for _time, _pri, kind, data in batch:
                if kind == "F":
                    heapq.heappush(piles, t)  # 桩在 t 时刻可用
                else:
                    waiting.append(data)

            while piles and piles[0] <= t and waiting:
                car = waiting[0]
                at, ct, wt = cars[car]
                if t - at > wt:  # 等待超时，失败
                    waiting.pop(0)
                    failed += 1
                    continue
                heapq.heappop(piles)
                finish = t + ct
                heapq.heappush(events, (finish, 0, "F", 0))  # 充电结束释放
                waiting.pop(0)

        return failed
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.PriorityQueue;

public class Solution {
    private static class Event implements Comparable<Event> {
        int time, pri, type, id;

        Event(int time, int pri, int type, int id) {
            this.time = time;
            this.pri = pri;
            this.type = type;
            this.id = id;
        }

        public int compareTo(Event o) {
            if (time != o.time) return Integer.compare(time, o.time);
            return Integer.compare(pri, o.pri);
        }
    }

    public int countFailedCharging(int n, int[][] cars) {
        PriorityQueue<Event> events = new PriorityQueue<>();
        for (int i = 0; i < cars.length; i++) {
            events.offer(new Event(cars[i][0], 1, 1, i));
        }

        PriorityQueue<Integer> piles = new PriorityQueue<>();
        for (int i = 0; i < n; i++) piles.offer(0);

        List<Integer> waiting = new ArrayList<>();
        int failed = 0;

        while (!events.isEmpty()) {
            int t = events.peek().time;
            List<Event> batch = new ArrayList<>();
            while (!events.isEmpty() && events.peek().time == t) {
                batch.add(events.poll());
            }
            batch.sort((a, b) -> Integer.compare(a.pri, b.pri));

            for (Event e : batch) {
                if (e.type == 0) {
                    piles.offer(t);
                } else {
                    waiting.add(e.id);
                }
            }

            while (!piles.isEmpty() && piles.peek() <= t && !waiting.isEmpty()) {
                int car = waiting.get(0);
                int at = cars[car][0], ct = cars[car][1], wt = cars[car][2];
                if (t - at > wt) {
                    waiting.remove(0);
                    failed++;
                    continue;
                }
                piles.poll();
                events.offer(new Event(t + ct, 0, 0, 0));
                waiting.remove(0);
            }
        }
        return failed;
    }
}
```

### C++

```cpp
#include <algorithm>
#include <queue>
#include <tuple>
#include <vector>

using namespace std;

class Solution {
 public:
  int countFailedCharging(int n, vector<vector<int>>& cars) {
    using Event = tuple<int, int, char, int>;
    priority_queue<Event, vector<Event>, greater<Event>> events;
    for (int i = 0; i < (int)cars.size(); i++) {
      events.push({cars[i][0], 1, 'A', i});
    }

    priority_queue<int, vector<int>, greater<int>> piles;
    for (int i = 0; i < n; i++) piles.push(0);

    vector<int> waiting;
    int failed = 0;

    while (!events.empty()) {
      int t = get<0>(events.top());
      vector<Event> batch;
      while (!events.empty() && get<0>(events.top()) == t) {
        batch.push_back(events.top());
        events.pop();
      }
      sort(batch.begin(), batch.end(),
           [](const Event& a, const Event& b) { return get<1>(a) < get<1>(b); });

      for (auto& e : batch) {
        if (get<2>(e) == 'F')
          piles.push(t);
        else
          waiting.push_back(get<3>(e));
      }

      while (!piles.empty() && piles.top() <= t && !waiting.empty()) {
        int car = waiting[0];
        int at = cars[car][0], ct = cars[car][1], wt = cars[car][2];
        if (t - at > wt) {
          waiting.erase(waiting.begin());
          failed++;
          continue;
        }
        piles.pop();
        events.push({t + ct, 0, 'F', 0});
        waiting.erase(waiting.begin());
      }
    }
    return failed;
  }
};
```
