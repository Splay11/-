# 驿站补给最短耗时

## 解题思路

油箱容量只有 $B\le 100$，把「当前驿站 + 剩余电量」看成一层状态，在这张状态图上跑 Dijkstra。

1. 状态 $(u,f)$ 表示在 $u$ 号驿站、电池还剩 $f$ 格。出发是 $(1,B)$，耗时 $0$。
2. 转移有两类：在当地充 `1` 格，到 $(u,f+1)$，加时 $s_u$（要求 $f<B$）；或走一条耗能 $a$、耗时 $w$ 的路，到 $(v,f-a)$，要求 $f\ge a$。
3. 边是双向的。第一次弹出 $C$ 号驿站的某个状态，就是最短时间。
4. 若所有 $(C,f)$ 都到不了，答案为 `-1`。耗能为 $0$ 的充电（$s_u=0$）只会把电量一格格加满，状态数有限，不会死循环。

## 复杂度分析

- 时间复杂度：$O\bigl((C\cdot B+E\cdot B)\log(C\cdot B)\bigr)$。
- 空间复杂度：$O(C\cdot B+E)$。

## 代码实现

### Python

```python
import heapq


def solve(n, F, p, edges):
  # 建无向图
  g = [[] for _ in range(n + 1)]
  for u, v, c, t in edges:
    g[u].append((v, c, t))
    g[v].append((u, c, t))
  INF = 10 ** 18
  dist = [[INF] * (F + 1) for _ in range(n + 1)]
  # 出发满电
  dist[1][F] = 0
  pq = [(0, 1, F)]
  while pq:
    d, u, f = heapq.heappop(pq)
    if d != dist[u][f]:
      continue
    if u == n:
      return d
    # 充 1 格
    if f < F:
      nd = d + p[u]
      if nd < dist[u][f + 1]:
        dist[u][f + 1] = nd
        heapq.heappush(pq, (nd, u, f + 1))
    for v, c, t in g[u]:
      if f >= c:
        nd = d + t
        nf = f - c
        if nd < dist[v][nf]:
          dist[v][nf] = nd
          heapq.heappush(pq, (nd, v, nf))
  return -1


n, m, F = map(int, input().split())
p = [0] + list(map(int, input().split()))
edges = []
for _ in range(m):
  u, v, c, t = map(int, input().split())
  edges.append((u, v, c, t))
print(solve(n, F, p, edges))
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
  static long solve(int n, int F, int[] p, int[][] edges) {
    ArrayList<int[]>[] g = new ArrayList[n + 1];
    for (int i = 1; i <= n; i++) g[i] = new ArrayList<int[]>();
    for (int[] e : edges) {
      g[e[0]].add(new int[] {e[1], e[2], e[3]});
      g[e[1]].add(new int[] {e[0], e[2], e[3]});
    }
    long INF = Long.MAX_VALUE / 4;
    long[][] dist = new long[n + 1][F + 1];
    for (int i = 1; i <= n; i++) Arrays.fill(dist[i], INF);
    dist[1][F] = 0;
    PriorityQueue<long[]> pq = new PriorityQueue<long[]>(new Comparator<long[]>() {
      public int compare(long[] a, long[] b) { return Long.compare(a[0], b[0]); }
    });
    pq.add(new long[] {0, 1, F});
    while (!pq.isEmpty()) {
      long[] cur = pq.poll();
      long d = cur[0];
      int u = (int) cur[1], f = (int) cur[2];
      if (d != dist[u][f]) continue;
      if (u == n) return d;
      if (f < F) {
        long nd = d + p[u];
        if (nd < dist[u][f + 1]) {
          dist[u][f + 1] = nd;
          pq.add(new long[] {nd, u, f + 1});
        }
      }
      for (int[] e : g[u]) {
        if (f >= e[1]) {
          long nd = d + e[2];
          int nf = f - e[1];
          if (nd < dist[e[0]][nf]) {
            dist[e[0]][nf] = nd;
            pq.add(new long[] {nd, e[0], nf});
          }
        }
      }
    }
    return -1;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] sp = br.readLine().trim().split("\\s+");
    int n = Integer.parseInt(sp[0]);
    int m = Integer.parseInt(sp[1]);
    int F = Integer.parseInt(sp[2]);
    String[] ps = br.readLine().trim().split("\\s+");
    int[] p = new int[n + 1];
    for (int i = 1; i <= n; i++) p[i] = Integer.parseInt(ps[i - 1]);
    int[][] edges = new int[m][4];
    for (int i = 0; i < m; i++) {
      String[] e = br.readLine().trim().split("\\s+");
      for (int j = 0; j < 4; j++) edges[i][j] = Integer.parseInt(e[j]);
    }
    System.out.println(solve(n, F, p, edges));
  }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Node {
  long long d;
  int u, f;
  Node(long long d, int u, int f) : d(d), u(u), f(f) {}
  bool operator>(const Node& o) const { return d > o.d; }
};

long long solve(int n, int F, const vector<int>& p,
                const vector<tuple<int, int, int, int> >& edges) {
  vector<vector<tuple<int, int, int> > > g(n + 1);
  for (size_t i = 0; i < edges.size(); i++) {
    int u = get<0>(edges[i]), v = get<1>(edges[i]);
    int c = get<2>(edges[i]), t = get<3>(edges[i]);
    g[u].push_back(make_tuple(v, c, t));
    g[v].push_back(make_tuple(u, c, t));
  }
  const long long INF = (1LL << 62);
  vector<vector<long long> > dist(n + 1, vector<long long>(F + 1, INF));
  dist[1][F] = 0;
  priority_queue<Node, vector<Node>, greater<Node> > pq;
  pq.push(Node(0, 1, F));
  while (!pq.empty()) {
    Node cur = pq.top();
    pq.pop();
    if (cur.d != dist[cur.u][cur.f]) continue;
    if (cur.u == n) return cur.d;
    if (cur.f < F) {
      long long nd = cur.d + p[cur.u];
      if (nd < dist[cur.u][cur.f + 1]) {
        dist[cur.u][cur.f + 1] = nd;
        pq.push(Node(nd, cur.u, cur.f + 1));
      }
    }
    for (size_t j = 0; j < g[cur.u].size(); j++) {
      int v = get<0>(g[cur.u][j]), c = get<1>(g[cur.u][j]), t = get<2>(g[cur.u][j]);
      if (cur.f >= c) {
        long long nd = cur.d + t;
        int nf = cur.f - c;
        if (nd < dist[v][nf]) {
          dist[v][nf] = nd;
          pq.push(Node(nd, v, nf));
        }
      }
    }
  }
  return -1;
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m, F;
  cin >> n >> m >> F;
  vector<int> p(n + 1);
  for (int i = 1; i <= n; i++) cin >> p[i];
  vector<tuple<int, int, int, int> > edges;
  for (int i = 0; i < m; i++) {
    int u, v, c, t;
    cin >> u >> v >> c >> t;
    edges.push_back(make_tuple(u, v, c, t));
  }
  cout << solve(n, F, p, edges) << "\n";
  return 0;
}
```
