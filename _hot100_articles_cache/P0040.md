其实看完样例，咱们又有BFS的基础，其实思路很清晰了！维护一个队列，一开始将序列化数组的第一个元素（也就是根）放入队列。接着每次将队列的队首弹出，将当前序列化数组的第一个元素拿出来当它的左儿子，将当前序列化数组的第二个元素拿出来当它的右儿子。此时当拿出来的元素非null时，需要加入队列。直到整个序列化数组为空

    if not tree_arr or tree_arr[0] is None:

    q = []

#include <iostream>
#include <vector>
#include <string>
#include <queue>
#include <sstream>
using namespace std;

// 定义二叉树节点类
class TreeNode {
public:
    int val;
    TreeNode* left;
    TreeNode* right;

    TreeNode(int val = 0, TreeNode* left = nullptr, TreeNode* right = nullptr) {
        this->val = val;
        this->left = left;
        this->right = right;
    }
};

TreeNode* build_tree(vector<string> tree_arr) {
    // 空树
    if (tree_arr.empty() || tree_arr[0] == "null") {
        return nullptr;
    }

    queue<TreeNode*> q;
    TreeNode* rt = new TreeNode(stoi(tree_arr[0]));
    q.push(rt);

    int index = 1;
    while (!q.empty() && index < (int)tree_arr.size()) {
        TreeNode* node = q.front();
        q.pop();

        // 左孩子
        if (index < (int)tree_arr.size() && tree_arr[index] != "null") {
            node->left = new TreeNode(stoi(tree_arr[index]));
            q.push(node->left);
        }
        index += 1;

        // 右孩子
        if (index < (int)tree_arr.size() && tree_arr[index] != "null") {
            node->right = new TreeNode(stoi(tree_arr[index]));
            q.push(node->right);
        }
        index += 1;
    }

    return rt;
}

// 将字符串处理成序列化数组(不一定要用到，万一要用到，得会)
vector<string> deserialize(string tree_str) {
    // 去掉首尾空格
    int left = 0;
    int right = (int)tree_str.size() - 1;
    while (left <= right && isspace((unsigned char)tree_str[left])) {
        left++;
    }
    while (right >= left && isspace((unsigned char)tree_str[right])) {
        right--;
    }
    tree_str = tree_str.substr(left, right - left + 1);

    if (tree_str == "[]") {
        return {};
    }

    // 将括号去除之后，按逗号分割
    // "[1,null,2]" -> "1,null,2" -> ["1","null","2"]
    string content = tree_str.substr(1, tree_str.size() - 2);
    vector<string> raw_arr;
    stringstream ss(content);
    string raw;
    while (getline(ss, raw, ',')) {
        raw_arr.push_back(raw);
    }

    vector<string> tree_arr;
    // 转int数组👇
    // ["1","null","2"] -> [1,None,2]
    for (string raw : raw_arr) {
        int start = 0;
        int end = (int)raw.size() - 1;
        while (start <= end && isspace((unsigned char)raw[start])) {
            start++;
        }
        while (end >= start && isspace((unsigned char)raw[end])) {
            end--;
        }
        string node_val = raw.substr(start, end - start + 1);

        // 兼容 null / None
        if (node_val == "null" || node_val == "None") {
            tree_arr.push_back("null");
        } else {
            tree_arr.push_back(node_val);
        }
    }

    return tree_arr;
}

int main() {
    string tree_str = "[1,2,3,null,4,null,5,6]";
    vector<string> tree_arr = deserialize(tree_str);
    TreeNode* rt = build_tree(tree_arr);

    return 0;
}

import java.util.*;

// 定义二叉树节点类
class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }

    TreeNode(int val) {
        this(val, null, null);
    }

    TreeNode() {
        this(0, null, null);
    }
}

public class Main {
    public static TreeNode build_tree(List<String> tree_arr) {
        // 空树
        if (tree_arr == null || tree_arr.size() == 0 || tree_arr.get(0).equals("null")) {
            return null;
        }

        Queue<TreeNode> q = new LinkedList<>();
        TreeNode rt = new TreeNode(Integer.parseInt(tree_arr.get(0)));
        q.offer(rt);

        int index = 1;
        while (!q.isEmpty() && index < tree_arr.size()) {
            TreeNode node = q.poll();

            // 左孩子
            if (index < tree_arr.size() && !tree_arr.get(index).equals("null")) {
                node.left = new TreeNode(Integer.parseInt(tree_arr.get(index)));
                q.offer(node.left);
            }
            index += 1;

            // 右孩子
            if (index < tree_arr.size() && !tree_arr.get(index).equals("null")) {
                node.right = new TreeNode(Integer.parseInt(tree_arr.get(index)));
                q.offer(node.right);
            }
            index += 1;
        }

        return rt;
    }

    // 将字符串处理成序列化数组(不一定要用到，万一要用到，得会)
    public static List<String> deserialize(String tree_str) {
        // 去掉首尾空格
        tree_str = tree_str.trim();
        if (tree_str.equals("[]")) {
            return new ArrayList<>();
        }

        // 将括号去除之后，按逗号分割
        // "[1,null,2]" -> "1,null,2" -> ["1","null","2"]
        String[] raw_arr = tree_str.substring(1, tree_str.length() - 1).split(",");
        List<String> tree_arr = new ArrayList<>();

        // 转int数组👇
        // ["1","null","2"] -> [1,None,2]
        for (String raw : raw_arr) {
            String node_val = raw.trim();
            // 兼容 null / None
            if (node_val.equals("null") || node_val.equals("None")) {
                tree_arr.add("null");
            } else {
                tree_arr.add(String.valueOf(Integer.parseInt(node_val)));
            }
        }

        return tree_arr;
    }

    public static void main(String[] args) {
        String tree_str = "[1,2,3,null,4,null,5,6]";
        List<String> tree_arr = deserialize(tree_str);
        TreeNode rt = build_tree(tree_arr);
    }
}
