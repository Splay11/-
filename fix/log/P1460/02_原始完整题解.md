## 思路：最短路-堆优化Dijkstra

本题是一道堆优化的 Dijkstra 模板题。

每次从 $1$ 号点到达 $x$ 号点后，需要从 $x$ 号点再返回 $1$ 号点。
有 $q(1\leq q < 10^5)$ 次询问，所以单次时间复杂度不能超过 $O(\log n)$

从 $1$ 号点到达 $x$ 号点的最短距离，必然也是从 $x$ 号点到达 $1$ 号点的最短距离。所以相当于到达每个点再返回的总距离就是从 $1$ 号点到达给定 $q$ 个点的总距离乘 $2$ 。

这就是一个单源最短路径问题，由于数据规模是 $1\leq n,m < 10^5$ ，所以需要堆优化的 Dijkstra 来解决。

时间复杂度：$O(m\log n)$

## 代码

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef pair<int, int> PII;
const int INF = 0x3f3f3f3f;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, T;
    cin >> n >> m >> T;

    vector<vector<PII>> g(n);
    for (int i = 0; i < m; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        u--; v--;
        g[u].emplace_back(v, w);
        g[v].emplace_back(u, w);
    }

    priority_queue<PII, vector<PII>, greater<PII>> q;

    vector<int> st(n, 0);
    vector<int> dist(n, INF);
    dist[0] = 0;

    q.emplace(0, 0);
    while (!q.empty()) {
        auto top = q.top(); q.pop();
        int u = top.second;
        if (st[u]) continue;
        st[u] = 1;

        for (auto& [v, w]: g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                q.emplace(dist[v], v);
            }
        }
    }

    long long ans = 0;
    while (T--) {
        int x; cin >> x;
        x--;
        ans += dist[x];
    }
    cout << ans * 2 << "\n";
    return 0;
}
```

### python

```python
import heapq

INF = float('inf')

def main():
    n, m, T = map(int, input().split())

    g = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        g[u].append((v, w))
        g[v].append((u, w))

    q = [(0, 0)]

    st = [0] * n
    dist = [INF] * n
    dist[0] = 0

    while q:
        top = heapq.heappop(q)
        u = top[1]
        if st[u]:
            continue
        st[u] = 1

        for v, w in g[u]:
            if dist[v] > dist[u] + w:
                dist[v] = dist[u] + w
                heapq.heappush(q, (dist[v], v))

    places = list(map(int, input().split()))
    ans = 0
    for i in places:
        x = i - 1
        ans += dist[x]

    print(ans * 2)


if __name__ == "__main__":
    main()

```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        final int INF = 0x3f3f3f3f;
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int T = scanner.nextInt();

        List<List<PII>> g = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            g.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            int w = scanner.nextInt();
            g.get(u).add(new PII(v, w));
            g.get(v).add(new PII(u, w));
        }

        PriorityQueue<PII> q = new PriorityQueue<>(Comparator.comparingInt(PII::getFirst));

        int[] st = new int[n];
        int[] dist = new int[n];
        Arrays.fill(dist, INF);
        dist[0] = 0;

        q.offer(new PII(0, 0));
        while (!q.isEmpty()) {
            PII top = q.poll();
            int u = top.getSecond();
            if (st[u] == 1) continue;
            st[u] = 1;

            for (PII p : g.get(u)) {
                int v = p.getFirst();
                int w = p.getSecond();
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    q.offer(new PII(dist[v], v));
                }
            }
        }

        long ans = 0;
        while (T-- > 0) {
            int x = scanner.nextInt() - 1;
            ans += dist[x];
        }
        System.out.println(ans * 2);
    }

    static class PII {
        private final int first;
        private final int second;

        public PII(int first, int second) {
            this.first = first;
            this.second = second;
        }

        public int getFirst() {
            return first;
        }

        public int getSecond() {
            return second;
        }
    }
}
```
