## 解题思路

每次配送都是从 `1` 号驿站走到目标驿站再返回 `1` 号。道路双向，最短路对称，因此一次往返的长度等于 `1` 号到该驿站最短路的两倍。

全部 $q$ 次配送互不影响，答案就是

$$
2\sum_{i=1}^{q} \mathrm{dist}(1, x_i).
$$

$n,m\le 10^5$，需要堆优化 Dijkstra 求出 `1` 号到所有驿站的最短路，再对 $q$ 个目标累加。

## 复杂度分析

时间复杂度 $O((n+m)\log n+q)$，空间复杂度 $O(n+m)$。

## 代码实现

### Python

```python
import heapq

INF = float("inf")


def main():
    n, m, T = map(int, input().split())
    g = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        g[u].append((v, w))
        g[v].append((u, w))
    q = [(0, 0)]
    st = [0] * n
    dist = [INF] * n
    dist[0] = 0
    while q:
        d, u = heapq.heappop(q)
        if st[u]:
            continue
        st[u] = 1
        for v, w in g[u]:
            if dist[v] > dist[u] + w:
                dist[v] = dist[u] + w
                heapq.heappush(q, (dist[v], v))
    places = list(map(int, input().split()))
    ans = 0
    for i in places:
        ans += dist[i - 1]
    print(ans * 2)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        final int INF = 0x3f3f3f3f;
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int T = scanner.nextInt();
        List<List<PII>> g = new ArrayList<>();
        for (int i = 0; i < n; i++) g.add(new ArrayList<>());
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            int w = scanner.nextInt();
            g.get(u).add(new PII(v, w));
            g.get(v).add(new PII(u, w));
        }
        PriorityQueue<PII> q = new PriorityQueue<>(Comparator.comparingInt(PII::getFirst));
        int[] st = new int[n];
        int[] dist = new int[n];
        Arrays.fill(dist, INF);
        dist[0] = 0;
        q.offer(new PII(0, 0));
        while (!q.isEmpty()) {
            PII top = q.poll();
            int u = top.getSecond();
            if (st[u] == 1) continue;
            st[u] = 1;
            for (PII p : g.get(u)) {
                int v = p.getFirst(), w = p.getSecond();
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    q.offer(new PII(dist[v], v));
                }
            }
        }
        long ans = 0;
        while (T-- > 0) {
            int x = scanner.nextInt() - 1;
            ans += dist[x];
        }
        System.out.println(ans * 2);
    }

    static class PII {
        private final int first, second;
        public PII(int first, int second) {
            this.first = first;
            this.second = second;
        }
        public int getFirst() { return first; }
        public int getSecond() { return second; }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef pair<int, int> PII;
const int INF = 0x3f3f3f3f;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, T;
    cin >> n >> m >> T;
    vector<vector<PII>> g(n);
    for (int i = 0; i < m; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        --u; --v;
        g[u].emplace_back(v, w);
        g[v].emplace_back(u, w);
    }
    priority_queue<PII, vector<PII>, greater<PII>> q;
    vector<int> st(n, 0), dist(n, INF);
    dist[0] = 0;
    q.emplace(0, 0);
    while (!q.empty()) {
        int u = q.top().second;
        q.pop();
        if (st[u]) continue;
        st[u] = 1;
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                q.emplace(dist[v], v);
            }
        }
    }
    long long ans = 0;
    while (T--) {
        int x;
        cin >> x;
        ans += dist[x - 1];
    }
    cout << ans * 2 << "\n";
    return 0;
}
```
