## 解题思路

将观测序列排序为 $b_1 \le b_2 \le \cdots \le b_n$。统计第 $k$ 小值恰好等于 $b_{i+1}$（$0$-下标为 $b_i$）的子序列数量。

要使 $b_i$ 成为第 $k$ 小（$i$ 为 $0$-下标，对应排序后第 $i+1$ 个）：

- 必须选中 $b_i$；
- 从它左边 $i$ 个元素中恰好选出 $k-1$ 个；
- 右边 $n-i-1$ 个元素可任选，共 $2^{n-i-1}$ 种。

因此当 $i \ge k-1$ 时，贡献为

$$
b_i \cdot \binom{i}{k-1} \cdot 2^{n-i-1}
$$

对所有这样的 $i$ 求和并模 $998244353$。预处理阶乘、逆元阶乘与 $2$ 的幂，即可 $O(n)$ 完成一组数据。

## 复杂度分析

设单个文件 $n$ 之和为 $\Sigma n$。预处理 $O(\max n)$，每组 $O(n)$，总时间复杂度 $O(\max n + \Sigma n)$，空间复杂度 $O(\max n)$。

## 代码实现

### Python

```python
MOD = 998244353

def modpow(a, e):
    res = 1
    a %= MOD
    while e:
        if e & 1:
            res = res * a % MOD
        a = a * a % MOD
        e >>= 1
    return res

maxn = 200000
fac = [1] * (maxn + 1)
invfac = [1] * (maxn + 1)
pow2 = [1] * (maxn + 1)
for i in range(1, maxn + 1):
    fac[i] = fac[i - 1] * i % MOD
invfac[maxn] = modpow(fac[maxn], MOD - 2)
for i in range(maxn, 0, -1):
    invfac[i - 1] = invfac[i] * i % MOD
for i in range(1, maxn + 1):
    pow2[i] = pow2[i - 1] * 2 % MOD

import sys
input = sys.stdin.readline
T = int(input())
for _ in range(T):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    for i in range(k - 1, n):
        comb = fac[i] * invfac[k - 1] % MOD * invfac[i - (k - 1)] % MOD
        ans = (ans + a[i] % MOD * comb % MOD * pow2[n - i - 1]) % MOD
    print(ans)
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final int MOD = 998244353;
    static long modpow(long a, long e) {
        long r = 1;
        a %= MOD;
        while (e > 0) {
            if ((e & 1) == 1) r = r * a % MOD;
            a = a * a % MOD;
            e >>= 1;
        }
        return r;
    }
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        int maxn = 200000;
        long[] fac = new long[maxn + 1], invfac = new long[maxn + 1], pow2 = new long[maxn + 1];
        fac[0] = 1;
        for (int i = 1; i <= maxn; i++) fac[i] = fac[i - 1] * i % MOD;
        invfac[maxn] = modpow(fac[maxn], MOD - 2);
        for (int i = maxn; i > 0; i--) invfac[i - 1] = invfac[i] * i % MOD;
        pow2[0] = 1;
        for (int i = 1; i <= maxn; i++) pow2[i] = pow2[i - 1] * 2 % MOD;

        while (T-- > 0) {
            String[] nk = br.readLine().split(" ");
            int n = Integer.parseInt(nk[0]);
            int k = Integer.parseInt(nk[1]);
            long[] a = new long[n];
            String[] parts = br.readLine().split(" ");
            for (int i = 0; i < n; i++) a[i] = Long.parseLong(parts[i]);
            Arrays.sort(a);
            long ans = 0;
            for (int i = k - 1; i < n; i++) {
                long comb = fac[i] * invfac[k - 1] % MOD * invfac[i - (k - 1)] % MOD;
                ans = (ans + a[i] % MOD * comb % MOD * pow2[n - i - 1]) % MOD;
            }
            System.out.println(ans);
        }
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 998244353;

ll modpow(ll a, ll e) {
    ll r = 1;
    while (e) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    int maxn = 200000;
    vector<ll> fac(maxn + 1), invfac(maxn + 1), pow2(maxn + 1);
    fac[0] = 1;
    for (int i = 1; i <= maxn; i++) fac[i] = fac[i - 1] * i % MOD;
    invfac[maxn] = modpow(fac[maxn], MOD - 2);
    for (int i = maxn; i > 0; i--) invfac[i - 1] = invfac[i] * i % MOD;
    pow2[0] = 1;
    for (int i = 1; i <= maxn; i++) pow2[i] = pow2[i - 1] * 2 % MOD;

    while (T--) {
        int n, k;
        cin >> n >> k;
        vector<ll> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        ll ans = 0;
        for (int i = k - 1; i < n; i++) {
            ll comb = fac[i] * invfac[k - 1] % MOD * invfac[i - (k - 1)] % MOD;
            ans = (ans + a[i] % MOD * comb % MOD * pow2[n - i - 1]) % MOD;
        }
        cout << ans << '\n';
    }
    return 0;
}
```
