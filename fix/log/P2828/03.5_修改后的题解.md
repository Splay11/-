## 解题思路

题目等价于：有一个长度为 $n$ 的排列 $a$，每次可以交换两个位置上的魔力等级，但要求这两个等级的差的绝对值不超过 $k$。若差为 $1$ 则消耗 $1$ 点魔力；若差大于 $1$ 且不超过 $k$ 则不消耗魔力。目标是将序列变成 $1,2,\dots,n$，求最小魔力消耗。

根据 $k$ 的不同取值分情况讨论：

1. **当 $k \ge 3$ 时**  
   - 可以通过一系列不消耗魔力的交换（差值 $\ge 2$ 且 $\le k$）来调整任意两个魔力等级的位置。  
   - 因此总能把序列排好序且不消耗任何魔力，答案为 $0$。

2. **当 $k = 1$ 时**  
   - 由于魔力等级互不相同，差值绝对值不超过 $1$ 等价于两数必须是相邻整数，即只能交换魔力等级相差 $1$ 的商品，且每次消耗 $1$ 点魔力。  
   - 此时任何一次交换本质上相当于一次相邻元素的交换（因为只有相邻整数才能交换），将序列排序的最小代价就等于原排列的逆序对数。  
   - 使用树状数组或归并排序计算逆序对数即可。

3. **当 $k = 2$ 时**  
   - 差值恰好为 $1$ 的交换消耗 $1$ 魔力，差值恰好为 $2$ 的交换不消耗魔力。  
   - 观察发现：所有相差 $2$ 的魔力等级之间可以免费交换，于是所有“奇数”魔力等级之间可以免费相互交换，所有“偶数”魔力等级之间也可以免费相互交换。只有奇偶之间的交换必然消耗魔力（因为奇偶差值为 $1$）。  
   - 最终有序排列 $1,2,3,\dots,n$ 要求奇数魔力等级必须位于奇数位置（第 $1,3,5,\dots$ 位），偶数魔力等级必须位于偶数位置（第 $2,4,6,\dots$ 位）。  
   - 利用免费交换，我们可以任意重排所有奇数内部的顺序，以及所有偶数内部的顺序。因此问题转化为：将初始排列中所有魔力等级为奇数的商品移到预定的奇数位置上，每次调换一个奇数和一个偶数消耗 $1$ 魔力，最小魔力消耗等于把初始奇数位置集合变换到目标奇数位置集合所需的总位移和。  
   - 具体做法：  
     - 遍历数组，记录魔力等级为奇数的商品所在的**位置**（位置从 $1$ 开始编号），得到列表 $B$。  
     - 设奇数个数为 $m$，目标奇数位置依次为 $1,3,5,\dots,2m-1$，记为 $A_i = 2i-1$。  
     - 最小魔力消耗 = $\sum_{i=1}^{m} |B_i - A_i|$。  
   - 该公式在 $O(n)$ 时间内即可计算。

## 复杂度分析

- 当 $k \ge 3$ 或 $k = 2$ 时，只需一次线性扫描，时间复杂度 $O(n)$；当 $k = 1$ 时，使用树状数组求逆序对，时间复杂度 $O(n \log n)$。
- 总的空间复杂度为 $O(n)$（存储排列或树状数组）。

## 代码实现

### Python

```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    k = int(data[1])
    a = list(map(int, data[2:]))

    if k >= 3:
        # k >= 3 时，可以不消耗魔力完成排序
        print(0)
    elif k == 1:
        # k = 1 只能相邻交换，最小代价 = 逆序数
        class BIT:
            def __init__(self, n):
                self.n = n
                self.tree = [0] * (n + 1)
            def update(self, i, delta):
                while i <= self.n:
                    self.tree[i] += delta
                    i += i & -i
            def query(self, i):
                s = 0
                while i > 0:
                    s += self.tree[i]
                    i -= i & -i
                return s

        bit = BIT(n)
        inv = 0
        for i in range(n):
            # 当前数字 a[i] 左侧已有数字中大于 a[i] 的个数
            inv += i - bit.query(a[i])
            bit.update(a[i], 1)
        print(inv)
    elif k == 2:
        # k = 2 时，所有奇数之间免费交换，所有偶数之间免费交换
        # 奇偶之间交换消耗1点魔力，目标奇数位置为 1,3,5,...
        pos_odd = []  # 存储原序列中奇数魔力等级的位置，从1开始
        for i in range(n):
            if a[i] % 2 == 1:
                pos_odd.append(i + 1)
        ans = 0
        for i, pos in enumerate(pos_odd):
            target = 2 * i + 1   # 第 i+1 个奇数应该处于的位置
            ans += abs(pos - target)
        print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 树状数组
    static class BIT {
        int n;
        int[] tree;
        BIT(int n) {
            this.n = n;
            tree = new int[n + 1];
        }
        void update(int i, int val) {
            for (; i <= n; i += i & -i) {
                tree[i] += val;
            }
        }
        int query(int i) {
            int sum = 0;
            for (; i > 0; i -= i & -i) {
                sum += tree[i];
            }
            return sum;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] firstLine = br.readLine().trim().split("\\s+");
        int n = Integer.parseInt(firstLine[0]);
        int k = Integer.parseInt(firstLine[1]);
        String[] secondLine = br.readLine().trim().split("\\s+");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(secondLine[i]);
        }

        if (k >= 3) {
            System.out.println(0);
        } else if (k == 1) {
            // k = 1：只能相邻交换，最小代价 = 逆序数
            BIT bit = new BIT(n);
            long inv = 0;
            for (int i = 0; i < n; i++) {
                inv += i - bit.query(a[i]);
                bit.update(a[i], 1);
            }
            System.out.println(inv);
        } else if (k == 2) {
            // k = 2：奇数内部、偶数内部免费交换，奇偶交换消耗1
            List<Integer> posOdd = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 == 1) {
                    posOdd.add(i + 1);
                }
            }
            long ans = 0;
            for (int i = 0; i < posOdd.size(); i++) {
                int target = 2 * i + 1;
                ans += Math.abs(posOdd.get(i) - target);
            }
            System.out.println(ans);
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

typedef long long ll;

// 树状数组
struct BIT {
    int n;
    vector<int> tree;
    BIT(int n): n(n), tree(n + 1, 0) {}
    void update(int i, int val) {
        for (; i <= n; i += i & -i) {
            tree[i] += val;
        }
    }
    int query(int i) {
        int res = 0;
        for (; i > 0; i -= i & -i) {
            res += tree[i];
        }
        return res;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    if (k >= 3) {
        cout << 0 << "\n";
    } else if (k == 1) {
        // 逆序对数
        BIT bit(n);
        ll inv = 0;
        for (int i = 0; i < n; i++) {
            inv += i - bit.query(a[i]);
            bit.update(a[i], 1);
        }
        cout << inv << "\n";
    } else if (k == 2) {
        // 奇数位置集合
        vector<int> posOdd;
        for (int i = 0; i < n; i++) {
            if (a[i] & 1) {
                posOdd.push_back(i + 1);
            }
        }
        ll ans = 0;
        int m = posOdd.size();
        for (int i = 0; i < m; i++) {
            int target = 2 * i + 1;
            ans += abs(posOdd[i] - target);
        }
        cout << ans << "\n";
    }

    return 0;
}
```
