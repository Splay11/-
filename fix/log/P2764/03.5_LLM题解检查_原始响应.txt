## 解题思路

1. **产品判定转化为数值**  
   对于第 $i$ 款产品，若 $r_i - s_i > d_i$，则定义其判定值 $v_i = 1$（优质产品）；否则定义 $v_i = -1$（普通产品）。  
   这样，一个连续子区间 $[l, r]$ 中优质产品多于普通产品，当且仅当该区间内所有 $v_i$ 的和大于 $0$。

2. **区间和转化为前缀和的比较**  
   令前缀和数组 $pre$，其中 $pre[0] = 0$，$pre[i] = pre[i-1] + v_i\ (1 \le i \le n)$。  
   对于任意区间 $[l, r]$，其和可表示为  
   $$sum = pre[r] - pre[l-1]$$  
   于是 $sum > 0$ 等价于 $pre[l-1] < pre[r]$。  

3. **问题转化为顺序对统计**  
   原问题转化为：在长度为 $n+1$ 的前缀和序列 $\{pre[0], pre[1], \dots, pre[n]\}$ 中，统计满足 $0 \le i < j \le n$ 且 $pre[i] < pre[j]$ 的对 $(i, j)$ 的个数，记作 $count$。  
   此时所有可能的连续子区间总数为  
   $$total = \frac{n(n+1)}{2}$$

4. **离散化与树状数组**  
   由于 $n \le 10^5$，可以使用树状数组（BIT）高效统计顺序对：  
   - 将所有 $pre$ 值进行离散化，得到从小到大的排名。  
   - 从 $j=0$ 到 $n$ 依次遍历前缀和数组：
     - 查询当前 BIT 中比 $pre[j]$ 小的元素个数（即查询排名在 $pre[j]$ 之前的总数），累加到 $count$。  
     - 将 $pre[j]$ 对应的排名插入 BIT（单点加 $1$）。  

5. **概率计算与模逆元**  
   所求概率为最简分数 $\frac{p}{q} = \frac{count}{total}$。  
   利用费马小定理，在模 $M = 10^9+7$ 意义下，输出  
   $$ans = (count \cdot total^{M-2}) \bmod M$$

## 复杂度分析

- **时间复杂度**：$O(n \log n)$  
  离散化需要对 $n+1$ 个前缀和排序，后续遍历过程中每次 BIT 的查询和更新均为 $O(\log n)$，总复杂度为 $O(n \log n)$。  

- **空间复杂度**：$O(n)$  
  需要存储数组 $v$、前缀和数组 $pre$、离散化数组以及树状数组，均与 $n$ 同阶。

## 代码实现

### Python

```python
import sys
input = sys.stdin.readline
MOD = 10**9 + 7

# 快速幂求模逆元
def modExp(a, b, mod):
    res = 1
    a %= mod
    while b:
        if b & 1:
            res = res * a % mod
        a = a * a % mod
        b //= 2
    return res

# 树状数组
class BIT:
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (n + 1)

    def update(self, idx, val):
        while idx <= self.n:
            self.tree[idx] += val
            idx += idx & -idx

    def query(self, idx):
        s = 0
        while idx:
            s += self.tree[idx]
            idx -= idx & -idx
        return s

def main():
    n = int(input())
    r = list(map(int, input().split()))
    s = list(map(int, input().split()))
    d = list(map(int, input().split()))

    # 将产品转化为 1（优质）或 -1（普通）
    v = []
    for i in range(n):
        if r[i] - s[i] > d[i]:
            v.append(1)
        else:
            v.append(-1)

    # 前缀和 pre[0]=0, pre[i]=pre[i-1]+v[i-1]
    pre = [0] * (n + 1)
    for i in range(1, n + 1):
        pre[i] = pre[i-1] + v[i-1]

    # 离散化所有前缀和值
    sorted_vals = sorted(set(pre))
    comp = {x: i + 1 for i, x in enumerate(sorted_vals)}

    bit = BIT(len(sorted_vals))
    count_pairs = 0

    # 统计所有满足 pre[i] < pre[j] 的 (i,j)
    for j in range(n + 1):
        idx = comp[pre[j]]
        count_pairs += bit.query(idx - 1)   # 当前 BIT 中比 pre[j] 小的数量
        bit.update(idx, 1)

    total = n * (n + 1) // 2                # 总区间数
    inv_total = modExp(total, MOD - 2, MOD)
    ans = (count_pairs % MOD) * inv_total % MOD
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final int MOD = 1000000007;

    // 快速幂求模逆元
    static long modExp(long a, long b, long mod) {
        long res = 1;
        a %= mod;
        while (b > 0) {
            if ((b & 1) == 1)
                res = (res * a) % mod;
            a = (a * a) % mod;
            b >>= 1;
        }
        return res;
    }

    // 树状数组
    static class BIT {
        int n;
        long[] tree;

        public BIT(int n) {
            this.n = n;
            tree = new long[n + 1];
        }

        public void update(int idx, long val) {
            while (idx <= n) {
                tree[idx] += val;
                idx += idx & -idx;
            }
        }

        public long query(int idx) {
            long sum = 0;
            while (idx > 0) {
                sum += tree[idx];
                idx -= idx & -idx;
            }
            return sum;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine().trim());

        String[] partsR = br.readLine().trim().split(" ");
        long[] r = new long[n];
        for (int i = 0; i < n; i++) r[i] = Long.parseLong(partsR[i]);

        String[] partsS = br.readLine().trim().split(" ");
        long[] s = new long[n];
        for (int i = 0; i < n; i++) s[i] = Long.parseLong(partsS[i]);

        String[] partsD = br.readLine().trim().split(" ");
        long[] d = new long[n];
        for (int i = 0; i < n; i++) d[i] = Long.parseLong(partsD[i]);

        // 转换为 1 或 -1
        int[] v = new int[n];
        for (int i = 0; i < n; i++) {
            v[i] = (r[i] - s[i] > d[i]) ? 1 : -1;
        }

        // 前缀和
        long[] pre = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            pre[i] = pre[i - 1] + v[i - 1];
        }

        // 离散化
        TreeSet<Long> set = new TreeSet<>();
        for (int i = 0; i <= n; i++) set.add(pre[i]);
        Map<Long, Integer> comp = new HashMap<>();
        int idx = 1;
        for (Long x : set) {
            comp.put(x, idx++);
        }

        BIT bit = new BIT(comp.size());
        long countPairs = 0;

        // 统计顺序对
        for (int j = 0; j <= n; j++) {
            int id = comp.get(pre[j]);
            countPairs += bit.query(id - 1);
            bit.update(id, 1);
        }

        long total = (long) n * (n + 1) / 2;
        long invTotal = modExp(total, MOD - 2, MOD);
        long ans = (countPairs % MOD) * invTotal % MOD;

        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 1e9 + 7;

// 快速幂
ll modExp(ll base, ll exp, ll mod) {
    ll res = 1;
    base %= mod;
    while (exp) {
        if (exp & 1) res = (res * base) % mod;
        base = (base * base) % mod;
        exp >>= 1;
    }
    return res;
}

// 树状数组
struct BIT {
    int n;
    vector<ll> tree;
    BIT(int n) : n(n), tree(n + 1, 0) {}

    void update(int idx, ll val) {
        for (; idx <= n; idx += idx & -idx)
            tree[idx] += val;
    }

    ll query(int idx) {
        ll res = 0;
        for (; idx; idx -= idx & -idx)
            res += tree[idx];
        return res;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<ll> r(n), s(n), d(n);
    for (int i = 0; i < n; i++) cin >> r[i];
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 0; i < n; i++) cin >> d[i];

    // 判定值 1 (优质) 或 -1 (普通)
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
        v[i] = (r[i] - s[i] > d[i]) ? 1 : -1;
    }

    // 前缀和
    vector<ll> pre(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        pre[i] = pre[i - 1] + v[i - 1];
    }

    // 离散化
    vector<ll> vals = pre;
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());

    auto getID = [&](ll x) {
        return lower_bound(vals.begin(), vals.end(), x) - vals.begin() + 1;
    };

    BIT bit(vals.size());
    ll countPairs = 0;

    // 统计顺序对
    for (int j = 0; j <= n; j++) {
        int id = getID(pre[j]);
        countPairs += bit.query(id - 1);   // 小于当前值的个数
        bit.update(id, 1);
    }

    ll total = (ll)n * (n + 1) / 2;
    ll invTotal = modExp(total, MOD - 2, MOD);
    ll ans = (countPairs % MOD) * invTotal % MOD;

    cout << ans << "\n";
    return 0;
}
```
