## 解题思路

本题需要在一个按照“纯元数”规则建成的无向图中，找出节点数最多的连通分量，并对每个这样的连通分量求最小生成树，最终输出这些最小生成树权值和的最小值。具体步骤如下：

1. **预处理素数（纯元数）**  
   由于数组元素 $a_i \le 10^6$，任意两数之和最大为 $2\times10^6$。使用线性筛或埃氏筛预处理区间 $[0, 2\times10^6+1]$ 内的所有素数，标记数组 `is_prime`。

2. **构建连通分量并统计大小**  
   - 初始化并查集，每个节点自成一个集合，集合大小初始为 $1$。
   - 双重循环枚举所有下标对 $i < j$，若 $a_i + a_j$ 为素数，则合并 $i$ 和 $j$ 所在的集合，并累加集合大小。

3. **确定最大连通分量**  
   遍历所有节点，找出最大的集合大小 `max_size`。对于每个集合的根节点（满足 `find(i) == i` 且大小等于 `max_size` 的连通块），单独处理。

4. **对每个最大连通分量求最小生成树**  
   对于每个符合要求的连通分量：
   - 提取该分量内的所有节点编号（记为 `nodes`）。
   - 枚举这些节点内部的所有点对 $(u, v)$，若 $a_u + a_v$ 是素数，则生成一条边，边权为 $a_u+a_v$。
   - 使用 Kruskal 算法：新建一个小的并查集维护本分量内的连通性，将所有边按权值升序排序，依次加入不形成环的边，直到选出 $|nodes|-1$ 条边，累加权值得到最小生成树的总权值。

5. **输出答案**  
   对所有最大连通分量的最小生成树总权值取最小值输出。题目保证每个连通分量至少存在一棵生成树，因此总能求出结果。

## 复杂度分析

- **时间复杂度**：  
  - 素数预处理 $O(M)$，其中 $M = 2\times10^6$。  
  - 枚举点对建图 $O(n^2)$。  
  - 并查集操作近似 $O(1)$。  
  - 对最大连通分量求最小生成树时，最坏情况下包含 $n$ 个节点，边数 $O(n^2)$，排序耗时 $O(n^2 \log n)$。  
  整体时间复杂度为 $O(n^2 \log n)$，在 $n \le 1000$ 时可高效运行。

- **空间复杂度**：  
  需要存储素数标记数组 $O(M)$，并查集父节点与大小数组 $O(n)$，存储边的数组最坏 $O(n^2)$。总空间复杂度 $O(M + n^2)$，约 $2\times10^6 + 10^6$ 量级，在限制内。

```python
import sys

def sieve(limit):
    """返回长度为 limit+1 的列表 is_prime，is_prime[x] 为 True 表示 x 是素数"""
    is_prime = [True] * (limit + 1)
    is_prime[0] = is_prime[1] = False
    for i in range(2, int(limit ** 0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, limit + 1, i):
                is_prime[j] = False
    return is_prime

class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.size = [1] * n

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        rx, ry = self.find(x), self.find(y)
        if rx == ry:
            return
        # 按大小合并，保持平衡
        if self.size[rx] < self.size[ry]:
            rx, ry = ry, rx
        self.parent[ry] = rx
        self.size[rx] += self.size[ry]

def kruskal(nodes, a, is_prime):
    """对连通分量内的节点 nodes 求最小生成树总权值"""
    m = len(nodes)
    edges = []
    # 枚举分量内所有点对
    for i in range(m):
        for j in range(i + 1, m):
            u, v = nodes[i], nodes[j]
            s = a[u] + a[v]
            if is_prime[s]:
                edges.append((s, i, j))
    edges.sort(key=lambda x: x[0])
    uf = UnionFind(m)
    total = 0
    edges_used = 0
    for w, i, j in edges:
        if uf.find(i) != uf.find(j):
            uf.union(i, j)
            total += w
            edges_used += 1
            if edges_used == m - 1:
                break
    return total

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    a = list(map(int, data[1:1+n]))
    max_sum = max(a) * 2 if a else 0
    limit = max(2000000, max_sum)  # 保证至少覆盖 2e6
    is_prime = sieve(limit)
    
    uf = UnionFind(n)
    # 建图，合并连通分量
    for i in range(n):
        for j in range(i + 1, n):
            if is_prime[a[i] + a[j]]:
                uf.union(i, j)
    
    # 统计连通分量大小
    comp_size = {}
    for i in range(n):
        root = uf.find(i)
        comp_size[root] = uf.size[root]  # 已经通过 union 维护 size，也可以用 comp_size[root]=uf.size[root]
    if not comp_size:
        print(0)
        return
    max_size = max(comp_size.values())
    
    # 找出所有大小为 max_size 的连通分量的根
    roots = [r for r, sz in comp_size.items() if sz == max_size]
    
    ans = float('inf')
    for root in roots:
        # 提取该分量内的所有节点
        nodes = [i for i in range(n) if uf.find(i) == root]
        mst_weight = kruskal(nodes, a, is_prime)
        if mst_weight < ans:
            ans = mst_weight
    print(ans)

if __name__ == "__main__":
    main()
```

```java
import java.util.*;

public class Main {
    static final int MAX_VAL = 2000000;
    static boolean[] isPrime;

    // 素数筛
    static void sieve(int limit) {
        isPrime = new boolean[limit + 1];
        Arrays.fill(isPrime, true);
        isPrime[0] = isPrime[1] = false;
        for (int i = 2; i * i <= limit; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= limit; j += i) {
                    isPrime[j] = false;
                }
            }
        }
    }

    // 并查集
    static class UnionFind {
        int[] parent, size;
        UnionFind(int n) {
            parent = new int[n];
            size = new int[n];
            for (int i = 0; i < n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }
        int find(int x) {
            if (parent[x] != x) {
                parent[x] = find(parent[x]);
            }
            return parent[x];
        }
        void union(int x, int y) {
            int rx = find(x), ry = find(y);
            if (rx == ry) return;
            if (size[rx] < size[ry]) {
                int tmp = rx; rx = ry; ry = tmp;
            }
            parent[ry] = rx;
            size[rx] += size[ry];
        }
    }

    // Kruskal 求连通分量最小生成树
    static int kruskal(List<Integer> nodes, int[] a) {
        int m = nodes.size();
        List<int[]> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            for (int j = i + 1; j < m; j++) {
                int u = nodes.get(i), v = nodes.get(j);
                int sum = a[u] + a[v];
                if (isPrime[sum]) {
                    edges.add(new int[]{sum, i, j});
                }
            }
        }
        edges.sort(Comparator.comparingInt(e -> e[0]));
        UnionFind uf = new UnionFind(m);
        int total = 0;
        int edgesUsed = 0;
        for (int[] e : edges) {
            if (uf.find(e[1]) != uf.find(e[2])) {
                uf.union(e[1], e[2]);
                total += e[0];
                edgesUsed++;
                if (edgesUsed == m - 1) break;
            }
        }
        return total;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int maxAi = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            if (a[i] > maxAi) maxAi = a[i];
        }
        int limit = Math.max(MAX_VAL, maxAi * 2);
        sieve(limit);

        UnionFind uf = new UnionFind(n);
        // 枚举所有点对合并
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (isPrime[a[i] + a[j]]) {
                    uf.union(i, j);
                }
            }
        }

        // 统计连通分量大小
        Map<Integer, Integer> compSize = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int root = uf.find(i);
            compSize.put(root, uf.size[root]);
        }
        int maxSize = 0;
        for (int sz : compSize.values()) {
            if (sz > maxSize) maxSize = sz;
        }

        int ans = Integer.MAX_VALUE;
        for (Map.Entry<Integer, Integer> entry : compSize.entrySet()) {
            if (entry.getValue() == maxSize) {
                int root = entry.getKey();
                List<Integer> nodes = new ArrayList<>();
                for (int i = 0; i < n; i++) {
                    if (uf.find(i) == root) nodes.add(i);
                }
                int mst = kruskal(nodes, a);
                if (mst < ans) ans = mst;
            }
        }
        System.out.println(ans);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAX_VAL = 2000000;
vector<bool> isPrime;

// 素数筛
void sieve(int limit) {
    isPrime.assign(limit + 1, true);
    isPrime[0] = isPrime[1] = false;
    for (int i = 2; i * i <= limit; ++i) {
        if (isPrime[i]) {
            for (int j = i * i; j <= limit; j += i) {
                isPrime[j] = false;
            }
        }
    }
}

// 并查集
struct UnionFind {
    vector<int> parent, sz;
    UnionFind(int n) : parent(n), sz(n, 1) {
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int x) {
        return parent[x] == x ? x : parent[x] = find(parent[x]);
    }
    void unite(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return;
        if (sz[rx] < sz[ry]) swap(rx, ry);
        parent[ry] = rx;
        sz[rx] += sz[ry];
    }
};

// 对分量 nodes 求最小生成树总权值
int kruskal(const vector<int>& nodes, const vector<int>& a) {
    int m = nodes.size();
    vector<tuple<int, int, int>> edges; // (weight, u_idx, v_idx)
    for (int i = 0; i < m; ++i) {
        for (int j = i + 1; j < m; ++j) {
            int u = nodes[i], v = nodes[j];
            int sum = a[u] + a[v];
            if (isPrime[sum]) {
                edges.emplace_back(sum, i, j);
            }
        }
    }
    sort(edges.begin(), edges.end());
    UnionFind uf(m);
    int total = 0, edgesUsed = 0;
    for (auto [w, i, j] : edges) {
        if (uf.find(i) != uf.find(j)) {
            uf.unite(i, j);
            total += w;
            ++edgesUsed;
            if (edgesUsed == m - 1) break;
        }
    }
    return total;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> a(n);
    int maxAi = 0;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        if (a[i] > maxAi) maxAi = a[i];
    }
    int limit = max(MAX_VAL, maxAi * 2);
    sieve(limit);

    UnionFind uf(n);
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            if (isPrime[a[i] + a[j]]) {
                uf.unite(i, j);
            }
        }
    }

    unordered_map<int, int> compSize; // root -> size
    for (int i = 0; i < n; ++i) {
        int root = uf.find(i);
        compSize[root] = uf.sz[root];
    }
    int maxSize = 0;
    for (auto& p : compSize) {
        if (p.second > maxSize) maxSize = p.second;
    }

    int ans = INT_MAX;
    for (auto& p : compSize) {
        if (p.second == maxSize) {
            int root = p.first;
            vector<int> nodes;
            for (int i = 0; i < n; ++i) {
                if (uf.find(i) == root) nodes.push_back(i);
            }
            int mst = kruskal(nodes, a);
            if (mst < ans) ans = mst;
        }
    }
    cout << ans << "\n";
    return 0;
}
```
