## 题目思路

显然只需要按速度的值排序即可，速度最小值（负数最大绝对值的）放在最左边，然后依次减慢，然后再是正向的由慢到快。

## 代码

**Python代码**

```python
n = int(input())
airps = []		#存储每个点的速度和编号
point = []		#存储每个坐标
for i in range(n):
	x, v = map(int, input().split())
	point.append(x)
	airps.append([v,i])		#速度放在第一维，sort默认按每一维依次排序。	airps[i][0]代表速度，airps[i][1]代表编号，即输入顺序
point.sort()	#坐标从小到大排序
airps.sort()	#sort默认按每一维依次排序，优先按第一维速度排
for i in range(n):
	airps[i].append(point[i])		#每个飞机对应从左到右的相应位置，存为a[i][2]
airps.sort(key = lambda x: x[1])	#按照编号排序，然后直接输出即可
for i in airps:
	print(i[2], i[0])	#输出位置和速度
```

**C++代码**

```c++
#include <iostream>
#include <algorithm>
using namespace std;
int n;
const int maxn=2e5+10;
int a[maxn];
struct  pp
{
	int id,x,v;
}p[maxn];
bool cmp1(pp a,pp b){
	return a.id<b.id;
}
bool cmp2(pp a,pp b){
	return a.v<b.v;
}
int main() {
	cin>>n;
	for (int i=1;i<=n;i++)
	{
		p[i].id=i;
		cin>>a[i];
		cin>>p[i].v;
	}
	sort(a+1,a+1+n);//坐标从小到大排序
	sort(p+1,p+1+n,cmp2);//速度从小到大排序
	for (int i=1;i<=n;i++){//速度匹配对应坐标
		p[i].x=a[i];
	}
	sort(p+1,p+1+n,cmp1);//编号排序
	for (int i=1;i<=n;i++)
		cout<<p[i].x<<" "<<p[i].v<<endl;
	return 0;
}
```

**Java代码**

```java
import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

class pp {
    int id, x, v;
}

class cmp1 implements Comparator<pp> {
    public int compare(pp a, pp b) {
        return a.id - b.id;
    }
}

class cmp2 implements Comparator<pp> {
    public int compare(pp a, pp b) {
        return a.v - b.v;
    }
}

class Main {
    public static int n;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        pp[] p = new pp[n + 1];
        for (int i = 1; i <= n; i++)
            p[i] = new pp();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i].id = i;
            a[i] = scanner.nextInt();
            p[i].v = scanner.nextInt();
        }
        Arrays.sort(a, 1, n + 1);//坐标从小到大排序
        Arrays.sort(p, 1, n + 1, new cmp2());//速度从小到大排序
        for (int i = 1; i <= n; i++) {//速度匹配对应坐标
            p[i].x = a[i];
        }
        Arrays.sort(p, 1, n + 1, new cmp1());//编号排序
        for (int i = 1; i <= n; i++)
            System.out.println(p[i].x + " " + p[i].v);
    }
}
```
