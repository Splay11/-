## 思路：构造

首先需要获取 $s$ 的字符集，这里用一个集合 $ch$ 来存储 $s$ 的字符集合。

对于 $t$ 来说，如果 $s$ 的字符集大小为 $1$ ，则说明无法构造。

对于字符集大小大于 $1$ 的情况，我们在构造时，优先考虑让每个字符先出现一遍。

这里令列表 $lst$ 为 $ch$ 中的字符集的列表形式。

在我们从前往后构造每个字符，当遍历到索引为 $i$ 的字符 $s[i]$ 时，遍历字符集 $ch$ ，如果在 $ch$ 中找到一个字符和 $s[i]$ 不相等，则将这个字符放到 $t[i]$ 中，然后将这个字符从 $ch$ 中删除。

如果我们遍历整个 $ch$ ，都不能找到一个字符和 $s[i]$ 不相等，则再考虑 $lst[0]$ 和 $lst[1]$ 中与 $s[i]$ 不相等的字符。这样，我们就一定可以使得 $ch$ 中的每个字符都在 $t$ 中出现一次，剩余的部分只需要找到一个和 $s[i]$ 不相同的即可。

时间复杂度：$O(n)$

## 代码

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() 
{
    string s;
    cin >> s;

    // ch 中存储 s 的字符集
    unordered_set<char> ch;
    for (char c : s) {
        ch.insert(c);
    }

    // 如果只有一种字符，那就不能构造出一个 t 了
    if (ch.size() == 1) {
        cout << "-1" << "\n";
        return 0;
    }

    vector<char> lst(ch.begin(), ch.end());

    // 否则按照如下的构造方式：
    // 对于每个字符，找到 set 中不与其相同的字符，每次遍历 set 最多两次即可
    // 如果 set 中不存在，则从 lst[0] 和 lst[1] 中找到一个与当前字符不同的字符即可

    string ans;
    for (char c : s) {
        bool flag = false;
        // 遍历 ch 字符集
        for (char j : ch) {
            if (j != c) {
                ans.push_back(j);
                ch.erase(j);
                flag = true;
                break;
            }
        }
        // 如果 ch 字符集中的字符找不到与其不同的，那么再在 lst[0] 和 lst[1] 中找一个即可
        if (!flag) {
            if (lst[0] != c) {
                ans.push_back(lst[0]);
            } else {
                ans.push_back(lst[1]);
            }
        }
    }

    cout << ans << "\n";

    return 0;
}
```

```python
                ans.append(lst[1])
```

### Java

```java
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();

        // ch 中存储 s 的字符集
        Set<Character> ch = new HashSet<>();
        for (char c : s.toCharArray()) {
            ch.add(c);
        }

        // 如果只有一种字符，那就不能构造出一个 t 了
        if (ch.size() == 1) {
            System.out.println("-1");
            return;
        }

        char[] lst = new char[ch.size()];
        int index = 0;
        for (char c : ch) {
            lst[index++] = c;
        }

        // 否则按照如下的构造方式：
        // 对于每个字符，找到 set 中不与其相同的字符，每次遍历 set 最多两次即可
        // 如果 set 中不存在，则从 lst[0] 和 lst[1] 中找到一个与当前字符不同的字符即可

        StringBuilder ans = new StringBuilder();
        for (char c : s.toCharArray()) {
            boolean flag = false;
            // 遍历 ch 字符集
            for (char j : ch) {
                if (j != c) {
                    ans.append(j);
                    ch.remove(j);
                    flag = true;
                    break;
                }
            }
            // 如果 ch 字符集中的字符找不到与其不同的，那么再在 lst[0] 和 lst[1] 中找一个即可
            if (!flag) {
                if (lst[0] != c) {
                    ans.append(lst[0]);
                } else {
                    ans.append(lst[1]);
                }
            }
        }

        System.out.println(ans.toString());
    }
}
```
