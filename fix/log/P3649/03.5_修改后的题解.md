## 解题思路

把每一次工作人员在时刻 $t$ 结束前手动点亮的路灯 $a_t$ 看作一个“初始亮点”。从下一个时刻（即时刻 $t+1$）开始，每个整数时刻的开始时，该亮点会向左右各传播一格。因此在第 $T$ 个时刻结束时，该手动点亮的路灯能够覆盖的街道区间为

$$
\big[\max(1,\ a_t - (T - t)),\ \min(n,\ a_t + (T - t))\big]
$$

若 $T < t$，则该路灯还未被手动点亮，不产生任何覆盖。

问题等价于：求最小的整数时刻 $T$，使得所有满足 $t \le T$ 的手动点亮路灯产生的覆盖区间的并集恰好包含整条街道 $[1, n]$。这显然具有单调性，可以使用**二分答案**来求解。

### 判定某时刻 $T$ 是否已全亮

对于给定的 $T$，需要检查所有 $t \le T$ 的路灯产生的区间能否覆盖 $[1, n]$。

观察左端点：$L_t = a_t - (T - t) = (a_t + t) - T$，其中 $(a_t + t)$ 是与 $T$ 无关的常量。因此可以先预处理所有手动点亮的路灯，按照 $(a_t + t)$ **升序排序**。这样在二分过程中，每次判定只需按序扫描，无需重新排序。

1. 维护当前已连续覆盖到的最右位置的下一个位置 `cur`，初始为 $1$。
2. 按预排好的顺序遍历所有手动点亮的路灯 $(a_t, t)$：
   - 若 $t > T$，跳过（该路灯尚未出现）；
   - 计算左端点 $L = a_t - (T - t)$，右端点 $R = a_t + (T - t)$；
   - 若 $R < 1$ 或 $L > n$，跳过（与街道无交集）；
   - 将区间与 $[1, n]$ 取交：$L = \max(L, 1)$，$R = \min(R, n)$；
   - 若 $L > cur$，说明出现未被任何区间覆盖的空隙，返回 `false`；
   - 否则用该区间向右扩展：若 $R + 1 > cur$，则更新 $cur = R + 1$；
   - 如果在某个时刻 $cur > n$，说明 $[1, n]$ 已被完整覆盖，返回 `true`。
3. 扫描结束后，若 $cur > n$ 则返回 `true`，否则返回 `false`。

### 二分边界

- 下界 $lo = 0$（虽然 $0$ 时刻不可能全部点亮，但方便统一写法）。
- 上界 $hi = n + m$：最坏情况下，手动点亮全部发生在最后的 $m$ 时刻，之后还需自动传播最多 $n$ 秒。
在 $[lo, hi]$ 上二分最小的可行 $T$ 即可。

## 复杂度分析

- 预排序：$O(m \log m)$。
- 每次判定：只需一趟线性扫描 $O(m)$。
- 二分次数：$\lceil \log_2(n + m + 1)\rceil \le 40$（$n \le 10^9$）。
- 总时间复杂度：$O(m \log m + m \log (n + m))$。题目保证单个测试文件内 $\sum m \le 2 \times 10^5$，可以高效通过。
- 空间复杂度：$O(m)$，主要用于存储手动点亮的路灯信息。

## 代码实现

### Python

```python
import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    t_cases = next(it)
    out_lines = []

    for _ in range(t_cases):
        n = next(it)
        m = next(it)
        a = [next(it) for _ in range(m)]

        # 预处理：按 keyL = a_t + t 排序
        # 同时存储 keyR = a_t - t，以及 t 本身
        pts = []
        for i, x in enumerate(a, start=1):
            pts.append((x + i, x - i, i, x))
        pts.sort(key=lambda v: v[0])  # 按 (a_t + t) 升序

        def ok(TT: int) -> bool:
            """判断第 TT 个时刻结束时，是否所有路灯均已点亮"""
            if TT <= 0:
                return False
            cur = 1                     # 下一个需要覆盖的位置
            for keyL, keyR, t, x in pts:
                if t > TT:             # 该路灯尚未被手动点亮
                    continue
                # 第 TT 时刻结束时，该路灯点亮的区间
                L = keyL - TT          # (a_t + t) - T
                R = keyR + TT          # (a_t - t) + T
                if R < 1 or L > n:     # 与街道无交集
                    continue
                if L < 1:
                    L = 1
                if R > n:
                    R = n
                if L > cur:            # 出现未被覆盖的空隙
                    return False
                if R + 1 > cur:
                    cur = R + 1
                if cur > n:            # 整条街已完全覆盖
                    return True
            return cur > n

        lo, hi = 0, n + m
        while lo < hi:
            mid = (lo + hi) // 2
            if ok(mid):
                hi = mid
            else:
                lo = mid + 1
        out_lines.append(str(lo))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static class Node {
        long keyL, keyR; // keyL = a_t + t, keyR = a_t - t
        int t;
        Node(long keyL, long keyR, int t) {
            this.keyL = keyL;
            this.keyR = keyR;
            this.t = t;
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder out = new StringBuilder();
        int T = Integer.parseInt(nextToken(br));
        while (T-- > 0) {
            long n = Long.parseLong(nextToken(br));
            int m = Integer.parseInt(nextToken(br));
            long[] a = new long[m + 1];
            for (int i = 1; i <= m; i++) {
                a[i] = Long.parseLong(nextToken(br));
            }

            // 预处理：按 keyL = a_t + t 排序
            ArrayList<Node> v = new ArrayList<>(m);
            for (int i = 1; i <= m; i++) {
                v.add(new Node(a[i] + i, a[i] - i, i));
            }
            v.sort(Comparator.comparingLong(o -> o.keyL));

            // 判定第 TT 秒结束时是否全亮
            java.util.function.LongPredicate ok = (long TT) -> {
                if (TT <= 0) return false;
                long cur = 1;
                for (Node nd : v) {
                    if (nd.t > TT) continue;          // 还未手动点亮
                    long L = nd.keyL - TT;            // (a_t + t) - TT
                    long R = nd.keyR + TT;            // (a_t - t) + TT
                    if (R < 1 || L > n) continue;    // 无交集
                    if (L < 1) L = 1;
                    if (R > n) R = n;
                    if (L > cur) return false;       // 出现空隙
                    if (R + 1 > cur) cur = R + 1;
                    if (cur > n) return true;        // 全覆盖
                }
                return cur > n;
            };

            long lo = 0, hi = n + m;
            while (lo < hi) {
                long mid = (lo + hi) >>> 1;
                if (ok.test(mid)) hi = mid;
                else lo = mid + 1;
            }
            out.append(lo).append('\n');
        }
        System.out.print(out.toString());
    }

    static String nextToken(BufferedReader br) throws IOException {
        int c;
        StringBuilder sb = new StringBuilder();
        while (true) {
            br.mark(1);
            c = br.read();
            if (c == -1) return sb.length() == 0 ? null : sb.toString();
            if (!Character.isWhitespace(c)) break;
        }
        while (c != -1 && !Character.isWhitespace(c)) {
            sb.append((char) c);
            br.mark(1);
            c = br.read();
        }
        return sb.toString();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while (T--) {
        int64 n;
        int m;
        cin >> n >> m;
        vector<int64> a(m + 1);
        for (int i = 1; i <= m; ++i) cin >> a[i];

        // 预处理：按 keyL = a_t + t 排序，同时存储 keyR = a_t - t
        struct Node {
            int64 keyL, keyR;
            int t;
        };
        vector<Node> v;
        v.reserve(m);
        for (int i = 1; i <= m; ++i) {
            v.push_back({a[i] + i, a[i] - i, i});
        }
        sort(v.begin(), v.end(), [](const Node& A, const Node& B) {
            return A.keyL < B.keyL;
        });

        // 判定第 TT 个时刻结束时是否全亮
        auto ok = [&](int64 TT) -> bool {
            if (TT <= 0) return false;
            int64 cur = 1;
            for (auto& nd : v) {
                if (nd.t > TT) continue;           // 尚未手动点亮
                int64 L = nd.keyL - TT;            // (a_t + t) - TT
                int64 R = nd.keyR + TT;            // (a_t - t) + TT
                if (R < 1 || L > n) continue;     // 与 [1,n] 无交
                if (L < 1) L = 1;
                if (R > n) R = n;
                if (L > cur) return false;        // 出现空档
                if (R + 1 > cur) cur = R + 1;
                if (cur > n) return true;         // 全覆盖
            }
            return cur > n;
        };

        int64 lo = 0, hi = n + m;
        while (lo < hi) {
            int64 mid = (lo + hi) >> 1;
            if (ok(mid))
                hi = mid;
            else
                lo = mid + 1;
        }
        cout << lo << "\n";
    }
    return 0;
}
```
