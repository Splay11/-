## 解题思路

我们考虑操作的性质。每次操作选择一个节点 $v$ 和一个非负整数 $x$，会将 $v$ 的子树中所有与 $v$ 距离为偶数的节点异或 $x$。  
由于异或运算满足自反律（$a \oplus a = 0$），我们可以自底向上地处理整棵树，使得每个节点通过一次操作变为 $0$。

关键观察：对于任意节点 $u$，若其深度（到根节点 $1$ 的距离）的奇偶性为 $p = d \bmod 2$，则只有那些“所在操作节点与 $u$ 距离为偶数”的操作才会影响 $u$，而这样的操作节点必须与 $u$ 同奇偶性。也就是说，所有深度奇偶性与 $u$ 相同的节点上施加的操作，都会对 $u$ 产生异或影响。因此，我们可以维护两个累积异或值：

- $cum[0]$：表示到目前为止，对所有深度为偶数的节点所施加操作的累积异或；
- $cum[1]$：表示到目前为止，对所有深度为奇数的节点所施加操作的累积异或。

当遍历到节点 $u$（深度 $d$）时，当前节点受到的历史操作总影响为 $cum[d \bmod 2]$，其真实数值为 $a_u \oplus cum[d \bmod 2]$。为了将它变为 $0$，我们必须在这时对 $u$ 执行一次操作，异或值 $x$ 必须等于这个真实数值。将 $x$ 累加到答案中，并更新 $cum[d \bmod 2] \gets cum[d \bmod 2] \oplus x$，这样子树中同奇偶性的后代节点也会受到这个操作的影响。

我们采用深度优先搜索（DFS）遍历整棵树。为避免递归栈溢出，使用显式栈模拟递归。栈中元素包含节点编号、父节点、深度、阶段标记（入栈处理/回溯撤销）以及本次施加的操作值 $x$。  
在入栈阶段，计算所需的 $x$ 并更新累积异或；处理完子节点后，在回溯阶段将累积异或恢复原值，以保证兄弟子树不受影响。

算法结束时，答案即为所有操作的 $x$ 之和。

## 复杂度分析

- 时间复杂度：每个节点仅入栈、出栈各一次，每次进行常数次异或和加法运算，因此总时间复杂度为 $O(n)$。  
- 空间复杂度：邻接表存储树需要 $O(n)$ 空间；显式栈最多存储 $2n$ 个元素，亦为 $O(n)$。因此总空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys

def solve() -> None:
    input = sys.stdin.readline
    n = int(input())
    a = [0] + list(map(int, input().split()))
    g = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)

    res = 0
    cum = [0, 0]  # cum[0] 偶深度累积异或, cum[1] 奇深度累积异或

    # 栈元素: (当前节点, 父节点, 深度, 阶段, 操作值x)
    stk = [(1, 0, 0, 0, 0)]
    while stk:
        u, p, d, stage, x = stk.pop()
        if stage == 0:
            pp = d & 1
            need = cum[pp] ^ a[u]  # 当前实际值，需要异或的x
            res += need
            cum[pp] ^= need
            # 回溯阶段入栈
            stk.append((u, p, d, 1, need))
            for v in g[u]:
                if v != p:
                    stk.append((v, u, d + 1, 0, 0))
        else:
            # 回溯，撤销当前节点对累积异或的影响
            pp = d & 1
            cum[pp] ^= x

    print(res)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        long[] a = new long[n + 1];
        StringTokenizer st = new StringTokenizer(in.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Long.parseLong(st.nextToken());
        }

        List<Integer>[] g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(in.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            g[u].add(v);
            g[v].add(u);
        }

        long res = 0;
        long[] cum = new long[2]; // cum[0] 偶深度, cum[1] 奇深度

        // 栈元素: [u, parent, depth, stage, x]
        Deque<long[]> stk = new ArrayDeque<>();
        stk.push(new long[]{1, 0, 0, 0, 0});

        while (!stk.isEmpty()) {
            long[] cur = stk.pop();
            int u = (int) cur[0];
            int p = (int) cur[1];
            int d = (int) cur[2];
            int stage = (int) cur[3];
            long x = cur[4];

            if (stage == 0) {
                int pp = d & 1;
                long need = cum[pp] ^ a[u];
                res += need;
                cum[pp] ^= need;
                // 回溯阶段
                stk.push(new long[]{u, p, d, 1, need});
                for (int v : g[u]) {
                    if (v != p) {
                        stk.push(new long[]{v, u, d + 1, 0, 0});
                    }
                }
            } else {
                int pp = d & 1;
                cum[pp] ^= x; // 撤销修改
            }
        }

        System.out.println(res);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<ll> a(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }

    vector<vector<int>> g(n + 1);
    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    ll res = 0;
    ll cum[2] = {0, 0}; // cum[0] 偶深度, cum[1] 奇深度

    struct StackFrame {
        int u, p, d, stg;
        ll x;
    };
    vector<StackFrame> stk;
    stk.reserve(2 * n);
    stk.push_back({1, 0, 0, 0, 0});

    while (!stk.empty()) {
        auto [u, p, d, stg, x] = stk.back();
        stk.pop_back();

        if (stg == 0) {
            int pp = d & 1;
            ll need = cum[pp] ^ a[u];
            res += need;
            cum[pp] ^= need;

            stk.push_back({u, p, d, 1, need});
            for (int v : g[u]) {
                if (v != p) {
                    stk.push_back({v, u, d + 1, 0, 0});
                }
            }
        } else {
            int pp = d & 1;
            cum[pp] ^= x; // 撤销操作
        }
    }

    cout << res << '\n';
    return 0;
}
```
