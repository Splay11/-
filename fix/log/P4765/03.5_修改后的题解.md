## 解题思路

设闸机状态串为 $s$，相邻权重为 $w_1,w_2,\dots,w_{n-1}$。协同收益为

$$
\sum_{i=1}^{n-1} w_i \cdot [s_i=s_{i+1}]
$$

翻转连续区间 $[l,r]$ 时：

* 区间内部相邻对两端同时翻转，相同关系不变
* 区间外部相邻对也不变
* 只有左右边界处的相邻对可能变化

因此一次翻转最多改变两个相邻对的贡献。

对每个相邻对 $i$，若它成为翻转边界：

* 原来 $s_i=s_{i+1}$，贡献变化为 $-w_i$
* 原来 $s_i\ne s_{i+1}$，贡献变化为 $+w_i$

任意两个不同相邻对都可以分别作为某区间的左右边界，所以只需在这些变化收益中取最大的两个正数（没有正数则不翻转）。

算法：

1. 计算初始协同收益 $base$
2. 同时维护最大的两个正收益 $mx_1,mx_2$（默认为 `0`）
3. 答案为 $base+mx_1+mx_2$

## 复杂度分析

时间复杂度 $O(n)$，空间复杂度 $O(1)$。

## 代码实现

### Python

```python
def solve(n, s, w):
    # 计算初始同态收益
    base = 0

    # 记录最大的两个正收益
    mx1 = 0
    mx2 = 0

    for i in range(n - 1):
        if s[i] == s[i + 1]:
            # 当前相邻对原本有贡献
            base += w[i]
            gain = -w[i]
        else:
            # 当前相邻对原本无贡献，若作为边界可新增贡献
            gain = w[i]

        # 维护最大的两个正收益
        if gain > mx1:
            mx2 = mx1
            mx1 = gain
        elif gain > mx2:
            mx2 = gain

    return base + mx1 + mx2


def main():
    # 输入
    n = int(input().strip())
    s = input().strip()
    w = list(map(int, input().split()))

    # 输出
    print(solve(n, s, w))


if __name__ == "__main__":
    main()
```

```java
import java.util.Scanner;

public class Main {

    // 计算翻转至多一次后的最大同态收益
    public static long solve(int n, String s, int[] w) {
        long base = 0L;

        // 记录最大的两个正收益
        long mx1 = 0L, mx2 = 0L;

        for (int i = 0; i < n - 1; i++) {
            long gain;
            if (s.charAt(i) == s.charAt(i + 1)) {
                // 当前相邻对原本有贡献
                base += w[i];
                gain = -w[i];
            } else {
                // 当前相邻对原本无贡献，若作为边界可新增贡献
                gain = w[i];
            }

            // 维护最大的两个正收益
            if (gain > mx1) {
                mx2 = mx1;
                mx1 = gain;
            } else if (gain > mx2) {
                mx2 = gain;
            }
        }

        return base + mx1 + mx2;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 输入
        int n = sc.nextInt();

        int[] w = new int[n - 1];
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// 计算翻转至多一次后的最大同态收益
long long solve(int n, const string& s, const vector<int>& w) {
    long long base = 0;

    // 记录最大的两个正收益
    long long mx1 = 0, mx2 = 0;

    for (int i = 0; i < n - 1; i++) {
        long long gain;
        if (s[i] == s[i + 1]) {
            // 当前相邻对原本有贡献
            base += w[i];
            gain = -1LL * w[i];
        } else {
            // 当前相邻对原本无贡献，若作为边界可新增贡献
            gain = w[i];
        }

        // 维护最大的两个正收益
        if (gain > mx1) {
            mx2 = mx1;
            mx1 = gain;
        } else if (gain > mx2) {
            mx2 = gain;
        }
    }

    return base + mx1 + mx2;
}

int main() {
    // 输入
    int n;
    cin >> n;

    string s;
    cin >> s;

    vector<int> w(n - 1);
    for (int i = 0; i < n - 1; i++) {
        cin >> w[i];
    }

    // 输出
    cout << solve(n, s, w) << '\n';

    return 0;
}
```
