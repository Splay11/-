## 解题思路

本题可以看作追捕者与小 C 在数轴上的追赶模拟。两者的运动速度均为每秒 1 个单位，但小 C 在拆除陷阱时需要原地停留一段时间，此时追捕者会继续移动。小 C 始终向终点 $L$ 方向前进，不会回头，因此只需要考虑位于小 C **当前位置及前方**的陷阱。

具体算法步骤如下：

1. **读取数据并整理陷阱**  
   将每组测试数据中的 $M$ 个陷阱的位置 $a_i$ 和对应的拆除时间 $t_i$ 配对，并按位置从小到大排序。  
   小 C 的初始位置为 $P$，追捕者初始位置为 $1$（对应时刻 $0$ 追捕者所在坐标）。

2. **依次模拟每个陷阱**  
   维护两个变量：  
   - `chaser`：当前追捕者的坐标；  
   - `agent`：小 C 当前所在的坐标。  
   遍历排序后的陷阱列表，对于每个陷阱 $\langle a,\,t \rangle$：  
   - 若 $a \ge agent$（即陷阱在小 C 的前方，才会被触发），执行以下过程：  
     - 小 C 从 `agent` 移动到 $a$ 需要时间 $a - agent$；拆除陷阱又消耗时间 $t$。  
     - 在这段时间里，追捕者同步移动，因此 `chaser` 增加 $(a - agent) + t$。  
     - 若 `chaser >= a`，说明在小 C 位于陷阱处时，追捕者已经追到或超过小 C，直接输出 `NO` 并结束当前测试数据。  
     - 否则，小 C 成功通过该陷阱，更新 `agent = a`。  
   - 若 $a < agent$，该陷阱在小 C 身后，不会遇到，直接跳过。

3. **模拟最后冲刺到终点**  
   遍历完所有陷阱后，小 C 从当前坐标 `agent` 向终点 $L$ 前进，所需时间为 $L - agent$。追捕者同样移动这段距离：  
   `chaser += L - agent`。  
   若 `chaser >= L`，说明在到达终点前已被追上，输出 `NO`；否则输出 `YES`。

该算法本质是对“小 C 停留的每一个点”做一次安全判断，因为在速度相同的情况下，只要在每个停留点及终点处追捕者没有超过小 C，那么整个过程就不会被追上。

## 复杂度分析

- **时间复杂度**：对于每组测试数据，需要对 $M$ 个陷阱进行排序，复杂度为 $O(M \log M)$；随后的模拟过程是线性扫描，复杂度为 $O(M)$。所有数据 $M$ 的总和不超过 $10^5$，因此总时间复杂度为 $O\left(\sum M \log M\right)$，可以在时限内完成。

- **空间复杂度**：需要存储 $M$ 个陷阱的位置和拆除时间，空间复杂度为 $O(M)$。

## 代码实现

### Python

```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    T = int(next(it))
    out_lines = []
    
    for _ in range(T):
        L = int(next(it))
        M = int(next(it))
        P = int(next(it))
        
        positions = [0] * M
        for i in range(M):
            positions[i] = int(next(it))
        times = [0] * M
        for i in range(M):
            times[i] = int(next(it))
            
        # 将陷阱信息组合并按位置排序
        traps = sorted(zip(positions, times), key=lambda x: x[0])
        
        chaser = 1          # 追捕者初始坐标
        agent = P           # 小 C 当前坐标
        escaped = True
        
        for pos, t in traps:
            if pos >= agent:        # 只考虑前方的陷阱
                # 追捕者同步移动：小C移动 (pos - agent) 并停留 t 秒
                chaser += (pos - agent) + t
                if chaser >= pos:   # 在陷阱处被追上
                    escaped = False
                    break
                agent = pos         # 小 C 通过陷阱
        
        if escaped:
            # 处理最后一段到终点的冲刺
            chaser += (L - agent)
            if chaser >= L:
                escaped = False
        
        out_lines.append("YES" if escaped else "NO")
    
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        StringBuilder sb = new StringBuilder();
        
        while (T-- > 0) {
            String[] lmp = br.readLine().trim().split(" ");
            long L = Long.parseLong(lmp[0]);
            int M = Integer.parseInt(lmp[1]);
            long P = Long.parseLong(lmp[2]);
            
            long[] pos = new long[M];
            String[] posStr = br.readLine().trim().split(" ");
            for (int i = 0; i < M; i++) {
                pos[i] = Long.parseLong(posStr[i]);
            }
            
            long[] time = new long[M];
            String[] timeStr = br.readLine().trim().split(" ");
            for (int i = 0; i < M; i++) {
                time[i] = Long.parseLong(timeStr[i]);
            }
            
            // 组装陷阱并按位置排序
            long[][] traps = new long[M][2];
            for (int i = 0; i < M; i++) {
                traps[i][0] = pos[i];
                traps[i][1] = time[i];
            }
            Arrays.sort(traps, Comparator.comparingLong(a -> a[0]));
            
            long chaser = 1;      // 追捕者坐标
            long agent = P;       // 小C坐标
            boolean ok = true;
            
            for (int i = 0; i < M; i++) {
                long p = traps[i][0];
                long t = traps[i][1];
                if (p >= agent) {          // 陷阱在前方
                    chaser += (p - agent) + t;
                    if (chaser >= p) {     // 在该陷阱处被追上
                        ok = false;
                        break;
                    }
                    agent = p;
                }
            }
            
            if (ok) {
                chaser += (L - agent);
                if (chaser >= L) {
                    ok = false;
                }
            }
            
            sb.append(ok ? "YES" : "NO").append("\n");
        }
        
        System.out.print(sb);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int T;
    cin >> T;
    while (T--) {
        ll L, P;
        int M;
        cin >> L >> M >> P;
        
        vector<pair<ll, ll>> traps(M);   // {位置, 拆除时间}
        for (int i = 0; i < M; ++i) {
            cin >> traps[i].first;
        }
        for (int i = 0; i < M; ++i) {
            cin >> traps[i].second;
        }
        
        // 按陷阱位置排序
        sort(traps.begin(), traps.end());
        
        ll chaser = 1;      // 追捕者坐标
        ll agent = P;       // 小C坐标
        bool escaped = true;
        
        for (auto &[pos, t] : traps) {
            if (pos >= agent) {          // 只处理前方的陷阱
                chaser += (pos - agent) + t;
                if (chaser >= pos) {     // 在陷阱处被追上
                    escaped = false;
                    break;
                }
                agent = pos;
            }
        }
        
        if (escaped) {
            chaser += (L - agent);
            if (chaser >= L) {
                escaped = false;
            }
        }
        
        cout << (escaped ? "YES" : "NO") << "\n";
    }
    
    return 0;
}
```
