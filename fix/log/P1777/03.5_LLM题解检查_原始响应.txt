## 解题思路

本题需要依次判断每个会话是否异常，异常条件有两条，只要满足任意一条即判定为异常。  
设会话的访问记录按顺序为 $c_1, c_2, \dots, c_b$，这些事件编号互不相同。

**规则一：严重性级别遗漏**

1. 扫描会话的所有访问记录，统计该会话访问过的**最大事件编号** $p$。由于严重性数组 $a$ 单调不减，该会话通过的最高严重性为 $x = a_p$（注意事件编号从 $1$ 开始）。
2. 规则要求：若访问了最高严重性为 $x$ 的事件，则所有严重性级别 **不超过** $x-500$ 的事件都必须被访问，否则异常。
3. 利用数组 $a$ 的单调性，使用二分查找（如 `upper_bound`）找到满足 $a_i \le x-500$ 的最大下标 $i$，记 $t = i$（这里 $t$ 同时也是满足条件的事件数量，且这些事件的编号为 $1 \sim t$）。  
   若 $x-500 < a_1$，则 $t=0$，表示没有任何事件的严重性低于 $x-500$，规则一自动满足。
4. 统计会话的访问记录中，事件编号 $\le t$ 的数量，设为 $cnt$。  
   - 若 $cnt = t$，说明编号为 $1 \sim t$ 的事件全部被访问，不触发规则一。  
   - 若 $cnt < t$，说明至少漏掉了一个严重性不超过 $x-500$ 的事件，**判定为异常**（规则一命中）。
5. 注意：因为同一会话内事件编号不重复，所以只需要比较数量即可，不存在重复访问的干扰。

**规则二：相邻事件编号跳变**

在扫描访问记录的同时，依次检查相邻两个访问的事件编号之差：

$$
|c_{k+1} - c_k| \ge r
$$

若存在任一相邻对满足上述条件，**直接判定为异常**（规则二命中）。

**汇总**

对每个会话：
- 首先在单次线性扫描中求出最大事件编号，并检测规则二；
- 若规则二不触发，再使用二分查找计算 $t$，并统计前缀访问数量，判断规则一；
- 任一规则触发，该会话异常计数加 $1$。

处理完所有 $m$ 个会话后，输出异常会话的总数。

## 复杂度分析

记所有会话的记录总数为 $S = \sum_{j=1}^{m} b_j$。

- 每个会话需要一次线性扫描（$O(b_j)$）和一次二分查找（$O(\log n)$），因此单个会话处理时间为 $O(b_j + \log n)$。
- 总时间复杂度为 $O(n + S + m \log n)$。根据题目限制 $S \le 2 \times 10^5$，可以高效通过。
- 空间复杂度主要存储严重性数组 $a$ 以及当前会话的访问记录，为 $O(n + \max b_j)$，满足题目要求。

## 代码实现

### Python

```python
import sys
from bisect import bisect_right

def is_abnormal(a, r, passed):
    """判断一个会话是否异常"""
    max_index = 0

    # 一次扫描：求最大事件编号，同时检查规则二（相邻编号跳变）
    for i in range(len(passed)):
        max_index = max(max_index, passed[i])
        if i > 0 and abs(passed[i] - passed[i - 1]) >= r:
            return True

    # 规则一：前缀事件是否全部被访问
    # 最高严重性为 a[max_index - 1]
    limit = a[max_index - 1] - 500
    # 严重性不超过 limit 的事件数量（即前缀长度 t）
    t = bisect_right(a, limit)

    # 统计访问记录中编号不超过 t 的事件个数
    cnt = 0
    for event in passed:
        if event <= t:
            cnt += 1

    # 如果前缀事件没有全部被访问，则违反规则一
    return cnt < t

def main():
    input = sys.stdin.buffer.readline
    n, r = map(int, input().split())
    a = list(map(int, input().split()))

    m = int(input())
    ans = 0

    for _ in range(m):
        b = int(input())
        passed = list(map(int, input().split()))
        if is_abnormal(a, r, passed):
            ans += 1

    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {

    // 二分查找：返回大于 target 的第一个位置，即 <= target 的元素个数
    static int upperBound(long[] a, long target) {
        int left = 0, right = a.length;
        while (left < right) {
            int mid = left + (right - left) / 2;
            if (a[mid] <= target)
                left = mid + 1;
            else
                right = mid;
        }
        return left;  // left 即为 <= target 的元素个数
    }

    static boolean isAbnormal(long[] a, int r, int[] passed) {
        int maxIndex = 0;

        // 一次扫描：求最大事件编号，同时检查规则二
        for (int i = 0; i < passed.length; i++) {
            maxIndex = Math.max(maxIndex, passed[i]);
            if (i > 0 && Math.abs(passed[i] - passed[i - 1]) >= r)
                return true;
        }

        // 规则一：必须访问的前缀长度 t
        long limit = a[maxIndex - 1] - 500;
        int t = upperBound(a, limit);   // 严重性 <= limit 的事件个数

        // 统计实际访问的前缀事件数量
        int cnt = 0;
        for (int event : passed) {
            if (event <= t)
                cnt++;
        }

        return cnt < t;  // 如有遗漏则异常
    }

    public static void main(String[] args) throws IOException {
        TokenReader reader = new TokenReader();
        int n = reader.nextInt();
        int r = reader.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++)
            a[i] = reader.nextLong();

        int m = reader.nextInt();
        int ans = 0;
        for (int i = 0; i < m; i++) {
            int b = reader.nextInt();
            int[] passed = new int[b];
            for (int j = 0; j < b; j++)
                passed[j] = reader.nextInt();
            if (isAbnormal(a, r, passed))
                ans++;
        }
        System.out.println(ans);
    }

    // 快速输入工具类
    static class TokenReader {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer;

        String next() throws IOException {
            while (tokenizer == null || !tokenizer.hasMoreTokens())
                tokenizer = new StringTokenizer(reader.readLine());
            return tokenizer.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        long nextLong() throws IOException {
            return Long.parseLong(next());
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cstdlib>
using namespace std;

bool isAbnormal(const vector<long long>& a, int r, const vector<int>& passed) {
    int maxIndex = 0;
    // 一次扫描：求最大事件编号，同时检查规则二
    for (int i = 0; i < (int)passed.size(); ++i) {
        maxIndex = max(maxIndex, passed[i]);
        if (i > 0 && abs(passed[i] - passed[i - 1]) >= r)
            return true;
    }

    // 规则一：必须访问的前缀长度 t
    long long limit = a[maxIndex - 1] - 500;
    int t = upper_bound(a.begin(), a.end(), limit) - a.begin(); // <= limit 的个数

    // 统计实际访问的前缀事件数量
    int cnt = 0;
    for (int event : passed)
        if (event <= t)
            cnt++;

    return cnt < t; // 有遗漏则异常
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, r;
    cin >> n >> r;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    int m;
    cin >> m;
    int ans = 0;
    for (int i = 0; i < m; ++i) {
        int b;
        cin >> b;
        vector<int> passed(b);
        for (int j = 0; j < b; ++j) cin >> passed[j];
        if (isAbnormal(a, r, passed))
            ans++;
    }
    cout << ans << '\n';
    return 0;
}
```
