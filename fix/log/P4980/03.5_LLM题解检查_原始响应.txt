## 解题思路

- **问题转化**：要求最小化子序列中最大元素的值，可以对这个最大值 $M$ 进行二分查找。  
  判定问题变为：在所有 $a_i \le M$ 的元素中，能否按原顺序选出一个长度为 $k$ 的子序列，且相邻两项的最大公约数大于 $1$。

- **特殊情况**：若 $k=1$，则无需考虑相邻关系，答案为整个序列中的最小值，可直接在二分时特判。

- **判定方法**：
  1. 预处理每个 $a_i$ 的不同质因子，将每个质因子映射到一个整数编号，压缩空间并加速查找。$a_i=1$ 没有质因子，在 $k \ge 2$ 时无法参与构成相容子序列。
  2. 从左到右动态规划：
     - 维护数组 `best[p]`，表示以含有质因子 $p$ 的元素结尾的最长相容子序列长度。
     - 对于当前元素 $a_i$（$a_i \le M$ 且 $a_i \neq 1$），设其质因子编号集合为 $F_i$，则当前元素可以接在任何一个以 $F_i$ 中质因子结尾的序列后面，得到：
       $$dp = 1 + \max_{p \in F_i} best[p]$$
       若没有前驱，则 $dp=1$。
     - 用 $dp$ 更新全局最大长度，同时对于所有 $p \in F_i$，令 $best[p] = \max(best[p], dp)$。
     - 如果全局最大长度 $\ge k$，则判定成功。
  3. 若能成功，缩小二分上界；否则提高下界。最终得到的最小可行 $M$ 即为答案。

- **预处理质因子**：  
  由于 $a_i \le 10^9$，只需试除 $\sqrt{10^9} \approx 31623$ 以内的素数。可使用筛法预先得到这些素数，加速每个元素的质因数分解。

## 复杂度分析

- 令 $\omega = \max_i \omega(a_i)$ 为每个元素的不同质因子个数，$\omega \approx 7$，可视为常数。
- 预处理质因数分解：$O(n \cdot \omega)$。
- 二分答案：值域大小 $O(\log (\max a_i))$，每次判定 DP 遍历 $n$ 个元素，每个元素处理其所有质因子，复杂度 $O(n \cdot \omega)$。总时间复杂度 $O(n \cdot \omega \cdot \log (\max a_i))$。
- 空间复杂度：存储每个元素的质因子编号，共 $O(n \cdot \omega)$；`best` 数组与质因子总数相关，同样 $O(n \cdot \omega)$。总体空间复杂度 $O(n \cdot \omega)$。

## 代码实现

### Python

```python
import sys

def sieve(limit):
    is_prime = [True] * (limit + 1)
    primes = []
    for i in range(2, limit + 1):
        if not is_prime[i]:
            continue
        primes.append(i)
        if i * i > limit:
            continue
        for j in range(i * i, limit + 1, i):
            is_prime[j] = False
    return primes

primes = sieve(32000)

def factorize(x):
    factors = []
    for p in primes:
        if p * p > x:
            break
        if x % p == 0:
            factors.append(p)
            while x % p == 0:
                x //= p
    if x > 1:
        factors.append(x)
    return factors

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    k = int(data[1])
    a = list(map(int, data[2:2 + n]))

    # 质因子映射
    pid = {}
    pc = 0
    fac_ids = [[] for _ in range(n)]
    for i, x in enumerate(a):
        if x == 1:
            continue
        for p in factorize(x):
            if p not in pid:
                pid[p] = pc
                pc += 1
            fac_ids[i].append(pid[p])

    # 去重并排序，用于二分
    vals = sorted(set(a))

    def check(M):
        if k == 1:
            return any(x <= M for x in a)
        best = [0] * pc
        mx = 0
        for i, x in enumerate(a):
            if x > M or x == 1:
                continue
            dp = 1
            for pid in fac_ids[i]:
                if best[pid] + 1 > dp:
                    dp = best[pid] + 1
            if dp > mx:
                mx = dp
            if mx >= k:
                return True
            for pid in fac_ids[i]:
                if best[pid] < dp:
                    best[pid] = dp
        return False

    lo, hi = 0, len(vals) - 1
    ans = -1
    while lo <= hi:
        mid = (lo + hi) // 2
        if check(vals[mid]):
            ans = vals[mid]
            hi = mid - 1
        else:
            lo = mid + 1
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static List<Integer> primes = new ArrayList<>();

    static void initPrimes() {
        int N = 32000;
        boolean[] isPrime = new boolean[N + 1];
        Arrays.fill(isPrime, true);
        for (int i = 2; i <= N; i++) {
            if (!isPrime[i]) continue;
            primes.add(i);
            if ((long) i * i > N) continue;
            for (int j = i * i; j <= N; j += i) {
                isPrime[j] = false;
            }
        }
    }

    static List<Integer> factorize(long x) {
        List<Integer> fac = new ArrayList<>();
        for (int p : primes) {
            if ((long) p * p > x) break;
            if (x % p == 0) {
                fac.add(p);
                while (x % p == 0) x /= p;
            }
        }
        if (x > 1) fac.add((int) x);
        return fac;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        long[] a = new long[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(st.nextToken());
        }

        initPrimes();
        Map<Integer, Integer> pidMap = new HashMap<>();
        int pc = 0;
        List<Integer>[] facIds = new List[n];
        for (int i = 0; i < n; i++) facIds[i] = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) continue;
            for (int p : factorize(a[i])) {
                Integer id = pidMap.get(p);
                if (id == null) {
                    id = pc++;
                    pidMap.put(p, id);
                }
                facIds[i].add(id);
            }
        }

        long[] vals = a.clone();
        Arrays.sort(vals);
        int idx = 0;
        for (int i = 0; i < n; i++) {
            if (i == 0 || vals[i] != vals[idx - 1]) {
                vals[idx++] = vals[i];
            }
        }
        int m = idx;

        int lo = 0, hi = m - 1;
        long ans = -1;
        while (lo <= hi) {
            int mid = (lo + hi) / 2;
            long M = vals[mid];
            boolean ok;
            if (k == 1) {
                ok = false;
                for (long x : a) {
                    if (x <= M) { ok = true; break; }
                }
            } else {
                int[] best = new int[pc];
                int mx = 0;
                ok = false;
                for (int i = 0; i < n; i++) {
                    if (a[i] > M || a[i] == 1) continue;
                    int dp = 1;
                    for (int pid : facIds[i]) {
                        if (best[pid] + 1 > dp) dp = best[pid] + 1;
                    }
                    if (dp > mx) mx = dp;
                    if (mx >= k) { ok = true; break; }
                    for (int pid : facIds[i]) {
                        if (best[pid] < dp) best[pid] = dp;
                    }
                }
            }
            if (ok) { ans = M; hi = mid - 1; }
            else lo = mid + 1;
        }
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> primes;
void init_primes() {
    const int N = 32000;
    vector<bool> is_prime(N + 1, true);
    for (int i = 2; i <= N; ++i) {
        if (!is_prime[i]) continue;
        primes.push_back(i);
        if (1LL * i * i > N) continue;
        for (int j = i * i; j <= N; j += i) {
            is_prime[j] = false;
        }
    }
}

vector<int> factorize(long long x) {
    vector<int> fac;
    for (int p : primes) {
        if (1LL * p * p > x) break;
        if (x % p == 0) {
            fac.push_back(p);
            while (x % p == 0) x /= p;
        }
    }
    if (x > 1) fac.push_back((int)x);
    return fac;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    init_primes();
    int n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    unordered_map<int, int> pid;
    int pc = 0;
    vector<vector<int>> fac_ids(n);
    for (int i = 0; i < n; ++i) {
        if (a[i] == 1) continue;
        for (int p : factorize(a[i])) {
            if (!pid.count(p)) pid[p] = pc++;
            fac_ids[i].push_back(pid[p]);
        }
    }

    auto check = [&](long long M) -> bool {
        if (k == 1) {
            for (long long x : a) if (x <= M) return true;
            return false;
        }
        vector<int> best(pc, 0);
        int mx = 0;
        for (int i = 0; i < n; ++i) {
            if (a[i] > M || a[i] == 1) continue;
            int dp = 1;
            for (int id : fac_ids[i]) dp = max(dp, best[id] + 1);
            mx = max(mx, dp);
            if (mx >= k) return true;
            for (int id : fac_ids[i]) {
                if (best[id] < dp) best[id] = dp;
            }
        }
        return false;
    };

    vector<long long> vals = a;
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());

    long long ans = -1;
    int lo = 0, hi = (int)vals.size() - 1;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        if (check(vals[mid])) {
            ans = vals[mid];
            hi = mid - 1;
        } else {
            lo = mid + 1;
        }
    }
    cout << ans << '\n';
    return 0;
}
```
