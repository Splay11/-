## 解题思路

1. **问题转化**  
   每次操作是在一个连续子段 $[l, r]$ 内任意重排元素，消耗为该子段内元素之和。  
   如果把整个序列划分成若干不相交的区间分别操作，且每个元素至多被操作一次，总消耗就是这些区间内所有元素之和。  
   设序列总和为 $S$，未被操作的元素（即保留在原位的元素）总和为 $S_{\text{keep}}$，则总消耗等于 $S - S_{\text{keep}}$。  
   因此最小化总消耗等价于最大化可保留在原位的元素之和。

2. **可保留元素的条件**  
   记目标序列为 $y$（非降序时 $y$ 为升序排列，非增序时 $y$ 为降序排列）。  
   对于位置 $i$（从 $1$ 到 $n$），若同时满足以下两个条件，则 $x_i$ 可被保留：
   - 前缀多重集合相同：$\{x_1, x_2, \dots, x_i\}$ 与 $\{y_1, y_2, \dots, y_i\}$ 作为多重集合完全一致；
   - 当前元素相等：$x_i = y_i$。  
   满足这两个条件的位置可以作为操作区间的“分割点”，使得该位置不需要参与任何操作。

3. **算法步骤**  
   - 计算 $S = \sum_{i=1}^n x_i$。  
   - 对原数组 $x$ 排序得到升序的 $y$，调用 `calc(x, y)` 计算升序下最大可保留和 $S_{\text{keep\_asc}}$。  
   - 将 $y$ 逆序得到降序的目标数组，同样调用 `calc` 得到 $S_{\text{keep\_desc}}$。  
   - 答案分别为 $S - S_{\text{keep\_asc}}$（非降序）和 $S - S_{\text{keep\_desc}}$（非增序）。

4. **前缀多重集合相同的判定**  
   使用差异计数器 `diff` 和哈希表 `cnt`。  
   - 遍历 $i=1$ 到 $n$，将 $x_i$ 加入“已出现”集合（`cnt[x]` 加 1），将 $y_i$ 从“已出现”集合中移除（`cnt[y]` 减 1）。  
   - 维护 `cnt` 中值不为 $0$ 的不同键的个数 `diff`。当 `diff == 0` 时，前缀多重集合完全相同。  
   - 若此时还满足 $x_i = y_i$，则累计 $x_i$ 到保留和。

## 复杂度分析

- 时间复杂度：排序需要 $O(n \log n)$，扫描统计需要 $O(n)$，整体 $O(n \log n)$。  
- 空间复杂度：需要存储目标数组和哈希表，$O(n)$。

## 代码实现

### Python

```python
import sys
from collections import defaultdict

def calc(x, y):
    cnt = defaultdict(int)
    diff = 0
    keep = 0
    for a, b in zip(x, y):
        # 插入 x 的前缀元素
        before = cnt[a]
        if before != 0:
            diff -= 1
        cnt[a] = before + 1
        if cnt[a] != 0:
            diff += 1

        # 移除 y 的前缀元素
        before = cnt[b]
        if before != 0:
            diff -= 1
        cnt[b] = before - 1
        if cnt[b] != 0:
            diff += 1

        if diff == 0 and a == b:
            keep += a
    return keep

def main():
    input = sys.stdin.readline
    n = int(input())
    x = list(map(int, input().split()))
    total = sum(x)

    # 非降序（升序）
    y = sorted(x)
    keep_asc = calc(x, y)

    # 非增序（降序）
    y = sorted(x, reverse=True)
    keep_desc = calc(x, y)

    print(total - keep_asc, total - keep_desc)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static long calc(long[] x, long[] y) {
        Map<Long, Integer> cnt = new HashMap<>();
        long diff = 0, keep = 0;
        for (int i = 0; i < x.length; i++) {
            long a = x[i], b = y[i];
            // 插入 a
            int before = cnt.getOrDefault(a, 0);
            if (before != 0) diff--;
            cnt.put(a, before + 1);
            if (cnt.get(a) != 0) diff++;

            // 移除 b
            before = cnt.getOrDefault(b, 0);
            if (before != 0) diff--;
            cnt.put(b, before - 1);
            if (cnt.get(b) != 0) diff++;

            if (diff == 0 && a == b) {
                keep += a;
            }
        }
        return keep;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        long[] x = new long[n];
        StringTokenizer st = new StringTokenizer(in.readLine());
        long total = 0;
        for (int i = 0; i < n; i++) {
            x[i] = Long.parseLong(st.nextToken());
            total += x[i];
        }

        // 非降序（升序）
        long[] y = x.clone();
        Arrays.sort(y);
        long keepAsc = calc(x, y);

        // 非增序（降序）
        for (int i = 0; i < n / 2; i++) {
            long tmp = y[i];
            y[i] = y[n - 1 - i];
            y[n - 1 - i] = tmp;
        }
        long keepDesc = calc(x, y);

        System.out.println((total - keepAsc) + " " + (total - keepDesc));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll calc(const vector<ll>& x, vector<ll> y) {
    int n = x.size();
    unordered_map<ll, int> cnt;
    cnt.reserve(n * 2);
    ll diff = 0, keep = 0;

    for (int i = 0; i < n; i++) {
        // 插入 x[i]
        {
            ll a = x[i];
            int before = cnt[a];
            if (before != 0) diff--;
            cnt[a] = before + 1;
            if (cnt[a] != 0) diff++;
        }
        // 移除 y[i]
        {
            ll b = y[i];
            int before = cnt[b];
            if (before != 0) diff--;
            cnt[b] = before - 1;
            if (cnt[b] != 0) diff++;
        }
        if (diff == 0 && x[i] == y[i]) {
            keep += x[i];
        }
    }
    return keep;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<ll> x(n);
    ll total = 0;
    for (int i = 0; i < n; i++) {
        cin >> x[i];
        total += x[i];
    }

    // 非降序（升序）
    vector<ll> y = x;
    sort(y.begin(), y.end());
    ll keepAsc = calc(x, y);

    // 非增序（降序）
    sort(y.begin(), y.end(), greater<ll>());
    ll keepDesc = calc(x, y);

    cout << total - keepAsc << " " << total - keepDesc << "\n";
    return 0;
}
```
