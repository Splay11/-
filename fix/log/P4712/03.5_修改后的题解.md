## 解题思路

- 操作允许交换下标为 $i$ 和 $i+d$ 的元素，因此一个元素只能在下标模 $d$ 同余的组内部移动。
- 将数组的所有下标按照 $i \bmod d$（0‑based）划分为 $d$ 个组，每一组内部的元素可以通过多次相邻交换得到任意排列顺序，相当于全排列。
- 要使最终序列的字典序最大，应采用贪心策略：对于每一组，将该组中的所有元素从大到小排序（降序），然后按照这组元素在原序列中出现的下标顺序，依次把排序后的最大值放到最靠前的位置。
- 具体实现：
  1. 遍历序列，将元素按 `i % d` 放入对应的组中。
  2. 对每个组进行降序排序。
  3. 再次遍历序列的下标 `i`，从 `i % d` 组中取出当前尚未使用的最大值填入输出序列对应位置，由于排序是从大到小，这样保证了越靠前的位置得到的元素越大，整体字典序也最大。
- 由于所有操作都在组内自洽，且排序和回填仅涉及原序列，该算法能给出字典序最大的最终序列。

## 复杂度分析

- 每个元素只会被放入分组一次、排序一次、回填一次。所有组的总元素个数为 $n$，各组排序的总复杂度为 $O(n \log n)$。  
- 总时间复杂度为 $O(n \log n)$。  
- 需要存储 $d$ 个分组和最终结果，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys


def solve_case(n, d, arr):
    groups = [[] for _ in range(d)]

    # 按下标模 d 分组
    for i in range(n):
        groups[i % d].append(arr[i])

    # 每组降序排序
    for i in range(d):
        groups[i].sort(reverse=True)

    idx = [0] * d
    res = [0] * n

    # 按原位置顺序回填，每次放当前组最大的剩余元素
    for i in range(n):
        g = i % d
        res[i] = groups[g][idx[g]]
        idx[g] += 1

    return res


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    p = 1
    ans = []

    for _ in range(t):
        n = data[p]
        d = data[p + 1]
        p += 2
        arr = data[p:p + n]
        p += n

        res = solve_case(n, d, arr)
        ans.append(" ".join(map(str, res)))

    sys.stdout.write("\n".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static int[] solveCase(int n, int d, int[] arr) {
        ArrayList<Integer>[] groups = new ArrayList[d];
        for (int i = 0; i < d; i++) {
            groups[i] = new ArrayList<>();
        }

        // 按下标模 d 分组
        for (int i = 0; i < n; i++) {
            groups[i % d].add(arr[i]);
        }

        // 每组降序排序
        for (int i = 0; i < d; i++) {
            groups[i].sort(Collections.reverseOrder());
        }

        int[] idx = new int[d];
        int[] res = new int[n];

        // 按位置顺序回填
        for (int i = 0; i < n; i++) {
            int g = i % d;
            res[i] = groups[g].get(idx[g]);
            idx[g]++;
        }

        return res;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        StringBuilder sb = new StringBuilder();

        int t = fs.nextInt();
        while (t-- > 0) {
            int n = fs.nextInt();
            int d = fs.nextInt();

            int[] arr = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = fs.nextInt();
            }

            int[] res = solveCase(n, d, arr);

            for (int i = 0; i < n; i++) {
                if (i > 0) sb.append(' ');
                sb.append(res[i]);
            }
            sb.append('\n');
        }

        System.out.print(sb.toString());
    }

    // 快速输入
    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        FastScanner(InputStream is) {
            in = is;
        }

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        int nextInt() throws IOException {
            int c;
            do {
                c = read();
            } while (c <= ' ');

            int sign = 1;
            if (c == '-') {
                sign = -1;
                c = read();
            }

            int val = 0;
            while (c > ' ') {
                val = val * 10 + c - '0';
                c = read();
            }
            return val * sign;
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;


vector<long long> solve_case(int n, int d, const vector<long long>& arr) {
    vector<vector<long long>> groups(d);

    // 按下标模 d 分组
    for (int i = 0; i < n; i++) {
        groups[i % d].push_back(arr[i]);
    }

    // 每组降序排序
    for (int i = 0; i < d; i++) {
        sort(groups[i].begin(), groups[i].end(), greater<long long>());
    }

    vector<int> idx(d, 0);
    vector<long long> res(n);

    // 按位置顺序回填
    for (int i = 0; i < n; i++) {
        int g = i % d;
        res[i] = groups[g][idx[g]];
        idx[g]++;
    }

    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, d;
        cin >> n >> d;

        vector<long long> arr(n);
        for (int i = 0; i < n; i++) {
            cin >> arr[i];
        }

        vector<long long> res = solve_case(n, d, arr);

        for (int i = 0; i < n; i++) {
            if (i) cout << ' ';
            cout << res[i];
        }
        cout << '\n';
    }

    return 0;
}
```
