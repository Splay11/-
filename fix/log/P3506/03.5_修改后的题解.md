## 解题思路

定义函数 $f(x) = \left(\sum_{s_i \ge x} c_i\right) - x$。题目要求的是最大的非负整数 $x$，使得 $f(x) \ge 0$。

**算法步骤：**

1. **理解单调性**  
   随着 $x$ 增大，满足 $s_i \ge x$ 的项目集合只会变小或不变，因此贡献值总和 $\sum c_i$ 单调不增，而 $x$ 单调增加，故 $f(x)$ 单调递减。这表明最优解可以通过贪心或排序方式求出。

2. **排序**  
   将所有项目按照得分 $s_i$ 从大到小排序，得到序列 $(s'_1, c'_1), (s'_2, c'_2), \dots, (s'_n, c'_n)$，其中 $s'_1 \ge s'_2 \ge \cdots \ge s'_n$。

3. **遍历与计算候选阈值**  
   从排好序的序列头部开始遍历，维护一个累计贡献和 `current_sum`。对于当前项目 $(s'_i, c'_i)$：
   - 将 $c'_i$ 累加到 `current_sum` 中。
   - 此时，前 $i$ 个项目的得分均不小于 $s'_i$，因此对于任意 $x \le s'_i$，这 $i$ 个项目都会对求和做出贡献，贡献总和至少为 `current_sum`。
   - 可选的 $x$ 必须同时满足 $x \le s'_i$ 和 $x \le \text{current\_sum}$，因此取两者的较小值 $\min(s'_i, \text{current\_sum})$ 作为当前步骤的候选阈值。
   - 用该候选阈值更新历史最大值。

4. **输出答案**  
   遍历结束后，记录的最大候选阈值即为可能的最大杰出贡献阈值。

**补充说明**：  
排序后从前向后遍历等价于按照阈值递减的顺序不断扩充项目集，每一步都能直接得到在该得分条件下可达到的最高阈值。该方法避免了二分查找，直观且高效。

## 复杂度分析

- **时间复杂度**：排序占用 $O(n \log n)$，遍历占用 $O(n)$，总时间复杂度 $O(n \log n)$。  
- **空间复杂度**：存储 $n$ 个项目需要 $O(n)$ 的额外空间，因此空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys

def solve():
    try:
        n = int(sys.stdin.readline())
        s = list(map(int, sys.stdin.readline().split()))
        c = list(map(int, sys.stdin.readline().split()))
    except (IOError, ValueError):
        return
    
    # 将得分和贡献值打包，按得分从大到小排序
    projects = list(zip(s, c))
    projects.sort(key=lambda x: x[0], reverse=True)
    
    max_threshold = 0
    current_sum = 0
    
    for score, contribution in projects:
        current_sum += contribution
        # 候选阈值必须不超过当前得分和当前贡献总和
        candidate = min(score, current_sum)
        if candidate > max_threshold:
            max_threshold = candidate
            
    print(max_threshold)

def main():
    try:
        T = int(sys.stdin.readline())
        for _ in range(T):
            solve()
    except (IOError, ValueError):
        return

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {

    static class Project {
        long score;
        long contribution;

        Project(long score, long contribution) {
            this.score = score;
            this.contribution = contribution;
        }
    }

    private static void solve(BufferedReader br) throws IOException {
        int n = Integer.parseInt(br.readLine());
        Project[] projects = new Project[n];

        StringTokenizer scoreSt = new StringTokenizer(br.readLine());
        StringTokenizer contSt = new StringTokenizer(br.readLine());

        for (int i = 0; i < n; i++) {
            long s = Long.parseLong(scoreSt.nextToken());
            long c = Long.parseLong(contSt.nextToken());
            projects[i] = new Project(s, c);
        }

        // 按得分从大到小排序
        Arrays.sort(projects, (p1, p2) -> Long.compare(p2.score, p1.score));

        long maxThreshold = 0;
        long currentSum = 0;

        for (int i = 0; i < n; i++) {
            currentSum += projects[i].contribution;
            long candidate = Math.min(projects[i].score, currentSum);
            if (candidate > maxThreshold) {
                maxThreshold = candidate;
            }
        }

        System.out.println(maxThreshold);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        while (T-- > 0) {
            solve(br);
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Project {
    long long score;
    long long contribution;
};

// 按得分降序排序
bool compareProjects(const Project& p1, const Project& p2) {
    return p1.score > p2.score;
}

void solve() {
    int n;
    cin >> n;
    vector<Project> projects(n);
    for (int i = 0; i < n; ++i) {
        cin >> projects[i].score;
    }
    for (int i = 0; i < n; ++i) {
        cin >> projects[i].contribution;
    }

    sort(projects.begin(), projects.end(), compareProjects);

    long long maxThreshold = 0;
    long long currentSum = 0;

    for (int i = 0; i < n; ++i) {
        currentSum += projects[i].contribution;
        long long candidate = min(projects[i].score, currentSum);
        if (candidate > maxThreshold) {
            maxThreshold = candidate;
        }
    }

    cout << maxThreshold << endl;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int T;
    cin >> T;
    while (T--) {
        solve();
    }

    return 0;
}
```
