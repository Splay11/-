## 解题思路

对每个路口维护到急救站的最短 $K$ 个距离。

- 自底向上计算 `down[v]`：子树内急救站对 $v$ 的前 $K$ 小距离（边权加上后与自身是否急救站合并）。
- 自顶向下计算 `up[v]`：来自父侧及其它兄弟子树的前 $K$ 小距离。对每个节点用前缀/后缀合并，避免枚举“去掉某一个儿子”时平方复杂度。
- $F(v)$ 等于把 `down` 的全部子贡献、`up[v]` 以及自身急救站贡献（距离 `0`）合并后前 $K$ 小之和。

有序数组合并每次 $O(K)$。树上任意两点路径唯一，子树侧与父侧覆盖全部急救站。

## 复杂度分析

时间复杂度 $O(nK)$，空间复杂度 $O(nK)$。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(1 << 25)
input = sys.stdin.readline

def mergeK(a, b, K):
	# 合并两个有序数组，取前K小
	res = []
	i = j = 0
	na, nb = len(a), len(b)
	while len(res) < K and (i < na or j < nb):
		if j >= nb or (i < na and a[i] <= b[j]):
			res.append(a[i]); i += 1
		else:
			res.append(b[j]); j += 1
	return res

def addWeight(a, w, K):
	t = min(len(a), K)
	return [a[i] + w for i in range(t)]

n, m, K = map(int, input().split())
is_store = [False] * (n + 1)
stores = list(map(int, input().split()))
for x in stores:
	is_store[x] = True

g = [[] for _ in range(n + 1)]
for _ in range(n - 1):
	u, v, w = map(int, input().split())
	g[u].append((v, w))
	g[v].append((u, w))

root = 1
parent = [0] * (n + 1)
pw = [0] * (n + 1)
order = []
st = [root]
parent[root] = 0
while st:
	v = st.pop()
	order.append(v)
	for to, w in g[v]:
		if to == parent[v]: continue
		parent[to] = v
		pw[to] = w
		st.append(to)

down = [[] for _ in range(n + 1)]
for v in reversed(order):
	cur = []
	if is_store[v]:
		cur = mergeK(cur, [0], K)
	for to, w in g[v]:
		if to == parent[v]: continue
		childAdd = addWeight(down[to], w, K)
		cur = mergeK(cur, childAdd, K)
	down[v] = cur

up = [[] for _ in range(n + 1)]
ans = [0] * (n + 1)

for v in order:
	children = []
	ws = []
	for to, w in g[v]:
		if to == parent[v]: continue
		children.append(to)
		ws.append(w)
	d = len(children)
	contrib = []
	for i in range(d):
		contrib.append(addWeight(down[children[i]], ws[i], K))
	pref = [None] * d
	suff = [None] * d
	for i in range(d):
		if i == 0: pref[i] = contrib[i]
		else: pref[i] = mergeK(pref[i-1], contrib[i], K)
	for i in range(d - 1, -1, -1):
		if i == d - 1: suff[i] = contrib[i]
		else: suff[i] = mergeK(contrib[i], suff[i+1], K)

	base = up[v][:]
	if is_store[v]:
		base = mergeK(base, [0], K)

	allChildren = pref[d-1] if d > 0 else []
	allAtV = mergeK(base, allChildren, K)
	ans[v] = sum(allAtV[:K])

	for i in range(d):
		if d == 1:
			others = []
		elif i == 0:
			others = suff[1]
		elif i == d - 1:
			others = pref[d-2]
		else:
			others = mergeK(pref[i-1], suff[i+1], K)
		cand = mergeK(base, others, K)
		up[children[i]] = addWeight(cand, ws[i], K)

print(' '.join(str(ans[i]) for i in range(1, n + 1)))
```

### Java

```java
import java.io.*;
import java.util.*;

/*
思路同C++：两次合并（down 自底向上 + up 自顶向下），
每次合并仅保留有序的前K小。用前缀/后缀合并加速“去掉某子”的场景。
*/
public class Main {
	static class FastScanner {
		BufferedInputStream in;
		byte[] buffer = new byte[1 << 16];
		int ptr = 0, len = 0;
		FastScanner(InputStream is) { in = new BufferedInputStream(is); }
		int read() throws IOException {
			if (ptr >= len) {
				len = in.read(buffer);
				ptr = 0;
				if (len <= 0) return -1;
			}
			return buffer[ptr++];
		}
		int nextInt() throws IOException {
			int c, sgn = 1, x = 0;
			do { c = read(); } while (c <= ' ');
			if (c == '-') { sgn = -1; c = read(); }
			for (; c > ' '; c = read()) x = x * 10 + (c - '0');
			return x * sgn;
		}
	}
	static int n, m, K;
	static ArrayList<int[]>[] g;
	static boolean[] isStore;

	static ArrayList<Long> mergeK(ArrayList<Long> a, ArrayList<Long> b) {
		ArrayList<Long> res = new ArrayList<>(Math.min(K, a.size() + b.size()));
		int i = 0, j = 0;
		while (res.size() < K && (i < a.size() || j < b.size())) {
			if (j >= b.size() || (i < a.size() && a.get(i) <= b.get(j))) {
				res.add(a.get(i++));
			} else {
				res.add(b.get(j++));
			}
		}
		return res;
	}
	static ArrayList<Long> addWeight(List<Long> a, long w) {
		int t = Math.min(a.size(), K);
		ArrayList<Long> res = new ArrayList<>(t);
		for (int i = 0; i < t; ++i) res.add(a.get(i) + w);
		return res;
	}

	public static void main(String[] args) throws Exception {
		FastScanner fs = new FastScanner(System.in);
		n = fs.nextInt(); m = fs.nextInt(); K = fs.nextInt();
		isStore = new boolean[n + 1];
		for (int i = 0; i < m; ++i) {
			int x = fs.nextInt();
			isStore[x] = true;
		}
		g = new ArrayList[n + 1];
		for (int i = 1; i <= n; ++i) g[i] = new ArrayList<>();
		for (int i = 0; i < n - 1; ++i) {
			int u = fs.nextInt(), v = fs.nextInt(), w = fs.nextInt();
			g[u].add(new int[]{v, w});
			g[v].add(new int[]{u, w});
		}

		int root = 1;
		int[] parent = new int[n + 1];
		int[] pw = new int[n + 1];
		int[] order = new int[n];
		{
			int sz = 0;
			int[] st = new int[n];
			st[sz++] = root;
			parent[root] = 0;
			int idx = 0;
			while (sz > 0) {
				int v = st[--sz];
				order[idx++] = v;
				for (int[] e : g[v]) {
					int to = e[0];
					if (to == parent[v]) continue;
					parent[to] = v;
					pw[to] = e[1];
					st[sz++] = to;
				}
			}
		}

		ArrayList<Long>[] down = new ArrayList[n + 1];
		for (int i = 1; i <= n; ++i) down[i] = new ArrayList<>();
		for (int idx = n - 1; idx >= 0; --idx) {
			int v = order[idx];
			ArrayList<Long> cur = new ArrayList<>();
			if (isStore[v]) {
				ArrayList<Long> self = new ArrayList<>();
				self.add(0L);
				cur = mergeK(cur, self);
			}
			for (int[] e : g[v]) {
				int to = e[0], w = e[1];
				if (to == parent[v]) continue;
				ArrayList<Long> childAdd = addWeight(down[to], w);
				cur = mergeK(cur, childAdd);
			}
			down[v] = cur;
		}

		ArrayList<Long>[] up = new ArrayList[n + 1];
		for (int i = 1; i <= n; ++i) up[i] = new ArrayList<>();
		long[] ans = new long[n + 1];

		for (int idx = 0; idx < n; ++idx) {
			int v = order[idx];
			ArrayList<Integer> childs = new ArrayList<>();
			ArrayList<Integer> ws = new ArrayList<>();
			for (int[] e : g[v]) {
				int to = e[0], w = e[1];
				if (to == parent[v]) continue;
				childs.add(to);
				ws.add(w);
			}
			int d = childs.size();
			ArrayList<Long>[] contrib = new ArrayList[d];
			for (int i = 0; i < d; ++i) {
				contrib[i] = addWeight(down[childs.get(i)], ws.get(i));
			}
			ArrayList<Long>[] pref = new ArrayList[d];
			ArrayList<Long>[] suff = new ArrayList[d];
			for (int i = 0; i < d; ++i) {
				if (i == 0) pref[i] = contrib[i];
				else pref[i] = mergeK(pref[i - 1], contrib[i]);
			}
			for (int i = d - 1; i >= 0; --i) {
				if (i == d - 1) suff[i] = contrib[i];
				else suff[i] = mergeK(contrib[i], suff[i + 1]);
			}

			ArrayList<Long> base = new ArrayList<>(up[v]);
			if (isStore[v]) {
				ArrayList<Long> self = new ArrayList<>();
				self.add(0L);
				base = mergeK(base, self);
			}
			ArrayList<Long> allChildren = d > 0 ? pref[d - 1] : new ArrayList<>();
			ArrayList<Long> allAtV = mergeK(base, allChildren);
			long sum = 0;
			for (int i = 0; i < Math.min(K, allAtV.size()); ++i) sum += allAtV.get(i);
			ans[v] = sum;

			for (int i = 0; i < d; ++i) {
				ArrayList<Long> others;
				if (d == 1) {
					others = new ArrayList<>();
				} else if (i == 0) {
					others = suff[1];
				} else if (i == d - 1) {
					others = pref[d - 2];
				} else {
					others = mergeK(pref[i - 1], suff[i + 1]);
				}
				ArrayList<Long> cand = mergeK(base, others);
				up[childs.get(i)] = addWeight(cand, ws.get(i));
			}
		}

		StringBuilder sb = new StringBuilder();
		for (int i = 1; i <= n; ++i) {
			if (i > 1) sb.append(' ');
			sb.append(ans[i]);
		}
		sb.append('\n');
		System.out.print(sb.toString());
	}
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

static inline vector<ll> mergeK(const vector<ll>& a, const vector<ll>& b, int K) {
	vector<ll> res;
	res.reserve(min(K, (int)a.size() + (int)b.size()));
	int i = 0, j = 0;
	while ((int)res.size() < K && (i < (int)a.size() || j < (int)b.size())) {
		if (j >= (int)b.size() || (i < (int)a.size() && a[i] <= b[j])) {
			res.push_back(a[i++]);
		} else {
			res.push_back(b[j++]);
		}
	}
	return res;
}

static inline vector<ll> addWeight(const vector<ll>& a, ll w, int K) {
	int take = min((int)a.size(), K);
	vector<ll> res;
	res.reserve(take);
	for (int i = 0; i < take; ++i) res.push_back(a[i] + w);
	return res;
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int n, m, K;
	if (!(cin >> n >> m >> K)) return 0;
	vector<char> isStore(n + 1, 0);
	for (int i = 0; i < m; ++i) {
		int x; cin >> x;
		isStore[x] = 1;
	}
	vector<vector<pair<int,int>>> g(n + 1);
	for (int i = 0; i < n - 1; ++i) {
		int u, v; int w;
		cin >> u >> v >> w;
		g[u].push_back({v, w});
		g[v].push_back({u, w});
	}

	// 预处理 parent 和遍历顺序（root=1）
	int root = 1;
	vector<int> parent(n + 1, 0);
	vector<int> parentW(n + 1, 0);
	vector<int> order; order.reserve(n);
	{
		vector<int> st; st.reserve(n);
		st.push_back(root);
		parent[root] = 0;
		while (!st.empty()) {
			int v = st.back(); st.pop_back();
			order.push_back(v);
			for (auto [to, w] : g[v]) if (to != parent[v]) {
				parent[to] = v;
				parentW[to] = w;
				st.push_back(to);
			}
		}
	}

	// 后序计算 down[v]
	vector<vector<ll>> down(n + 1);
	for (int idx = (int)order.size() - 1; idx >= 0; --idx) {
		int v = order[idx];
		vector<ll> cur; // 升序，至多K
		if (isStore[v]) {
			cur = mergeK(cur, vector<ll>{0}, K);
		}
		for (auto [to, w] : g[v]) {
			if (to == parent[v]) continue;
			vector<ll> childAdd = addWeight(down[to], w, K);
			cur = mergeK(cur, childAdd, K);
		}
		down[v].swap(cur);
	}

	// 前序计算 up[v] 并得到答案
	vector<vector<ll>> up(n + 1);
	vector<ll> ans(n + 1, 0);
	// root 的 up 为空
	up[root].clear();

	for (int v : order) {
		// 预计算每个子节点的贡献（down[child] + 边权）
		vector<int> childs;
		vector<int> ws;
		for (auto [to, w] : g[v]) if (to != parent[v]) {
			childs.push_back(to);
			ws.push_back(w);
		}
		int d = (int)childs.size();
		vector<vector<ll>> contrib(d);
		for (int i = 0; i < d; ++i) {
			contrib[i] = addWeight(down[childs[i]], ws[i], K);
		}
		// 前缀/后缀合并
		vector<vector<ll>> pref(d), suff(d);
		for (int i = 0; i < d; ++i) {
			if (i == 0) pref[i] = contrib[i];
			else pref[i] = mergeK(pref[i-1], contrib[i], K);
		}
		for (int i = d - 1; i >= 0; --i) {
			if (i == d - 1) suff[i] = contrib[i];
			else suff[i] = mergeK(contrib[i], suff[i+1], K);
		}

		// 基础候选：来自父侧 up[v] 以及 v 自身若为仓储节点的0
		vector<ll> base = up[v];
		if (isStore[v]) base = mergeK(base, vector<ll>{0}, K);

		// 所有子贡献合并
		vector<ll> allChildren;
		if (d > 0) allChildren = pref[d-1];

		// 计算 F(v)
		vector<ll> allAtV = mergeK(base, allChildren, K);
		ll sum = 0;
		for (int i = 0; i < (int)allAtV.size() && i < K; ++i) sum += allAtV[i];
		ans[v] = sum;

		// 传递给每个子节点的 up[child]
		for (int i = 0; i < d; ++i) {
			vector<ll> others;
			if (i == 0) {
				if (d >= 2) others = suff[1];
			} else if (i == d - 1) {
				if (d >= 2) others = pref[d-2];
			} else {
				others = mergeK(pref[i-1], suff[i+1], K);
			}
			vector<ll> cand = mergeK(base, others, K);
			up[childs[i]] = addWeight(cand, ws[i], K);
		}
	}

	// 输出
	for (int i = 1; i <= n; ++i) {
		if (i > 1) cout << ' ';
		cout << ans[i];
	}
	cout << '\n';
	return 0;
}
```
