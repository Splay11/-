## 解题思路

用哈希表维护二元组异或值。把四条报文校验码组成的四元组 $(i,j,p,q)$ 拆成左右两对：

- 左侧 $(i,j)$ 满足 $i < j < p$；
- 右侧 $(p,q)$ 满足 $p < q$。

条件 $a_i \oplus a_j \oplus a_p \oplus a_q = k$ 等价于

$$
a_i \oplus a_j = k \oplus a_p \oplus a_q.
$$

从左到右枚举第三个下标 $p$：

1. 哈希表 $cnt$ 维护所有 $i < j < p$ 的二元组异或 $a_i \oplus a_j$ 的出现次数；
2. 枚举 $q > p$，令 $need = k \oplus a_p \oplus a_q$，把 $cnt[need]$ 加入答案；
3. 处理完当前 $p$ 后，把所有 $(i,p)$ 插入哈希表。

表中只保存 $j < p$ 的二元组，因此四个下标天然满足 $i < j < p < q$。

## 复杂度分析

单组时间复杂度 $O(n^2)$，空间复杂度 $O(n^2)$。在 $\sum n \le 4 \times 10^3$ 下可通过。

## 代码实现

### Python

```python
import sys


def count_quadruples(a, k):
    n = len(a)
    cnt = {}
    ans = 0

    # 枚举第三个下标 p
    for p in range(n):
        # 枚举第四个下标 q
        for q in range(p + 1, n):
            need = k ^ a[p] ^ a[q]
            ans += cnt.get(need, 0)

        # 将所有 (i, p) 加入哈希表
        for i in range(p):
            x = a[i] ^ a[p]
            cnt[x] = cnt.get(x, 0) + 1

    return ans


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    idx = 1
    res = []

    for _ in range(t):
        n = data[idx]
        k = data[idx + 1]
        idx += 2

        a = data[idx:idx + n]
        idx += n

        res.append(str(count_quadruples(a, k)))

    print("\n".join(res))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

    static long countQuadruples(int[] a, int k) {
        int n = a.length;
        Map<Integer, Long> cnt = new HashMap<>();
        long ans = 0;

        for (int p = 0; p < n; p++) {
            for (int q = p + 1; q < n; q++) {
                int need = k ^ a[p] ^ a[q];
                ans += cnt.getOrDefault(need, 0L);
            }

            for (int i = 0; i < p; i++) {
                int x = a[i] ^ a[p];
                cnt.put(x, cnt.getOrDefault(x, 0L) + 1);
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();
        StringBuilder sb = new StringBuilder();

        for (int tc = 0; tc < T; tc++) {
            int n = sc.nextInt();
            int k = sc.nextInt();

            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }

            sb.append(countQuadruples(a, k)).append('\n');
        }

        System.out.print(sb.toString());
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

long long countQuadruples(vector<int>& a, int k) {
    int n = a.size();
    unordered_map<int, long long> cnt;
    cnt.reserve(n * n);

    long long ans = 0;

    for (int p = 0; p < n; p++) {
        for (int q = p + 1; q < n; q++) {
            int need = k ^ a[p] ^ a[q];
            if (cnt.count(need)) {
                ans += cnt[need];
            }
        }

        for (int i = 0; i < p; i++) {
            int x = a[i] ^ a[p];
            cnt[x]++;
        }
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, k;
        cin >> n >> k;

        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }

        cout << countQuadruples(a, k) << '\n';
    }

    return 0;
}
```
