## 解题思路

$n$ 不超过 $3\times 10^3$，可以直接模拟。用数组 $a[1..n]$ 保存当前序列，并用布尔表 $f[i][j]$ 记录编号 $i$ 是否在位置 $j$ 出现过。初始时 $f[i][i]=\mathrm{true}$。

对每次操作 $[l,r]$：把 $a[l..r]$ 暂存，再把 $a[r+1..n]$ 依次前移到从 $l$ 开始的位置，最后把暂存片段接到末尾。这等价于把区间 $[l,r]$ 挪到当前序列尾部。因为只有位置 $l$ 到 $n$ 可能变化，更新 $f$ 时只需扫描这一段。

全部操作结束后，对每个编号统计 $f[i][1..n]$ 中真值的个数。

## 复杂度分析

每次操作移动并更新 $O(n)$ 个位置，共 $q$ 次。

- 时间复杂度：$O(n^2+nq)$。
- 空间复杂度：$O(n^2)$。

## 代码实现

### Python

```python
n, q = map(int, input().split())

a = [i for i in range(n + 1)]
f = [[False] * (n + 1) for _ in range(n + 1)]

for i in range(1, n + 1):
    f[i][i] = True

b = [0] * (n + 1)

for _ in range(q):
    l, r = map(int, input().split())

    for i in range(l, r + 1):
        b[i] = a[i]

    j = l
    for i in range(r + 1, n + 1):
        a[j] = a[i]
        j += 1

    for i in range(l, n + 1):
        a[j] = b[i]
        j += 1
        if j > n:
            break

    for i in range(l, n + 1):
        f[a[i]][i] = True

result = []
for i in range(1, n + 1):
    ans = sum(f[i][j] for j in range(1, n + 1))
    result.append(str(ans))

print(" ".join(result))
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n + 1];
        boolean[][] f = new boolean[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = i;
            f[i][i] = true;
        }
        int[] b = new int[n + 1];
        for (int t = 0; t < q; t++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            for (int i = l; i <= r; i++) b[i] = a[i];
            int j = l;
            for (int i = r + 1; i <= n; i++) {
                a[j] = a[i];
                j++;
            }
            for (int i = l; i <= n; i++) {
                a[j] = b[i];
                j++;
                if (j > n) break;
            }
            for (int i = l; i <= n; i++) {
                f[a[i]][i] = true;
            }
        }
        StringJoiner sj = new StringJoiner(" ");
        for (int i = 1; i <= n; i++) {
            int ans = 0;
            for (int j = 1; j <= n; j++) if (f[i][j]) ans++;
            sj.add(Integer.toString(ans));
        }
        System.out.println(sj);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    cin >> n >> q;
    vector<int> a(n + 1), b(n + 1);
    vector<vector<char>> f(n + 1, vector<char>(n + 1, 0));
    for (int i = 1; i <= n; i++) {
        a[i] = i;
        f[i][i] = 1;
    }
    while (q--) {
        int l, r;
        cin >> l >> r;
        for (int i = l; i <= r; i++) b[i] = a[i];
        int j = l;
        for (int i = r + 1; i <= n; i++) {
            a[j] = a[i];
            j++;
        }
        for (int i = l; i <= n; i++) {
            a[j] = b[i];
            j++;
            if (j > n) break;
        }
        for (int i = l; i <= n; i++) f[a[i]][i] = 1;
    }
    for (int i = 1; i <= n; i++) {
        int ans = 0;
        for (int j = 1; j <= n; j++) ans += f[i][j];
        if (i > 1) cout << ' ';
        cout << ans;
    }
    cout << '\n';
    return 0;
}
```
