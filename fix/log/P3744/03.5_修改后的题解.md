## 解题思路

1. 将给定的表达式视为一个关于 $t$ 的多项式：
   $$P(t)=\prod_{i=0}^{n-1}(t+a_i)$$
   其中 $n$ 为因子的个数，$a_i$ 是每个因子里的常数项。对于因子 $(t+d)$，$a_i = d$；对于因子 $(t-d)$，$a_i = -d$，$d\in\{1,\dots,9\}$。

2. 设完全展开后 $t$ 的一次项系数为 $c_1$，则有：
   $$c_1 = P'(0)$$
   因为 $P(t)=c_0+c_1t+c_2t^2+\cdots$，求导得 $P'(t)=c_1+2c_2t+\cdots$，代入 $t=0$ 即得 $c_1$。

3. 利用乘积求导法则：
   $$P'(t)=\sum_{i=0}^{n-1}\;\prod_{j\neq i}(t+a_j)$$
   因此
   $$c_1 = P'(0) = \sum_{i=0}^{n-1}\;\prod_{j\neq i} a_j$$
   问题转化为：对于所有 $i$，求去掉 $a_i$ 后剩下 $n-1$ 个数的乘积，再将这些乘积求和。

4. 使用前后缀积在线性时间内完成计算：
   - 前缀积：$L[i] = a_0\cdot a_1\cdots a_{i-1} \bmod 10007$，规定 $L[0]=1$。
   - 后缀积：$R[i] = a_{i+1}\cdot a_{i+2}\cdots a_{n-1} \bmod 10007$，规定 $R[n-1]=1$。
   则 $\prod_{j\neq i} a_j \equiv L[i] \times R[i] \pmod{10007}$。
   
5. 最终答案即为：
   $$\text{ans} = \sum_{i=0}^{n-1} \big(L[i]\times R[i]\big) \bmod 10007$$

6. 解析输入字符串：输入字符串由若干个长度为 $5$ 的子串（如 `(t+3)`）拼接而成，总长度为 $5n$。对于第 $k$ 个因子（$0\le k<n$）：
   - 起始下标为 $5k$；
   - 下标 $5k+2$ 处是符号 `'+'` 或 `'-'`；
   - 下标 $5k+3$ 处是一位数字 `1` 到 `9`。
   由此可得 $a_k = \pm d$。为便于模运算，将其映射到 $[0,10006]$ 范围：`((val % MOD) + MOD) % MOD`。

7. 最后输出 $\text{ans} \bmod 10007$。

## 复杂度分析

- 时间复杂度：$O(n)$，其中 $n$ 为因子个数，也是字符串长度的 $1/5$。解析、前缀积、后缀积、累加均为单层循环。
- 空间复杂度：$O(n)$，需要存储常数数组 $a$ 以及前后缀积数组 $L,R$，每个长度均为 $n$。

## 代码实现

### Python

```python
import sys

MOD = 10007

def main():
    s = sys.stdin.readline().strip()
    n = len(s) // 5          # 因子个数
    a = []

    # 解析每个因子 (t+d) 或 (t-d)
    for k in range(n):
        base = 5 * k
        op = s[base + 2]                 # '+' 或 '-'
        d = ord(s[base + 3]) - ord('0')  # 数字 1..9
        val = -d if op == '-' else d
        # 映射到 [0, MOD) 区间
        a.append((val % MOD + MOD) % MOD)

    # 前缀积 L[i] = a[0] * ... * a[i-1] (mod MOD), L[0]=1
    L = [1] * n
    for i in range(1, n):
        L[i] = (L[i - 1] * a[i - 1]) % MOD

    # 后缀积 R[i] = a[i+1] * ... * a[n-1] (mod MOD), R[n-1]=1
    R = [1] * n
    for i in range(n - 2, -1, -1):
        R[i] = (R[i + 1] * a[i + 1]) % MOD

    # 求和
    ans = 0
    for i in range(n):
        ans = (ans + L[i] * R[i]) % MOD

    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;

public class Main {
    static final int MOD = 10007;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int n = s.length() / 5;   // 因子个数

        int[] a = new int[n];
        for (int k = 0; k < n; k++) {
            int base = 5 * k;
            char op = s.charAt(base + 2);        // '+' 或 '-'
            int d = s.charAt(base + 3) - '0';    // 数字 1..9
            int val = (op == '-') ? -d : d;
            // 映射到 [0, MOD) 区间
            a[k] = ((val % MOD) + MOD) % MOD;
        }

        // 前缀积 L[i] = a[0] * ... * a[i-1] (mod MOD), L[0]=1
        int[] L = new int[n];
        L[0] = 1;
        for (int i = 1; i < n; i++) {
            L[i] = (int)((1L * L[i - 1] * a[i - 1]) % MOD);
        }

        // 后缀积 R[i] = a[i+1] * ... * a[n-1] (mod MOD), R[n-1]=1
        int[] R = new int[n];
        R[n - 1] = 1;
        for (int i = n - 2; i >= 0; i--) {
            R[i] = (int)((1L * R[i + 1] * a[i + 1]) % MOD);
        }

        // 求和
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += (1L * L[i] * R[i]) % MOD;
            if (ans >= (1L << 60)) ans %= MOD;  // 防止累加过程中溢出
        }
        ans %= MOD;
        if (ans < 0) ans += MOD;

        System.out.println(ans);
    }
}
```

```cpp
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    const int MOD = 10007;

    string s;
    if (!getline(cin, s)) return 0;

    int n = (int)s.size() / 5;          // 因子个数，每个 '(t+d)' 长度为5
    vector<int> a(n);

    // 解析每个因子 (t+d) 或 (t-d)
    for (int k = 0; k < n; ++k) {
        int base = 5 * k;
        char op = s[base + 2];          // '+' 或 '-'
        int d = s[base + 3] - '0';      // 数字 1..9
        int val = (op == '-') ? -d : d;
        // 映射到 [0, MOD) 区间
        a[k] = (val % MOD + MOD) % MOD;
    }

    // 前缀积 L[i] = a[0] * ... * a[i-1] (mod MOD), L[0]=1
    vector<int> L(n, 1), R(n, 1);
    for (int i = 1; i < n; ++i) {
        L[i] = (int)((1LL * L[i - 1] * a[i - 1]) % MOD);
    }

    // 后缀积 R[i] = a[i+1] * ... * a[n-1] (mod MOD), R[n-1]=1
    for (int i = n - 2; i >= 0; --i) {
        R[i] = (int)((1LL * R[i + 1] * a[i + 1]) % MOD);
    }

    // 求和
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += 1LL * L[i] * R[i] % MOD;
        if (ans >= (1LL << 60)) ans %= MOD;  // 防止长整数溢出
    }
    ans = (ans % MOD + MOD) % MOD;
    cout << ans << "\n";

    return 0;
}
```
