## 解题思路

给定长度为 $n$ 的品类序列 $a_1, a_2, \ldots, a_n$，以及 $m$ 个起始货位 $b_i$。对每个询问，求区间 $[b_i, n]$ 中不同品类的个数。

这是后缀不同元素计数问题。从右向左扫描集装箱：

1. 用布尔数组 `vis` 标记每个品类是否已经在当前后缀中出现过。
2. 若 $a_i$ 是第一次出现，则种类计数 `cnt` 加 `1`。
3. 令 $f_i$ 表示从第 $i$ 个集装箱到第 $n$ 个集装箱的不同品类数，扫描到位置 $i$ 时令 $f_i = \textit{cnt}$。

预处理结束后，每个询问 $b_i$ 直接输出 $f_{b_i}$ 即可。由于 $a_i \le 10^5$，`vis` 可用大小为 $10^5+1$ 的布尔数组实现。

## 复杂度分析

预处理扫描一遍序列，每个询问 $O(1)$ 回答。

- 时间复杂度：$O(n + m)$。
- 空间复杂度：$O(n + U)$，其中 $U \le 10^5$ 为品类编号上界。

## 代码实现

### Python

```python
import sys


def solve(n, m, arr, queries):
    MAXV = 100000
    vis = [False] * (MAXV + 1)
    f = [0] * (n + 1)
    cnt = 0
    # 从右到左预处理后缀不同品类数
    for i in range(n - 1, -1, -1):
        v = arr[i]
        if not vis[v]:
            vis[v] = True
            cnt += 1
        f[i] = cnt
    out_lines = []
    for b in queries:
        out_lines.append(str(f[b - 1]))  # 输入为 1-based
    return "\n".join(out_lines)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]
    arr = data[2:2 + n]
    queries = data[2 + n:2 + n + m]
    print(solve(n, m, arr, queries))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;

public class Main {
    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        int nextInt() throws IOException {
            int c, sgn = 1, x = 0;
            do { c = read(); } while (c <= ' ');
            if (c == '-') { sgn = -1; c = read(); }
            while (c > ' ') {
                x = x * 10 + (c - '0');
                c = read();
            }
            return x * sgn;
        }
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        StringBuilder out = new StringBuilder();

        int n = fs.nextInt();
        int m = fs.nextInt();
        int[] a = new int[n];
        int MAXV = 100000;
        for (int i = 0; i < n; i++) a[i] = fs.nextInt();

        boolean[] vis = new boolean[MAXV + 1];
        int[] f = new int[n + 1];
        int cnt = 0;
        for (int i = n - 1; i >= 0; i--) {
            int v = a[i];
            if (!vis[v]) {
                vis[v] = true;
                cnt++;
            }
            f[i] = cnt;
        }

        for (int i = 0; i < m; i++) {
            int b = fs.nextInt();
            out.append(f[b - 1]).append('\n');
        }
        System.out.print(out.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    const int MAXV = 100000;
    vector<char> vis(MAXV + 1, 0);
    vector<int> f(n + 1, 0);
    int cnt = 0;
    for (int i = n - 1; i >= 0; --i) {
        int v = a[i];
        if (!vis[v]) {
            vis[v] = 1;
            ++cnt;
        }
        f[i] = cnt;
    }

    for (int i = 0; i < m; ++i) {
        int b;
        cin >> b;
        cout << f[b - 1] << "\n";
    }
    return 0;
}
```
