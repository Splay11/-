## 解题思路

将商品按余数 $r = a_i \bmod k$ 分桶。结对规则只取决于余数：

- 余数为 $0$ 的商品可以同类两两结对；若 $k$ 为偶数，余数为 $k/2$ 的商品同样可以同类两两结对。这类桶最终最多留下 `1` 件。为使剩余和最大，应撤下该桶中最小的 $2\lfloor |V_r|/2 \rfloor$ 件。
- 对于互补余数对 $r$ 与 $s=(k-r)\bmod k$（$1 \le r < s \le k-1$）：两类之间可以跨类结对，最终至少有一类会被清空，因此必须从两侧各撤下 $m=\min(|V_r|,|V_s|)$ 件。为使剩余和最大，两侧都撤下最小的 $m$ 件。

每个桶升序排序并做前缀和后，$O(1)$ 求出应撤下的体积和。没有互补桶的余数则全部保留。

## 复杂度分析

- **时间复杂度**：$O(n\log n)$。瓶颈是各余数桶内部排序。
- **空间复杂度**：$O(n)$。需要存储分桶结果与前缀和。

## 代码实现

### Python

```python
import sys

input_data = sys.stdin.read().strip().split()
it = iter(input_data)
t = int(next(it))

out_lines = []
for _ in range(t):
    n = int(next(it)); k = int(next(it))
    buckets = {}
    for _i in range(n):
        a = int(next(it))
        r = a % k if k != 0 else 0
        buckets.setdefault(r, []).append(a)

    info = {}
    for r, vec in buckets.items():
        vec.sort()
        pref = [0]
        for x in vec:
            pref.append(pref[-1] + x)
        info[r] = (vec, pref, pref[-1])

    ans = 0
    vis = set()

    for r in list(info.keys()):
        if r in vis:
            continue

        if r == 0 or (k % 2 == 0 and r * 2 == k):
            v, pref, total = info[r]
            s = len(v)
            pairs = s // 2
            remove_cnt = 2 * pairs
            removed_sum = pref[remove_cnt]
            ans += total - removed_sum
            vis.add(r)
            continue

        s = (k - (r % k)) % k
        if s not in info:
            ans += info[r][2]
            vis.add(r)
            continue

        if r == s:
            v, pref, total = info[r]
            sz = len(v)
            pairs = sz // 2
            remove_cnt = 2 * pairs
            removed_sum = pref[remove_cnt]
            ans += total - removed_sum
            vis.add(r)
            continue

        if r < s:
            vr, prefr, totalr = info[r]
            vs, prefs, totals = info[s]
            m = min(len(vr), len(vs))
            keep = (totalr - prefr[m]) + (totals - prefs[m])
            ans += keep
            vis.add(r); vis.add(s)

    out_lines.append(str(ans))

sys.stdout.write("\n".join(out_lines))
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        String next() throws IOException {
            StringBuilder sb = new StringBuilder();
            int c;
            while ((c = read()) != -1 && c <= ' ') {}
            if (c == -1) return null;
            do {
                sb.append((char) c);
                c = read();
            } while (c != -1 && c > ' ');
            return sb.toString();
        }
        int nextInt() throws IOException { return Integer.parseInt(next()); }
        long nextLong() throws IOException { return Long.parseLong(next()); }
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        String ts = fs.next();
        if (ts == null) return;
        int t = Integer.parseInt(ts);
        StringBuilder out = new StringBuilder();

        while (t-- > 0) {
            int n = fs.nextInt();
            long k = fs.nextLong();
            HashMap<Long, ArrayList<Long>> buckets = new HashMap<>(n * 2);

            for (int i = 0; i < n; i++) {
                long a = fs.nextLong();
                long r = (k == 0) ? 0 : (a % k);
                buckets.computeIfAbsent(r, key -> new ArrayList<>()).add(a);
            }

            final class Info {
                ArrayList<Long> v;
                long[] pref;
                long total;
            }
            HashMap<Long, Info> info = new HashMap<>(buckets.size() * 2);
            for (Map.Entry<Long, ArrayList<Long>> e : buckets.entrySet()) {
                long r = e.getKey();
                ArrayList<Long> v = e.getValue();
                v.sort(Comparator.naturalOrder());
                Info inf = new Info();
                inf.v = v;
                inf.pref = new long[v.size() + 1];
                for (int i = 0; i < v.size(); i++) {
                    inf.pref[i + 1] = inf.pref[i] + v.get(i);
                }
                inf.total = inf.pref[v.size()];
                info.put(r, inf);
            }

            long ans = 0L;
            HashSet<Long> vis = new HashSet<>(info.size() * 2);

            for (long r : info.keySet()) {
                if (vis.contains(r)) continue;

                if (r == 0 || (k % 2 == 0 && r * 2 == k)) {
                    Info inf = info.get(r);
                    int s = inf.v.size();
                    int pairs = s / 2;
                    int removeCnt = 2 * pairs;
                    long removed = inf.pref[removeCnt];
                    ans += (inf.total - removed);
                    vis.add(r);
                    continue;
                }

                long s = (k - (r % k)) % k;
                if (!info.containsKey(s)) {
                    ans += info.get(r).total;
                    vis.add(r);
                    continue;
                }

                if (r == s) {
                    Info inf = info.get(r);
                    int sz = inf.v.size();
                    int pairs = sz / 2;
                    int removeCnt = 2 * pairs;
                    long removed = inf.pref[removeCnt];
                    ans += (inf.total - removed);
                    vis.add(r);
                    continue;
                }

                if (r < s) {
                    Info ir = info.get(r);
                    Info isv = info.get(s);
                    int m = Math.min(ir.v.size(), isv.v.size());
                    long keep = (ir.total - ir.pref[m]) + (isv.total - isv.pref[m]);
                    ans += keep;
                    vis.add(r);
                    vis.add(s);
                }
            }

            out.append(ans).append('\n');
        }

        System.out.print(out.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    if (!(cin >> t)) return 0;
    while (t--) {
        int n;
        long long k;
        cin >> n >> k;
        unordered_map<long long, vector<long long>> buckets;
        buckets.reserve(n * 2);

        for (int i = 0; i < n; ++i) {
            long long a; cin >> a;
            long long r = k == 0 ? 0 : (a % k);
            buckets[r].push_back(a);
        }

        struct Info {
            vector<long long> v;
            vector<long long> pref;
            long long total = 0;
        };
        unordered_map<long long, Info> info;
        info.reserve(buckets.size() * 2);

        for (auto &kv : buckets) {
            auto r = kv.first;
            auto &vec = kv.second;
            sort(vec.begin(), vec.end());
            Info inf;
            inf.v = move(vec);
            inf.pref.resize(inf.v.size() + 1, 0);
            for (size_t i = 0; i < inf.v.size(); ++i) {
                inf.pref[i + 1] = inf.pref[i] + inf.v[i];
            }
            inf.total = inf.pref.back();
            info[r] = move(inf);
        }

        long long ans = 0;
        unordered_set<long long> vis;
        vis.reserve(info.size() * 2);

        for (auto &kv : info) {
            long long r = kv.first;
            if (vis.count(r)) continue;

            if (r == 0 || (k % 2 == 0 && r * 2 == k)) {
                auto &inf = info[r];
                long long s = (long long)inf.v.size();
                long long pairs = s / 2;
                long long remove_cnt = 2 * pairs;
                long long removed_sum = inf.pref[remove_cnt];
                ans += (inf.total - removed_sum);
                vis.insert(r);
                continue;
            }

            long long s = (k - (r % k)) % k;
            if (!info.count(s)) {
                ans += info[r].total;
                vis.insert(r);
                continue;
            }

            if (r == s) {
                auto &inf = info[r];
                long long sz = (long long)inf.v.size();
                long long pairs = sz / 2;
                long long remove_cnt = 2 * pairs;
                long long removed_sum = inf.pref[remove_cnt];
                ans += (inf.total - removed_sum);
                vis.insert(r);
                continue;
            }

            if (r < s) {
                auto &ir = info[r];
                auto &isv = info[s];
                long long m = min<long long>(ir.v.size(), isv.v.size());
                long long keep = (ir.total - ir.pref[m]) + (isv.total - isv.pref[m]);
                ans += keep;
                vis.insert(r);
                vis.insert(s);
            }
        }

        cout << ans << "\n";
    }
    return 0;
}
```
