### 思路

* 将数组按余数分组：对每个数计算 $r = a_i \bmod k$，分类到 $V_r$。
* 对于 $r = 0$ 以及（若 $k$ 为偶数）$r = \frac{k}{2}$：同类元素可以两两配对移除，最多只能留下 $1$ 个。要最大化剩余和，应移除该类中最小的 $2 \left\lfloor \frac{|V_r|}{2} \right\rfloor$ 个元素。
* 对于互补余数对 $r$ 与 $s = (k-r) \bmod k$（取 $1 \le r < s \le k-1$）：

  * 最终必须让两类中至少有一类变为空，因此必然会从两边各移除 $m = \min(|V_r|, |V_s|)$ 个元素。
  * 为使剩余和最大，移除两侧中最小的 $m$ 个元素（总移除和最小）。
  * 余下的就是较大一侧的剩余元素（或两边都空）。
* 将每个 $V_r$ 升序排序并做前缀和 $P_r$。记 $\text{sum}(V_r) = P_r[|V_r|]$。

### C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    if (!(cin >> t)) return 0;
    while (t--) {
        int n;
        long long k;
        cin >> n >> k;
        unordered_map<long long, vector<long long>> buckets;
        buckets.reserve(n * 2);

        for (int i = 0; i < n; ++i) {
            long long a; cin >> a;
            long long r = k == 0 ? 0 : (a % k);
            buckets[r].push_back(a);
        }

        // 对每个余数组升序并构建前缀和
        struct Info {
            vector<long long> v;
            vector<long long> pref;
            long long total = 0;
        };
        unordered_map<long long, Info> info;
        info.reserve(buckets.size() * 2);

        for (auto &kv : buckets) {
            auto r = kv.first;
            auto &vec = kv.second;
            sort(vec.begin(), vec.end());
            Info inf;
            inf.v = move(vec);
            inf.pref.resize(inf.v.size() + 1, 0);
            for (size_t i = 0; i < inf.v.size(); ++i) {
                inf.pref[i + 1] = inf.pref[i] + inf.v[i];
            }
            inf.total = inf.pref.back();
            info[r] = move(inf);
        }

        long long ans = 0;
        unordered_set<long long> vis;
        vis.reserve(info.size() * 2);

        for (auto &kv : info) {
            long long r = kv.first;
            if (vis.count(r)) continue;

            // 特殊类：r = 0 或 r = k/2 (k 偶)
            if (r == 0 || (k % 2 == 0 && r * 2 == k)) {
                auto &inf = info[r];
                long long s = (long long)inf.v.size();
                long long pairs = s / 2;
                long long remove_cnt = 2 * pairs;
                long long removed_sum = inf.pref[remove_cnt];
                ans += (inf.total - removed_sum);
                vis.insert(r);
                continue;
            }

            long long s = (k - (r % k)) % k;
            // 如果没有互补类，全部保留
            if (!info.count(s)) {
                ans += info[r].total;
                vis.insert(r);
                continue;
            }

            // 成对处理 r 与 s（仅处理 r < s，防止重复）
            if (r == s) {
                // 安全防御，理论上已在上面特殊类处理
                auto &inf = info[r];
                long long sz = (long long)inf.v.size();
                long long pairs = sz / 2;
                long long remove_cnt = 2 * pairs;
                long long removed_sum = inf.pref[remove_cnt];
                ans += (inf.total - removed_sum);
                vis.insert(r);
                continue;
            }

            if (r < s) {
                auto &ir = info[r];
                auto &isv = info[s];
                long long m = min<long long>(ir.v.size(), isv.v.size());
                long long keep = (ir.total - ir.pref[m]) + (isv.total - isv.pref[m]);
                ans += keep;
                vis.insert(r);
                vis.insert(s);
            } else {
                // 将在 r<s 的时候处理，这里只标记以避免重复
                // 不做任何事
            }
        }

        cout << ans << "\n";
    }
    return 0;
}
```

```python
import sys

input_data = sys.stdin.read().strip().split()
it = iter(input_data)
t = int(next(it))

out_lines = []
for _ in range(t):
    n = int(next(it)); k = int(next(it))
    buckets = {}
    for _i in range(n):
        a = int(next(it))
        r = a % k if k != 0 else 0
        buckets.setdefault(r, []).append(a)

    info = {}
    for r, vec in buckets.items():
        vec.sort()
        pref = [0]
        for x in vec:
            pref.append(pref[-1] + x)
        info[r] = (vec, pref, pref[-1])  # (v, pref, total)

    ans = 0
    vis = set()

    for r in list(info.keys()):
        if r in vis:
            continue

        # 特殊类：r = 0 或 r = k/2 (k 偶)
        if r == 0 or (k % 2 == 0 and r * 2 == k):
            v, pref, total = info[r]
            s = len(v)
            pairs = s // 2
            remove_cnt = 2 * pairs
            removed_sum = pref[remove_cnt]
            ans += total - removed_sum
            vis.add(r)
            continue

        s = (k - (r % k)) % k
        if s not in info:
            # 没有互补类，全部保留
            ans += info[r][2]
            vis.add(r)
            continue

        if r == s:
            # 理论上已在特殊类处理，这里兜底
            v, pref, total = info[r]
            sz = len(v)
            pairs = sz // 2
            remove_cnt = 2 * pairs
            removed_sum = pref[remove_cnt]
            ans += total - removed_sum
            vis.add(r)
            continue

        if r < s:
            vr, prefr, totalr = info[r]
            vs, prefs, totals = info[s]
            m = min(len(vr), len(vs))
            keep = (totalr - prefr[m]) + (totals - prefs[m])
            ans += keep
            vis.add(r); vis.add(s)
        else:
            # 在 r<s 时处理
            pass

    out_lines.append(str(ans))

sys.stdout.write("\n".join(out_lines))
```

### Java

```java
import java.io.*;
import java.util.*;

/*
  思路：
  - 按 a_i % k 分桶并排序，构建前缀和
  - 处理 r=0 与（若 k 为偶数）r=k/2：移除最小的 2 * floor(sz/2) 个
  - 处理互补余数对 r 与 s=k-r（只在 r<s 时做）：各移除 m=min(sz_r, sz_s) 个最小的
  - 累加剩余和
*/
public class Main {
    static final class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        String next() throws IOException {
            StringBuilder sb = new StringBuilder();
            int c;
            while ((c = read()) != -1 && c <= ' ') {}
            if (c == -1) return null;
            do {
                sb.append((char)c);
                c = read();
            } while (c != -1 && c > ' ');
            return sb.toString();
        }
        int nextInt() throws IOException { return Integer.parseInt(next()); }
        long nextLong() throws IOException { return Long.parseLong(next()); }
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        String ts = fs.next();
        if (ts == null) return;
        int t = Integer.parseInt(ts);
        StringBuilder out = new StringBuilder();

        while (t-- > 0) {
            int n = fs.nextInt();
            long k = fs.nextLong();
            HashMap<Long, ArrayList<Long>> buckets = new HashMap<>(n * 2);

            for (int i = 0; i < n; i++) {
                long a = fs.nextLong();
                long r = (k == 0) ? 0 : (a % k);
                buckets.computeIfAbsent(r, key -> new ArrayList<>()).add(a);
            }

            // 排序并构建前缀和
            final class Info {
                ArrayList<Long> v;
                long[] pref;
                long total;
            }
            HashMap<Long, Info> info = new HashMap<>(buckets.size() * 2);
            for (Map.Entry<Long, ArrayList<Long>> e : buckets.entrySet()) {
                long r = e.getKey();
                ArrayList<Long> v = e.getValue();
                v.sort(Comparator.naturalOrder());
                Info inf = new Info();
                inf.v = v;
                inf.pref = new long[v.size() + 1];
                for (int i = 0; i < v.size(); i++) {
                    inf.pref[i + 1] = inf.pref[i] + v.get(i);
                }
                inf.total = inf.pref[v.size()];
                info.put(r, inf);
            }

            long ans = 0L;
            HashSet<Long> vis = new HashSet<>(info.size() * 2);

            for (long r : info.keySet()) {
                if (vis.contains(r)) continue;

                // 特殊类：r = 0 或 r = k/2（k 偶）
                if (r == 0 || (k % 2 == 0 && r * 2 == k)) {
                    Info inf = info.get(r);
                    int s = inf.v.size();
                    int pairs = s / 2;
                    int removeCnt = 2 * pairs;
                    long removed = inf.pref[removeCnt];
                    ans += (inf.total - removed);
                    vis.add(r);
                    continue;
                }

                long s = (k - (r % k)) % k;
                if (!info.containsKey(s)) {
                    // 没有互补类，全部保留
                    ans += info.get(r).total;
                    vis.add(r);
                    continue;
                }

                if (r == s) {
                    // 兜底（理论上已在特殊类处理）
                    Info inf = info.get(r);
                    int sz = inf.v.size();
                    int pairs = sz / 2;
                    int removeCnt = 2 * pairs;
                    long removed = inf.pref[removeCnt];
                    ans += (inf.total - removed);
                    vis.add(r);
                    continue;
                }

                if (r < s) {
                    Info ir = info.get(r);
                    Info isv = info.get(s);
                    int m = Math.min(ir.v.size(), isv.v.size());
                    long keep = (ir.total - ir.pref[m]) + (isv.total - isv.pref[m]);
                    ans += keep;
                    vis.add(r);
                    vis.add(s);
                } else {
                    // 等待 r<s 的时候处理
                }
            }

            out.append(ans).append('\n');
        }

        System.out.print(out.toString());
    }
}
```
