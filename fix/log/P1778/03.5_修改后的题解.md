## 解题思路

这是一个带有状态的最短路问题。可以将每个节点 $i$ 和当前持有的通道类型组合成一个状态 $(i, \text{type})$，其中 $\text{type} \in \{0, 1\}$（内部用 $0$ 表示题目中的 `1` 类型，$1$ 表示 `2` 类型）。使用变种的 **Dijkstra 算法** 求解：

1. 定义 $dist[i][0]$ 表示从起点 $1$ 出发，到达节点 $i$ 且当前通道类型为 `1` 的最短时间。  
   $dist[i][1]$ 表示到达节点 $i$ 且当前类型为 `2` 的最短时间。

2. 初始状态：从节点 $1$ 出发，可以任意选择一种通道类型且无切换延迟，因此  
   $dist[1][0] = dist[1][1] = 0$ ，将两个状态加入优先队列。

3. 每次从堆中取出当前总时间最小的状态 $(u, t)$，若该状态已处理则跳过。否则遍历节点 $u$ 的所有邻边 $(u, v, w, \text{nt})$：
   - 若 $\text{nt} \neq t$，说明需要在节点 $u$ 花费 $c_u$ 的时间切换通道类型，否则切换延迟为 $0$。
   - 新的时间为 $dist[u][t] + w + \text{trans}$（$\text{trans}$ 为切换延迟）。
   - 如果新时间小于 $dist[v][\text{nt}]$，则更新并将 $(v, \text{nt})$ 加入堆中。

4. 最终答案即为 $\min(dist[n][0],\ dist[n][1])$。

题目保证图连通，因此一定存在可行路径。

## 复杂度分析

图中每个节点最多对应 $2$ 个状态，所有状态和边都会在 Dijkstra 过程中被处理。使用小根堆优化时，堆中元素个数为 $O(n)$。

- **时间复杂度**：$O((n + m) \log n)$ 或等价地 $O(m \log n)$，其中 $n$ 为节点数，$m$ 为链路数。
- **空间复杂度**：$O(n + m)$，用于存储邻接表、距离数组和访问标记。

## 代码实现

### Python

```python
import heapq

INF = int(1e18)

class Edge:
    def __init__(self, to, w, t):
        self.to = to
        self.w = w
        self.t = t

class State:
    def __init__(self, d, u, t):
        self.d = d
        self.u = u
        self.t = t

    def __lt__(self, other):
        return self.d < other.d

def dijkstra(n, c, g):
    # dist[i][0] 对应通道类型 1，dist[i][1] 对应通道类型 2
    dist = [[INF] * 2 for _ in range(n + 1)]
    vis = [[False] * 2 for _ in range(n + 1)]
    dist[1][0] = dist[1][1] = 0
    heap = []
    heapq.heappush(heap, State(0, 1, 0))
    heapq.heappush(heap, State(0, 1, 1))

    while heap:
        cur = heapq.heappop(heap)
        u, t = cur.u, cur.t
        if vis[u][t]:
            continue
        vis[u][t] = True

        for e in g[u]:
            v, w, nxt_t = e.to, e.w, e.t
            trans = 0 if nxt_t == t else c[u]   # 若通道类型不同，在 u 处花费 c_u 切换
            new_dist = dist[u][t] + trans + w
            if new_dist < dist[v][nxt_t]:
                dist[v][nxt_t] = new_dist
                heapq.heappush(heap, State(new_dist, v, nxt_t))

    return min(dist[n][0], dist[n][1])


if __name__ == "__main__":
    n, m = map(int, input().split())
    # c[1..n] 切换延迟
    c = [0] + list(map(int, input().split()))
    g = [[] for _ in range(n + 1)]

    for _ in range(m):
        u, v, w, t = map(int, input().split())
        t -= 1          # 内部用 0/1 表示
        g[u].append(Edge(v, w, t))
        g[v].append(Edge(u, w, t))

    print(dijkstra(n, c, g))
```

### Java

```java
import java.util.*;

class Edge {
    int to, w, t;
    Edge(int to, int w, int t) {
        this.to = to;
        this.w = w;
        this.t = t;
    }
}

class State implements Comparable<State> {
    long dist;
    int u, t;
    State(long dist, int u, int t) {
        this.dist = dist;
        this.u = u;
        this.t = t;
    }
    public int compareTo(State other) {
        return Long.compare(this.dist, other.dist);
    }
}

public class Main {
    static final long INF = (long) 1e18;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        // c[1..n] 切换延迟
        int[] c = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            c[i] = sc.nextInt();
        }

        List<Edge>[] g = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            g[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            int t = sc.nextInt() - 1;   // 内部用 0/1
            g[u].add(new Edge(v, w, t));
            g[v].add(new Edge(u, w, t));
        }

        // dist[i][0] 类型1, dist[i][1] 类型2
        long[][] dist = new long[n + 1][2];
        boolean[][] vis = new boolean[n + 1][2];
        for (int i = 0; i <= n; i++) {
            dist[i][0] = dist[i][1] = INF;
        }
        dist[1][0] = dist[1][1] = 0;

        PriorityQueue<State> pq = new PriorityQueue<>();
        pq.offer(new State(0, 1, 0));
        pq.offer(new State(0, 1, 1));

        while (!pq.isEmpty()) {
            State cur = pq.poll();
            int u = cur.u, t = cur.t;
            if (vis[u][t]) continue;
            vis[u][t] = true;

            for (Edge e : g[u]) {
                int v = e.to, nxtT = e.t;
                long trans = (nxtT == t) ? 0 : c[u];  // 不同则加切换延迟 c[u]
                long newDist = dist[u][t] + trans + e.w;
                if (newDist < dist[v][nxtT]) {
                    dist[v][nxtT] = newDist;
                    pq.offer(new State(newDist, v, nxtT));
                }
            }
        }

        System.out.println(Math.min(dist[n][0], dist[n][1]));
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;
using LL = long long;
const LL INF = 1e18;

struct Edge {
    int to, w, t;
};

struct State {
    LL d;
    int u, t;
    // 小根堆需要重载大于号
    bool operator>(const State& other) const {
        return d > other.d;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    // c[1..n] 切换延迟
    vector<int> c(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> c[i];
    }

    vector<vector<Edge>> g(n + 1);
    for (int i = 0; i < m; ++i) {
        int u, v, w, t;
        cin >> u >> v >> w >> t;
        t--;   // 内部用 0/1
        g[u].push_back({v, w, t});
        g[v].push_back({u, w, t});
    }

    // dist[i][0] 类型1, dist[i][1] 类型2
    vector<vector<LL>> dist(n + 1, vector<LL>(2, INF));
    vector<vector<bool>> vis(n + 1, vector<bool>(2, false));
    dist[1][0] = dist[1][1] = 0;

    priority_queue<State, vector<State>, greater<State>> pq;
    pq.push({0, 1, 0});
    pq.push({0, 1, 1});

    while (!pq.empty()) {
        State cur = pq.top();
        pq.pop();
        int u = cur.u, t = cur.t;
        if (vis[u][t]) continue;
        vis[u][t] = true;

        for (auto &e : g[u]) {
            int v = e.to, nxtT = e.t;
            LL trans = (nxtT == t) ? 0 : c[u];   // 通道类型不同，加 c[u]
            LL newDist = dist[u][t] + trans + e.w;
            if (newDist < dist[v][nxtT]) {
                dist[v][nxtT] = newDist;
                pq.push({newDist, v, nxtT});
            }
        }
    }

    cout << min(dist[n][0], dist[n][1]) << '\n';
    return 0;
}
```
