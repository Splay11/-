## 解题思路

每次操作可以选取两个已有的数字标记进行逻辑与，并将结果加入集合。由于新值可以反复参与运算，最终可达的任意标记，必然可以表示为初始标记的某个**非空子集**的按位与结果。因此，问题转化为：

> 在数值范围 $[0, 1023]$ 内，统计有多少个整数 $mask$ 能够被表示为至少一个初始标记的非空子集的按位与。

**可达判定方法**  
若一个 $mask$ 可以由某个子集按位与得到，则该子集中的每个标记都必须包含 $mask$ 的所有 $1$ 位，即 $(mark \mathbin{\&} mask) = mask$。设 $\mathcal{S}(mask)$ 为所有满足此条件的初始标记的集合。如果 $\mathcal{S}(mask)$ 非空，并且 $\mathcal{S}(mask)$ 中所有元素的按位与结果恰好等于 $mask$，那么 $mask$ 就是可达的。

**超集动态规划**  
值域很小（$V=1024$，仅 $10$ 个二进制位），我们可以使用超集 DP 高效实现：

1. 维护两个数组：
   - `g[mask]`：所有包含 $mask$ 的初始标记的按位与结果，初始化为全 $1$（即 $1023$）。
   - `exist[mask]`：是否存在一个初始标记刚好等于 $mask$，初始为 `false`。
2. 对于每个输入的标记 $x$，令 `g[x] = x`，`exist[x] = true`。
3. 从低到高枚举二进制位 $bit = 0 \dots 9$，再枚举所有 $mask$：
   - 若 `mask` 的第 $bit$ 位为 $0$，则考虑其超集 `super = mask | (1 << bit)`；
   - 如果 `exist[super]` 为真，则将 `super` 的按位与信息合并到 `mask`：`g[mask] &= g[super]`，并置 `exist[mask] = true`。
4. 遍历结束后，`g[mask]` 就存储了所有包含 $mask$ 的初始标记的按位与。
5. 答案即为满足 `exist[mask] == true` 且 `g[mask] == mask` 的 $mask$ 的个数。

这种做法的正确性依赖于：若超集的按位与能贡献给子集，则子集的存在性和按位与值都能被正确向下传递。

## 复杂度分析

设值域大小 $V = 1024$，二进制位数 $B = 10$。

- 对于每组测试数据：
  - 初始化 `g` 和 `exist` 数组：$O(V)$
  - 读入 $n$ 个标记并设置初值：$O(n)$
  - 超集 DP 两层循环：$O(V \cdot B) = O(1024 \times 10) = O(1)$ 常数开销
  - 统计答案：$O(V)$
- 单组时间复杂度 $O(n + V \cdot B)$。所有测试样例的 $n$ 之和 $\sum n \le 2 \times 10^5$，总时间复杂度 $O(\sum n + T \cdot V \cdot B)$，在给定约束下非常高效。
- 空间复杂度：需要两个长度为 $V$ 的数组，即 $O(V) = O(1)$ 额外空间。

## 代码实现

### Python

```python
def max_distinct_marks(marks):
    MAX_MASK = 1023
    SIZE = 1024
    g = [MAX_MASK] * SIZE      # g[mask] 表示包含 mask 的所有初始标记的按位与
    exist = [False] * SIZE

    for x in marks:
        g[x] = x
        exist[x] = True

    # 超集 DP
    for bit in range(10):
        for mask in range(SIZE):
            if (mask & (1 << bit)) == 0:
                super_mask = mask | (1 << bit)
                if exist[super_mask]:
                    g[mask] &= g[super_mask]
                    exist[mask] = True

    ans = sum(1 for mask in range(SIZE) if exist[mask] and g[mask] == mask)
    return ans


def main():
    import sys
    data = sys.stdin.read().split()
    it = iter(data)
    t = int(next(it))
    out = []
    for _ in range(t):
        n = int(next(it))
        marks = [int(next(it)) for _ in range(n)]
        out.append(str(max_distinct_marks(marks)))
    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedInputStream;
import java.io.IOException;

public class Main {
    static final int MAX_MASK = 1023;
    static final int SIZE = 1024;

    static int maxDistinctMarks(int[] marks) {
        int[] g = new int[SIZE];
        boolean[] exist = new boolean[SIZE];
        for (int i = 0; i < SIZE; i++) {
            g[i] = MAX_MASK;
        }

        for (int x : marks) {
            g[x] = x;
            exist[x] = true;
        }

        for (int bit = 0; bit < 10; bit++) {
            for (int mask = 0; mask < SIZE; mask++) {
                if ((mask & (1 << bit)) == 0) {
                    int superMask = mask | (1 << bit);
                    if (exist[superMask]) {
                        g[mask] &= g[superMask];
                        exist[mask] = true;
                    }
                }
            }
        }

        int ans = 0;
        for (int mask = 0; mask < SIZE; mask++) {
            if (exist[mask] && g[mask] == mask) {
                ans++;
            }
        }
        return ans;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();
        int tc = fs.nextInt();
        StringBuilder sb = new StringBuilder();

        for (int t = 0; t < tc; t++) {
            int n = fs.nextInt();
            int[] marks = new int[n];
            for (int i = 0; i < n; i++) {
                marks[i] = fs.nextInt();
            }
            sb.append(maxDistinctMarks(marks)).append('\n');
        }
        System.out.print(sb.toString());
    }
}

class FastScanner {
    private final BufferedInputStream in = new BufferedInputStream(System.in);
    private final byte[] buffer = new byte[1 << 16];
    private int ptr = 0, len = 0;

    private int read() throws IOException {
        if (ptr >= len) {
            len = in.read(buffer);
            ptr = 0;
            if (len <= 0) return -1;
        }
        return buffer[ptr++];
    }

    int nextInt() throws IOException {
        int c;
        do { c = read(); } while (c <= ' ');
        int sign = 1;
        if (c == '-') {
            sign = -1;
            c = read();
        }
        int val = 0;
        while (c > ' ') {
            val = val * 10 + (c - '0');
            c = read();
        }
        return val * sign;
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int maxDistinctMarks(const vector<int>& marks) {
    const int MAX_MASK = 1023;
    const int SIZE = 1024;
    vector<int> g(SIZE, MAX_MASK);
    vector<bool> exist(SIZE, false);

    for (int x : marks) {
        g[x] = x;
        exist[x] = true;
    }

    for (int bit = 0; bit < 10; ++bit) {
        for (int mask = 0; mask < SIZE; ++mask) {
            if ((mask & (1 << bit)) == 0) {
                int superMask = mask | (1 << bit);
                if (exist[superMask]) {
                    g[mask] &= g[superMask];
                    exist[mask] = true;
                }
            }
        }
    }

    int ans = 0;
    for (int mask = 0; mask < SIZE; ++mask) {
        if (exist[mask] && g[mask] == mask) ans++;
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        vector<int> marks(n);
        for (int i = 0; i < n; ++i) cin >> marks[i];
        cout << maxDistinctMarks(marks) << '\n';
    }
    return 0;
}
```
