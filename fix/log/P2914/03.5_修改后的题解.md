## 解题思路

本题要求对 $n \times n$ 的高程矩阵进行最大值压缩，采样窗口的宽和高分别为 $w$ 和 $h$，步长与窗口尺寸完全相同（水平步长 $w$，垂直步长 $h$）。从左上角出发，每次在窗口内取最大值，当右侧或下侧不足以放下完整窗口时停止。

**算法步骤：**

1. **确定输出矩阵尺寸**  
   由于窗口水平和垂直滑动步长等于窗口尺寸，且只在剩余空间足够容纳完整窗口时才继续，因此输出矩阵的行数为 $\lfloor n/h \rfloor$，列数为 $\lfloor n/w \rfloor$。

2. **遍历所有窗口位置**  
   使用两层循环，外层循环控制行起点 $i$，范围为 `0, h, 2h, …` 直到 $i + h \le n$；内层循环控制列起点 $j$，范围为 `0, w, 2w, …` 直到 $j + w \le n$。

3. **计算当前窗口的最大值**  
   对于每个 $(i, j)$ 确定的窗口，遍历窗口内的所有元素 `matrix[x][y]`（其中 $x \in [i, i+h)$，$y \in [j, j+w)$），记录并返回最大值。

4. **保存结果**  
   将每个窗口的最大值按行优先顺序填入输出矩阵的对应位置。

5. **边界处理**  
   当 $n$ 不能被 $h$ 或 $w$ 整除时，剩余的底部若干行或右侧若干列将不被处理，自动忽略。

此方法等价于对二维矩阵进行无重叠的最大池化（max pooling），池化核大小即为 $(h, w)$，步长等于核大小。

## 复杂度分析

- **时间复杂度**：$O(n^2)$  
  需要进行 $\frac{n}{h} \times \frac{n}{w}$ 次窗口处理，每个窗口需遍历 $h \times w$ 个元素，总操作次数为 $\frac{n}{h} \times \frac{n}{w} \times h \times w = n^2$，与输入矩阵的元素总数同阶。

- **空间复杂度**：$O(1)$ 额外辅助空间  
  不考虑存储输出矩阵所需的空间，算法只需常数级别的临时变量用于记录最大值和循环控制，不随输入规模增长。

## 代码实现

### Python

```python
def max_pooling(matrix, pool_w, pool_h):
    n = len(matrix)
    result = []
    # 按步长 h 遍历行
    for i in range(0, n - pool_h + 1, pool_h):
        row = []
        # 按步长 w 遍历列
        for j in range(0, n - pool_w + 1, pool_w):
            max_val = matrix[i][j]
            # 遍历窗口内所有元素找最大值
            for x in range(i, i + pool_h):
                for y in range(j, j + pool_w):
                    if matrix[x][y] > max_val:
                        max_val = matrix[x][y]
            row.append(max_val)
        result.append(row)
    return result


if __name__ == "__main__":
    matrix = eval(input())          # 读取二维列表
    w, h = eval(input())            # 读取窗口宽度和高度
    ans = max_pooling(matrix, w, h)
    print(ans)                      # 直接输出 Python 列表格式
```

### Java

```java
import java.util.*;
import java.util.regex.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String line1 = sc.nextLine().trim();
        String line2 = sc.nextLine().trim();

        // 解析二维矩阵
        List<Integer> nums = extractInts(line1);
        int n = (int) Math.sqrt(nums.size());
        int[][] matrix = new int[n][n];
        int idx = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                matrix[i][j] = nums.get(idx++);
            }
        }

        // 解析窗口大小
        List<Integer> sizes = extractInts(line2);
        int w = sizes.get(0);
        int h = sizes.get(1);

        // 执行最大值压缩
        int[][] result = maxPooling(matrix, w, h);

        // 输出为 Python 列表格式
        printPythonMatrix(result);
    }

    // 从字符串中提取所有整数（支持负数）
    private static List<Integer> extractInts(String s) {
        List<Integer> list = new ArrayList<>();
        Matcher m = Pattern.compile("-?\\d+").matcher(s);
        while (m.find()) {
            list.add(Integer.parseInt(m.group()));
        }
        return list;
    }

    // 最大值池化
    private static int[][] maxPooling(int[][] mat, int w, int h) {
        int n = mat.length;
        int rows = n / h;
        int cols = n / w;
        int[][] res = new int[rows][cols];

        for (int i = 0; i < rows; i++) {
            int rStart = i * h;
            for (int j = 0; j < cols; j++) {
                int cStart = j * w;
                int maxVal = mat[rStart][cStart];
                for (int x = rStart; x < rStart + h; x++) {
                    for (int y = cStart; y < cStart + w; y++) {
                        if (mat[x][y] > maxVal) {
                            maxVal = mat[x][y];
                        }
                    }
                }
                res[i][j] = maxVal;
            }
        }
        return res;
    }

    // 按 Python 列表格式打印二维数组
    private static void printPythonMatrix(int[][] mat) {
        StringBuilder sb = new StringBuilder();
        sb.append("[");
        for (int i = 0; i < mat.length; i++) {
            sb.append("[");
            for (int j = 0; j < mat[i].length; j++) {
                sb.append(mat[i][j]);
                if (j < mat[i].length - 1) sb.append(", ");
            }
            sb.append("]");
            if (i < mat.length - 1) sb.append(", ");
        }
        sb.append("]");
        System.out.println(sb.toString());
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <regex>
#include <cmath>
using namespace std;

vector<int> extractInts(const string& s) {
    vector<int> nums;
    regex re("-?\\d+");
    auto begin = sregex_iterator(s.begin(), s.end(), re);
    auto end = sregex_iterator();
    for (auto it = begin; it != end; ++it) {
        nums.push_back(stoi(it->str()));
    }
    return nums;
}

vector<vector<int>> maxPooling(const vector<vector<int>>& mat, int w, int h) {
    int n = mat.size();
    int rows = n / h;
    int cols = n / w;
    vector<vector<int>> res(rows, vector<int>(cols));

    for (int i = 0; i < rows; i++) {
        int rStart = i * h;
        for (int j = 0; j < cols; j++) {
            int cStart = j * w;
            int maxVal = mat[rStart][cStart];
            for (int x = rStart; x < rStart + h; x++) {
                for (int y = cStart; y < cStart + w; y++) {
                    if (mat[x][y] > maxVal) {
                        maxVal = mat[x][y];
                    }
                }
            }
            res[i][j] = maxVal;
        }
    }
    return res;
}

void printPythonMatrix(const vector<vector<int>>& mat) {
    cout << "[";
    for (size_t i = 0; i < mat.size(); i++) {
        cout << "[";
        for (size_t j = 0; j < mat[i].size(); j++) {
            cout << mat[i][j];
            if (j < mat[i].size() - 1) cout << ", ";
        }
        cout << "]";
        if (i < mat.size() - 1) cout << ", ";
    }
    cout << "]" << endl;
}

int main() {
    string line1, line2;
    getline(cin, line1);
    getline(cin, line2);

    // 解析二维矩阵
    vector<int> nums = extractInts(line1);
    int n = (int) sqrt(nums.size());
    vector<vector<int>> matrix(n, vector<int>(n));
    int idx = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            matrix[i][j] = nums[idx++];
        }
    }

    // 解析窗口大小
    vector<int> sizes = extractInts(line2);
    int w = sizes[0];
    int h = sizes[1];

    vector<vector<int>> result = maxPooling(matrix, w, h);

    printPythonMatrix(result);
    return 0;
}
```
