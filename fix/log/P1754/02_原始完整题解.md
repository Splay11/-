## 题解

本题是 codeforces 难度分为 1900分的原题 [链接](https://codeforces.com/contest/404/problem/D)

由于本题是有后效性的动态规划，直接递推比较难写，这边采用记忆化搜索的方法来写

定义 $dfs(i,j)$ 表示考虑 $s[0] ... s[i]$，$j=0/1/2$ ，分别为：$s[i]$ 不能是雷/ $s[i]$ 一定是雷/ $s[i+1]$ 是雷

返回在这种情况下的方案数。 

分类讨论即可。 

递归边界：$dfs(-1,0) = dfs(-1,2) = 1, dfs(-1,1) = 0$

递归入口：$dfs(n-1,0) + dfs(n-1,1)  $



## AC代码


### C++
```cpp
#include <iostream>
#include <vector>
#include <cstring>
using namespace std;

#define int long long
const int mod = 1000000007;

int f(int i, int j, const string& s, vector<vector<int>>& dp) {
    if (i < 0) {
        if (j == 1) return 0;
        return 1;
    }
    int& res = dp[i][j];
    if (res != -1) return res;
    char b = s[i];
    if (j == 0) {
        switch (b) {
            case '0':
                return res = f(i-1, 0, s, dp);
            case '1':
                return res = f(i-1, 1, s, dp);
            case '2':
            case '*':
                return res = 0;
            default:
                return res = (f(i-1, 0, s, dp) + f(i-1, 1, s, dp)) % mod;
        }
    } else if (j == 1) {
        switch (b) {
            case '0':
            case '1':
            case '2':
                return res = 0;
            default:
                return res = f(i-1, 2, s, dp);
        }
    } else { // j == 2
        switch (b) {
            case '0':
                return res = 0;
            case '1':
                return res = f(i-1, 0, s, dp);
            case '2':
                return res = f(i-1, 1, s, dp);
            case '*':
                return res = f(i-1, 2, s, dp);
            default:
                return res = (f(i-1, 0, s, dp) + f(i-1, 1, s, dp) + f(i-1, 2, s, dp)) % mod;
        }
    }
}

signed main() {
    ios::sync_with_stdio(false);
    std::cin.tie(0);
    string s;
    cin >> s;
    int n = s.size();
    vector<vector<int>> dp(n, vector<int>(3, -1));
    int result = (f(n-1, 0, s, dp) + f(n-1, 1, s, dp)) % mod;
    cout << result << endl;
    return 0;
}

```

import java.util.*;

public class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        long res = 0;
        String str = sc.next();
        int n = str.length(); // 表示字符串的数量

        long mod = 1000000007;

        char[] ch = str.toCharArray();
    
        for(int i = 0; i < n; i++){
            if(ch[i] == '2'){
                ch[i-1] = '*';
                ch[i+1] = '*';
            }else if(ch[i] == '0'){
                if(i > 0 && ch[i-1] == '?')
                    ch[i-1] = 'x';
                if(i+1 < n && ch[i+1]== '?')
                    ch[i+1] = 'x';

            }else if(ch[i] == '1'){
                if(i > 0 && i+1 < n && ch[i-1] == '?' && ch[i+1] == '?'){
                    ch[i-1] = '-';
                    ch[i+1] = '-';
                }else if(i > 0 && ch[i-1] == '?'){
                        ch[i-1] = 'x';
                }else if(i+1 < n && ch[i+1] == '?'){
                        ch[i+1] = 'x';
                }
            }
        }

        for(int i = 0; i < n; i++){
            if(ch[i] == '?')
                res += 1;
            else if(ch[i] == '-')
                res += 0.5;
        }
        // for(int i = 0; i < n; i++){
        //     System.out.println(ch[i]);
        // }
        long temp = 1;
        for(int i = 0; i < res; i++){
            temp = (temp * 2) % mod;
        }
        System.out.println(temp);
    }
}
