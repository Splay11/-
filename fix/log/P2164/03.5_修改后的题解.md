## 解题思路

每个 $a_i<2^{30}$，按位独立考虑。对每一位，维护：

- $pre1[i][j]$：从 $i$ 往左、在与 $i$ 同奇偶的位置上，最近一次第 $j$ 位为 `1` 的下标；
- $pre0[i][j]$：同理，最近一次为 `0` 的下标。

因为 $\\&0$ 会把该位清零，$|1$ 会把该位置一，而 $\\&1$ 与 $|0$ 不改变该位，所以只需比较区间内“有效的 $0$”和“有效的 $1$”谁更靠后。

$op=1$ 从 $\\&$ 起交替，$op=2$ 从 $|$ 起交替。单点、两点区间需要特判。按位拼出答案。

## 复杂度分析

时间复杂度 $O((n+q)\cdot 30)$，空间复杂度 $O(n\cdot 30)$。

## 代码实现

### Python

```python
N = 200005
pre1 = [[0] * 32 for _ in range(N)]
pre0 = [[0] * 32 for _ in range(N)]

n, q = map(int, input().split())
a = [0] + list(map(int, input().split()))

for j in range(30):
    for i in range(1, n + 1):
        if i >= 3:
            pre1[i][j] = pre1[i - 2][j]
            pre0[i][j] = pre0[i - 2][j]
        if (a[i] >> j) & 1:
            pre1[i][j] = i
        else:
            pre0[i][j] = i

for _ in range(q):
    op, l, r = map(int, input().split())
    ans = 0
    id = l % 2
    if op == 1:
        for j in range(30):
            if l == r:
                if (a[l] >> j) & 1:
                    ans += 1 << j
            elif pre1[r - ((r % 2) != id)][j] < l:
                continue
            elif pre0[r - ((r % 2) == id)][j] < l:
                if pre1[r - ((r % 2) != id)][j] >= l:
                    ans += 1 << j
            elif pre1[r - ((r % 2) != id)][j] - pre0[r - ((r % 2) == id)][j] > 0:
                ans += 1 << j
    else:
        for j in range(30):
            if l == r:
                if (a[l] >> j) & 1:
                    ans += 1 << j
            elif l == r - 1:
                if (a[l] >> j) & 1 or (a[r] >> j) & 1:
                    ans += 1 << j
            elif pre1[r - ((r % 2) == id)][j] < l and not ((a[l] >> j) & 1):
                continue
            elif pre1[r - ((r % 2) == id)][j] - pre0[r - ((r % 2) != id)][j] > 0 or \
                    (pre0[r - ((r % 2) != id)][j] < l and (a[l] >> j) & 1):
                ans += 1 << j
    print(ans)
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer line1 = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(line1.nextToken());
        int q = Integer.parseInt(line1.nextToken());
        int[] a = new int[n + 1];
        int[][] pre1 = new int[n + 1][32];
        int[][] pre0 = new int[n + 1][32];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        for (int j = 0; j < 30; j++) {
            for (int i = 1; i <= n; i++) {
                if (i >= 3) {
                    pre1[i][j] = pre1[i - 2][j];
                    pre0[i][j] = pre0[i - 2][j];
                }
                if (((a[i] >> j) & 1) == 1) {
                    pre1[i][j] = i;
                } else {
                    pre0[i][j] = i;
                }
            }
        }
        for (int qi = 0; qi < q; qi++) {
            StringTokenizer query = new StringTokenizer(br.readLine());
            int op = Integer.parseInt(query.nextToken());
            int l = Integer.parseInt(query.nextToken());
            int r = Integer.parseInt(query.nextToken());
            int ans = 0;
            int id = l % 2;
            if (op == 1) {
                for (int j = 0; j < 30; j++) {
                    if (l == r) {
                        if (((a[l] >> j) & 1) == 1) {
                            ans += 1 << j;
                        }
                    } else if (pre1[r - ((r % 2 != id) ? 1 : 0)][j] < l) {
                        continue;
                    } else if (pre0[r - ((r % 2 == id) ? 1 : 0)][j] < l) {
                        if (pre1[r - ((r % 2 != id) ? 1 : 0)][j] >= l) {
                            ans += 1 << j;
                        }
                    } else if (pre1[r - ((r % 2 != id) ? 1 : 0)][j] - pre0[r - ((r % 2 == id) ? 1 : 0)][j] > 0) {
                        ans += 1 << j;
                    }
                }
            } else {
                for (int j = 0; j < 30; j++) {
                    if (l == r) {
                        if (((a[l] >> j) & 1) == 1) {
                            ans += 1 << j;
                        }
                    } else if (l == r - 1) {
                        if (((a[l] >> j) & 1) == 1 || ((a[r] >> j) & 1) == 1) {
                            ans += 1 << j;
                        }
                    } else if (pre1[r - ((r % 2 == id) ? 1 : 0)][j] < l && ((a[l] >> j) & 1) == 0) {
                        continue;
                    } else if (pre1[r - ((r % 2 == id) ? 1 : 0)][j] - pre0[r - ((r % 2 != id) ? 1 : 0)][j] > 0
                            || (pre0[r - ((r % 2 != id) ? 1 : 0)][j] < l && ((a[l] >> j) & 1) == 1)) {
                        ans += 1 << j;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
#define N 200005
int a[N];
int pre1[N][32], pre0[N][32];
int n, q, op, l, r;

int main() {
    cin >> n >> q;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int j = 0; j < 30; j++) {
        for (int i = 1; i <= n; i++) {
            if (i >= 3) {
                pre1[i][j] = pre1[i - 2][j];
                pre0[i][j] = pre0[i - 2][j];
            }
            if ((a[i] >> j) & 1) pre1[i][j] = i;
            else pre0[i][j] = i;
        }
    }
    while (q--) {
        cin >> op >> l >> r;
        int ans = 0;
        int id = l % 2;
        if (op == 1) {
            for (int j = 0; j < 30; j++) {
                if (l == r) {
                    if ((a[l] >> j) & 1) ans += 1 << j;
                } else if (pre1[r - ((r % 2) != id)][j] < l) {
                    continue;
                } else if (pre0[r - ((r % 2) == id)][j] < l) {
                    if (pre1[r - ((r % 2) != id)][j] >= l) ans += 1 << j;
                } else if (pre1[r - ((r % 2) != id)][j] - pre0[r - ((r % 2) == id)][j] > 0) {
                    ans += 1 << j;
                }
            }
        } else {
            for (int j = 0; j < 30; j++) {
                if (l == r) {
                    if ((a[l] >> j) & 1) ans += 1 << j;
                } else if (l == r - 1) {
                    if (((a[l] >> j) & 1) || ((a[r] >> j) & 1)) ans += 1 << j;
                } else if (pre1[r - ((r % 2) == id)][j] < l && !((a[l] >> j) & 1)) {
                    continue;
                } else if (pre1[r - ((r % 2) == id)][j] - pre0[r - ((r % 2) != id)][j] > 0
                           || (pre0[r - ((r % 2) != id)][j] < l && ((a[l] >> j) & 1))) {
                    ans += 1 << j;
                }
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
