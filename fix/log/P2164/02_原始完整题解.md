# 题目大意
给定一个长度为n的数组a和q次查询，每次查询有三种操作：

对于查询(op=1)，从al开始，依次交替进行按位与(&)和按位或(|)运算，直到ar。
对于查询(op=2)，从al开始，依次交替进行按位或(|)和按位与(&)运算，直到ar。
每次查询后输出计算结果。
# 思路
使用位运算和预处理：

因为每个元素都是小于 2^30 的整数，可以将每个元素看成30位的二进制表示，并按位分别处理每一位的 & 和 | 运算。
我们维护两个预处理数组 pre1 和 pre0 来帮助计算：
pre1[i][j]：表示从位置 i 开始，距离 i 最近的（同类间隔）为 1 的下标（针对第 j 位）。$\\$
pre0[i][j]：表示从位置 i 开始，距离 i 最近的（同类间隔）为 0 的下标（针对第 j 位）。$\\$
通过预处理数组，可以快速判断在一个范围内某一位是否可以通过连续的 & 和 | 运算得到 1 或 0。
需要注意的是op等于2的时候al也是|的，特殊判断一下，(1或0)&1还是他本身，(1或0)|0还是他本身。所以&0和|1才是有效的，找到哪个在前面即可
## 代码如下
### cpp
```cpp
#include<bits/stdc++.h>
using namespace std;

#define N 200005  // 定义数组大小为 200005
int a[N];  // 存储输入的数组
int pre1[N][32], pre0[N][32];  // 预处理数组：pre1[i][j] 表示i位置j位是1的最近位置，pre0[i][j] 表示i位置j位是0的最近位置
int n, q;  // n 表示数组大小，q 表示查询次数
int op, l, r;  // 查询的操作类型以及区间 [l, r]
int ans = 0;  // 每次查询的结果

int main() {
    // 输入 n 和 q
    cin >> n >> q;

    // 输入数组 a
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    // 预处理：分别记录每一位二进制上1和0出现的最近位置
    for (int j = 0; j < 30; j++) {  // 针对每个数字的30个二进制位
        for (int i = 1; i <= n; i++) {
            // 如果 i >= 3，直接继承前两个位置的 pre1 和 pre0 状态
            if (i >= 3) {
                pre1[i][j] = pre1[i - 2][j];
                pre0[i][j] = pre0[i - 2][j];
            }
            // 根据当前数的第 j 位是 1 还是 0 更新 pre1 和 pre0
            if ((a[i] >> j) & 1)
                pre1[i][j] = i;  // 如果第 j 位是1，更新 pre1[i][j] 为当前 i
            else
                pre0[i][j] = i;  // 如果第 j 位是0，更新 pre0[i][j] 为当前 i
        }
    }

    // 处理 q 次查询
    while (q--) {
        // 输入每次查询的操作类型 op 和区间 [l, r]
        cin >> op >> l >> r;
        ans = 0;  // 每次查询的结果

        int id = l % 2;  // 记录 l 的奇偶性，用于判断 & 和 | 的起始操作

        if (op == 1) {  // 操作1：以 & 开始，交替进行 & 和 |
            for (int j = 0; j < 30; j++) {  // 针对每个数字的30个二进制位
                if (l == r) {  // 特殊情况：只有一个元素
                    if ((a[l] >> j) & 1) {  // 如果该位是1，则将其加入 ans
                        ans += 1 << j;
                    }
                }
                else if (pre1[r - ((r % 2) != id)][j] < l) {
                    // 如果最后一个满足 & 的 1 位置在 l 之前，跳过此位
                    continue;
                }
                else if (pre0[r - ((r % 2) == id)][j] < l) {
                    // 如果0的位置在l之前，且1的位置在l之后，加上此位的值
                    if (pre1[r - ((r % 2) != id)][j] >= l)
                        ans += 1 << j;
                }
                else if (pre1[r - ((r % 2) != id)][j] - pre0[r - ((r % 2) == id)][j] > 0) {
                    // 如果满足条件，加上此位的值
                    ans += 1 << j;
                }
            }
        }
        else {  // 操作2：以 | 开始，交替进行 | 和 &
            for (int j = 0; j < 30; j++) {  // 针对每个数字的30个二进制位
                if (l == r) {  // 特殊情况：只有一个元素
                    if ((a[l] >> j) & 1) {  // 如果该位是1，则将其加入 ans
                        ans += 1 << j;
                    }
                }
                else if (l == r - 1) {  // 特殊情况：只有两个元素
                    if ((a[l] >> j) & 1 || (a[r] >> j) & 1) {  // 如果任意一个位是1，则加入 ans
                        ans += 1 << j;
                    }
                }
                else if (pre1[r - ((r % 2) == id)][j] < l && !((a[l] >> j) & 1)) {
                    // 如果该位上没有 1 且 l 到 r 范围内没有 1，则跳过
                    continue;
                }
                else if (pre1[r - ((r % 2) == id)][j] - pre0[r - ((r % 2) != id)][j] > 0 || 
                         (pre0[r - ((r % 2) != id)][j] < l && ((a[l] >> j) & 1))) {
                    // 如果满足条件，则加上此位的值
                    ans += 1 << j;
                }
            }
        }

        // 输出本次查询结果
        cout << ans << '\n';
    }
}

### python
```python
# 定义常量
N = 200005

# 输入数组和预处理数组
a = [0] * N
pre1 = [[0] * 32 for _ in range(N)]  # 预处理1的数组
pre0 = [[0] * 32 for _ in range(N)]  # 预处理0的数组

# 输入
n, q = map(int, input().split())  # 输入 n 和 q
a = [0] + list(map(int, input().split()))  # 输入数组 a，1-based 索引

# 预处理每一位的最近 1 和 0
for j in range(30):  # 针对每个数字的30个二进制位
    for i in range(1, n + 1):
        if i >= 3:
            # 预处理奇偶交替位的 pre1 和 pre0
            pre1[i][j] = pre1[i - 2][j]
            pre0[i][j] = pre0[i - 2][j]
        if (a[i] >> j) & 1:  # 如果第 j 位是1
            pre1[i][j] = i
        else:  # 如果第 j 位是0
            pre0[i][j] = i

# 处理 q 次查询
for _ in range(q):
    op, l, r = map(int, input().split())  # 输入查询
    ans = 0  # 每次查询的结果
    id = l % 2  # 判断 l 是奇数还是偶数，决定起始是 & 还是 |
    
    if op == 1:  # 操作1：以 & 开始
        for j in range(30):  # 针对每个数字的30个二进制位
            if l == r:  # 特殊情况：只有一个元素
                if (a[l] >> j) & 1:  # 如果该位是1，则将其加入 ans
                    ans += 1 << j
            elif pre1[r - ((r % 2) != id)][j] < l:
                continue  # 跳过此位
            elif pre0[r - ((r % 2) == id)][j] < l:
                if pre1[r - ((r % 2) != id)][j] >= l:
                    ans += 1 << j
            elif pre1[r - ((r % 2) != id)][j] - pre0[r - ((r % 2) == id)][j] > 0:
                ans += 1 << j
    else:  # 操作2：以 | 开始
        for j in range(30):  # 针对每个数字的30个二进制位
            if l == r:  # 特殊情况：只有一个元素
                if (a[l] >> j) & 1:  # 如果该位是1，则将其加入 ans
                    ans += 1 << j
            elif l == r - 1:  # 特殊情况：只有两个元素
                if (a[l] >> j) & 1 or (a[r] >> j) & 1:
                    ans += 1 << j
            elif pre1[r - ((r % 2) == id)][j] < l and not ((a[l] >> j) & 1):
                continue  # 跳过不满足条件的位
            elif pre1[r - ((r % 2) == id)][j] - pre0[r - ((r % 2) != id)][j] > 0 or \
                 (pre0[r - ((r % 2) != id)][j] < l and (a[l] >> j) & 1):
                ans += 1 << j
    
    # 输出查询结果
    print(ans)

```java
import java.util.*;
import java.io.*;

public class Main {
    
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer line1 = new StringTokenizer(br.readLine());
        
        int n = Integer.parseInt(line1.nextToken());
        int q = Integer.parseInt(line1.nextToken());
        
        int[] a = new int[n+1];
        int[][] pre1 = new int[n+1][32];  // 预处理1的数组
        int[][] pre0 = new int[n+1][32];  // 预处理0的数组

        // 输入数组 a
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }

        // 预处理每一位的最近 1 和 0
        for (int j = 0; j < 30; j++) {
            for (int i = 1; i <= n; i++) {
                if (i >= 3) {
                    // 预处理奇偶交替位的 pre1 和 pre0
                    pre1[i][j] = pre1[i - 2][j];
                    pre0[i][j] = pre0[i - 2][j];
                }
                if ((a[i] >> j & 1) == 1)  // 如果第 j 位是1
                    pre1[i][j] = i;
                else  // 如果第 j 位是0
                    pre0[i][j] = i;
            }
        }

        // 处理 q 次查询
        for (int i = 0; i < q; i++) {
            StringTokenizer query = new StringTokenizer(br.readLine());
            int op = Integer.parseInt(query.nextToken());
            int l = Integer.parseInt(query.nextToken());
            int r = Integer.parseInt(query.nextToken());
            
            int ans = 0;  // 每次查询的结果
            int id = l % 2;  // 判断 l 是奇数还是偶数，决定起始是 & 还是 |
            
            if (op == 1) {  // 操作1：以 & 开始
                for (int j = 0; j < 30; j++) {  // 针对每个数字的30个二进制位
                    if (l == r) {  // 特殊情况：只有一个元素
                        if ((a[l] >> j & 1) == 1) {  // 如果该位是1，则将其加入 ans
                            ans += 1 << j;
                        }
                    } else if (pre1[r - ((r % 2 != id) ? 1 : 0)][j] < l) {
                        continue;  // 跳过此位
                    } else if (pre0[r - ((r % 2 == id) ? 1 : 0)][j] < l) {
                        if (pre1[r - ((r % 2 != id) ? 1 : 0)][j] >= l)
                            ans += 1 << j;
                    } else if (pre1[r - ((r % 2 != id) ? 1 : 0)][j] - pre0[r - ((r % 2 == id) ? 1 : 0)][j] > 0) {
                        ans += 1 << j;
                    }
                }
            } else {  // 操作2：以 | 开始
                for (int j = 0; j < 30; j++) {  // 针对每个数字的30个二进制位
                    if (l == r) {  // 特殊情况：只有一个元素
                        if ((a[l] >> j & 1) == 1) {  // 如果该位是1，则将其加入 ans
                            ans += 1 << j;
                        }
                    } else if (l == r - 1) {  // 特殊情况：只有两个元素
                        if ((a[l] >> j & 1) == 1 || (a[r] >> j & 1) == 1) {
                            ans += 1 << j;
                        }
                    } else if (pre1[r - ((r % 2 == id) ? 1 : 0)][j] < l && (a[l] >> j & 1) == 0) {
                        continue;  // 跳过不满足条件的位
                    } else if (pre1[r - ((r % 2 == id) ? 1 : 0)][j] - pre0[r - ((r % 2 != id) ? 1 : 0)][j] > 0 || 
                               (pre0[r - ((r % 2 != id) ? 1 : 0)][j] < l && (a[l] >> j & 1) == 1)) {
                        ans += 1 << j;
                    }
                }
            }
            
            // 输出查询结果
            System.out.println(ans);
        }
    }
}
