## 解题思路

我们需要构造一个 $1$ 到 $n$ 的排列，使得它的逆序对总数（混乱度）恰好等于 $k$。  
最大可能的逆序对数为 $\frac{n(n-1)}{2}$，题目保证给出的 $k$ 在合法范围内。

可以采用贪心策略从大到小放置数字。设当前剩余未使用的数字集合为 $\{1, 2, \dots, s\}$（$s$ 初始为 $n$）。

- 如果我们将剩余数字中的最大值 $s$ 放在当前答案序列的下一个位置，由于它比后面所有 $s-1$ 个数字都大，因此会与它们形成 $s-1$ 个逆序对。
- 贪心思路：只要当前还需要构造的逆序对数 $k \ge s - 1$ 且 $s > 1$，就将 $s$ 放入答案，令 $k \leftarrow k - (s-1)$，并缩减剩余集合 $s \leftarrow s - 1$，继续尝试放入下一个最大值。
- 当 $k < s - 1$ 或 $s = 1$ 时，无法再通过放下最大值并消耗 $s-1$ 个逆序对的方式继续构造。此时我们在剩余的数字 $1 \dots s$ 上直接构造出恰好 $k$ 个逆序对的片段：
  - 若 $k = 0$：剩余数字按递增顺序排列，即 $1, 2, \dots, s$，逆序对数为 $0$。
  - 若 $k > 0$：将剩余数字排列为 $[k+1,\ 1,\ 2,\ \dots,\ k,\ k+2,\ \dots,\ s]$。这个片段中，第一个数字 $k+1$ 会与后面的 $1, 2, \dots, k$ 形成恰好 $k$ 个逆序对，而后方所有数字保持递增，不再产生新的逆序对。

将构造出的片段拼接到已获得的答案序列末尾，即可得到满足条件的完整排列。

由于每个数字最多被处理和放置一次，整个过程的时间复杂度为 $O(n)$。

## 复杂度分析

- 时间复杂度：对于每个测试用例，我们需要 $O(n)$ 的时间构造答案序列。所有测试用例的 $n$ 之和不超过 $2 \times 10^5$，因此总时间复杂度为 $O(\sum n)$，可以满足要求。
- 空间复杂度：需要存储长度为 $n$ 的答案序列，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    T = int(next(it))
    out_lines = []
    for _ in range(T):
        n = int(next(it))
        k = int(next(it))
        ans = []
        s = n
        while s >= 1:
            if k >= s - 1 and s > 1:
                ans.append(s)
                k -= s - 1
                s -= 1
            else:
                if k == 0:
                    ans.extend(range(1, s + 1))
                else:
                    ans.append(k + 1)
                    ans.extend(range(1, k + 1))
                    ans.extend(range(k + 2, s + 1))
                break
        out_lines.append(' '.join(map(str, ans)))
    sys.stdout.write('\n'.join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int n = Integer.parseInt(st.nextToken());
            long k = Long.parseLong(st.nextToken());
            ArrayList<Integer> ans = new ArrayList<>();
            int s = n;
            while (s >= 1) {
                if (k >= s - 1 && s > 1) {
                    ans.add(s);
                    k -= (s - 1);
                    s--;
                } else {
                    if (k == 0) {
                        for (int i = 1; i <= s; i++) {
                            ans.add(i);
                        }
                    } else {
                        ans.add((int) (k + 1));
                        for (int i = 1; i <= k; i++) {
                            ans.add(i);
                        }
                        for (int i = (int) (k + 2); i <= s; i++) {
                            ans.add(i);
                        }
                    }
                    break;
                }
            }
            for (int i = 0; i < ans.size(); i++) {
                if (i > 0) sb.append(' ');
                sb.append(ans.get(i));
            }
            sb.append('\n');
        }
        System.out.print(sb);
    }
}
```

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        int n;
        long long k;
        cin >> n >> k;
        vector<int> ans;
        int s = n;
        while (s >= 1) {
            if (k >= s - 1 && s > 1) {
                ans.push_back(s);
                k -= (s - 1);
                --s;
            } else {
                if (k == 0) {
                    for (int i = 1; i <= s; ++i)
                        ans.push_back(i);
                } else {
                    ans.push_back((int)(k + 1));
                    for (int i = 1; i <= k; ++i)
                        ans.push_back(i);
                    for (int i = (int)(k + 2); i <= s; ++i)
                        ans.push_back(i);
                }
                break;
            }
        }
        for (size_t i = 0; i < ans.size(); ++i) {
            if (i > 0) cout << ' ';
            cout << ans[i];
        }
        cout << '\n';
    }
    return 0;
}
