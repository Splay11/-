## 解题思路

本题要求统计给定标识符中满足以下“优雅”条件的数量：

- 首字符必须是大写英文字母；
- 末字符必须是小写英文字母；
- 对于任意字母（例如 a 和 A），其大写形式与小写形式不能同时出现在标识符中。

我们可以依次检查每个标识符：

1. **首尾条件检查**  
   直接判断字符串的第一个字符是否为大写字母（`'A'` ~ `'Z'`），最后一个字符是否为小写字母（`'a'` ~ `'z'`），若不满足则直接跳过该标识符。

2. **大小写冲突检查（位掩码法）**  
   使用两个长度为 26 位的整数掩码 `lowerMask` 和 `upperMask`，分别记录标识符中出现过的小写字母和大写字母。  
   扫描字符串中的每一个字符：
   - 若是小写字母（如 `c`），计算其索引 `idx = c - 'a'`，检查 `upperMask` 的第 `idx` 位是否已经置位：若置位，说明同一字母的大写已经出现过，发生冲突，该标识符不优雅；否则将 `lowerMask` 的第 `idx` 位置位。
   - 若是大写字母，同理计算 `idx = c - 'A'`，检查 `lowerMask` 的第 `idx` 位是否已经置位：若置位，发生冲突；否则将 `upperMask` 的第 `idx` 位置位。
   若扫描过程中未发现冲突，则该标识符为优雅标识符。

3. **统计答案**  
   对每个满足条件的标识符计数，最终输出总数。

该方法只需要一次线性扫描即可完成判断，且使用的掩码操作非常高效。

## 复杂度分析

- 时间复杂度：$O(\sum_{i=1}^n l_i)$，即所有标识符的总长度。每个字符仅被遍历一次，每次掩码检查与更新均为 $O(1)$ 时间。
- 空间复杂度：$O(1)$，仅使用了固定数量的整数变量，不随输入规模增长。

## 代码实现

### Python

```python
import sys

def is_elegant(s: str) -> bool:
    # 首字符必须是大写，末字符必须是小写
    if not s or not ('A' <= s[0] <= 'Z'):
        return False
    if not ('a' <= s[-1] <= 'z'):
        return False

    lower_mask = 0
    upper_mask = 0
    for ch in s:
        if 'a' <= ch <= 'z':
            idx = ord(ch) - ord('a')
            if (upper_mask >> idx) & 1:  # 同一字母的大写已出现过，冲突
                return False
            lower_mask |= (1 << idx)
        elif 'A' <= ch <= 'Z':
            idx = ord(ch) - ord('A')
            if (lower_mask >> idx) & 1:  # 同一字母的小写已出现过，冲突
                return False
            upper_mask |= (1 << idx)
        # 根据题意，字符串仅由字母组成，无需处理其他情况
    return True

def main():
    data = sys.stdin.read().strip().splitlines()
    it = iter(data)
    n = int(next(it))
    elegant_cnt = 0
    for _ in range(n):
        length = int(next(it))      # 读取长度但不强制使用
        s = next(it).strip()
        if is_elegant(s):
            elegant_cnt += 1
    print(elegant_cnt)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;

public class Main {
    static boolean isElegant(String s) {
        // 首字符必须是大写，末字符必须是小写
        if (s == null || s.isEmpty()) return false;
        char first = s.charAt(0);
        char last = s.charAt(s.length() - 1);
        if (first < 'A' || first > 'Z') return false;
        if (last < 'a' || last > 'z') return false;

        int lowerMask = 0, upperMask = 0;
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            if (ch >= 'a' && ch <= 'z') {
                int idx = ch - 'a';
                if ((upperMask & (1 << idx)) != 0) return false; // 冲突
                lowerMask |= (1 << idx);
            } else if (ch >= 'A' && ch <= 'Z') {
                int idx = ch - 'A';
                if ((lowerMask & (1 << idx)) != 0) return false; // 冲突
                upperMask |= (1 << idx);
            }
            // 字符串仅由字母组成，无需处理其他情况
        }
        return true;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        int n = Integer.parseInt(line.trim());
        int elegantCnt = 0;
        for (int i = 0; i < n; i++) {
            // 读取长度
            String lenLine = br.readLine();
            while (lenLine != null && lenLine.isEmpty()) lenLine = br.readLine();
            // int length = Integer.parseInt(lenLine.trim()); // 此处可读取但不使用
            // 读取字符串
            String s = br.readLine();
            while (s != null && s.isEmpty()) s = br.readLine();
            if (isElegant(s)) {
                elegantCnt++;
            }
        }
        System.out.println(elegantCnt);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

bool isElegant(const string& s) {
    // 首字符必须是大写，末字符必须是小写
    if (s.empty() || !(s[0] >= 'A' && s[0] <= 'Z'))
        return false;
    if (!(s.back() >= 'a' && s.back() <= 'z'))
        return false;

    int lowerMask = 0, upperMask = 0;
    for (char ch : s) {
        if (ch >= 'a' && ch <= 'z') {
            int idx = ch - 'a';
            if (upperMask & (1 << idx)) // 同一字母的大写已出现，冲突
                return false;
            lowerMask |= (1 << idx);
        } else if (ch >= 'A' && ch <= 'Z') {
            int idx = ch - 'A';
            if (lowerMask & (1 << idx)) // 同一字母的小写已出现，冲突
                return false;
            upperMask |= (1 << idx);
        }
        // 字符串仅由字母组成，无需处理其他情况
    }
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    int elegantCnt = 0;
    for (int i = 0; i < n; ++i) {
        int length;
        string s;
        cin >> length >> s; // 读入长度（可忽略）和字符串
        if (isElegant(s))
            ++elegantCnt;
    }
    cout << elegantCnt << '\n';
    return 0;
}
```
