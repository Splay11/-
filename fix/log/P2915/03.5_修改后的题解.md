## 解题思路

1. **定义纯净数**  
   称一个正整数 $x$ 为纯净数，当且仅当其十进制表示中不出现数字 `2` 和 `4`。  
   用一个指示函数 $f[x]$ 表示 $x$ 是否为纯净数：  
   $$
   f[x] = \begin{cases}
   1 & \text{若 }x\text{ 是纯净数},\\
   0 & \text{否则}.
   \end{cases}
   $$

2. **生成函数**  
   构造形式幂级数（生成函数）  
   $$
   G(t) = \sum_{i=0}^{\infty} f[i]\, t^i .
   $$  
   则 $G^3(t)$ 中 $t^n$ 的系数表示将 $n$ 拆分为三个纯净数（允许相等且有序）的方法数，记为 $E_1[n]$。

3. **容斥原理计算互异有序三元组**  
   - 令 $E_1[n]$ 为有序三元组 $(a,b,c)$ 且允许相等的方案数，即 $G^3$ 中 $t^n$ 的系数。  
   - 令 $E_2[n]$ 为恰有一对相等的有序三元组（例如 $a = b \neq c$）的方案数。可以由 $G(t) \cdot G(t^2)$ 中 $t^n$ 的系数得到。  
   - 令 $E_3[n]$ 为三数全相等的方案数，即 $n$ 能被 $3$ 整除且 $n/3$ 是纯净数时取 $1$，否则为 $0$。  

   根据容斥原理，有序且两两互异的三元组数量为：  
   $$
   F[n] = E_1[n] - 3E_2[n] + 2E_3[n].
   $$

4. **无序三元组计数**  
   有序互异三元组的每一种排列会被计算 $3! = 6$ 次，因此无序且两两不同的三元组数量为：  
   $$
   \text{ans}[n] = \frac{F[n]}{6}.
   $$

5. **利用 FFT 加速卷积**  
   - 先预处理出所有不超过询问最大值 $\max(n)$ 的纯净数标识 $f[i]$。  
   - 构造数组 $A[i] = f[i]$ 和 $B[2i] = f[i]$，分别对应 $G(t)$ 和 $G(t^2)$ 的系数。  
   - 对 $A$ 和 $B$ 做快速傅里叶变换（FFT），计算 $A^3$ 得到 $E_1$，计算 $A \cdot B$ 得到 $E_2$。  
   - 结合 $E_3$ 与容斥公式，即可一次性回答所有询问。

---

## 复杂度分析

- **预处理纯净数**：需要检查 $1$ 到 $\max(n)$ 的每一个数，每个数扫描其十进制位，总体时间复杂度为 $O(N \log_{10} N)$，其中 $N = \max(n) \le 10^6$，常数极小，可以忽略不计。
- **FFT 卷积**：FFT 的长度 $L$ 取大于 $3N$ 的最小 $2$ 的幂，$L = O(N)$。一次 FFT 的复杂度为 $O(L \log L)$，而总共需要 $3$ 次正变换和 $2$ 次逆变换，因此时间复杂度为 $O(N \log N)$。对于 $N = 10^6$，该部分在时限内完全可行。
- **总时间复杂度**：$O(N \log N + T)$，其中 $T$ 是询问组数，$T \le 10^4$，回答询问为 $O(1)$。
- **空间复杂度**：需要存储 $A,B$ 及其中间变换结果，长度为 $O(N)$，因此空间复杂度为 $O(N)$。

## 代码实现

### Python

```python
import sys
import threading
import numpy as np

def main():
    input = sys.stdin.readline
    T = int(input().strip())
    qs = [int(input().strip()) for _ in range(T)]
    mx = max(qs)

    # 标记纯净数（不包含数字2和4）
    good = np.zeros(mx + 1, dtype=int)
    for i in range(1, mx + 1):
        x = i
        ok = True
        while x > 0:
            d = x % 10
            if d == 2 or d == 4:
                ok = False
                break
            x //= 10
        good[i] = ok

    # 确定FFT所需长度
    lim = 1
    while lim < 3 * mx + 1:
        lim <<= 1

    A = np.zeros(lim, dtype=np.int64)
    B = np.zeros(lim, dtype=np.int64)
    A[:mx + 1] = good
    B[:2 * mx + 1:2] = good

    # FFT 计算卷积
    FA = np.fft.rfft(A)
    FB = np.fft.rfft(B)

    # E1 = A * A * A
    E1 = np.fft.irfft(FA * FA * FA, n=lim).round().astype(np.int64)[:mx + 1]
    # E2 = A * B
    E2 = np.fft.irfft(FA * FB, n=lim).round().astype(np.int64)[:mx + 1]

    # 预处理所有答案
    ans = [0] * (mx + 1)
    for n in range(mx + 1):
        e3 = 1 if n % 3 == 0 and good[n // 3] == 1 else 0
        tot = E1[n] - 3 * E2[n] + 2 * e3
        ans[n] = tot // 6

    # 输出
    print("\n".join(str(ans[q]) for q in qs))

if __name__ == "__main__":
    threading.Thread(target=main).start()
```

```java
import java.util.*;

public class Main {
    static class Complex {
        double real, imag;

        Complex(double real, double imag) {
            this.real = real;
            this.imag = imag;
        }

        Complex add(Complex other) {
            return new Complex(this.real + other.real, this.imag + other.imag);
        }

        Complex subtract(Complex other) {
            return new Complex(this.real - other.real, this.imag - other.imag);
        }

        Complex multiply(Complex other) {
            return new Complex(this.real * other.real - this.imag * other.imag,
                    this.imag * other.real + this.real * other.imag);
        }

        Complex divide(double scalar) {
            return new Complex(this.real / scalar, this.imag / scalar);
        }
    }

    static void fft(List<Complex> a, boolean inv) {
        int n = a.size(), j = 0;
        for (int i = 1; i < n; ++i) {
            int bit = n >> 1;
            for (; (j & bit) != 0; bit >>= 1) j ^= bit;
            j |= bit;
            if (i < j) Collections.swap(a, i, j);
        }
        final double PI = Math.acos(-1);
        for (int len = 2; len <= n; len <<= 1) {
            double ang = 2 * PI / len * (inv ? -1 : 1);
            Complex wlen = new Complex(Math.cos(ang), Math.sin(ang));
            for (int i = 0; i < n; i += len) {
                Complex w = new Complex(1, 0);
                for (int k = 0; k < len / 2; ++k) {
                    Complex u = a.get(i + k);
                    Complex v = a.get(i + k + len / 2).multiply(w);
                    a.set(i + k, u.add(v));
                    a.set(i + k + len / 2, u.subtract(v));
                    w = w.multiply(wlen);
                }
            }
        }
        if (inv) {
            for (int i = 0; i < n; ++i) {
                a.set(i, a.get(i).divide(n));
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        int[] qs = new int[T];
        int mx = 0;
        for (int i = 0; i < T; ++i) {
            qs[i] = sc.nextInt();
            mx = Math.max(mx, qs[i]);
        }

        // 标记纯净数（不包含数字2和4）
        boolean[] good = new boolean[mx + 1];
        for (int i = 1; i <= mx; ++i) {
            int x = i;
            boolean ok = true;
            while (x > 0) {
                int d = x % 10;
                if (d == 2 || d == 4) {
                    ok = false;
                    break;
                }
                x /= 10;
            }
            good[i] = ok;
        }

        int lim = 1;
        while (lim < 3 * mx + 1) lim <<= 1;

        List<Complex> A = new ArrayList<>(Collections.nCopies(lim, new Complex(0, 0)));
        List<Complex> B = new ArrayList<>(Collections.nCopies(lim, new Complex(0, 0)));
        for (int i = 0; i <= mx; ++i) {
            if (good[i]) {
                A.set(i, new Complex(1, 0));
                B.set(2 * i, new Complex(1, 0));
            }
        }

        fft(A, false);
        fft(B, false);

        long[] E1 = new long[mx + 1];
        long[] E2 = new long[mx + 1];

        {   // E1 = A * A * A
            List<Complex> F = new ArrayList<>(A);
            for (int i = 0; i < lim; ++i) {
                F.set(i, F.get(i).multiply(F.get(i)).multiply(A.get(i)));
            }
            fft(F, true);
            for (int i = 0; i <= mx; ++i) {
                E1[i] = Math.round(F.get(i).real);
            }
        }

        {   // E2 = A * B
            List<Complex> F = new ArrayList<>(A);
            for (int i = 0; i < lim; ++i) {
                F.set(i, F.get(i).multiply(B.get(i)));
            }
            fft(F, true);
            for (int i = 0; i <= mx; ++i) {
                E2[i] = Math.round(F.get(i).real);
            }
        }

        for (int n : qs) {
            long e3 = (n % 3 == 0 && good[n / 3]) ? 1 : 0;
            long tot = E1[n] - 3 * E2[n] + 2 * e3;
            System.out.println(tot / 6);
        }

        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using cd = complex<double>;

// 自底向上FFT
void fft(vector<cd>& a, bool inv) {
    int n = a.size(), j = 0;
    for (int i = 1; i < n; ++i) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j |= bit;
        if (i < j) swap(a[i], a[j]);
    }
    const double PI = acos(-1);
    for (int len = 2; len <= n; len <<= 1) {
        double ang = 2 * PI / len * (inv ? -1 : 1);
        cd wlen(cos(ang), sin(ang));
        for (int i = 0; i < n; i += len) {
            cd w(1);
            for (int j = 0; j < len / 2; ++j) {
                cd u = a[i + j];
                cd v = a[i + j + len / 2] * w;
                a[i + j] = u + v;
                a[i + j + len / 2] = u - v;
                w *= wlen;
            }
        }
    }
    if (inv) {
        for (cd& x : a) x /= n;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    vector<int> qs(T);
    int mx = 0;
    for (int i = 0; i < T; ++i) {
        cin >> qs[i];
        mx = max(mx, qs[i]);
    }

    // 标记纯净数（不包含数字2和4）
    vector<char> good(mx + 1, 0);
    for (int i = 1; i <= mx; ++i) {
        int x = i;
        bool ok = true;
        while (x) {
            int d = x % 10;
            if (d == 2 || d == 4) {
                ok = false;
                break;
            }
            x /= 10;
        }
        good[i] = ok;
    }

    // 确定FFT长度
    int lim = 1;
    while (lim < 3 * mx + 1) lim <<= 1;

    vector<cd> A(lim), B(lim);
    for (int i = 0; i <= mx; ++i) {
        if (good[i]) {
            A[i] = 1;
            B[2 * i] = 1;
        }
    }

    fft(A, false);
    fft(B, false);

    vector<long long> E1(mx + 1), E2(mx + 1);

    {   // E1 = A * A * A
        vector<cd> F = A;
        for (int i = 0; i < lim; ++i) F[i] *= F[i] * A[i];
        fft(F, true);
        for (int i = 0; i <= mx; ++i)
            E1[i] = llround(F[i].real());
    }

    {   // E2 = A * B
        vector<cd> F = A;
        for (int i = 0; i < lim; ++i) F[i] *= B[i];
        fft(F, true);
        for (int i = 0; i <= mx; ++i)
            E2[i] = llround(F[i].real());
    }

    // 回答询问
    for (int n : qs) {
        long long e3 = (n % 3 == 0 && good[n / 3]) ? 1 : 0;
        long long tot = E1[n] - 3 * E2[n] + 2 * e3;
        cout << tot / 6 << "\n";
    }

    return 0;
}
```
