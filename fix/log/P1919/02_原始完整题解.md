## 思路
此题如果使用暴力解法时间复杂度将会是阶乘级，肯定不满足，所以我们使用状压动态规划来解决此题.这里建议先学习前置知识:[二进制集合操作](https://oi-wiki.org/math/binary-set/)

首先对于数n中的数为进行单独看待，最多有15个数位，同时m最大为100，将15个数位中的每一个数选择以及不选择的排列组合视作一个状态，并用二进制的形式压缩为整数表示，并对任意一个状态，统计对于m取模值为j的组合数量，即定义 $f[i][j]$ 为在n中选了下标二进制集合为 i 且模 m 为 j 的数的组合数量。

状态的转移即为在所有的数位中挑选一个未曾选择的数位加入到数字的最后一位然后进行状态更新，可以证明此转移可以枚举到所有情况。

但是如此转移有一个**错误**，那就是会将所有数位都视作不同，但是实际情况如果数位中的数字相同那么这两个数位便相同，所以我们要进行去重，对于相同的数位，我们要除去相同数量的阶乘，这就是同一种数位之间位置的排列组合种数，相同数位位置的排列组合数字都是相同，被视作同一个数字

```cpp
void solve() {
    string n;
    cin >> n;
    int m;
    cin >> m;
    vector<array<ll, 100>> f(1 << n.size()); //定义 f[i][j] 为在n中选了下标二进制集合为 i 且模 m 为 j 的数的数量
    for (int i = 1; i < (1 << n.size()); i++) {
        if (lowbit(i) == i && n[__lg(i)] > '0') f[i][(n[__lg(i)] - '0') % m] = 1;
        //当只选一位 1 << i 的时候，显然只有选 n[i] 这一种可能，特判掉前缀0
        for (int j = 0; j < n.size(); j++) { //考虑转移到 i | (1 << j)
            if (i >> j & 1) continue; //如果当前已经选了 j 这一位显然不能重复选
            for (int k = 0; k < m; k++) { //枚举当前状态 i 模 m 的值
                f[i | (1 << j)][(k * 10 + n[j] - '0') % m] += f[i][k]; //在状态 i 模 m 为 k 的所有数后面追加 n[j]，转移到 f[i | (1 << j)][(k * 10 + n[j] - '0') % m]
            }
        }
    }
    ll ans = f[(1 << n.size()) - 1][0]; // 获取答案

    array<int, 10> cnt = {0};
    //如果把去重部分放在 dp 里面的话很麻烦
    //但是注意到本题计算的都是 n 的排列，里面数位的出现频率不变
    //此时每个去重后唯一的数字都在当前答案内贡献了 cnt[i]的阶乘之积 次
    //这个也很容易理解，就是假设数字里一共有 k 个相同的数位 digit
    //那么在之前的 dp 里，因为我们认为相同位上选择的下标不同数字就不同，因此这 k 个数位在 n 所在的索引的全排列都被计入了答案一次
    //所以直接在当前的答案上除就可以得到去重后的答案了
    for (char c : n) cnt[c - '0']++;

    for (int i = 0; i < 10; i++) {
        for (int j = 1; j <= cnt[i]; j++) ans /= j;
    }

    cout << ans << endl;
}

int main(void) {
    ios::sync_with_stdio(false);
    cin.tie(0); cout.tie(0);
    int t = 1;

    while (t--) {
    	solve();
	}
}
```

### python
```cpp
from math import factorial

def solve():
    n = input().strip()
    m = int(input())
    f = [[0] * 100 for _ in range(1 << len(n))]

    for i in range(1, 1 << len(n)):
        if i & -i == i and n[i.bit_length() - 1] > '0':
            f[i][(int(n[i.bit_length() - 1]) % m)] = 1

        for j in range(len(n)):
            if i >> j & 1:
                continue
            for k in range(m):
                f[i | (1 << j)][(k * 10 + int(n[j]) - int('0')) % m] += f[i][k]

    ans = f[(1 << len(n)) - 1][0]

    cnt = [0] * 10
    for c in n:
        cnt[int(c)] += 1

    for i in range(10):
        for j in range(1, cnt[i] + 1):
            ans //= j

    print(ans)

def main():
    t = 1
    while t > 0:
        solve()
        t -= 1

if __name__ == "__main__":
    main()
```

```java
import java.util.Scanner;

public class Main {
    static long lowbit(long x) {
        return x & -x;
    }

    public static void solve() {
        Scanner scanner = new Scanner(System.in);
        String n = scanner.next();
        int m = scanner.nextInt();
        long[][] f = new long[1 << n.length()][100];

        for (int i = 1; i < (1 << n.length()); i++) {
            if (lowbit(i) == i && n.charAt(Long.numberOfTrailingZeros(i)) > '0') {
                f[i][(n.charAt(Long.numberOfTrailingZeros(i)) - '0') % m] = 1;
            }

            for (int j = 0; j < n.length(); j++) {
                if ((i >> j & 1) == 1) continue;
                for (int k = 0; k < m; k++) {
                    f[i | (1 << j)][(k * 10 + n.charAt(j) - '0') % m] += f[i][k];
                }
            }
        }

        long ans = f[(1 << n.length()) - 1][0];

        int[] cnt = new int[10];
        for (char c : n.toCharArray()) {
            cnt[c - '0']++;
        }

        for (int i = 0; i < 10; i++) {
            for (int j = 1; j <= cnt[i]; j++) {
                ans /= j;
            }
        }

        System.out.println(ans);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = 1;

        while (t-- > 0) {
            solve();
        }
    }
}
```
