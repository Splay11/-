## 解题思路

每个站点的配额必须落在 $[l,r]$ 内，共 $n$ 个站点，目标和为 $m$。

可行区间的最小、最大总和分别为

$$
S_{\min}=n\cdot l,\quad S_{\max}=n\cdot r.
$$

仅当 $S_{\min}\le m\le S_{\max}$ 时有解，否则输出 $r+1$。

有解时用贪心构造：先把所有站点都设为下界 $l$，此时总和为 $S_{\min}$，还需要补 $\mathrm{rem}=m-S_{\min}$（必有 $\mathrm{rem}\ge 0$）。从前到后，每个站点最多还能增加 $r-l$，实际增加 $\min(\mathrm{rem},\, r-l)$，直到 $\mathrm{rem}$ 变为 $0$。得到任意一组可行配额即可。

## 复杂度分析

单组构造一次线性遍历，时间复杂度 $O(n)$。所有测试的 $n$ 之和不超过 $2\times 10^5$。空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys

def build_array(n: int, m: int, l: int, r: int):
    smin = n * l
    smax = n * r
    if m < smin or m > smax:
        return None  # 无解
    a = [l] * n  # 先全置为下界
    rem = m - smin
    if rem == 0:
        return a
    cap = r - l  # 每个站点最多还能增加的量
    for i in range(n):
        if rem == 0:
            break
        add = cap if rem >= cap else rem
        a[i] += add
        rem -= add
    return a  # 此时 rem 必为 0

def main():
    data = sys.stdin.read().strip().split()
    t = int(data[0])
    idx = 1
    out_lines = []
    for _ in range(t):
        n = int(data[idx]); m = int(data[idx+1]); l = int(data[idx+2]); r = int(data[idx+3])
        idx += 4
        ans = build_array(n, m, l, r)
        if ans is None:
            out_lines.append(str(r + 1))  # 无解输出 r+1
        else:
            out_lines.append(' '.join(map(str, ans)))
    sys.stdout.write('\n'.join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    public static long[] buildArray(int n, long m, long l, long r) {
        long smin = (long) n * l;
        long smax = (long) n * r;
        if (m < smin || m > smax) return null;

        long[] a = new long[n];
        Arrays.fill(a, l);
        long rem = m - smin;
        if (rem == 0) return a;

        long cap = r - l;
        for (int i = 0; i < n && rem > 0; i++) {
            long add = Math.min(rem, cap);
            a[i] += add;
            rem -= add;
        }
        return a;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) {
            sc.close();
            return;
        }
        int T = sc.nextInt();
        StringBuilder sb = new StringBuilder();
        for (int tc = 0; tc < T; tc++) {
            int n = sc.nextInt();
            long m = sc.nextLong();
            long l = sc.nextLong();
            long r = sc.nextLong();

            long[] ans = buildArray(n, m, l, r);
            if (ans == null) {
                sb.append(r + 1).append('\n');
            } else {
                for (int i = 0; i < n; i++) {
                    if (i > 0) sb.append(' ');
                    sb.append(ans[i]);
                }
                sb.append('\n');
            }
        }
        System.out.print(sb.toString());
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<long long> buildArray(int n, long long m, long long l, long long r) {
    long long smin = 1LL * n * l;
    long long smax = 1LL * n * r;
    if (m < smin || m > smax) return {};

    vector<long long> a(n, l);
    long long rem = m - smin;
    if (rem == 0) return a;

    long long cap = r - l;
    for (int i = 0; i < n && rem > 0; ++i) {
        long long add = min(rem, cap);
        a[i] += add;
        rem -= add;
    }
    return a;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    if (!(cin >> T)) return 0;
    while (T--) {
        int n;
        long long m, l, r;
        cin >> n >> m >> l >> r;

        vector<long long> ans = buildArray(n, m, l, r);
        if (ans.empty()) {
            cout << (r + 1) << '\n';
        } else {
            for (int i = 0; i < n; ++i) {
                if (i) cout << ' ';
                cout << ans[i];
            }
            cout << '\n';
        }
    }
    return 0;
}
```
