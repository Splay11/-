## 解题思路

1. **合法相邻关系**：禁止子串 `'ee'`、`'lm'`、`'ml'`、`'ll'`、`'mm'`，意味着两个 `'e'` 不能相邻，任意两个非 `'e'` 字符也不能相邻。因此合法串必须是 `'e'` 与非 `'e'`（即 `'l'` 或 `'m'`）严格交替。
2. **两种模式**：
   - 模式一：以 `'e'` 开头，形如 $e,x,e,x,\ldots$，需要 $\lceil n/2 \rceil$ 个 `'e'` 和 $\lfloor n/2 \rfloor$ 个非 `'e'`。
   - 模式二：以非 `'e'` 开头，形如 $x,e,x,e,\ldots$，需要 $\lfloor n/2 \rfloor$ 个 `'e'` 和 $\lceil n/2 \rceil$ 个非 `'e'`。
3. **判定**：统计 `'e'`、`'l'`、`'m'` 的个数。若数量恰好匹配某一模式，则按该模式填空；非 `'e'` 位置按剩余个数依次填 `'l'` 再填 `'m'`。否则输出 `No`。

## 复杂度分析

- **时间复杂度**：$O(n)$。计数与构造均为线性。
- **空间复杂度**：$O(n)$。需要存放长度为 $n$ 的答案字符串。

## 代码实现

### Python

```python
# -*- coding: utf-8 -*-
s = input().strip()
n = len(s)
cnt_e = s.count('e')
cnt_l = s.count('l')
cnt_m = s.count('m')
cnt_non = cnt_l + cnt_m

need_e1 = (n + 1) // 2
need_non1 = n // 2
need_non2 = (n + 1) // 2
need_e2 = n // 2

res = [''] * n
possible = False

if cnt_e == need_e1 and cnt_non == need_non1:
    for i in range(n):
        if i % 2 == 0:
            res[i] = 'e'
        else:
            if cnt_l > 0:
                res[i] = 'l'
                cnt_l -= 1
            else:
                res[i] = 'm'
                cnt_m -= 1
    possible = True
elif cnt_e == need_e2 and cnt_non == need_non2:
    for i in range(n):
        if i % 2 == 0:
            if cnt_l > 0:
                res[i] = 'l'
                cnt_l -= 1
            else:
                res[i] = 'm'
                cnt_m -= 1
        else:
            res[i] = 'e'
    possible = True

if possible:
    print(''.join(res))
else:
    print("No")
```

### Java

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        int cntE = 0, cntL = 0, cntM = 0;
        for (char c : s.toCharArray()) {
            if (c == 'e') cntE++;
            else if (c == 'l') cntL++;
            else if (c == 'm') cntM++;
        }
        int cntNon = cntL + cntM;
        int needE1 = (n + 1) / 2;
        int needNon1 = n / 2;
        int needNon2 = (n + 1) / 2;
        int needE2 = n / 2;
        char[] ans = new char[n];
        boolean possible = false;
        if (cntE == needE1 && cntNon == needNon1) {
            for (int i = 0; i < n; i++) {
                if (i % 2 == 0) {
                    ans[i] = 'e';
                } else {
                    if (cntL > 0) {
                        ans[i] = 'l';
                        cntL--;
                    } else {
                        ans[i] = 'm';
                        cntM--;
                    }
                }
            }
            possible = true;
        } else if (cntE == needE2 && cntNon == needNon2) {
            for (int i = 0; i < n; i++) {
                if (i % 2 == 0) {
                    if (cntL > 0) {
                        ans[i] = 'l';
                        cntL--;
                    } else {
                        ans[i] = 'm';
                        cntM--;
                    }
                } else {
                    ans[i] = 'e';
                }
            }
            possible = true;
        }
        if (possible) {
            System.out.println(new String(ans));
        } else {
            System.out.println("No");
        }
    }
}
```

```cpp
#include <iostream>
#include <string>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string s;
    cin >> s;
    int n = s.size();
    int cnt_e = 0, cnt_l = 0, cnt_m = 0;
    for (char c : s) {
        if (c == 'e') cnt_e++;
        else if (c == 'l') cnt_l++;
        else if (c == 'm') cnt_m++;
    }
    int cnt_non = cnt_l + cnt_m;
    int need_e1 = (n + 1) / 2;
    int need_non1 = n / 2;
    int need_non2 = (n + 1) / 2;
    int need_e2 = n / 2;
    string ans;
    bool possible = false;
    if (cnt_e == need_e1 && cnt_non == need_non1) {
        ans.resize(n);
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) ans[i] = 'e';
            else {
                if (cnt_l > 0) { ans[i] = 'l'; cnt_l--; }
                else { ans[i] = 'm'; cnt_m--; }
            }
        }
        possible = true;
    } else if (cnt_e == need_e2 && cnt_non == need_non2) {
        ans.resize(n);
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                if (cnt_l > 0) { ans[i] = 'l'; cnt_l--; }
                else { ans[i] = 'm'; cnt_m--; }
            } else ans[i] = 'e';
        }
        possible = true;
    }
    if (possible) cout << ans;
    else cout << "No";
    return 0;
}
```
