## 解题思路

题目要求构造一个长度为 $n$ 的序列 $a$，使得：
- 所有元素的绝对值恰好是 $1 \sim n$ 的一个排列；
- 所有元素之和为 $0$。

相当于给 $1,2,\dots,n$ 每个数分配一个正号或负号，使得带符号的数之和为 $0$。  
设正数部分之和为 $S_+$，负数绝对值部分之和为 $S_-$，则 $S_+ = S_-$，且 $S_+ + S_- = 1+2+\cdots+n = \frac{n(n+1)}{2}$。  
因此 $\frac{n(n+1)}{2}$ 必须是偶数，即 $n(n+1)$ 能被 $4$ 整除，**无解条件**为 $n \bmod 4 \in \{1,2\}$。此时输出 $-1$。

当有解时，采用以下构造方法：
- 对于每四个连续整数 $x,\ x+1,\ x+2,\ x+3$，可以取符号为 $x,\ -(x+1),\ -(x+2),\ x+3$，它们的和为 $0$。
- 若 $n \bmod 4 = 0$，直接从 $x=1$ 开始，每 $4$ 个一组构造，长度为 $n$ 的序列即满足要求。
- 若 $n \bmod 4 = 3$，先固定前三个数为 $1,\,2,\,-3$（和为 $0$），然后从 $x=4$ 开始，对后面的数按上述四元组构造，直到 $n$。

这样每个测试用例可以在 $O(n)$ 时间内构造出序列（或无解判断）。

## 复杂度分析

对于单个测试用例：
- 判断 $n \bmod 4$ 仅需常数时间；
- 构造序列需要遍历 $1 \sim n$，时间复杂度为 $O(n)$；
- 存储结果需要长度为 $n$ 的数组，空间复杂度为 $O(n)$。

本题有 $q$ 个测试用例，且所有 $n$ 的总和不超过 $2 \times 10^5$，因此总时间复杂度 $O(\sum n) \le O(2 \times 10^5)$，总空间复杂度与最大 $n$ 同阶，均可高效通过。

## 代码实现

### Python

```python
import sys

def build(n):
    # 1..n 的和为奇数时无解，即 n % 4 为 1 或 2
    if n % 4 in (1, 2):
        return None
    res = []
    if n % 4 == 3:
        # n=3 时的基础构造：[1, 2, -3]
        res.extend([1, 2, -3])
        start = 4
    else:
        start = 1
    # 每四个一组：x, -(x+1), -(x+2), x+3
    for x in range(start, n + 1, 4):
        res.extend([x, -(x + 1), -(x + 2), x + 3])
    return res

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    q = int(data[0])
    out_lines = []
    idx = 1
    for _ in range(q):
        n = int(data[idx]); idx += 1
        seq = build(n)
        if seq is None:
            out_lines.append("-1")
        else:
            out_lines.append(" ".join(map(str, seq)))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static int[] build(int n) {
        // 无解情况：n mod 4 = 1 或 2
        if (n % 4 == 1 || n % 4 == 2) return null;
        int[] res = new int[n];
        int idx = 0;
        int start;
        if (n % 4 == 3) {
            // n=3 的基础构造：[1, 2, -3]
            res[idx++] = 1;
            res[idx++] = 2;
            res[idx++] = -3;
            start = 4;
        } else {
            start = 1;
        }
        // 每四个一组：x, -(x+1), -(x+2), x+3
        for (int x = start; x <= n; x += 4) {
            res[idx++] = x;
            res[idx++] = -(x + 1);
            res[idx++] = -(x + 2);
            res[idx++] = x + 3;
        }
        return res;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(br.readLine().trim());
        StringBuilder out = new StringBuilder();
        for (int i = 0; i < q; i++) {
            int n = Integer.parseInt(br.readLine().trim());
            int[] seq = build(n);
            if (seq == null) {
                out.append("-1\n");
            } else {
                for (int j = 0; j < n; j++) {
                    if (j > 0) out.append(' ');
                    out.append(seq[j]);
                }
                out.append('\n');
            }
        }
        System.out.print(out);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> build(int n) {
    // 无解：n(n+1)/2 为奇数，即 n % 4 为 1 或 2
    if (n % 4 == 1 || n % 4 == 2) return {};
    vector<int> res;
    int start;
    if (n % 4 == 3) {
        // n=3 时的基础构造：[1, 2, -3]
        res = {1, 2, -3};
        start = 4;
    } else {
        start = 1;
    }
    // 每四个一组：x, -(x+1), -(x+2), x+3
    for (int x = start; x <= n; x += 4) {
        res.push_back(x);
        res.push_back(-(x + 1));
        res.push_back(-(x + 2));
        res.push_back(x + 3);
    }
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<int> seq = build(n);
        if (seq.empty()) {
            cout << -1 << '\n';
        } else {
            for (int i = 0; i < n; ++i) {
                if (i) cout << ' ';
                cout << seq[i];
            }
            cout << '\n';
        }
    }
    return 0;
}
```
