## 解题思路

1. **演变规则分析**  
   设当前序列长度为 $L$，其中数字 $1$ 的个数为 $c_1$，数字 $0$ 的个数为 $c_0 = L - c_1$。  
   - 每个 `0` 变为 `1`，贡献长度 $1$；  
   - 每个 `1` 变为 `01`，贡献长度 $2$。  
   一轮演变后，新序列长度 $L' = c_0 \times 1 + c_1 \times 2 = L + c_1$。

2. **状态转移**  
   设第 $k$ 轮后（$k \ge 0$）序列长度为 $L_k$，其中 $1$ 的个数为 $B_k$，$0$ 的个数为 $A_k$。初始时：  
   - $L_0 = |s|$；  
   - $B_0 = s$ 中 `1` 的个数，$A_0 = L_0 - B_0$。  
   根据演变规则，一轮后的变化为：  
   - $A_{k+1} = B_k$（所有 $1$ 产生 $1$ 个 $0$）；  
   - $B_{k+1} = A_k + B_k = L_k$（每个 $0$ 产生 $1$ 个 $1$，每个 $1$ 也产生 $1$ 个 $1$）。  
   于是有 $L_{k+1} = A_{k+1} + B_{k+1} = B_k + L_k$。  
   又因为 $B_k = L_{k-1}$（对 $k \ge 1$），代入得递推关系：
   \[
   L_{k+1} = L_k + L_{k-1} \qquad (k \ge 1).
   \]

3. **与斐波那契数列的关系**  
   定义标准斐波那契数列 $F_0 = 0,\ F_1 = 1,\ F_n = F_{n-1} + F_{n-2}$。由初始条件 $L_0$、$L_1 = L_0 + B_0$ 以及递推式，可证明对于 $n \ge 1$ 有：
   \[
   L_n = L_1 \cdot F_n + L_0 \cdot F_{n-1}.
   \]
   其中 $F_n$ 和 $F_{n-1}$ 可用快速倍增算法在 $O(\log n)$ 时间内求出。

4. **快速倍增算法**  
   快速倍增法通过递归计算 $(F_k, F_{k+1})$ 来得到 $(F_{2k}, F_{2k+1})$ 或 $(F_{2k+1}, F_{2k+2})$，结合位运算可以在 $O(\log n)$ 内求出 $F_n$ 和 $F_{n+1}$。所有运算均在模 $10^9+7$ 下进行。

5. **答案计算**  
   - 若 $n = 1$，答案即为 $L_1 \bmod (10^9+7)$；  
   - 若 $n > 1$，由公式 $L_n = (L_1 \cdot F_n + L_0 \cdot F_{n-1}) \bmod (10^9+7)$ 得到结果。  
   $F_{n-1}$ 可由 $F_{n+1} - F_n$ 在模意义下计算。

6. **多组数据处理**  
   题目保证 $\sum n \le 3 \times 10^5$，每组数据单独计算并输出结果。

## 复杂度分析

- **时间复杂度**：对于每组数据，统计初始序列中 `1` 的个数需要 $O(|s|)$ 时间，使用快速倍增算法求斐波那契数需要 $O(\log n)$ 时间。总时间复杂度为 $O(\sum|s| + T\log n)$，满足题目限制。
- **空间复杂度**：快速倍增算法的递归深度为 $O(\log n)$，因此每组数据的空间复杂度为 $O(\log n)$。若实现为迭代形式可降至 $O(1)$。整体空间复杂度 $O(\log n)$。

## 代码实现

### Python

```python
import sys

MOD = 10**9 + 7

def fast_fib(n):
    """快速倍增算法，返回 (F_n, F_{n+1}) mod MOD"""
    if n == 0:
        return (0, 1)
    a, b = fast_fib(n >> 1)
    # 计算 F_{2k} = F_k * (2 * F_{k+1} - F_k)
    c = (a * ((2 * b - a) % MOD)) % MOD
    # 计算 F_{2k+1} = F_k^2 + F_{k+1}^2
    d = (a * a + b * b) % MOD
    if n & 1:
        return (d, (c + d) % MOD)
    else:
        return (c, d)

def main():
    data = sys.stdin.read().split()
    t = int(data[0])
    idx = 1
    out = []
    for _ in range(t):
        s = data[idx]
        n = int(data[idx + 1])
        idx += 2
        
        L0 = len(s)
        B0 = s.count('1')          # 初始序列中1的个数
        L1 = L0 + B0               # 第一轮后的长度
        
        if n == 0:
            out.append(str(L0 % MOD))
            continue
        
        fn, fn_plus1 = fast_fib(n) # F_n, F_{n+1}
        fn_minus1 = (fn_plus1 - fn) % MOD
        
        if n == 1:
            ans = L1 % MOD
        else:
            ans = (L1 % MOD * fn + L0 % MOD * fn_minus1) % MOD
        out.append(str(ans))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1000000007L;

    // 快速倍增，返回 [F_n, F_{n+1}]
    static long[] fastFib(long n) {
        if (n == 0) return new long[]{0, 1};
        long[] p = fastFib(n >> 1);
        long a = p[0], b = p[1];
        long c = (a * ((2 * b % MOD - a + MOD) % MOD)) % MOD;
        long d = (a * a + b * b) % MOD;
        if ((n & 1) == 1)
            return new long[]{d, (c + d) % MOD};
        else
            return new long[]{c, d};
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        int T = Integer.parseInt(br.readLine().trim());
        while (T-- > 0) {
            String[] parts = br.readLine().split(" ");
            String s = parts[0];
            long n = Long.parseLong(parts[1]);

            long L0 = s.length();
            long B0 = 0;
            for (int i = 0; i < s.length(); i++) {
                if (s.charAt(i) == '1')
                    B0++;
            }
            long L1 = L0 + B0;

            if (n == 0) {
                sb.append(L0 % MOD).append('\n');
                continue;
            }

            long[] fib = fastFib(n);     // [F_n, F_{n+1}]
            long fn = fib[0];
            long fnPlus1 = fib[1];
            long fnMinus1 = (fnPlus1 - fn + MOD) % MOD;

            long ans;
            if (n == 1) {
                ans = L1 % MOD;
            } else {
                ans = ((L1 % MOD) * fn % MOD + (L0 % MOD) * fnMinus1 % MOD) % MOD;
            }
            sb.append(ans).append('\n');
        }
        System.out.print(sb);
    }
}
```

```cpp
const long long MOD = 1000000007LL;

// 快速倍增返回 pair<F_n, F_{n+1}>
pair<long long, long long> fastFib(long long n) {
    if (n == 0) return {0, 1};
    auto p = fastFib(n >> 1);
    long long a = p.first, b = p.second;
    long long c = (a * ((2 * b % MOD - a + MOD) % MOD)) % MOD;
    long long d = (a * a % MOD + b * b % MOD) % MOD;
    if (n & 1)
        return {d, (c + d) % MOD};
    else
        return {c, d};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        string s;
        long long n;
        cin >> s >> n;

        long long L0 = s.size();
        long long B0 = 0;
        for (char c : s)
            if (c == '1') B0++;
        long long L1 = L0 + B0;

        } else {
            ans = ((L1 % MOD) * fn % MOD + (L0 % MOD) * fnMinus1 % MOD) % MOD;
        }
        cout << ans << '\n';
    }
    return 0;
}
```
