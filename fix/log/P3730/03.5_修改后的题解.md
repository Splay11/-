## 解题思路

灯珠亮度只有 `1` 和 `2`，任意连续片段的乘积都是 `2` 的幂。因此“各段乘积相同”等价于“各段中亮度为 `2` 的灯珠个数相同”。

设当前亮度为 `2` 的灯珠总数为 $c_2$，要分成 $z$ 段且每段恰好 $k$ 个 `2`，必须 $c_2 = z \cdot k$，即 $c_2$ 能被 $z$ 整除。

分情况判断：

1. 若 $z > n$，无法分成多于灯珠个数的非空段，答案为 `No`。
2. 若 $c_2 = 0$，整条灯带全是 `1`，任意非空片段乘积都是 `1`。只要 $z \le n$ 即可，答案为 `Yes`。
3. 若 $c_2 \bmod z \ne 0$，无法均分，答案为 `No`。
4. 否则令 $k = c_2 / z$。前 $z-1$ 段一共需要 $(z-1)k$ 个 `2`，且最后一段必须非空，因此这些 `2` 必须全部落在前 $n-1$ 颗灯珠中。记前 $n-1$ 颗中 `2` 的个数为 $c_2$ 减去末位是否为 `2`。若 $(z-1)k$ 不超过该前缀个数，则一定能切出合法分段。

维护方面：读入时统计 $c_2$，每次把位置 $x$ 改为 $y$ 时，只在 `1` 与 `2` 互变时更新计数。每次询问均可 $O(1)$ 完成。

## 复杂度分析

每组数据在读入时扫描一遍灯带，之后每次维护常数时间判断。

- 时间复杂度：$O(n+m)$。
- 空间复杂度：$O(n)$。

在 $\sum n$ 与 $\sum m$ 均不超过 $2 \times 10^5$ 的限制下可以通过。

## 代码实现

### Python

```python
import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    count_twos = a.count(2)

    for _ in range(m):
        x, y, z = map(int, sys.stdin.readline().split())
        x -= 1

        if a[x] != y:
            if a[x] == 2:
                count_twos -= 1
            else:
                count_twos += 1
            a[x] = y

        if z > n:
            print("No")
            continue

        if count_twos == 0:
            print("Yes")
            continue

        if count_twos % z != 0:
            print("No")
        else:
            k = count_twos // z
            prefix_twos_n_minus_1 = count_twos
            if a[n - 1] == 2:
                prefix_twos_n_minus_1 -= 1
            if (z - 1) * k <= prefix_twos_n_minus_1:
                print("Yes")
            else:
                print("No")


def main():
    try:
        num_tests = int(sys.stdin.readline())
        for _ in range(num_tests):
            solve()
    except (IOError, ValueError):
        pass


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

public class Main {

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }
    }

    public static void solve(FastReader sc, PrintWriter out) {
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        long countTwos = 0;

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            if (a[i] == 2) {
                countTwos++;
            }
        }

        for (int kOp = 0; kOp < m; kOp++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            long z = sc.nextLong();
            x--;

            if (a[x] != y) {
                if (a[x] == 2) {
                    countTwos--;
                } else {
                    countTwos++;
                }
                a[x] = y;
            }

            if (z > n) {
                out.println("No");
                continue;
            }

            if (countTwos == 0) {
                out.println("Yes");
                continue;
            }

            if (countTwos % z != 0) {
                out.println("No");
            } else {
                long k = countTwos / z;
                long prefixTwosNMinus1 = countTwos;
                if (a[n - 1] == 2) {
                    prefixTwosNMinus1--;
                }
                if ((z - 1) * k <= prefixTwosNMinus1) {
                    out.println("Yes");
                } else {
                    out.println("No");
                }
            }
        }
    }

    public static void main(String[] args) {
        FastReader sc = new FastReader();
        PrintWriter out = new PrintWriter(System.out);
        int t = sc.nextInt();
        while (t-- > 0) {
            solve(sc, out);
        }
        out.flush();
    }
}
```

```cpp
    std::cin >> n >> m;
    std::vector<int> a(n);

    for (int i = 0; i < n; ++i) {
        std::cin >> a[i];
        if (a[i] == 2) {
            count_twos++;
        }
    }

    for (int k_op = 0; k_op < m; ++k_op) {
        long long x, y, z;
        std::cin >> x >> y >> z;
        --x;

        if (a[x] != y) {
            if (a[x] == 2) {
                count_twos--;
            } else {
                count_twos++;
            }
            a[x] = y;
        }

        if (z > n) {
            std::cout << "No\n";
            continue;
        }

        if (count_twos == 0) {
            std::cout << "Yes\n";
            continue;
        }

        if (count_twos % z != 0) {
            std::cout << "No\n";
        } else {
            long long k = count_twos / z;
            long long prefix_twos_n_minus_1 = count_twos;

            } else {

            }
        }
    }
}

int main() {
```
