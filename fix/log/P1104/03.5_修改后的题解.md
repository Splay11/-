## 解题思路

1. 首先判断可行性：若总工作量 `total` 为奇数，则无法平分为完全相等的两部分，方案数为 0。
2. 令 `n = total / 2`，表示每名程序员需要承担的工作量。问题转化为：将 `n` 拆分成 `i` 个**正整数**（作为第一名程序员的子任务序列），将另一个 `n` 拆分成 `k - i` 个正整数（作为第二名程序员的子任务序列），其中 `1 ≤ i ≤ k-1`。
3. 将一个正整数 `n` 拆分成 `m` 个正整数之和，且顺序有意义（即拆分序列），其方案数等价于在 `n` 个 1 之间的 `n-1` 个空隙中插入 `m-1` 个隔板，方案数为组合数 **C(n-1, m-1)**。
4. 因此，对于固定的 `i`，第一部分的方案数为 `C(n-1, i-1)`，第二部分的方案数为 `C(n-1, k-i-1)`。总方案数即为：
   \[
   \text{ans} = \sum_{i=1}^{k-1} \binom{n-1}{i-1} \times \binom{n-1}{k-i-1} \pmod{10^9+7}
   \]
5. 由于结果很大，需要取模 $10^9+7$（一个质数）。组合数通过预处理阶乘和逆元来快速计算：
   - 预处理 `fac[0..N]`，`fac[i] = i! mod MOD`。
   - 利用费马小定理求逆元：`inv(a) = a^(MOD-2) mod MOD`，用快速幂实现。
   - 则 `C(n, m) = fac[n] * inv(fac[m]) * inv(fac[n-m]) mod MOD`，当 `m > n` 时结果为 0。
6. 遍历所有可能的 `i` 累加即可得到答案。

## 复杂度分析

- 时间复杂度：$O(N + k)$，其中 $N = total$ 用于预处理阶乘和逆元（最大 $2 \times 10^5$），$k$ 为枚举分割点的循环次数，每次组合数查询为 $O(1)$。
- 空间复杂度：$O(N)$，即存储阶乘数组所需的空间。

## 代码实现

### Python

```python
MOD = 1000000007

def main():
    total, k = map(int, input().split())
    if total % 2 != 0:
        print(0)
        return
    
    n = total // 2
    max_n = total  # 足够覆盖所有需要的阶乘值
    
    # 预处理阶乘
    fac = [1] * (max_n + 1)
    for i in range(1, max_n + 1):
        fac[i] = fac[i-1] * i % MOD
    
    # 快速幂
    def pow_mod(a, b):
        ans = 1
        base = a % MOD
        while b > 0:
            if b & 1:
                ans = ans * base % MOD
            base = base * base % MOD
            b >>= 1
        return ans
    
    # 组合数计算 C(n, m)
    def comb(n_val, m_val):
        if m_val > n_val or m_val < 0:
            return 0
        x = fac[n_val]
        y = fac[m_val] * fac[n_val - m_val] % MOD
        return x * pow_mod(y, MOD - 2) % MOD
    
    ans = 0
    for i in range(1, k):
        ans = (ans + comb(n - 1, i - 1) * comb(n - 1, k - i - 1)) % MOD
    
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    static final int MOD = 1000000007;
    static long[] fac;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int total = sc.nextInt();
        int k = sc.nextInt();
        sc.close();

        if (total % 2 != 0) {
            System.out.println(0);
            return;
        }

        int n = total / 2;
        int maxN = total;  // 足够覆盖所有组合数需求
        fac = new long[maxN + 1];
        fac[0] = 1;
        for (int i = 1; i <= maxN; i++) {
            fac[i] = fac[i - 1] * i % MOD;
        }

        long ans = 0;
        for (int i = 1; i < k; i++) {
            long left = comb(n - 1, i - 1);
            long right = comb(n - 1, k - i - 1);
            ans = (ans + left * right) % MOD;
        }
        System.out.println(ans);
    }

    static long powMod(long a, long b) {
        long ans = 1;
        a %= MOD;
        while (b > 0) {
            if ((b & 1) == 1) {
                ans = ans * a % MOD;
            }
            a = a * a % MOD;
            b >>= 1;
        }
        return ans;
    }

    static long comb(int n, int m) {
        if (m > n || m < 0) return 0;
        long x = fac[n];
        long y = fac[m] * fac[n - m] % MOD;
        return x * powMod(y, MOD - 2) % MOD;
    }
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

const int MOD = 1000000007;
vector<long long> fac;

long long powMod(long long a, long long b) {
    long long ans = 1;
    a %= MOD;
    while (b > 0) {
        if (b & 1) ans = ans * a % MOD;
        a = a * a % MOD;
        b >>= 1;
    }
    return ans;
}

long long comb(int n, int m) {
    if (m > n || m < 0) return 0;
    long long x = fac[n];
    long long y = fac[m] * fac[n - m] % MOD;
    return x * powMod(y, MOD - 2) % MOD;
}

int main() {
    int total, k;
    cin >> total >> k;

    if (total % 2 != 0) {
        cout << 0 << endl;
        return 0;
    }

    int n = total / 2;
    int maxN = total;  // 预处理到 total 即可
    fac.resize(maxN + 1);
    fac[0] = 1;
    for (int i = 1; i <= maxN; ++i) {
        fac[i] = fac[i - 1] * i % MOD;
    }

    long long ans = 0;
    for (int i = 1; i < k; ++i) {
        long long left = comb(n - 1, i - 1);
        long long right = comb(n - 1, k - i - 1);
        ans = (ans + left * right) % MOD;
    }

    cout << ans << endl;
    return 0;
}
```
