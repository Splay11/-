## 解题思路

题目要求在 $n \times n$ 的战场网格中，找到那些只需部署一个单位就能消灭一个或多个敌群的“关键逃生口”，并计算在每个关键逃生口部署单位后能消灭的敌军总数。  
我们可以将问题拆解为以下几个步骤：

1. **识别敌群（连通块）**  
   战场中的敌方单位 `'x'` 通过上下左右四个方向连成一片，形成“敌群”。使用广度优先搜索（BFS）或深度优先搜索遍历棋盘，为每个 `'x'` 连通块计算大小（即该敌群的敌方单位数量）。

2. **统计敌群的逃生口（气）**  
   遍历每个敌群时，检查其上下左右相邻的格子：  
   - 相邻的 `'x'` 属于同一敌群，继续延伸搜索。  
   - 相邻的 `'.'` 就是该敌群的“逃生口”（气）。  
   - 相邻的 `'o'`（我方单位）或棋盘边界不会贡献逃生口。  
   对于同一个敌群，同一个空地 `'.'` 可能被多次访问，需要去重。可以为每个空地记录上一次被哪个敌群统计过（使用一个全局数组 `mark`），如果本次敌群编号与记录不同，则视为该敌群的一个新逃生口，并将气数加一。

3. **判断唯一逃生口并累加消灭数**  
   若某个敌群的逃生口数量恰好为 $1$，则说明只要我方在这个唯一的 `.` 位置部署单位，该敌群的所有相邻格子都会变成我方单位、敌方单位或边界，从而被立即消灭，消灭的敌军数量等于该敌群的大小。  
   多个敌群可能共享同一个逃生口，此时在该逃生口部署单位会同时消灭所有相关的敌群，总消灭数为这些敌群大小的总和。使用哈希表（或字典）将逃生口坐标映射到一个累加器，每当发现一个拥有唯一逃生口的敌群时，就在该逃生口的累加值上加上敌群大小。

4. **整理并输出结果**  
   所有累加完毕后，哈希表中记录的每一个逃生口及其对应的消灭总数，就是题目要求输出的“可部署位置”。按照坐标 $(r_i, c_i)$ 升序排序（先按行、再按列），并输出满足条件的位置数量 $k$，然后每行输出行号、列号（均从 $1$ 开始）和消灭总数 $s_i$。

## 复杂度分析

- **时间复杂度**：$O(n^2)$。整个算法对每个格子最多访问常数次（BFS 遍历所有 `'x'` 格子一次，检查邻居时访问相邻空地），$n$ 最大为 $1000$，可以高效运行。
- **空间复杂度**：$O(n^2)$。需要存储棋盘、访问标记数组、去重标记数组以及记录结果的数据结构，均与棋盘大小成正比。

## 代码实现

### Python

```python
from collections import deque
import sys

def main():
    data = sys.stdin.read().strip().splitlines()
    n = int(data[0])
    g = data[1:1+n]

    def idx(r, c):
        return r * n + c

    dr = (-1, 1, 0, 0)
    dc = (0, 0, -1, 1)

    visX = [False] * (n * n)      # 敌方连通块访问标记
    mark = [-1] * (n * n)        # 空地唯一气去重（记录最近统计的敌群编号）
    capture = {}                 # 空地索引 -> 累计消灭数
    gid = 0                      # 当前敌群编号

    for r in range(n):
        for c in range(n):
            if g[r][c] != 'x':
                continue
            id0 = idx(r, c)
            if visX[id0]:
                continue

            gid += 1
            q = deque()
            q.append((r, c))
            visX[id0] = True
            size = 0
            libCount = 0
            lastLib = -1

            while q:
                ur, uc = q.popleft()
                size += 1
                for k in range(4):
                    vr = ur + dr[k]
                    vc = uc + dc[k]
                    if not (0 <= vr < n and 0 <= vc < n):
                        continue
                    ch = g[vr][vc]
                    if ch == 'x':
                        nid = idx(vr, vc)
                        if not visX[nid]:
                            visX[nid] = True
                            q.append((vr, vc))
                    elif ch == '.':
                        e = idx(vr, vc)
                        if mark[e] != gid:
                            mark[e] = gid
                            libCount += 1
                            lastLib = e
                    # 'o' 忽略

            if libCount == 1:
                capture[lastLib] = capture.get(lastLib, 0) + size

    ans = []
    for e, cnt in capture.items():
        r, c = divmod(e, n)
        ans.append((r + 1, c + 1, cnt))
    ans.sort()

    out_lines = [str(len(ans))]
    for a, b, cnt in ans:
        out_lines.append(f"{a} {b} {cnt}")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static int n;
    static char[][] g;
    static boolean[] visX;
    static int[] mark;
    static final int[] dr = {-1, 1, 0, 0};
    static final int[] dc = {0, 0, -1, 1};

    static int idx(int r, int c) {
        return r * n + c;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine().trim());
        g = new char[n][];
        for (int i = 0; i < n; i++) {
            g[i] = br.readLine().trim().toCharArray();
        }

        visX = new boolean[n * n];
        mark = new int[n * n];
        Arrays.fill(mark, -1);

        HashMap<Integer, Integer> capture = new HashMap<>();
        int gid = 0;

        for (int r = 0; r < n; r++) {
            for (int c = 0; c < n; c++) {
                if (g[r][c] != 'x') continue;
                int id0 = idx(r, c);
                if (visX[id0]) continue;

                gid++;
                ArrayDeque<int[]> q = new ArrayDeque<>();
                q.add(new int[]{r, c});
                visX[id0] = true;

                int size = 0;
                int libCount = 0;
                int lastLib = -1;

                while (!q.isEmpty()) {
                    int[] cur = q.poll();
                    int ur = cur[0], uc = cur[1];
                    size++;
                    for (int k = 0; k < 4; k++) {
                        int vr = ur + dr[k], vc = uc + dc[k];
                        if (vr < 0 || vr >= n || vc < 0 || vc >= n) continue;
                        char ch = g[vr][vc];
                        if (ch == 'x') {
                            int nid = idx(vr, vc);
                            if (!visX[nid]) {
                                visX[nid] = true;
                                q.add(new int[]{vr, vc});
                            }
                        } else if (ch == '.') {
                            int e = idx(vr, vc);
                            if (mark[e] != gid) {
                                mark[e] = gid;
                                libCount++;
                                lastLib = e;
                            }
                        }
                        // 'o' 忽略
                    }
                }

                if (libCount == 1) {
                    capture.put(lastLib, capture.getOrDefault(lastLib, 0) + size);
                }
            }
        }

        List<int[]> ans = new ArrayList<>();
        for (Map.Entry<Integer, Integer> kv : capture.entrySet()) {
            int e = kv.getKey(), cnt = kv.getValue();
            int r = e / n, c = e % n;
            ans.add(new int[]{r + 1, c + 1, cnt});
        }
        ans.sort(Comparator.<int[]>comparingInt(a -> a[0]).thenComparingInt(a -> a[1]));

        StringBuilder sb = new StringBuilder();
        sb.append(ans.size()).append('\n');
        for (int[] t : ans) {
            sb.append(t[0]).append(' ').append(t[1]).append(' ').append(t[2]).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    if (!(cin >> n)) return 0;
    vector<string> g(n);
    for (int i = 0; i < n; ++i) cin >> g[i];

    auto idx = [&](int r, int c) { return r * n + c; };
    const int dr[4] = {-1, 1, 0, 0};
    const int dc[4] = {0, 0, -1, 1};

    vector<char> visX(n * n, 0);
    vector<int> mark(n * n, -1);
    unordered_map<int, int> capture;
    int gid = 0;

    for (int r = 0; r < n; ++r) {
        for (int c = 0; c < n; ++c) {
            if (g[r][c] != 'x') continue;
            int id = idx(r, c);
            if (visX[id]) continue;

            gid++;
            queue<pair<int, int>> q;
            q.push({r, c});
            visX[id] = 1;
            int size = 0;
            int libCount = 0;
            int lastLib = -1;

            while (!q.empty()) {
                auto [ur, uc] = q.front(); q.pop();
                size++;
                for (int k = 0; k < 4; ++k) {
                    int vr = ur + dr[k], vc = uc + dc[k];
                    if (vr < 0 || vr >= n || vc < 0 || vc >= n) continue;
                    char ch = g[vr][vc];
                    if (ch == 'x') {
                        int nid = idx(vr, vc);
                        if (!visX[nid]) {
                            visX[nid] = 1;
                            q.push({vr, vc});
                        }
                    } else if (ch == '.') {
                        int e = idx(vr, vc);
                        if (mark[e] != gid) {
                            mark[e] = gid;
                            libCount++;
                            lastLib = e;
                        }
                    }
                    // 'o' 忽略
                }
            }

            if (libCount == 1) {
                capture[lastLib] += size;
            }
        }
    }

    vector<tuple<int, int, int>> ans;
    ans.reserve(capture.size());
    for (auto &kv : capture) {
        int e = kv.first, cnt = kv.second;
        int r = e / n, c = e % n;
        ans.push_back({r + 1, c + 1, cnt});
    }
    sort(ans.begin(), ans.end());

    cout << ans.size() << "\n";
    for (auto &t : ans) {
        int a, b, cnt;
        tie(a, b, cnt) = t;
        cout << a << ' ' << b << ' ' << cnt << "\n";
    }
    return 0;
}
```
