## 解题思路

1. **幽灵字母的判定**  
   原始信息中的幽灵字母会完全丢失，因此残存字符串 $s$ 中**不可能出现**该字母。换言之，若某个字母在 $s$ 中出现过，它不可能是幽灵字母。只有 $s$ 中未出现的字母才可能作为幽灵字母。

2. **固定幽灵字母后的计数**  
   假设幽灵字母为 $c$（$c$ 未在 $s$ 中出现），我们要在 $s$ 的基础上还原原始信息。原始信息可以看作在 $s$ 的相邻字符之间以及字符串首尾插入若干个 $c$：
   - 相邻两个字符不能相同，因此：
     * 若 $s_i \neq s_{i+1}$，这两个字符之间**可以选择插入一个 $c$ 或不插入**，有 $2$ 种选择。
     * 若 $s_i = s_{i+1}$，则必须在它们之间**插入一个 $c$**，否则相邻相同字符违反规定，仅有 $1$ 种选择。
   - 字符串两端（最左侧和最右侧）可以各自选择插入一个 $c$ 或不插入，**各**有 $2$ 种选择。
   - 此外，每个空位最多只能插入一个 $c$，否则会出现相邻的 $cc$。

   设 $diff$ 为 $s$ 中相邻字符**不同**的位置个数，即：
   $$ diff = |\{ i \mid 1 \le i < m,\ s_i \ne s_{i+1} \}| $$
   则对于一个幽灵字母 $c$，可以构造的合法原始信息数量为：
   $$ 2^{diff} \times 2 \times 2 = 2^{diff + 2} $$

3. **对所有幽灵字母求和**  
   设 $kind$ 为 $s$ 中包含的不同字母种类数，则可能成为幽灵字母的字母数量为 $26 - kind$。  
   最终答案为：
   $$ ans = (26 - kind) \times 2^{diff + 2} \bmod (10^9+7) $$

4. **实现细节**  
   - 遍历 $s$，用布尔数组标记出现过的字母，统计 $kind$。
   - 同时扫描相邻字符，统计 $diff$。
   - 预处理 $2^k \bmod (10^9+7)$ 的表，最大 $k$ 为所有测试用例中 $m$ 的最大值加 $2$。
   - 对每组数据直接计算上述公式即可。

## 复杂度分析

设所有测试字符串的总长度为 $S$，最大字符串长度为 $m_{\max}$。

- **时间复杂度**：每个字符串只需线性扫描一次，预处理 $2^k$ 需要 $O(m_{\max})$，因此总时间复杂度为 $O(S + m_{\max})$。
- **空间复杂度**：需要长度为 $m_{\max}+3$ 的幂数组以及少量辅助变量，空间复杂度为 $O(m_{\max})$。

## 代码实现

### Python

```python
import sys

MOD = 10 ** 9 + 7


def solve() -> None:
    data = sys.stdin.read().split()
    if not data:
        return
    t = int(data[0])
    strings = []
    max_len = 0
    idx = 1
    for _ in range(t):
        m = int(data[idx])
        s = data[idx + 1]
        idx += 2
        strings.append(s)
        if m > max_len:
            max_len = m

    # 预处理 2 的幂
    pow2 = [1] * (max_len + 3)
    for i in range(1, max_len + 3):
        pow2[i] = pow2[i - 1] * 2 % MOD

    out_lines = []
    for s in strings:
        vis = [False] * 26
        kind = 0
        # 统计出现的字母种类
        for ch in s:
            x = ord(ch) - 97
            if not vis[x]:
                vis[x] = True
                kind += 1

        # 统计相邻字符不同的位置数
        diff = 0
        for i in range(len(s) - 1):
            if s[i] != s[i + 1]:
                diff += 1

        miss = 26 - kind  # 可能的幽灵字母数量
        ans = miss * pow2[diff + 2] % MOD
        out_lines.append(str(ans))

    sys.stdout.write("\n".join(out_lines))


if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1000000007L;

    static long solveCase(String s, long[] pow2) {
        boolean[] vis = new boolean[26];
        int kind = 0;
        for (int i = 0; i < s.length(); i++) {
            int x = s.charAt(i) - 'a';
            if (!vis[x]) {
                vis[x] = true;
                kind++;
            }
        }

        int diff = 0;
        for (int i = 0; i + 1 < s.length(); i++) {
            if (s.charAt(i) != s.charAt(i + 1)) {
                diff++;
            }
        }

        int miss = 26 - kind;
        return miss * pow2[diff + 2] % MOD;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        String[] strings = new String[t];
        int maxLen = 0;

        for (int i = 0; i < t; i++) {
            int m = Integer.parseInt(br.readLine());
            String s = br.readLine();
            strings[i] = s;
            if (m > maxLen) maxLen = m;
        }

        long[] pow2 = new long[maxLen + 3];
        pow2[0] = 1;
        for (int i = 1; i < pow2.length; i++) {
            pow2[i] = pow2[i - 1] * 2 % MOD;
        }

        StringBuilder sb = new StringBuilder();
        for (String s : strings) {
            sb.append(solveCase(s, pow2)).append('\n');
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

const long long MOD = 1000000007LL;

long long solve_case(const string& s, const vector<long long>& pow2) {
    vector<int> vis(26, 0);
    int kind = 0;
    for (char ch : s) {
        int x = ch - 'a';
        if (!vis[x]) {
            vis[x] = 1;
            kind++;
        }
    }

    int diff = 0;
    for (int i = 0; i + 1 < (int)s.size(); i++) {
        if (s[i] != s[i + 1]) {
            diff++;
        }
    }

    int miss = 26 - kind;
    return miss * pow2[diff + 2] % MOD;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    vector<string> strings(t);
    int max_len = 0;
    for (int i = 0; i < t; i++) {
        int m;
        string s;
        cin >> m >> s;
        strings[i] = s;
        max_len = max(max_len, m);
    }

    vector<long long> pow2(max_len + 3, 1);
    for (int i = 1; i < (int)pow2.size(); i++) {
        pow2[i] = pow2[i - 1] * 2 % MOD;
    }

    for (const auto& s : strings) {
        cout << solve_case(s, pow2) << '\n';
    }

    return 0;
}
```
