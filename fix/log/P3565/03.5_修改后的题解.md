## 解题思路

巡检编号区间 $[l,r]$ 中满足 $i \equiv x \pmod k$ 的整数构成公差为 $k$ 的等差数列。

令 $f$ 为不小于 $l$ 的第一个合法数：

$$
f = l + ((x-l) \bmod k),
$$

其中余数取在 $[0,k-1]$。若 $f>r$ 则答案为 `0`，否则

$$
\mathrm{ans}=\left\lfloor\frac{r-f}{k}\right\rfloor+1.
$$

每组 $O(1)$ 计算。

## 复杂度分析

时间复杂度 $O(T)$，空间复杂度 $O(1)$。

## 代码实现

### Python

```python
import sys

def mod_nonneg(a: int, m: int) -> int:
    r = a % m
    if r < 0:
        r += m
    return r

data = sys.stdin.read().strip().split()
it = iter(data)
try:
    T = int(next(it))
except StopIteration:
    sys.exit(0)

out_lines = []
for _ in range(T):
    l = int(next(it)); r = int(next(it)); k = int(next(it)); x = int(next(it))
    f = l + mod_nonneg(x - l, k)
    if f <= r:
        ans = (r - f) // k + 1
    else:
        ans = 0
    out_lines.append(str(ans))

sys.stdout.write("\n".join(out_lines))
```

### Java

```java
import java.io.*;

public class Main {
    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        long nextLong() throws IOException {
            int c; do { c = read(); } while (c <= ' ' && c != -1);
            boolean neg = false;
            if (c == '-') { neg = true; c = read(); }
            long x = 0;
            while (c > ' ') {
                x = x * 10 + (c - '0');
                c = read();
            }
            return neg ? -x : x;
        }
        int nextInt() throws IOException { return (int) nextLong(); }
    }
    static long modNonneg(long a, long m) {
        long r = a % m;
        if (r < 0) r += m;
        return r;
    }
    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        StringBuilder sb = new StringBuilder();
        int T = fs.nextInt();
        while (T-- > 0) {
            long l = fs.nextLong();
            long r = fs.nextLong();
            long k = fs.nextLong();
            long x = fs.nextLong();
            long f = l + modNonneg(x - l, k);
            long ans = 0;
            if (f <= r) ans = (r - f) / k + 1;
            sb.append(ans).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

#include <bits/stdc++.h>
using namespace std;

long long mod_nonneg(long long a, long long m) {
    long long r = a % m;
    if (r < 0) r += m;
    return r;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    if (!(cin >> T)) return 0;
    while (T--) {
        long long l, r, k, x;
        cin >> l >> r >> k >> x;
        long long f = l + mod_nonneg(x - l, k);
        long long ans = 0;
        if (f <= r) ans = (r - f) / k + 1;
        cout << ans << '\n';
    }
    return 0;
}
