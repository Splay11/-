## 解题思路

问题要求进行 $m$ 轮淘汰，每轮在当前序列中找到一个能力值最小的人员，若有多个则淘汰原下标最小的那个。直接模拟时间复杂度过高，需要更高效的做法。

注意到淘汰规则实际上是按照「能力值从小到大，同能力值按下标从小到大」的顺序依次删除元素。整个淘汰过程等价于：

1. 将每个人员视为一个 `(能力值, 原下标)` 的二元组，按字典序从小到大排序；
2. 淘汰前 $m$ 个二元组对应的人员；
3. 剩下的留在原序列中的人员，按原始顺序输出其能力值。

进一步考虑到数据范围 $1 \le p_i \le n$，值域很小，可以直接用**桶排序**的思想避免显式排序。

具体步骤如下：

- **分桶**：创建 $n+1$ 个桶，桶 $v$ 存储所有能力值为 $v$ 的人员的原始下标。按 $i$ 从 $1$ 到 $n$ 遍历原序列，将下标 $i-1$（或 $i$，视具体实现）加入 `bucket[p[i]]` 中。因为遍历顺序就是下标递增顺序，所以每个桶内的下标天然有序（从小到大）。
- **模拟淘汰**：用一个布尔数组 `removed` 标记哪些下标已被淘汰。从能力值 $v = 1$ 到 $n$ 依次处理：
  - 若当前桶内元素个数 $\le m$，则该桶内所有下标均被淘汰，对应标记为 `true`，$m$ 减去该桶大小；
  - 否则，只淘汰该桶内**前 $m$ 个**下标（因为它们对应能力值均为 $v$，且下标最小），然后将 $m$ 置为 $0$ 并终止处理。
- **收集答案**：再次按 $i$ 从 $1$ 到 $n$ 遍历原序列，若 $i-1$ 未被标记为淘汰，则将该能力值加入结果列表。
- 最后输出结果列表，元素之间用空格分隔。

该方法严格遵循题目淘汰规则，且每组数据只需线性时间。

## 复杂度分析

- **时间复杂度**：对于每组数据，分桶遍历原数组一次，模拟淘汰最多遍历值域和所有下标一次，收集答案再遍历一次，每一步均为 $O(n)$。总时间复杂度 $O(n)$。  
  全体 $T$ 组数据的总复杂度为 $O(\sum n)$，在题目约束下可以通过。

- **空间复杂度**：额外使用了 $n+1$ 个桶（存储下标，总空间 $O(n)$）以及一个布尔数组（长度 $n$），因此空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys

def solve_case(n, m, p):
    # bucket[v] 存储所有能力值为 v 的人员的原下标（从小到大）
    buckets = [[] for _ in range(n + 1)]
    for i in range(n):
        buckets[p[i]].append(i)

    removed = [False] * n

    # 按能力值从小到大淘汰
    for v in range(1, n + 1):
        if m == 0:
            break
        size = len(buckets[v])
        if size <= m:
            for idx in buckets[v]:
                removed[idx] = True
            m -= size
        else:
            for i in range(m):
                removed[buckets[v][i]] = True
            m = 0
            break

    ans = []
    for i in range(n):
        if not removed[i]:
            ans.append(str(p[i]))
    return " ".join(ans)

def main():
    input = sys.stdin.readline
    t = int(input().strip())
    out = []
    for _ in range(t):
        n, m = map(int, input().split())
        p = list(map(int, input().split()))
        out.append(solve_case(n, m, p))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static String solveCase(int n, int m, int[] p) {
        // bucket[v] 存储所有能力值为 v 的人员的原下标（从小到大）
        List<Integer>[] bucket = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            bucket[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            bucket[p[i]].add(i);
        }

        boolean[] removed = new boolean[n];

        // 按能力值从小到大淘汰
        for (int v = 1; v <= n && m > 0; v++) {
            int size = bucket[v].size();
            if (size <= m) {
                for (int idx : bucket[v]) {
                    removed[idx] = true;
                }
                m -= size;
            } else {
                for (int i = 0; i < m; i++) {
                    removed[bucket[v].get(i)] = true;
                }
                m = 0;
                break;
            }
        }

        StringBuilder sb = new StringBuilder();
        boolean first = true;
        for (int i = 0; i < n; i++) {
            if (!removed[i]) {
                if (!first) sb.append(' ');
                sb.append(p[i]);
                first = false;
            }
        }
        return sb.toString();
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder out = new StringBuilder();
        int T = Integer.parseInt(br.readLine().trim());
        for (int t = 0; t < T; t++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int n = Integer.parseInt(st.nextToken());
            int m = Integer.parseInt(st.nextToken());
            int[] p = new int[n];
            st = new StringTokenizer(br.readLine());
            for (int i = 0; i < n; i++) {
                p[i] = Integer.parseInt(st.nextToken());
            }
            out.append(solveCase(n, m, p));
            if (t != T - 1) out.append('\n');
        }
        System.out.print(out);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

string solveCase(int n, int m, const vector<int>& p) {
    // bucket[v] 存储所有能力值为 v 的人员的原下标（从小到大）
    vector<vector<int>> bucket(n + 1);
    for (int i = 0; i < n; i++) {
        bucket[p[i]].push_back(i);
    }

    vector<bool> removed(n, false);

    // 按能力值从小到大淘汰
    for (int v = 1; v <= n && m > 0; v++) {
        int sz = (int)bucket[v].size();
        if (sz <= m) {
            for (int idx : bucket[v]) {
                removed[idx] = true;
            }
            m -= sz;
        } else {
            for (int i = 0; i < m; i++) {
                removed[bucket[v][i]] = true;
            }
            m = 0;
            break;
        }
    }

    string ans;
    bool first = true;
    for (int i = 0; i < n; i++) {
        if (!removed[i]) {
            if (!first) ans += " ";
            ans += to_string(p[i]);
            first = false;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while (T--) {
        int n, m;
        cin >> n >> m;
        vector<int> p(n);
        for (int i = 0; i < n; i++) {
            cin >> p[i];
        }
        cout << solveCase(n, m, p) << '\n';
    }
    return 0;
}
```
