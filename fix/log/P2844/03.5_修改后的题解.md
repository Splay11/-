## 解题思路

覆盖两块三角形作业面等价于覆盖它们的六个顶点。最小覆盖圆只取决于点集凸包，点数仅为 `6`，可暴力枚举候选圆：

1. 枚举两点，以它们的连线为直径作圆，检查是否覆盖全部六点；
2. 枚举三点，若不共线则作其外接圆，再检查是否覆盖全部六点。

在所有能覆盖的候选圆中取半径最小者。判断点是否在圆内时允许 $10^{-9}$ 量级误差。

## 复杂度分析

点数 $k=6$，枚举两点 $O(k^2)$、三点 $O(k^3)$，每次覆盖检查 $O(k)$，时间与空间均为常数级。

## 代码实现

### Python

```python
import math

class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

def distance(a, b):
    return math.sqrt((a.x - b.x)**2 + (a.y - b.y)**2)

def covers(O, r, pts):
    for pt in pts:
        if distance(O, pt) > r + 1e-9:
            return False
    return True

def circle_from_2points(a, b):
    center = Point((a.x + b.x) / 2.0, (a.y + b.y) / 2.0)
    r = distance(a, b) / 2.0
    return center, r

def circle_from_3points(a, b, c):
    D = 2 * (a.x * (b.y - c.y) + b.x * (c.y - a.y) + c.x * (a.y - b.y))
    if abs(D) < 1e-9:
        return None, -1
    a_sq = a.x * a.x + a.y * a.y
    b_sq = b.x * b.x + b.y * b.y
    c_sq = c.x * c.x + c.y * c.y
    center_x = (a_sq * (b.y - c.y) + b_sq * (c.y - a.y) + c_sq * (a.y - b.y)) / D
    center_y = (a_sq * (c.x - b.x) + b_sq * (a.x - c.x) + c_sq * (b.x - a.x)) / D
    center = Point(center_x, center_y)
    r = distance(center, a)
    return center, r

def main():
    pts = []
    arr1 = list(map(int, input().split()))
    pts.append(Point(arr1[0], arr1[1]))
    pts.append(Point(arr1[2], arr1[3]))
    pts.append(Point(arr1[4], arr1[5]))
    arr2 = list(map(int, input().split()))
    pts.append(Point(arr2[0], arr2[1]))
    pts.append(Point(arr2[2], arr2[3]))
    pts.append(Point(arr2[4], arr2[5]))

    ans = float('inf')
    n = len(pts)
    for i in range(n):
        for j in range(i + 1, n):
            center, r = circle_from_2points(pts[i], pts[j])
            if covers(center, r, pts):
                ans = min(ans, r)
    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                center, r = circle_from_3points(pts[i], pts[j], pts[k])
                if r < 0:
                    continue
                if covers(center, r, pts):
                    ans = min(ans, r)
    print(ans)

if __name__ == '__main__':
    main()
```

```java
import java.util.*;

    }

    static boolean covers(Point O, double r, ArrayList<Point> pts) {
        for (Point pt : pts) {
            if (distance(O, pt) > r + 1e-9) return false;
        }
        return true;

    static double[] circleFrom2Points(Point a, Point b) {

    static double[] circleFrom3Points(Point a, Point b, Point c) {
        double D = 2 * (a.x * (b.y - c.y) + b.x * (c.y - a.y) + c.x * (a.y - b.y));
        if (Math.abs(D) < 1e-9) return null;
        double aSq = a.x * a.x + a.y * a.y;
        double bSq = b.x * b.x + b.y * b.y;

        double r = distance(new Point(centerX, centerY), a);
        return new double[]{centerX, centerY, r};
    }

    public static void main(String[] args) {

                double[] circle = circleFrom2Points(pts.get(i), pts.get(j));
```

```cpp
#include <iostream>
#include <cmath>
#include <vector>
#include <limits>
using namespace std;

struct Point {
    double x, y;
};

double distance(const Point &a, const Point &b) {
    return sqrt((a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y));
}

bool covers(const Point &O, double r, const vector<Point> &pts) {
    for (auto &pt : pts) {
        if (distance(O, pt) > r + 1e-9) return false;
    }
    return true;
}

pair<Point, double> circleFrom2Points(const Point &a, const Point &b) {
    Point center = {(a.x + b.x) / 2.0, (a.y + b.y) / 2.0};
    return {center, distance(a, b) / 2.0};
}

pair<Point, double> circleFrom3Points(const Point &a, const Point &b, const Point &c) {
    double D = 2 * (a.x * (b.y - c.y) + b.x * (c.y - a.y) + c.x * (a.y - b.y));
    if (fabs(D) < 1e-9) return {{0, 0}, -1};
    double aSq = a.x * a.x + a.y * a.y;
    double bSq = b.x * b.x + b.y * b.y;
    double cSq = c.x * c.x + c.y * c.y;
    Point center;
    center.x = (aSq * (b.y - c.y) + bSq * (c.y - a.y) + cSq * (a.y - b.y)) / D;
    center.y = (aSq * (c.x - b.x) + bSq * (a.x - c.x) + cSq * (b.x - a.x)) / D;
    return {center, distance(center, a)};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    vector<Point> pts(6);
    for (int i = 0; i < 6; i++) cin >> pts[i].x >> pts[i].y;
    double ans = numeric_limits<double>::infinity();
    for (int i = 0; i < 6; i++) {
        for (int j = i + 1; j < 6; j++) {
            auto [center, r] = circleFrom2Points(pts[i], pts[j]);
            if (covers(center, r, pts)) ans = min(ans, r);
        }
    }
    for (int i = 0; i < 6; i++) {
        for (int j = i + 1; j < 6; j++) {
            for (int k = j + 1; k < 6; k++) {
                auto [center, r] = circleFrom3Points(pts[i], pts[j], pts[k]);
                if (r < 0) continue;
                if (covers(center, r, pts)) ans = min(ans, r);
            }
        }
    }
    cout.precision(13);
    cout << ans << "\n";
    return 0;
}
```
