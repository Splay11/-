# 题解

## 题面描述

给定二维平面上两个三角形$△ABC$和$△DEF$，要求求出一个圆，该圆的半径最小，同时能够完全覆盖这两个三角形。  
输入时，第一行提供$△ABC$三个顶点的坐标：$x_A,y_A,x_B,y_B,x_C,y_C$；第二行提供$△DEF$三个顶点的坐标：$x_D,y_D,x_E,y_E,x_F,y_F$。保证输入的三个点构成非退化的三角形。  
输出结果为覆盖两个三角形的最小圆的半径。答案只要满足相对误差不超过$10^{-6}$即可。

---

## 解题思路

整个问题可以转化为求一组点的**最小覆盖圆**问题。关键观察如下：

- $△ABC$和$△DEF$是凸的，因此它们的并集的**凸包**即为所有顶点构成的凸包。  
- 由于最小覆盖圆仅与集合的凸包上的点有关，所以可以直接使用所有$6$个顶点来求解。  
- 采用求解最小覆盖圆的经典方法：  

  1. 枚举所有两个点组合，取这两个点构成**直径**的圆，若该圆可以包含所有点，则更新答案；  
  2. 枚举所有三个点组合，若这三点不共线，则计算它们的**外接圆**（$△PQR$的外接圆），若该圆可以包含所有点，则更新答案。  
  3. 同时也需要考虑特殊情况，如只有一个点的情况。  

- 由于点的个数最多为$6$个，所以该暴力算法的复杂度可接受。

主要公式说明：

- **两点间距离**：  
  $$
  d(P,Q)=\sqrt{(x_P-x_Q)^2+(y_P-y_Q)^2}
  $$

- **两点构成圆**：  
  圆心为$O=\displaystyle\frac{P+Q}{2}$，半径为$\displaystyle r=\frac{d(P,Q)}{2}$。

- **三点外接圆**的求法：  
  给定不共线的三点$P(x_P,y_P)$、$Q(x_Q,y_Q)$、$R(x_R,y_R)$，  
  计算辅助量：  
  $D$=$2\cdot ( $x_P(y_Q-y_R$)+$x_Q$($y_R-y_P$)+$x_R$($y_P-y_Q) $)$
  若$D=0$则三点共线，无外接圆。否则，  
  圆心$O(x,y)$的坐标为：  
  ![image](/file/2/M5tInkkkF7u7IHbwWI5XR.png) 
  半径为：  
  $r=d(O,P)$

## 代码分析

**基本思路：**  
用一个函数检查给定的圆（由圆心$O$和半径$r$决定）能否覆盖所有点，对于每一个候选圆都使用该函数进行验证。  

**暴力枚举：**  
- 遍历每一对点，构造候选圆，更新最小半径。  
- 遍历每一组三点，若不共线则计算外接圆，再更新最小半径。

**数值误差处理：**  
在判断点是否在圆内时，允许一个非常小的误差（如$10^{-9}$）。

## C++

```cpp
#include <iostream>
#include <cmath>
#include <vector>
#include <limits>
using namespace std;

// 定义一个点的结构体
struct Point {
    double x, y;
};

// 计算两点之间的距离
double distance(const Point &a, const Point &b) {
    return sqrt((a.x - b.x)*(a.x - b.x) + (a.y - b.y)*(a.y - b.y));
}

// 检查圆心 O 和半径 r 是否可以覆盖所有点 pts
bool covers(const Point &O, double r, const vector<Point> &pts) {
    for (auto &pt : pts) {
        if(distance(O, pt) > r + 1e-9) return false;
    }
    return true;
}

// 计算由两点构成的圆
pair<Point,double> circleFrom2Points(const Point &a, const Point &b) {
    Point center = {(a.x + b.x)/2.0, (a.y + b.y)/2.0};
    double r = distance(a, b) / 2.0;
    return {center, r};
}

// 计算由三点构成的外接圆，若三点共线返回 {center, -1}
pair<Point,double> circleFrom3Points(const Point &a, const Point &b, const Point &c) {
    double D = 2 * (a.x * (b.y - c.y) + 
                    b.x * (c.y - a.y) + 
                    c.x * (a.y - b.y));
    // 共线判断
    if(fabs(D) < 1e-9) return {{0,0}, -1};
    
    double aSq = a.x*a.x + a.y*a.y;
    double bSq = b.x*b.x + b.y*b.y;
    double cSq = c.x*c.x + c.y*c.y;
    
    Point center;
    center.x = (aSq*(b.y - c.y) + bSq*(c.y - a.y) + cSq*(a.y - b.y)) / D;
    center.y = (aSq*(c.x - b.x) + bSq*(a.x - c.x) + cSq*(b.x - a.x)) / D;
    
    double r = distance(center, a);
    return {center, r};
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    vector<Point> pts(6);
    // 输入 $△ABC$ 的三个顶点
    cin >> pts[0].x >> pts[0].y >> pts[1].x >> pts[1].y >> pts[2].x >> pts[2].y;
    // 输入 $△DEF$ 的三个顶点
    cin >> pts[3].x >> pts[3].y >> pts[4].x >> pts[4].y >> pts[5].x >> pts[5].y;
    
    double ans = numeric_limits<double>::infinity();
    
    // 枚举所有两点组合构造圆
    for (int i = 0; i < 6; i++) {
        for (int j = i+1; j < 6; j++) {
            auto [center, r] = circleFrom2Points(pts[i], pts[j]);
            if(covers(center, r, pts)) {
                ans = min(ans, r);
            }
        }
    }
    
    // 枚举所有三点组合构造圆（外接圆）
    for (int i = 0; i < 6; i++) {
        for (int j = i+1; j < 6; j++) {
            for (int k = j+1; k < 6; k++) {
                auto [center, r] = circleFrom3Points(pts[i], pts[j], pts[k]);
                if(r < 0) continue; // 三点共线则跳过
                if(covers(center, r, pts)) {
                    ans = min(ans, r);
                }
            }
        }
    }
    
    cout.precision(13);
    cout << ans << "\n";
    return 0;
}
```

```python
import math

# 定义点类
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

# 计算两点间的距离
def distance(a, b):
    return math.sqrt((a.x - b.x)**2 + (a.y - b.y)**2)

# 检查圆心 O 和半径 r 是否能覆盖所有点 pts
def covers(O, r, pts):
    for pt in pts:
        if distance(O, pt) > r + 1e-9:
            return False
    return True

# 由两个点构造圆
def circle_from_2points(a, b):
    center = Point((a.x + b.x) / 2.0, (a.y + b.y) / 2.0)
    r = distance(a, b) / 2.0
    return center, r

# 由三个点构造外接圆，三点共线时返回 (None, -1)
def circle_from_3points(a, b, c):
    D = 2 * (a.x * (b.y - c.y) + b.x * (c.y - a.y) + c.x * (a.y - b.y))
    if abs(D) < 1e-9:
        return None, -1
    a_sq = a.x * a.x + a.y * a.y
    b_sq = b.x * b.x + b.y * b.y
    c_sq = c.x * c.x + c.y * c.y
    center_x = (a_sq*(b.y - c.y) + b_sq*(c.y - a.y) + c_sq*(a.y - b.y)) / D
    center_y = (a_sq*(c.x - b.x) + b_sq*(a.x - c.x) + c_sq*(b.x - a.x)) / D
    center = Point(center_x, center_y)
    r = distance(center, a)
    return center, r

def main():
    # 输入6个点的坐标
    pts = []
    # 输入 $△ABC$ 的顶点
    arr1 = list(map(int, input().split()))
    pts.append(Point(arr1[0], arr1[1]))
    pts.append(Point(arr1[2], arr1[3]))
    pts.append(Point(arr1[4], arr1[5]))
    # 输入 $△DEF$ 的顶点
    arr2 = list(map(int, input().split()))
    pts.append(Point(arr2[0], arr2[1]))
    pts.append(Point(arr2[2], arr2[3]))
    pts.append(Point(arr2[4], arr2[5]))
    
    ans = float('inf')
    n = len(pts)
    
    # 枚举所有两点组合构造圆
    for i in range(n):
        for j in range(i+1, n):
            center, r = circle_from_2points(pts[i], pts[j])
            if covers(center, r, pts):
                ans = min(ans, r)
                
    # 枚举所有三点组合构造外接圆
    for i in range(n):
        for j in range(i+1, n):
            for k in range(j+1, n):
                center, r = circle_from_3points(pts[i], pts[j], pts[k])
                if r < 0:  # 三点共线
                    continue
                if covers(center, r, pts):
                    ans = min(ans, r)
                    
    # 输出结果，保留适当的精度
    print(ans)

if __name__ == '__main__':
    main()
```

```java
import java.util.*;
import java.io.*;

public class Main {
    // 定义点类
    static class Point {
        double x, y;
        Point(double x, double y){
            this.x = x;
            this.y = y;
        }
    }
    
    // 计算两点间的距离
    static double distance(Point a, Point b) {
        return Math.sqrt((a.x - b.x)*(a.x - b.x) + (a.y - b.y)*(a.y - b.y));
    }
    
    // 检查以圆心 O 和半径 r 构成的圆是否覆盖所有点 pts
    static boolean covers(Point O, double r, ArrayList<Point> pts) {
        for (Point pt : pts) {
            if(distance(O, pt) > r + 1e-9)
                return false;
        }
        return true;
    }
    
    // 由两个点构造圆，返回一个数组：[center_x, center_y, r]
    static double[] circleFrom2Points(Point a, Point b) {
        double centerX = (a.x + b.x) / 2.0;
        double centerY = (a.y + b.y) / 2.0;
        double r = distance(a, b) / 2.0;
        return new double[]{centerX, centerY, r};
    }
    
    // 由三个点构造外接圆，如果三点共线则返回 null
    static double[] circleFrom3Points(Point a, Point b, Point c) {
        double D = 2 * (a.x * (b.y - c.y) + b.x * (c.y - a.y) + c.x * (a.y - b.y));
        if(Math.abs(D) < 1e-9)
            return null;
        double aSq = a.x * a.x + a.y * a.y;
        double bSq = b.x * b.x + b.y * b.y;
        double cSq = c.x * c.x + c.y * c.y;
        
        double centerX = (aSq*(b.y - c.y) + bSq*(c.y - a.y) + cSq*(a.y - b.y)) / D;
        double centerY = (aSq*(c.x - b.x) + bSq*(a.x - c.x) + cSq*(b.x - a.x)) / D;
        double r = distance(new Point(centerX, centerY), a);
        return new double[]{centerX, centerY, r};
    }
    
    public static void main(String[] args) throws IOException{
        Scanner sc = new Scanner(System.in);
        ArrayList<Point> pts = new ArrayList<>();
        
        // 输入 $△ABC$ 的顶点
        double xA = sc.nextDouble(), yA = sc.nextDouble();
        double xB = sc.nextDouble(), yB = sc.nextDouble();
        double xC = sc.nextDouble(), yC = sc.nextDouble();
        pts.add(new Point(xA, yA));
        pts.add(new Point(xB, yB));
        pts.add(new Point(xC, yC));
        
        // 输入 $△DEF$ 的顶点
        double xD = sc.nextDouble(), yD = sc.nextDouble();
        double xE = sc.nextDouble(), yE = sc.nextDouble();
        double xF = sc.nextDouble(), yF = sc.nextDouble();
        pts.add(new Point(xD, yD));
        pts.add(new Point(xE, yE));
        pts.add(new Point(xF, yF));
        
        double ans = Double.POSITIVE_INFINITY;
        int n = pts.size();
        
        // 枚举所有两点组合构造圆
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                double[] circle = circleFrom2Points(pts.get(i), pts.get(j));
                Point center = new Point(circle[0], circle[1]);
                double r = circle[2];
                if(covers(center, r, pts)) {
                    ans = Math.min(ans, r);
                }
            }
        }
        
        // 枚举所有三点组合构造外接圆
        for (int i = 0; i < n; i++){
            for (int j = i+1; j < n; j++){
                for (int k = j+1; k < n; k++){
                    double[] circle = circleFrom3Points(pts.get(i), pts.get(j), pts.get(k));
                    if(circle == null) continue; // 三点共线
                    Point center = new Point(circle[0], circle[1]);
                    double r = circle[2];
                    if(covers(center, r, pts)){
                        ans = Math.min(ans, r);
                    }
                }
            }
        }
        
        // 输出答案，使用足够精度
        System.out.printf("%.13f\n", ans);
    }
}
```
