## 解题思路

题目要求统计每个位置 $i$ 作为**均衡值**出现的连续子区间个数。  
对于某个区间 $[l,r]$（$l\le i\le r$），设比 $a_i$ 小的元素个数为 $x$，比 $a_i$ 大的元素个数为 $y$，则 $a_i$ 是该区间的均衡值当且仅当  
$$
x-y=0\quad \text{或}\quad x-y=1
$$

为了高效统计，固定位置 $i$ 后，将区间拆成左右两部分：

- 左侧：$[l,i-1]$，其贡献和记为 $L$；
- 右侧：$[i+1,r]$，其贡献和记为 $R$。

定义每个元素相对于 $a_i$ 的贡献值：

- 若 $a_j < a_i$，贡献为 $+1$；
- 若 $a_j > a_i$，贡献为 $-1$。

于是 $L$ 和 $R$ 分别是左侧、右侧所有元素的贡献之和。  
总贡献 $x-y = L+R$。条件变为  
$$
L+R=0\quad \text{或}\quad L+R=1
$$
即
$$
L = -R\quad \text{或}\quad L = 1-R
$$

因此，对于固定的 $i$，可以先向左扫描，用计数数组 `cnt` 统计每一种左侧贡献和 $L$ 的出现次数（包含左侧为空时 $L=0$ 的情况）。然后从 $i$ 开始向右扫描，依次累加右侧贡献和 $R$，对于当前的 $R$，答案增加
$$
\mathrm{cnt}[-R] + \mathrm{cnt}[1-R]
$$
即可得到所有包含 $i$ 且满足条件的区间个数。

对每个 $i$ 执行上述过程，最终输出每个位置的答案。

## 复杂度分析

对于每个位置 $i$，分别向左、向右各扫描一次，总扫描长度为 $O(n)$，因此单个位置复杂度 $O(n)$。  
对 $n$ 个位置均做如上操作，总时间复杂度为 $O(n^2)$。  
题目保证所有测试数据的 $n$ 总和不超过 $5000$，因此 $O(n^2)$ 的算法可以顺利通过。

空间方面，用于计数的数组大小为 $O(n)$，额外空间复杂度也为 $O(n)$。

## 代码实现

### Python

```python
import sys


def solve_case(n, a):
    # ans[i] 表示第 i 个位置的元素作为均衡值的子区间个数
    ans = [0] * n

    # 贡献和的范围在 [-n, n]，开一个足够大的数组做计数
    offset = n + 2
    size = 2 * n + 5

    for i in range(n):
        cnt = [0] * size

        # 左侧为空：贡献和为 0
        cnt[offset] = 1

        # 向左扫描，统计左侧贡献和出现次数
        cur = 0
        for l in range(i - 1, -1, -1):
            if a[l] < a[i]:
                cur += 1
            else:
                cur -= 1
            cnt[cur + offset] += 1

        # 向右扫描，r=i 表示右侧为空
        cur = 0
        for r in range(i, n):
            if r > i:
                if a[r] < a[i]:
                    cur += 1
                else:
                    cur -= 1

            # 统计满足 L + cur == 0 或 1 的方案数
            ans[i] += cnt[-cur + offset] + cnt[1 - cur + offset]

    return ans


def main():
    data = list(map(int, sys.stdin.read().split()))
    t = data[0]
    idx = 1
    out = []

    for _ in range(t):
        n = data[idx]
        idx += 1
        a = data[idx:idx + n]
        idx += n

        res = solve_case(n, a)
        out.append(" ".join(map(str, res)))

    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedInputStream;
import java.util.StringJoiner;

public class Main {

    public static long[] solveCase(int n, int[] a) {
        // ans[i] 表示第 i 个位置的元素作为均衡值的子区间个数
        long[] ans = new long[n];

        int offset = n + 2;
        int size = 2 * n + 5;

        for (int i = 0; i < n; i++) {
            int[] cnt = new int[size];

            // 左侧为空：贡献和为 0
            cnt[offset] = 1;

            // 向左扫描，统计左侧贡献和出现次数
            int cur = 0;
            for (int l = i - 1; l >= 0; l--) {
                if (a[l] < a[i]) {
                    cur++;
                } else {
                    cur--;
                }
                cnt[cur + offset]++;
            }

            // 向右扫描，r=i 表示右侧为空
            cur = 0;
            for (int r = i; r < n; r++) {
                if (r > i) {
                    if (a[r] < a[i]) {
                        cur++;
                    } else {
                        cur--;
                    }
                }

                // 统计满足 left_sum + cur == 0 或 1 的方案数
                ans[i] += cnt[-cur + offset];
                ans[i] += cnt[1 - cur + offset];
            }
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();

        int T = fs.nextInt();
        StringBuilder sb = new StringBuilder();

        while (T-- > 0) {
            int n = fs.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = fs.nextInt();
            }

            long[] ans = solveCase(n, a);

            StringJoiner sj = new StringJoiner(" ");
            for (int i = 0; i < n; i++) {
                sj.add(String.valueOf(ans[i]));
            }
            sb.append(sj).append('\n');
        }

        System.out.print(sb.toString());
    }

    static class FastScanner {
        private final BufferedInputStream in = new BufferedInputStream(System.in);
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        private int read() throws Exception {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) {
                    return -1;
                }
            }
            return buffer[ptr++];
        }

        int nextInt() throws Exception {
            int c;
            do {
                c = read();
            } while (c <= ' ');

            int sign = 1;
            if (c == '-') {
                sign = -1;
                c = read();
            }

            int val = 0;
            while (c > ' ') {
                val = val * 10 + (c - '0');
                c = read();
            }
            return val * sign;
        }
    }
}
```

```cpp
using namespace std;

vector<long long> solveCase(int n, const vector<int>& a) {
    // ans[i] 表示第 i 个位置的元素作为均衡值的子区间个数
    vector<long long> ans(n, 0);

    int offset = n + 2;
    int size = 2 * n + 5;

    for (int i = 0; i < n; i++) {
        vector<int> cnt(size, 0);

        // 左侧为空：贡献和为 0
        cnt[offset] = 1;

        // 向左扫描，统计左侧贡献和出现次数
        int cur = 0;
        for (int l = i - 1; l >= 0; l--) {
            if (a[l] < a[i]) {
                cur++;
            } else {
                cur--;
            }
            cnt[cur + offset]++;
        }

        // 向右扫描，r=i 表示右侧为空
        cur = 0;
        for (int r = i; r < (int)n; r++) {
            if (r > i) {
                if (a[r] < a[i]) {
                    cur++;
                } else {
                    cur--;
                }
            }

            // 统计满足 L + cur == 0 或 1 的方案数
            ans[i] += cnt[-cur + offset];
            ans[i] += cnt[1 - cur + offset];
        }
    }

    return ans;
}

int main() {
    int T;
    cin >> T;

    while (T--) {
        int n;
        cin >> n;

        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }

        vector<long long> ans = solveCase(n, a);

        for (int i = 0; i < n; i++) {
            if (i) {
                cout << ' ';

}
```
