## 解题思路

问题要求在最多修改 $k$ 个基站频率的前提下，使相邻基站频率差的最大值（干扰度）尽可能小。由于可以任意设置修改后的频率值，当我们将目标干扰度限制为某个阈值 $L$ 时，若一条边两端基站的初始频率差 $|f_u-f_v|>L$，则该边会“破坏”我们的目标，必须修改这条边至少一个端点的频率；反之，若 $|f_u-f_v|\le L$，则即使两端点都不修改，该边也能满足要求。

将初始频率差大于 $L$ 的边称为 **坏边**。问题转化为：在坏边构成的边集上，是否存在一个顶点覆盖（选出的顶点集与每条坏边相邻），且顶点覆盖的大小不超过 $k$？树上最小顶点覆盖可以通过树形 DP 在 $O(n)$ 时间内求出。再对 $L$ 进行二分查找即可得到最小的可行干扰度。

具体步骤：

1. 计算每条树边的权值 $w_e=|f_u-f_v|$，并记录最大权值 $W=\max w_e$。
2. 在 $[0,W]$ 范围内二分 $L$。对于每个候选 $L$，进行树形 DP：
   - 任选一个节点（例如 $1$ 号基站）作为树根。
   - 定义状态：$dp[u][0]$ 表示在 $u$ 的子树内，不修改 $u$ 的频率时覆盖所有坏边所需的最小修改点数；$dp[u][1]$ 表示修改 $u$ 的频率时覆盖所有坏边的最小修改点数。
   - 初始化：$dp[u][1]=1$（计入 $u$ 本身的修改），$dp[u][0]=0$。
   - 对于 $u$ 的每个子节点 $v$ 及边权 $w$，从 $v$ 的 DP 结果 $(dp[v][0], dp[v][1])$ 向 $u$ 转移：
     * 若 $w>L$（坏边）：
       - 若不修改 $u$，则必须修改 $v$，故 $dp[u][0] \mathrel{+}= dp[v][1]$；
       - 若修改 $u$，则子节点 $v$ 可修改也可不修改，故 $dp[u][1] \mathrel{+}= \min(dp[v][0], dp[v][1])$。
     * 若 $w\le L$（好边）：
       - 无论 $u$ 是否修改，该边都已满足条件，子节点 $v$ 可任选状态，因此 $dp[u][0]$ 和 $dp[u][1]$ 均累加 $\min(dp[v][0], dp[v][1])$。
   - 最终整棵树所需的最小修改点数为 $\min(dp[1][0], dp[1][1])$。
3. 若 $\min(dp[1][0], dp[1][1]) \le k$，则当前 $L$ 可行，缩右边界；否则缩左边界。二分结束后输出得到的下界即为最小干扰度。

## 复杂度分析

- 每次 DP 遍历树的所有节点和边，时间复杂度为 $O(n)$。
- 二分搜索的范围为 $[0, W]$，其中 $W\le \max f_i\le 100$，二分次数为 $O(\log W)$。
- 总时间复杂度 $O(n \log W)$，空间复杂度 $O(n)$（存储邻接表和递归栈）。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(1 << 20)

def dfs(u, p, L, adj):
    dp0, dp1 = 0, 1  # dp1 初值 +1，表示修改 u 的频率
    for v, w in adj[u]:
        if v == p:
            continue
        c0, c1 = dfs(v, u, L, adj)
        if w > L:          # 坏边：必须至少改一端
            dp0 += c1       # 不修改 u 则必须修改 v
            dp1 += min(c0, c1)  # 修改 u，v 任意
        else:               # 好边：无需额外修改
            t = min(c0, c1)
            dp0 += t
            dp1 += t
    return dp0, dp1

def check(L, adj, n, k):
    dp0, dp1 = dfs(1, 0, L, adj)
    return min(dp0, dp1) <= k

def main():
    data = list(map(int, sys.stdin.read().strip().split()))
    it = iter(data)
    n = next(it); k = next(it)
    f = [0] + [next(it) for _ in range(n)]        # 基站初始频率
    adj = [[] for _ in range(n + 1)]
    maxw = 0
    for _ in range(n - 1):
        u = next(it); v = next(it)
        w = abs(f[u] - f[v])
        if w > maxw:
            maxw = w
        adj[u].append((v, w))
        adj[v].append((u, w))
    lo, hi = 0, maxw
    while lo < hi:
        mid = (lo + hi) // 2
        if check(mid, adj, n, k):
            hi = mid
        else:
            lo = mid + 1
    print(lo)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static class Edge {
        int to, w;
        Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    static List<Edge>[] g;
    static int n, k;

    static int[] dfs(int u, int p, int L) {
        int dp0 = 0; // 不修改 u 的频率
        int dp1 = 1; // 修改 u 的频率
        for (Edge e : g[u]) {
            int v = e.to, w = e.w;
            if (v == p) continue;
            int[] child = dfs(v, u, L);
            if (w > L) {                        // 坏边
                dp0 += child[1];                // 不修改 u 必改 v
                dp1 += Math.min(child[0], child[1]); // 修改 u，v 任意
            } else {                            // 好边
                int t = Math.min(child[0], child[1]);
                dp0 += t;
                dp1 += t;
            }
        }
        return new int[]{dp0, dp1};
    }

    static boolean check(int L) {
        int[] res = dfs(1, 0, L);
        return Math.min(res[0], res[1]) <= k;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());

        int[] f = new int[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            f[i] = Integer.parseInt(st.nextToken());
        }

        g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();

        int maxw = 0;
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int w = Math.abs(f[u] - f[v]);
            maxw = Math.max(maxw, w);
            g[u].add(new Edge(v, w));
            g[v].add(new Edge(u, w));
        }

        int lo = 0, hi = maxw;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (check(mid)) hi = mid;
            else lo = mid + 1;
        }
        System.out.println(lo);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Edge {
    int to, w;
};

int n, k;
vector<vector<Edge>> g;

pair<int, int> dfs(int u, int p, int L) {
    int dp0 = 0; // 不修改 u 的频率
    int dp1 = 1; // 修改 u 的频率
    for (const auto &e : g[u]) {
        int v = e.to, w = e.w;
        if (v == p) continue;
        auto [c0, c1] = dfs(v, u, L);
        if (w > L) {                 // 坏边
            dp0 += c1;               // 不修改 u 必改 v
            dp1 += min(c0, c1);      // 修改 u，v 任意
        } else {                     // 好边
            int t = min(c0, c1);
            dp0 += t;
            dp1 += t;
        }
    }
    return {dp0, dp1};
}

bool check(int L) {
    auto [dp0, dp1] = dfs(1, 0, L);
    return min(dp0, dp1) <= k;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> k;
    vector<int> f(n + 1);
    for (int i = 1; i <= n; i++) cin >> f[i];

    g.assign(n + 1, {});
    int maxw = 0;
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        int w = abs(f[u] - f[v]);
        maxw = max(maxw, w);
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }

    int lo = 0, hi = maxw;
    while (lo < hi) {
        int mid = (lo + hi) / 2;
        if (check(mid)) hi = mid;
        else lo = mid + 1;
    }
    cout << lo << "\n";
    return 0;
}
```
