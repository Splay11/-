## 解题思路

1. 读取输入的一行 JSON 字符串，解析为一个字典，其中每个键表示观测站编号（字符串），对应的值又是一个字典，描述该观测站下各物种的个体数量。
2. 对于每一个观测站：
   - 计算该站所有物种的个体总数 $S = \sum c_i$。若 $S = 0$，则定义熵为 $0.0$。
   - 对于每个物种，其个体数量为 $c_i$，比例 $p_i = c_i / S$。当 $c_i = 0$ 时，跳过该物种（因为 $p_i = 0$ 的项贡献为 $0$）。
   - 根据信息熵公式计算该观测站的熵值：
     $$H = -\sum_{i=1}^{n} p_i \log_2 p_i$$
     实际计算中，只对 $c_i > 0$ 的项累加 $- \frac{c_i}{S} \log_2 \frac{c_i}{S}$。
   - 将计算结果四舍五入保留三位小数，存入结果字典，键为观测站编号，值为 `{"entropy": 熵值}`。
3. 所有观测站处理完毕后，将结果字典序列化为 JSON 字符串并输出。注意数值必须保留三位小数（例如 `1.200`、`0.000`），确保 JSON 格式合法。

## 复杂度分析

- 设观测站总数为 $M$，所有观测站包含的物种数之和为 $N$（即所有计数的总个数）。  
- 遍历所有观测站及其物种计数，时间复杂度 $O(N)$。  
- 仅使用一个字典保存结果，空间复杂度 $O(M)$，可以满足 $M \le 10^4$ 的限制。

## 代码实现

### Python

```python
import sys
import json
import math

def station_entropy(counts):
    """计算一个观测站的熵值，返回四舍五入保留三位小数的浮点数"""
    total = sum(counts.values())
    if total == 0:
        return 0.0

    entropy = 0.0
    for c in counts.values():
        if c == 0:
            continue
        p = c / total
        entropy -= p * math.log2(p)

    return round(entropy, 3)

def main():
    raw = sys.stdin.read().strip()
    if not raw:
        print("{}")
        return

    data = json.loads(raw)          # 解析输入的 JSON 字典
    result = {}
    for station_id, species_counts in data.items():
        result[station_id] = {"entropy": station_entropy(species_counts)}

    # 输出 JSON，保留三位小数的数值会由 json.dumps 自动处理
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
```

### Java

```java
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.*;

public class Main {
    // 计算单个观测站的熵值，保留三位小数
    private static double calcEntropy(Map<String, Integer> counts) {
        long total = 0;
        for (int c : counts.values()) {
            total += c;
        }
        if (total == 0) {
            return 0.0;
        }

        double entropy = 0.0;
        for (int c : counts.values()) {
            if (c == 0) continue;
            double p = (double) c / total;
            entropy -= p * (Math.log(p) / Math.log(2));   // log2(p)
        }

        // 四舍五入到三位小数
        return Math.round(entropy * 1000.0) / 1000.0;
    }

    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        String input = sc.useDelimiter("\\A").next();   // 读取全部输入
        sc.close();

        ObjectMapper mapper = new ObjectMapper();
        Map<String, Map<String, Integer>> data = mapper.readValue(input,
            new com.fasterxml.jackson.core.type.TypeReference<Map<String, Map<String, Integer>>>() {});

        Map<String, Map<String, Double>> result = new LinkedHashMap<>();
        for (Map.Entry<String, Map<String, Integer>> entry : data.entrySet()) {
            String stationId = entry.getKey();
            double entropy = calcEntropy(entry.getValue());
            Map<String, Double> inner = new HashMap<>();
            inner.put("entropy", entropy);
            result.put(stationId, inner);
        }

        String jsonOutput = mapper.writeValueAsString(result);
        System.out.println(jsonOutput);
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <cmath>
#include <map>
#include <sstream>
#include <iomanip>
#include <nlohmann/json.hpp>

using json = nlohmann::json;

// 对 JSON 字符串中的特殊字符进行转义
std::string jsonEscape(const std::string &s) {
    std::string out;
    for (char ch : s) {
        if (ch == '\\') out += "\\\\";
        else if (ch == '"') out += "\\\"";
        else out += ch;
    }
    return out;
}

// 计算一个观测站的熵值，保留三位小数
double stationEntropy(const std::map<std::string, int> &counts) {
    long long total = 0;
    for (const auto &kv : counts) total += kv.second;
    if (total == 0) return 0.0;

    double entropy = 0.0;
    for (const auto &kv : counts) {
        if (kv.second == 0) continue;
        double p = static_cast<double>(kv.second) / total;
        entropy -= p * std::log2(p);
    }

    // 四舍五入到三位小数
    return std::round(entropy * 1000.0) / 1000.0;
}

int main() {
    std::string line;
    // 读取全部标准输入（可能包含空格，但 JSON 通常在一行内）
    std::getline(std::cin, line, '\0'); // 读至 EOF
    if (line.empty()) {
        std::cout << "{}" << std::endl;
        return 0;
    }

    json data = json::parse(line);   // 解析输入的 JSON 对象

    std::ostringstream out;
    out << "{";
    bool first = true;

    for (auto it = data.begin(); it != data.end(); ++it) {
        const std::string &stationId = it.key();
        const json &speciesJson = it.value();

        std::map<std::string, int> counts;
        for (auto sp = speciesJson.begin(); sp != speciesJson.end(); ++sp) {
            counts[sp.key()] = sp.value().get<int>();
        }

        double entropy = stationEntropy(counts);

        if (!first) out << ",";
        first = false;

        out << "\"" << jsonEscape(stationId) << "\":{\"entropy\":"
            << std::fixed << std::setprecision(3) << entropy << "}";
    }

    out << "}";
    std::cout << out.str() << std::endl;

    return 0;
}
```
