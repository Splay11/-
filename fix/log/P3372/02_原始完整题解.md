### 题面描述（重述）

有 $n$ 个城市，给出 $m$ 条建桥计划，每条连接两城市 $u$、$v$ 并有利润 $w$（$w>0$ 盈利，$w<0$ 亏损）。从这些计划中选若干条，使所有城市互相可达（连通即可，不限条数）。求能获得的最大总利润。

保证原始 $m$ 条计划连通。

### 思路与结论

* 关键结论：最优解一定包含所有利润为正的边。

  * 证明要点：添加一条正边不会破坏连通性且能增加总利润，因此任何缺少正边的可行解都不是最优。
* 于是策略是：

  1. 把所有正边都选上（并计入答案），并用并查集把它们连起来。
  2. 此时若图已连通，结束。
  3. 否则只在非正边（$w≤0$）里选尽可能“损失最小”的边来把剩余的连通分量连起来。即对 $w≤0$ 的边按权值从大到小（$0$、$-1$、$-2$、…）做 $Kruskal$，只在连接不同连通分量时使用该边并把权值加到答案上。
* 正确性：这等价于“先全取正边，再在 $w≤0$ 的边上取一个最大权生成森林使整体连通”。因为 $w≤0$ 边只在必要时才加，且优先选损失最小的。
* 复杂度：

  * 并查集合并与查找近似 $O(α(n))$。
  * 排序仅对 $w≤0$ 的边，最坏 $O(m log m)$。
  * 总体满足 $n≤1e5$, $m≤1e6$ 的规模。


### C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

struct DSU {
	vector<int> p, sz;
	DSU(int n = 0) { init(n); }
	void init(int n) {
		p.resize(n + 1);
		sz.assign(n + 1, 1);
		iota(p.begin(), p.end(), 0);
	}
	int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }
	bool unite(int a, int b) {
		a = find(a); b = find(b);
		if (a == b) return false;
		if (sz[a] < sz[b]) swap(a, b);
		p[b] = a;
		sz[a] += sz[b];
		return true;
	}
};

struct Edge {
	int u, v;
	long long w;
};

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	int n, m;
	if (!(cin >> n >> m)) return 0;

	DSU dsu(n);
	long long ans = 0;
	vector<Edge> nonpos; // 存放 w <= 0 的边

	for (int i = 0; i < m; ++i) {
		int u, v;
		long long w;
		cin >> u >> v >> w;
		if (w > 0) {
			ans += w;           // 所有正边都计入答案
			dsu.unite(u, v);    // 并查集合并
		} else {
			nonpos.push_back({u, v, w});
		}
	}

	// 将非正边按权值从大到小排序（0, -1, -2, ...）
	sort(nonpos.begin(), nonpos.end(), [](const Edge& a, const Edge& b) {
		return a.w > b.w;
	});

	// 只在需要连通不同分量时使用这条（非正）边
	for (const auto& e : nonpos) {
		if (dsu.unite(e.u, e.v)) {
			ans += e.w; // w<=0，可能为0或减少总利润
		}
	}

	cout << ans << '\n';
	return 0;
}
```

### Python 

```python
import sys

def input():
	return sys.stdin.buffer.readline()

class DSU:
	def __init__(self, n):
		self.p = list(range(n + 1))
		self.sz = [1] * (n + 1)
	def find(self, x):
		while self.p[x] != x:
			self.p[x] = self.p[self.p[x]]
			x = self.p[x]
		return x
	def unite(self, a, b):
		a = self.find(a); b = self.find(b)
		if a == b: return False
		if self.sz[a] < self.sz[b]:
			a, b = b, a
		self.p[b] = a
		self.sz[a] += self.sz[b]
		return True

def main():
	n_m = input().split()
	while len(n_m) < 2:
		n_m += input().split()
	n, m = map(int, n_m)

	dsu = DSU(n)
	ans = 0
	nonpos = []

	for _ in range(m):
		u, v, w = map(int, input().split())
		if w > 0:
			ans += w
			dsu.unite(u, v)
		else:
			nonpos.append((w, u, v))

	# 非正边按 w 从大到小排序
	nonpos.sort(key=lambda x: x[0], reverse=True)

	for w, u, v in nonpos:
		if dsu.unite(u, v):
			ans += w

	print(ans)

if __name__ == "__main__":
	go = main()
```

```java
import java.io.*;
import java.util.*;

public class Main {
	static final class FastScanner {
		private final InputStream in;
		private final byte[] buffer = new byte[1 << 20];
		private int ptr = 0, len = 0;
		FastScanner(InputStream is) { in = is; }
		private int read() throws IOException {
			if (ptr >= len) {
				len = in.read(buffer);
				ptr = 0;
				if (len <= 0) return -1;
			}
			return buffer[ptr++];
		}
		long nextLong() throws IOException {
			int c; do { c = read(); } while (c <= 32);
			int sign = 1;
			if (c == '-') { sign = -1; c = read(); }
			long x = 0;
			while (c > 32) {
				x = x * 10 + (c - '0');
				c = read();
			}
			return x * sign;
		}
		int nextInt() throws IOException { return (int) nextLong(); }
	}

	static final class DSU {
		int[] p, sz;
		DSU(int n) {
			p = new int[n + 1];
			sz = new int[n + 1];
			for (int i = 0; i <= n; i++) { p[i] = i; sz[i] = 1; }
		}
		int find(int x) {
			while (p[x] != x) {
				p[x] = p[p[x]];
				x = p[x];
			}
			return x;
		}
		boolean unite(int a, int b) {
			a = find(a); b = find(b);
			if (a == b) return false;
			if (sz[a] < sz[b]) { int t = a; a = b; b = t; }
			p[b] = a;
			sz[a] += sz[b];
			return true;
		}
	}

	// 对并行数组 (w, u, v) 按 w 降序快排
	static void quickSort(long[] w, int[] u, int[] v, int l, int r) {
		int i = l, j = r;
		long pivot = w[(l + r) >>> 1];
		while (i <= j) {
			while (w[i] > pivot) i++;
			while (w[j] < pivot) j--;
			if (i <= j) {
				long tw = w[i]; w[i] = w[j]; w[j] = tw;
				int tu = u[i]; u[i] = u[j]; u[j] = tu;
				int tv = v[i]; v[i] = v[j]; v[j] = tv;
				i++; j--;
			}
		}
		if (l < j) quickSort(w, u, v, l, j);
		if (i < r) quickSort(w, u, v, i, r);
	}

	public static void main(String[] args) throws Exception {
		FastScanner fs = new FastScanner(System.in);
		int n = fs.nextInt(), m = fs.nextInt();

		DSU dsu = new DSU(n);
		long ans = 0;

		// 最多 m 条非正边
		long[] w = new long[m];
		int[] u = new int[m];
		int[] v = new int[m];
		int cnt = 0;

		for (int i = 0; i < m; i++) {
			int uu = fs.nextInt();
			int vv = fs.nextInt();
			long ww = fs.nextLong();
			if (ww > 0) {
				ans += ww;
				dsu.unite(uu, vv);
			} else {
				u[cnt] = uu;
				v[cnt] = vv;
				w[cnt] = ww;
				cnt++;
			}
		}

		if (cnt > 0) {
			quickSort(w, u, v, 0, cnt - 1);
			for (int i = 0; i < cnt; i++) {
				if (dsu.unite(u[i], v[i])) {
					ans += w[i]; // w<=0
				}
			}
		}

		System.out.println(ans);
	}
}
```
