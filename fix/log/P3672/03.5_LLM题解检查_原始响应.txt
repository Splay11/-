## 解题思路

1. **可行性判断**  
   如果将网络拆分成若干个服务器数量均为偶数的子网，则所有子网的服务器数之和也为偶数。因此，若总服务器数 $n$ 为奇数，无论如何拆分都不可能满足条件，此时方案数为 $0$。  
   当 $n$ 为偶数时，才可能存在合法方案。

2. **子树大小与边的性质**  
   将整个网络看成一棵无根树，任选服务器 $1$ 作为根，进行一次遍历，得到每个节点与父节点之间的父子关系，以及以每个节点为根的子树大小 `sz[u]`。  
   对于连接父节点 $p$ 和子节点 $v$ 的光纤（边），如果切断该光纤，则 $v$ 及其子树将形成一个独立的子网，该子网的服务器数量正好等于 `sz[v]`。  
   为了保证切割后每个子网的大小均为偶数，**所有被切断的光纤所对应的子树大小都必须为偶数**。反之，若切断的所有边都满足子树大小为偶数，则剩下的部分也必定是偶数（因为总数偶数减去若干偶数还是偶数），因此所有子网大小均为偶数。

3. **统计候选边数**  
   在整棵树中，除了根节点外，每个节点都对应一条到其父节点的光纤。设满足 `sz[v] % 2 == 0` 的节点 $v$（$v \neq 1$）的个数为 $m$，则我们有 $m$ 条“偶子树光纤”。  
   题目要求**恰好**切断 $k$ 条光纤，如果 $k > m$，则无法选出足够的合法光纤，方案数为 $0$。

4. **组合计数**  
   从 $m$ 条候选光纤中任意选择 $k$ 条进行切断，均会得到一个合法的拆分方案。不同的选择对应不同的光纤切断方案。因此答案即为组合数 $C(m, k)$。  
   利用阶乘与逆元预处理至 $n$，即可在 $O(1)$ 时间内计算组合数，模数为 $10^9+7$。

## 复杂度分析

- **时间复杂度**：$O(n)$。一次遍历建树并计算子树大小，组合数预处理和查询均为 $O(n)$ 或 $O(1)$。  
- **空间复杂度**：$O(n)$。需要存储邻接表、父子关系、子树大小以及阶乘和逆元数组。

## 代码实现

### Python

```python
import sys

MOD = 10**9 + 7

def mod_pow(a, e):
    r = 1
    while e:
        if e & 1:
            r = r * a % MOD
        a = a * a % MOD
        e >>= 1
    return r

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    k = int(next(it))
    
    # 构建邻接表
    g = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u = int(next(it))
        v = int(next(it))
        g[u].append(v)
        g[v].append(u)
    
    # 总数为奇数时无解
    if n % 2 == 1:
        print(0)
        return
    
    # 迭代 DFS 获取父节点和遍历顺序
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v in g[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            stack.append(v)
    
    # 逆序累加子树大小
    sz = [1] * (n + 1)
    for u in reversed(order):
        p = parent[u]
        if p != -1:
            sz[p] += sz[u]
    
    # 统计偶子树边数 m (不含根)
    m = sum(1 for v in range(2, n + 1) if sz[v] % 2 == 0)
    
    if k > m:
        print(0)
        return
    
    # 预处理阶乘与逆元
    fac = [1] * (n + 1)
    for i in range(1, n + 1):
        fac[i] = fac[i - 1] * i % MOD
    
    inv_fac = [1] * (n + 1)
    inv_fac[n] = mod_pow(fac[n], MOD - 2)
    for i in range(n - 1, -1, -1):
        inv_fac[i] = inv_fac[i + 1] * (i + 1) % MOD
    
    def C(N, K):
        if K < 0 or K > N:
            return 0
        return fac[N] * inv_fac[K] % MOD * inv_fac[N - K] % MOD
    
    print(C(m, k))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;
import java.util.StringTokenizer;

public class Main {
    static final long MOD = 1_000_000_007L;
    
    static long modPow(long a, long e) {
        long res = 1;
        while (e > 0) {
            if ((e & 1) == 1) res = res * a % MOD;
            a = a * a % MOD;
            e >>= 1;
        }
        return res;
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        
        // 构建邻接表
        ArrayList<Integer>[] g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            g[u].add(v);
            g[v].add(u);
        }
        
        if (n % 2 == 1) {
            System.out.println(0);
            return;
        }
        
        // 迭代 DFS 获取父节点与遍历顺序
        int[] parent = new int[n + 1];
        int[] order = new int[n];
        int[] stack = new int[n];
        int top = 0, idx = 0;
        stack[top++] = 1;
        parent[1] = -1;
        while (top > 0) {
            int u = stack[--top];
            order[idx++] = u;
            for (int v : g[u]) {
                if (v == parent[u]) continue;
                parent[v] = u;
                stack[top++] = v;
            }
        }
        
        // 逆序累加子树大小
        int[] sz = new int[n + 1];
        for (int i = 1; i <= n; i++) sz[i] = 1;
        for (int i = idx - 1; i >= 0; i--) {
            int u = order[i];
            int p = parent[u];
            if (p != -1) sz[p] += sz[u];
        }
        
        // 统计偶子树边数 m
        int m = 0;
        for (int v = 2; v <= n; v++) {
            if (sz[v] % 2 == 0) m++;
        }
        
        if (k > m) {
            System.out.println(0);
            return;
        }
        
        // 预处理阶乘与逆元
        long[] fac = new long[n + 1];
        long[] ifac = new long[n + 1];
        fac[0] = 1;
        for (int i = 1; i <= n; i++) {
            fac[i] = fac[i - 1] * i % MOD;
        }
        ifac[n] = modPow(fac[n], MOD - 2);
        for (int i = n - 1; i >= 0; i--) {
            ifac[i] = ifac[i + 1] * (i + 1) % MOD;
        }
        
        long ans = fac[m] * ifac[k] % MOD * ifac[m - k] % MOD;
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
const long long MOD = 1000000007LL;

long long modpow(long long a, long long e) {
    long long r = 1;
    while (e) {
        if (e & 1) r = r * a % MOD;
        a = a * a % MOD;
        e >>= 1;
    }
    return r;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n, k;
    cin >> n >> k;
    
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    
    if (n % 2 == 1) {
        cout << 0 << '\n';
        return 0;
    }
    
    // 迭代 DFS 获取父节点与遍历顺序
    vector<int> parent(n + 1, 0), order;
    order.reserve(n);
    vector<int> st;
    st.reserve(n);
    st.push_back(1);
    parent[1] = -1;
    while (!st.empty()) {
        int u = st.back();
        st.pop_back();
        order.push_back(u);
        for (int v : g[u]) {
            if (v == parent[u]) continue;
            parent[v] = u;
            st.push_back(v);
        }
    }
    
    // 逆序累加子树大小
    vector<int> sz(n + 1, 1);
    for (int i = (int)order.size() - 1; i >= 0; i--) {
        int u = order[i];
        int p = parent[u];
        if (p != -1) sz[p] += sz[u];
    }
    
    // 统计偶子树边数 m
    int m = 0;
    for (int v = 2; v <= n; v++) {
        if (sz[v] % 2 == 0) m++;
    }
    
    if (k > m) {
        cout << 0 << '\n';
        return 0;
    }
    
    // 预处理阶乘与逆元
    vector<long long> fac(n + 1), ifac(n + 1);
    fac[0] = 1;
    for (int i = 1; i <= n; i++) {
        fac[i] = fac[i - 1] * i % MOD;
    }
    ifac[n] = modpow(fac[n], MOD - 2);
    for (int i = n - 1; i >= 0; i--) {
        ifac[i] = ifac[i + 1] * (i + 1) % MOD;
    }
    
    auto C = [&](int N, int K) -> long long {
        if (K < 0 || K > N) return 0;
        return fac[N] * ifac[K] % MOD * ifac[N - K] % MOD;
    };
    
    cout << C(m, k) << '\n';
    return 0;
}
```
