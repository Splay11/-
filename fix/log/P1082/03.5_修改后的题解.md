## 解题思路

本题要求统计二叉树中有多少棵子树是**饱和谱系**（即满二叉树）。根据定义，饱和谱系必须满足：每个非叶子节点都有两个子节点，并且所有叶子节点的深度相同。

我们可以使用动态规划自底向上判断每个节点为根的子树是否为饱和谱系，并记录其高度。具体思路如下：

1. **状态定义**：  
   令 `dp[i]` 表示以细胞 `i` 为根的子树是饱和谱系时的**高度**（叶子高度为 1）。若该子树不是饱和谱系，则 `dp[i] = 0`。

2. **状态转移**：
   - **叶子节点**（左右子节点均为 `-1`）：子树一定是饱和谱系，高度为 `1`，即 `dp[i] = 1`。
   - **非叶子节点**：若左子节点 `l` 和右子节点 `r` 均存在，且它们各自的子树都是饱和谱系，并且高度相等（即 `dp[l] == dp[r]` 且 `dp[l] != 0`），则以当前节点为根的子树也是饱和谱系，高度为 `dp[i] = dp[l] + 1`。
   - **其他情况**（某个子节点缺失，或左右子树不是饱和谱系，或左右高度不等）：该子树不是饱和谱系，`dp[i] = 0`。

3. **计算顺序**：  
   采用后序遍历（DFS），先递归计算左右孩子的 `dp` 值，再根据转移方程确定当前节点的 `dp` 值。

4. **寻找根节点**：  
   输入给出的是一棵二叉树的节点关系，数据保证只有一个节点没有父节点（根节点）。可以通过统计每个节点的入度（被当作子节点的次数），入度为 `0` 的节点即为根节点。

5. **统计答案**：  
   遍历所有节点，统计 `dp[i] != 0` 的节点个数，即为满足条件的细胞数量。

## 复杂度分析

- **时间复杂度**：$O(n)$。每个节点被访问一次，递归或迭代均在线性时间内完成。
- **空间复杂度**：$O(n)$。需要存储树的结构（左右孩子编号）、`dp` 数组以及递归调用栈（最坏情况为链状树，递归深度 $O(n)$）。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    left = [0] * (n + 1)
    right = [0] * (n + 1)
    indeg = [0] * (n + 1)

    for i in range(1, n + 1):
        l = int(next(it))
        r = int(next(it))
        left[i] = l
        right[i] = r
        if l != -1:
            indeg[l] += 1
        if r != -1:
            indeg[r] += 1

    # 寻找根节点（入度为0）
    root = 1
    for i in range(1, n + 1):
        if indeg[i] == 0:
            root = i
            break

    dp = [0] * (n + 1)

    def dfs(u):
        l = left[u]
        r = right[u]
        if l == -1 and r == -1:
            dp[u] = 1
            return
        if l != -1:
            dfs(l)
        if r != -1:
            dfs(r)
        if l != -1 and r != -1 and dp[l] == dp[r] and dp[l] != 0:
            dp[u] = dp[l] + 1
        else:
            dp[u] = 0

    dfs(root)

    ans = sum(1 for v in dp[1:] if v != 0)
    print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.*;

public class Main {
    static int[] left, right, dp;
    static int n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        left = new int[n + 1];
        right = new int[n + 1];
        int[] indeg = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            left[i] = l;
            right[i] = r;
            if (l != -1) indeg[l]++;
            if (r != -1) indeg[r]++;
        }

        int root = 1;
        for (int i = 1; i <= n; i++) {
            if (indeg[i] == 0) {
                root = i;
                break;
            }
        }

        dp = new int[n + 1];
        dfs(root);

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (dp[i] != 0) ans++;
        }
        System.out.println(ans);
    }

    static void dfs(int u) {
        int l = left[u];
        int r = right[u];
        // 叶子节点
        if (l == -1 && r == -1) {
            dp[u] = 1;
            return;
        }
        if (l != -1) dfs(l);
        if (r != -1) dfs(r);
        // 左右子树均为饱和谱系且高度相等
        if (l != -1 && r != -1 && dp[l] == dp[r] && dp[l] != 0) {
            dp[u] = dp[l] + 1;
        } else {
            dp[u] = 0;
        }
    }
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

vector<int> left_child, right_child, dp;
int n;

void dfs(int u) {
    int l = left_child[u];
    int r = right_child[u];
    // 叶子节点
    if (l == -1 && r == -1) {
        dp[u] = 1;
        return;
    }
    if (l != -1) dfs(l);
    if (r != -1) dfs(r);
    // 左右子树均为饱和谱系且高度相等
    if (l != -1 && r != -1 && dp[l] == dp[r] && dp[l] != 0) {
        dp[u] = dp[l] + 1;
    } else {
        dp[u] = 0;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    left_child.resize(n + 1);
    right_child.resize(n + 1);
    vector<int> indeg(n + 1, 0);

    for (int i = 1; i <= n; i++) {
        int l, r;
        cin >> l >> r;
        left_child[i] = l;
        right_child[i] = r;
        if (l != -1) indeg[l]++;
        if (r != -1) indeg[r]++;
    }

    int root = 1;
    for (int i = 1; i <= n; i++) {
        if (indeg[i] == 0) {
            root = i;
            break;
        }
    }

    dp.resize(n + 1, 0);
    dfs(root);

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (dp[i] != 0) ans++;
    }
    cout << ans << endl;
    return 0;
}
```
