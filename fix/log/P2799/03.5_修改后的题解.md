## 解题思路

1. **位图表示**：把每台探针的覆盖串看成长度为 $m$ 的位图，`'1'` 表示覆盖对应探测点。
2. **规模限制**：一次最多启用 $8$ 台，$n\le 25$，枚举大小为 $1\sim 8$ 的所有组合即可，组合数 $\sum_{k=1}^{8}C(n,k)$ 可接受。
3. **覆盖计算**：对每个组合，将所选探针的位图做按位或，统计 `'1'` 的个数即为覆盖点数。
4. **维护答案**：记录最大覆盖数，以及达到该覆盖数的方案个数。

## 复杂度分析

- **时间复杂度**：$O\bigl(\sum_{k=1}^{8} C(n,k)\cdot k\bigr)$。最坏约数百万次位运算合并。
- **空间复杂度**：$O(n)$。只需保存 $n$ 台探针的位图。

## 代码实现

### Python

```python
import sys
import itertools

def main():
    input_line = sys.stdin.readline().strip()
    if not input_line:
        return
    n, m = map(int, input_line.split())
    cams = []
    for _ in range(n):
        s = sys.stdin.readline().strip()
        cam = int(s, 2)
        cams.append(cam)

    maxCoverage = 0
    ways = 0
    for k in range(1, min(n, 8) + 1):
        for comb in itertools.combinations(range(n), k):
            union = 0
            for idx in comb:
                union |= cams[idx]
            cov = bin(union).count('1')
            if cov > maxCoverage:
                maxCoverage = cov
                ways = 1
            elif cov == maxCoverage:
                ways += 1
    print(maxCoverage, ways)

if __name__ == '__main__':
    main()
```

### Java

```java
import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = br.readLine().trim().split("\\s+");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        BitSet[] cams = new BitSet[n];
        for (int i = 0; i < n; i++) {
            String s = br.readLine().trim();
            cams[i] = new BitSet(m);
            for (int j = 0; j < m; j++) {
                if (s.charAt(j) == '1') {
                    cams[i].set(j);
                }
            }
        }
        int[] maxCoverage = new int[]{0};
        long[] ways = new long[]{0};
        dfs(0, 0, new BitSet(m), n, m, cams, maxCoverage, ways);
        System.out.println(maxCoverage[0] + " " + ways[0]);
    }

    public static void dfs(int start, int count, BitSet cur, int n, int m, BitSet[] cams, int[] maxCoverage, long[] ways) {
        if (count > 0) {
            int cov = cur.cardinality();
            if (cov > maxCoverage[0]) {
                maxCoverage[0] = cov;
                ways[0] = 1;
            } else if (cov == maxCoverage[0]) {
                ways[0]++;
            }
        }
        if (count == 8) return;
        for (int i = start; i < n; i++) {
            BitSet next = (BitSet) cur.clone();
            next.or(cams[i]);
            dfs(i + 1, count + 1, next, n, m, cams, maxCoverage, ways);
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <bitset>
#include <functional>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<bitset<128>> cams(n);
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < m; j++) {
            if (s[j] == '1') cams[i].set(j);
        }
    }
    int maxCoverage = 0;
    long long ways = 0;
    function<void(int, int, bitset<128>&)> dfs = [&](int start, int count, bitset<128> &cur) {
        if (count > 0) {
            int cov = cur.count();
            if (cov > maxCoverage) {
                maxCoverage = cov;
                ways = 1;
            } else if (cov == maxCoverage) {
                ways++;
            }
        }
        if (count == 8) return;
        for (int i = start; i < n; i++) {
            bitset<128> next = cur | cams[i];
            dfs(i + 1, count + 1, next);
        }
    };
    bitset<128> empty;
    dfs(0, 0, empty);
    cout << maxCoverage << " " << ways << endl;
    return 0;
}
```
