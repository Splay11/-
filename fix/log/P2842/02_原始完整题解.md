# 题解

## 题面描述

题目给定 $T$ 组测试数据，每组数据中有 $n$ 个任务和 $m$ 个工人。  
每个任务有两个属性：  
- $a_i$：完成该任务所要求的最小能力值  
- $p_i$：完成该任务获得的收益  

每个工人有一个能力值 $b_i$，只有当工人的能力不低于任务所要求的能力值时，才能完成该任务。  
同一个任务可以被多个工人完成，每完成一次获得一次任务的收益，因此任务的总收益为 $p_i$ 乘以该任务被完成的次数。  
要求在每个工人最多只能分配一个任务的前提下，求出最大可能的总收益。

---

## 解题思路

1. **贪心思想**：  
   每个工人能够完成所有满足条件（即工人能力 $b_i \ge a_j$）的任务，收益为对应任务的 $p_j$。为了使总收益最大化，应该让每个工人选择所能完成的任务中收益最高的那一个。

2. **排序与预处理**：  
   - 首先对所有任务按照 $a_i$ 从小到大排序。  
   - 遍历排序后的任务列表，使用一个变量（记为 $max\_profit$）记录到当前任务为止出现的最大收益，这样可以构造出一个“能力阈值到最大收益”的映射函数 $f(x)$，即对于能力值 $x$，所有要求不超过 $x$ 的任务中最大的收益是多少。  
   - 对于每个工人，其最优收益就是 $f(b_i)$。

3. **统计总收益**：  
   对所有工人计算各自能获得的最高任务收益之和，即为答案。

4. **时间复杂度**：  
   - 对任务排序的复杂度为 $O(n \log n)$，  
   - 对工人遍历的复杂度为 $O(m)$，  
   总体复杂度可接受。

---

## 代码分析

- **预处理任务**：  
  排序后，遍历任务数组时令变量 $best$ 保存当前遇到的最大收益，如果当前任务的收益 $p_i$ 比 $best$ 大，则更新 $best$，然后将当前任务对应的最大收益存入映射数组中。这样，当处理工人时，遍历任务同时更新“能力-收益”对照表，可以用双指针或二分查找方式迅速得到满足能力要求的最大收益。

- **工人匹配**：  
  遍历每个工人的能力值 $b_i$，利用二分查找或者预处理好的“能力-收益”表来获得 $f(b_i)$，然后累加得到最终收益。

- **问题本质**：  
  本题的本质是对于每个工人找到其对应的“可行任务”中收益最高的任务，并对所有工人的局部最优决策进行累加，从而形成全局最优收益。数据规模较大时，应尽量使用预处理和高效搜索算法（如排序、二分查找等）。

## C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
 
struct Task {
    int require;    // a_i：任务所需能力
    int profit;     // p_i：任务的收益
};
 
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
 
    int T;
    cin >> T; // 读取测试数据组数T
    
    while(T--){
        int n, m;
        cin >> n >> m; // 读取任务数n和工人数m
        
        vector<Task> tasks(n);
        // 读取每个任务的要求能力a_i
        for(int i = 0; i < n; i++){
            cin >> tasks[i].require;
        }
        // 读取每个任务的收益p_i
        for(int i = 0; i < n; i++){
            cin >> tasks[i].profit;
        }
 
        vector<int> workers(m);
        // 读取每个工人的能力值b_i
        for(int i = 0; i < m; i++){
            cin >> workers[i];
        }
 
        // 对任务按照要求能力从小到大排序
        sort(tasks.begin(), tasks.end(), [](const Task& a, const Task& b) {
            return a.require < b.require;
        });
 
        // 对工人能力排序，便于二分查找
        sort(workers.begin(), workers.end());
 
        // 使用变量best记录遍历任务时的最大收益
        vector<pair<int,int>> profitTable; // 存储 (require, bestProfit) 对
        int best = 0;
        for(auto &task : tasks){
            if(task.profit > best) {
                best = task.profit;
                // 遇到更高收益的任务时记录下来
                profitTable.push_back({task.require, best});
            }
        }
 
        long long ans = 0;
        // 对每个工人寻找能够完成任务中的最大收益
        for(auto ability : workers){
            
            int l = 0, r = profitTable.size()-1;
            int idx = -1;
            while(l <= r){
                int mid = (l + r) >> 1;
                if(profitTable[mid].first <= ability){
                    idx = mid;
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            // 如果找到可完成的任务，则将对应最大收益加入答案
            if(idx != -1){
                ans += profitTable[idx].second;
            }
        }
 
        cout << ans << "\n"; // 输出本组测试数据的最大收益
    }
 
    return 0;
}
```

## Python

```python
# 读取输入采用 sys.stdin 提高速度
import sys
import bisect

# 输入所有数据
input_data = sys.stdin.read().split()
index = 0

T = int(input_data[index])  # 测试数据组数T
index += 1
results = []

for _ in range(T):
    n = int(input_data[index])    # 任务数n
    m = int(input_data[index+1])  # 工人数m
    index += 2
    
    # 读取任务要求能力a_i
    a = list(map(int, input_data[index:index+n]))
    index += n
    # 读取任务收益p_i
    p = list(map(int, input_data[index:index+n]))
    index += n
    # 读取工人能力b_i
    b = list(map(int, input_data[index:index+m]))
    index += m
    
    tasks = sorted(zip(a, p), key=lambda x: x[0])  # 按照a_i排序
    # 构造能力-收益表，profit_table 存储 (require, best_profit)
    profit_table = []
    best = 0
    for require, profit in tasks:
        if profit > best:
            best = profit
            profit_table.append((require, best))
    
    # 为二分查找构造一个要求能力的列表
    requires = [item[0] for item in profit_table]
    
    ans = 0
    # 对每个工人进行查找
    for ability in sorted(b):
        # bisect_right 返回第一个比 ability 大的位置
        pos = bisect.bisect_right(requires, ability)
        if pos:
            ans += profit_table[pos - 1][1]
    results.append(ans)

# 输出所有测试数据的结果
sys.stdout.write("\n".join(map(str, results)))
```

## Java

```java
import java.io.*;
import java.util.*;
// 定义任务类，包含任务要求能力a和收益p
class Task {
    int require;
    int profit;
    public Task(int require, int profit) {
        this.require = require;
        this.profit = profit;
    }
}
public class Main {
    public static void main(String[] args) throws IOException {
        // 使用 BufferedReader 和 BufferedWriter 提高输入输出效率
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        int T = Integer.parseInt(br.readLine().trim()); // 测试数据组数T
        while(T-- > 0) {
            String[] parts = br.readLine().trim().split("\\s+");
            int n = Integer.parseInt(parts[0]); // 任务数n
            int m = Integer.parseInt(parts[1]); // 工人数m
            
            // 读取任务所需能力a_i
            String[] aStr = br.readLine().trim().split("\\s+");
            int[] a = new int[n];
            for(int i = 0; i < n; i++){
                a[i] = Integer.parseInt(aStr[i]);
            }
            
            // 读取任务收益p_i
            String[] pStr = br.readLine().trim().split("\\s+");
            int[] p = new int[n];
            for(int i = 0; i < n; i++){
                p[i] = Integer.parseInt(pStr[i]);
            }
            
            // 读取工人能力b_i
            String[] bStr = br.readLine().trim().split("\\s+");
            int[] b = new int[m];
            for(int i = 0; i < m; i++){
                b[i] = Integer.parseInt(bStr[i]);
            }
            
            // 构造任务数组，并按照任务要求能力a升序排序
            Task[] tasks = new Task[n];
            for (int i = 0; i < n; i++){
                tasks[i] = new Task(a[i], p[i]);
            }
            Arrays.sort(tasks, new Comparator<Task>() {
                public int compare(Task t1, Task t2) {
                    return Integer.compare(t1.require, t2.require);
                }
            });
            
            // 构造利润映射表 profitTable，存储 (require, bestProfit)
            ArrayList<int[]> profitTable = new ArrayList<>();
            int best = 0;
            for(Task task : tasks) {
                if(task.profit > best) {
                    best = task.profit;
                    profitTable.add(new int[]{task.require, best});
                }
            }
            
            // 对工人能力数组进行排序
            Arrays.sort(b);
            long ans = 0;
            // 对每个工人利用二分查找匹配合适的任务
            for (int ability : b) {
                int idx = binarySearch(profitTable, ability);
                if(idx != -1) {
                    ans += profitTable.get(idx)[1];
                }
            }
            bw.write(ans + "\n");
        }
        bw.flush();
        bw.close();
        br.close();
    }
    
    // 二分查找: 在 profitTable 中查找满足 require <= ability 的最大下标
    private static int binarySearch(ArrayList<int[]> profitTable, int ability) {
        int l = 0, r = profitTable.size() - 1;
        int idx = -1;
        while(l <= r) {
            int mid = (l + r) / 2;
            if(profitTable.get(mid)[0] <= ability) {
                idx = mid;
                l = mid + 1;
            } else {
                r = mid - 1;
            }
        }
        return idx;
    }
}
```
