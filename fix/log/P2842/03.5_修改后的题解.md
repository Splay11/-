## 解题思路

每名工程师最多承担一项任务，且同一任务可被多人完成，因此工程师之间互不影响：每名工程师应选择自己能接的任务中收益最大的那一项。

将任务按能力要求 $a_i$ 升序排序。从左到右扫描时维护当前已出现的最大收益，得到单调的“能力阈值 $\to$ 最大收益”表。对每名工程师的能力 $b_j$，在该表上二分找到 $\le b_j$ 的最大阈值，累加对应收益。

## 复杂度分析

时间复杂度 $O((n+m)\log n)$（排序与二分），空间复杂度 $O(n+m)$。

## 代码实现

### Python

```python
import sys
import bisect

input_data = sys.stdin.read().split()
index = 0

T = int(input_data[index])
index += 1
results = []

for _ in range(T):
    n = int(input_data[index])
    m = int(input_data[index + 1])
    index += 2
    a = list(map(int, input_data[index:index + n]))
    index += n
    p = list(map(int, input_data[index:index + n]))
    index += n
    b = list(map(int, input_data[index:index + m]))
    index += m

    tasks = sorted(zip(a, p), key=lambda x: x[0])
    profit_table = []
    best = 0
    for require, profit in tasks:
        if profit > best:
            best = profit
            profit_table.append((require, best))

    requires = [item[0] for item in profit_table]
    ans = 0
    for ability in sorted(b):
        pos = bisect.bisect_right(requires, ability)
        if pos:
            ans += profit_table[pos - 1][1]
    results.append(ans)

sys.stdout.write("\n".join(map(str, results)))
```

### Java

```java
import java.io.*;
import java.util.*;

class Task {
    int require;
    int profit;
    public Task(int require, int profit) {
        this.require = require;
        this.profit = profit;
    }
}

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        int T = Integer.parseInt(br.readLine().trim());
        while (T-- > 0) {
            String[] parts = br.readLine().trim().split("\\s+");
            int n = Integer.parseInt(parts[0]);
            int m = Integer.parseInt(parts[1]);
            String[] aStr = br.readLine().trim().split("\\s+");
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = Integer.parseInt(aStr[i]);
            String[] pStr = br.readLine().trim().split("\\s+");
            int[] p = new int[n];
            for (int i = 0; i < n; i++) p[i] = Integer.parseInt(pStr[i]);
            String[] bStr = br.readLine().trim().split("\\s+");
            int[] b = new int[m];
            for (int i = 0; i < m; i++) b[i] = Integer.parseInt(bStr[i]);

            Task[] tasks = new Task[n];
            for (int i = 0; i < n; i++) tasks[i] = new Task(a[i], p[i]);
            Arrays.sort(tasks, Comparator.comparingInt(t -> t.require));

            ArrayList<int[]> profitTable = new ArrayList<>();
            int best = 0;
            for (Task task : tasks) {
                if (task.profit > best) {
                    best = task.profit;
                    profitTable.add(new int[]{task.require, best});
                }
            }

            Arrays.sort(b);
            long ans = 0;
            for (int ability : b) {
                int idx = binarySearch(profitTable, ability);
                if (idx != -1) ans += profitTable.get(idx)[1];
            }
            bw.write(ans + "\n");
        }
        bw.flush();
        bw.close();
        br.close();
    }

    private static int binarySearch(ArrayList<int[]> profitTable, int ability) {
        int l = 0, r = profitTable.size() - 1, idx = -1;
        while (l <= r) {
            int mid = (l + r) / 2;
            if (profitTable.get(mid)[0] <= ability) {
                idx = mid;
                l = mid + 1;
            } else {
                r = mid - 1;
            }
        }
        return idx;
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Task {
    int require;
    int profit;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while (T--) {
        int n, m;
        cin >> n >> m;
        vector<Task> tasks(n);
        for (int i = 0; i < n; i++) cin >> tasks[i].require;
        for (int i = 0; i < n; i++) cin >> tasks[i].profit;
        vector<int> workers(m);
        for (int i = 0; i < m; i++) cin >> workers[i];

        sort(tasks.begin(), tasks.end(), [](const Task& a, const Task& b) {
            return a.require < b.require;
        });

        vector<pair<int,int>> profitTable;
        int best = 0;
        for (auto &task : tasks) {
            if (task.profit > best) {
                best = task.profit;
                profitTable.push_back({task.require, best});
            }
        }

        long long ans = 0;
        for (auto ability : workers) {
            int l = 0, r = (int)profitTable.size() - 1, idx = -1;
            while (l <= r) {
                int mid = (l + r) >> 1;
                if (profitTable[mid].first <= ability) {
                    idx = mid;
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            if (idx != -1) ans += profitTable[idx].second;
        }
        cout << ans << "\n";
    }
    return 0;
}
```
