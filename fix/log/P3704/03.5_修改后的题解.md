## 解题思路

这是一个经典的**最长公共子串**问题，可以使用**动态规划**配合**滚动数组**高效求解。题目保证最长公共连续片段存在且唯一，因此只需在遍历过程中记录全局最优解即可。

设文本序列 $S$ 的长度为 $n$，$T$ 的长度为 $m$（$1 \le n,m < 5000$）。定义：

- `dp[j]`：以 $S$ 当前处理到的第 $i$ 个字符（下标从 1 开始）和 $T$ 的第 $j$ 个字符**结尾**的最长公共后缀长度。  
  为了节省空间，我们使用一维数组 `dp[0..m]`，外层循环扫描 $S$ 的每个字符，内层从右向左更新 `dp`。

**状态转移：**

- 当 $S[i-1] = T[j-1]$ 时，`dp[j] = dp[j-1] + 1`（累积公共长度）。
- 当字符不同时，`dp[j] = 0`（公共子串中断）。

**全局最优记录：**

- 用 `bestLen` 记录当前找到的最长公共子串长度，`endPos` 记录该子串在 $S$ 中的结束位置（下标从 1 开始）。
- 每次更新 `dp[j]` 后，若 `dp[j] > bestLen`，则更新 `bestLen` 和 `endPos = i`。

**结果提取：**

- 最长公共连续片段为 $S[endPos-bestLen : endPos]$（Python 切片语义）。

**为什么内层从右向左遍历？**  
因为 `dp[j]` 依赖于上一轮（$i-1$）的 `dp[j-1]`，正序更新会覆盖旧值；倒序则保证读取的 `dp[j-1]` 仍是上一行的值，从而正确实现二维 DP 的一维优化。

## 复杂度分析

- 时间复杂度：$O(n \times m)$，需要双重循环扫描两个序列。
- 空间复杂度：$O(m)$，仅使用一维长度为 $m+1$ 的数组，等价于 $O(\min(n, m))$。

## 代码实现

### Python

```python
import sys

def longest_common_substring(s: str, t: str) -> str:
    """返回 s 和 t 的最长公共连续片段"""
    m = len(t)
    dp = [0] * (m + 1)      # dp[j]：以 t[j-1] 结尾的公共后缀长度
    best_len, end_pos = 0, 0  # 全局最优长度及在 s 中的结束位置(下标从1开始)

    for i in range(1, len(s) + 1):
        ch = s[i - 1]
        # 倒序更新，确保 dp[j-1] 仍是上一轮的值
        for j in range(m, 0, -1):
            if ch == t[j - 1]:
                dp[j] = dp[j - 1] + 1
                if dp[j] > best_len:
                    best_len = dp[j]
                    end_pos = i
            else:
                dp[j] = 0

    return s[end_pos - best_len : end_pos]

if __name__ == "__main__":
    raw = sys.stdin.read().strip()
    # 输入格式："S","T"，用双引号分割后提取第2和第4个字段
    parts = raw.split('"')
    # parts[0] 为空，parts[1] 是 S，parts[2] 是逗号及其可能的空格，parts[3] 是 T
    S = parts[1]
    T = parts[3]
    ans = longest_common_substring(S, T)
    print(f'"{ans}"')
```

### Java

```java
import java.io.*;

class Main {
    // 求两个字符串的最长公共连续片段
    public static String lcs(String a, String b) {
        int m = b.length();
        int[] dp = new int[m + 1];  // dp[j]：以 b[j-1] 结尾的公共后缀长度
        int bestLen = 0, endPos = 0;

        for (int i = 1; i <= a.length(); i++) {
            char c = a.charAt(i - 1);
            for (int j = m; j >= 1; j--) {
                if (c == b.charAt(j - 1)) {
                    dp[j] = dp[j - 1] + 1;
                    if (dp[j] > bestLen) {
                        bestLen = dp[j];
                        endPos = i;
                    }
                } else {
                    dp[j] = 0;
                }
            }
        }
        return a.substring(endPos - bestLen, endPos);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String raw = br.readLine().trim();  // 读取输入行
        // 按双引号分割，提取 S 和 T
        String[] parts = raw.split("\"");
        String S = parts[1];
        String T = parts[3];
        String ans = lcs(S, T);
        System.out.println("\"" + ans + "\"");
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

// 返回 s 和 t 的最长公共连续片段
string lcs(const string& s, const string& t) {
    int m = t.size();
    vector<int> dp(m + 1, 0);   // dp[j]：以 t[j-1] 结尾的公共后缀长度
    int bestLen = 0, endPos = 0;

    for (int i = 1; i <= (int)s.size(); ++i) {
        char c = s[i - 1];
        for (int j = m; j >= 1; --j) {
            if (c == t[j - 1]) {
                dp[j] = dp[j - 1] + 1;
                if (dp[j] > bestLen) {
                    bestLen = dp[j];
                    endPos = i;
                }
            } else {
                dp[j] = 0;
            }
        }
    }
    return s.substr(endPos - bestLen, bestLen);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string raw;
    getline(cin, raw);   // 读取一行
    // 按双引号分割，提取两个字符串
    size_t first = raw.find('"');
    size_t second = raw.find('"', first + 1);
    size_t third = raw.find('"', second + 1);
    size_t fourth = raw.find('"', third + 1);
    string S = raw.substr(first + 1, second - first - 1);
    string T = raw.substr(third + 1, fourth - third - 1);

    string ans = lcs(S, T);
    cout << '"' << ans << '"' << "\n";
    return 0;
}
```
