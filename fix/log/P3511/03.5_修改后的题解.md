## 解题思路

类型 `1` 询问的是区间内出现奇数次的不同取值的异或。把区间里每个位置上的数逐个异或，相同值出现偶数次会互相抵消，剩下的恰好是出现奇数次的那些值。令前缀异或 $S[i]=a_1\oplus a_2\oplus\cdots\oplus a_i$，则

$$
O(l,r)=S[r]\oplus S[l-1].
$$

类型 `2` 询问的是出现偶数次（且至少两次）的不同取值的异或。令 $D(l,r)$ 为区间内每个不同取值各取一次再异或，则偶数次取值与奇数次取值互不相交，且并起来正好是全部出现过的取值，因此

$$
E(l,r)=D(l,r)\oplus O(l,r).
$$

$D(l,r)$ 可以离线按右端点递增计算：用树状数组在“每个值最后一次出现的位置”上挂该值本身。扫到位置 $i$、值为 $x$ 时，若 $x$ 曾出现在 $p$，先在 $p$ 处异或掉 $x$ 以移除旧位置，再在 $i$ 处异或上 $x$。此时对 $[l,r]$ 的区间异或就是该前缀内出现过的不同值的异或。

实现时：类型 `1` 直接用前缀异或回答；类型 `2` 按 $r$ 分桶，扫到 $r$ 时用树状数组求出 $D$ 再与 $O$ 异或。

## 复杂度分析

预处理前缀异或线性；离线扫描每个位置一次树状数组更新，每个询问一次查询。

- 时间复杂度：$O((n+q)\log n)$。
- 空间复杂度：$O(n+q)$。

在 $n,q\le 10^5$ 的限制下可以通过。

## 代码实现

### Python

```python
import sys

class BIT:
    def __init__(self, n):
        self.n = n
        self.t = [0] * (n + 1)
    def add(self, i, v):
        n = self.n
        while i <= n:
            self.t[i] ^= v
            i += i & -i
    def sum(self, i):
        r = 0
        while i > 0:
            r ^= self.t[i]
            i -= i & -i
        return r
    def range(self, l, r):
        if l > r:
            return 0
        return self.sum(r) ^ self.sum(l - 1)

data = sys.stdin.buffer.read().split()
it = iter(data)
try:
    n = int(next(it))
    q = int(next(it))
except StopIteration:
    sys.exit()

a = [0] * (n + 1)
pre = [0] * (n + 1)
for i in range(1, n + 1):
    a[i] = int(next(it))
    pre[i] = pre[i - 1] ^ a[i]

ans = [0] * q
bucket = [[] for _ in range(n + 1)]
for idx in range(q):
    op = int(next(it)); l = int(next(it)); r = int(next(it))
    if op == 1:
        ans[idx] = pre[r] ^ pre[l - 1]
    else:
        bucket[r].append((l, idx))

bit = BIT(n)
last = {}

for r in range(1, n + 1):
    x = a[r]
    if x in last:
        bit.add(last[x], x)
    bit.add(r, x)
    last[x] = r

    for l, idx in bucket[r]:
        D = bit.range(l, r)
        O = pre[r] ^ pre[l - 1]
        ans[idx] = D ^ O

out = '\n'.join(str(v) for v in ans)
sys.stdout.write(out + '\n')
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static class BIT {
        int n;
        int[] t;
        BIT(int n) { this.n = n; this.t = new int[n + 1]; }
        void add(int i, int v) {
            for (; i <= n; i += i & -i) t[i] ^= v;
        }
        int sum(int i) {
            int r = 0;
            for (; i > 0; i -= i & -i) r ^= t[i];
            return r;
        }
        int range(int l, int r) {
            if (l > r) return 0;
            return sum(r) ^ sum(l - 1);
        }
    }

    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        int nextInt() throws IOException {
            int c, sgn = 1, x = 0;
            do { c = read(); } while (c <= 32);
            if (c == '-') { sgn = -1; c = read(); }
            while (c > 32) {
                x = x * 10 + (c - '0');
                c = read();
            }
            return x * sgn;
        }
    }

    static class Q {
        int l, idx;
        Q(int l, int idx) { this.l = l; this.idx = idx; }
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        int n = fs.nextInt();
        int q = fs.nextInt();
        int[] a = new int[n + 1];
        int[] pre = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = fs.nextInt();
            pre[i] = pre[i - 1] ^ a[i];
        }

        long[] ans = new long[q];
        @SuppressWarnings("unchecked")
        ArrayList<Q>[] bucket = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) bucket[i] = new ArrayList<>();

        for (int i = 0; i < q; i++) {
            int op = fs.nextInt(), l = fs.nextInt(), r = fs.nextInt();
            if (op == 1) {
                ans[i] = pre[r] ^ pre[l - 1];
            } else {
                bucket[r].add(new Q(l, i));
            }
        }

        BIT bit = new BIT(n);
        HashMap<Integer, Integer> last = new HashMap<>(n * 2);
        for (int r = 1; r <= n; r++) {
            int x = a[r];
            Integer p = last.get(x);
            if (p != null) bit.add(p, x);
            bit.add(r, x);
            last.put(x, r);

            for (Q qq : bucket[r]) {
                int l = qq.l, idx = qq.idx;
                int D = bit.range(l, r);
                int O = pre[r] ^ pre[l - 1];
                ans[idx] = D ^ O;
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < q; i++) {
            sb.append(ans[i]).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct BIT {
    int n;
    vector<int> t;
    BIT(int n=0): n(n), t(n+1, 0) {}
    void add(int i, int v) {
        for (; i <= n; i += i & -i) t[i] ^= v;
    }
    int sum(int i) {
        int r = 0;
        for (; i > 0; i -= i & -i) r ^= t[i];
        return r;
    }
    int range(int l, int r) {
        if (l > r) return 0;
        return sum(r) ^ sum(l - 1);
    }
};

struct Q {
    int l, idx;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    if (!(cin >> n >> q)) return 0;
    vector<int> a(n + 1), pre(n + 1, 0);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        pre[i] = pre[i - 1] ^ a[i];
    }

    vector<long long> ans(q);
    vector<vector<Q>> bucket(n + 1);
    for (int i = 0; i < q; ++i) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {
            ans[i] = pre[r] ^ pre[l - 1];
        } else {
            bucket[r].push_back({l, i});
        }
    }

    BIT bit(n);
    unordered_map<int,int> last;
    last.reserve(n * 2);

    for (int r = 1; r <= n; ++r) {
        int x = a[r];
        auto it = last.find(x);
        if (it != last.end()) {
            bit.add(it->second, x);
        }
        bit.add(r, x);
        last[x] = r;

        for (auto &qq : bucket[r]) {
            int l = qq.l;
            int idx = qq.idx;
            int D = bit.range(l, r);
            int O = pre[r] ^ pre[l - 1];
            ans[idx] = D ^ O;
        }
    }

    for (int i = 0; i < q; ++i) {
        cout << ans[i] << '\n';
    }
    return 0;
}
```
