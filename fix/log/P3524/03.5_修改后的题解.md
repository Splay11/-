## 解题思路

本题要求模拟一个按固定顺序 `A → L → I` 循环选取零件的组装过程，零件数量由输入序列中有效字符的数量决定。

解题步骤可以分为以下几步：

1. **统计有效零件数量**：遍历原料序列，统计字符 `'A'`、`'L'`、`'I'` 各自出现的次数，分别记为 `cntA`、`cntL`、`cntI`，忽略其他无关大写字母。
2. **判断是否全空**：若 `cntA + cntL + cntI = 0`，说明序列中不包含任何有效零件，直接输出 `-1`。
3. **循环组装模拟**：
   - 使用指针 `p` 表示当前期望抓取的零件类型，初始 `p = 0`（对应 `A`）。后续按 `0 → A, 1 → L, 2 → I` 的映射关系工作。
   - 当还有剩余零件（即 `total = cntA + cntL + cntI > 0`）时，从 `p` 开始顺序检查最多 3 个位置（即 `p, (p+1)%3, (p+2)%3`），选择第一个计数大于 0 的字符作为本次组装的零件。
   - 将该字符添加到结果序列末尾，对应的计数减 1，总剩余数减 1，并将指针 `p` 更新为当前选中字符类型的下一个位置（`(idx + 1) % 3`），以保证下次从正确位置继续循环。
4. **输出结果**：重复上述过程直到所有有效零件用完，所得字符串即为组装序列。每个测试用例输出一行结果；若结果为空串，则输出 `-1`。

## 复杂度分析

对于每个测试用例，设原料序列长度为 $m$，有效零件总数为 $tot \le m$。

- **时间复杂度**：$O(m)$。统计阶段需要遍历一次字符串，耗时 $O(m)$；构造结果时，每次找到下一个可用零件至多尝试 3 次，总计操作次数不超过 $3 \times tot \le 3m$，因此总时间复杂度为 $O(m)$。
- **空间复杂度**：$O(m)$（用于存储结果字符串）。如果忽略结果字符串所占空间（视为输出所需），则额外辅助空间为 $O(1)$，仅使用了常数个计数变量与指针。

## 代码实现

### Python

```python
import sys

def transform_ali(s: str) -> str:
    # 统计 A, L, I 的数量
    cnt = [0, 0, 0]  # 0 -> A, 1 -> L, 2 -> I
    for ch in s:
        if ch == 'A':
            cnt[0] += 1
        elif ch == 'L':
            cnt[1] += 1
        elif ch == 'I':
            cnt[2] += 1

    total = sum(cnt)
    if total == 0:
        return "-1"

    res = []
    p = 0  # 当前起始指针，0:A, 1:L, 2:I
    while total > 0:
        placed = False
        for k in range(3):
            idx = (p + k) % 3
            if cnt[idx] > 0:
                # 将对应字符加入结果
                if idx == 0:
                    res.append('A')
                elif idx == 1:
                    res.append('L')
                else:
                    res.append('I')
                cnt[idx] -= 1
                total -= 1
                p = (idx + 1) % 3  # 下一次从下一个字母开始
                placed = True
                break
        if not placed:  # 理论上不会发生，防御性break
            break

    return "".join(res)


def main():
    data = sys.stdin.read().splitlines()
    if not data:
        return
    n = int(data[0].strip())
    out_lines = []
    for i in range(1, n + 1):
        s = data[i] if i < len(data) else ""
        out_lines.append(transform_ali(s))
    sys.stdout.write("\n".join(out_lines))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static String transformALI(String s) {
        // 统计 A, L, I 的数量
        int[] cnt = new int[3]; // 0:A, 1:L, 2:I
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == 'A') cnt[0]++;
            else if (c == 'L') cnt[1]++;
            else if (c == 'I') cnt[2]++;
        }

        int total = cnt[0] + cnt[1] + cnt[2];
        if (total == 0) {
            return "-1";
        }

        StringBuilder ans = new StringBuilder();
        int p = 0; // 当前起始指针
        while (total > 0) {
            boolean placed = false;
            for (int k = 0; k < 3; k++) {
                int idx = (p + k) % 3;
                if (cnt[idx] > 0) {
                    // 添加对应字符
                    if (idx == 0) ans.append('A');
                    else if (idx == 1) ans.append('L');
                    else ans.append('I');
                    cnt[idx]--;
                    total--;
                    p = (idx + 1) % 3; // 从下一个字母继续
                    placed = true;
                    break;
                }
            }
            if (!placed) break;
        }

        return ans.toString();
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null || line.isEmpty()) return;
        int n = Integer.parseInt(line.trim());

        StringBuilder out = new StringBuilder();
        for (int i = 0; i < n; i++) {
            String s = br.readLine();
            if (s == null) s = "";
            out.append(transformALI(s));
            if (i + 1 < n) out.append('\n');
        }
        System.out.print(out.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

string transformALI(const string& s) {
    // 只统计 A, L, I 的数量，其他字符忽略
    int cnt[3] = {0, 0, 0}; // 0->A, 1->L, 2->I
    for (char c : s) {
        if (c == 'A') cnt[0]++;
        else if (c == 'L') cnt[1]++;
        else if (c == 'I') cnt[2]++;
    }

    int total = cnt[0] + cnt[1] + cnt[2];
    if (total == 0) return "-1";

    string ans;
    int p = 0; // 当前起始指针
    while (total > 0) {
        bool placed = false;
        for (int k = 0; k < 3; ++k) {
            int idx = (p + k) % 3;
            if (cnt[idx] > 0) {
                ans.push_back(idx == 0 ? 'A' : (idx == 1 ? 'L' : 'I'));
                cnt[idx]--;
                total--;
                p = (idx + 1) % 3; // 下一次从下一个字母开始
                placed = true;
                break;
            }
        }
        if (!placed) break; // 防御性break
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    cin.ignore(); // 忽略换行符
    for (int i = 0; i < n; ++i) {
        string s;
        getline(cin, s);
        cout << transformALI(s) << "\n";
    }
    return 0;
}
```
