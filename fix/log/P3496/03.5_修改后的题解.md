## 解题思路

1. **反向计数**  
   记 `X` 的个数为 $\mathrm{cntX}$，总填充方案数为 $2^{\mathrm{cntX}}$。定义“不达标”为最终串中**没有**连续 $k$ 个 `B`。答案为 $2^{\mathrm{cntX}} - \mathrm{bad}$（取模）。

2. **线性 DP 统计不达标方案**  
   记 `okB[i]`：第 $i$ 位可以是 `B`（即 $s_i \ne$ `W`）；`okW[i]`：第 $i$ 位可以是 `W`（即 $s_i \ne$ `B`）。`Wpre[i]` 为前 $i$ 位中 `W` 的个数。

   - $dp[i]$：前 $i$ 位且不含连续 $k$ 个 `B` 的方案数；
   - $g[i]$：前 $i$ 位且第 $i$ 位为 `W` 的合法方案数。边界 $dp[0]=g[0]=1$。

   转移（$i=1..n$）：
   - 放 `W`：若 `okW[i]`，则 $g[i]=dp[i-1]$，并计入 $dp[i]$；
   - 放 `B`：若 `okB[i]`，先加上 $dp[i-1]$；若最后 $k$ 位都可以是 `B`（区间 $[i-k+1,i]$ 无 `W`），则减去以 $i$ 结尾的非法 $k$ 连 `B`，数量为 $g[i-k]$。

3. **特判 $k=1$**  
   达标即“存在一个 `B`”。若原串已有固定 `B`，答案为 $2^{\mathrm{cntX}}$；否则减去全填 `W` 的 `1` 种方案。

## 复杂度分析

- **时间复杂度**：$O(n)$。预处理与 DP 各扫描一次。
- **空间复杂度**：$O(n)$。需要前缀数组与 DP 数组。

## 代码实现

### Python

```python
import sys

MOD = 10**9 + 7

def mod_pow2(x):
    return pow(2, x, MOD)

def solve():
    data = sys.stdin.read().strip().split()
    n = int(data[0]); k = int(data[1])
    s = data[2].strip()

    cntX = s.count('X')
    total = mod_pow2(cntX)

    if k == 1:
        if 'B' in s:
            print(total % MOD)
        else:
            print((total - 1 + MOD) % MOD)
        return

    Wpre = [0] * (n + 1)
    for i, ch in enumerate(s, 1):
        Wpre[i] = Wpre[i - 1] + (1 if ch == 'W' else 0)

    dp = [0] * (n + 1)
    g = [0] * (n + 1)
    dp[0] = g[0] = 1

    for i in range(1, n + 1):
        ch = s[i - 1]
        okB = ch != 'W'
        okW = ch != 'B'

        if okW:
            g[i] = dp[i - 1]
            dp[i] = (dp[i] + dp[i - 1]) % MOD
        else:
            g[i] = 0

        if okB:
            add = dp[i - 1]
            if i >= k and Wpre[i] - Wpre[i - k] == 0:
                add -= g[i - k]
            dp[i] = (dp[i] + add) % MOD

    bad = dp[n] % MOD
    ans = (total - bad) % MOD
    print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1_000_000_007L;

    static long pow2(long e) {
        long a = 2, r = 1;
        while (e > 0) {
            if ((e & 1) == 1) r = (r * a) % MOD;
            a = (a * a) % MOD;
            e >>= 1;
        }
        return r;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        String s = br.readLine().trim();

        int cntX = 0;
        boolean hasB = false;
        for (char c : s.toCharArray()) {
            if (c == 'X') cntX++;
            if (c == 'B') hasB = true;
        }
        long total = pow2(cntX);

        if (k == 1) {
            long ans = hasB ? total : (total - 1 + MOD) % MOD;
            System.out.println(ans);
            return;
        }

        int[] Wpre = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            Wpre[i] = Wpre[i - 1] + (s.charAt(i - 1) == 'W' ? 1 : 0);
        }

        long[] dp = new long[n + 1];
        long[] g = new long[n + 1];
        dp[0] = g[0] = 1;

        for (int i = 1; i <= n; i++) {
            char ch = s.charAt(i - 1);
            boolean okB = ch != 'W';
            boolean okW = ch != 'B';

            if (okW) {
                g[i] = dp[i - 1];
                dp[i] = (dp[i] + dp[i - 1]) % MOD;
            } else g[i] = 0;

            if (okB) {
                long add = dp[i - 1];
                if (i >= k && (Wpre[i] - Wpre[i - k] == 0)) {
                    add = (add - g[i - k]) % MOD;
                }
                dp[i] = (dp[i] + add) % MOD;
            }
        }

        long bad = (dp[n] % MOD + MOD) % MOD;
        long ans = (total - bad) % MOD;
        if (ans < 0) ans += MOD;
        System.out.println(ans);
    }
}
```
