## 解题思路

- 对于每个盒子 $i$（代码中盒子编号从 $0$ 到 $k-1$），先计算该盒子内所有卡片数字的总和，记作 $S_i$。
- 遍历该盒子的每一张卡片，卡片上的数字记为 $a$。如果抽出这张卡片，该盒子的剩余总分为 $S_i - a$。用一个集合记录当前盒子所有可能的剩余总分（去重）。
- 使用一个哈希表（例如 Python 的字典或 C++ 的 `unordered_map`）维护已处理盒子产生的所有剩余总分，键为剩余总分，值为该总分对应的盒子编号。
- 对于当前盒子的每个剩余总分 $x$，检查哈希表中是否已存在键 $x$，且该键对应的盒子编号与当前盒子编号不同。如果存在，则找到了一对可匹配盒子，记录答案为 `YES`。
- 如果所有盒子处理完毕后仍未找到，则输出 `NO`。
- 注意：对于同一盒子内，不同的卡片可能得到相同的剩余总分，因此先用集合并去重，避免重复检查和提高效率。

## 复杂度分析

- 时间复杂度：令所有盒子的卡片总数 $N = \sum n_i \le 5\times 10^5$。对于每组测试数据，我们需要遍历每个盒子的卡片计算总分、生成剩余总分集合，并进行哈希表操作。每次集合与哈希表的插入、查找操作均摊为 $O(1)$。因此总时间复杂度为 $O(N)$，在 $T\le 10$ 组数据下依然能高效运行。  
- 空间复杂度：需要存储每个盒子卡片序列、剩余总分集合以及全局哈希表。最坏情况下所有可能的剩余总分都会存入哈希表，因此空间复杂度为 $O(N)$。

## 代码实现

### Python

```python
import sys

def main():
    data = sys.stdin.read().split()
    it = iter(data)
    T = int(next(it))          # 测试数据组数
    out_lines = []
    for _ in range(T):
        k = int(next(it))      # 盒子数量
        seen = {}              # 剩余总分 -> 盒子编号
        found = False
        for box_idx in range(k):
            n = int(next(it))  # 当前盒子卡片数
            arr = [int(next(it)) for _ in range(n)]
            total = sum(arr)   # 当前盒子的总分
            cur_sums = set()   # 删除一张卡片后的剩余总分集合
            for a in arr:
                cur_sums.add(total - a)
            # 检查是否与之前盒子的剩余总分匹配
            for s in cur_sums:
                if s in seen and seen[s] != box_idx:
                    found = True
            # 记录当前盒子的剩余总分
            for s in cur_sums:
                seen[s] = box_idx
        out_lines.append("YES" if found else "NO")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();                 // 测试数据组数
        StringBuilder sb = new StringBuilder();
        for (int t = 0; t < T; t++) {
            int k = sc.nextInt();             // 盒子数量
            Map<Long, Integer> seen = new HashMap<>(); // 剩余总分 -> 盒子编号
            boolean found = false;
            for (int boxIdx = 0; boxIdx < k; boxIdx++) {
                int n = sc.nextInt();         // 当前盒子卡片数
                long total = 0;
                int[] arr = new int[n];
                for (int j = 0; j < n; j++) {
                    arr[j] = sc.nextInt();
                    total += arr[j];
                }
                Set<Long> curSums = new HashSet<>(); // 删除一张卡片后的剩余总分
                for (int a : arr) {
                    curSums.add(total - a);
                }
                // 检查是否与之前盒子的剩余总分匹配
                for (long s : curSums) {
                    if (seen.containsKey(s) && seen.get(s) != boxIdx) {
                        found = true;
                    }
                }
                // 记录当前盒子的剩余总分
                for (long s : curSums) {
                    seen.put(s, boxIdx);
                }
            }
            sb.append(found ? "YES" : "NO").append("\n");
        }
        System.out.print(sb);
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <unordered_map>
#include <unordered_set>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;                    // 测试数据组数
    cin >> T;
    while (T--) {
        int k;                // 盒子数量
        cin >> k;
        unordered_map<ll, int> seen; // 剩余总分 -> 盒子编号
        bool found = false;
        for (int boxIdx = 0; boxIdx < k; ++boxIdx) {
            int n;            // 当前盒子卡片数
            cin >> n;
            ll total = 0;
            int* arr = new int[n];
            for (int j = 0; j < n; ++j) {
                cin >> arr[j];
                total += arr[j];
            }
            unordered_set<ll> curSums; // 删除一张卡片后的剩余总分
            for (int j = 0; j < n; ++j) {
                curSums.insert(total - arr[j]);
            }
            // 检查是否与之前盒子的剩余总分匹配
            for (ll s : curSums) {
                if (seen.count(s) && seen[s] != boxIdx) {
                    found = true;
                }
            }
            // 记录当前盒子的剩余总分
            for (ll s : curSums) {
                seen[s] = boxIdx;
            }
            delete[] arr;
        }
        cout << (found ? "YES" : "NO") << "\n";
    }
    return 0;
}
```
