## 解题思路

本题需要从带有情感标签的文本反馈中，计算每个单词的关联度得分 $S$（卡方统计量），并输出得分最高的前 $k$ 个单词。具体算法步骤如下：

1. **输入读取与校验**  
   首先读取文档总数 $N$，接着读取接下来的 $N$ 行数据（每行为一个制表符分隔的标签和文本），最后读取一个整数 $k$。  
   如果行数不足、标签既不是 `positive` 也不是 `negative`，或者最后一行无法解析为整数，则判定输入格式错误，输出 $-1$ 并终止。

2. **构建词袋模型并统计频数**  
   - 使用集合（set）对每篇文本进行单词去重，大小写敏感。  
   - 统计两类的文档总数：$n_p$（`positive` 类文档数）和 $n_n$（`negative` 类文档数）。  
   - 对于每个单词，记录两个观测频数：  
     $A$：`positive` 类中包含该单词的文档数；  
     $B$：`negative` 类中包含该单词的文档数。  
   - 派生频数：  
     $C = n_p - A$（`positive` 类中不含该单词的文档数），  
     $D = n_n - B$（`negative` 类中不含该单词的文档数）。

3. **计算关联度得分 $S$**  
   设总文档数 $N = n_p + n_n$。对于每个单元格 $(i,j)$，计算期望频数：  
   $$E_{11} = \frac{(A+B) \times n_p}{N},\quad E_{12} = \frac{(A+B) \times n_n}{N},\quad E_{21} = \frac{(C+D) \times n_p}{N},\quad E_{22} = \frac{(C+D) \times n_n}{N}$$  
   关联度得分定义为：  
   $$S = \sum_{i=1}^{2}\sum_{j=1}^{2}\frac{(O_{ij}-E_{ij})^2}{E_{ij}}$$  
   其中 $O_{11}=A$，$O_{12}=B$，$O_{21}=C$，$O_{22}=D$。当某个期望频数 $E_{ij}=0$ 时，该项不计入（避免除以零）。

4. **排序与输出**  
   将所有单词按 $S$ 降序排列；若得分相同，则按单词的字典序升序排列。  
   输出排序后的前 $k$ 个单词（若 $k$ 超过单词总数，则输出全部，虽然题目保证了 $k$ 不超过不同单词总数）。

## 复杂度分析

- **时间复杂度**：$O(N \cdot L + M \log M)$  
  其中 $N$ 为文档数，$L$ 为单篇文档最大单词数（分词与去重），$M$ 为不同单词总数。  
  对每篇文档进行分词、去重、统计频数需要 $O(N \cdot L)$，计算 $M$ 个单词的 $S$ 值需要 $O(M)$，排序需要 $O(M \log M)$。

- **空间复杂度**：$O(N \cdot L + M)$  
  存储文档文本（可选）及单词频数统计表，每个单词的频数信息占用常数空间，主要的存储开销来自单词字典和去重用的集合。

## 代码实现

### Python

```python
# -*- coding: utf-8 -*-
import sys

def main():
    lines = sys.stdin.read().splitlines()
    if not lines:
        print(-1)
        return

    try:
        N = int(lines[0])
    except:
        print(-1)
        return

    # 需要 N 行文档 + 1 行 k 值
    if len(lines) != N + 2:
        print(-1)
        return

    positive_count = 0
    negative_count = 0
    word_counts = {}  # key: word, value: [A, B]

    for i in range(1, N + 1):
        line = lines[i]
        if "\t" in line:
            label, text = line.split("\t", 1)
        else:
            # 兼容无制表符的情况：第一个单词为标签，其余为文本
            parts = line.split()
            if len(parts) < 2:
                print(-1)
                return
            label = parts[0]
            text = " ".join(parts[1:])

        if label == "positive":
            positive_count += 1
            is_positive = True
        elif label == "negative":
            negative_count += 1
            is_positive = False
        else:
            print(-1)
            return

        # 词袋处理：集合去重
        words = set(text.split())
        for word in words:
            if word not in word_counts:
                word_counts[word] = [0, 0]
            if is_positive:
                word_counts[word][0] += 1
            else:
                word_counts[word][1] += 1

    total = positive_count + negative_count
    features = []

    for word, (A, B) in word_counts.items():
        C = positive_count - A
        D = negative_count - B

        row1 = A + B  # 含有该单词的文档总数
        row2 = C + D  # 不含该单词的文档总数
        col1 = positive_count
        col2 = negative_count

        S = 0.0
        # 计算四个单元格对 S 的贡献
        if row1 > 0 and col1 > 0:
            E = row1 * col1 / total
            if E > 0:
                S += (A - E) ** 2 / E
        if row1 > 0 and col2 > 0:
            E = row1 * col2 / total
            if E > 0:
                S += (B - E) ** 2 / E
        if row2 > 0 and col1 > 0:
            E = row2 * col1 / total
            if E > 0:
                S += (C - E) ** 2 / E
        if row2 > 0 and col2 > 0:
            E = row2 * col2 / total
            if E > 0:
                S += (D - E) ** 2 / E

        features.append((word, S))

    # 按 S 降序，相同时按单词字典序升序
    features.sort(key=lambda x: (-x[1], x[0]))

    try:
        k = int(lines[-1])
    except:
        print(-1)
        return

    out_count = min(k, len(features))
    for i in range(out_count):
        print(features[i][0])

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

class Feature {
    String word;
    double score;
    Feature(String w, double s) { word = w; score = s; }
}

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null) {
            System.out.println(-1);
            return;
        }

        int N;
        try {
            N = Integer.parseInt(line.trim());
        } catch (Exception e) {
            System.out.println(-1);
            return;
        }

        List<String> docs = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            String doc = br.readLine();
            if (doc == null) {
                System.out.println(-1);
                return;
            }
            docs.add(doc);
        }

        String kLine = br.readLine();
        if (kLine == null) {
            System.out.println(-1);
            return;
        }
        int k;
        try {
            k = Integer.parseInt(kLine.trim());
        } catch (Exception e) {
            System.out.println(-1);
            return;
        }

        int positiveCount = 0, negativeCount = 0;
        Map<String, int[]> wordCount = new HashMap<>(); // [A, B]

        for (String doc : docs) {
            String label, text;
            if (doc.contains("\t")) {
                String[] parts = doc.split("\t", 2);
                if (parts.length < 2) {
                    System.out.println(-1);
                    return;
                }
                label = parts[0];
                text = parts[1];
            } else {
                String[] parts = doc.split(" ");
                if (parts.length < 2) {
                    System.out.println(-1);
                    return;
                }
                label = parts[0];
                text = String.join(" ", Arrays.copyOfRange(parts, 1, parts.length));
            }

            boolean isPositive;
            if (label.equals("positive")) {
                positiveCount++;
                isPositive = true;
            } else if (label.equals("negative")) {
                negativeCount++;
                isPositive = false;
            } else {
                System.out.println(-1);
                return;
            }

            String[] words = text.split(" ");
            Set<String> wordSet = new HashSet<>(Arrays.asList(words));
            for (String word : wordSet) {
                int[] counts = wordCount.getOrDefault(word, new int[2]);
                if (isPositive) counts[0]++;
                else counts[1]++;
                wordCount.put(word, counts);
            }
        }

        int total = positiveCount + negativeCount;
        List<Feature> features = new ArrayList<>();

        for (Map.Entry<String, int[]> entry : wordCount.entrySet()) {
            String word = entry.getKey();
            int A = entry.getValue()[0];
            int B = entry.getValue()[1];
            int C = positiveCount - A;
            int D = negativeCount - B;

            int row1 = A + B;
            int row2 = C + D;
            int col1 = positiveCount;
            int col2 = negativeCount;

            double S = 0.0;
            if (row1 > 0 && col1 > 0) {
                double E = (double) row1 * col1 / total;
                if (E > 0) S += (A - E) * (A - E) / E;
            }
            if (row1 > 0 && col2 > 0) {
                double E = (double) row1 * col2 / total;
                if (E > 0) S += (B - E) * (B - E) / E;
            }
            if (row2 > 0 && col1 > 0) {
                double E = (double) row2 * col1 / total;
                if (E > 0) S += (C - E) * (C - E) / E;
            }
            if (row2 > 0 && col2 > 0) {
                double E = (double) row2 * col2 / total;
                if (E > 0) S += (D - E) * (D - E) / E;
            }

            features.add(new Feature(word, S));
        }

        features.sort((a, b) -> {
            if (Math.abs(a.score - b.score) > 1e-9)
                return Double.compare(b.score, a.score);
            return a.word.compareTo(b.word);
        });

        for (int i = 0; i < k && i < features.size(); i++)
            System.out.println(features.get(i).word);
    }
}
```

### C++

```cpp
#include <iostream>
#include <sstream>
#include <vector>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <cmath>
#include <algorithm>
using namespace std;

struct Feature {
    string word;
    double score;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    if (!(cin >> N)) {
        cout << -1;
        return 0;
    }
    cin.ignore(); // 忽略 N 之后的换行符

    vector<string> docs(N);
    for (int i = 0; i < N; i++) {
        if (!getline(cin, docs[i])) {
            cout << -1;
            return 0;
        }
    }

    int k;
    if (!(cin >> k)) {
        cout << -1;
        return 0;
    }

    int positiveCount = 0, negativeCount = 0;
    unordered_map<string, pair<int, int>> wordCount; // first: A, second: B

    for (const string& doc : docs) {
        string label, text;
        size_t pos = doc.find('\t');
        if (pos != string::npos) {
            label = doc.substr(0, pos);
            text = doc.substr(pos + 1);
        } else {
            istringstream iss(doc);
            if (!(iss >> label)) { cout << -1; return 0; }
            getline(iss, text);
            if (!text.empty() && text[0] == ' ') text.erase(0, 1);
        }

        bool isPositive;
        if (label == "positive") {
            positiveCount++;
            isPositive = true;
        } else if (label == "negative") {
            negativeCount++;
            isPositive = false;
        } else {
            cout << -1;
            return 0;
        }

        istringstream textStream(text);
        unordered_set<string> wordSet;
        string word;
        while (textStream >> word)
            wordSet.insert(word);

        for (const string& w : wordSet) {
            if (isPositive)
                wordCount[w].first++;
            else
                wordCount[w].second++;
        }
    }

    int total = positiveCount + negativeCount;
    vector<Feature> features;

    for (const auto& p : wordCount) {
        const string& word = p.first;
        int A = p.second.first;
        int B = p.second.second;
        int C = positiveCount - A;
        int D = negativeCount - B;

        int row1 = A + B;
        int row2 = C + D;
        int col1 = positiveCount;
        int col2 = negativeCount;

        double S = 0.0;
        if (row1 > 0 && col1 > 0) {
            double E = (double)row1 * col1 / total;
            if (E > 0) S += (A - E) * (A - E) / E;
        }
        if (row1 > 0 && col2 > 0) {
            double E = (double)row1 * col2 / total;
            if (E > 0) S += (B - E) * (B - E) / E;
        }
        if (row2 > 0 && col1 > 0) {
            double E = (double)row2 * col1 / total;
            if (E > 0) S += (C - E) * (C - E) / E;
        }
        if (row2 > 0 && col2 > 0) {
            double E = (double)row2 * col2 / total;
            if (E > 0) S += (D - E) * (D - E) / E;
        }

        features.push_back({word, S});
    }

    sort(features.begin(), features.end(), [](const Feature& a, const Feature& b) {
        if (fabs(a.score - b.score) > 1e-9)
            return a.score > b.score;
        return a.word < b.word;
    });

    for (int i = 0; i < k && i < (int)features.size(); i++)
        cout << features[i].word << "\n";

    return 0;
}
```
