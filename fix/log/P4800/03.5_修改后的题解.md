## 解题思路

对于固定的地点 $i$，题目要求对每个 $j$ 计算出 $S(i,j)$，即满足 $k \neq i$、$k \neq j$ 且 $d(i,k) \le d(i,j)$ 的地点 $k$ 的个数。

把定义变形一下会更清晰：

- 考虑地点 $i$，将除 $i$ 之外的所有地点 $j$ 按照亲近度 $d(i,j)$ 从小到大排序；
- 若某个地点 $j$ 的亲近度为 $d(i,j)$，那么所有亲近度 $\le d(i,j)$ 的地点（不含 $i$ 自己）都会被统计，但要去掉 $j$ 自身，因此：
  $$
  S(i,j) = \bigl(\#\{k \neq i \mid d(i,k) \le d(i,j)\}\bigr) - 1
  $$

### 算法步骤

对于每一个地点 $i$：

1. 计算 $i$ 到所有其他地点 $j$ $(j \neq i)$ 的亲近度：
   $$
   d(i,j) = (x_i - x_j)^2 + (y_i - y_j)^2
   $$
   将计算结果连同 $j$ 的编号一起记录。

2. 将所有记录按亲近度从小到大排序。

3. 对排序后的记录按“相同亲近度”分组。  
   假设某一组在排序数组中的位置范围是 $[l, r]$（下标从 $0$ 开始），那么属于该组的任意地点 $j$，其亲近度与组内其他地点相同。此时满足亲近度 $\le d(i,j)$ 的其他地点一共有 $r+1$ 个（包含 $j$ 自己），因此：
   $$
   S(i,j) = (r+1) - 1 = r
   $$

   同一组内的所有地点 $j$ 答案都相同，且均等于该组最后一个位置的下标 $r$。

4. 最后令 $S(i,i) = 0$（对角线上保持为 $0$）。

### 正确性说明

对固定 $i$，排序后前 $r+1$ 个记录恰好是所有亲近度 $\le d(i,j)$ 的其他地点（不包含 $i$），这当中包含了 $j$ 自身。题目要求统计除 $i,j$ 之外的地点，所以去掉 $j$ 后数量即为 $r$。因此按组统一赋值是正确的。

## 复杂度分析

对于每组测试数据：

- 共有 $n$ 个地点，对每个地点 $i$ 需要计算 $n-1$ 个亲近度并排序，单次复杂度为 $O(n \log n)$；
- 共 $n$ 个地点，总时间复杂度为 $O(n^2 \log n)$。

- 结果矩阵 $S$ 需要 $O(n^2)$ 的空间；
- 排序时临时数组需要 $O(n)$ 的空间。

因此整体空间复杂度为 $O(n^2)$。

## 代码实现

### Python

```python
import sys

def solve_case(points):
    # 地点数量
    n = len(points)

    # S[i][j] 表示题目要求的 S(i,j)
    S = [[0] * n for _ in range(n)]

    # 依次固定地点 i
    for i in range(n):
        dist_list = []

        xi, yi = points[i]

        # 计算其他地点与 i 的亲近度
        for j in range(n):
            if i == j:
                continue
            xj, yj = points[j]
            dx = xi - xj
            dy = yi - yj
            d2 = dx * dx + dy * dy
            dist_list.append((d2, j))

        # 按亲近度从小到大排序
        dist_list.sort()

        # 按相同亲近度分组处理
        m = len(dist_list)
        l = 0
        while l < m:
            r = l
            # 找到与 dist_list[l] 亲近度相同的这一整组
            while r + 1 < m and dist_list[r + 1][0] == dist_list[l][0]:
                r += 1

            # 这一组中任意地点 j 的答案都是 r
            for k in range(l, r + 1):
                j = dist_list[k][1]
                S[i][j] = r

            l = r + 1

    return S

def main():
    input = sys.stdin.readline
    T = int(input().strip())
    out_lines = []

    for _ in range(T):
        n = int(input().strip())
        points = []
        for _ in range(n):
            x, y = map(int, input().split())
            points.append((x, y))

        ans = solve_case(points)

        for i in range(n):
            out_lines.append(' '.join(map(str, ans[i])))

    sys.stdout.write('\n'.join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;
import java.util.Collections;
import java.util.StringTokenizer;

public class Main {

    // 用来保存“亲近度 + 地点编号”
    static class Node implements Comparable<Node> {
        long dist;
        int idx;

        Node(long dist, int idx) {
            this.dist = dist;
            this.idx = idx;
        }

        @Override
        public int compareTo(Node other) {
            return Long.compare(this.dist, other.dist);
        }
    }

    // 返回一组测试数据的答案矩阵 S
    static int[][] solveCase(long[][] points) {
        int n = points.length;
        int[][] S = new int[n][n];

        // 依次固定地点 i
        for (int i = 0; i < n; i++) {
            ArrayList<Node> list = new ArrayList<>();

            long xi = points[i][0];
            long yi = points[i][1];

            // 计算其他地点与 i 的亲近度
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }
                long dx = xi - points[j][0];
                long dy = yi - points[j][1];
                long d2 = dx * dx + dy * dy;
                list.add(new Node(d2, j));
            }

            // 按亲近度从小到大排序
            Collections.sort(list);

            // 按相同亲近度分组处理
            int m = list.size();
            int l = 0;
            while (l < m) {
                int r = l;

                // 找到同一亲近度的一整组
                while (r + 1 < m && list.get(r + 1).dist == list.get(l).dist) {
                    r++;
                }

                // 这一组中所有地点的答案都为 r
                for (int k = l; k <= r; k++) {
                    int j = list.get(k).idx;
                    S[i][j] = r;
                }

                l = r + 1;
            }
        }

        return S;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();

        int T = Integer.parseInt(br.readLine().trim());

        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine().trim());
            long[][] points = new long[n][2];

            for (int i = 0; i < n; i++) {
                StringTokenizer st = new StringTokenizer(br.readLine());
                points[i][0] = Long.parseLong(st.nextToken());
                points[i][1] = Long.parseLong(st.nextToken());
            }

            int[][] ans = solveCase(points);

            // 输出矩阵 S
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (j > 0) {
                        sb.append(' ');
                    }
                    sb.append(ans[i][j]);
                }
                sb.append('\n');
            }
        }

        System.out.print(sb.toString());
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

// 用来保存“亲近度 + 地点编号”
struct Node {
    long long dist;
    int idx;

    bool operator < (const Node& other) const {
        return dist < other.dist;
    }
};

// 返回一组测试数据的答案矩阵 S
vector<vector<int>> solve_case(const vector<pair<long long, long long>>& points) {
    int n = (int)points.size();
    vector<vector<int>> S(n, vector<int>(n, 0));

    // 依次固定地点 i
    for (int i = 0; i < n; i++) {
        vector<Node> dist_list;
        long long xi = points[i].first;
        long long yi = points[i].second;

        // 计算其他所有地点与 i 的亲近度
        for (int j = 0; j < n; j++) {
            if (i == j) {
                continue;
            }
            long long dx = xi - points[j].first;
            long long dy = yi - points[j].second;
            long long d2 = dx * dx + dy * dy;
            dist_list.push_back({d2, j});
        }

        // 按亲近度从小到大排序
        sort(dist_list.begin(), dist_list.end());

        // 按相同亲近度分组处理
        int m = (int)dist_list.size();
        int l = 0;
        while (l < m) {
            int r = l;

            // 找到亲近度相同的一整组
            while (r + 1 < m && dist_list[r + 1].dist == dist_list[l].dist) {
                r++;
            }

            // 这一组内所有地点的答案都为 r
            for (int k = l; k <= r; k++) {
                int j = dist_list[k].idx;
                S[i][j] = r;
            }

            l = r + 1;
        }
    }

    return S;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n;
        cin >> n;

        vector<pair<long long, long long>> points(n);

        // 读入所有地点坐标
        for (int i = 0; i < n; i++) {
            cin >> points[i].first >> points[i].second;
        }

        vector<vector<int>> ans = solve_case(points);

        // 输出矩阵 S
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (j > 0) {
                    cout << ' ';
                }
                cout << ans[i][j];
            }
            cout << '\n';
        }
    }

    return 0;
}
```
