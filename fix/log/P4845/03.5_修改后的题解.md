## 解题思路

1. **问题转化**  
   对于成员 $u$，若存在两个后代 $x$ 和 $y$（允许 $x = y$），满足 $LCA(x, y) = u$ 且 $dist(x, u) = dist(y, u) = d$，则 $(x, y)$ 构成的路径长度为 $2d + 1$。  
   要使路径上的纹章字符串为回文串，等价于从 $u$ 出发向 $x$ 方向走的 $d$ 个字符与向 $y$ 方向走的 $d$ 个字符完全相同。  
   因此，问题转化为：对每个成员 $u$，在其不同儿子的子树中各选取一条长度为 $d$ 的从儿子出发的向下路径，若两条路径的纹章字符串相等，则可构成以 $u$ 为中心的镜渊路径，路径长度为 $2d + 1$。  
   特殊情形 $x = y = u$ 时，路径仅包含自身，长度为 $1$，总是合法，故每个成员的镜渊半径至少为 $1$。

2. **树上字符串哈希**  
   采用双模数哈希，预处理从根到每个节点 $v$ 的哈希值 $H_v$ 以及深度 $dep_v$。  
   利用哈希前缀性质，可以在 $O(1)$ 时间内求出从 $u$ 的下一层（即 $u$ 的某个儿子）到后代 $x$ 这段向下路径的字符串哈希值。

3. **启发式合并（DSU on tree）**  
   - 预处理重儿子：通过一次 DFS 求出每个节点子树大小，确定重儿子。  
   - 对每个节点 $u$，先后处理其所有儿子子树。
     - 先加入重儿子子树中的所有节点：对每个节点 $x$ 计算相对深度 $d = dep_x - dep_u$ 及对应的哈希值，存入哈希表 `map`（键为 $(d, hash)$，值为所属儿子编号）。
     - 再依次处理轻儿子子树：
       1. **查询**：对于该儿子子树中的每个节点 $x$，若 `map` 中已存在相同的键，且所属儿子不同，则可以将答案 $ans_u$ 更新为 $\max(ans_u, 2d + 1)$。
       2. **加入**：将该儿子子树的所有节点以相同方式插入 `map`。
   - 由于重儿子先加入，每个轻儿子子树中的节点在其祖先处总共只会被枚举 $O(\log n)$ 次，保证了时间复杂度。

4. **答案收集**  
   最终 $ans_u$ 即为成员 $u$ 的镜渊半径。特别地，若未找到任何长度大于 $1$ 的合法路径，$ans_u$ 保持初始值 $1$。

## 复杂度分析

- **时间复杂度**：预处理哈希、倍增数组需要 $O(n \log n)$，启发式合并过程中每个节点被枚举 $O(\log n)$ 次，总复杂度为 $O(n \log n)$。  
- **空间复杂度**：存储树、深度、哈希、倍增数组以及辅助哈希表，需要 $O(n \log n)$ 的空间。

```python
import sys
sys.setrecursionlimit(300000)

MOD1 = 1000000007
MOD2 = 1000000009
BASE = 911382323

def solve(n, s, edges):
    g = [[] for _ in range(n + 1)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    LOG = 18
    fa = [[0] * (n + 1) for _ in range(LOG)]
    dep = [0] * (n + 1)
    size = [1] * (n + 1)
    son = [0] * (n + 1)

    h1 = [0] * (n + 1)
    h2 = [0] * (n + 1)
    pow1 = [1] * (n + 2)
    pow2 = [1] * (n + 2)

    for i in range(1, n + 2):
        pow1[i] = pow1[i - 1] * BASE % MOD1
        pow2[i] = pow2[i - 1] * BASE % MOD2

    order = [1]
    fa[0][1] = 0

    # 建树并求父亲、深度和根到点哈希
    for u in order:
        for v in g[u]:
            if v == fa[0][u]:
                continue
            fa[0][v] = u
            dep[v] = dep[u] + 1
            c = ord(s[v - 1]) - ord('a') + 1
            h1[v] = (h1[u] * BASE + c) % MOD1
            h2[v] = (h2[u] * BASE + c) % MOD2
            order.append(v)

    # 倍增父亲
    for k in range(1, LOG):
        for i in range(1, n + 1):
            fa[k][i] = fa[k - 1][fa[k - 1][i]]

    # 求子树大小和重儿子
    for u in reversed(order):
        size[u] = 1
        for v in g[u]:
            if v == fa[0][u]:
                continue
            size[u] += size[v]
            if size[v] > size[son[u]]:
                son[u] = v

    dfn = [0] * (n + 1)
    rev = [0] * (n + 1)
    timer = 0

    # DFS 序，方便枚举子树
    def dfs_dfn(u):
        nonlocal timer
        timer += 1
        dfn[u] = timer
        rev[timer] = u
        for v in g[u]:
            if v != fa[0][u]:
                dfs_dfn(v)

    dfs_dfn(1)

    ans = [1] * (n + 1)

    # 求 ancestor 到 x 的下一层后到 x 的字符串哈希
    def get_hash(ancestor, x):
        length = dep[x] - dep[ancestor]
        a = ancestor
        v1 = (h1[x] - h1[a] * pow1[length]) % MOD1
        v2 = (h2[x] - h2[a] * pow2[length]) % MOD2
        return v1, v2

    # 枚举子树中的所有节点
    def nodes_in_subtree(u):
        for i in range(dfn[u], dfn[u] + size[u]):
            yield rev[i]

    # 递归处理每个节点
    def dfs(u):
        for v in g[u]:
            if v != fa[0][u]:
                dfs(v)

        mp = {}

        # 先加入重儿子子树，减少总枚举次数
        if son[u]:
            for x in nodes_in_subtree(son[u]):
                d = dep[x] - dep[u]
                key = (d, get_hash(u, x))
                mp[key] = son[u]

        # 处理其他儿子子树
        for v in g[u]:
            if v == fa[0][u] or v == son[u]:
                continue

            # 先查询，避免同一儿子子树内部互相匹配
            for x in nodes_in_subtree(v):
                d = dep[x] - dep[u]
                key = (d, get_hash(u, x))
                if key in mp and mp[key] != v:
                    ans[u] = max(ans[u], 2 * d + 1)

            # 再加入当前儿子子树
            for x in nodes_in_subtree(v):
                d = dep[x] - dep[u]
                key = (d, get_hash(u, x))
                mp[key] = v

    dfs(1)
    return ans[1:]


def main():
    input = sys.stdin.readline
    n = int(input())
    s = input().strip()

    edges = []
    for _ in range(n - 1):
        u, v = map(int, input().split())
        edges.append((u, v))

    ans = solve(n, s, edges)
    print(*ans)


if __name__ == "__main__":
    main()
```

```java
import java.util.*;

public class Main {
    static final long MOD1 = 1000000007L;
    static final long MOD2 = 1000000009L;
    static final long BASE = 911382323L;

    static int n, LOG = 18, timer = 0;
    static String str;
    static ArrayList<Integer>[] g;
    static int[][] fa;
    static int[] dep, size, son, dfn, rev, ans;
    static long[] h1, h2, pow1, pow2;

    // 哈希键
    static class Key {
        int depth;
        long a, b;

        Key(int depth, long a, long b) {
            this.depth = depth;
            this.a = a;
            this.b = b;
        }

        public boolean equals(Object o) {
            if (!(o instanceof Key)) return false;
            Key k = (Key) o;
            return depth == k.depth && a == k.a && b == k.b;
        }

        public int hashCode() {
            long res = depth;
            res = res * 1000003 + a;
            res = res * 1000003 + b;
            return (int)(res ^ (res >>> 32));
        }
    }

    // 建树，求父亲、深度、哈希
    static void build() {
        int[] q = new int[n];
        int head = 0, tail = 0;
        q[tail++] = 1;

        while (head < tail) {
            int u = q[head++];
            for (int v : g[u]) {
                if (v == fa[0][u]) continue;
                fa[0][v] = u;
                dep[v] = dep[u] + 1;

                int c = str.charAt(v - 1) - 'a' + 1;
                h1[v] = (h1[u] * BASE + c) % MOD1;
                h2[v] = (h2[u] * BASE + c) % MOD2;

                q[tail++] = v;
            }
        }

        // 倍增父亲
        for (int k = 1; k < LOG; k++) {
            for (int i = 1; i <= n; i++) {
                fa[k][i] = fa[k - 1][fa[k - 1][i]];
            }
        }

        // 求子树大小和重儿子
        for (int i = n - 1; i >= 0; i--) {
            int u = q[i];
            size[u] = 1;
            for (int v : g[u]) {
                if (v == fa[0][u]) continue;
                size[u] += size[v];
                if (son[u] == 0 || size[v] > size[son[u]]) {
                    son[u] = v;
                }
            }
        }
    }

    // 求 DFS 序
    static void dfsDfn(int u) {
        dfn[u] = ++timer;
        rev[timer] = u;

        for (int v : g[u]) {
            if (v != fa[0][u]) {
                dfsDfn(v);
            }
        }
    }

    // 求 ancestor 的下一层到 x 的字符串哈希
    static Key getKey(int ancestor, int x) {
        int len = dep[x] - dep[ancestor];

        long a = (h1[x] - h1[ancestor] * pow1[len]) % MOD1;
        if (a < 0) a += MOD1;

        long b = (h2[x] - h2[ancestor] * pow2[len]) % MOD2;
        if (b < 0) b += MOD2;

        return new Key(len, a, b);
    }

    // 处理每个节点
    static void dfs(int u) {
        for (int v : g[u]) {
            if (v != fa[0][u]) {
                dfs(v);
            }
        }

        HashMap<Key, Integer> map = new HashMap<>();

        // 先加入重儿子子树
        if (son[u] != 0) {
            int v = son[u];
            for (int i = dfn[v]; i < dfn[v] + size[v]; i++) {
                int x = rev[i];
                Key key = getKey(u, x);
                map.put(key, v);
            }
        }

        // 再处理轻儿子子树
        for (int v : g[u]) {
            if (v == fa[0][u] || v == son[u]) continue;

            // 先查询是否能与之前儿子子树匹配
            for (int i = dfn[v]; i < dfn[v] + size[v]; i++) {
                int x = rev[i];
                Key key = getKey(u, x);

                if (map.containsKey(key) && map.get(key) != v) {
                    int d = dep[x] - dep[u];
                    ans[u] = Math.max(ans[u], 2 * d + 1);
                }
            }

            // 再加入当前儿子子树
            for (int i = dfn[v]; i < dfn[v] + size[v]; i++) {
                int x = rev[i];
                Key key = getKey(u, x);
                map.put(key, v);
            }
        }
    }

    static int[] solve() {
        pow1 = new long[n + 2];
        pow2 = new long[n + 2];
        pow1[0] = pow2[0] = 1;

        for (int i = 1; i <= n + 1; i++) {
            pow1[i] = pow1[i - 1] * BASE % MOD1;
            pow2[i] = pow2[i - 1] * BASE % MOD2;
        }

        build();
        dfsDfn(1);

        Arrays.fill(ans, 1);
        dfs(1);

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        n = sc.nextInt();
        str = sc.next();

        g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            g[i] = new ArrayList<>();
        }

        for (int i = 1; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            g[u].add(v);
            g[v].add(u);
        }

        fa = new int[LOG][n + 1];
        dep = new int[n + 1];
        size = new int[n + 1];
        son = new int[n + 1];
        dfn = new int[n + 1];
        rev = new int[n + 1];
        ans = new int[n + 1];
        h1 = new long[n + 1];
        h2 = new long[n + 1];

        int[] res = solve();

        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            if (i > 1) sb.append(' ');
            sb.append(res[i]);
        }
        System.out.println(sb);
    }
}
```

```cpp
using namespace std;

const long long MOD1 = 1000000007LL;
const long long MOD2 = 1000000009LL;
const long long BASE = 911382323LL;

struct Key {
    int d;
    long long a, b;

    bool operator==(const Key& other) const {
        return d == other.d && a == other.a && b == other.b;
    }
};

// 自定义哈希函数
struct KeyHash {
    size_t operator()(const Key& k) const {
        long long res = k.d;
        res = res * 1000003 + k.a;
        res = res * 1000003 + k.b;
        return (size_t)(res ^ (res >> 32));
    }
};

void build() {
    vector<int> order;
    order.push_back(1);

    for (int i = 0; i < (int)order.size(); i++) {
        int u = order[i];

        for (int v : g[u]) {
            if (v == fa[0][u]) continue;

            fa[0][v] = u;
            dep[v] = dep[u] + 1;

            h1[v] = (h1[u] * BASE + c) % MOD1;
            h2[v] = (h2[u] * BASE + c) % MOD2;

            order.push_back(v);
        }
    }

        for (int i = 1; i <= n; i++) {
            fa[k][i] = fa[k - 1][fa[k - 1][i]];

    }

    }
}

void dfs(int u) {
    for (int v : g[u]) {
        if (v != fa[0][u]) {
            dfs(v);
        }
    }

    unordered_map<Key, int, KeyHash> mp;

    // 先加入重儿子子树
    if (son[u]) {

            int x = rev[i];
            Key key = getKey(u, x);
            mp[key] = v;
        }

    // 处理轻儿子子树
    for (int v : g[u]) {

        }
    }
}
```
