## 解题思路

本题要求判断一个字符串在“镜像翻转”后是否与原串相同。  
镜像翻转定义为：先将字符串左右反转，再将每个字母替换为它的镜像字母（仅当字母有镜像对应时才能替换）。  
根据题意，镜像配对规则如下：

- `n` ↔ `u`，`w` ↔ `m`，`p` ↔ `d`，`q` ↔ `b`
- `x` 和 `o` 自身的镜像就是自己
- 其余小写字母没有镜像

因此，一个字符串是“镜中词”当且仅当从左到右与从右到左的每一对字符满足镜像关系。具体地，对于字符串 `s`，我们从两端向中间进行双指针检查：

1. 预处理一个映射表 `mirror`，存放每个字母的镜像字母。若字母没有镜像，则不放入表中。
2. 令 `i = 0`，`j = len(s) - 1`，当 `i <= j` 时循环：
   - 若 `s[i] == s[j]`：该字符必须能自身镜像，即只能是 `'x'` 或 `'o'`。否则字符串不满足条件。
   - 若 `s[i] != s[j]`：必须满足 `mirror[s[i]] == s[j]`（即这两个字符恰好互为镜像）。注意，如果 `s[i]` 不在映射表中，或它的镜像不是 `s[j]`，则直接返回 `"no"`。
   - 每轮 `i` 加 1，`j` 减 1。
3. 所有位置检查通过后，返回 `"yes"`。

该方法可以一次性处理所有询问，且对每个字符串只进行一趟扫描。

## 复杂度分析

- 时间复杂度：$O\left(\sum |s|\right)$，其中 $\sum |s|$ 是所有字符串的总长度。双指针遍历每个字符串一次，每个字符被访问常数次。  
- 空间复杂度：$O(1)$。镜像映射表仅存放常数对字符关系，与输入规模无关。

## 代码实现

### Python

```python
import sys

def solve():
    T = int(sys.stdin.readline())
    # 镜像映射表，仅包含题面给出的镜像对
    mirror = {
        'n': 'u', 'u': 'n',
        'w': 'm', 'm': 'w',
        'p': 'd', 'd': 'p',
        'q': 'b', 'b': 'q',
        'x': 'x', 'o': 'o'
    }
    out = []
    for _ in range(T):
        s = sys.stdin.readline().strip()
        n = len(s)
        i, j = 0, n - 1
        ok = True
        while i <= j:
            if s[i] == s[j]:
                # 相等字符必须自身镜像 'x' 或 'o'
                if s[i] != 'x' and s[i] != 'o':
                    ok = False
                    break
            else:
                # 不等字符必须是一对镜像字母
                if mirror.get(s[i], '') != s[j]:
                    ok = False
                    break
            i += 1
            j -= 1
        out.append("yes" if ok else "no")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        // 建立镜像映射表
        HashMap<Character, Character> mirror = new HashMap<>();
        mirror.put('n', 'u'); mirror.put('u', 'n');
        mirror.put('w', 'm'); mirror.put('m', 'w');
        mirror.put('p', 'd'); mirror.put('d', 'p');
        mirror.put('q', 'b'); mirror.put('b', 'q');
        mirror.put('x', 'x'); mirror.put('o', 'o');

        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            String s = br.readLine();
            int i = 0, j = s.length() - 1;
            boolean ok = true;
            while (i <= j) {
                char left = s.charAt(i), right = s.charAt(j);
                if (left == right) {
                    if (left != 'x' && left != 'o') {
                        ok = false;
                        break;
                    }
                } else {
                    // 检查 left 是否有镜像且镜像等于 right
                    Character mir = mirror.get(left);
                    if (mir == null || mir != right) {
                        ok = false;
                        break;
                    }
                }
                i++;
                j--;
            }
            sb.append(ok ? "yes\n" : "no\n");
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <unordered_map>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    // 镜像映射表
    unordered_map<char, char> mirror = {
        {'n', 'u'}, {'u', 'n'},
        {'w', 'm'}, {'m', 'w'},
        {'p', 'd'}, {'d', 'p'},
        {'q', 'b'}, {'b', 'q'},
        {'x', 'x'}, {'o', 'o'}
    };

    int T;
    cin >> T;
    while (T--) {
        string s;
        cin >> s;
        int i = 0, j = s.size() - 1;
        bool ok = true;
        while (i <= j) {
            if (s[i] == s[j]) {
                // 自身镜像只能是 'x' 或 'o'
                if (s[i] != 'x' && s[i] != 'o') {
                    ok = false;
                    break;
                }
            } else {
                // 必须是一对镜像字母
                auto it = mirror.find(s[i]);
                if (it == mirror.end() || it->second != s[j]) {
                    ok = false;
                    break;
                }
            }
            i++;
            j--;
        }
        cout << (ok ? "yes" : "no") << '\n';
    }
    return 0;
}
```
