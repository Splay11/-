## 解题思路

1. 把每个时刻 `hh:mm:ss` 拆成时、分、秒三个整数，换算成当天从零点起经过的秒数：
   $$t = hh\times 3600 + mm\times 60 + ss.$$
2. 对相邻两个时刻 $t_1$、$t_2$，要求最小正向间隔。若 $t_2 < t_1$，说明后一时刻在钟面上更早，已经跨过一次零点，令 $t_2 \leftarrow t_2 + 24\times 3600$；若 $t_2 = t_1$，间隔为 `0`。
3. 间隔秒数为 $\Delta = t_2 - t_1$。天线每 `60` 秒转一圈，因此圈数为 $\Delta / 60$。按题目要求保留两位小数输出。
4. 一共有 $n-1$ 对相邻时刻，依次计算即可。比较两个时刻的先后时，按小时、分钟、秒钟依次比较，与直接比较秒数等价。

## 复杂度分析

设记录个数为 $n$。

- 时间复杂度：$O(n)$。
- 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
def interval_seconds(time_1, time_2):
    h1, m1, s1 = map(int, time_1.split(':'))
    h2, m2, s2 = map(int, time_2.split(':'))
    # 后一时刻严格更早时，视为已跨过零点
    if h1 > h2 or (h1 == h2 and m1 > m2) or (h1 == h2 and m1 == m2 and s1 > s2):
        h2 += 24
    return (h2 - h1) * 3600 + (m2 - m1) * 60 + (s2 - s1)


n = int(input())
times = input().split()
res = []
for i in range(n - 1):
    circles = interval_seconds(times[i], times[i + 1]) / 60
    res.append(f'{circles:.2f}')
print(' '.join(res))
```

### Java

```java
import java.util.Scanner;

public class Main {
    static int intervalSeconds(String time1, String time2) {
        String[] p1 = time1.split(":");
        String[] p2 = time2.split(":");
        int h1 = Integer.parseInt(p1[0]);
        int m1 = Integer.parseInt(p1[1]);
        int s1 = Integer.parseInt(p1[2]);
        int h2 = Integer.parseInt(p2[0]);
        int m2 = Integer.parseInt(p2[1]);
        int s2 = Integer.parseInt(p2[2]);

        if (h1 > h2 || (h1 == h2 && m1 > m2) || (h1 == h2 && m1 == m2 && s1 > s2)) {
            h2 += 24;
        }
        return (h2 - h1) * 3600 + (m2 - m1) * 60 + (s2 - s1);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] times = new String[n];
        for (int i = 0; i < n; i++) {
            times[i] = sc.next();
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n - 1; i++) {
            double circles = intervalSeconds(times[i], times[i + 1]) / 60.0;
            if (i > 0) {
                sb.append(' ');
            }
            sb.append(String.format("%.2f", circles));
        }
        System.out.println(sb.toString());
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <sstream>
#include <iomanip>

using namespace std;

vector<string> split(const string& s, char delim) {
    vector<string> res;
    stringstream ss(s);
    string t;
    while (getline(ss, t, delim)) {
        res.push_back(t);
    }
    return res;
}

int intervalSeconds(const string& time1, const string& time2) {
    vector<string> p1 = split(time1, ':');
    vector<string> p2 = split(time2, ':');
    int h1 = stoi(p1[0]), m1 = stoi(p1[1]), s1 = stoi(p1[2]);
    int h2 = stoi(p2[0]), m2 = stoi(p2[1]), s2 = stoi(p2[2]);

    if (h1 > h2 || (h1 == h2 && m1 > m2) || (h1 == h2 && m1 == m2 && s1 > s2)) {
        h2 += 24;
    }
    return (h2 - h1) * 3600 + (m2 - m1) * 60 + (s2 - s1);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<string> times(n);
    for (int i = 0; i < n; i++) {
        cin >> times[i];
    }

    cout << fixed << setprecision(2);
    for (int i = 0; i < n - 1; i++) {
        double circles = intervalSeconds(times[i], times[i + 1]) / 60.0;
        if (i) {
            cout << ' ';
        }
        cout << circles;
    }
    cout << '\n';
    return 0;
}
```
