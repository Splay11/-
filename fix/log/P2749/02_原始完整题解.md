## 题解

### 题目描述

给定 $n$ 个区间，每个区间表示为 $[l_i, r_i]$，计算有多少对不同的区间满足这两个区间有交集。

### 思路分析

这道题要求计算所有区间对中有交集的数目，我们可以将其转化为总对数减去不相交的对数。具体做法是：首先将所有区间按左端点升序排序，然后利用离散化和树状数组高效统计每个区间之前有多少区间的右端点小于当前区间的左端点（即不相交的情况），最后用组合数公式计算总对数并减去不相交对数得到答案。

1. **问题转化**：计算所有区间对中满足相交条件的对数，等价于总对数减去互不相交的对数。
2. **互不相交的条件**：对于两个区间 $[l_u, r_u]$ 和 $[l_v, r_v]$，若 $r_u < l_v$ 或 $r_v < l_u$，则它们不相交。
3. **排序处理**：按左端点升序排序后，只需统计每个区间之前有多少区间的右端点小于当前区间的左端点。
4. **离散化与树状数组**：通过离散化处理右端点，使用树状数组高效维护和查询前缀和，统计符合要求的区间数目。


#### C++
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long ll;
typedef pair<int, int> pii;

class FenwickTree {
private:
    vector<int> tree;
public:
    FenwickTree(int size) : tree(size + 1, 0) {}

    void update(int idx, int delta) {
        idx += 1; // 树状数组从1开始索引
        while (idx < tree.size()) {
            tree[idx] += delta;
            idx += idx & -idx;
        }
    }

    int query(int idx) {
        idx += 1;
        int res = 0;
        while (idx > 0) {
            res += tree[idx];
            idx -= idx & -idx;
        }
        return res;
    }
};
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<pii> intervals(n);
    vector<int> r_values;

    for (int i = 0; i < n; ++i) {
        cin >> intervals[i].first >> intervals[i].second;
        r_values.push_back(intervals[i].second);
    }

    // 按左端点升序排序
    sort(intervals.begin(), intervals.end());

    // 离散化右端点
    sort(r_values.begin(), r_values.end());
    r_values.erase(unique(r_values.begin(), r_values.end()), r_values.end());

    FenwickTree ft(r_values.size());
    ll non_overlap = 0;

    for (const auto& interval : intervals) {
        int l = interval.first;
        int r = interval.second;

        // 计算x = l - 1
        int x = l - 1;
        int pos = upper_bound(r_values.begin(), r_values.end(), x) - r_values.begin();
        non_overlap += ft.query(pos - 1);

        // 找到r对应的离散化位置
        int idx = lower_bound(r_values.begin(), r_values.end(), r) - r_values.begin();
        ft.update(idx, 1);
    }

    ll total = (ll)n * (n - 1) / 2;
    ll ans = total - non_overlap;
    cout << ans << "\n";

    return 0;
}
```

#### Python
```python
import bisect

n = int(input())
intervals = []
r_values = []

for _ in range(n):
    l, r = map(int, input().split())
    intervals.append((l, r))
    r_values.append(r)

# 按左端点排序
intervals.sort()

# 离散化右端点
r_values_sorted = sorted(list(set(r_values)))
size = len(r_values_sorted)

# 树状数组
class FenwickTree:
    def __init__(self, size):
        self.n = size
        self.tree = [0] * (self.n + 1)
    
    def update(self, idx, delta):
        idx += 1  # 转为1-based索引
        while idx <= self.n:
            self.tree[idx] += delta
            idx += idx & -idx
    
    def query(self, idx):
        idx += 1  # 转为1-based索引
        res = 0
        while idx > 0:
            res += self.tree[idx]
            idx -= idx & -idx
        return res

ft = FenwickTree(size)
non_overlap = 0

for l, r in intervals:
    x = l - 1
    pos = bisect.bisect_right(r_values_sorted, x)
    non_overlap += ft.query(pos - 1)
    
    # 插入当前r
    idx = bisect.bisect_left(r_values_sorted, r)
    ft.update(idx, 1)

total = n * (n - 1) // 2
print(total - non_overlap)
```

#### Java
```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<int[]> intervals = new ArrayList<>();
        List<Integer> rValues = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            intervals.add(new int[]{l, r});
            rValues.add(r);
        }

        // 按左端点排序
        intervals.sort((a, b) -> a[0] != b[0] ? Integer.compare(a[0], b[0]) : Integer.compare(a[1], b[1]));

        // 离散化右端点
        Set<Integer> uniqueR = new TreeSet<>(rValues);
        List<Integer> sortedR = new ArrayList<>(uniqueR);
        int size = sortedR.size();

        FenwickTree ft = new FenwickTree(size);
        long nonOverlap = 0;

        for (int[] interval : intervals) {
            int l = interval[0];
            int r = interval[1];

            int x = l - 1;
            int pos = bisectRight(sortedR, x);
            nonOverlap += ft.query(pos - 1);

            int idx = Collections.binarySearch(sortedR, r);
            if (idx < 0) idx = -(idx + 1);
            ft.update(idx, 1);
        }

        long total = (long) n * (n - 1) / 2;
        System.out.println(total - nonOverlap);
    }

    private static int bisectRight(List<Integer> list, int x) {
        int l = 0, r = list.size();
        while (l < r) {
            int m = (l + r) >>> 1;
            if (list.get(m) > x) r = m;
            else l = m + 1;
        }
        return l;
    }

    static class FenwickTree {
        int[] tree;
        int n;

        public FenwickTree(int size) {
            this.n = size;
            tree = new int[n + 1];
        }

        public void update(int idx, int delta) {
            idx++; // 转为1-based索引
            while (idx <= n) {
                tree[idx] += delta;
                idx += idx & -idx;
            }
        }

        public int query(int idx) {
            idx++; // 转为1-based索引
            int res = 0;
            while (idx > 0) {
                res += tree[idx];
                idx -= idx & -idx;
            }
            return res;
        }
    }
}
```
