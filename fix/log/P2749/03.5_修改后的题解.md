## 解题思路

$n$ 段泊位占用的总对数为 $\dfrac{n(n-1)}{2}$。冲突对数等于总对数减去互不相交的对数。

两段 $[l_u,r_u]$、$[l_v,r_v]$ 不相交当且仅当 $r_u < l_v$ 或 $r_v < l_u$。将所有占用按左端点升序排序后，对当前段只需统计**已经处理过、且右端点小于当前左端点**的段数，这些就是与当前段不相交的对。

右端点值域达到 `1000000000`，先离散化，再用树状数组维护已出现右端点的个数，即可在 $O(\log n)$ 内完成前缀查询与插入。

## 复杂度分析

时间复杂度 $O(n \log n)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import bisect

n = int(input())
intervals = []
r_values = []

for _ in range(n):
    l, r = map(int, input().split())
    intervals.append((l, r))
    r_values.append(r)

intervals.sort()
r_values_sorted = sorted(list(set(r_values)))
size = len(r_values_sorted)

class FenwickTree:
    def __init__(self, size):
        self.n = size
        self.tree = [0] * (self.n + 1)

    def update(self, idx, delta):
        idx += 1
        while idx <= self.n:
            self.tree[idx] += delta
            idx += idx & -idx

    def query(self, idx):
        idx += 1
        res = 0
        while idx > 0:
            res += self.tree[idx]
            idx -= idx & -idx
        return res

ft = FenwickTree(size)
non_overlap = 0

for l, r in intervals:
    x = l - 1
    pos = bisect.bisect_right(r_values_sorted, x)
    non_overlap += ft.query(pos - 1)
    idx = bisect.bisect_left(r_values_sorted, r)
    ft.update(idx, 1)

total = n * (n - 1) // 2
print(total - non_overlap)
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<int[]> intervals = new ArrayList<>();
        List<Integer> rValues = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            intervals.add(new int[]{l, r});
            rValues.add(r);
        }

        intervals.sort((a, b) -> a[0] != b[0] ? Integer.compare(a[0], b[0]) : Integer.compare(a[1], b[1]));

        Set<Integer> uniqueR = new TreeSet<>(rValues);
        List<Integer> sortedR = new ArrayList<>(uniqueR);
        int size = sortedR.size();

        FenwickTree ft = new FenwickTree(size);
        long nonOverlap = 0;

        for (int[] interval : intervals) {
            int l = interval[0];
            int r = interval[1];
            int x = l - 1;
            int pos = bisectRight(sortedR, x);
            nonOverlap += ft.query(pos - 1);
            int idx = Collections.binarySearch(sortedR, r);
            if (idx < 0) idx = -(idx + 1);
            ft.update(idx, 1);
        }

        long total = (long) n * (n - 1) / 2;
        System.out.println(total - nonOverlap);
    }

    private static int bisectRight(List<Integer> list, int x) {
        int l = 0, r = list.size();
        while (l < r) {
            int m = (l + r) >>> 1;
            if (list.get(m) > x) r = m;
            else l = m + 1;
        }
        return l;
    }

    static class FenwickTree {
        int[] tree;
        int n;

        public FenwickTree(int size) {
            this.n = size;
            tree = new int[n + 1];
        }

        public void update(int idx, int delta) {
            idx++;
            while (idx <= n) {
                tree[idx] += delta;
                idx += idx & -idx;
            }
        }

        public int query(int idx) {
            idx++;
            int res = 0;
            while (idx > 0) {
                res += tree[idx];
                idx -= idx & -idx;
            }
            return res;
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long ll;
typedef pair<int, int> pii;

class FenwickTree {
private:
    vector<int> tree;
public:
    FenwickTree(int size) : tree(size + 1, 0) {}

    void update(int idx, int delta) {
        idx += 1;
        while (idx < (int)tree.size()) {
            tree[idx] += delta;
            idx += idx & -idx;
        }
    }

    int query(int idx) {
        idx += 1;
        int res = 0;
        while (idx > 0) {
            res += tree[idx];
            idx -= idx & -idx;
        }
        return res;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<pii> intervals(n);
    vector<int> r_values;

    for (int i = 0; i < n; ++i) {
        cin >> intervals[i].first >> intervals[i].second;
        r_values.push_back(intervals[i].second);
    }

    sort(intervals.begin(), intervals.end());
    sort(r_values.begin(), r_values.end());
    r_values.erase(unique(r_values.begin(), r_values.end()), r_values.end());

    FenwickTree ft(r_values.size());
    ll non_overlap = 0;

    for (const auto& interval : intervals) {
        int l = interval.first;
        int r = interval.second;
        int x = l - 1;
        int pos = upper_bound(r_values.begin(), r_values.end(), x) - r_values.begin();
        non_overlap += ft.query(pos - 1);
        int idx = lower_bound(r_values.begin(), r_values.end(), r) - r_values.begin();
        ft.update(idx, 1);
    }

    ll total = (ll)n * (n - 1) / 2;
    cout << total - non_overlap << "\n";
    return 0;
}
```
