## 解题思路

本题的核心是找出网格中所有由同类昆虫构成的连通区域（即“栖息群落”），并在划分群落的同时统计每个群落边界上与多少种不同类别的昆虫相邻。具体步骤如下：

1. **连通域划分与编号**  
   利用广度优先搜索（BFS）或深度优先搜索（DFS）遍历整个网格，将上下左右相邻且昆虫类别相同的格子归为同一个栖息群落。使用一个二维数组 `comp` 记录每个格子所属的群落编号（从 0 开始递增），未访问的格子初始标记为 -1。

2. **统计邻接的异类昆虫种类**  
   在 BFS 搜索某个群落的过程中，对于每个格子检查其四个方向的邻居：
   - 若邻居与当前群落昆虫类别相同且未被访问，则将其加入队列并赋予相同的群落编号；
   - 若邻居昆虫类别不同，则将该邻居的类别（小写字母）记录到当前群落对应的“相邻异类集合”中。
   搜索完整个群落后，该集合的大小即为该群落相邻的其他昆虫种类数。将该值存入数组 `answer`（按群落编号索引）。

3. **输出答案**  
   对于网格中的每个格子，查表得到其所在群落的编号 `comp[i][j]`，输出 `answer[comp[i][j]]`。每行 `m` 个整数，空格分隔，行末无多余空格。

## 复杂度分析

- **时间复杂度**：$O(n \times m)$  
  每个格子恰好被访问一次（BFS 入队一次、出队一次），每次检查最多四个邻居，常数时间。总操作次数与网格大小成正比。

- **空间复杂度**：$O(n \times m)$  
  需要存储网格 `grid`（输入）、群落编号数组 `comp`（$n \times m$ 的整数矩阵）、群落答案数组（大小最多 $n \times m$，即最坏每个格子单独成落）以及 BFS 队列（最坏容纳所有格子）。额外使用的集合空间最多为不同昆虫种类的数量（不超过 26 种小写字母），可视为常数。因此整体空间开销与网格规模线性相关。

## 代码实现

### Python

```python
from collections import deque

def main():
    import sys
    input = sys.stdin.readline

    n, m = map(int, input().split())
    grid = [input().strip() for _ in range(n)]

    # 记录每个格子所属的群落编号，-1 表示未访问
    comp = [[-1] * m for _ in range(n)]
    # answer[idx] 表示编号为 idx 的群落相邻的其他昆虫种类数
    answer = []

    directions = [(1, 0), (-1, 0), (0, 1), (0, -1)]
    comp_id = 0

    for i in range(n):
        for j in range(m):
            if comp[i][j] == -1:
                current_species = grid[i][j]  # 当前群落的昆虫类别
                neighbor_set = set()          # 记录相邻的不同类别
                q = deque()
                q.append((i, j))
                comp[i][j] = comp_id

                while q:
                    x, y = q.popleft()
                    for dx, dy in directions:
                        nx, ny = x + dx, y + dy
                        if nx < 0 or nx >= n or ny < 0 or ny >= m:
                            continue
                        if grid[nx][ny] == current_species and comp[nx][ny] == -1:
                            comp[nx][ny] = comp_id
                            q.append((nx, ny))
                        elif grid[nx][ny] != current_species:
                            neighbor_set.add(grid[nx][ny])

                answer.append(len(neighbor_set))
                comp_id += 1

    # 输出结果
    out_lines = []
    for i in range(n):
        row = [str(answer[comp[i][j]]) for j in range(m)]
        out_lines.append(" ".join(row))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = br.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);

        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = br.readLine().toCharArray();
        }

        int[][] comp = new int[n][m];
        for (int[] row : comp) {
            Arrays.fill(row, -1);
        }

        List<Integer> answer = new ArrayList<>();
        int[] dx = {1, -1, 0, 0};
        int[] dy = {0, 0, 1, -1};

        int compId = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (comp[i][j] == -1) {
                    char currentSpecies = grid[i][j];
                    Set<Character> neighborSet = new HashSet<>();
                    Queue<int[]> queue = new LinkedList<>();
                    queue.add(new int[]{i, j});
                    comp[i][j] = compId;

                    while (!queue.isEmpty()) {
                        int[] cur = queue.poll();
                        int x = cur[0], y = cur[1];
                        for (int d = 0; d < 4; d++) {
                            int nx = x + dx[d];
                            int ny = y + dy[d];
                            if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                            if (grid[nx][ny] == currentSpecies && comp[nx][ny] == -1) {
                                comp[nx][ny] = compId;
                                queue.add(new int[]{nx, ny});
                            } else if (grid[nx][ny] != currentSpecies) {
                                neighborSet.add(grid[nx][ny]);
                            }
                        }
                    }
                    answer.add(neighborSet.size());
                    compId++;
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append(answer.get(comp[i][0]));
            for (int j = 1; j < m; j++) {
                sb.append(" ").append(answer.get(comp[i][j]));
            }
            sb.append("\n");
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <set>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;
    vector<string> grid(n);
    for (int i = 0; i < n; ++i) {
        cin >> grid[i];
    }

    // comp[i][j] 表示格子 (i,j) 所属的群落编号，-1 表示未访问
    vector<vector<int>> comp(n, vector<int>(m, -1));
    vector<int> answer;            // answer[idx] = 群落 idx 相邻的其他昆虫种类数
    int compId = 0;

    const int dx[4] = {1, -1, 0, 0};
    const int dy[4] = {0, 0, 1, -1};

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (comp[i][j] == -1) {
                char currentSpecies = grid[i][j];
                set<char> neighborSet;          // 记录相邻的不同昆虫类别
                queue<pair<int, int>> q;
                q.push({i, j});
                comp[i][j] = compId;

                while (!q.empty()) {
                    auto [x, y] = q.front();
                    q.pop();
                    for (int d = 0; d < 4; ++d) {
                        int nx = x + dx[d];
                        int ny = y + dy[d];
                        if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                        if (grid[nx][ny] == currentSpecies && comp[nx][ny] == -1) {
                            comp[nx][ny] = compId;
                            q.push({nx, ny});
                        } else if (grid[nx][ny] != currentSpecies) {
                            neighborSet.insert(grid[nx][ny]);
                        }
                    }
                }
                answer.push_back(neighborSet.size());
                ++compId;
            }
        }
    }

    // 输出结果
    for (int i = 0; i < n; ++i) {
        cout << answer[comp[i][0]];
        for (int j = 1; j < m; ++j) {
            cout << " " << answer[comp[i][j]];
        }
        cout << "\n";
    }

    return 0;
}
```
