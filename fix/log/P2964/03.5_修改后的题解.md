## 解题思路

对于一个长度为 $n$ 的排列 $b$，定义其相邻元素之差的绝对值序列 $d$，其中 $d_i = |b_{i+1} - b_i|$（$1 \le i \le n-1$）。  
要求 $d$ 构成公差为 $c$ 的等差数列，即对于所有 $i$，$d_{i+1} - d_i = c$。

首先分析 $d$ 序列的合法取值范围：

- 由于 $b$ 是 $1$ 到 $n$ 的排列，任意两个不同数的差的绝对值至少为 $1$，至多为 $n-1$，因此 $1 \le d_i \le n-1$。
- 设等差数列的首项为 $A$，则有 $d_i = A + (i-1) \cdot c$，$i = 1,2,\dots,n-1$。
- 要使所有 $d_i$ 都在 $[1, n-1]$ 内，必须对 $c$ 加以限制。可以证明，只有当 $c \in \{-1, 0, 1\}$ 时，才存在满足条件的 $A$ 和序列。

根据 $c$ 的三种可能取值分别构造：

1. **$c = 0$**：所有 $d_i$ 相等。最简单的方式是令 $b = [1, 2, \dots, n]$，此时 $d_i = 1$（常数），公差为 $0$。
2. **$c = -1$**：需要 $d$ 为递减序列：$n-1, n-2, \dots, 1$。可通过交替从两端取数构造：  
   $b = [1, n, 2, n-1, 3, n-2, \dots]$。  
   此时 $d_i = n-i$，首项为 $n-1$，公差为 $-1$。
3. **$c = 1$**：需要 $d$ 为递增序列：$1, 2, \dots, n-1$。将 $c = -1$ 构造出的序列逆序即可得到满足条件的排列。
4. **其他 $c$ 值**：无解，输出 `NO`。

## 复杂度分析

- 时间复杂度：以上构造均可在线性或接近线性的时间内完成，具体为 $O(n)$。
- 空间复杂度：需要存储长度为 $n$ 的排列或进行输出，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    c = int(data[1])

    if c == 0:
        # 公差为0，直接输出升序排列
        print("YES")
        print(*range(1, n + 1))
    elif c == -1:
        # 公差为-1，交替从两端取数
        print("YES")
        res = []
        L, R = 1, n
        for i in range(1, n + 1):
            if i % 2 == 1:
                res.append(L)
                L += 1
            else:
                res.append(R)
                R -= 1
        print(*res)
    elif c == 1:
        # 公差为1，先按c=-1构造，再逆序
        res = []
        L, R = 1, n
        for i in range(1, n + 1):
            if i % 2 == 1:
                res.append(L)
                L += 1
            else:
                res.append(R)
                R -= 1
        res.reverse()
        print("YES")
        print(*res)
    else:
        # 其他公差无解
        print("NO")

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int c = Integer.parseInt(st.nextToken());
        br.close();

        if (c == 0) {
            // 公差为0
            System.out.println("YES");
            StringBuilder sb = new StringBuilder();
            for (int i = 1; i <= n; i++) {
                sb.append(i);
                if (i != n) sb.append(" ");
            }
            System.out.println(sb);
        } else if (c == -1) {
            // 公差为-1，交替取两端
            System.out.println("YES");
            StringBuilder sb = new StringBuilder();
            int L = 1, R = n;
            for (int i = 1; i <= n; i++) {
                if ((i & 1) == 1) {
                    sb.append(L++);
                } else {
                    sb.append(R--);
                }
                if (i != n) sb.append(" ");
            }
            System.out.println(sb);
        } else if (c == 1) {
            // 公差为1，先按c=-1构造再逆序
            List<Integer> p = new ArrayList<>();
            int L = 1, R = n;
            for (int i = 1; i <= n; i++) {
                if ((i & 1) == 1) {
                    p.add(L++);
                } else {
                    p.add(R--);
                }
            }
            Collections.reverse(p);
            System.out.println("YES");
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < n; i++) {
                sb.append(p.get(i));
                if (i != n - 1) sb.append(" ");
            }
            System.out.println(sb);
        } else {
            // 其他公差无解
            System.out.println("NO");
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, c;
    cin >> n >> c;

    if (c == 0) {
        // 公差为0，输出升序排列
        cout << "YES\n";
        for (int i = 1; i <= n; i++) {
            cout << i << (i == n ? '\n' : ' ');
        }
    }
    else if (c == -1) {
        // 公差为-1，交替从两端取数
        cout << "YES\n";
        int L = 1, R = n;
        for (int i = 1; i <= n; i++) {
            if (i & 1) {
                cout << L++;
            } else {
                cout << R--;
            }
            cout << (i == n ? '\n' : ' ');
        }
    }
    else if (c == 1) {
        // 公差为1，先按c=-1构造，再逆序
        vector<int> p;
        int L = 1, R = n;
        for (int i = 1; i <= n; i++) {
            if (i & 1) p.push_back(L++);
            else       p.push_back(R--);
        }
        reverse(p.begin(), p.end());
        cout << "YES\n";
        for (int i = 0; i < n; i++) {
            cout << p[i] << (i + 1 == n ? '\n' : ' ');
        }
    }
    else {
        // 其他公差无解
        cout << "NO\n";
    }

    return 0;
}
```
