## 解题思路

第 $i$ 台仪器的一次有效实验，占用时长的下界为 $c_i^L = \min(a_i, x)$，上界为 $c_i^R = a_i$。需要在 $l_i \le v_i \le r_i$ 且 $\sum v_i = m$ 的前提下，分别最小化与最大化总占用时长。

先给每台仪器分配下界次数 $l_i$，剩余可分配次数为 $\textit{left} = m - \sum l_i$，第 $i$ 台还能再增加的次数为 $\textit{cap}_i = r_i - l_i$。

- 求最小总时长：按 $c_i^L$ 从小到大，把剩余次数优先分给单位占用更小的仪器。
- 求最大总时长：按 $c_i^R$ 从大到小，把剩余次数优先分给额定时长更大的仪器。

每次分配量为 $\min(\textit{left}, \textit{cap}_i)$，直到剩余次数用完。由交换论证可知，把一次额外实验从较差系数挪到更优系数只会让目标更优，因此该贪心正确。累加过程中数值可能很大，需使用 64 位整数。

## 复杂度分析

排序占主导，随后线性分配一次。

- 时间复杂度：$O(n \log n)$。
- 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
from typing import List


def calc_min_total_time(n: int, x: int, m: int,
                        a: List[int], q: List[int], r: List[int]) -> int:
    """最小总占用时长：单位系数为 min(a_i, x)"""
    coeff = [min(a[i], x) for i in range(n)]
    base = 0
    caps = []
    for i in range(n):
        base += q[i] * coeff[i]
        caps.append((coeff[i], r[i] - q[i]))
    left = m - sum(q)
    caps.sort(key=lambda t: t[0])
    res = base
    for c, cap in caps:
        if left <= 0:
            break
        add = min(left, cap)
        res += add * c
        left -= add
    return res


def calc_max_total_time(n: int, x: int, m: int,
                        a: List[int], q: List[int], r: List[int]) -> int:
    """最大总占用时长：单位系数为 a_i"""
    coeff = [a[i] for i in range(n)]
    base = 0
    caps = []
    for i in range(n):
        base += q[i] * coeff[i]
        caps.append((coeff[i], r[i] - q[i]))
    left = m - sum(q)
    caps.sort(key=lambda t: -t[0])
    res = base
    for c, cap in caps:
        if left <= 0:
            break
        add = min(left, cap)
        res += add * c
        left -= add
    return res


def main():
    n, x, m = map(int, input().split())
    a = list(map(int, input().split()))
    q = list(map(int, input().split()))
    r = list(map(int, input().split()))
    L = calc_min_total_time(n, x, m, a, q, r)
    R = calc_max_total_time(n, x, m, a, q, r)
    print(L, R)


if __name__ == "__main__":
    main()
```

```java
import java.io.*;
import java.util.*;

public class Main {

    static long calcMin(int n, int x, long m, int[] a, long[] q, long[] r) {
        long base = 0;
        long left = m;
        for (int i = 0; i < n; i++) left -= q[i];

        long[][] arr = new long[n][2];
        for (int i = 0; i < n; i++) {
            long coef = Math.min(a[i], x);
            base += q[i] * coef;
            arr[i][0] = coef;
            arr[i][1] = r[i] - q[i];
        }
        Arrays.sort(arr, new Comparator<long[]>() {
            public int compare(long[] p1, long[] p2) {
                return Long.compare(p1[0], p2[0]);
            }
        });
        long res = base;
        for (int i = 0; i < n && left > 0; i++) {
            long add = Math.min(left, arr[i][1]);
            res += add * arr[i][0];
            left -= add;
        }
        return res;
    }

    static long calcMax(int n, int x, long m, int[] a, long[] q, long[] r) {
        long base = 0;
        long left = m;
        for (int i = 0; i < n; i++) left -= q[i];

        long[][] arr = new long[n][2];
        for (int i = 0; i < n; i++) {
            long coef = a[i];
            base += q[i] * coef;
            arr[i][0] = coef;
            arr[i][1] = r[i] - q[i];
        }
        Arrays.sort(arr, new Comparator<long[]>() {
            public int compare(long[] p1, long[] p2) {
                return Long.compare(p2[0], p1[0]);
            }
        });
        long res = base;
        for (int i = 0; i < n && left > 0; i++) {
            long add = Math.min(left, arr[i][1]);
            res += add * arr[i][0];
            left -= add;
        }
        return res;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;

        st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int x = Integer.parseInt(st.nextToken());
        long m = Long.parseLong(st.nextToken());

        int[] a = new int[n];
        long[] q = new long[n];
        long[] r = new long[n];

        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) a[i] = Integer.parseInt(st.nextToken());

        for (int i = 0; i < n; i++) q[i] = Long.parseLong(st.nextToken());

    }
```

```cpp
#include <bits/stdc++.h>
using namespace std;

long long calc_min_total_time(int n, long long x, long long m,
                              const vector<long long>& a,
                              const vector<long long>& q,
                              const vector<long long>& r) {
    vector<pair<long long, long long>> vec;
    vec.reserve(n);
    long long base = 0;
    long long left = m;
    for (int i = 0; i < n; ++i) left -= q[i];
    for (int i = 0; i < n; ++i) {
        long long coef = min(a[i], x);
        base += q[i] * coef;
        vec.push_back({coef, r[i] - q[i]});
    }
    sort(vec.begin(), vec.end(), [](const auto& p1, const auto& p2) {
        return p1.first < p2.first;
    });
    long long res = base;
    for (auto &p : vec) {
        if (left <= 0) break;
        long long add = min(left, p.second);
        res += add * p.first;
        left -= add;
    }
    return res;
}

                              const vector<long long>& q,
                              const vector<long long>& r) {
    vector<pair<long long, long long>> vec;
    vec.reserve(n);
    long long base = 0;
    long long left = m;
    for (int i = 0; i < n; ++i) left -= q[i];
    for (int i = 0; i < n; ++i) {
        long long coef = a[i];
        base += q[i] * coef;
        vec.push_back({coef, r[i] - q[i]});

}
```
