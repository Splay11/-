## 解题思路

本题可以被转化为：首先花费一定代价从起点 $1$ 到达某个全能补给站，之后再免费（通行费用为 $0$）走到终点 $n$。因此，最优路径的总费用等于从起点到某个补给站的**最小费用**，前提是该补给站能够通过后续的免费航线到达终点。

具体分为以下三步：

1. **正向 Dijkstra 求起点到所有星球的最小费用**  
   以星球 $1$ 为源点，对原图（有向边，边权为通行费用）运行一次堆优化 Dijkstra，得到 `dist[u]` 表示从起点 $1$ 到星球 $u$ 的最小通行费用总和。  
   若起点本身无法到达某个星球，其 `dist` 值为无穷大。

2. **反向 BFS 求能到达终点的星球集合**  
   从终点 $n$ 出发，在**反向图**（所有有向边反转方向）上做 BFS，标记所有能够到达终点的节点 `can[u]`。由于到达补给站后所有航线免费，我们只关心拓扑连通性，BFS 不考虑边权。

3. **枚举补给站取最优值**  
   遍历所有全能补给站 $p$：  
   - 若 `can[p]` 为真（即从 $p$ 可以通过免费航线到达终点），且 `dist[p]` 不为无穷大，则总费用为 `dist[p]`。  
   - 答案为所有可行补给站对应的 `dist[p]` 的最小值。  
   若没有可行的补给站，输出 $-1$。

该做法的正确性基于：一旦抵达补给站，后续费用为零，因此总开销完全由“从起点到该补给站的费用”决定，且必须保证后续能抵达终点。

## 复杂度分析

- **时间复杂度**：$O((n+m)\log n)$。其中 Dijkstra 算法需要 $O((n+m)\log n)$，反向 BFS 需要 $O(n+m)$，枚举补给站需要 $O(k)$。  
- **空间复杂度**：$O(n+m)$。用于存储正向图、反向图、距离数组和可达性标记。

## 代码实现

### Python

```python
import sys
import heapq
from collections import deque

INF = 10 ** 18

def main():
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    # 读入全能补给站
    supply = [False] * n
    supply_nodes = []
    for _ in range(k):
        r = int(next(it)) - 1   # 转为 0-based 编号
        supply[r] = True
        supply_nodes.append(r)

    # 建图：正向带权图 + 反向无权图（仅用于 BFS）
    g = [[] for _ in range(n)]   # 正图：存 (v, w)
    rg = [[] for _ in range(n)]  # 反图：存可到达当前节点的结点列表

    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        g[u].append((v, w))
        rg[v].append(u)

    # 第一步：Dijkstra 求从起点1到所有星球的最小费用
    dist = [INF] * n
    dist[0] = 0   # 起点为星球1，0-index 为 0
    pq = [(0, 0)]
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist[u]:
            continue
        for v, w in g[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))

    # 第二步：反向 BFS 标记能到达终点 n 的所有星球
    can_reach = [False] * n
    can_reach[n - 1] = True   # 终点自身可达
    q = deque([n - 1])
    while q:
        u = q.popleft()
        for v in rg[u]:       # 反向边
            if not can_reach[v]:
                can_reach[v] = True
                q.append(v)

    # 第三步：枚举所有补给站，取最小 dist[p]
    ans = INF
    for p in supply_nodes:
        if can_reach[p] and dist[p] < ans:
            ans = dist[p]

    print(-1 if ans >= INF else ans)

if __name__ == '__main__':
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long INF = (long) 1e18;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());

        // 读入全能补给站
        boolean[] supply = new boolean[n];
        int[] supplyNodes = new int[k];
        for (int i = 0; i < k; i++) {
            int r = Integer.parseInt(br.readLine()) - 1; // 转为 0-based
            supply[r] = true;
            supplyNodes[i] = r;
        }

        // 建图：正向图 (v, w) 和反向图 (仅存结点)
        List<int[]>[] g = new ArrayList[n];
        List<Integer>[] rg = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            g[i] = new ArrayList<>();
            rg[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            int w = Integer.parseInt(st.nextToken());
            g[u].add(new int[]{v, w});
            rg[v].add(u);   // 反向边
        }

        // 第一步：Dijkstra 求从起点到所有星球的最小费用
        long[] dist = new long[n];
        Arrays.fill(dist, INF);
        dist[0] = 0;   // 起点为星球1，0-index 为 0
        PriorityQueue<long[]> pq = new PriorityQueue<>(
            (a, b) -> Long.compare(a[0], b[0]));
        pq.offer(new long[]{0, 0});

        while (!pq.isEmpty()) {
            long[] top = pq.poll();
            long d = top[0];
            int u = (int) top[1];
            if (d != dist[u]) continue;
            for (int[] e : g[u]) {
                int v = e[0], w = e[1];
                long nd = d + w;
                if (nd < dist[v]) {
                    dist[v] = nd;
                    pq.offer(new long[]{nd, v});
                }
            }
        }

        // 第二步：反向 BFS 标记能到达终点的结点
        boolean[] canReach = new boolean[n];
        canReach[n - 1] = true;  // 终点自身可达
        ArrayDeque<Integer> q = new ArrayDeque<>();
        q.offer(n - 1);

        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : rg[u]) {  // 沿反向边走
                if (!canReach[v]) {
                    canReach[v] = true;
                    q.offer(v);
                }
            }
        }

        // 第三步：枚举所有补给站，取最小 dist[p]
        long ans = INF;
        for (int p : supplyNodes) {
            if (canReach[p] && dist[p] < ans) {
                ans = dist[p];
            }
        }

        System.out.println(ans >= INF ? -1 : ans);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <deque>
using namespace std;
using ll = long long;

const ll INF = 1e18;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, k;
    cin >> n >> m >> k;

    // 读入全能补给站
    vector<bool> supply(n, false);
    vector<int> supplyNodes;
    supplyNodes.reserve(k);
    for (int i = 0; i < k; ++i) {
        int r;
        cin >> r;
        --r;  // 转为 0-based
        supply[r] = true;
        supplyNodes.push_back(r);
    }

    // 建图：正向带权图 + 反向无权图
    vector<vector<pair<int, int>>> g(n);   // 正图: (v, w)
    vector<vector<int>> rg(n);              // 反图: 可到达该结点的结点列表

    for (int i = 0; i < m; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        --u; --v;
        g[u].push_back({v, w});
        rg[v].push_back(u);
    }

    // 第一步：Dijkstra 求从起点到所有星球的最小费用
    vector<ll> dist(n, INF);
    dist[0] = 0;   // 起点为星球1，0-index 为 0
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<>> pq;
    pq.push({0, 0});

    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d != dist[u]) continue;
        for (auto [v, w] : g[u]) {
            ll nd = d + w;
            if (nd < dist[v]) {
                dist[v] = nd;
                pq.push({nd, v});
            }
        }
    }

    // 第二步：反向 BFS 标记能到达终点的结点
    vector<bool> canReach(n, false);
    canReach[n - 1] = true;  // 终点自身可达
    deque<int> q;
    q.push_back(n - 1);

    while (!q.empty()) {
        int u = q.front();
        q.pop_front();
        for (int v : rg[u]) {  // 沿反向边走
            if (!canReach[v]) {
                canReach[v] = true;
                q.push_back(v);
            }
        }
    }

    // 第三步：枚举所有补给站，取最小 dist[p]
    ll ans = INF;
    for (int p : supplyNodes) {
        if (canReach[p] && dist[p] < ans) {
            ans = dist[p];
        }
    }

    cout << (ans >= INF ? -1 : ans) << '\n';
    return 0;
}
```
