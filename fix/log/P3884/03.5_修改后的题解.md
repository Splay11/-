## 解题思路

均衡号的二进制长度 $L$ 必为偶数 $L=2k$，且最高位为 $1$、$1$ 的个数恰好为 $k$。

长度 $2k$ 的最小均衡号形如 $1\,0^k\,1^{k-1}$。

给定 $x$：

* 若其长度 $L$ 为奇数，答案为长度 $L+1$ 的最小均衡号；
* 若 $L=2k$ 为偶数：若 $x$ 本身已均衡则返回 $x$；否则从低到高找可以把 $0$ 改成 $1$ 的位置，使高位与 $x$ 相同、低位放成最小合法后缀。若同长度无解，则取长度 $L+2$ 的最小均衡号。

## 复杂度分析

单组 $O(L)$（$L\le 30$），总时间 $O(T\cdot L)$，额外空间 $O(1)$。

## 代码实现

### Python

```python
# 题意：多组 x，输出不小于 x 的最小“二进制均衡数”
# 思路：见题解。实现两个函数：
# 1) min_perfect_by_len(L): 计算偶数长度 L 的最小均衡数
# 2) next_same_len(x, L, k): 在固定长度 L 内，构造最小的 y>=x 且 popcount=k；失败返回 -1

import sys

def min_perfect_by_len(L: int) -> int:
    """给定偶数长度 L=2k，返回该长度的最小均衡数：1 0^k 1^(k-1)"""
    k = L // 2
    high = 1 << (L - 1)                # 最高位 1
    low = (1 << (k - 1)) - 1 if k >= 2 else 0  # 低位 (k-1) 个 1
    return high | low

def next_same_len(x: int, L: int, k: int) -> int:
    """同长度内构造答案；不可行返回 -1"""
    # 如果 x 已经是均衡数
    if bin(x).count("1") == k:
        return x
    # 从低位到高位枚举可提升的位置 p（x 的第 p 位必须是 0）
    for p in range(L):
        if (x >> p) & 1:  # 该位是 1，不能把 0→1，跳过
            continue
        ones_hi = bin(x >> (p + 1)).count("1")  # 高于 p 的 1 的个数
        remOnes = k - (ones_hi + 1)           # 当前位设为 1 后，低位还需多少个 1
        if 0 <= remOnes <= p:
            # 高位保持与 x 相同，清掉低 p+1 位
            high = (x >> (p + 1)) << (p + 1)
            # 位置 p 设为 1，低位放成最小合法后缀：最低 remOnes 位全为 1
            low = (1 << remOnes) - 1 if remOnes > 0 else 0
            return high | (1 << p) | low
    return -1  # 同长度无解

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    ans = []
    idx = 1
    for _ in range(t):
        x = data[idx]; idx += 1
        L = x.bit_length()
        if L % 2 == 1:
            # 下一偶数长度的最小均衡数
            y = min_perfect_by_len(L + 1)
            ans.append(str(y))
            continue
        k = L // 2
        y = next_same_len(x, L, k)
        if y != -1:
            ans.append(str(y))
        else:
            # 同长度无解，跳到下一偶数长度（L+2）
            y2 = min_perfect_by_len(L + 2)
            ans.append(str(y2))
    sys.stdout.write("\n".join(ans))

if __name__ == "__main__":
    solve()
```

### Java

```java
// 题意同上，按位贪心 + 组合构造
// 注意：答案可能超过 2^31-1（例如长度 32 的最小均衡数），需使用 long。
import java.io.*;
import java.util.*;

public class Main {

    // 计算偶数长度 L=2k 的最小均衡数：1 0^k 1^(k-1)
    static long minPerfectByLen(int L) {
        int k = L / 2;
        long high = 1L << (L - 1);                  // 最高位 1
        long low = (k >= 2) ? ((1L << (k - 1)) - 1) : 0L; // (k-1) 个 1 在最低位
        return high | low;
    }

    // 在固定长度 L 内，构造最小的 y >= x 且 popcount = k；不可行返回 -1
    static long nextSameLen(long x, int L, int k) {
        if (Long.bitCount(x) == k) return x;
        for (int p = 0; p < L; p++) {
            if (((x >> p) & 1L) == 1L) continue; // 该位是 1，无法 0→1
            int ones_hi = Long.bitCount(x >> (p + 1));  // 高位 1 的个数
            int remOnes = k - (ones_hi + 1);            // 该位设 1 后低位需多少个 1
            if (remOnes >= 0 && remOnes <= p) {
                long high = (x >> (p + 1)) << (p + 1);  // 清低 p+1 位
                long low = (remOnes > 0) ? ((1L << remOnes) - 1) : 0L; // 最小后缀
                return high | (1L << p) | low;
            }
        }
        return -1L; // 同长度无解
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        int T = (int) fs.nextLong();
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < T; i++) {
            long x = fs.nextLong();
            int L = 64 - Long.numberOfLeadingZeros(x);
            if ((L & 1) == 1) {
                long y = minPerfectByLen(L + 1);
                sb.append(y).append('\n');
            } else {
                int k = L / 2;
                long y = nextSameLen(x, L, k);
                if (y != -1L) {
                    sb.append(y).append('\n');
                } else {
                    long y2 = minPerfectByLen(L + 2);
                    sb.append(y2).append('\n');
                }
            }
        }
        System.out.print(sb.toString());
    }

    // 简单快读（基于缓冲输入），应对 T<=1e5
    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        long nextLong() throws IOException {
            int c; do { c = read(); } while (c <= ' '); // 跳过空白
            int sign = 1;
            if (c == '-') { sign = -1; c = read(); }
            long val = 0;
            while (c > ' ') {
                val = val * 10 + (c - '0');
                c = read();
            }
            return val * sign;
        }
    }
}
```

### C++

```cpp
// 题意同上：按位贪心 + 组合构造最小合法后缀
// 注意使用 long long，答案可能超过 2^31-1。
#include <bits/stdc++.h>
using namespace std;

// 偶数长度 L=2k 的最小均衡数：1 0^k 1^(k-1)
long long minPerfectByLen(int L) {
    int k = L / 2;
    long long high = 1LL << (L - 1);                 // 最高位 1
    long long low = (k >= 2) ? ((1LL << (k - 1)) - 1) : 0LL; // 低位 (k-1) 个 1
    return high | low;
}

// 同长度内构造最小 y>=x 且 popcount=k；失败返回 -1
long long nextSameLen(long long x, int L, int k) {
    if (__builtin_popcountll(x) == k) return x;
    for (int p = 0; p < L; ++p) {
        if ((x >> p) & 1LL) continue; // 非 0 位不能 0→1
        int ones_hi = __builtin_popcountll(x >> (p + 1)); // 高位 1 的个数
        int remOnes = k - (ones_hi + 1);                  // 低位还需的 1
        if (0 <= remOnes && remOnes <= p) {
            long long high = (x >> (p + 1)) << (p + 1);   // 清低 p+1 位
            long long low = (remOnes > 0) ? ((1LL << remOnes) - 1) : 0LL; // 最小后缀
            return high | (1LL << p) | low;
        }
    }
    return -1LL; // 同长度无解
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T; 
    if (!(cin >> T)) return 0;
    while (T--) {
        long long x; cin >> x;
        int L = 64 - __builtin_clzll(x); // bit_length
        if (L & 1) {
            cout << minPerfectByLen(L + 1) << '\n';
        } else {
            int k = L / 2;
            long long y = nextSameLen(x, L, k);
            if (y != -1LL) cout << y << '\n';
            else cout << minPerfectByLen(L + 2) << '\n';
        }
    }
    return 0;
}
```
