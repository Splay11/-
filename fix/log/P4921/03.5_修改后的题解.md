## 解题思路

设全体账号掩码按位与为 $A$。吊销某个账号 $a_i$ 后，剩余掩码按位与只可能在某些二进制位上从 `0` 变成 `1`，不会把 $A$ 中已是 `1` 的位改成 `0`。

因此吊销 $a_i$ 后结果严格变大，当且仅当存在某一二进制位：

- 该位在 $A$ 中为 `0`；
- 全体掩码里只有 $a_i$ 这一位是 `0`。

按位统计即可：

1. 枚举二进制位 `0` 到 `30`；
2. 统计该位为 `0` 的元素个数；
3. 若恰好 `1` 个，则吊销该账号后这一位会变成 `1`；
4. 用标记数组记录下标，避免同一位置被多个位重复计数。

## 复杂度分析

时间复杂度 $O(31n)$，即 $O(n)$；空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys


def count_valid(a):
    n = len(a)
    ok = [False] * n

    for bit in range(31):
        zero_count = 0
        zero_pos = -1

        for i in range(n):
            if ((a[i] >> bit) & 1) == 0:
                zero_count += 1
                zero_pos = i

        if zero_count == 1:
            ok[zero_pos] = True

    return sum(ok)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    idx = 1
    ans = []

    for _ in range(t):
        n = data[idx]
        idx += 1
        a = data[idx:idx + n]
        idx += n

        ans.append(str(count_valid(a)))

    print("\n".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static int countValid(int[] a) {
        int n = a.length;
        boolean[] ok = new boolean[n];

        for (int bit = 0; bit <= 30; bit++) {
            int zeroCount = 0;
            int zeroPos = -1;

            for (int i = 0; i < n; i++) {
                if (((a[i] >> bit) & 1) == 0) {
                    zeroCount++;
                    zeroPos = i;
                }
            }

            if (zeroCount == 1) {
                ok[zeroPos] = true;
            }
        }

        int ans = 0;
        for (boolean v : ok) {
            if (v) ans++;
        }
        return ans;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        StringBuilder sb = new StringBuilder();

        int T = fs.nextInt();

        for (int tc = 0; tc < T; tc++) {
            int n = fs.nextInt();
            int[] a = new int[n];

            for (int i = 0; i < n; i++) {
                a[i] = fs.nextInt();
            }

            sb.append(countValid(a)).append('\n');
        }

        System.out.print(sb.toString());
    }

    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        FastScanner(InputStream is) {
            in = is;
        }

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        int nextInt() throws IOException {
            int c;
            do {
                c = read();
            } while (c <= ' ');

            int sign = 1;
            if (c == '-') {
                sign = -1;
                c = read();
            }

            int val = 0;
            while (c > ' ') {
                val = val * 10 + c - '0';
                c = read();
            }

            return val * sign;
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int countValid(const vector<int>& a) {
    int n = a.size();
    vector<int> ok(n, 0);

    for (int bit = 0; bit <= 30; bit++) {
        int zeroCount = 0;
        int zeroPos = -1;

        for (int i = 0; i < n; i++) {
            if (((a[i] >> bit) & 1) == 0) {
                zeroCount++;
                zeroPos = i;
            }
        }

        if (zeroCount == 1) {
            ok[zeroPos] = 1;
        }
    }

    int ans = 0;
    for (int v : ok) {
        ans += v;
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n;
        cin >> n;

        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }

        cout << countValid(a) << '\n';
    }

    return 0;
}
```
