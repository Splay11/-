## 解题思路

本题要求统计长度为 $k$ 的连续子段中，有多少个恰好构成“完整阶”。根据定义，一个序列 $B$ 是完整阶的**充要条件**为：

1. $B$ 中所有元素互不相同；
2. $\max(B) - \min(B) = |B| - 1$。

而当 $|B| = k$ 时，第二个条件等价于 $\max - \min = k - 1$。因此，我们可以通过**固定长度的滑动窗口**依次检查每一个长度为 $k$ 的子数组，并利用以下两个技巧进行高效判断：

- **统计互异元素个数**：用一个桶（计数数组）`cnt[]` 记录当前窗口内每个值的出现次数，并维护一个变量 `diff` 表示窗口内**不同元素**的个数。每加入一个新元素 $x$，若 `cnt[x]` 从 $0$ 变为 $1$，则 `diff` 加 $1$；每移出窗口一个元素 $y$，若 `cnt[y]` 从 $1$ 变为 $0$，则 `diff` 减 $1$。当且仅当 `diff == k` 时，窗口内元素互异。
- **快速获取窗口最值**：由于窗口固定且只进行左右移动，可以使用**两个双端队列（单调队列）**分别维护窗口内的最大值和最小值：
  - **最大值队列**`max_q`：从队首到队尾单调递减，队首即为当前窗口最大值的索引。
  - **最小值队列**`min_q`：从队首到队尾单调递增，队首即为当前窗口最小值的索引。
  每次滑动时，将新元素索引加入队列，并从队尾弹出破坏单调性的元素；同时检查队首索引是否已滑出窗口（`< i - k + 1`），若是则弹出队首。这样可以在 $O(1)$ 时间内得到窗口的最值。

算法流程如下：

1. 读入 $n, k$ 和序列 $A$。初始化桶 `cnt[]` 大小根据值域（$1 \le A_i \le 10^6$）设定，`diff = 0`，答案 `ans = 0`。
2. 遍历 $i = 0$ 到 $n-1$，执行：
   - **加入** $A[i]$：`cnt[A[i]]++`，若 `cnt[A[i]] == 1` 则 `diff++`。
   - 维护 `max_q`：弹出队尾所有满足 $A[\text{队尾}] \le A[i]$ 的索引，再将 $i$ 入队。
   - 维护 `min_q`：弹出队尾所有满足 $A[\text{队尾}] \ge A[i]$ 的索引，再将 $i$ 入队。
   - 当 $i \ge k$ 时，**移出**左边界元素 $A[i-k]$：`cnt[A[i-k]]--`，若 `cnt[A[i-k]] == 0` 则 `diff--`。
   - **清理过期元素**：将 `max_q`、`min_q` 中队首索引小于 $i-k+1$ 的元素弹出（因为当前窗口左端点为 $i-k+1$）。
   - 当 $i \ge k-1$ 时，窗口长度恰好为 $k$，此时若 `diff == k`（互异）且 $A[max_q.\text{队首}] - A[min_q.\text{队首}] == k-1$，则 `ans++`。
3. 输出 `ans`。

该方法只遍历数组一次，每步操作均为均摊 $O(1)$。

## 复杂度分析

- **时间复杂度**：$O(n)$，每个元素入队、出队、入桶、出桶均摊 $O(1)$，总体线性。
- **空间复杂度**：$O(n + V)$，其中 $V$ 是元素值域上限（$10^6$），用于存储计数数组；两个双端队列最多存储 $n$ 个索引，也属于 $O(n)$。

## 代码实现

### Python

```python
import sys
from collections import deque

def main():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    k = int(next(it))
    A = [int(next(it)) for _ in range(n)]

    MAX_VAL = 1000005
    cnt = [0] * MAX_VAL
    diff = 0
    ans = 0

    max_q = deque()  # 维护最大值，队首索引对应元素最大
    min_q = deque()  # 维护最小值，队首索引对应元素最小

    for i in range(n):
        x = A[i]
        # 加入当前元素
        cnt[x] += 1
        if cnt[x] == 1:
            diff += 1

        # 更新最大值队列
        while max_q and A[max_q[-1]] <= x:
            max_q.pop()
        max_q.append(i)

        # 更新最小值队列
        while min_q and A[min_q[-1]] >= x:
            min_q.pop()
        min_q.append(i)

        # 当窗口长度超过k，移除左边界元素
        if i >= k:
            left_val = A[i - k]
            cnt[left_val] -= 1
            if cnt[left_val] == 0:
                diff -= 1

        # 清理已经滑出窗口的索引
        left = i - k + 1
        while max_q and max_q[0] < left:
            max_q.popleft()
        while min_q and min_q[0] < left:
            min_q.popleft()

        # 窗口长度达到k时进行判断
        if i >= k - 1:
            if diff == k:
                max_val = A[max_q[0]]
                min_val = A[min_q[0]]
                if max_val - min_val == k - 1:
                    ans += 1

    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        // 快速输入
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] first = br.readLine().split(" ");
        int n = Integer.parseInt(first[0]);
        int k = Integer.parseInt(first[1]);
        int[] A = new int[n];
        String[] nums = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            A[i] = Integer.parseInt(nums[i]);
        }

        final int MAX_VAL = 1000005;
        int[] cnt = new int[MAX_VAL];
        int diff = 0;
        int ans = 0;

        Deque<Integer> maxQ = new ArrayDeque<>(); // 单调递减，队首最大
        Deque<Integer> minQ = new ArrayDeque<>(); // 单调递增，队首最小

        for (int i = 0; i < n; i++) {
            int x = A[i];
            // 加入
            cnt[x]++;
            if (cnt[x] == 1) diff++;

            // 更新最大值队列
            while (!maxQ.isEmpty() && A[maxQ.peekLast()] <= x) {
                maxQ.pollLast();
            }
            maxQ.offerLast(i);

            // 更新最小值队列
            while (!minQ.isEmpty() && A[minQ.peekLast()] >= x) {
                minQ.pollLast();
            }
            minQ.offerLast(i);

            // 移除左边界
            if (i >= k) {
                int leftVal = A[i - k];
                cnt[leftVal]--;
                if (cnt[leftVal] == 0) diff--;
            }

            // 清理过期索引
            int left = i - k + 1;
            while (!maxQ.isEmpty() && maxQ.peekFirst() < left) {
                maxQ.pollFirst();
            }
            while (!minQ.isEmpty() && minQ.peekFirst() < left) {
                minQ.pollFirst();
            }

            // 判断窗口
            if (i >= k - 1) {
                if (diff == k) {
                    int maxVal = A[maxQ.peekFirst()];
                    int minVal = A[minQ.peekFirst()];
                    if (maxVal - minVal == k - 1) {
                        ans++;
                    }
                }
            }
        }

        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;
    vector<int> A(n);
    for (int i = 0; i < n; ++i) {
        cin >> A[i];
    }

    const int MAX_VAL = 1000005;
    vector<int> cnt(MAX_VAL, 0);
    int diff = 0;
    int ans = 0;

    deque<int> max_q, min_q; // 分别维护最大值和最小值索引

    for (int i = 0; i < n; ++i) {
        int x = A[i];
        // 加入当前元素
        cnt[x]++;
        if (cnt[x] == 1) diff++;

        // 维护递减的最大值队列
        while (!max_q.empty() && A[max_q.back()] <= x) {
            max_q.pop_back();
        }
        max_q.push_back(i);

        // 维护递增的最小值队列
        while (!min_q.empty() && A[min_q.back()] >= x) {
            min_q.pop_back();
        }
        min_q.push_back(i);

        // 窗口长度超出k，移除左边界
        if (i >= k) {
            int left_val = A[i - k];
            cnt[left_val]--;
            if (cnt[left_val] == 0) diff--;
        }

        // 弹出过期索引
        int left = i - k + 1;
        while (!max_q.empty() && max_q.front() < left) {
            max_q.pop_front();
        }
        while (!min_q.empty() && min_q.front() < left) {
            min_q.pop_front();
        }

        // 判断长度为k的窗口
        if (i >= k - 1) {
            if (diff == k) {
                int max_val = A[max_q.front()];
                int min_val = A[min_q.front()];
                if (max_val - min_val == k - 1) {
                    ans++;
                }
            }
        }
    }

    cout << ans << '\n';
    return 0;
}
```
