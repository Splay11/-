## 解题思路

每件样品的风险指数是它的不同质因子个数（`1` 的指数为 `0`）。下线一段长度为 $k$ 的连续样品后，剩余指数和等于总指数和减去被下线窗口的指数和。因此只需让被下线窗口的指数和最小。

对每个 $a_i$ 分解质因数得到风险指数，再用长度为 $k$ 的滑动窗口求出最小窗口和，答案为总和减去该最小值。

## 复杂度分析

时间复杂度 $O(n\sqrt{A})$，其中 $A$ 为 $a_i$ 的上界；空间复杂度 $O(n)$。

## 代码实现

### Python

```python
def count_prime_factors(a):
    cnt = 0
    if a == 1:
        return 0
    i = 2
    while i * i <= a:
        if a % i == 0:
            cnt += 1
            while a % i == 0:
                a //= i
        i += 1
    if a > 1:
        cnt += 1
    return cnt

def main():
    import sys
    input = sys.stdin.readline
    n, k = map(int, input().split())
    arr = list(map(int, input().split()))

    weights = []
    total_sum = 0
    for num in arr:
        w = count_prime_factors(num)
        weights.append(w)
        total_sum += w

    window_sum = sum(weights[:k])
    min_window_sum = window_sum
    for i in range(k, n):
        window_sum += weights[i] - weights[i - k]
        if window_sum < min_window_sum:
            min_window_sum = window_sum

    print(total_sum - min_window_sum)

if __name__ == '__main__':
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static int countPrimeFactors(int a) {
        int cnt = 0;
        if (a == 1) return 0;
        for (int i = 2; i * i <= a; i++) {
            if (a % i == 0) {
                cnt++;
                while (a % i == 0) {
                    a /= i;
                }
            }
        }
        if (a > 1) cnt++;
        return cnt;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        int n = Integer.parseInt(s[0]);
        int k = Integer.parseInt(s[1]);
        int[] a = new int[n];
        String[] nums = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(nums[i]);
        }

        int[] weights = new int[n];
        int totalSum = 0;
        for (int i = 0; i < n; i++) {
            weights[i] = countPrimeFactors(a[i]);
            totalSum += weights[i];
        }

        int windowSum = 0;
        for (int i = 0; i < k; i++) {
            windowSum += weights[i];
        }
        int minWindowSum = windowSum;
        for (int i = k; i < n; i++) {
            windowSum += weights[i] - weights[i - k];
            minWindowSum = Math.min(minWindowSum, windowSum);
        }

        System.out.println(totalSum - minWindowSum);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

int countPrimeFactors(int a) {
    int cnt = 0;
    if (a == 1) return 0;
    for (int j = 2; j * j <= a; j++) {
        if (a % j == 0) {
            ++cnt;
            while (a % j == 0) {
                a /= j;
            }
        }
    }
    if (a > 1) ++cnt;
    return cnt;
}

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    vector<int> w(n, 0);
    int totalSum = 0;

    for (int i = 0; i < n; i++) {
        cin >> a[i];
        w[i] = countPrimeFactors(a[i]);
        totalSum += w[i];
    }

    int windowSum = 0;
    for (int i = 0; i < k; i++) {
        windowSum += w[i];
    }
    int minWindowSum = windowSum;
    for (int i = k; i < n; i++) {
        windowSum += w[i] - w[i - k];
        if (windowSum < minWindowSum)
            minWindowSum = windowSum;
    }

    cout << totalSum - minWindowSum << endl;
    return 0;
}
```
