## 解题思路

最终要求所有编号的最大公约数大于 1，即

$$
\gcd(a_1,a_2,\ldots,a_n)>1
$$

设这个公共因子为 $d>1$，则 $d$ 一定含有某个质因子 $p$。因此，所有编号也一定能被某个质数 $p$ 整除。

对于固定的质数 $p$：

- 原数组中已经能被 $p$ 整除的元素可以直接保留；
- 不能被 $p$ 整除的元素需要修改成 $p$ 的倍数；
- 若原数组中有 $cnt_p$ 个元素能被 $p$ 整除，则需要修改的元素数量为 $n-cnt_p$。

所以问题转化为：

> 找到一个质数 $p$，使原数组中能被 $p$ 整除的元素数量最多。

设这个最大数量为 $max\_count$，答案就是 $n-max\_count$。

具体实现：

1. 统计每个编号 $x$ 出现的次数，记为 $freq[x]$。
2. 使用埃氏筛求出不超过数组最大值的所有质数。
3. 对于每个质数 $p$，枚举它的所有倍数 $p,2p,3p,\ldots$，累加这些倍数在 $freq$ 中的出现次数，得到能被 $p$ 整除的元素数量。
4. 取所有质数对应的数量最大值，最后用 $n$ 减去该最大值。

特别地，编号 $1$ 不会被任何质数整除，因此如果所有编号都是 $1$，最大数量为 $0$，需要修改 $n$ 个货箱。

## 复杂度分析

设数组中的最大值为 $M=\max(a_1,a_2,\ldots,a_n)$。

使用埃氏筛求质数的时间复杂度为：

$$
O(M\log\log M)
$$

枚举所有质数的倍数，总时间复杂度同样为：

$$
O(M\log\log M)
$$

统计数组元素频率需要 $O(n)$ 时间。

因此总时间复杂度为：

$$
O(n+M\log\log M)
$$

空间复杂度为：

$$
O(M)
$$

在 $n\le 120000$、$M\le 900000$ 的数据范围下可以通过。

## 代码实现

### Python

```python
import sys

def min_operations(n, nums):
    max_value = max(nums)

    # 统计每个编号出现的次数
    freq = [0] * (max_value + 1)
    for x in nums:
        freq[x] += 1

    # 埃氏筛求质数
    is_prime = [True] * (max_value + 1)
    if max_value >= 0:
        is_prime[0] = False
    if max_value >= 1:
        is_prime[1] = False

    i = 2
    while i * i <= max_value:
        if is_prime[i]:
            for j in range(i * i, max_value + 1, i):
                is_prime[j] = False
        i += 1

    max_count = 0

    # 枚举每个质数，统计能被它整除的元素数量
    for p in range(2, max_value + 1):
        if is_prime[p]:
            count = 0

            # 枚举质数 p 的所有倍数
            for multiple in range(p, max_value + 1, p):
                count += freq[multiple]

            max_count = max(max_count, count)

    return n - max_count


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    nums = data[1:1 + n]

    print(min_operations(n, nums))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    public static int minOperations(int[] nums) {
        int n = nums.length;
        int maxValue = 0;

        for (int x : nums) {
            maxValue = Math.max(maxValue, x);
        }

        // 统计每个编号出现的次数
        int[] freq = new int[maxValue + 1];
        for (int x : nums) {
            freq[x]++;
        }

        // 埃氏筛求质数
        boolean[] isPrime = new boolean[maxValue + 1];
        Arrays.fill(isPrime, true);

        if (maxValue >= 0) {
            isPrime[0] = false;
        }
        if (maxValue >= 1) {
            isPrime[1] = false;
        }

        for (int i = 2; i * i <= maxValue; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= maxValue; j += i) {
                    isPrime[j] = false;
                }
            }
        }

        int maxCount = 0;

        // 枚举每个质数，统计能被它整除的元素数量
        for (int p = 2; p <= maxValue; p++) {
            if (isPrime[p]) {
                int count = 0;

                // 枚举质数 p 的所有倍数
                for (int multiple = p; multiple <= maxValue; multiple += p) {
                    count += freq[multiple];
                }

                maxCount = Math.max(maxCount, count);
            }
        }

        return n - maxCount;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = 0;
        int[] nums = null;
        int idx = 0;

        String line;
        while ((line = br.readLine()) != null) {
            if (line.trim().isEmpty()) {
                continue;
            }

            StringTokenizer st = new StringTokenizer(line);

            if (n == 0) {
                n = Integer.parseInt(st.nextToken());
                nums = new int[n];
            }

            while (st.hasMoreTokens() && idx < n) {
                nums[idx++] = Integer.parseInt(st.nextToken());
            }

            if (idx == n) {
                break;
            }
        }

        System.out.println(minOperations(nums));
    }
}
```

```cpp
int minOperations(const vector<int>& nums) {

    int maxValue = *max_element(nums.begin(), nums.end());

    vector<int> freq(maxValue + 1, 0);
    for (int x : nums) {

    vector<bool> isPrime(maxValue + 1, true);

    if (maxValue >= 0) {
        isPrime[0] = false;
    }
    if (maxValue >= 1) {
        isPrime[1] = false;
    }

    for (int i = 2; i * i <= maxValue; i++) {
        if (isPrime[i]) {
            for (int j = i * i; j <= maxValue; j += i) {
                isPrime[j] = false;
            }
        }
    }

    int maxCount = 0;

    // 枚举每个质数，统计能被它整除的元素数量
    for (int p = 2; p <= maxValue; p++) {
```
