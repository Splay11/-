## 解题思路

我们首先考虑在不插入任何元素的情况下，最多能进行多少次删除。  
由于删除只发生在相邻且相等的两个元素之间，且删除后左右序列会拼接，这类似于经典的“消除相邻相同对”问题，可以用**栈**来模拟：

- 从左到右扫描序列。
- 如果栈非空且栈顶元素等于当前元素，则这两个元素可以消除，将栈顶弹出，删除次数加 $1$（可累计）。
- 否则将当前元素压入栈中。

最终栈中剩下的序列就是无法再消除的“简化序列”，设其长度为 $m$。  
原始序列长度为 $n$，那么原序列通过上述过程已经执行了 $(n-m)/2$ 次删除（因为每一对被消除的元素数量为 $2$）。

记 $base = \frac{n-m}{2}$ 为不插入时的最大删除次数。

现在考虑我们可以在任意位置插入**一个**任意正整数。  
由于原序列相邻相等就消，插入位置实际上只需考虑简化后的序列 $b$（即栈中剩余元素）。因为任何在原始序列中已被消除的位置之间插入，不会产生新的消除（那些位置已经为空）。所以插入等价于在简化序列 $b$ 的某个间隙或两端插入一个数。

在 $b$ 中，任意相邻元素都不同（否则栈模拟时会继续消除）。如果我们插入一个数 $x$ 到某位置，使得 $x$ 与它旁边某个元素相等，那么可以先消除这一对，然后两侧可能因为相等而继续消除，就像从中间向两边扩散消除。

例如 $b = [1, 2, 3, 2, 1]$，在中间的 $3$ 旁边插入 $3$，可得：
1. 先消除插入的 $3$ 和中心的 $3$；
2. 然后两侧的 $2$ 和 $2$ 相邻，消除；
3. 再两侧的 $1$ 和 $1$ 相邻，消除。

这个过程恰好相当于在 $b$ 中以某个位置为中心，向两边扩展消除尽可能多的元素。插入一个数最多能带来的额外消除次数，等于 $b$ 中以某个位置为中心的最长**奇数长度回文**的半径（半径定义为从中心到一侧的字符数，包含中心自身）。因为要使得两边对称消除，中心必须对应插入的那个数，且两侧必须是对称相等的序列。

因此，我们可以使用 **Manacher 算法**（仅处理奇数长度回文）在线性时间内求出简化序列 $b$ 中每个位置作为中心的最大回文半径，记 $maxRadius$ 为其最大值。

最终答案：
- 若 $b$ 为空（即 $m=0$），说明原序列已完全消除，插入一个数无法产生新的消除，答案为 $base$。
- 否则答案为 $base + maxRadius$。

## 复杂度分析

- 每组数据：栈模拟需要 $O(n)$ 时间。
- Manacher 求最长奇回文半径需要 $O(m)$ 时间，$m \le n$。
- 每组数据总时间复杂度 $O(n)$。
- 所有测试数据 $n$ 之和不超过 $4\times10^5$，因此总时间 $O(\sum n)$ 可以接受。
- 空间复杂度：栈与 Manacher 辅助数组均为 $O(n)$。

**时间复杂度**：$O(n)$  
**空间复杂度**：$O(n)$

## 代码实现

### Python

```python
import sys


def odd_manacher(arr):
    """返回整数数组 arr 的最长奇回文半径（半径 = 从中心向一侧的长度，含中心）"""
    n = len(arr)
    d = [0] * n
    l, r = 0, -1
    ans = 0

    for i in range(n):
        if i > r:
            k = 1
        else:
            k = min(d[l + r - i], r - i + 1)

        while i - k >= 0 and i + k < n and arr[i - k] == arr[i + k]:
            k += 1

        d[i] = k
        ans = max(ans, k)

        if i + k - 1 > r:
            l = i - k + 1
            r = i + k - 1

    return ans


def solve_one(n, a):
    stack = []
    # 用栈模拟原序列的全部消除
    for x in a:
        if stack and stack[-1] == x:
            stack.pop()
        else:
            stack.append(x)

    m = len(stack)
    base = (n - m) // 2          # 不插入时的删除次数

    if m == 0:                  # 已经全消完，插入一个元素不能带来新消除
        return base

    extra = odd_manacher(stack) # 插入一次最多带来的额外消除次数
    return base + extra


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    idx = 1
    out = []
    for _ in range(t):
        n = data[idx]
        idx += 1
        a = data[idx:idx + n]
        idx += n
        out.append(str(solve_one(n, a)))
    print("\n".join(out))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static class FastScanner {
        private final InputStream in = System.in;
        private final byte[] buf = new byte[1 << 16];
        private int ptr = 0, len = 0;

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buf);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buf[ptr++] & 0xFF;
        }

        int nextInt() throws IOException {
            int c = read();
            while (c != -1 && c <= ' ') c = read();
            if (c == -1) return -1;
            int sign = 1;
            if (c == '-') {
                sign = -1;
                c = read();
            }
            int val = 0;
            while (c != -1 && c > ' ') {
                val = val * 10 + c - '0';
                c = read();
            }
            return val * sign;
        }
    }

    static int oddManacher(int[] arr, int n) {
        int[] d = new int[n];
        int l = 0, r = -1;
        int ans = 0;

        for (int i = 0; i < n; i++) {
            int k = (i > r) ? 1 : Math.min(d[l + r - i], r - i + 1);
            while (i - k >= 0 && i + k < n && arr[i - k] == arr[i + k]) {
                k++;
            }
            d[i] = k;
            ans = Math.max(ans, k);
            if (i + k - 1 > r) {
                l = i - k + 1;
                r = i + k - 1;
            }
        }
        return ans;
    }

    static int solveOne(int n, int[] a) {
        int[] stack = new int[n];
        int top = 0;
        for (int x : a) {
            if (top > 0 && stack[top - 1] == x) {
                top--;
            } else {
                stack[top++] = x;
            }
        }

        int m = top;
        int base = (n - m) / 2;

        if (m == 0) return base;

        int[] reduced = Arrays.copyOf(stack, m);
        int extra = oddManacher(reduced, m);
        return base + extra;
    }

    public static void main(String[] args) throws IOException {
        FastScanner fs = new FastScanner();
        StringBuilder sb = new StringBuilder();
        int T = fs.nextInt();
        for (int tc = 0; tc < T; tc++) {
            int n = fs.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = fs.nextInt();
            }
            sb.append(solveOne(n, a)).append('\n');
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 求整数数组 arr 的最长奇回文半径（半径 = 从中心向一侧的长度，含中心）
int oddManacher(const vector<int>& arr) {
    int n = arr.size();
    vector<int> d(n);
    int l = 0, r = -1;
    int ans = 0;

    for (int i = 0; i < n; ++i) {
        int k = (i > r) ? 1 : min(d[l + r - i], r - i + 1);
        while (i - k >= 0 && i + k < n && arr[i - k] == arr[i + k]) {
            ++k;
        }
        d[i] = k;
        ans = max(ans, k);
        if (i + k - 1 > r) {
            l = i - k + 1;
            r = i + k - 1;
        }
    }
    return ans;
}

int solveOne(int n, const vector<int>& a) {
    vector<int> st;
    for (int x : a) {
        if (!st.empty() && st.back() == x) {
            st.pop_back();
        } else {
            st.push_back(x);
        }
    }

    int m = st.size();
    int base = (n - m) / 2;   // 不插入时的删除次数

    if (m == 0) return base;  // 已完全消除，插入无额外收益

    int extra = oddManacher(st);
    return base + extra;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        cout << solveOne(n, a) << '\n';
    }
    return 0;
}
```
