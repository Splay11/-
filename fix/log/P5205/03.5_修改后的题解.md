## 解题思路

设无限循环序列满足：

$$
b_i = a_{((i-1)\bmod n)+1}
$$

定义前缀余数：

$$
P_0 = 0
$$

$$
P_i = \left(P_{i-1} + b_i\right)\bmod K
$$

其中余数均为标准余数，落在 $[0,K-1]$。

对于起点 $s$，第 $t$ 步检测码为：

$$
C_t = \left(\sum_{j=0}^{t-1} b_{s+j}\right)\bmod K = \left(P_{s+t-1} - P_{s-1}\right)\bmod K
$$

因此对于两个步数 $x,y$：

$$
C_x = C_y \iff P_{s+x-1} \equiv P_{s+y-1} \pmod K
$$

由于前缀数组存储的已经是标准余数，所以等价于：

$$
P_{s+x-1} = P_{s+y-1}
$$

于是起点 $s$ 的最长有效步数 $L_s$，就是序列：

$$
P_s, P_{s+1}, \dots, P_{s+n-1}
$$

从左端开始，所有元素两两不同的最长前缀长度。

算法步骤：

1. 因为起点 $s$ 最大为 $n$，对应的最大下标为 $2n-1$，所以只需要计算前缀余数到下标 $2n-1$。
2. 使用双指针滑动窗口维护一段所有前缀余数互不相同的窗口 $[left,right]$。
3. 对于每个起点 $left$，窗口右端点最多扩展到 $left+n-1$，保证步数不超过 $n$。
4. 当 $prefix[right+1]$ 已经存在于窗口中时，不能继续扩展。
5. 此时：
   $$
   L_{left} = right-left+1
   $$
6. 将 $L_{left}$ 累加，然后删除 $prefix[left]$，继续处理下一个起点。

由于所有测试数据的 $n$ 之和不超过 $2\times 10^5$，滑动窗口右端点单调移动，整体效率可以满足要求。

## 复杂度分析

对于每组测试数据：

- 构造前缀余数数组需要 $O(n)$ 时间。
- 滑动窗口扫描左右端点总共移动 $O(n)$ 次。
- 因此每组数据时间复杂度为 $O(n)$。

所有测试数据总时间复杂度为：

$$
O\left(\sum n\right)
$$

空间复杂度：

- 需要存储长度为 $2n$ 的前缀数组，以及一个哈希集合。
- 空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys


def solve_case(n, K, a):
    # prefix[i] 表示无限循环序列前 i 项之和对 K 取模
    # 实际只需要用到下标 2n-1
    prefix = [0] * (2 * n)

    for i in range(1, 2 * n):
        prefix[i] = (prefix[i - 1] + a[(i - 1) % n]) % K

    seen = set()
    right = 0
    answer = 0

    # 枚举每个起点 s
    for left in range(1, n + 1):
        limit = left + n - 1

        # 保证窗口内前缀余数互不相同，且步数不超过 n
        while right + 1 <= limit and prefix[right + 1] not in seen:
            right += 1
            seen.add(prefix[right])

        # 当前窗口长度就是起点 left 的最长有效步数
        answer += right - left + 1

        # 左端点右移，删除离开窗口的前缀余数
        if left <= right:
            seen.remove(prefix[left])

    return answer


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0

    T = data[idx]
    idx += 1

    output = []

    for _ in range(T):
        n = data[idx]
        K = data[idx + 1]
        idx += 2

        a = data[idx:idx + n]
        idx += n

        output.append(str(solve_case(n, K, a)))

    sys.stdout.write("\n".join(output))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.HashSet;

public class Main {

    static long solveCase(int n, int K, long[] a) {
        int[] prefix = new int[2 * n];

        // 构造循环序列的前缀余数
        for (int i = 1; i < 2 * n; i++) {
            long value = a[(i - 1) % n] % K;

            // Java 取模可能为负数，需要调整到 [0, K-1]
            if (value < 0) {
                value += K;
            }

            prefix[i] = (int) ((prefix[i - 1] + value) % K);
        }

        HashSet<Integer> seen = new HashSet<>();
        int right = 0;
        long answer = 0;

        // 枚举每个起点 s
        for (int left = 1; left <= n; left++) {
            int limit = left + n - 1;

            // 保证窗口内前缀余数互不相同，且步数不超过 n
            while (right + 1 <= limit && !seen.contains(prefix[right + 1])) {
                right++;
                seen.add(prefix[right]);
            }

            // 当前窗口长度就是起点 left 的最长有效步数
            answer += right - left + 1L;

            // 左端点右移，删除离开窗口的前缀余数
            if (left <= right) {
                seen.remove(prefix[left]);
            }
        }

        return answer;
    }

    public static void main(String[] args) throws Exception {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in))
        );

        input.nextToken();
        int T = (int) input.nval;

        StringBuilder output = new StringBuilder();

        for (int tc = 0; tc < T; tc++) {
            input.nextToken();
            int n = (int) input.nval;

            input.nextToken();
            int K = (int) input.nval;

            long[] a = new long[n];

            for (int i = 0; i < n; i++) {
                input.nextToken();
                a[i] = (long) input.nval;
            }

            output.append(solveCase(n, K, a)).append('\n');
        }

        System.out.print(output);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <unordered_set>
using namespace std;

long long solveCase(int n, int K, const vector<long long>& a) {
    vector<int> prefix(2 * n, 0);

    // 构造循环序列的前缀余数
    for (int i = 1; i < 2 * n; i++) {
        long long value = a[(i - 1) % n] % K;

        // C++ 取模可能为负数，需要调整到 [0, K-1]
        if (value < 0) {
            value += K;
        }

        prefix[i] = (int) ((prefix[i - 1] + value) % K);
    }

    unordered_set<int> seen;
    seen.reserve(2 * n + 1);

    int right = 0;
    long long answer = 0;

    // 枚举每个起点 s
    for (int left = 1; left <= n; left++) {
        int limit = left + n - 1;

        // 保证窗口内前缀余数互不相同，且步数不超过 n
        while (right + 1 <= limit && seen.find(prefix[right + 1]) == seen.end()) {
            right++;
            seen.insert(prefix[right]);
        }

        // 当前窗口长度就是起点 left 的最长有效步数
        answer += right - left + 1LL;

        // 左端点右移，删除离开窗口的前缀余数
        if (left <= right) {
            seen.erase(prefix[left]);
        }
    }

    return answer;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, K;
        cin >> n >> K;

        vector<long long> a(n);

        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }

        cout << solveCase(n, K, a) << '\n';
    }

    return 0;
}
```
