## 解题思路

每次调度给相邻两个桩位同时加 $1$，相当于同时翻转这两个桩位的奇偶性。因此：

- 奇数个奇数桩位的奇偶性不变；偶数个偶数桩位的奇偶性也不变。
- 若奇数桩位个数与偶数桩位个数都是奇数，则既不能全变成奇数，也不能全变成偶数，直接输出 `No`。
- 否则至少有一种目标可行：奇数个数为偶数时，可以把全部变成偶数；否则把全部变成奇数。

构造方案时，从 $1$ 号桩位出发做后序遍历。处理完某个子桩位后，若它仍未达到目标奇偶性，就对“父桩位—该子桩位”这条边调度一次，并给父桩位加 $1$。子桩位之后不会再被检查，因此不必改它的存储值。调度次数不超过 $n-1$。

## 复杂度分析

时间复杂度 $O(\sum n)$，空间复杂度 $O(\sum n)$。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    q = int(input())
    for _ in range(q):
        n = int(input())
        a = list(map(int, input().split()))
        g = [[] for _ in range(n + 1)]
        for _ in range(n - 1):
            u, v = map(int, input().split())
            g[u].append(v)
            g[v].append(u)
        c = sum(i % 2 for i in a)
        if c % 2 != 0 and (n - c) % 2 != 0:
            print("No")
            continue
        print("Yes")
        ans = []
        f = 0 if c % 2 == 0 else 1

        def dfs(root, fa):
            for e in g[root]:
                if e == fa:
                    continue
                dfs(e, root)
                if a[e - 1] % 2 != f:
                    ans.append((root, e))
                    a[root - 1] += 1

        dfs(1, -1)
        print(len(ans))
        for x, y in ans:
            print(x, y)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static List<Integer>[] g;
    static int[] a;
    static int f;
    static List<int[]> ans;

    static void dfs(int root, int fa) {
        for (int e : g[root]) {
            if (e == fa) continue;
            dfs(e, root);
            if (a[e] % 2 != f) {
                ans.add(new int[] {root, e});
                a[root]++;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            int n = sc.nextInt();
            a = new int[n + 1];
            int c = 0;
            for (int i = 1; i <= n; i++) {
                a[i] = sc.nextInt();
                c += a[i] % 2;
            }
            g = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt(), v = sc.nextInt();
                g[u].add(v);
                g[v].add(u);
            }
            if (c % 2 != 0 && (n - c) % 2 != 0) {
                System.out.println("No");
                continue;
            }
            System.out.println("Yes");
            ans = new ArrayList<>();
            f = (c % 2 == 0) ? 0 : 1;
            dfs(1, -1);
            System.out.println(ans.size());
            for (int[] e : ans) {
                System.out.println(e[0] + " " + e[1]);
            }
        }
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<int> a(n + 1);
        int c = 0;
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
            c += a[i] % 2;
        }
        vector<vector<int>> g(n + 1);
        for (int i = 0; i < n - 1; i++) {
            int u, v;
            cin >> u >> v;
            g[u].push_back(v);
            g[v].push_back(u);
        }
        if (c % 2 != 0 && (n - c) % 2 != 0) {
            cout << "No\n";
            continue;
        }
        cout << "Yes\n";
        int f = (c % 2 == 0) ? 0 : 1;
        vector<pair<int, int>> ans;
        function<void(int, int)> dfs = [&](int root, int fa) {
            for (int e : g[root]) {
                if (e == fa) continue;
                dfs(e, root);
                if (a[e] % 2 != f) {
                    ans.push_back({root, e});
                    a[root]++;
                }
            }
        };
        dfs(1, -1);
        cout << ans.size() << '\n';
        for (auto [x, y] : ans) cout << x << ' ' << y << '\n';
    }
    return 0;
}
```
