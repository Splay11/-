## 题目思路

思维题。

将所有节点尝试都转化为全奇或者全偶

转化为全奇时，我们只需要从根节点开始遍历一遍树，在回溯时，如果子节点不是奇数，那么就操作该条边一次，将子节点变成奇数。最终，如果能够满足奇偶相同，那么根节点一定是奇数，否则就不能满足。
转化为全偶时同理

```c
#include <bits/stdc++.h>
using namespace std;

const int maxn=1e5+10;
int n;
int cnt=0;
int a[maxn];
// 链式前向星存图 
int head[maxn],ver[maxn<<1],Next[maxn<<1],tot=1;

struct node{
	int l,r;
} res[maxn];

void add(int x,int y){
	ver[++tot]=y;Next[tot]=head[x];head[x]=tot;
}

void dfs(int x,int fa,int val){
	for(int i=head[x],y;i;i=Next[i]){
		y=ver[i];
		if(y==fa) continue;
		dfs(y,x,val);
		if(a[y]%2!=val){
			cnt++;
			res[cnt]=(node){x,y};
			a[x]++;
			a[y]++;
		}
	}
}

int main() {
	int q;
	scanf("%d",&q);
	while(q--){
		cnt=0;
		tot=1;
		memset(head,0,sizeof(head));
		scanf("%d",&n);
		for(int i=1;i<=n;++i){
			scanf("%d",a+i);
		}
		for(int i=1,u,v;i<n;++i){
			scanf("%d %d",&u,&v);
			add(u,v);
			add(v,u);
		}
		// 尝试全奇 
		dfs(1,-1,1);
		if(a[1]%2==0){
			for(int i=1;i<=cnt;++i){
				a[res[i].l]--;
				a[res[i].r]--;
			}
			cnt=0;
			// 尝试全偶 
			dfs(1,-1,0);
			if(a[1]%2==1){
				puts("No");
			}else{
				puts("Yes");
				printf("%d\n",cnt);
				for(int i=1;i<=cnt;++i){
					printf("%d %d\n",res[i].l,res[i].r);
				}
			}
		}else{
			puts("Yes");
			printf("%d\n",cnt);
			for(int i=1;i<=cnt;++i){
				printf("%d %d\n",res[i].l,res[i].r);
			}
		}
	}
	
    return 0;
}
```

```Java
import java.util.*;

class Main {
    static class Node {
        int l, r;

        Node(int l, int r) {
            this.l = l;
            this.r = r;
        }
    }

    static final int maxn = 100010;
    static int n;
    static int cnt = 0;
    static int[] a = new int[maxn];
    // 链式前向星存图
    static int[] head = new int[maxn];
    static int[] ver = new int[maxn << 1];
    static int[] Next = new int[maxn << 1];
    static int tot = 1;
    static Node[] res = new Node[maxn];

    static void add(int x, int y) {
        ver[++tot] = y;
        Next[tot] = head[x];
        head[x] = tot;
    }

    static void dfs(int x, int fa, int val) {
        for (int i = head[x], y; i != 0; i = Next[i]) {
            y = ver[i];
            if (y == fa)
                continue;
            dfs(y, x, val);
            if (a[y] % 2 != val) {
                cnt++;
                res[cnt] = new Node(x, y);
                a[x]++;
                a[y]++;
            }
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        while (q-- > 0) {
            cnt = 0;
            tot = 1;
            Arrays.fill(head, 0);
            n = scanner.nextInt();
            for (int i = 1; i <= n; ++i) {
                a[i] = scanner.nextInt();
            }
            for (int i = 1, u, v; i < n; ++i) {
                u = scanner.nextInt();
                v = scanner.nextInt();
                add(u, v);
                add(v, u);
            }
            // 尝试全奇
            dfs(1, -1, 1);
            if (a[1] % 2 == 0) {
                for (int i = 1; i <= cnt; ++i) {
                    a[res[i].l]--;
                    a[res[i].r]--;
                }
                cnt = 0;
                // 尝试全偶
                dfs(1, -1, 0);
                if (a[1] % 2 == 1) {
                    System.out.println("No");
                } else {
                    System.out.println("Yes");
                    System.out.println(cnt);
                    for (int i = 1; i <= cnt; ++i) {
                        System.out.println(res[i].l + " " + res[i].r);
                    }
                }
            } else {
                System.out.println("Yes");
                System.out.println(cnt);
                for (int i = 1; i <= cnt; ++i) {
                    System.out.println(res[i].l + " " + res[i].r);
                }
            }
        }
        scanner.close();
    }
}
```

```python
def solve():
    q = int(input())
    for _ in range(q):
        n = int(input())
        a = list(map(int, input().split()))
        g = [[] for _ in range(n + 1)]
        for _ in range(n - 1):
            u, v = map(int, input().split())
            g[u].append(v)
            g[v].append(u)
        c = sum(i % 2 for i in a)
        if c % 2 != 0 and (n - c) % 2 != 0:
            print("No")
            continue
        print("Yes")
        ans = list()
        f = 0
        if c % 2 == 0:
            f = 0
        else:
            f = 1
        def dfs(root: int, fa: int) -> None:
            for e in g[root]:
                if fa == e:
                    continue
                dfs(e, root)
                if a[e - 1] % 2 != f:
                    ans.append((root, e))
                    a[root - 1] += 1
        dfs(1, -1)
        print(len(ans))
        for x, y in ans:
            print(x, y)



if __name__ == "__main__":
    solve()
```
