## 解题思路

询问 $x$ 要求把所有满足 $(x\ \&\ i)=i$ 的展位金额求和，即展位编号 $i$ 是 $x$ 的按位子集。

$n,x\le 2\times 10^5<2^{18}$，把 $18$ 位拆成低 $10$ 位与高 $8$ 位。对每个高位组 $h$ 开长度为 $1024$ 的数组，把展位 $i$ 的金额放入 $(h_i,l_i)$，再对低位做子集和。处理后 `groups[h][mask]` 表示高位为 $h$、低位为 `mask` 子集的金额和。

回答 $x=(h_x,l_x)$ 时，枚举 $h_x$ 的所有子集 $s$，累加 `groups[s][l_x]`。

## 复杂度分析

设实际出现的高位组数为 $g$（$g\le 256$）。每组数据时间复杂度

$$
O(n + g\times 10\times 1024 + q\times 256)
$$

空间复杂度 $O(g\times 1024)$。

## 代码实现

### Python

```python
import sys


# 计算一组测试数据所有询问的答案
def solve_case(n, q, a, queries):
    LOW_BITS = 10
    LOW_SIZE = 1 << LOW_BITS
    LOW_MASK = LOW_SIZE - 1
    HIGH_SIZE = 1 << (18 - LOW_BITS)

    # groups[h] 是一个长度为 1024 的数组
    # groups[h][l] 表示高位为 h、低位恰好为 l 的权值和
    groups = [None] * HIGH_SIZE

    # 按下标 i 的二进制高低位拆分，把 a_i 放入对应位置
    for i in range(1, n + 1):
        h = i >> LOW_BITS
        l = i & LOW_MASK
        if groups[h] is None:
            groups[h] = [0] * LOW_SIZE
        groups[h][l] += a[i - 1]

    # 对每个出现过的高位组做低位上的子集和预处理
    # 处理后 groups[h][mask] 表示低位是 mask 的子集的所有贡献和
    for h in range(HIGH_SIZE):
        if groups[h] is None:
            continue
        dp = groups[h]
        for b in range(LOW_BITS):
            bit = 1 << b
            for mask in range(LOW_SIZE):
                if mask & bit:
                    dp[mask] += dp[mask ^ bit]

    ans = []

    # 回答每个询问
    for x in queries:
        hx = x >> LOW_BITS
        lx = x & LOW_MASK

        total = 0

        # 枚举 hx 的所有子集 s
        s = hx
        while True:
            if groups[s] is not None:
                total += groups[s][lx]
            if s == 0:
                break
            s = (s - 1) & hx

        ans.append(str(total))

    return ans


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    idx = 1
    out = []

    for _ in range(t):
        n = data[idx]
        q = data[idx + 1]
        idx += 2

        a = data[idx: idx + n]
        idx += n

        queries = data[idx: idx + q]
        idx += q

        out.extend(solve_case(n, q, a, queries))

    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
```

```java
import java.io.BufferedInputStream;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

public class Main {

    // 计算一组测试数据所有询问的答案
    public static List<Long> solveCase(int n, int q, long[] a, int[] queries) {
        final int LOW_BITS = 10;
        final int LOW_SIZE = 1 << LOW_BITS;
        final int LOW_MASK = LOW_SIZE - 1;
        final int HIGH_SIZE = 1 << (18 - LOW_BITS);

        // groups[h] 是长度为 1024 的数组
        // groups[h][l] 表示高位为 h、低位恰好为 l 的权值和
        long[][] groups = new long[HIGH_SIZE][];

        // 按下标 i 拆分高低位，把 a_i 放入对应位置
        for (int i = 1; i <= n; i++) {
            int h = i >> LOW_BITS;
            int l = i & LOW_MASK;
            if (groups[h] == null) {
                groups[h] = new long[LOW_SIZE];
            }
            groups[h][l] += a[i];
        }

        // 对每个出现过的高位组做低位上的子集和预处理
        for (int h = 0; h < HIGH_SIZE; h++) {
            if (groups[h] == null) {
                continue;
            }
            long[] dp = groups[h];
            for (int b = 0; b < LOW_BITS; b++) {
                int bit = 1 << b;
                for (int mask = 0; mask < LOW_SIZE; mask++) {
                    if ((mask & bit) != 0) {
                        dp[mask] += dp[mask ^ bit];
                    }
                }
            }
        }

        List<Long> ans = new ArrayList<>();

        // 回答每个询问
        for (int x : queries) {
            int hx = x >> LOW_BITS;
            int lx = x & LOW_MASK;

            long total = 0;

            // 枚举 hx 的所有子集
            int s = hx;
            while (true) {
                if (groups[s] != null) {
                    total += groups[s][lx];
                }
                if (s == 0) {
                    break;
                }
                s = (s - 1) & hx;
            }

            ans.add(total);
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();

        int T = fs.nextInt();
        StringBuilder sb = new StringBuilder();

        for (int tc = 0; tc < T; tc++) {
            int n = fs.nextInt();
            int q = fs.nextInt();

            // a 从 1 开始存，方便和题目下标一致
            long[] a = new long[n + 1];
            for (int i = 1; i <= n; i++) {
                a[i] = fs.nextLong();
            }

            int[] queries = new int[q];
            for (int i = 0; i < q; i++) {
                queries[i] = fs.nextInt();
            }

            List<Long> ans = solveCase(n, q, a, queries);
            for (long v : ans) {
                sb.append(v).append('\n');
            }
        }

        System.out.print(sb.toString());
    }

    // 读入工具，使用 BufferedInputStream，适合本题数据范围
    static class FastScanner {
        private final BufferedInputStream in = new BufferedInputStream(System.in);
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0;
        private int len = 0;

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) {
                    return -1;
                }
            }
            return buffer[ptr++];
        }

        public long nextLong() throws IOException {
            int c;
            do {
                c = read();
            } while (c <= ' ');

            int sign = 1;
            if (c == '-') {
                sign = -1;
                c = read();
            }

            long val = 0;
            while (c > ' ') {
                val = val * 10 + c - '0';
                c = read();
            }
            return val * sign;
        }

        public int nextInt() throws IOException {
            return (int) nextLong();
        }
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;


// 计算一组测试数据所有询问的答案
vector<long long> solve_case(int n, int q, const vector<long long>& a, const vector<int>& queries) {
    const int LOW_BITS = 10;
    const int LOW_SIZE = 1 << LOW_BITS;
    const int LOW_MASK = LOW_SIZE - 1;
    const int HIGH_SIZE = 1 << (18 - LOW_BITS);

    // groups[h] 是一个长度为 1024 的数组
    // groups[h][l] 表示高位为 h、低位恰好为 l 的权值和
    vector<vector<long long>> groups(HIGH_SIZE);

    // 按下标 i 拆分高低位，把 a_i 放入对应位置
    for (int i = 1; i <= n; i++) {
        int h = i >> LOW_BITS;
        int l = i & LOW_MASK;
        if (groups[h].empty()) {
            groups[h].assign(LOW_SIZE, 0);
        }
        groups[h][l] += a[i];
    }

    // 对每个出现过的高位组做低位上的子集和预处理
    for (int h = 0; h < HIGH_SIZE; h++) {
        if (groups[h].empty()) {
            continue;
        }
        for (int b = 0; b < LOW_BITS; b++) {
            int bit = 1 << b;
            for (int mask = 0; mask < LOW_SIZE; mask++) {
                if (mask & bit) {
                    groups[h][mask] += groups[h][mask ^ bit];
                }
            }
        }
    }

    vector<long long> ans;
    ans.reserve(q);

    // 回答每个询问
    for (int x : queries) {
        int hx = x >> LOW_BITS;
        int lx = x & LOW_MASK;

        long long total = 0;

        // 枚举 hx 的所有子集
        int s = hx;
        while (true) {
            if (!groups[s].empty()) {
                total += groups[s][lx];
            }
            if (s == 0) {
                break;
            }
            s = (s - 1) & hx;
        }

        ans.push_back(total);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, q;
        cin >> n >> q;

        // a 从 1 开始存，方便和题目下标一致
        vector<long long> a(n + 1);
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
        }

        vector<int> queries(q);
        for (int i = 0; i < q; i++) {
            cin >> queries[i];
        }

        vector<long long> ans = solve_case(n, q, a, queries);
        for (long long v : ans) {
            cout << v << '\n';
        }
    }

    return 0;
}
```
