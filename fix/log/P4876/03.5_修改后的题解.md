## 解题思路

本题需要判断每个位置 $i$ 是否存在某个 $j > i$ 使得 $a_i \oplus a_j \ge M$，其中 $M$ 是序列的最大值。这是一个典型的“对每个元素查询其后缀中与其异或能得到的最大值”问题，可以使用**二进制字典树（Trie）**高效解决。

核心思路分为以下步骤：

1. **求基准值**：先遍历整个序列，找到最大值 $M$。后续所有判断都以 $M$ 为阈值。
2. **反向遍历 + 字典树维护后缀**：  
   从右向左遍历序列。在处理位置 $i$ 时，字典树中已经存储了所有 $j > i$ 的元素。  
   - 对于当前元素 $a_i$，在字典树中查询能与其异或得到的**最大可能值** $\text{best} = \max_{j > i} (a_i \oplus a_j)$。
   - 如果 $\text{best} \ge M$，则位置 $i$ 是优质位置，答案字符为 `'1'`；否则为 `'0'`。
   - 查询结束后，将 $a_i$ 插入字典树，供左侧位置使用。
3. **字典树的构建与查询**：  
   因为 $a_i \le 10^9 < 2^{30}$，所以我们用 $30$ 到 $0$ 共 $31$ 个二进制位表示每个数。  
   - 插入：从根节点开始，依次按照当前二进制位（从高位到低位）决定走 0 分枝或 1 分枝，若节点不存在则新建。
   - 查询最大异或值：从根节点开始，对于每一位，**优先选择与当前位相反的二进制位**（使得异或结果为 $1$），如果相反位不存在则走相同位。按位累计结果即可得到最大异或值。

由于每个元素插入和查询各需遍历 $31$ 个二进制位，总复杂度足够处理 $n \le 2 \times 10^5$ 的规模。

## 复杂度分析

- **时间复杂度**：$O(n \log C)$，其中 $C$ 是数值上限，本题 $C \le 10^9$，$\log C \le 31$。每个数插入和查询最多 $31$ 次位操作，总时间复杂度可以看作 $O(31n)$。
- **空间复杂度**：$O(n \log C)$，字典树最多有 $n \times 31$ 个节点，每个节点只需存储两个子节点指针，空间开销与 $n$ 成线性关系。

## 代码实现

### Python

```python
import sys


def solve_array(a):
    n = len(a)
    M = max(a)  # 基准值

    # 字典树：ch0[node] 和 ch1[node] 分别表示 0/1 子节点，-1 表示不存在
    ch0 = [-1]
    ch1 = [-1]

    def insert(x):
        """将数值 x 插入二进制字典树"""
        node = 0
        for b in range(30, -1, -1):
            bit = (x >> b) & 1
            if bit == 0:
                if ch0[node] == -1:
                    ch0[node] = len(ch0)
                    ch0.append(-1)
                    ch1.append(-1)
                node = ch0[node]
            else:
                if ch1[node] == -1:
                    ch1[node] = len(ch1)
                    ch0.append(-1)
                    ch1.append(-1)
                node = ch1[node]

    def query_max_xor(x):
        """查询与 x 异或能得到的最大值"""
        node = 0
        res = 0
        for b in range(30, -1, -1):
            bit = (x >> b) & 1
            # 优先走相反位，使当前位异或结果为 1
            if bit == 0:
                if ch1[node] != -1:
                    res |= 1 << b
                    node = ch1[node]
                else:
                    node = ch0[node]
            else:
                if ch0[node] != -1:
                    res |= 1 << b
                    node = ch0[node]
                else:
                    node = ch1[node]
        return res

    ans = ['0'] * n

    # 从右向左遍历，维护后缀字典树
    for i in range(n - 1, -1, -1):
        if i < n - 1:  # 不是最后一个元素时，字典树中已有右侧元素
            best = query_max_xor(a[i])
            if best >= M:
                ans[i] = '1'

        # 将当前元素插入字典树，供左侧位置使用
        insert(a[i])

    return ''.join(ans)


def main():
    input = sys.stdin.readline
    n = int(input())
    a = list(map(int, input().split()))
    print(solve_array(a))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static int[][] child;
    static int nodeCnt;

    // 将数 x 插入二进制字典树
    static void insert(int x) {
        int node = 0;
        for (int b = 30; b >= 0; b--) {
            int bit = (x >> b) & 1;
            if (child[node][bit] == 0) {
                child[node][bit] = nodeCnt++;
            }
            node = child[node][bit];
        }
    }

    // 查询与 x 异或能得到的最大值
    static int queryMaxXor(int x) {
        int node = 0;
        int res = 0;
        for (int b = 30; b >= 0; b--) {
            int bit = (x >> b) & 1;
            int want = bit ^ 1;           // 相反的位
            // 优先走相反位，使当前位异或结果为 1
            if (child[node][want] != 0) {
                res |= 1 << b;
                node = child[node][want];
            } else {
                node = child[node][bit];
            }
        }
        return res;
    }

    static String solveArray(int[] a) {
        int n = a.length;
        int M = 0;  // 基准值
        for (int x : a) {
            M = Math.max(M, x);
        }

        // 最多 n * 31 个节点，0 号节点为根
        child = new int[n * 31 + 5][2];
        nodeCnt = 1;

        char[] ans = new char[n];
        Arrays.fill(ans, '0');

        // 从右向左遍历，维护后缀字典树
        for (int i = n - 1; i >= 0; i--) {
            if (i < n - 1) {
                int best = queryMaxXor(a[i]);
                if (best >= M) {
                    ans[i] = '1';
                }
            }
            insert(a[i]);
        }

        return new String(ans);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        System.out.println(solveArray(a));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<array<int, 2>> child;

// 将数 x 插入二进制字典树
void insertNumber(int x) {
    int node = 0;
    for (int b = 30; b >= 0; b--) {
        int bit = (x >> b) & 1;
        if (child[node][bit] == -1) {
            child[node][bit] = (int)child.size();
            child.push_back({-1, -1});
        }
        node = child[node][bit];
    }
}

// 查询与 x 异或能得到的最大值
int queryMaxXor(int x) {
    int node = 0;
    int res = 0;
    for (int b = 30; b >= 0; b--) {
        int bit = (x >> b) & 1;
        int want = bit ^ 1;          // 相反的位
        // 优先走相反位，使当前位异或结果为 1
        if (child[node][want] != -1) {
            res |= 1 << b;
            node = child[node][want];
        } else {
            node = child[node][bit];
        }
    }
    return res;
}

string solveArray(vector<int>& a) {
    int n = (int)a.size();
    int M = *max_element(a.begin(), a.end());  // 基准值

    child.clear();
    child.push_back({-1, -1});  // 根节点

    string ans(n, '0');

    // 从右向左遍历，维护后缀字典树
    for (int i = n - 1; i >= 0; i--) {
        if (i < n - 1) {
            int best = queryMaxXor(a[i]);
            if (best >= M) {
                ans[i] = '1';
            }
        }
        insertNumber(a[i]);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    cout << solveArray(a) << '\n';
    return 0;
}
```
