## 解题思路

设样本读数序列为 $a_1,\dots,a_n$。对每个 $k$（$1\le k\le n$），求所有长度为 $k$ 的子序列 $b$ 的峰值 $f(b)=\max(b)$ 之和，答案对 $998244353$ 取模。

将读数按非降序排序，记为 $A_1\le A_2\le \cdots\le A_n$。把每个长度为 $k$ 的子序列唯一地归属给其最大值在排序后的最右位置 $i$。当峰值归属在位置 $i$ 时，其余 $k-1$ 个元素只能从 $[1,i-1]$ 中任取（都不超过 $A_i$），共有

$$
\binom{i-1}{k-1}
$$

种选择。因此

$$
\text{ans}_k=\sum_{i=k}^{n} A_i \cdot \binom{i-1}{k-1}.
$$

相等读数同样成立，因为“最右位置”保证划分唯一。

实现时先排序。组合数用杨辉三角逐行递推：$\binom{i}{0}=\binom{i}{i}=1$，$\binom{i}{j}=\binom{i-1}{j-1}+\binom{i-1}{j}$（取模）。只保留上一行与当前行，空间为 $O(n)$。枚举位置 $i$ 时，用上一行的 $\binom{i-1}{k-1}$ 把 $A_i$ 累加到所有 $k\le i$ 的答案上。

## 复杂度分析

排序 $O(n\log n)$，组合数递推与累加总计 $O(n^2)$。当 $n\le 5\times 10^3$ 时可行。

- 时间复杂度：$O(n^2)$。
- 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
import sys

MOD = 998244353

def solve_all(a):
    n = len(a)
    a.sort()
    ans = [0] * (n + 1)
    prev = [0] * (n + 1)
    prev[0] = 1
    for i in range(1, n + 1):
        ai = a[i - 1] % MOD
        for k in range(1, i + 1):
            ans[k] = (ans[k] + ai * prev[k - 1]) % MOD
        cur = [0] * (n + 1)
        cur[0] = 1
        for j in range(1, i):
            cur[j] = (prev[j - 1] + prev[j]) % MOD
        cur[i] = 1
        prev = cur
    return ans[1:]

def main():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    n = int(next(it))
    arr = [int(next(it)) for _ in range(n)]
    res = solve_all(arr)
    print(" ".join(str(x) for x in res))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
    static final long MOD = 998244353L;

    static long[] solveAll(long[] a) {
        int n = a.length;
        Arrays.sort(a);
        long[] ans = new long[n + 1];
        long[] prev = new long[n + 1];
        prev[0] = 1;
        for (int i = 1; i <= n; i++) {
            long ai = a[i - 1] % MOD;
            for (int k = 1; k <= i; k++) {
                ans[k] = (ans[k] + ai * prev[k - 1]) % MOD;
            }
            long[] cur = new long[n + 1];
            cur[0] = 1;
            for (int j = 1; j < i; j++) {
                cur[j] = (prev[j - 1] + prev[j]) % MOD;
            }
            cur[i] = 1;
            prev = cur;
        }
        long[] ret = new long[n];
        for (int i = 1; i <= n; i++) ret[i - 1] = ans[i];
        return ret;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        long[] a = new long[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) a[i] = Long.parseLong(st.nextToken());
        long[] res = solveAll(a);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < res.length; i++) {
            if (i > 0) sb.append(' ');
            sb.append(res[i]);
        }
        System.out.println(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353LL;

vector<long long> solveAll(vector<long long> a) {
    int n = (int)a.size();
    sort(a.begin(), a.end());
    vector<long long> ans(n + 1, 0);
    vector<long long> prev(n + 1, 0);
    prev[0] = 1;
    for (int i = 1; i <= n; ++i) {
        long long ai = a[i - 1] % MOD;
        for (int k = 1; k <= i; ++k) {
            ans[k] = (ans[k] + ai * prev[k - 1]) % MOD;
        }
        vector<long long> cur(n + 1, 0);
        cur[0] = 1;
        for (int j = 1; j < i; ++j) {
            cur[j] = (prev[j - 1] + prev[j]) % MOD;
        }
        cur[i] = 1;
        prev.swap(cur);
    }
    ans.erase(ans.begin());
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    if (!(cin >> n)) return 0;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    vector<long long> res = solveAll(a);
    for (int i = 0; i < (int)res.size(); ++i) {
        if (i) cout << ' ';
        cout << res[i];
    }
    cout << '\n';
    return 0;
}
```
