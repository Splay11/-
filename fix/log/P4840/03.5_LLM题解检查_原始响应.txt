## 解题思路

我们要统计所有配对 $(i, j)$ 中满足 $x_i \times y_j \ge K$ 的个数。两个数列均为非负整数，我们可以利用排序和二分查找高效计算。

1. 将传感器 $Y$ 的读数数组 $y$ **升序排序**。
2. 遍历传感器 $X$ 的每个读数 $x_i$，根据不同情况累加合格配对个数：
   - 若 $K = 0$，任意非负乘积均 $\ge 0$，所有 $m$ 个 $y_j$ 都合格，贡献 $m$。
   - 若 $K > 0$ 且 $x_i = 0$，乘积必为 $0 < K$，无合格配对，跳过。
   - 否则（$x_i > 0$），需要 $y_j \ge \lceil K / x_i \rceil$。计算阈值 `need = (K + x_i - 1) // x_i`（向上取整）。在排序后的 $y$ 中用**二分查找**找到第一个 `>= need` 的位置 $pos$，则后缀 $y[pos \dots m-1]$ 全部合格，贡献 $m - pos$。
3. 将所有 $x_i$ 的贡献累加，即为该测试用例的答案。

这样只需对 $y$ 排序一次，然后每个 $x_i$ 进行一次二分查找，时间复杂度很低，可以应对题目中所有测试数据的总规模。

## 复杂度分析

- **时间复杂度**：排序 $y$ 需 $O(m \log m)$，再对 $n$ 个 $x_i$ 各做一次二分查找，每次 $O(\log m)$，因此总时间复杂度为 $O((n + m) \log m)$。所有测试用例的 $n + m$ 总和不超过 $2 \times 10^5$，足够高效。
- **空间复杂度**：除了存储输入数据外，只使用了常数级别的额外空间，即 $O(1)$ 额外空间（排序所需栈空间不计，若考虑则为 $O(\log m)$，但远小于题目限制）。

## 代码实现

### Python

```python
import sys
from bisect import bisect_left

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0
    T = data[idx]; idx += 1
    out_lines = []
    for _ in range(T):
        n = data[idx]; m = data[idx+1]; K = data[idx+2]
        idx += 3
        x = data[idx:idx+n]; idx += n
        y = data[idx:idx+m]; idx += m
        
        y.sort()
        ans = 0
        for xi in x:
            if K == 0:
                ans += m
            elif xi == 0:
                continue
            else:
                need = (K + xi - 1) // xi   # 向上取整
                pos = bisect_left(y, need)
                ans += m - pos
        out_lines.append(str(ans))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static long countPairs(int[] x, int[] y, long K) {
        Arrays.sort(y);
        int m = y.length;
        long ans = 0;
        for (int xi : x) {
            if (K == 0) {
                ans += m;
            } else if (xi == 0) {
                continue;
            } else {
                long need = (K + xi - 1) / xi;   // 向上取整
                int pos = lowerBound(y, need);
                ans += m - pos;
            }
        }
        return ans;
    }

    static int lowerBound(int[] arr, long target) {
        int l = 0, r = arr.length;
        while (l < r) {
            int mid = (l + r) >>> 1;
            if (arr[mid] >= target) r = mid;
            else l = mid + 1;
        }
        return l;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        int T = fs.nextInt();
        StringBuilder sb = new StringBuilder();
        for (int tc = 0; tc < T; tc++) {
            int n = fs.nextInt();
            int m = fs.nextInt();
            long K = fs.nextLong();
            int[] x = new int[n];
            int[] y = new int[m];
            for (int i = 0; i < n; i++) x[i] = fs.nextInt();
            for (int j = 0; j < m; j++) y[j] = fs.nextInt();
            sb.append(countPairs(x, y, K)).append('\n');
        }
        System.out.print(sb);
    }

    static class FastScanner {
        private final InputStream in;
        private final byte[] buf = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buf);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buf[ptr++];
        }
        long nextLong() throws IOException {
            int c;
            do { c = read(); } while (c <= ' ');
            long x = 0;
            while (c > ' ') {
                x = x * 10 + (c - '0');
                c = read();
            }
            return x;
        }
        int nextInt() throws IOException {
            return (int) nextLong();
        }
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

long long countPairs(vector<int>& x, vector<int>& y, long long K) {
    sort(y.begin(), y.end());
    int m = y.size();
    long long ans = 0;
    for (int xi : x) {
        if (K == 0) {
            ans += m;
        } else if (xi == 0) {
            continue;
        } else {
            long long need = (K + xi - 1) / xi; // 向上取整
            int pos = lower_bound(y.begin(), y.end(), need) - y.begin();
            ans += m - pos;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while (T--) {
        int n, m;
        long long K;
        cin >> n >> m >> K;
        vector<int> x(n), y(m);
        for (int i = 0; i < n; i++) cin >> x[i];
        for (int j = 0; j < m; j++) cin >> y[j];
        cout << countPairs(x, y, K) << '\n';
    }

}
```
