## 解题思路

每台调音器只能一路乘 `2` 或一路除 `2`（下取整），可达刻度数量约为 $O(\log)$ 级，适合动态规划。

定义 $dp[j]$ 表示：已经处理完当前前缀，且最后一个刻度恰好为 $j$ 时的最少操作次数。为了保证序列非空前缀合法，用一个哨兵状态 $dp[1]=0$ 作为“虚拟前驱”。

处理下一个初始刻度 $val$ 时，枚举前一个状态 $(k, v)$（表示前一个刻度为 $k$、代价为 $v$）：

- 若 $val \ge k$，则不断把 $val$ 除以 `2`，得到所有 $\ge k$ 的可达值 $l$，代价为 $v$ 加上除法次数；
- 若 $val < k$，则不断把 $val$ 乘 `2` 直到 $\ge k$，只能得到这一个乘后的值，代价为 $v$ 加上乘法次数。

同一刻度 $j$ 若存在另一个 $j' \ge j$ 且代价不更优，则 $j'$ 对后续转移无用，按刻度从小到大扫描后只保留代价单调不升的状态。滚动更新即可。

最终答案为所有存活状态代价的最小值。

## 复杂度分析

每个数的可达状态数为 $O(\log A)$ 量级，转移时再扫描上一层状态。

- 时间复杂度：$O(n \cdot (\log A)^2)$ 量级（含无效状态剔除）。
- 空间复杂度：$O(\log A)$。

其中 $A \le 20000$。

## 代码实现

### Python

```python
from collections import defaultdict

n = int(input())
a = list(map(int, input().split()))

dp = defaultdict(lambda: float('inf'))
dp[1] = 0

for val in a:
    new_dp = defaultdict(lambda: float('inf'))
    arr = []
    for k, v in dp.items():
        l, r = val, val
        cnt = 0
        while l >= k:
            if new_dp[l] == float('inf'):
                new_dp[l] = v + cnt
                arr.append(l)
            new_dp[l] = min(new_dp[l], v + cnt)
            l //= 2
            cnt += 1
        cnt = 0
        while r < k:
            r *= 2
            cnt += 1
        if new_dp[r] == float('inf'):
            new_dp[r] = v + cnt
            arr.append(r)
        new_dp[r] = min(new_dp[r], v + cnt)
    arr.sort()
    dp.clear()
    minv = new_dp[arr[0]]
    dp[arr[0]] = minv
    for i in range(1, len(arr)):
        if new_dp[arr[i]] > minv:
            continue
        minv = new_dp[arr[i]]
        dp[arr[i]] = minv

print(min(dp.values()))
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();

        Map<Long, Long> dp = new HashMap<>();
        dp.put(1L, 0L);

        for (long val : a) {
            Map<Long, Long> newDp = new HashMap<>();
            List<Long> arr = new ArrayList<>();
            for (Map.Entry<Long, Long> e : dp.entrySet()) {
                long k = e.getKey(), v = e.getValue();
                long l = val, r = val, cnt = 0;
                while (l >= k) {
                    if (!newDp.containsKey(l)) {
                        newDp.put(l, v + cnt);
                        arr.add(l);
                    }
                    newDp.put(l, Math.min(newDp.get(l), v + cnt));
                    l /= 2;
                    cnt++;
                }
                cnt = 0;
                while (r < k) {
                    r *= 2;
                    cnt++;
                }
                if (!newDp.containsKey(r)) {
                    newDp.put(r, v + cnt);
                    arr.add(r);
                }
                newDp.put(r, Math.min(newDp.get(r), v + cnt));
            }
            Collections.sort(arr);
            dp.clear();
            long minv = newDp.get(arr.get(0));
            dp.put(arr.get(0), minv);
            for (int i = 1; i < arr.size(); i++) {
                if (newDp.get(arr.get(i)) > minv) continue;
                minv = newDp.get(arr.get(i));
                dp.put(arr.get(i), minv);
            }
        }

        long res = Long.MAX_VALUE;
        for (long v : dp.values()) res = Math.min(res, v);
        System.out.println(res);
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

#define int long long
signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    unordered_map<int, int> dp;
    dp[1] = 0;
    for (int val : a) {
        unordered_map<int, int> new_dp;
        vector<int> arr;
        for (auto [k, v] : dp) {
            int l = val, r = val, cnt = 0;
            while (l >= k) {
                if (!new_dp.count(l)) {
                    new_dp[l] = v + cnt;
                    arr.push_back(l);
                }
                new_dp[l] = min(new_dp[l], v + cnt);
                l /= 2;
                cnt++;
            }
            cnt = 0;
            while (r < k) {
                r *= 2;
                cnt++;
            }
            if (!new_dp.count(r)) {
                new_dp[r] = v + cnt;
                arr.push_back(r);
            }
            new_dp[r] = min(new_dp[r], v + cnt);
        }
        sort(arr.begin(), arr.end());
        unordered_map<int, int>().swap(dp);
        int minv = new_dp[arr[0]];
        dp[arr[0]] = minv;
        for (int i = 1; i < (int)arr.size(); i++) {
            if (new_dp[arr[i]] > minv) continue;
            minv = new_dp[arr[i]];
            dp[arr[i]] = minv;
        }
    }
    int res = (int)1e18;
    for (auto [k, v] : dp) res = min(res, v);
    cout << res << "\n";
    return 0;
}
```
