## 解题思路

1. 枚举所有可能的平方数，范围仅限于不超过 $10^9$ 的正整数。  
   设 $i$ 从 $1$ 到 $\lfloor\sqrt{10^9}\rfloor \approx 31622$，那么 $x = i^2$ 即为待考察的平方数。

2. 对于每一个平方数 $x$，将其转换为十进制字符串 $s$。  
   枚举所有可能的分割位置 $\text{cut}$（$1 \le \text{cut} < \text{len}(s)$），将字符串分成左右两个非空子串：
   - 左片段 $s[0:\text{cut}]$ 对应的整数记作 $a$；
   - 右片段 $s[\text{cut}:]$ 对应的整数记作 $b$。

3. 验证左右片段是否合法：
   - 两个片段均必须是无前导零的十进制表示。唯一的例外是片段为单个字符 `'0'`，这是合法的。
   - 将片段转换为整数 $a$ 和 $b$，判断它们是否都是平方数。
   
   为了实现快速平方数判定，我们预处理出所有不超过 $10^9$ 的平方数并存入哈希集合（如 `set` 或 `unordered_set`）。

4. 若某个 $x$ 存在至少一个分割位置满足上述条件，则 $x$ 是一个双子平方数，将其收集到一个列表 $\text{good}$ 中。

5. 预处理完毕后，$\text{good}$ 列表中的元素是全部不超过 $10^9$ 的双子平方数。对 $\text{good}$ 进行排序（实际上生成过程已经有序）。

6. 对于每组询问，给定上界 $n$，使用二分查找（如 `upper_bound` 或 `bisect_right`）计算 $\text{good}$ 中 $\le n$ 的元素个数，即为答案。

## 复杂度分析

- 预处理时间复杂度：  
  最大平方根 $M = \lfloor\sqrt{10^9}\rfloor \approx 3\times 10^4$。  
  枚举每个平方数时需检查其所有分割位置，每个数字最多 $10$ 个分割点，每次分割进行哈希查询，因此预处理复杂度为 $O(M \cdot \log_{10} (10^9)) = O(M)$。

- 单次询问时间复杂度：  
  在双子平方数列表上二分查找，时间复杂度为 $O(\log K)$，其中 $K$ 为双子平方数的个数（远小于 $M$，实际不足 $300$ 个）。

- 空间复杂度：  
  需要存储 $O(M)$ 个平方数在哈希集合中，以及 $O(K)$ 的双子平方数列表，整体空间复杂度为 $O(M)$。

## 代码实现

### Python

```python
import sys
from bisect import bisect_right

def valid_part(part: str) -> bool:
    """片段无前导零，除非本身是单个 '0'"""
    if len(part) == 1:
        return True
    return part[0] != '0'

def build_good_numbers(limit: int) -> list:
    max_root = int(limit ** 0.5)
    
    # 所有平方数集合，用于 O(1) 判断
    square_set = {i * i for i in range(max_root + 1)}
    
    good = []
    for i in range(1, max_root + 1):
        x = i * i
        s = str(x)
        m = len(s)
        
        # 枚举所有分割点
        for cut in range(1, m):
            left = s[:cut]
            right = s[cut:]
            
            if not valid_part(left) or not valid_part(right):
                continue
            
            a = int(left)
            b = int(right)
            if a in square_set and b in square_set:
                good.append(x)
                break  # 只要存在一个合法分割即可
                
    return good

def count_le(good: list, n: int) -> int:
    return bisect_right(good, n)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    T = int(input_data[0])
    queries = [int(v) for v in input_data[1:1+T]]
    
    # 最大上界 10^9
    good = build_good_numbers(10**9)
    
    out_lines = []
    for n in queries:
        out_lines.append(str(count_le(good, n)))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 判断片段无前导零（单个字符 "0" 合法）
    static boolean validPart(String part) {
        if (part.length() == 1) return true;
        return part.charAt(0) != '0';
    }

    static List<Integer> buildGoodNumbers(int limit) {
        int maxRoot = (int) Math.sqrt(limit);
        
        // 所有平方数集合
        Set<Integer> squareSet = new HashSet<>();
        for (int i = 0; i <= maxRoot; i++) {
            squareSet.add(i * i);
        }
        
        List<Integer> good = new ArrayList<>();
        for (int i = 1; i <= maxRoot; i++) {
            int x = i * i;
            String s = String.valueOf(x);
            int m = s.length();
            
            for (int cut = 1; cut < m; cut++) {
                String left = s.substring(0, cut);
                String right = s.substring(cut);
                
                if (!validPart(left) || !validPart(right)) continue;
                
                int a = Integer.parseInt(left);
                int b = Integer.parseInt(right);
                if (squareSet.contains(a) && squareSet.contains(b)) {
                    good.add(x);
                    break;
                }
            }
        }
        return good;
    }

    // 二分查找 <= n 的元素个数
    static int countLe(List<Integer> good, int n) {
        int left = 0, right = good.size();
        while (left < right) {
            int mid = left + (right - left) / 2;
            if (good.get(mid) <= n) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }
        return left;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        
        List<Integer> good = buildGoodNumbers(1_000_000_000);
        
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < T; i++) {
            int n = Integer.parseInt(br.readLine().trim());
            sb.append(countLe(good, n)).append('\n');
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <unordered_set>
#include <algorithm>
#include <cmath>

using namespace std;

// 片段无前导零（单个字符 "0" 合法）
bool validPart(const string& part) {
    if (part.size() == 1) return true;
    return part[0] != '0';
}

vector<int> buildGoodNumbers(int limit) {
    int maxRoot = (int) sqrt(limit);
    
    // 所有平方数集合
    unordered_set<int> squareSet;
    for (int i = 0; i <= maxRoot; i++) {
        squareSet.insert(i * i);
    }
    
    vector<int> good;
    for (int i = 1; i <= maxRoot; i++) {
        int x = i * i;
        string s = to_string(x);
        int m = (int) s.size();
        
        for (int cut = 1; cut < m; cut++) {
            string left = s.substr(0, cut);
            string right = s.substr(cut);
            
            if (!validPart(left) || !validPart(right)) continue;
            
            int a = stoi(left);
            int b = stoi(right);
            if (squareSet.count(a) && squareSet.count(b)) {
                good.push_back(x);
                break;
            }
        }
    }
    return good;
}

// 利用 upper_bound 得到 <= n 的个数
int countLe(const vector<int>& good, int n) {
    return upper_bound(good.begin(), good.end(), n) - good.begin();
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int T;
    cin >> T;
    
    vector<int> good = buildGoodNumbers(1000000000);
    
    while (T--) {
        int n;
        cin >> n;
        cout << countLe(good, n) << '\n';
    }
    
    return 0;
}
```
