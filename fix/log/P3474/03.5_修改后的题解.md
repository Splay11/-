## 解题思路

1. **读取与解析输入**
   - 从标准输入读取完整的 JSON 字符串（可能跨越多行）。
   - 解析为顶层对象，其键为传感器节点的唯一标识（字符串），值为一个内层对象。内层对象的键是信号类型名称（字符串），值为该类型的接收次数（非负整数）。

2. **计算分布复杂度（信息熵）**
   - 对于每个传感器节点，遍历其内层对象的全部信号类型：
     - 计算总接收次数 $C = \sum c_i$。
     - 若 $C = 0$，则该节点的分布复杂度直接定义为 $0$。
     - 若 $C > 0$，对于每个 $c_i > 0$ 的类型，计算频率 $p_i = c_i / C$。
     - 按公式累加熵值：$H = -\sum_{i, c_i > 0} p_i \log_2 p_i$。约定 $\log_2$ 的底为 $2$，忽略 $c_i = 0$ 的项。
   - 最终将结果 $H$ 使用四舍五入保留三位小数。

3. **构造输出**
   - 对每个节点，生成一个字典 `{"entropy": 保留三位小数的值}`。
   - 将所有节点与对应字典组装成一个 JSON 对象，保持节点出现的顺序（若解析时保留插入顺序）。
   - 将结果 JSON 对象输出至标准输出，使用双引号且紧凑格式。

4. **注意事项**
   - 输入保证是合法的 JSON，且数据范围满足节点数 $\le 1000$，每个节点的信号类型数 $\le 100$，接收次数为非负整数 $\le 10^6$。
   - Python 可直接使用内置 `json` 模块；Java 和 C++ 代码中通过手动解析 JSON 以保证不依赖第三方库，解析逻辑仅处理字符串键、整数或嵌套对象，与输入格式严格对齐。

## 复杂度分析

- 时间复杂度：设节点数为 $N$，每个节点最多有 $M$ 种信号类型。遍历所有节点并累加 $c_i$、计算 $p_i \log_2 p_i$，总操作次数为 $O(N \times M)$。JSON 解析过程与输入长度线性相关，也是 $O(N \times M)$。因此总体时间复杂度为 $O(N \times M)$。
- 空间复杂度：需要存储输入的计数信息以及输出结果，额外空间为 $O(N \times M)$。此外，解析过程中的辅助空间与输入规模成正比。总体空间复杂度为 $O(N \times M)$。

## 代码实现

### Python

```python
import sys
import json
import math

def compute_entropy(signal_counts):
    """
    根据信号类型 -> 接收次数的字典，计算分布复杂度（信息熵）
    """
    total = sum(signal_counts.values())
    if total == 0:
        return 0.0

    H = 0.0
    for c in signal_counts.values():
        if c <= 0:
            continue
        p = c / total
        H -= p * math.log2(p)
    return H

def main():
    raw = sys.stdin.read().strip()
    data = json.loads(raw)          # 顶层对象，键=节点ID，值=信号类型计数字典

    result = {}
    for node_id, signal_counts in data.items():
        H = compute_entropy(signal_counts)
        # 保留三位小数，若为 0 也会显示 0.000
        result[node_id] = {"entropy": round(H, 3)}

    # 输出紧凑型 JSON，双引号
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    // 全局指针，用于解析 JSON 字符串
    private static int idx;
    private static String json;

    public static void main(String[] args) {
        // 读取全部标准输入
        Scanner sc = new Scanner(System.in);
        json = sc.useDelimiter("\\A").next();
        sc.close();
        idx = 0;

        // 解析顶层对象，得到 节点ID -> 信号类型计数字典
        Map<String, Map<String, Integer>> data = parseTopLevel();

        // 计算结果，保留输入顺序
        Map<String, Map<String, Object>> result = new LinkedHashMap<>();
        for (Map.Entry<String, Map<String, Integer>> entry : data.entrySet()) {
            String node = entry.getKey();
            Map<String, Integer> counts = entry.getValue();
            double entropy = computeEntropy(counts);
            // 构建 {"entropy": 保留三位小数}
            Map<String, Object> inner = new LinkedHashMap<>();
            inner.put("entropy", formatDouble(entropy));
            result.put(node, inner);
        }

        // 输出 JSON
        System.out.println(mapToJson(result));
    }

    /** 解析顶层对象： { "node": {...}, ... } */
    private static Map<String, Map<String, Integer>> parseTopLevel() {
        skipWhitespace();
        expect('{');
        Map<String, Map<String, Integer>> map = new LinkedHashMap<>();
        if (peek() == '}') {
            idx++;
            return map;
        }
        while (true) {
            skipWhitespace();
            String key = parseString();
            skipWhitespace();
            expect(':');
            skipWhitespace();
            Map<String, Integer> inner = parseInnerObject();
            map.put(key, inner);
            skipWhitespace();
            char c = json.charAt(idx);
            if (c == ',') {
                idx++;
                continue;
            } else if (c == '}') {
                idx++;
                break;
            } else {
                throw new RuntimeException("Unexpected char: " + c);
            }
        }
        return map;
    }

    /** 解析内层对象： { "sig": int, ... } */
    private static Map<String, Integer> parseInnerObject() {
        skipWhitespace();
        expect('{');
        Map<String, Integer> map = new LinkedHashMap<>();
        if (peek() == '}') {
            idx++;
            return map;
        }
        while (true) {
            skipWhitespace();
            String key = parseString();
            skipWhitespace();
            expect(':');
            skipWhitespace();
            int val = parseInteger();
            map.put(key, val);
            skipWhitespace();
            char c = json.charAt(idx);
            if (c == ',') {
                idx++;
                continue;
            } else if (c == '}') {
                idx++;
                break;
            } else {
                throw new RuntimeException("Unexpected char in inner: " + c);
            }
        }
        return map;
    }

    /** 解析双引号包围的字符串 */
    private static String parseString() {
        skipWhitespace();
        expect('"');
        StringBuilder sb = new StringBuilder();
        while (idx < json.length()) {
            char c = json.charAt(idx);
            if (c == '"') {
                idx++;
                return sb.toString();
            }
            sb.append(c);
            idx++;
        }
        throw new RuntimeException("Unterminated string");
    }

    /** 解析整数（含可能的负号，但题目保证非负） */
    private static int parseInteger() {
        skipWhitespace();
        int sign = 1;
        if (json.charAt(idx) == '-') {
            sign = -1;
            idx++;
        }
        int num = 0;
        while (idx < json.length() && Character.isDigit(json.charAt(idx))) {
            num = num * 10 + (json.charAt(idx) - '0');
            idx++;
        }
        return sign * num;
    }

    private static void skipWhitespace() {
        while (idx < json.length() && Character.isWhitespace(json.charAt(idx))) {
            idx++;
        }
    }

    private static char peek() {
        skipWhitespace();
        return json.charAt(idx);
    }

    private static void expect(char expected) {
        if (json.charAt(idx) != expected) {
            throw new RuntimeException("Expected " + expected + " but got " + json.charAt(idx));
        }
        idx++;
    }

    /** 计算分布复杂度（信息熵） */
    private static double computeEntropy(Map<String, Integer> counts) {
        int total = 0;
        for (int c : counts.values()) {
            total += c;
        }
        if (total == 0) return 0.0;

        double H = 0.0;
        for (int c : counts.values()) {
            if (c <= 0) continue;
            double p = (double) c / total;
            H -= p * (Math.log(p) / Math.log(2)); // log2(p)
        }
        return H;
    }

    /** 将 double 格式化为保留三位小数的字符串（如 1.234, 0.000） */
    private static String formatDouble(double val) {
        // 使用 Locale.US 保证小数点格式
        return String.format(Locale.US, "%.3f", val);
    }

    /** 将结果 Map 转化为紧凑 JSON 字符串 */
    private static String mapToJson(Map<String, Map<String, Object>> map) {
        StringBuilder sb = new StringBuilder();
        sb.append("{");
        boolean first = true;
        for (Map.Entry<String, Map<String, Object>> entry : map.entrySet()) {
            if (!first) sb.append(",");
            sb.append("\"").append(entry.getKey()).append("\":{");
            // 内层只有一个键 "entropy"
            sb.append("\"entropy\":").append(entry.getValue().get("entropy").toString());
            sb.append("}");
            first = false;
        }
        sb.append("}");
        return sb.toString();
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <map>
#include <vector>
#include <cmath>
#include <iomanip>
#include <sstream>
#include <cctype>

using namespace std;

class JSONParser {
public:
    JSONParser(const string& s) : json(s), idx(0) {}

    // 解析顶层对象，返回节点ID及其信号的计数字典，并保持插入顺序
    vector<pair<string, map<string, int>>> parseTopLevel() {
        skipWhitespace();
        expect('{');
        vector<pair<string, map<string, int>>> result;
        if (peek() == '}') { idx++; return result; }
        while (true) {
            skipWhitespace();
            string key = parseString();
            skipWhitespace();
            expect(':');
            skipWhitespace();
            map<string, int> inner = parseInnerObject();
            result.emplace_back(key, inner);
            skipWhitespace();
            char c = json[idx];
            if (c == ',') {
                idx++;
                continue;
            } else if (c == '}') {
                idx++;
                break;
            } else {
                throw runtime_error("Unexpected char in top-level");
            }
        }
        return result;
    }

private:
    string json;
    size_t idx;

    void skipWhitespace() {
        while (idx < json.size() && isspace(json[idx])) idx++;
    }

    char peek() {
        skipWhitespace();
        return json[idx];
    }

    void expect(char ch) {
        if (json[idx] != ch) throw runtime_error("Expected char");
        idx++;
    }

    string parseString() {
        skipWhitespace();
        expect('"');
        string s;
        while (idx < json.size() && json[idx] != '"') {
            s.push_back(json[idx++]);
        }
        if (idx < json.size() && json[idx] == '"') idx++;
        return s;
    }

    int parseInteger() {
        skipWhitespace();
        int sign = 1;
        if (json[idx] == '-') { sign = -1; idx++; }
        int num = 0;
        while (idx < json.size() && isdigit(json[idx])) {
            num = num * 10 + (json[idx] - '0');
            idx++;
        }
        return sign * num;
    }

    map<string, int> parseInnerObject() {
        skipWhitespace();
        expect('{');
        map<string, int> m;
        if (peek() == '}') { idx++; return m; }
        while (true) {
            skipWhitespace();
            string key = parseString();
            skipWhitespace();
            expect(':');
            skipWhitespace();
            int val = parseInteger();
            m[key] = val;
            skipWhitespace();
            char c = json[idx];
            if (c == ',') {
                idx++;
                continue;
            } else if (c == '}') {
                idx++;
                break;
            } else {
                throw runtime_error("Unexpected char in inner object");
            }
        }
        return m;
    }
};

// 计算分布复杂度（信息熵）
double computeEntropy(const map<string, int>& counts) {
    int total = 0;
    for (const auto& p : counts) total += p.second;
    if (total == 0) return 0.0;
    double H = 0.0;
    for (const auto& p : counts) {
        int c = p.second;
        if (c <= 0) continue;
        double prob = static_cast<double>(c) / total;
        H -= prob * log2(prob);
    }
    return H;
}

// 将 double 保留三位小数转为字符串（避免科学计数法，确保小数点后三位）
string formatDouble(double val) {
    ostringstream oss;
    oss << fixed << setprecision(3) << val;
    return oss.str();
}

// 将结果转换为紧凑 JSON 字符串，保持节点插入顺序
string mapToJson(const vector<pair<string, double>>& entropies) {
    ostringstream oss;
    oss << "{";
    bool first = true;
    for (const auto& p : entropies) {
        if (!first) oss << ",";
        oss << "\"" << p.first << "\":{\"entropy\":" << formatDouble(p.second) << "}";
        first = false;
    }
    oss << "}";
    return oss.str();
}

int main() {
    // 读取全部标准输入
    string input, line;
    while (getline(cin, line)) {
        input += line;
    }
    JSONParser parser(input);
    auto data = parser.parseTopLevel();

    // 计算每个节点的熵，按序存储
    vector<pair<string, double>> entropies;
    for (const auto& nodePair : data) {
        double H = computeEntropy(nodePair.second);
        entropies.emplace_back(nodePair.first, H);
    }

    // 输出 JSON
    cout << mapToJson(entropies) << endl;
    return 0;
}
```
