## 解题思路

本题要求动态维护社交网络中的连通群体，并判断每次添加好友关系后是否**首次形成闭环群体**。一个连通群体是闭环群体，当且仅当它内部的用户数量等于好友关系数量，即图论中一棵树加上一条边形成恰好一个环（基环树）的结构。由于网络由无向边构成且保证没有自环，该条件等价于：在一个原本是树的连通块中添加一条连接块内两个用户的边。

利用**并查集（DSU）**维护连通块，并记录每个连通块是否已经形成过闭环群体（最初均未形成）。处理每次操作 $(u, v)$：

1. **处理重边**：如果 $(u, v)$ 或 $(v, u)$ 之前已经添加过（输入保证无自环），根据题意该操作无效，直接输出 `No`。用一个集合记录所有出现过的无序边对。
2. **检查连通性**：
   - **若 $u$ 和 $v$ 不在同一连通块**：合并两个连通块，此时两棵树合成为一棵更大的树，边数仍等于点数减一，不满足闭环条件，输出 `No`。
   - **若 $u$ 和 $v$ 已经在同一连通块**：说明该连通块原本是一棵树（边数 = 点数 - 1），加上这条边后恰好边数等于点数，形成闭环。
     - 如果该连通块之前**从未被标记为已形成过闭环**，则本次操作使其**首次形成闭环群体**，输出 `Yes`，并将该连通块的根节点标记为已形成。
     - 如果该连通块**已经被标记过**，说明之前已经输出过 `Yes`，此后不再输出，直接输出 `No`。

3. 将当前边 $(u, v)$ 加入边集合，防止后续重边。

通过上述方法，每次操作可以在几乎 $O(1)$ 的时间内完成判定和合并。

## 复杂度分析

- 时间复杂度：$O((n + m) \alpha(n))$ 或 $O((n + m) \log n)$，其中 $\alpha(n)$ 为反阿克曼函数。并查集的查找与合并操作近似为常数时间，边集合的插入与查询在基于哈希的实现中期望为 $O(1)$，总复杂度与操作数和用户数近乎线性。
- 空间复杂度：$O(n + m)$。并查集需要 $O(n)$ 的空间存储父节点和大小，标记集合需要 $O(n)$，边集合需要 $O(m)$ 的空间记录不重复的边。

## 代码实现

### Python

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    parent = list(range(n + 1))
    size = [1] * (n + 1)

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(x, y):
        x = find(x)
        y = find(y)
        if x == y:
            return False
        if size[x] < size[y]:
            x, y = y, x
        parent[y] = x
        size[x] += size[y]
        return True

    closed = set()       # 记录已形成过闭环的连通块根节点
    edges = set()        # 记录已存在的无向边

    out = []
    for _ in range(m):
        u, v = map(int, input().split())
        # 保证无自环，只处理重边
        edge = (u, v) if u < v else (v, u)
        if edge in edges:
            out.append("No")
            continue

        ru = find(u)
        rv = find(v)
        if ru != rv:
            union(u, v)
            out.append("No")
        else:
            # 已经在同一连通块
            root = ru
            if root not in closed:
                out.append("Yes")
                closed.add(root)
            else:
                out.append("No")
            # 此处再进行一次 union 调用无实际效果，与参考逻辑保持对齐
            union(u, v)

        edges.add(edge)

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.*;

public class Main {
    static class DSU {
        int[] parent, size;
        DSU(int n) {
            parent = new int[n + 1];
            size = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }
        int find(int x) {
            while (parent[x] != x) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }
            return x;
        }
        boolean union(int x, int y) {
            x = find(x);
            y = find(y);
            if (x == y) return false;
            if (size[x] < size[y]) {
                int t = x;
                x = y;
                y = t;
            }
            parent[y] = x;
            size[x] += size[y];
            return true;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        DSU dsu = new DSU(n);
        Set<Integer> closed = new HashSet<>();
        Set<String> edges = new HashSet<>();

        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            // 保证无自环，统一无序边编码
            String edge = u < v ? u + "," + v : v + "," + u;
            if (edges.contains(edge)) {
                System.out.println("No");
                continue;
            }

            int ru = dsu.find(u);
            int rv = dsu.find(v);
            if (ru != rv) {
                dsu.union(u, v);
                System.out.println("No");
            } else {
                int root = ru;
                if (!closed.contains(root)) {
                    System.out.println("Yes");
                    closed.add(root);
                } else {
                    System.out.println("No");
                }
                dsu.union(u, v); // 保持行为一致
            }
            edges.add(edge);
        }
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct DSU {
    vector<int> parent, sz;
    DSU(int n) : parent(n + 1), sz(n + 1, 1) {
        for (int i = 1; i <= n; i++) parent[i] = i;
    }
    int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }
    bool unite(int x, int y) {
        x = find(x); y = find(y);
        if (x == y) return false;
        if (sz[x] < sz[y]) swap(x, y);
        parent[y] = x;
        sz[x] += sz[y];
        return true;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    DSU dsu(n);
    unordered_set<int> closed;
    set<pair<int, int>> edges;  // 有序存放避免重复判断复杂

    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        if (u > v) swap(u, v);
        if (edges.count({u, v})) {
            cout << "No\n";
            continue;
        }
        int ru = dsu.find(u), rv = dsu.find(v);
        if (ru != rv) {
            dsu.unite(u, v);
            cout << "No\n";
        } else {
            int root = ru;
            if (closed.find(root) == closed.end()) {
                cout << "Yes\n";
                closed.insert(root);
            } else {
                cout << "No\n";
            }
            dsu.unite(u, v);
        }
        edges.insert({u, v});
    }
    return 0;
}
```
