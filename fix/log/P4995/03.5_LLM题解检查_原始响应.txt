## 解题思路

本题需要模拟石板字母序列的演化过程。每一轮演化从左到右扫描序列，对每个位置的字母，统计其在当前位置**左侧**出现的次数 \(x\) 和**右侧**出现的次数 \(y\)（均不计当前位置本身）；若 \(x = y\)，则将该字母替换为字母表中的下一个字母（`z` 的下一个字母循环至 `a`），否则保持不变。每次替换立即生效，会影响后续位置的计算。

直接模拟 \(r\) 轮是容易实现的：一轮扫描需要 \(O(m)\) 时间。但 \(m\) 和 \(r\) 均可达到 \(10^5\)，全部问询的 \(m\) 总和与 \(r\) 总和不超过 \(10^5\)，因此最坏情况下单组数据的 \(m\) 和 \(r\) 都可能接近 \(10^5\)，若逐轮模拟 \(r\) 次，总操作次数可能达到 \(O(m \cdot r)\)，需要优化。

观察演化规则可以发现：

1. 如果没有位置满足 \(x = y\)，则本轮序列不会发生任何变化，后续所有轮次也不会再变化（已进入稳定态），可以直接提前结束。
2. 若存在发生变化的位置，由于每次变化都是将字母变成下一个字母（`z` 变 `a`），且变化规则对称，该过程具有周期性。可以证明：一旦序列开始重复出现，其周期必然是 26 的因子；通常经过少数步后，序列会进入一个长度为 26 的循环，每个会发生改变的位置在每一轮都恰好向前走一步（模 26）。

因此可以采用“周期检测 + 跳转”的策略：

- 使用一个滑动窗口（例如队列）保存最近 27 个序列状态（含当前状态）；每完成一轮变换后，将新序列加入窗口尾部。
- 若窗口大小达到 27 且最前端的序列与最后端的序列完全相同，说明序列已经进入长度为 26 的周期（即序列每 26 轮循环一次）。
- 此时，找出在窗口中首尾两个相同序列之间，从第一个序列变换到第二个序列时发生变化的所有位置。这些位置在后续每一轮中都会稳定地递增 1（模 26）。设剩余待执行的轮数为 \(rem = r - done\)，则对这些位置直接加上 \(rem \bmod 26\) 即可得到最终序列，避免继续逐轮模拟。
- 如果在进入周期前已经稳定（某轮变换后序列不再变化），则直接结束。

具体单轮变换的模拟实现：维护一个长度为 26 的整型频次数组（统计每个字母在当前序列中出现的总次数），以及一个左侧出现次数的数组。从左到右处理每个位置：
- 当前字母为 \(c\)，则其右侧出现次数 = 总次数 `freq[c]` - 左侧次数 `left[c]` - 1（减去的 1 是当前位置本身）。
- 若 `left[c] == 右侧次数`，则将该位置更新为 `(c + 1) % 26`，同时更新 `freq` 和 `left` 数组以反映改变。
- 否则只增加 `left[c]`，继续处理下一位置。

利用上述优化，可以在限制时间内完成所有询问。

## 复杂度分析

- **时间复杂度**：对于每组测试数据，单轮变换需要扫描整个序列，时间复杂度为 \(O(m)\)。进入周期前的轮数在实际数据下很小（通常不超过数十轮），跳转后只需一次 \(O(m)\) 的扫描来更新变化的位置。所有测试数据的 \(m\) 总和不超过 \(10^5\)，\(r\) 总和也不超过 \(10^5\)，因此总复杂度为 \(O(\sum m + \text{小的额外模拟步数}) \approx O(10^5)\)，可以高效通过。
- **空间复杂度**：需要存储当前序列、频次数组以及保存最近序列状态的窗口。窗口至多保存 27 个长度为 \(m\) 的字符串，空间复杂度为 \(O(m)\)（在总和约束下可接受）。频次数组和左侧数组为常数大小 \(O(26)\)。

## 代码实现

### Python

```python
import sys
from collections import deque

def apply_once(seq):
    """对当前序列执行一轮演化，返回新序列和是否发生变化"""
    n = len(seq)
    arr = list(seq)
    freq = [0] * 26
    left = [0] * 26
    for ch in arr:
        freq[ord(ch) - 97] += 1

    changed = False
    for i in range(n):
        c = ord(arr[i]) - 97
        # 右侧出现次数 = 总次数 - 左侧次数 - 1（当前位置）
        if left[c] == freq[c] - left[c] - 1:
            freq[c] -= 1
            nc = (c + 1) % 26
            arr[i] = chr(nc + 97)
            freq[nc] += 1
            left[nc] += 1
            changed = True
        else:
            left[c] += 1
    return ''.join(arr), changed

def solve_one(z, r):
    """对单组数据计算 r 轮演化后的序列"""
    seq = z
    win = deque()
    win.append(seq)
    done = 0
    while done < r:
        nxt, changed = apply_once(seq)
        done += 1
        if not changed:       # 已经稳定
            seq = nxt
            break
        seq = nxt
        win.append(seq)
        if len(win) > 27:
            win.popleft()
        # 检查是否进入周期 26
        if len(win) == 27 and win[0] == win[-1]:
            rem = r - done
            s0 = win[0]
            s1 = win[1]
            # 找出从 s0 到 s1 发生变化的位置
            arr = list(seq)
            add = rem % 26
            for i in range(len(arr)):
                if s0[i] != s1[i]:
                    arr[i] = chr((ord(arr[i]) - 97 + add) % 26 + 97)
            seq = ''.join(arr)
            break
    return seq

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    out_lines = []
    for _ in range(q):
        m = int(next(it))
        r = int(next(it))
        z = next(it)
        out_lines.append(solve_one(z, r))
    sys.stdout.write('\n'.join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 执行一轮演化，返回新序列以及是否发生变化
    static String applyOnce(String s, boolean[] changed) {
        int n = s.length();
        char[] arr = s.toCharArray();
        int[] freq = new int[26];
        int[] left = new int[26];
        for (char ch : arr) freq[ch - 'a']++;

        boolean chg = false;
        for (int i = 0; i < n; i++) {
            int c = arr[i] - 'a';
            if (left[c] == freq[c] - left[c] - 1) {
                freq[c]--;
                int nc = (c + 1) % 26;
                arr[i] = (char) ('a' + nc);
                freq[nc]++;
                left[nc]++;
                chg = true;
            } else {
                left[c]++;
            }
        }
        changed[0] = chg;
        return new String(arr);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(br.readLine().trim());
        StringBuilder out = new StringBuilder();
        while (q-- > 0) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int m = Integer.parseInt(st.nextToken());
            int r = Integer.parseInt(st.nextToken());
            String z = br.readLine().trim();

            ArrayDeque<String> window = new ArrayDeque<>();
            window.addLast(z);
            int done = 0;
            boolean[] changed = new boolean[1];
            while (done < r) {
                String nxt = applyOnce(z, changed);
                done++;
                if (!changed[0]) {      // 已稳定
                    z = nxt;
                    break;
                }
                z = nxt;
                window.addLast(z);
                if (window.size() > 27) window.removeFirst();
                if (window.size() == 27 && window.peekFirst().equals(window.peekLast())) {
                    int rem = r - done;
                    String s0 = window.peekFirst();
                    Iterator<String> it = window.iterator();
                    it.next(); // skip first
                    String s1 = it.next(); // the second state
                    int add = rem % 26;
                    char[] arr = z.toCharArray();
                    for (int i = 0; i < m; i++) {
                        if (s0.charAt(i) != s1.charAt(i)) {
                            arr[i] = (char) ('a' + (arr[i] - 'a' + add) % 26);
                        }
                    }
                    z = new String(arr);
                    break;
                }
            }
            out.append(z).append('\n');
        }
        System.out.print(out);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 执行一轮演化，返回新序列，参数changed指示是否发生变化
string apply_once(const string &s, bool &changed) {
    int n = (int)s.size();
    array<int, 26> freq{}, left{};
    for (char ch : s) ++freq[ch - 'a'];
    string res = s;
    changed = false;
    for (int i = 0; i < n; ++i) {
        int c = res[i] - 'a';
        if (left[c] == freq[c] - left[c] - 1) {
            --freq[c];
            int nc = (c + 1) % 26;
            res[i] = char('a' + nc);
            ++freq[nc];
            ++left[nc];
            changed = true;
        } else {
            ++left[c];
        }
    }
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int q;
    cin >> q;
    while (q--) {
        int m, r;
        string z;
        cin >> m >> r >> z;

        deque<string> window;
        window.push_back(z);
        int done = 0;
        bool changed;
        while (done < r) {
            string nxt = apply_once(z, changed);
            ++done;
            if (!changed) {      // 已稳定
                z = nxt;
                break;
            }
            z.swap(nxt);
            window.push_back(z);
            if ((int)window.size() > 27) window.pop_front();
            if ((int)window.size() == 27 && window.front() == window.back()) {
                int rem = r - done;
                string s0 = window[0];
                string s1 = window[1];
                int add = rem % 26;
                for (int i = 0; i < m; ++i) {
                    if (s0[i] != s1[i]) {
                        z[i] = char('a' + (z[i] - 'a' + add) % 26);
                    }
                }
                break;
            }
        }
        cout << z << '\n';
    }
    return 0;
}
```
