## 解题思路

### 关键观察

题目要求统计有序对 $(a,b)$（$1\le a,b\le n$）且满足 $(a|b)\le K$。
注意这是“数值上的 $\le$”，而不是“按位包含关系”。例如：$K=5\,(0101_2)$，$(a|b)=3\,(0011_2)$ 也满足 $\le K$（虽然 3 在某些位上为 1 而 $K$ 为 0）。

因此，不能简单地只数“只用到了 $K$ 的 1 位的数”。更稳妥的做法是**二进制数位 DP（Digit DP）**：

我们从最高位到最低位逐位决定 $(a_i,b_i)\in\{0,1\}\times\{0,1\}$，同时维持三种“前缀是否紧贴上界”的状态：

* $ta$：到当前位为止，$a$ 的前缀是否仍与 $n$ 的前缀相等（紧）；
* $tb$：同理，针对 $b$ 与 $n$；
* $tk$：$(a|b)$ 的前缀是否仍与 $K$ 的前缀相等。

转移规则（在处理第 $i$ 位，最高位到最低位）：

* 若 $ta=1$，则本位 $a_i\le n_i$；若 $tb=1$，则 $b_i\le n_i$；
* 若 $tk=1$，则 $(a_i|b_i)\le K_i$；
* 新的紧致标记：

  * $ta' = ta \land (a_i=n_i)$
  * $tb' = tb \land (b_i=n_i)$
  * $tk' = tk \land ((a_i|b_i)=K_i)$

当 $tk=0$ 时，前缀已经严格小于 $K$，后续位对 $K$ 不再有约束；同理 $ta=0$ 或 $tb=0$ 时，相应数对 $n$ 也不再有约束。

上面 DP 自然地统计**区间 $[0,n]\times[0,n]$** 中满足 $(a|b)\le K$ 的对数，记为 $F(n,K)$。

但题目要求的是 $1\le a,b\le n$。用**容斥**排除 0：

* 固定 $a=0$ 的对数：$(0|b)=b\le K$ 且 $0\le b\le n$，所以是 $\min(n,K)+1$（包含 $b=0$）。
* 固定 $b=0$ 同理，也是 $\min(n,K)+1$。
* $(a,b)=(0,0)$ 被减了两次，要加回 1。

于是答案

$$
\text{Ans} \equiv F(n,K) - 2\cdot(\min(n,K)+1) + 1 \pmod{10^9+7}.
$$

## 复杂度分析

* 设二进制位数 $B\approx 61$。
* 每组数据：时间 $O(B\cdot 8\cdot 4)\approx O(B)$，空间 $O(8)$。
* 对于 $t\le 10^4$：总时间约 $2\times 10^7$ 级别，三种语言均可通过。

## 代码实现

### Python

```python
import sys

MOD = 10**9 + 7

def F(N, K):
    # 三维DP：dp[ta][tb][tk]
    dp = [[[0]*2 for _ in range(2)] for __ in range(2)]
    dp[1][1][1] = 1  # 初始：三个前缀都紧贴
    for i in range(60, -1, -1):  # 处理第 i 位
        ni = (N >> i) & 1
        ki = (K >> i) & 1
        ndp = [[[0]*2 for _ in range(2)] for __ in range(2)]
        for ta in (0, 1):
            for tb in (0, 1):
                for tk in (0, 1):
                    cur = dp[ta][tb][tk]
                    if cur == 0:
                        continue
                    maxa = ni if ta == 1 else 1  # 若紧，则本位不能超过 n_i
                    maxb = ni if tb == 1 else 1
                    for aa in range(maxa + 1):
                        for bb in range(maxb + 1):
                            if tk == 1 and ((aa | bb) > ki):
                                continue  # 若对K紧，则(aa|bb)不能超过 k_i
                            nta = 1 if (ta == 1 and aa == ni) else 0
                            ntb = 1 if (tb == 1 and bb == ni) else 0
                            ntk = 1 if (tk == 1 and ((aa | bb) == ki)) else 0
                            ndp[nta][ntb][ntk] = (ndp[nta][ntb][ntk] + cur) % MOD
        dp = ndp
    total = 0
    for ta in (0, 1):
        for tb in (0, 1):
            for tk in (0, 1):
                total = (total + dp[ta][tb][tk]) % MOD
    return total

def solve_case(n, k):
    total = F(n, k)  # 0<=a,b<=n
    z = min(n, k) + 1  # a=0 或 b=0 的计数
    ans = (total - 2 * (z % MOD) + 1) % MOD
    return ans

def main():
    data = list(map(int, sys.stdin.read().strip().split()))
    t = data[0]
    out = []
    idx = 1
    for _ in range(t):
        n = data[idx]; k = data[idx+1]; idx += 2
        out.append(str(solve_case(n, k)))
    print("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java 代码

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1_000_000_007L;

    static long F(long N, long K) {
        long[][][] dp = new long[2][2][2];
        dp[1][1][1] = 1; // 初始三紧

        for (int i = 60; i >= 0; --i) {
            int ni = (int)((N >> i) & 1L);
            int ki = (int)((K >> i) & 1L);
            long[][][] ndp = new long[2][2][2];

            for (int ta = 0; ta <= 1; ta++) {
                for (int tb = 0; tb <= 1; tb++) {
                    for (int tk = 0; tk <= 1; tk++) {
                        long cur = dp[ta][tb][tk];
                        if (cur == 0) continue;
                        int maxa = (ta == 1) ? ni : 1;
                        int maxb = (tb == 1) ? ni : 1;

                        for (int aa = 0; aa <= maxa; aa++) {
                            for (int bb = 0; bb <= maxb; bb++) {
                                if (tk == 1 && ((aa | bb) > ki)) continue;
                                int nta = (ta == 1 && aa == ni) ? 1 : 0;
                                int ntb = (tb == 1 && bb == ni) ? 1 : 0;
                                int ntk = (tk == 1 && ((aa | bb) == ki)) ? 1 : 0;
                                ndp[nta][ntb][ntk] += cur;
                                if (ndp[nta][ntb][ntk] >= MOD) ndp[nta][ntb][ntk] -= MOD;
                            }
                        }
                    }
                }
            }
            dp = ndp;
        }

        long total = 0;
        for (int ta = 0; ta <= 1; ta++)
            for (int tb = 0; tb <= 1; tb++)
                for (int tk = 0; tk <= 1; tk++) {
                    total += dp[ta][tb][tk];
                    if (total >= MOD) total -= MOD;
                }
        return total;
    }

    static long solveOne(long n, long k) {
        long total = F(n, k);                    // 0<=a,b<=n
        long z = Math.min(n, k) + 1;             // a=0 或 b=0 的个数
        long ans = (total - (2L * (z % MOD)) % MOD + MOD) % MOD;
        ans = (ans + 1) % MOD;                   // 加回 (0,0)
        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int t = Integer.parseInt(s.trim());
        StringBuilder sb = new StringBuilder();
        for (int _i = 0; _i < t; _i++) {
            String line;
            // 读取到非空行
            do { line = br.readLine(); } while (line != null && line.trim().isEmpty());
            StringTokenizer st = new StringTokenizer(line);
            long n = Long.parseLong(st.nextToken());
            long k = Long.parseLong(st.nextToken());
            sb.append(solveOne(n, k)).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

static const long long MOD = 1000000007LL;

long long F(long long N, long long K) {
    long long dp[2][2][2] = {};
    dp[1][1][1] = 1; // 初始三紧

    for (int i = 60; i >= 0; --i) {
        int ni = (int)((N >> i) & 1LL);
        int ki = (int)((K >> i) & 1LL);
        long long ndp[2][2][2] = {};
        for (int ta = 0; ta <= 1; ++ta) {
            for (int tb = 0; tb <= 1; ++tb) {
                for (int tk = 0; tk <= 1; ++tk) {
                    long long cur = dp[ta][tb][tk];
                    if (!cur) continue;
                    int maxa = (ta ? ni : 1);
                    int maxb = (tb ? ni : 1);
                    for (int aa = 0; aa <= maxa; ++aa) {
                        for (int bb = 0; bb <= maxb; ++bb) {
                            if (tk && ((aa | bb) > ki)) continue;
                            int nta = (ta && aa == ni) ? 1 : 0;
                            int ntb = (tb && bb == ni) ? 1 : 0;
                            int ntk = (tk && ((aa | bb) == ki)) ? 1 : 0;
                            ndp[nta][ntb][ntk] += cur;
                            if (ndp[nta][ntb][ntk] >= MOD) ndp[nta][ntb][ntk] -= MOD;
                        }
                    }
                }
            }
        }
        memcpy(dp, ndp, sizeof(dp));
    }
    long long total = 0;
    for (int ta = 0; ta <= 1; ++ta)
        for (int tb = 0; tb <= 1; ++tb)
            for (int tk = 0; tk <= 1; ++tk) {
                total += dp[ta][tb][tk];
                if (total >= MOD) total -= MOD;
            }
    return total;
}

long long solve_one(long long n, long long k) {
    long long total = F(n, k);                 // 0<=a,b<=n
    long long z = min(n, k) + 1;               // a=0 或 b=0 的数量
    long long ans = (total - (2LL * (z % MOD)) % MOD + MOD) % MOD;
    ans = (ans + 1) % MOD;                     // 加回 (0,0)
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t; 
    if (!(cin >> t)) return 0;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        cout << solve_one(n, k) << "\n";
    }
    return 0;
}
```
