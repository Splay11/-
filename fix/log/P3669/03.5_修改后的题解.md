## 解题思路

需要统计 $1 \le a,b \le n$ 且 $(a \mid b) \le K$ 的有序对个数。这里的 $\le$ 是数值比较，不能只统计“只用到 $K$ 中为 `1` 的那些二进制位”的数。

用数位 DP：从最高位到最低位决定 $(a_i,b_i)$，并维护三个紧致标记：

- $ta$：当前前缀是否仍与 $n$ 的前缀相等；
- $tb$：对 $b$ 同理；
- $tk$：$(a \mid b)$ 的前缀是否仍与 $K$ 的前缀相等。

转移时，若对应标记为 `1`，则本位不能超过上界。该 DP 统计的是 $[0,n] \times [0,n]$ 中满足 $(a \mid b) \le K$ 的对数 $F(n,K)$。

题目要求 $a,b \ge 1$，用容斥去掉 `0`：

- 固定 $a=0$ 的个数为 $\min(n,K)+1$（含 $b=0$），$b=0$ 同理；
- $(0,0)$ 被减了两次，加回 `1`。

答案为

$$
F(n,K) - 2(\min(n,K)+1) + 1 \pmod{10^9+7}.
$$

## 复杂度分析

设二进制位数 $B \approx 61$。

- 时间复杂度：$O(B)$ 每组。
- 空间复杂度：$O(1)$。

对于 $t \le 10^4$ 可以通过。

## 代码实现

### Python

```python
import sys

MOD = 10**9 + 7

def F(N, K):
    dp = [[[0] * 2 for _ in range(2)] for __ in range(2)]
    dp[1][1][1] = 1
    for i in range(60, -1, -1):
        ni = (N >> i) & 1
        ki = (K >> i) & 1
        ndp = [[[0] * 2 for _ in range(2)] for __ in range(2)]
        for ta in (0, 1):
            for tb in (0, 1):
                for tk in (0, 1):
                    cur = dp[ta][tb][tk]
                    if cur == 0:
                        continue
                    maxa = ni if ta == 1 else 1
                    maxb = ni if tb == 1 else 1
                    for aa in range(maxa + 1):
                        for bb in range(maxb + 1):
                            if tk == 1 and ((aa | bb) > ki):
                                continue
                            nta = 1 if (ta == 1 and aa == ni) else 0
                            ntb = 1 if (tb == 1 and bb == ni) else 0
                            ntk = 1 if (tk == 1 and ((aa | bb) == ki)) else 0
                            ndp[nta][ntb][ntk] = (ndp[nta][ntb][ntk] + cur) % MOD
        dp = ndp
    total = 0
    for ta in (0, 1):
        for tb in (0, 1):
            for tk in (0, 1):
                total = (total + dp[ta][tb][tk]) % MOD
    return total

def solve_case(n, k):
    total = F(n, k)
    z = min(n, k) + 1
    ans = (total - 2 * (z % MOD) + 1) % MOD
    return ans

def main():
    data = list(map(int, sys.stdin.read().strip().split()))
    t = data[0]
    out = []
    idx = 1
    for _ in range(t):
        n = data[idx]; k = data[idx + 1]; idx += 2
        out.append(str(solve_case(n, k)))
    print("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1_000_000_007L;

    static long F(long N, long K) {
        long[][][] dp = new long[2][2][2];
        dp[1][1][1] = 1;
        for (int i = 60; i >= 0; --i) {
            int ni = (int)((N >> i) & 1L);
            int ki = (int)((K >> i) & 1L);
            long[][][] ndp = new long[2][2][2];
            for (int ta = 0; ta <= 1; ta++) {
                for (int tb = 0; tb <= 1; tb++) {
                    for (int tk = 0; tk <= 1; tk++) {
                        long cur = dp[ta][tb][tk];
                        if (cur == 0) continue;
                        int maxa = (ta == 1) ? ni : 1;
                        int maxb = (tb == 1) ? ni : 1;
                        for (int aa = 0; aa <= maxa; aa++) {
                            for (int bb = 0; bb <= maxb; bb++) {
                                if (tk == 1 && ((aa | bb) > ki)) continue;
                                int nta = (ta == 1 && aa == ni) ? 1 : 0;
                                int ntb = (tb == 1 && bb == ni) ? 1 : 0;
                                int ntk = (tk == 1 && ((aa | bb) == ki)) ? 1 : 0;
                                ndp[nta][ntb][ntk] += cur;
                                if (ndp[nta][ntb][ntk] >= MOD) ndp[nta][ntb][ntk] -= MOD;
                            }
                        }
                    }
                }
            }
            dp = ndp;
        }
        long total = 0;
        for (int ta = 0; ta <= 1; ta++)
            for (int tb = 0; tb <= 1; tb++)
                for (int tk = 0; tk <= 1; tk++) {
                    total += dp[ta][tb][tk];
                    if (total >= MOD) total -= MOD;
                }
        return total;
    }

    static long solveOne(long n, long k) {
        long total = F(n, k);
        long z = Math.min(n, k) + 1;
        long ans = (total - (2L * (z % MOD)) % MOD + MOD) % MOD;
        ans = (ans + 1) % MOD;
        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine().trim());
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < t; i++) {
            String line;
            do { line = br.readLine(); } while (line != null && line.trim().isEmpty());
            StringTokenizer st = new StringTokenizer(line);
            long n = Long.parseLong(st.nextToken());
            long k = Long.parseLong(st.nextToken());
            sb.append(solveOne(n, k)).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

static const long long MOD = 1000000007LL;

long long F(long long N, long long K) {
    long long dp[2][2][2] = {};
    dp[1][1][1] = 1;
    for (int i = 60; i >= 0; --i) {
        int ni = (int)((N >> i) & 1LL);
        int ki = (int)((K >> i) & 1LL);
        long long ndp[2][2][2] = {};
        for (int ta = 0; ta <= 1; ++ta) {
            for (int tb = 0; tb <= 1; ++tb) {
                for (int tk = 0; tk <= 1; ++tk) {
                    long long cur = dp[ta][tb][tk];
                    if (!cur) continue;
                    int maxa = (ta ? ni : 1);
                    int maxb = (tb ? ni : 1);
                    for (int aa = 0; aa <= maxa; ++aa) {
                        for (int bb = 0; bb <= maxb; ++bb) {
                            if (tk && ((aa | bb) > ki)) continue;
                            int nta = (ta && aa == ni) ? 1 : 0;
                            int ntb = (tb && bb == ni) ? 1 : 0;
                            int ntk = (tk && ((aa | bb) == ki)) ? 1 : 0;
                            ndp[nta][ntb][ntk] += cur;
                            if (ndp[nta][ntb][ntk] >= MOD) ndp[nta][ntb][ntk] -= MOD;
                        }
                    }
                }
            }
        }
        memcpy(dp, ndp, sizeof(dp));
    }
    long long total = 0;
    for (int ta = 0; ta <= 1; ++ta)
        for (int tb = 0; tb <= 1; ++tb)
            for (int tk = 0; tk <= 1; ++tk) {
                total += dp[ta][tb][tk];
                if (total >= MOD) total -= MOD;
            }
    return total;
}

long long solve_one(long long n, long long k) {
    long long total = F(n, k);
    long long z = min(n, k) + 1;
    long long ans = (total - (2LL * (z % MOD)) % MOD + MOD) % MOD;
    ans = (ans + 1) % MOD;
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    if (!(cin >> t)) return 0;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        cout << solve_one(n, k) << "\n";
    }
    return 0;
}
```
