## 解题思路

本题给出了一棵包含 $n$ 个节点的树，每个节点 $i$ 有一个处理能力 $w_i$。对于每一台服务器 $i$，若它单独宕机，则该节点以及与其相连的所有边都会被移除，原树会分裂成若干个连通分量（子网）。每个连通分量的“子网处理能力”定义为该连通分量内所有存活服务器的处理能力的最大值；整个数据中心的“总剩余处理能力”为所有子网的子网处理能力之和。我们需要对每个 $i$ 求出仅移除 $i$ 后的总剩余处理能力。

删除一个节点 $u$ 后，剩余的连通分量包括：
- 每个儿子 $v$ 所在的子树（独立成为一个连通分量）；
- $u$ 的父节点一侧的剩余部分（如果有父节点）构成一个连通分量。

因此，问题可以转化为：对于每个节点 $u$，我们需要知道以 $u$ 为根的子树中（不删除任何节点）处理能力的最大值，以及当 $u$ 被删除后，它“父侧”部分的处理能力最大值。前者可以通过一次自底向上的树形 DP 求出，后者可以通过第二次自顶向下的 DP 求出。具体步骤如下：

1. **第一次 DFS（自底向上）**：  
   令 `down[u]` 表示以 $u$ 为根的子树中所有节点处理能力的最大值。  
   初始化 `down[u] = w[u]`，然后递归计算所有儿子 $v$ 的 `down[v]`，并更新 `down[u] = max(down[u], down[v])`。

2. **第二次 DFS（自顶向下）**：  
   令 `up[u]` 表示删除 $u$ 之后，其父侧连通分量的处理能力最大值。对于根节点，没有父侧，令 `up[root] = 0`。  
   在遍历到节点 $u$ 时，先计算以 $u$ 宕机时的答案：  
   `ans[u] = up[u] + Σ_{v 是 u 的儿子} down[v]`。  
   接着，为 $u$ 的所有儿子 $v$ 计算 `up[v]`。删除 $u$ 后，$v$ 的父侧连通分量包括：原来的父侧部分（最大值 `up[u]`）、$u$ 节点本身（处理能力 $w[u]$）以及 $u$ 除 $v$ 以外的其他儿子子树。因此，`up[v]` 需要取这三部分的最大值：  
   `use = max(up[u], w[u])`，并从 $u$ 的所有儿子中找出除了 $v$ 之外的 `down` 最大值（记作 `best`），则 `up[v] = max(use, best)`。  
   为了高效得到 `best`，可以在遍历 $u$ 的儿子时预先求出所有儿子 `down` 的最大值 `mx1` 和次大值 `mx2`。若 `down[v] == mx1`，则 `best = mx2`；否则 `best = mx1`。

按照上述 DP 过程，最终数组 `ans[i]` 即为服务器 $i$ 宕机后数据中心的总剩余处理能力。

## 复杂度分析

- **时间复杂度**：两次 DFS 遍历整棵树，每个节点和每条边均被访问常数次，总时间复杂度为 $O(n)$。
- **空间复杂度**：需要存储邻接表、`down` 数组、`up` 数组和 `ans` 数组，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    w = [int(next(it)) for _ in range(n)]
    adj = [[] for _ in range(n)]
    for _ in range(n - 1):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    down = [0] * n     # 子树内最大处理能力
    up = [0] * n       # 父侧连通分量的最大处理能力
    ans = [0] * n

    def dfs_down(u, parent):
        down[u] = w[u]
        for v in adj[u]:
            if v == parent:
                continue
            dfs_down(v, u)
            if down[v] > down[u]:
                down[u] = down[v]

    def dfs_up(u, parent):
        # 当前节点宕机后的总剩余处理能力
        total = up[u]
        mx1 = -1   # 所有儿子 down 的最大值
        mx2 = -1   # 所有儿子 down 的次大值
        for v in adj[u]:
            if v == parent:
                continue
            total += down[v]
            if down[v] > mx1:
                mx2 = mx1
                mx1 = down[v]
            elif down[v] > mx2:
                mx2 = down[v]
        ans[u] = total

        # 计算来自父侧的基准最大值
        use = up[u]
        if w[u] > use:
            use = w[u]

        for v in adj[u]:
            if v == parent:
                continue
            best = mx2 if down[v] == mx1 else mx1
            up[v] = use
            if best > up[v]:
                up[v] = best
            dfs_up(v, u)

    dfs_down(0, -1)
    dfs_up(0, -1)

    print(' '.join(str(x) for x in ans))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static int n;
    static int[] w;
    static List<Integer>[] adj;
    static int[] down, up;
    static long[] ans;

    static void dfsDown(int u, int parent) {
        down[u] = w[u];
        for (int v : adj[u]) {
            if (v == parent) continue;
            dfsDown(v, u);
            if (down[v] > down[u]) {
                down[u] = down[v];
            }
        }
    }

    static void dfsUp(int u, int parent) {
        long total = up[u];
        int mx1 = -1, mx2 = -1;
        for (int v : adj[u]) {
            if (v == parent) continue;
            total += down[v];
            if (down[v] > mx1) {
                mx2 = mx1;
                mx1 = down[v];
            } else if (down[v] > mx2) {
                mx2 = down[v];
            }
        }
        ans[u] = total;

        int use = up[u];
        if (w[u] > use) use = w[u];

        for (int v : adj[u]) {
            if (v == parent) continue;
            int best = (down[v] == mx1) ? mx2 : mx1;
            up[v] = use;
            if (best > up[v]) up[v] = best;
            dfsUp(v, u);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine().trim());
        w = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            w[i] = Integer.parseInt(st.nextToken());
        }

        adj = new ArrayList[n];
        for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            adj[u].add(v);
            adj[v].add(u);
        }

        down = new int[n];
        up = new int[n];
        ans = new long[n];

        dfsDown(0, -1);
        dfsUp(0, -1);

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append(ans[i]);
            if (i + 1 < n) sb.append(' ');
        }
        System.out.println(sb);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int n;
vector<int> w;
vector<vector<int>> adj;
vector<int> down, up;
vector<ll> ans;

void dfsDown(int u, int p) {
    down[u] = w[u];
    for (int v : adj[u]) {
        if (v == p) continue;
        dfsDown(v, u);
        if (down[v] > down[u]) {
            down[u] = down[v];
        }
    }
}

void dfsUp(int u, int p) {
    ll total = up[u];
    int mx1 = -1, mx2 = -1;
    for (int v : adj[u]) {
        if (v == p) continue;
        total += down[v];
        if (down[v] > mx1) {
            mx2 = mx1;
            mx1 = down[v];
        } else if (down[v] > mx2) {
            mx2 = down[v];
        }
    }
    ans[u] = total;

    int use = max(up[u], w[u]);
    for (int v : adj[u]) {
        if (v == p) continue;
        int best = (down[v] == mx1) ? mx2 : mx1;
        up[v] = max(use, best);
        dfsUp(v, u);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    w.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> w[i];
    }
    adj.assign(n, {});
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        --u; --v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    down.assign(n, 0);
    up.assign(n, 0);
    ans.assign(n, 0);

    dfsDown(0, -1);
    dfsUp(0, -1);

    for (int i = 0; i < n; i++) {
        cout << ans[i] << (i + 1 < n ? ' ' : '\n');
    }
    return 0;
}
```
