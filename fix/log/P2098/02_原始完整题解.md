# 思维
对于单个价值i*m+j,i变化的价值是m,j的是1,因为m>=1，所以优先考虑i变化也就是向下走，其次在是向右走，那么还有多余的步数，考虑上下横跳，或者左右横跳，同理，左右横跳大于上下横跳，也就是当m=1时上下横跳，其余左右横跳即可
## 代码如下
### cpp
```cpp
#include <bits/stdc++.h>
using namespace std;
#define int long long
int fun(int n,int a1,int d){
	return n*(2*a1+(n-1)*d)/2;
}
signed main() {
	ios::sync_with_stdio(0);
	cin.tie(0);
	int t;
	cin>>t;
	while(t--){
		int n,m,k;
		cin>>n>>m>>k;
		int res=0;
		if(k>=n-1){//a1 =0*m+0 a2=1*m+0
			res+=fun(n-1,m,m);
			k-=n-1;
		}
		else{//到不了(n-1,1)
			res+=fun(k,m,m);
			k-=k;
		}
		/*(n-1,0)-(n-1,m-1)*/
		if(k>=m-1){//a1 =(n-1)*m+0 a2=(n-1)*m+1
			res+=fun(m-1,(n-1)*m+1,1);
			k-=m-1;
		}
		else{//到不了(n-1,1)
			res+=fun(k,(n-1)*m+1,1);
			k-=k;
		}
		int sum1=0,sum2=0;
		if(m!=1){
			sum2=((n-1)*m+m-1);
			sum1=((n-1)*m+m-2);
		}
		else{
			sum2=((n-1)*m+m-1);
			sum1=((n-2)*m+m-1);
		}
		res+=(k+1)/2*sum1;
		res+=(k/2)*sum2;
		cout<<res<<'\n';
	}
	
	return 0;
}
```
### python
```py
import sys
import string
import random

def fun(n, a1, d):
    return n * (2 * a1 + (n - 1) * d) // 2

def random_lowercase_string(n):
    return ''.join(random.choice(string.ascii_lowercase) for _ in range(n))

def msd(t, n, mx):
    res = []
    for _ in range(t):
        q = random.randint(1, mx)
        res.append([n, q])
        res.append(random_lowercase_string(n))
        for _ in range(q):
            op = random.randint(1, 2)
            if op == 1:
                id = random.randint(1, n)
                c = random.choice(string.ascii_lowercase)
                res.append([op, id, c])
            else:
                if n == 1:
                    l = r = 1
                else:
                    l = random.randint(1, n // 2)
                    r = random.randint(n // 2, n)
                res.append([op, l, r])
    return res

def main():
    input = sys.stdin.read().split()
    idx = 0
    t = int(input[idx])
    idx += 1
    for _ in range(t):
        n = int(input[idx])
        m = int(input[idx + 1])
        k = int(input[idx + 2])
        idx += 3
        res = 0
        if k >= n - 1:
            res += fun(n - 1, m, m)
            k -= n - 1
        else:
            res += fun(k, m, m)
            k = 0
        if k >= m - 1:
            res += fun(m - 1, (n - 1) * m + 1, 1)
            k -= m - 1
        else:
            res += fun(k, (n - 1) * m + 1, 1)
            k = 0
        if m != 1:
            sum2 = (n - 1) * m + m - 1
            sum1 = (n - 1) * m + m - 2
        else:
            sum2 = (n - 1) * m + m - 1
            sum1 = (n - 2) * m + m - 1
        res += ((k + 1) // 2) * sum1
        res += (k // 2) * sum2
        print(res)

if __name__ == "__main__":
    main()
```

### java
```java
import java.io.*;
import java.util.*;

public class Main {
    public static long fun(long n, long a1, long d){
        return n * (2 * a1 + (n - 1) * d) / 2;
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter pw = new PrintWriter(System.out);
        StringTokenizer st = new StringTokenizer(br.readLine());
        int t = Integer.parseInt(st.nextToken());
        while(t-- > 0){
            st = new StringTokenizer(br.readLine());
            long n = Long.parseLong(st.nextToken());
            long m = Long.parseLong(st.nextToken());
            long k = Long.parseLong(st.nextToken());
            long res = 0;
            if(k >= n - 1){
                res += fun(n - 1, m, m);
                k -= n - 1;
            }
            else{
                res += fun(k, m, m);
                k = 0;
            }
            if(k >= m - 1){
                res += fun(m - 1, (n - 1) * m + 1, 1);
                k -= m - 1;
            }
            else{
                res += fun(k, (n - 1) * m + 1, 1);
                k = 0;
            }
            long sum1 = 0, sum2 = 0;
            if(m != 1){
                sum2 = (n - 1) * m + m - 1;
                sum1 = (n - 1) * m + m - 2;
            }
            else{
                sum2 = (n - 1) * m + m - 1;
                sum1 = (n - 2) * m + m - 1;
            }
            res += ((k + 1) / 2) * sum1;
            res += (k / 2) * sum2;
            pw.println(res);
        }
        pw.flush();
    }
}
```
