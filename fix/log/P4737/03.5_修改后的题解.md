## 解题思路

题目定义：在给定的单词序列 $s_1, s_2, \dots, s_n$ 中，一个单词 $s_i$ 是“可组合的”，当且仅当存在两个下标 $j$ 和 $k$（$j \neq k$），使得 $s_j + s_k = s_i$。等价地，存在一个切分位置 $p$（$1 \le p < |s_i|$），将 $s_i$ 分成前缀 $x = s_i[0:p]$ 和后缀 $y = s_i[p:]$，且 $x$ 和 $y$ 都曾在列表中出现过；若 $x = y$，则要求该字符串在列表中至少出现两次（以保证来自不同下标）。

根据这个定义，我们可以对每组测试数据执行如下操作：

1. **统计词频**  
   遍历该组所有单词，用哈希表 `cnt` 记录每个字符串的出现次数。

2. **逐个判断单词是否可组合**  
   对于每一个单词 $s$，枚举所有可能的切分点 $p$（$1 \le p < |s|$）：
   - 令前缀 `left = s[0:p]`，后缀 `right = s[p:]`。
   - 如果 `left != right`，只需检查 `cnt[left] > 0` 且 `cnt[right] > 0`。
   - 如果 `left == right`，需要检查 `cnt[left] >= 2`，以保证能够取到两个不同位置的相同单词。
   - 只要找到任何一个满足条件的切分点，该单词即为“可组合的”，计数并跳出循环。

3. **输出每组答案**  
   统计该组中可组合的单词个数并输出。

因为每个单词的所有切分点都会被枚举一次，而每次查询都在哈希表中进行，均摊 $O(1)$，所以总时间复杂度与所有字符串的长度之和成正比。

## 复杂度分析

设所有测试数据中字符串的总长度之和为 $L = \sum |s_i|$。  
对于每个单词，最多枚举 $|s|-1$ 次切分，因此总枚举次数为 $O(L)$。每次切分后哈希表查询为 $O(1)$。  
- 时间复杂度：$O(L)$  
- 空间复杂度：$O(L)$，用于存储词频哈希表。

## 代码实现

### Python

```python
import sys

def solve(strings):
    cnt = {}
    for s in strings:
        cnt[s] = cnt.get(s, 0) + 1
    
    ans = 0
    for s in strings:
        m = len(s)
        ok = False
        for p in range(1, m):
            left = s[:p]
            right = s[p:]
            if left != right:
                if cnt.get(left, 0) > 0 and cnt.get(right, 0) > 0:
                    ok = True
                    break
            else:
                if cnt.get(left, 0) >= 2:
                    ok = True
                    break
        if ok:
            ans += 1
    return ans

def main():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    T = int(next(it))
    out_lines = []
    for _ in range(T):
        n = int(next(it))
        words = [next(it) for _ in range(n)]
        out_lines.append(str(solve(words)))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static int solve(String[] strings) {
        Map<String, Integer> cnt = new HashMap<>();
        for (String s : strings) {
            cnt.put(s, cnt.getOrDefault(s, 0) + 1);
        }
        
        int ans = 0;
        for (String s : strings) {
            int m = s.length();
            boolean ok = false;
            for (int p = 1; p < m; p++) {
                String left = s.substring(0, p);
                String right = s.substring(p);
                if (!left.equals(right)) {
                    if (cnt.getOrDefault(left, 0) > 0 && cnt.getOrDefault(right, 0) > 0) {
                        ok = true;
                        break;
                    }
                } else {
                    if (cnt.getOrDefault(left, 0) >= 2) {
                        ok = true;
                        break;
                    }
                }
            }
            if (ok) ans++;
        }
        return ans;
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine());
            String[] words = new String[n];
            for (int i = 0; i < n; i++) {
                words[i] = br.readLine();
            }
            sb.append(solve(words)).append('\n');
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <unordered_map>
using namespace std;

int solve(const vector<string>& strings) {
    unordered_map<string, int> cnt;
    for (const string& s : strings) {
        cnt[s]++;
    }
    
    int ans = 0;
    for (const string& s : strings) {
        int m = s.size();
        bool ok = false;
        for (int p = 1; p < m; ++p) {
            string left = s.substr(0, p);
            string right = s.substr(p);
            if (left != right) {
                if (cnt.count(left) && cnt.count(right)) {
                    ok = true;
                    break;
                }
            } else {
                if (cnt[left] >= 2) {
                    ok = true;
                    break;
                }
            }
        }
        if (ok) ans++;
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        vector<string> words(n);
        for (int i = 0; i < n; ++i) {
            cin >> words[i];
        }
        cout << solve(words) << '\n';
    }
    return 0;
}
```
