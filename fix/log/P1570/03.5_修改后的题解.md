## 解题思路

用两张哈希表模拟接口的重载登记与调用：

- `sig`：键为「接口名 + 按顺序拼接的参数类型」，值为是否已登记。
- `name`：键为接口名，值为该名字是否出现过。

每次读入操作类型 $op$ 和一行字符串，在 `'('` 处切开，分别解析名字与参数列表。

**登记（$op=1$）**  
从空格分开返回类型与接口名，再按逗号拆出每个参数，只保留参数类型（每个片段的第一个单词）。把「名字 + 各类型」拼成签名。若签名已在 `sig` 中，输出 `redefinition of xxx.`；否则写入 `sig` 与 `name`，输出 `Yes.`。

**调用（$op=2$）**  
字符串左侧即为接口名，括号内按逗号拆出类型。同样拼成签名。若签名命中，输出 `Yes.`；否则若名字在 `name` 中出现过，输出 `xxx was not declared in this scope.`；否则输出 `There is no function xxx.`。

空参数列表（括号内为空）视为类型列表为空，签名就是接口名本身。判断重复时不看返回类型和参数名，因此 `void emit(int id)` 与 `String emit(int value)` 会被判为同一签名。

## 复杂度分析

时间复杂度 $O(q \cdot L)$，其中 $L$ 为单条字符串长度。空间复杂度 $O(q \cdot L)$。

## 代码实现

### Python

```python
q = int(input())
sig = dict()
name = dict()

for _ in range(q):
    op = int(input())
    if op == 1:
        s = input().split('(')
        pre = s[0].split()
        suf = s[1][:-1].split(',')
        key = pre[1]
        types = []
        for part in suf:
            if part == '':
                continue
            types.append(part.split()[0])
        for t in types:
            key += t
        if sig.get(key, 0) == 0:
            sig[key] = 1
            name[pre[1]] = 1
            print('Yes.')
        else:
            print('redefinition of ' + pre[1] + '.')
    else:
        s = input().split('(')
        parts = s[1][:-1].split(',')
        key = s[0]
        for part in parts:
            if part == '':
                continue
            key += part.split()[0]
        if sig.get(key, 0) != 0:
            print('Yes.')
        else:
            if name.get(s[0], 0) != 0:
                print(s[0] + ' was not declared in this scope.')
            else:
                print('There is no function ' + s[0] + '.')
```

### Java

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(reader.readLine());
        Map<String, List<String[]>> map = new HashMap<>();
        while (q-- > 0) {
            int op = Integer.parseInt(reader.readLine());
            String s = reader.readLine();
            String[] sb = s.split("[\\s+,()]");
            if (op == 1) {
                if (!map.containsKey(sb[1])) {
                    List<String[]> list = new ArrayList<>();
                    list.add(sb);
                    map.put(sb[1], list);
                    System.out.println("Yes.");
                } else {
                    List<String[]> list = map.get(sb[1]);
                    boolean dup = false;
                    for (String[] tmp : list) {
                        boolean mismatch = false;
                        if (tmp.length == sb.length) {
                            for (int i = 2; i < sb.length; i += 2) {
                                if (!sb[i].equals(tmp[i])) {
                                    mismatch = true;
                                    break;
                                }
                            }
                            if (!mismatch) {
                                dup = true;
                                break;
                            }
                        }
                    }
                    if (dup) {
                        System.out.println("redefinition of " + sb[1] + ".");
                    } else {
                        System.out.println("Yes.");
                        list.add(sb);
                    }
                }
            } else {
                if (map.containsKey(sb[0])) {
                    List<String[]> list = map.get(sb[0]);
                    boolean hit = false;
                    for (String[] tmp : list) {
                        boolean mismatch = false;
                        if (tmp.length == sb.length * 2) {
                            for (int i = 1; i < sb.length; i++) {
                                if (!sb[i].equals(tmp[i * 2])) {
                                    mismatch = true;
                                    break;
                                }
                            }
                            if (!mismatch) {
                                hit = true;
                                break;
                            }
                        }
                    }
                    if (hit) {
                        System.out.println("Yes.");
                    } else {
                        System.out.println(sb[0] + " was not declared in this scope.");
                    }
                } else {
                    System.out.println("There is no function " + sb[0] + ".");
                }
            }
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <sstream>
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int q;
    cin >> q;
    unordered_map<string, int> sig, name;
    for (int i = 0; i < q; i++) {
        int op;
        cin >> op;
        cin.get();
        string line;
        getline(cin, line);
        if (op == 1) {
            size_t lp = line.find('(');
            string left = line.substr(0, lp);
            string inside = line.substr(lp + 1, line.size() - lp - 2);
            istringstream ls(left);
            string ret, fname;
            ls >> ret >> fname;
            string key = fname;
            if (!inside.empty()) {
                stringstream ss(inside);
                string part;
                while (getline(ss, part, ',')) {
                    if (part.empty()) {
                        continue;
                    }
                    istringstream ps(part);
                    string ty;
                    ps >> ty;
                    key += ty;
                }
            }
            if (!sig.count(key)) {
                sig[key] = 1;
                name[fname] = 1;
                cout << "Yes.\n";
            } else {
                cout << "redefinition of " << fname << ".\n";
            }
        } else {
            size_t lp = line.find('(');
            string fname = line.substr(0, lp);
            string inside = line.substr(lp + 1, line.size() - lp - 2);
            string key = fname;
            if (!inside.empty()) {
                stringstream ss(inside);
                string part;
                while (getline(ss, part, ',')) {
                    if (part.empty()) {
                        continue;
                    }
                    istringstream ps(part);
                    string ty;
                    ps >> ty;
                    key += ty;
                }
            }
            if (sig.count(key)) {
                cout << "Yes.\n";
            } else if (name.count(fname)) {
                cout << fname << " was not declared in this scope.\n";
            } else {
                cout << "There is no function " << fname << ".\n";
            }
        }
    }
    return 0;
}
```
