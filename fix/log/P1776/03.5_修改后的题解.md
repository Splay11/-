## 解题思路

本题的目标是通过最少的变换次数，使得长度为偶数 $n$ 的序列中，奇数的个数与偶数的个数相等（即各有 $m = n/2$ 个）。每次变换可以对任意一个数字执行以下两种操作之一：乘以 $2$，或除以 $2$ 并向下取整。

关键观察和算法步骤：

1. **统计初始奇偶个数**  
   - 遍历整个序列，用 `cnt_odd` 记录当前奇数的个数，用 `cnt_even` 记录当前偶数的个数。  
   - 若 `cnt_odd == cnt_even`，说明已经满足要求，最少变换次数为 $0$。

2. **处理奇数多于偶数的情况（`cnt_odd > cnt_even`）**  
   - 由于奇数乘以 $2$ 一定变成偶数，且每次变换只乘一次 $2$，因此将一个奇数变为偶数只需要 $1$ 次操作。  
   - 需要将 $(cnt_odd - m)$ 个奇数变为偶数，因此最少操作次数就等于 $cnt_odd - m$。

3. **处理偶数多于奇数的情况（`cnt_even > cnt_odd`）**  
   - 将偶数变为奇数需要通过“除以 $2$ 并向下取整”的操作，连续除 $2$ 直到得到奇数。每个偶数都有一个确定的最小操作次数 `cost`。  
   - 遍历所有偶数，对于每个偶数 `v`，重复 `v //= 2` 并累加操作次数，直到 `v` 变为奇数，记录该 `cost`。将这些 `cost` 存入列表 `costs` 中。  
   - 为了用最少的操作次数补齐缺少的奇数，应优先选择那些转化为奇数所需操作次数最少的偶数。将 `costs` 排序，取最小的 $(cnt_even - m)$ 个 `cost`，它们的和即为所求答案。

这样分类讨论即可得到全局最少变换次数。

## 复杂度分析

- 统计每个偶数变为奇数所需操作次数时，每个偶数需要 $O(\log v)$ 的时间（最多约 $30$ 次除法）。因此预处理所有偶数的 `cost` 的总时间为 $O(n \log U)$，其中 $U = 10^9$。
- 对 `costs` 进行排序需要 $O(n \log n)$ 的时间。
- 整体时间复杂度为 $O(n \log n)$。
- 额外使用了 $O(n)$ 的空间存储 `costs` 列表，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))

    cnt_odd = 0
    cnt_even = 0
    costs = []

    for v in a:
        if v % 2 == 1:
            cnt_odd += 1
        else:
            cnt_even += 1
            # 计算将该偶数通过除以2变成奇数所需的操作次数
            c = 0
            while v % 2 == 0:
                v //= 2
                c += 1
            costs.append(c)

    m = n // 2

    if cnt_odd == cnt_even:
        print(0)
    elif cnt_odd > cnt_even:
        # 奇数多，将超出的奇数变为偶数，每个需要 1 次乘 2 操作
        print(cnt_odd - m)
    else:
        # 偶数多，选择所需操作次数最少的若干偶数变为奇数
        costs.sort()
        need = cnt_even - m
        print(sum(costs[:need]))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];

        int cntOdd = 0, cntEven = 0;
        List<Integer> costs = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            int v = a[i];
            if (v % 2 == 1) {
                cntOdd++;
            } else {
                cntEven++;
                int c = 0;
                while (v % 2 == 0) {
                    v /= 2;
                    c++;
                }
                costs.add(c);
            }
        }

        int m = n / 2;

        if (cntOdd == cntEven) {
            System.out.println(0);
        } else if (cntOdd > cntEven) {
            // 奇数多，将多余的奇数乘2变成偶数
            System.out.println(cntOdd - m);
        } else {
            // 偶数多，挑选操作次数最小的偶数转为奇数
            Collections.sort(costs);
            int need = cntEven - m;
            long ans = 0;
            for (int i = 0; i < need; i++) {
                ans += costs.get(i);
            }
            System.out.println(ans);
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n;
    cin >> n;
    vector<int> a(n);

    int cnt_odd = 0, cnt_even = 0;
    vector<int> costs;

    for (int i = 0; i < n; i++) {
        cin >> a[i];
        int v = a[i];
        if (v % 2 == 1) {
            cnt_odd++;
        } else {
            cnt_even++;
            int c = 0;
            while (v % 2 == 0) {
                v /= 2;
                c++;
            }
            costs.push_back(c);
        }
    }

    int m = n / 2;

    if (cnt_odd == cnt_even) {
        cout << 0 << '\n';
    } else if (cnt_odd > cnt_even) {
        // 奇数多，每个多余奇数乘2一次即可
        cout << cnt_odd - m << '\n';
    } else {
        // 偶数多，取操作次数最小的若干偶数
        sort(costs.begin(), costs.end());
        int need = cnt_even - m;
        long long ans = 0;
        for (int i = 0; i < need; i++) {
            ans += costs[i];
        }
        cout << ans << '\n';
    }

    return 0;
}
```
