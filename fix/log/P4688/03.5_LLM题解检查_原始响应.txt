## 解题思路

题目给出了样本之间的“同频”关系，该关系满足自反性、对称性和传递性，因此构成一个等价关系。所有样本被划分为若干个同频等价类，每个等价类内部的样本频率完全相同，不同等价类的频率互不相同。

实验人员确定**至少存在一个零频样本**，并已知：
- 若某个样本是零频样本，则它所在的整个等价类内所有样本都是零频样本（因为同一等价类频率相同）。
- 由于不同等价类频率不同，至多只能有一个等价类频率为 0，即**零频样本必然全部来自恰好一个等价类**。

因此，在所有可能情形下：
- 零频样本数量可以是任意一个等价类的大小。
- 最小值即为所有等价类中大小最小的那个值。
- 最大值即为所有等价类中大小最大的那个值。

具体算法步骤：
1. 读入样本数 `n` 和 `n×n` 的同频矩阵 `a`，其中 `a[i][j]=1` 表示样本 `i` 与样本 `j` 同频。
2. 使用 DFS（或 BFS）遍历所有样本，统计每个等价类（连通块）的大小。
3. 遍历过程中，对每个等价类的大小记录当前的最小值和最大值。
4. 输出这两个值。

## 复杂度分析

- 时间复杂度：$O(n^2)$。因为需要遍历邻接矩阵中所有元素，每个样本和关系最多被检查常数次。
- 空间复杂度：$O(n)$。需要长度为 $n$ 的访问标记数组，以及 DFS 栈（或 BFS 队列）在最坏情况下的空间均为 $O(n)$。

## 代码实现

### Python

```python
# 零频样本计数
# 使用 DFS 统计等价类大小

def dfs(start, a, vis, n):
    stack = [start]
    vis[start] = True
    cnt = 0

    while stack:
        u = stack.pop()
        cnt += 1
        # 枚举所有样本，找与当前样本同频的点
        for v in range(n):
            if a[u][v] == 1 and not vis[v]:
                vis[v] = True
                stack.append(v)

    return cnt


def solve(n, a):
    vis = [False] * n
    mn = n      # 初始化为可能的最大值
    mx = 1      # 初始化为可能的最小值

    # 枚举每个等价类
    for i in range(n):
        if not vis[i]:
            size = dfs(i, a, vis, n)
            mn = min(mn, size)
            mx = max(mx, size)

    return mn, mx


def main():
    n = int(input().strip())
    a = [list(map(int, input().split())) for _ in range(n)]

    mn, mx = solve(n, a)

    print(mn, mx)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    // 使用 DFS 统计一个等价类的大小
    public static int dfs(int start, int[][] a, boolean[] vis, int n) {
        Stack<Integer> stack = new Stack<>();
        stack.push(start);
        vis[start] = true;
        int cnt = 0;

        while (!stack.isEmpty()) {
            int u = stack.pop();
            cnt++;

            // 枚举所有样本，找与当前样本同频的点
            for (int v = 0; v < n; v++) {
                if (a[u][v] == 1 && !vis[v]) {
                    vis[v] = true;
                    stack.push(v);
                }
            }
        }

        return cnt;
    }

    // 求解零频样本数量的最小值和最大值
    public static int[] solve(int n, int[][] a) {
        boolean[] vis = new boolean[n];
        int mn = n;
        int mx = 1;

        // 枚举每个等价类
        for (int i = 0; i < n; i++) {
            if (!vis[i]) {
                int size = dfs(i, a, vis, n);
                mn = Math.min(mn, size);
                mx = Math.max(mx, size);
            }
        }

        return new int[]{mn, mx};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[][] a = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] = sc.nextInt();
            }
        }

        int[] ans = solve(n, a);

        System.out.println(ans[0] + " " + ans[1]);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <stack>
using namespace std;

// 使用 DFS 统计一个等价类的大小
int dfs(int start, const vector<vector<int>>& a, vector<bool>& vis, int n) {
    stack<int> st;
    st.push(start);
    vis[start] = true;
    int cnt = 0;

    while (!st.empty()) {
        int u = st.top();
        st.pop();
        cnt++;

        // 枚举所有样本，找与当前样本同频的点
        for (int v = 0; v < n; v++) {
            if (a[u][v] == 1 && !vis[v]) {
                vis[v] = true;
                st.push(v);
            }
        }
    }

    return cnt;
}

// 求解零频样本数量的最小值和最大值
pair<int, int> solve(int n, const vector<vector<int>>& a) {
    vector<bool> vis(n, false);
    int mn = n;
    int mx = 1;

    // 枚举每个等价类
    for (int i = 0; i < n; i++) {
        if (!vis[i]) {
            int size = dfs(i, a, vis, n);
            mn = min(mn, size);
            mx = max(mx, size);
        }
    }

    return {mn, mx};
}

int main() {
    int n;
    cin >> n;
    vector<vector<int>> a(n, vector<int>(n));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> a[i][j];
        }
    }

    pair<int, int> ans = solve(n, a);

    cout << ans.first << " " << ans.second << endl;

    return 0;
}
```
