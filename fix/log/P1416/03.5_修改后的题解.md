## 解题思路

这是带括号与乘除优先级的中缀表达式求值。使用双栈模拟：一个栈存操作数，一个栈存运算符。

从左到右扫描字符串，跳过空格后分类处理：

- 遇到数字：解析完整非负整数并压入操作数栈；
- 遇到 `(`：压入运算符栈；
- 遇到 `)`：不断弹出运算符并计算，直到弹出对应的 `(`；
- 遇到 `+`、`-`、`*`、`/`：当栈顶运算符优先级不低于当前运算符时，先弹出并计算，再把当前运算符入栈。

优先级规定：`*`、`/` 为 `2`，`+`、`-` 为 `1`，`(` 不参与出栈比较。扫描结束后把剩余运算符全部计算完，栈顶即为答案。

除法按题意向上取整：设被除数为 $a$、除数为 $b$（$b>0$），结果为 $\lceil a/b \rceil$。实现时对非负 $a$ 使用 $(a+b-1)/b$，对负 $a$ 使用向零取整（与向上取整一致）。

## 复杂度分析

每个字符入栈出栈至多常数次。

- 时间复杂度：$O(n)$。
- 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
def ceil_div(a, b):
    if b < 0:
        a, b = -a, -b
    if a >= 0:
        return (a + b - 1) // b
    return -((-a) // b)


def precedence(op):
    if op in "+-":
        return 1
    if op in "*/":
        return 2
    return 0


def apply_op(values, ops):
    b = values.pop()
    a = values.pop()
    op = ops.pop()
    if op == "+":
        values.append(a + b)
    elif op == "-":
        values.append(a - b)
    elif op == "*":
        values.append(a * b)
    else:
        values.append(ceil_div(a, b))


def evaluate(expr):
    values = []
    ops = []
    i = 0
    n = len(expr)
    while i < n:
        ch = expr[i]
        if ch == " ":
            i += 1
            continue
        if ch == "(":
            ops.append(ch)
        elif ch == ")":
            while ops[-1] != "(":
                apply_op(values, ops)
            ops.pop()
        elif ch in "+-*/":
            while ops and ops[-1] != "(" and precedence(ops[-1]) >= precedence(ch):
                apply_op(values, ops)
            ops.append(ch)
        else:
            x = 0
            while i < n and expr[i].isdigit():
                x = x * 10 + ord(expr[i]) - 48
                i += 1
            values.append(x)
            i -= 1
        i += 1
    while ops:
        apply_op(values, ops)
    return values[0]


print(evaluate(input().rstrip("\n")))
```

### Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Main {
    static int ceilDiv(int a, int b) {
        if (b < 0) {
            a = -a;
            b = -b;
        }
        if (a >= 0) {
            return (a + b - 1) / b;
        }
        return a / b;
    }

    static int precedence(char op) {
        if (op == '+' || op == '-') {
            return 1;
        }
        if (op == '*' || op == '/') {
            return 2;
        }
        return 0;
    }

    static void applyOp(Deque<Integer> values, Deque<Character> ops) {
        int b = values.pollLast();
        int a = values.pollLast();
        char op = ops.pollLast();
        if (op == '+') {
            values.addLast(a + b);
        } else if (op == '-') {
            values.addLast(a - b);
        } else if (op == '*') {
            values.addLast(a * b);
        } else {
            values.addLast(ceilDiv(a, b));
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String expr = sc.nextLine();
        Deque<Integer> values = new ArrayDeque<>();
        Deque<Character> ops = new ArrayDeque<>();
        int n = expr.length();
        for (int i = 0; i < n; i++) {
            char ch = expr.charAt(i);
            if (ch == ' ') {
                continue;
            }
            if (ch == '(') {
                ops.addLast(ch);
            } else if (ch == ')') {
                while (ops.peekLast() != '(') {
                    applyOp(values, ops);
                }
                ops.pollLast();
            } else if (ch == '+' || ch == '-' || ch == '*' || ch == '/') {
                while (!ops.isEmpty() && ops.peekLast() != '('
                        && precedence(ops.peekLast()) >= precedence(ch)) {
                    applyOp(values, ops);
                }
                ops.addLast(ch);
            } else {
                int x = 0;
                while (i < n && Character.isDigit(expr.charAt(i))) {
                    x = x * 10 + (expr.charAt(i) - '0');
                    i++;
                }
                values.addLast(x);
                i--;
            }
        }
        while (!ops.isEmpty()) {
            applyOp(values, ops);
        }
        System.out.println(values.peekLast());
        sc.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int ceil_div(int a, int b) {
    if (b < 0) {
        a = -a;
        b = -b;
    }
    if (a >= 0) {
        return (a + b - 1) / b;
    }
    return a / b;
}

int precedence(char op) {
    if (op == '+' || op == '-') {
        return 1;
    }
    if (op == '*' || op == '/') {
        return 2;
    }
    return 0;
}

void apply_op(vector<int> &values, vector<char> &ops) {
    int b = values.back();
    values.pop_back();
    int a = values.back();
    values.pop_back();
    char op = ops.back();
    ops.pop_back();
    if (op == '+') {
        values.push_back(a + b);
    } else if (op == '-') {
        values.push_back(a - b);
    } else if (op == '*') {
        values.push_back(a * b);
    } else {
        values.push_back(ceil_div(a, b));
    }
}

int main() {
    string expr;
    getline(cin, expr);
    vector<int> values;
    vector<char> ops;
    int n = (int)expr.size();
    for (int i = 0; i < n; i++) {
        char ch = expr[i];
        if (ch == ' ') {
            continue;
        }
        if (ch == '(') {
            ops.push_back(ch);
        } else if (ch == ')') {
            while (ops.back() != '(') {
                apply_op(values, ops);
            }
            ops.pop_back();
        } else if (ch == '+' || ch == '-' || ch == '*' || ch == '/') {
            while (!ops.empty() && ops.back() != '(' && precedence(ops.back()) >= precedence(ch)) {
                apply_op(values, ops);
            }
            ops.push_back(ch);
        } else {
            int x = 0;
            while (i < n && isdigit(expr[i])) {
                x = x * 10 + (expr[i] - '0');
                i++;
            }
            values.push_back(x);
            i--;
        }
    }
    while (!ops.empty()) {
        apply_op(values, ops);
    }
    cout << values.back() << "\n";
    return 0;
}
```
