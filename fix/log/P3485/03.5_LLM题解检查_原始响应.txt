## 解题思路

本题要求在不超过上限 $N$ 的所有平稳数（十进制表示从左向右每一位数字单调不减）中，统计各位数字之和能被 $k$ 整除的数的个数。由于 $N$ 高达 $10^{18}$ 且查询次数 $Q \le 500$，需要预处理后高效回答每个 $k$ 的查询。

核心思路是 **数位 DP + 预处理和向量**：

1. **定义状态**  
   设 $ways[L][mn][s]$ 表示：长度为 $L$（允许前导零）、所有位置数字均满足单调不减、且第一位数字至少为 $mn$ 的数列中，数位和恰好为 $s$ 的方案数。  
   其中 $L \in [0, \text{len}(N)]$，$mn, x \in [0,9]$，$s$ 取值范围 $0 \sim 9L$。

2. **状态转移**  
   - 边界：$ways[0][mn][0] = 1$，其余 $0$；
   - 对于 $L \ge 1$，当前位取 $x$（$x \ge mn$），剩余 $L-1$ 位的最小允许值变为 $x$，故有：
     \[
     ways[L][mn][s] = \sum_{x=mn}^{9} \sum_{s_R} ways[L-1][x][s_R] \cdot [x + s_R == s]
     \]
     实现时直接对 $ways[L-1][x]$ 向量做偏移累加即可。预处理规模大约 $18 \times 10 \times (9 \times 18)$ 级别，非常小。

3. **回答查询**  
   对于给定的模数 $k$，统计答案分为两部分（设 $N$ 有 $n$ 位，各位数字依次为 $d_0,d_1,\dots,d_{n-1}$）：
   
   - **位数少于 $n$ 的平稳数**：枚举长度 $L \in [1, n-1]$，首位数字 $f \in [1,9]$，剩余 $L-1$ 位的方案由 $ways[L-1][f]$ 给出。在这些方案中，数位和为 $s_R$，总数为 $f + s_R$。需要满足 $(f + s_R) \bmod k = 0$，即 $s_R \bmod k = (-f) \bmod k$。通过按模聚合 $ways$ 向量可快速计数。
   
   - **位数等于 $n$ 的平稳数**：使用数位 DP 处理与 $N$ 等长的平稳数。维护当前前缀的最小允许值 $mn$（初始为 $1$，因为首位不能为 $0$ 以保证长度）以及前缀数字和模 $k$ 的值 $S$。从左到右扫描第 $i$ 位：
     * 若当前位选择数字 $x \in [mn, d_i-1]$，那么剩下 $rem = n-i-1$ 位由 $ways[rem][x]$ 表示，其数位和 $s_R$ 需要满足 $(S + x + s_R) \bmod k = 0$；
     * 然后令 $x = d_i$，若 $d_i < mn$ 则不可能继续等于前缀，直接结束；否则更新 $S = (S + d_i) \bmod k$，$mn = d_i$ 继续下一轮。
   - 若最后能走完整个数字且最终 $S \bmod k = 0$，说明 $N$ 本身也是一个合法的平稳数（且满足模条件），答案加 $1$。

   “按模聚合”函数 $sum\_by\_mod(vec, d, r)$ 用于求出向量 $vec$ 中下标模 $d$ 余 $r$ 的所有元素之和。由于 $k \le 200$ 且 $vec$ 长度不超过 $162$，每次查询的复杂度很低。

## 复杂度分析

- **预处理**：状态数约为 $\sum_{L=0}^{18} 10 \cdot (9L+1) \approx 1.7\times 10^4$，每次转移枚举 $x$ 和 $s_R$，整体复杂度 $O(\sum_{L} 10 \cdot 10 \cdot 9L) \approx O(10^5)$，非常快。
- **单次查询**：需要遍历位数 $O(n)$，每个位置枚举 $x$ 并调用 $sum\_by\_mod$ 进行 $O(9n/k)$ 次加法。最坏情况下单次查询约 $O(10 \cdot n \cdot 162)$ 操作，所有 $Q$ 次查询总复杂度 $O(Q \cdot n \cdot 162)$。对于 $Q \le 500,\ n \le 18$，完全可行。
- **空间复杂度**：存储 $ways$ 三维向量，大小为 $O(L_{max} \cdot 10 \cdot 9L_{max}) \approx O(10^4)$，可忽略不计。

## 代码实现

### Python

```python
import sys

def sum_by_mod(vec, d, r):
    """返回向量 vec 中下标模 d 余 r 的元素之和"""
    s = 0
    for i in range(r, len(vec), d):
        s += vec[i]
    return s

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    mstr = data[0]          # 上限 N 的字符串形式
    Q = int(data[1])
    queries = list(map(int, data[2:2+Q]))
    digits = [int(c) for c in mstr]
    n = len(digits)

    max_rem = max(0, n - 1)   # 最大剩余位数
    # ways[L][mn] 是一个长度为 9*L + 1 的列表
    ways = [[None] * 10 for _ in range(max_rem + 1)]
    for L in range(max_rem + 1):
        for mn in range(10):
            ways[L][mn] = [0] * (9 * L + 1)
    for mn in range(10):
        ways[0][mn][0] = 1

    for L in range(1, max_rem + 1):
        for mn in range(10):
            cur = ways[L][mn]
            for x in range(mn, 10):
                prev = ways[L-1][x]
                for sR, cnt in enumerate(prev):
                    cur[x + sR] += cnt

    def solve_one(k):
        ans = 0
        # 1. 位数小于 n 的平稳数
        for L in range(1, n):
            for f in range(1, 10):
                vec = ways[L-1][f]
                need = (-f) % k
                ans += sum_by_mod(vec, k, need)

        # 2. 位数等于 n 的平稳数
        mn = 1       # 第一位至少是 1
        Smod = 0     # 前缀和模 k
        ok = True
        for i, bound in enumerate(digits):
            for x in range(mn, bound):
                rem = n - i - 1
                vec = ways[rem][x]
                need = (-(Smod + x)) % k
                ans += sum_by_mod(vec, k, need)
            if bound < mn:
                ok = False
                break
            Smod = (Smod + bound) % k
            mn = bound

        if ok and Smod % k == 0:
            ans += 1
        return ans

    out = [str(solve_one(k)) for k in queries]
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static long sumByMod(long[] vec, int d, int r) {
        long ans = 0;
        for (int i = r; i < vec.length; i += d) {
            ans += vec[i];
        }
        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        String mstr = st.nextToken();          // 上限 N
        int Q = Integer.parseInt(st.nextToken());
        int[] queries = new int[Q];
        for (int i = 0; i < Q; i++) {
            queries[i] = Integer.parseInt(br.readLine().trim());
        }
        int n = mstr.length();
        int[] digits = new int[n];
        for (int i = 0; i < n; i++) {
            digits[i] = mstr.charAt(i) - '0';
        }

        int maxRem = Math.max(0, n - 1);
        long[][][] ways = new long[maxRem + 1][10][];
        for (int L = 0; L <= maxRem; L++) {
            for (int mn = 0; mn <= 9; mn++) {
                ways[L][mn] = new long[9 * L + 1];
            }
        }
        for (int mn = 0; mn <= 9; mn++) {
            ways[0][mn][0] = 1;
        }
        for (int L = 1; L <= maxRem; L++) {
            for (int mn = 0; mn <= 9; mn++) {
                long[] cur = ways[L][mn];
                for (int x = mn; x <= 9; x++) {
                    long[] prev = ways[L - 1][x];
                    for (int sR = 0; sR < prev.length; sR++) {
                        cur[x + sR] += prev[sR];
                    }
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int qi = 0; qi < Q; qi++) {
            int k = queries[qi];
            long ans = 0;
            // 1. 位数小于 n
            for (int L = 1; L < n; L++) {
                for (int f = 1; f <= 9; f++) {
                    long[] vec = ways[L - 1][f];
                    int need = ((k - (f % k)) % k);
                    ans += sumByMod(vec, k, need);
                }
            }
            // 2. 位数等于 n
            int mn = 1;
            int Smod = 0;
            boolean ok = true;
            for (int i = 0; i < n; i++) {
                int bound = digits[i];
                for (int x = mn; x < bound; x++) {
                    int rem = n - i - 1;
                    long[] vec = ways[rem][x];
                    int need = (k - ((Smod + x) % k)) % k;
                    ans += sumByMod(vec, k, need);
                }
                if (bound < mn) {
                    ok = false;
                    break;
                }
                Smod = (Smod + bound) % k;
                mn = bound;
            }
            if (ok && Smod % k == 0) {
                ans++;
            }
            sb.append(ans).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 返回向量 vec 中下标模 d 余 r 的所有元素之和
long long sum_by_mod(const vector<long long>& vec, int d, int r) {
    long long ans = 0;
    for (int i = r; i < (int)vec.size(); i += d) ans += vec[i];
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string mstr;       // 上限 N
    int Q;
    cin >> mstr >> Q;
    vector<int> digits;
    for (char c : mstr) digits.push_back(c - '0');
    int n = (int)digits.size();

    int maxRem = max(0, n - 1);
    // ways[L][mn] 是一个长度为 9*L+1 的向量
    vector<vector<vector<long long>>> ways(maxRem + 1, vector<vector<long long>>(10));
    for (int L = 0; L <= maxRem; ++L) {
        for (int mn = 0; mn <= 9; ++mn) {
            ways[L][mn].assign(9 * L + 1, 0LL);
        }
    }
    for (int mn = 0; mn <= 9; ++mn) ways[0][mn][0] = 1;

    for (int L = 1; L <= maxRem; ++L) {
        for (int mn = 0; mn <= 9; ++mn) {
            auto& cur = ways[L][mn];
            for (int x = mn; x <= 9; ++x) {
                const auto& prev = ways[L - 1][x];
                for (int sR = 0; sR < (int)prev.size(); ++sR) {
                    cur[x + sR] += prev[sR];
                }
            }
        }
    }

    auto solve_one = [&](int k) -> long long {
        long long ans = 0;
        // 1. 位数小于 n 的平稳数
        for (int L = 1; L < n; ++L) {
            for (int f = 1; f <= 9; ++f) {
                const auto& vec = ways[L - 1][f];
                int need = (k - (f % k)) % k;
                ans += sum_by_mod(vec, k, need);
            }
        }
        // 2. 位数等于 n 的平稳数
        int mn = 1;
        int Smod = 0;
        bool ok = true;
        for (int i = 0; i < n; ++i) {
            int bound = digits[i];
            for (int x = mn; x < bound; ++x) {
                int rem = n - i - 1;
                const auto& vec = ways[rem][x];
                int need = (k - ((Smod + x) % k)) % k;
                ans += sum_by_mod(vec, k, need);
            }
            if (bound < mn) {
                ok = false;
                break;
            }
            Smod = (Smod + bound) % k;
            mn = bound;
        }
        if (ok && Smod % k == 0) ans += 1;
        return ans;
    };

    for (int qi = 0; qi < Q; ++qi) {
        int k;
        cin >> k;
        cout << solve_one(k) << "\n";
    }
    return 0;
}
```
