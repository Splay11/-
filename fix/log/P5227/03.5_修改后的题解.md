## 解题思路

把每座已有泵站的覆盖范围写成区间 $[p_i-w_i,\ p_i+w_i]$。只需覆盖干线 $[0,S]$，因此先截断到

$[\max(0,p_i-w_i),\ \min(S,p_i+w_i)]$。

按左端点从小到大排序，维护变量 $right$，表示从 $0$ 起已经连续覆盖到的最右位置。

依次处理区间 $[l,r]$：

* 若 $r\le right$，该区间完全落在已覆盖部分中，跳过。
* 若 $l\le right$，它可以接上已覆盖部分，将 $right$ 更新为 $\max(right,r)$。
* 若 $l>right$，则 $[right,l]$ 尚未覆盖。每座新泵站最多连续覆盖长度 $2R$，因此至少需要

$$
cnt=\left\lceil\frac{l-right}{2R}\right\rceil
$$

座新泵站。为了让新泵站尽可能向右延伸，从当前 $right$ 起连续放置，覆盖位置推进到 $right+cnt\cdot 2R$，再与当前已有区间取较大值更新 $right$。

贪心依据：对当前最左侧未覆盖点，新泵站在覆盖它的前提下应尽量靠右，不会使后续更差。

处理完所有已有区间后，若 $[right,S]$ 仍未覆盖，再补

$$
\left\lceil\frac{S-right}{2R}\right\rceil
$$

座即可。

不能把各段空隙单独计算后再相加。一座新泵站可能跨过已被旧泵站覆盖的区域，同时补上左右两侧，因此必须维护连续覆盖到的最右位置。

## 复杂度分析

对 $n$ 个已有覆盖区间排序，时间复杂度为 $O(n\log n)$，随后线性扫描为 $O(n)$。

总时间复杂度 $O(n\log n)$，空间复杂度 $O(n)$。

在 $n\le 10^5$ 的范围内可以通过。

## 代码实现

### Python

```python
def min_trees(L, m, D, trees):
    intervals = []

    # 将已有泵站转化为覆盖区间
    for x, c in trees:
        left = max(0, x - c)
        right = min(L, x + c)
        intervals.append((left, right))

    # 按左端点排序
    intervals.sort()

    right = 0
    ans = 0
    length = 2 * D

    for left, end in intervals:
        # 当前区间已经完全被覆盖
        if end <= right:
            continue

        # 中间存在空缺，需要新增泵站
        if left > right:
            need = (left - right + length - 1) // length
            ans += need
            right += need * length

        # 利用已有区间继续向右扩展
        if end > right:
            right = end

        if right >= L:
            break

    # 处理最后剩余的未覆盖部分
    if right < L:
        ans += (L - right + length - 1) // length

    return ans


def main():
    L, m, D = map(int, input().split())

    trees = []
    for _ in range(m):
        x, c = map(int, input().split())
        trees.append((x, c))

    print(min_trees(L, m, D, trees))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static long minTrees(long L, int m, long D, long[][] trees) {
        long[][] intervals = new long[m][2];

        // 将已有泵站转化为覆盖区间
        for (int i = 0; i < m; i++) {
            long x = trees[i][0];
            long c = trees[i][1];

            intervals[i][0] = Math.max(0, x - c);
            intervals[i][1] = Math.min(L, x + c);
        }

        // 按区间左端点排序
        Arrays.sort(intervals, new Comparator<long[]>() {
            public int compare(long[] a, long[] b) {
                return Long.compare(a[0], b[0]);
            }
        });

        long right = 0;
        long ans = 0;
        long length = 2 * D;

        for (int i = 0; i < m; i++) {
            long left = intervals[i][0];
            long end = intervals[i][1];

            if (end <= right) {
                continue;
            }

            if (left > right) {
                long need = (left - right + length - 1) / length;
                ans += need;
                right += need * length;
            }

            if (end > right) {
                right = end;
            }

            if (right >= L) {
                break;
            }
        }

        if (right < L) {
            ans += (L - right + length - 1) / length;
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        StringTokenizer st = new StringTokenizer(br.readLine());
        long L = Long.parseLong(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        long D = Long.parseLong(st.nextToken());

        long[][] trees = new long[m][2];

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            trees[i][0] = Long.parseLong(st.nextToken());
            trees[i][1] = Long.parseLong(st.nextToken());
        }

        System.out.println(minTrees(L, m, D, trees));
    }
}
```

```cpp
#include <iostream>

long long minTrees(long long L, int m, long long D,
                   vector<pair<long long, long long>>& trees) {
    vector<pair<long long, long long>> intervals;

    // 将已有泵站转化为覆盖区间
    for (auto tree : trees) {
        long long x = tree.first;
        long long c = tree.second;

        long long left = max(0LL, x - c);
        long long right = min(L, x + c);

        intervals.push_back({left, right});
    }

    sort(intervals.begin(), intervals.end());

    long long right = 0;
    long long ans = 0;
    long long length = 2 * D;

    for (auto interval : intervals) {
        long long left = interval.first;
        long long end = interval.second;

        if (end <= right) {
            continue;
        }

        if (left > right) {
            long long need = (left - right + length - 1) / length;
            ans += need;
            right += need * length;
        }

        if (end > right) {
            right = end;
        }

        if (right >= L) {
            break;
        }
    }

    if (right < L) {
        ans += (L - right + length - 1) / length;
    }

    return ans;
}

int main() {
    long long L, D;
    int m;

    cin >> L >> m >> D;

    vector<pair<long long, long long>> trees(m);

    for (int i = 0; i < m; i++) {
        cin >> trees[i].first >> trees[i].second;
    }
```
