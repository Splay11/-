## 解题思路

先求每个点子树内的同色连通块个数。令 $dp_u$ 表示以 $u$ 为根的子树中同色连通块的数量。初始 $dp_u=1$（$u$ 自己构成一块）。对每个儿子 $v$：

\[
dp_u \leftarrow dp_u + dp_v - [color_u = color_v]
\]

即若 $u$ 与 $v$ 同色，则 $v$ 所在连通块与 $u$ 合并，块数减一；否则直接拼上 $v$ 的所有块。

整棵树的同色连通块数为 $dp_1$。断开边 $(u,v)$（$u$ 为父，$v$ 为子）后：
- 子树一侧的块数为 $dp_v$；
- 另一侧的块数为 $dp_1 - dp_v$，若 $color_u = color_v$，则原来被合并的那一块会在断开后重新出现，需要再加 `1`。

该边权值为两侧块数之差的绝对值。对所有边求和即可。答案在最坏情况下可达 $O(n^2)$ 量级，需使用 64 位整数。

## 复杂度分析

时间复杂度 $O(n)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys

sys.setrecursionlimit(300000)

n = int(input())
a = input().strip()
e = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    x, y = map(int, input().split())
    e[x].append(y)
    e[y].append(x)

dp = [0] * (n + 1)
ans = 0

def dfs(u, fa):
    dp[u] = 1
    for v in e[u]:
        if v == fa:
            continue
        dfs(v, u)
        dp[u] += dp[v]
        if a[u - 1] == a[v - 1]:
            dp[u] -= 1

def dfs1(u, fa):
    global ans
    for v in e[u]:
        if v == fa:
            continue
        dfs1(v, u)
        x = dp[1] - dp[v]
        if a[u - 1] == a[v - 1]:
            x += 1
        ans += abs(x - dp[v])

dfs(1, -1)
dfs1(1, -1)
print(ans)
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static List<Integer>[] e = new List[200005];
    static String a;
    static long ans = 0;
    static int[] dp = new int[200005];

    public static void dfs(int u, int fa) {
        dp[u] = 1;
        for (int v : e[u]) {
            if (v == fa) continue;
            dfs(v, u);
            dp[u] += dp[v];
            if (a.charAt(u - 1) == a.charAt(v - 1)) dp[u]--;
        }
    }

    public static void dfs1(int u, int fa) {
        for (int v : e[u]) {
            if (v == fa) continue;
            dfs1(v, u);
            int x = dp[1] - dp[v];
            if (a.charAt(u - 1) == a.charAt(v - 1)) {
                x += 1;
            }
            ans += Math.abs(x - dp[v]);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i <= n; i++) {
            e[i] = new ArrayList<>();
        }
        a = sc.next();
        for (int i = 0; i < n - 1; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            e[x].add(y);
            e[y].add(x);
        }
        dfs(1, -1);
        dfs1(1, -1);
        System.out.println(ans);
    }
}
```

#include <iostream>
#include <vector>
#include <string>
#include <cmath>
using namespace std;

const int maxn = 200005;
string a;
vector<int> e[maxn];
long long ans = 0;
int dp[maxn];

void dfs(int u, int fa) {
    dp[u] = 1;
    for (int v : e[u]) {
        if (v == fa) continue;
        dfs(v, u);
        dp[u] += dp[v];
        if (a[u - 1] == a[v - 1]) dp[u]--;
    }
}

void dfs1(int u, int fa) {
    for (int v : e[u]) {
        if (v == fa) continue;
        dfs1(v, u);
        int x = dp[1] - dp[v];
        if (a[u - 1] == a[v - 1]) x += 1;
        ans += abs(x - dp[v]);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    cin >> a;
    for (int i = 0; i < n - 1; i++) {
        int x, y;
        cin >> x >> y;
        e[x].push_back(y);
        e[y].push_back(x);
    }
    dfs(1, -1);
    dfs1(1, -1);
    cout << ans << endl;
    return 0;
}
