## 解题思路

先求出每个数的指数强度，再在强度序列上求长度不超过 $k$ 的最大连续子段和。

指数强度是质因数分解中的最大指数。由于 $a_i\le 10^7$ 且 $\sum n\le 2\times 10^5$，用筛法预处理每个数的最小质因子 `spf`，再沿最小质因子分解并记录最大指数，单个数约为 $O(\log a_i)$。

得到强度序列 $w$ 后，用前缀和 $P$ 把子段和写成 $P[i]-P[j]$，其中 $i-k\le j<i$。对每个右端点 $i$，需要窗口内最小的前缀和。用单调队列维护该窗口，整体 $O(n)$。

## 复杂度分析

筛法预处理值域 $V=10^7$，之后每组线性扫描。

- 时间复杂度：预处理 $O(V\log\log V)$，所有询问合计 $O(\sum n)$。
- 空间复杂度：$O(V)$。

## 代码实现

### Python

```python
import sys
from collections import deque

MAX_A = 10000001
spf = list(range(MAX_A))

def sieve():
    for i in range(2, int(MAX_A**0.5) + 1):
        if spf[i] == i:
            for j in range(i * i, MAX_A, i):
                if spf[j] == j:
                    spf[j] = i

def get_weight(x):
    if x == 1:
        return 0
    max_exp = 0
    while x > 1:
        p = spf[x]
        current_exp = 0
        while x > 1 and x % p == 0:
            current_exp += 1
            x //= p
        if current_exp > max_exp:
            max_exp = current_exp
    return max_exp

def solve():
    n, k = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    weights = [get_weight(num) for num in a]
    prefix_sum = [0] * (n + 1)
    for i in range(n):
        prefix_sum[i + 1] = prefix_sum[i] + weights[i]
    max_weight_sum = 0
    dq = deque([0])
    for i in range(1, n + 1):
        if dq and dq[0] < i - k:
            dq.popleft()
        max_weight_sum = max(max_weight_sum, prefix_sum[i] - prefix_sum[dq[0]])
        while dq and prefix_sum[dq[-1]] >= prefix_sum[i]:
            dq.pop()
        dq.append(i)
    print(max_weight_sum)

sieve()
T = int(sys.stdin.readline())
for _ in range(T):
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final int MAX_A = 10000001;
    static int[] spf = new int[MAX_A];

    static void sieve() {
        for (int i = 0; i < MAX_A; i++) {
            spf[i] = i;
        }
        for (int i = 2; i * i < MAX_A; i++) {
            if (spf[i] == i) {
                for (int j = i * i; j < MAX_A; j += i) {
                    if (spf[j] == j) {
                        spf[j] = i;
                    }
                }
            }
        }
    }

    static int getWeight(int x) {
        if (x == 1) return 0;
        int maxExp = 0;
        while (x > 1) {
            int p = spf[x];
            int currentExp = 0;
            while (x > 1 && x % p == 0) {
                currentExp++;
                x /= p;
            }
            if (currentExp > maxExp) {
                maxExp = currentExp;
            }
        }
        return maxExp;
    }

    static void solve(BufferedReader br) throws IOException {
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        int[] a = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        long[] weights = new long[n];
        for (int i = 0; i < n; i++) {
            weights[i] = getWeight(a[i]);
        }
        long[] prefixSum = new long[n + 1];
        for (int i = 0; i < n; i++) {
            prefixSum[i + 1] = prefixSum[i] + weights[i];
        }
        long maxWeightSum = 0;
        Deque<Integer> dq = new ArrayDeque<>();
        dq.addLast(0);
        for (int i = 1; i <= n; i++) {
            if (!dq.isEmpty() && dq.peekFirst() < i - k) {
                dq.pollFirst();
            }
            maxWeightSum = Math.max(maxWeightSum, prefixSum[i] - prefixSum[dq.peekFirst()]);
            while (!dq.isEmpty() && prefixSum[dq.peekLast()] >= prefixSum[i]) {
                dq.pollLast();
            }
            dq.addLast(i);
        }
        System.out.println(maxWeightSum);
    }

    public static void main(String[] args) throws IOException {
        sieve();
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        while (T-- > 0) {
            solve(br);
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <numeric>
#include <algorithm>
#include <deque>

const int MAX_A = 10000001;
int spf[MAX_A];

void sieve() {
    std::iota(spf, spf + MAX_A, 0);
    for (int i = 2; i * i < MAX_A; ++i) {
        if (spf[i] == i) {
            for (int j = i * i; j < MAX_A; j += i) {
                if (spf[j] == j) {
                    spf[j] = i;
                }
            }
        }
    }
}

int get_weight(int x) {
    if (x == 1) return 0;
    int max_exp = 0;
    while (x > 1) {
        int p = spf[x];
        int current_exp = 0;
        while (x > 1 && x % p == 0) {
            current_exp++;
            x /= p;
        }
        if (current_exp > max_exp) {
            max_exp = current_exp;
        }
    }
    return max_exp;
}

void solve() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> a(n);
    std::vector<long long> weights(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> a[i];
        weights[i] = get_weight(a[i]);
    }
    std::vector<long long> prefix_sum(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        prefix_sum[i + 1] = prefix_sum[i] + weights[i];
    }
    long long max_weight_sum = 0;
    std::deque<int> dq;
    dq.push_back(0);
    for (int i = 1; i <= n; ++i) {
        if (!dq.empty() && dq.front() < i - k) {
            dq.pop_front();
        }
        max_weight_sum = std::max(max_weight_sum, prefix_sum[i] - prefix_sum[dq.front()]);
        while (!dq.empty() && prefix_sum[dq.back()] >= prefix_sum[i]) {
            dq.pop_back();
        }
        dq.push_back(i);
    }
    std::cout << max_weight_sum << "\n";
}

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(NULL);
    sieve();
    int T;
    std::cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
```
