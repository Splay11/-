## 解题思路

实验室有重量分别为 $1$ 克到 $n$ 克的砝码各一枚。对于每次目标重量 $k$，需要判断是否能选出若干枚砝码，使总重量恰好等于 $k$，并输出一种选法。

由于砝码重量是 $\{1,2,\dots,n\}$，这是一个连续整数集合。设 $S = \frac{n(n+1)}{2}$ 表示所有砝码的总重量。

1. 若 $k > S$，显然无法凑出目标重量，直接输出 `-1`。
2. 若 $k \le S$，则一定可以构造出解。构造方法采用贪心策略：**从大到小**扫描砝码重量 $i = n, n-1, \dots, 1$。对于当前重量 $i$，如果 $k \ge i$，则选择这枚砝码，并将剩余重量更新为 $k = k - i$。当 $k$ 变为 $0$ 时停止。
   - 该贪心的正确性基于 $\{1,\dots,n\}$ 是规范的硬币系统，在 $k \le S$ 时总能用最大可选砝码的贪心方式得到一种解。
3. 按上述方法选出的砝码重量列表是降序的，最后将其反转即可得到升序的输出。

每次称量任务相互独立，按照上述步骤对每个 $k$ 处理即可。输出时，若存在合法选法，先输出所选砝码的数量，再在下一行输出升序排列的砝码重量；否则仅输出一行 `-1`。

## 复杂度分析

- 对于单个称量任务 $k$，贪心扫描需要检查最多 $n$ 个砝码，时间复杂度 $O(n)$。
- 由于题目保证所有测试数据的 $n \times m$ 之和不超过 $2 \times 10^6$，总操作次数可以接受。
- 需要存储选择的砝码列表，最坏情况下选择所有砝码，空间复杂度 $O(n)$。

总体时间复杂度 $O(\sum (n \times m))$，空间复杂度 $O(n)$（不计输入输出缓冲区）。

## 代码实现

### Python

```python
import sys

def solve_one(n: int, k: int):
    """返回一个列表，包含所选砝码的升序重量；若无法凑出则返回空列表"""
    S = n * (n + 1) // 2
    if k > S:
        return []
    chosen = []
    # 从大到小贪心选取
    for w in range(n, 0, -1):
        if k >= w:
            chosen.append(w)
            k -= w
            if k == 0:
                break
    if k != 0:          # 理论上不会发生（k <= S 时必然可表示）
        return []
    chosen.reverse()    # 转为升序
    return chosen

def main():
    data = list(map(int, sys.stdin.read().split()))
    it = iter(data)
    T = next(it)
    out_lines = []
    for _ in range(T):
        n = next(it)
        m = next(it)
        for _ in range(m):
            k = next(it)
            res = solve_one(n, k)
            if not res:
                out_lines.append("-1")
            else:
                out_lines.append(str(len(res)))
                out_lines.append(" ".join(map(str, res)))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    // 返回所选砝码的升序重量列表；无解返回空列表
    static List<Integer> chooseWeights(int n, long k) {
        long S = 1L * n * (n + 1) / 2;
        if (k > S) return Collections.emptyList();
        List<Integer> ans = new ArrayList<>();
        for (int w = n; w >= 1; w--) {
            if (k >= w) {
                ans.add(w);
                k -= w;
                if (k == 0) break;
            }
        }
        if (k != 0) return Collections.emptyList(); // 理论上不会发生
        Collections.reverse(ans); // 升序
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        ArrayList<String> out = new ArrayList<>();

        for (int tc = 0; tc < T; tc++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int n = Integer.parseInt(st.nextToken());
            int m = Integer.parseInt(st.nextToken());
            for (int q = 0; q < m; q++) {
                long k = Long.parseLong(br.readLine().trim());
                List<Integer> res = chooseWeights(n, k);
                if (res.isEmpty()) {
                    out.add("-1");
                } else {
                    out.add(String.valueOf(res.size()));
                    StringBuilder sb = new StringBuilder();
                    for (int i = 0; i < res.size(); i++) {
                        if (i > 0) sb.append(' ');
                        sb.append(res.get(i));
                    }
                    out.add(sb.toString());
                }
            }
        }

        StringBuilder all = new StringBuilder();
        for (String line : out) {
            all.append(line).append('\n');
        }
        System.out.print(all);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 返回所选砝码的升序重量；无解返回空向量
vector<int> choose_weights(int n, long long k) {
    long long S = 1LL * n * (n + 1) / 2;
    if (k > S) return {};
    vector<int> ans;
    for (int w = n; w >= 1; --w) {
        if (k >= w) {
            ans.push_back(w);
            k -= w;
            if (k == 0) break;
        }
    }
    if (k != 0) return {}; // 理论上不会发生
    reverse(ans.begin(), ans.end()); // 转为升序
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    vector<string> out;
    while (T--) {
        int n, m;
        cin >> n >> m;
        for (int i = 0; i < m; ++i) {
            long long k;
            cin >> k;
            vector<int> res = choose_weights(n, k);
            if (res.empty()) {
                out.emplace_back("-1");
            } else {
                out.emplace_back(to_string(res.size()));
                string line;
                for (size_t j = 0; j < res.size(); ++j) {
                    if (j) line.push_back(' ');
                    line += to_string(res[j]);
                }
                out.emplace_back(line);
            }
        }
    }
    for (auto &s : out) cout << s << "\n";
    return 0;
}
```
