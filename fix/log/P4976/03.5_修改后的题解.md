## 解题思路

用 Pipeline 把 StandardScaler 与线性 SVM 串起来，保证交叉验证每一折都只在当前训练折上拟合标准化，避免泄漏。标签 `0` / `1` 分别表示合格与次品。

步骤：

1. 将训练表拆成特征 $X$ 与标签 $y \in \{0,1\}$。
2. StandardScaler 标准化特征。
3. SVC 固定 kernel = "linear"、random_state = `42`。
4. StratifiedKFold：`3` 折、shuffle = True、random_state = `42`。
5. GridSearchCV 在 $C \in \{0.1, 1.0, 10.0\}$ 中按 accuracy 选参；并列时取较小的 $C$。
6. refit = True，用最佳 $C$ 在完整训练集上重训，对测试集逐行输出 `0` / `1`。

Java / C++ 实现同一管线：先标准化，再对三个 $C$ 做分层 `3` 折交叉验证，最后用最优 $C$ 训练线性软间隔 SVM 并预测。

## 复杂度分析

网格 `3` 个 $C$、交叉验证 `3` 折，共训练 `9` 次，外加完整训练集重训 `1` 次。设线性 SVM 训练开销为 $T_{\mathrm{svm}}$，总时间约为 $O(10 \cdot T_{\mathrm{svm}} + q \cdot m)$。空间 $O(n m + q m)$。在 $n \le 200$、$m \le 10$ 下可运行。

## 代码实现

### Python

```python
import sys
import numpy as np
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVC
from sklearn.model_selection import StratifiedKFold, GridSearchCV


def svm_predict(train_data, test_data):
    train_arr = np.array(train_data, dtype=float)
    test_arr = np.array(test_data, dtype=float)

    X = train_arr[:, :-1]
    y = train_arr[:, -1].astype(int)

    model = Pipeline([
        ("scaler", StandardScaler()),
        ("svc", SVC(kernel="linear", random_state=42))
    ])

    param_grid = {
        "svc__C": [0.1, 1.0, 10.0]
    }

    cv = StratifiedKFold(
        n_splits=3,
        shuffle=True,
        random_state=42
    )

    grid = GridSearchCV(
        estimator=model,
        param_grid=param_grid,
        scoring="accuracy",
        cv=cv,
        refit=True
    )

    grid.fit(X, y)
    preds = grid.predict(test_arr)
    return preds.astype(int)


def main():
    data = sys.stdin.read().strip().split()
    idx = 0

    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1

    train_data = []
    for _ in range(n):
        row = []
        for _ in range(m + 1):
            row.append(float(data[idx]))
            idx += 1
        train_data.append(row)

    q = int(data[idx]); idx += 1

    test_data = []
    for _ in range(q):
        row = []
        for _ in range(m):
            row.append(float(data[idx]))
            idx += 1
        test_data.append(row)

    preds = svm_predict(train_data, test_data)
    for p in preds:
        print(int(p))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final double[] CS = {0.1, 1.0, 10.0};

    static void scaleFit(double[][] X, double[] mean, double[] std) {
        int n = X.length, m = X[0].length;
        Arrays.fill(mean, 0);
        Arrays.fill(std, 0);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) mean[j] += X[i][j];
        }
        for (int j = 0; j < m; j++) mean[j] /= n;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                double d = X[i][j] - mean[j];
                std[j] += d * d;
            }
        }
        for (int j = 0; j < m; j++) {
            std[j] = Math.sqrt(std[j] / n);
            if (std[j] < 1e-12) std[j] = 1;
        }
    }

    static double[][] applyScale(double[][] X, double[] mean, double[] std) {
        int n = X.length, m = X[0].length;
        double[][] Z = new double[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) Z[i][j] = (X[i][j] - mean[j]) / std[j];
        }
        return Z;
    }

    static class LinearSVM {
        double[] w;
        double b;

        void fit(double[][] X, int[] y, double C) {
            int n = X.length, d = X[0].length;
            w = new double[d];
            b = 0;
            int T = Math.max(2000, 40 * n);
            for (int t = 1; t <= T; t++) {
                int i = (t - 1) % n;
                double eta = 1.0 / (t + 10);
                double yv = y[i] == 1 ? 1.0 : -1.0;
                double pred = b;
                for (int j = 0; j < d; j++) pred += w[j] * X[i][j];
                for (int j = 0; j < d; j++) w[j] *= (1 - eta);
                if (yv * pred < 1) {
                    for (int j = 0; j < d; j++) w[j] += eta * C * yv * X[i][j];
                    b += eta * C * yv;
                }
            }
        }

        int predictOne(double[] x) {
            double s = b;
            for (int j = 0; j < w.length; j++) s += w[j] * x[j];
            return s >= 0 ? 1 : 0;
        }
    }

    static int[][] stratifiedFolds(int[] y, int k, long seed) {
        List<Integer> pos = new ArrayList<>();
        List<Integer> neg = new ArrayList<>();
        for (int i = 0; i < y.length; i++) {
            if (y[i] == 1) pos.add(i); else neg.add(i);
        }
        Random rnd = new Random(seed);
        Collections.shuffle(pos, rnd);
        Collections.shuffle(neg, rnd);
        int[][] folds = new int[y.length][1];
        int[] assign = new int[y.length];
        for (int i = 0; i < pos.size(); i++) assign[pos.get(i)] = i % k;
        for (int i = 0; i < neg.size(); i++) assign[neg.get(i)] = i % k;
        folds = new int[1][y.length];
        folds[0] = assign;
        return folds;
    }

    static double[][] rows(double[][] X, List<Integer> idx) {
        double[][] R = new double[idx.size()][X[0].length];
        for (int i = 0; i < idx.size(); i++) R[i] = X[idx.get(i)];
        return R;
    }

    static int[] rowsY(int[] y, List<Integer> idx) {
        int[] R = new int[idx.size()];
        for (int i = 0; i < idx.size(); i++) R[i] = y[idx.get(i)];
        return R;
    }

    static double cvAcc(double[][] X, int[] y, double C) {
        int n = y.length, k = 3;
        int[] assign = stratifiedFolds(y, k, 42)[0];
        int correct = 0, total = 0;
        for (int f = 0; f < k; f++) {
            List<Integer> tr = new ArrayList<>();
            List<Integer> va = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                if (assign[i] == f) va.add(i); else tr.add(i);
            }
            if (tr.isEmpty() || va.isEmpty()) continue;
            double[][] Xtr = rows(X, tr);
            int[] ytr = rowsY(y, tr);
            double[] mean = new double[X[0].length], std = new double[X[0].length];
            scaleFit(Xtr, mean, std);
            double[][] Ztr = applyScale(Xtr, mean, std);
            LinearSVM svm = new LinearSVM();
            svm.fit(Ztr, ytr, C);
            double[][] Xva = rows(X, va);
            int[] yva = rowsY(y, va);
            double[][] Zva = applyScale(Xva, mean, std);
            for (int i = 0; i < Zva.length; i++) {
                if (svm.predictOne(Zva[i]) == yva[i]) correct++;
                total++;
            }
        }
        return total == 0 ? 0 : (double) correct / total;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        int n = fs.nextInt(), m = fs.nextInt();
        double[][] X = new double[n][m];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) X[i][j] = fs.nextDouble();
            y[i] = (int) Math.round(fs.nextDouble());
        }
        int q = fs.nextInt();
        double[][] Xt = new double[q][m];
        for (int i = 0; i < q; i++) {
            for (int j = 0; j < m; j++) Xt[i][j] = fs.nextDouble();
        }

        double bestC = CS[0], bestAcc = -1;
        for (double C : CS) {
            double acc = cvAcc(X, y, C);
            if (acc > bestAcc + 1e-12) {
                bestAcc = acc;
                bestC = C;
            }
        }

        double[] mean = new double[m], std = new double[m];
        scaleFit(X, mean, std);
        LinearSVM svm = new LinearSVM();
        svm.fit(applyScale(X, mean, std), y, bestC);
        double[][] Zt = applyScale(Xt, mean, std);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < q; i++) sb.append(svm.predictOne(Zt[i])).append('\n');
        System.out.print(sb.toString());
    }

    static class FastScanner {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;
        FastScanner(InputStream in) { br = new BufferedReader(new InputStreamReader(in)); }
        String next() throws IOException {
            while (st == null || !st.hasMoreTokens()) st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }
        int nextInt() throws IOException { return Integer.parseInt(next()); }
        double nextDouble() throws IOException { return Double.parseDouble(next()); }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<double> CS = {0.1, 1.0, 10.0};

void scale_fit(const vector<vector<double>>& X, vector<double>& mean, vector<double>& stdev) {
    int n = (int)X.size(), m = (int)X[0].size();
    mean.assign(m, 0); stdev.assign(m, 0);
    for (int i = 0; i < n; i++) for (int j = 0; j < m; j++) mean[j] += X[i][j];
    for (int j = 0; j < m; j++) mean[j] /= n;
    for (int i = 0; i < n; i++) for (int j = 0; j < m; j++) {
        double d = X[i][j] - mean[j];
        stdev[j] += d * d;
    }
    for (int j = 0; j < m; j++) {
        stdev[j] = sqrt(stdev[j] / n);
        if (stdev[j] < 1e-12) stdev[j] = 1;
    }
}

vector<vector<double>> apply_scale(const vector<vector<double>>& X, const vector<double>& mean, const vector<double>& stdev) {
    int n = (int)X.size(), m = (int)X[0].size();
    vector<vector<double>> Z(n, vector<double>(m));
    for (int i = 0; i < n; i++) for (int j = 0; j < m; j++) Z[i][j] = (X[i][j] - mean[j]) / stdev[j];
    return Z;
}

struct LinearSVM {
    vector<double> w;
    double b = 0;
    void fit(const vector<vector<double>>& X, const vector<int>& y, double C) {
        int n = (int)X.size(), d = (int)X[0].size();
        w.assign(d, 0); b = 0;
        int T = max(2000, 40 * n);
        for (int t = 1; t <= T; t++) {
            int i = (t - 1) % n;
            double eta = 1.0 / (t + 10);
            double yv = y[i] == 1 ? 1.0 : -1.0;
            double pred = b;
            for (int j = 0; j < d; j++) pred += w[j] * X[i][j];
            for (int j = 0; j < d; j++) w[j] *= (1 - eta);
            if (yv * pred < 1) {
                for (int j = 0; j < d; j++) w[j] += eta * C * yv * X[i][j];
                b += eta * C * yv;
            }
        }
    }
    int predict_one(const vector<double>& x) const {
        double s = b;
        for (int j = 0; j < (int)w.size(); j++) s += w[j] * x[j];
        return s >= 0 ? 1 : 0;
    }
};

vector<int> stratified_assign(const vector<int>& y, int k, uint64_t seed) {
    vector<int> pos, neg;
    for (int i = 0; i < (int)y.size(); i++) (y[i] == 1 ? pos : neg).push_back(i);
    mt19937_64 rng(seed);
    shuffle(pos.begin(), pos.end(), rng);
    shuffle(neg.begin(), neg.end(), rng);
    vector<int> assign(y.size());
    for (int i = 0; i < (int)pos.size(); i++) assign[pos[i]] = i % k;
    for (int i = 0; i < (int)neg.size(); i++) assign[neg[i]] = i % k;
    return assign;
}

double cv_acc(const vector<vector<double>>& X, const vector<int>& y, double C) {
    int n = (int)y.size(), k = 3;
    vector<int> assign = stratified_assign(y, k, 42);
    int correct = 0, total = 0;
    for (int f = 0; f < k; f++) {
        vector<vector<double>> Xtr, Xva;
        vector<int> ytr, yva;
        for (int i = 0; i < n; i++) {
            if (assign[i] == f) { Xva.push_back(X[i]); yva.push_back(y[i]); }
            else { Xtr.push_back(X[i]); ytr.push_back(y[i]); }
        }
        if (Xtr.empty() || Xva.empty()) continue;
        vector<double> mean, stdev;
        scale_fit(Xtr, mean, stdev);
        auto Ztr = apply_scale(Xtr, mean, stdev);
        LinearSVM svm; svm.fit(Ztr, ytr, C);
        auto Zva = apply_scale(Xva, mean, stdev);
        for (int i = 0; i < (int)Zva.size(); i++) {
            if (svm.predict_one(Zva[i]) == yva[i]) correct++;
            total++;
        }
    }
    return total == 0 ? 0 : (double)correct / total;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<vector<double>> X(n, vector<double>(m));
    vector<int> y(n);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) cin >> X[i][j];
        double lab; cin >> lab;
        y[i] = (int)llround(lab);
    }
    int q; cin >> q;
    vector<vector<double>> Xt(q, vector<double>(m));
    for (int i = 0; i < q; i++) for (int j = 0; j < m; j++) cin >> Xt[i][j];

    double bestC = CS[0], bestAcc = -1;
    for (double C : CS) {
        double acc = cv_acc(X, y, C);
        if (acc > bestAcc + 1e-12) { bestAcc = acc; bestC = C; }
    }
    vector<double> mean, stdev;
    scale_fit(X, mean, stdev);
    LinearSVM svm;
    svm.fit(apply_scale(X, mean, stdev), y, bestC);
    auto Zt = apply_scale(Xt, mean, stdev);
    for (int i = 0; i < q; i++) cout << svm.predict_one(Zt[i]) << '\n';
    return 0;
}
```
