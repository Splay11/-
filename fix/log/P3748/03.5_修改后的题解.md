## 解题思路

对每个节点 $u$ 维护两个量：

- $down[u]$：从 $u$ 向下走到其子树内某个哨所的最大距离；若子树内没有哨所，记为负无穷（实现中用 $-10^9$）。
- $ans[u]$：以 $u$ 为根的子树的哨所跨度。

设 $u$ 的儿子为若干 $v$。若 $down[v]$ 有效，则该儿子能提供一条通往哨所的链，到 $u$ 的距离为 $down[v]+1$。子树内的最大跨度来自两种情况：

1. 完全落在某个儿子子树内，即 $ans[v]$。
2. 路径经过 $u$：收集所有可用的向下深度（每个儿子贡献 $down[v]+1$，若 $u$ 自身是哨所再贡献 `0`），取其中最大的两条相加。

若有效深度不足两条，经过 $u$ 的贡献按 `0` 计。

为避免递归栈过深，用栈从根 `1` 生成父子关系与遍历顺序，再按后序（逆序）计算上述转移。

## 复杂度分析

每个节点与每条边处理常数次。

- 时间复杂度：$O(n)$。
- 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        color[i] = int(next(it))

    g = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u = int(next(it)); v = int(next(it))
        g[u].append(v); g[v].append(u)

    NEG_INF = -10**9
    parent = [0] * (n + 1)
    order = []
    st = [1]
    parent[1] = 0
    while st:
        u = st.pop()
        order.append(u)
        for v in g[u]:
            if v != parent[u]:
                parent[v] = u
                st.append(v)

    down = [NEG_INF] * (n + 1)
    ans = [0] * (n + 1)

    for u in reversed(order):
        best1, best2 = (0, NEG_INF) if color[u] == 1 else (NEG_INF, NEG_INF)
        best_child_ans = 0
        for v in g[u]:
            if v == parent[u]:
                continue
            best_child_ans = max(best_child_ans, ans[v])
            if down[v] != NEG_INF:
                cand = down[v] + 1
                if cand > best1:
                    best2 = best1; best1 = cand
                elif cand > best2:
                    best2 = cand

        down[u] = 0 if color[u] == 1 else NEG_INF
        for v in g[u]:
            if v == parent[u]:
                continue
            if down[v] != NEG_INF:
                down[u] = max(down[u], down[v] + 1)

        through_u = 0
        if best2 != NEG_INF:
            through_u = best1 + best2

        ans[u] = max(best_child_ans, through_u)

    print(" ".join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static class FastScanner {
        BufferedInputStream in = new BufferedInputStream(System.in);
        byte[] buffer = new byte[1 << 16];
        int ptr = 0, len = 0;
        int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        int nextInt() throws IOException {
            int c, sgn = 1, x = 0;
            do { c = read(); } while (c <= 32);
            if (c == '-') { sgn = -1; c = read(); }
            while (c > 32) {
                x = x * 10 + (c - '0');
                c = read();
            }
            return x * sgn;
        }
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();
        int n;
        try {
            n = fs.nextInt();
        } catch (Exception e) {
            return;
        }
        int[] color = new int[n + 1];
        for (int i = 1; i <= n; i++) color[i] = fs.nextInt();

        ArrayList<Integer>[] g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int u = fs.nextInt(), v = fs.nextInt();
            g[u].add(v); g[v].add(u);
        }

        final int NEG_INF = -1000000000;
        int[] parent = new int[n + 1];
        int[] order = new int[n];
        int idx = 0;
        Deque<Integer> st = new ArrayDeque<>();
        st.push(1); parent[1] = 0;
        while (!st.isEmpty()) {
            int u = st.pop();
            order[idx++] = u;
            for (int v : g[u]) if (v != parent[u]) {
                parent[v] = u;
                st.push(v);
            }
        }

        int[] down = new int[n + 1];
        int[] ans = new int[n + 1];
        Arrays.fill(down, NEG_INF);

        for (int t = n - 1; t >= 0; t--) {
            int u = order[t];
            int best1 = (color[u] == 1 ? 0 : NEG_INF);
            int best2 = NEG_INF;
            int bestChildAns = 0;
            for (int v : g[u]) if (v != parent[u]) {
                bestChildAns = Math.max(bestChildAns, ans[v]);
                if (down[v] != NEG_INF) {
                    int cand = down[v] + 1;
                    if (cand > best1) { best2 = best1; best1 = cand; }
                    else if (cand > best2) { best2 = cand; }
                }
            }
            down[u] = (color[u] == 1 ? 0 : NEG_INF);
            for (int v : g[u]) if (v != parent[u]) {
                if (down[v] != NEG_INF)
                    down[u] = Math.max(down[u], down[v] + 1);
            }
            int throughU = 0;
            if (best2 != NEG_INF) {
                throughU = best1 + best2;
            }
            ans[u] = Math.max(bestChildAns, throughU);
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            if (i > 1) sb.append(' ');
            sb.append(ans[i]);
        }
        sb.append('\n');
        System.out.print(sb.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    if (!(cin >> n)) return 0;
    vector<int> color(n + 1);
    for (int i = 1; i <= n; ++i) cin >> color[i];
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; ++i) {
        int u, v; cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    const int NEG_INF = -1000000000;
    vector<int> parent(n + 1, 0), order; order.reserve(n);
    vector<int> st; st.push_back(1); parent[1] = 0;
    while (!st.empty()) {
        int u = st.back(); st.pop_back();
        order.push_back(u);
        for (int v : g[u]) if (v != parent[u]) {
            parent[v] = u;
            st.push_back(v);
        }
    }
    vector<int> down(n + 1, NEG_INF);
    vector<int> ans(n + 1, 0);
    for (int i = (int)order.size() - 1; i >= 0; --i) {
        int u = order[i];
        int best1 = NEG_INF, best2 = NEG_INF;
        if (color[u] == 1) best1 = 0;

        for (int v : g[u]) if (v != parent[u]) {
            bestChildAns = max(bestChildAns, ans[v]);

                int cand = down[v] + 1;
                if (cand > best1) { best2 = best1; best1 = cand; }
                else if (cand > best2) { best2 = cand; }
            }
        }
        down[u] = (color[u] ? 0 : NEG_INF);

        int throughU = 0;

    for (int i = 1; i <= n; ++i) {
        if (i > 1) cout << ' ';
        cout << ans[i];
    }
    cout << '\n';
    return 0;
```
