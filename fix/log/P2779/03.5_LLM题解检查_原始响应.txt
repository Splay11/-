## 解题思路

本题需要在一棵树上分配 26 种颜色的彩灯，希望最大化“和谐度”（即最长优美路径的节点数）。分配方案可以任意安排，因此问题可以拆解为两个独立的上界：

1. **树上最长简单路径的限制**：任何优美路径必然是一条树上的简单路径，其长度不可能超过树的直径（最长简单路径的节点数）。记树的直径节点数为 $D$，则和谐度一定满足 $\le D$。
2. **全局彩灯总数的回文限制**：一条路径上的颜色序列为回文，等价于在路径内部每种颜色的数量满足回文条件——偶数长度时所有颜色出现次数均为偶数，奇数长度时至多一个颜色出现奇数次。因此，从全部 $n$ 盏彩灯中能选出的最长回文序列长度 $L$ 为：将所有颜色的偶数部分全部累加，若存在奇数个的颜色则可再额外放置一个作为中心。即
   $$L = \sum_{i=1}^{26} \left(c_i - (c_i \bmod 2)\right) + \begin{cases} 1, & \text{若存在某个 } c_i \text{ 为奇数} \\ 0, & \text{否则} \end{cases}$$
   由于树上的任何路径的彩灯集合必定是全集的一个子集，因此和谐度不会超过 $L$。

综合两个上界，最大可能的和谐度即为 $\min(D, L)$。并且我们可以构造出达到该值的方案：将构成最长回文串的彩灯沿着树的直径依次分配，直径两端余下的节点放置剩余彩灯即可。

算法步骤如下：
- 读入 26 种彩灯的数量 $c_1 \dots c_{26}$，同时累加计算出最长回文长度 $L$。
- 读入整数 $n$ 和 $n-1$ 条边，构建树的邻接表。
- 使用两次 BFS（或 DFS）求树的直径：第一次从任意节点（如节点 1）出发，找到最远节点 $u$；第二次从 $u$ 出发，找到最远节点 $v$，此时的距离（边数）$+1$ 即为直径节点数 $D$。
- 最终答案为 $\min(D, L)$。

## 复杂度分析

- 计算 $L$ 只需遍历 26 种颜色，时间复杂度 $O(1)$。
- 求树的直径：两次 BFS 各遍历所有节点和边，时间复杂度 $O(n)$。
- 总体时间复杂度 $O(n)$，空间复杂度 $O(n)$（主要用于邻接表和 BFS 队列/距离数组）。

满足 $n \le 10^5$ 的数据范围。

## 代码实现

### Python

```python
import sys
from collections import deque

def main():
    # 读入26种彩灯的数量
    counts = list(map(int, sys.stdin.readline().split()))
    n = int(sys.stdin.readline().strip())
    
    # 构建邻接表
    tree = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v = map(int, sys.stdin.readline().split())
        tree[u].append(v)
        tree[v].append(u)
    
    # BFS 返回 (最远节点, 距离)
    def bfs(start):
        visited = [False] * (n + 1)
        dist = [0] * (n + 1)
        q = deque([start])
        visited[start] = True
        while q:
            u = q.popleft()
            for v in tree[u]:
                if not visited[v]:
                    visited[v] = True
                    dist[v] = dist[u] + 1
                    q.append(v)
        max_node = 1
        max_dist = 0
        for i in range(1, n + 1):
            if dist[i] > max_dist:
                max_dist = dist[i]
                max_node = i
        return max_node, max_dist
    
    # 第一次 BFS 从节点 1 开始
    u, _ = bfs(1)
    # 第二次 BFS 得到直径边数
    v, diameter_edges = bfs(u)
    diameter_nodes = diameter_edges + 1   # 直径节点数
    
    # 计算最长可构造回文长度 L
    even_sum = 0
    has_odd = False
    for cnt in counts:
        even_sum += cnt - (cnt & 1)
        if cnt % 2 == 1:
            has_odd = True
    L = even_sum + (1 if has_odd else 0)
    
    # 和谐度上界取最小
    ans = min(diameter_nodes, L)
    print(ans)

if __name__ == '__main__':
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static int n;
    static ArrayList<Integer>[] tree;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        // 读入26种彩灯数量
        String[] parts = br.readLine().split(" ");
        int[] counts = new int[26];
        for (int i = 0; i < 26; i++) {
            counts[i] = Integer.parseInt(parts[i]);
        }
        n = Integer.parseInt(br.readLine().trim());
        tree = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            tree[i] = new ArrayList<>();
        }
        for (int i = 1; i <= n - 1; i++) {
            String[] edge = br.readLine().split(" ");
            int u = Integer.parseInt(edge[0]);
            int v = Integer.parseInt(edge[1]);
            tree[u].add(v);
            tree[v].add(u);
        }

        // 两次 BFS 求直径
        int[] first = bfs(1);
        int[] second = bfs(first[0]);
        int diameterEdges = second[1];
        int diameterNodes = diameterEdges + 1;

        // 计算最长回文长度 L
        int evenSum = 0;
        boolean hasOdd = false;
        for (int cnt : counts) {
            evenSum += cnt - (cnt & 1);
            if (cnt % 2 == 1) {
                hasOdd = true;
            }
        }
        int L = evenSum + (hasOdd ? 1 : 0);

        int ans = Math.min(diameterNodes, L);
        System.out.println(ans);
    }

    static int[] bfs(int start) {
        boolean[] visited = new boolean[n + 1];
        int[] dist = new int[n + 1];
        Queue<Integer> queue = new LinkedList<>();
        queue.add(start);
        visited[start] = true;
        while (!queue.isEmpty()) {
            int u = queue.poll();
            for (int v : tree[u]) {
                if (!visited[v]) {
                    visited[v] = true;
                    dist[v] = dist[u] + 1;
                    queue.add(v);
                }
            }
        }
        int maxNode = start, maxDist = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] > maxDist) {
                maxDist = dist[i];
                maxNode = i;
            }
        }
        return new int[]{maxNode, maxDist};
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
using namespace std;

int n;
vector<vector<int>> tree;

pair<int,int> bfs(int start) {
    vector<bool> visited(n + 1, false);
    vector<int> dist(n + 1, 0);
    queue<int> q;
    q.push(start);
    visited[start] = true;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : tree[u]) {
            if (!visited[v]) {
                visited[v] = true;
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }
    int maxNode = start, maxDist = 0;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > maxDist) {
            maxDist = dist[i];
            maxNode = i;
        }
    }
    return {maxNode, maxDist};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int counts[26];
    for (int i = 0; i < 26; i++) {
        cin >> counts[i];
    }
    cin >> n;
    tree.resize(n + 1);
    for (int i = 1; i <= n - 1; i++) {
        int u, v;
        cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }
    
    // 求直径节点数
    pair<int,int> first = bfs(1);
    pair<int,int> second = bfs(first.first);
    int diameterEdges = second.second;
    int diameterNodes = diameterEdges + 1;
    
    // 计算最长回文长度 L
    int evenSum = 0;
    bool hasOdd = false;
    for (int i = 0; i < 26; i++) {
        evenSum += counts[i] - (counts[i] & 1);
        if (counts[i] % 2 == 1)
            hasOdd = true;
    }
    int L = evenSum + (hasOdd ? 1 : 0);
    
    int ans = min(diameterNodes, L);
    cout << ans << "\n";
    return 0;
}
```
