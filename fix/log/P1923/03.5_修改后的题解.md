## 解题思路

根据题意，一个网格**有效**需要同时满足两个条件：

1. 行数 $n$ 和列数 $m$ 均为**偶数**；
2. 网格中存在一个 $2 \times 2$ 的连续子网格，其中四个位置全部为 `1`（检测点）。

因此，解题步骤非常直接：

- 首先判断 $n$ 和 $m$ 是否都是偶数。如果不是，直接输出 `NO`；
- 若满足偶数条件，则遍历所有可能的 $2 \times 2$ 子网格：以每个 $(i, j)$ 作为左上角，其中 $0 \le i < n-1$，$0 \le j < m-1$，检查 `grid[i][j]`、`grid[i][j+1]`、`grid[i+1][j]`、`grid[i+1][j+1]` 是否均为 `'1'`；
- 一旦找到一个全为 `1` 的 $2 \times 2$ 子网格，即可判定有效，输出 `YES`；
- 若枚举完毕仍未找到，输出 `NO`。

## 复杂度分析

设网格大小为 $n \times m$。

需要枚举所有可能的 $2 \times 2$ 子网格，总数为 $(n-1) \times (m-1)$ 个，每次检查只需常数时间。

- **时间复杂度**：$O(n \cdot m)$  
- **空间复杂度**：$O(1)$（不计输入存储）

对于 $n, m \le 500$ 的数据范围，该复杂度完全可行。

## 代码实现

### Python

```python
def is_valid(n, m, grid):
    # 行数和列数必须均为偶数
    if n % 2 != 0 or m % 2 != 0:
        return False

    # 枚举所有 2x2 子网格
    for i in range(n - 1):
        for j in range(m - 1):
            # 检查四个位置是否全为 '1'
            if (grid[i][j] == '1' and
                grid[i][j + 1] == '1' and
                grid[i + 1][j] == '1' and
                grid[i + 1][j + 1] == '1'):
                return True

    return False


def main():
    T = int(input())  # 测试数据组数
    for _ in range(T):
        n, m = map(int, input().split())
        grid = [input().strip() for _ in range(n)]

        if is_valid(n, m, grid):
            print("YES")
        else:
            print("NO")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    // 判断网格是否有效
    public static boolean isValid(int n, int m, String[] grid) {
        // 行数和列数必须均为偶数
        if (n % 2 != 0 || m % 2 != 0) {
            return false;
        }

        // 枚举所有 2x2 子网格
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < m - 1; j++) {
                // 检查四个位置是否全为 '1'
                if (grid[i].charAt(j) == '1' &&
                    grid[i].charAt(j + 1) == '1' &&
                    grid[i + 1].charAt(j) == '1' &&
                    grid[i + 1].charAt(j + 1) == '1') {
                    return true;
                }
            }
        }

        return false;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt(); // 测试数据组数
        while (T-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();

            String[] grid = new String[n];
            for (int i = 0; i < n; i++) {
                grid[i] = sc.next();
            }

            if (isValid(n, m, grid)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }

        sc.close();
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// 判断网格是否有效
bool isValid(int n, int m, const vector<string>& grid) {
    // 行数和列数必须均为偶数
    if (n % 2 != 0 || m % 2 != 0) {
        return false;
    }

    // 枚举所有 2x2 子网格
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < m - 1; j++) {
            // 检查四个位置是否全为 '1'
            if (grid[i][j] == '1' &&
                grid[i][j + 1] == '1' &&
                grid[i + 1][j] == '1' &&
                grid[i + 1][j + 1] == '1') {
                return true;
            }
        }
    }

    return false;
}

int main() {
    int T;
    cin >> T; // 测试数据组数

    while (T--) {
        int n, m;
        cin >> n >> m;

        vector<string> grid(n);
        for (int i = 0; i < n; i++) {
            cin >> grid[i];
        }

        if (isValid(n, m, grid)) {
            cout << "YES" << '\n';
        } else {
            cout << "NO" << '\n';
        }
    }

    return 0;
}
```
