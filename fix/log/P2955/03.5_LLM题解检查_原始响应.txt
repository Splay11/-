## 解题思路

本题要求构造一个与给定排列 $p$ 不同的排列 $q$，使得 $q$ 的“活跃位置”个数与 $p$ 的活跃位置个数完全相同。  
活跃位置的定义为：位置 $i$（$1 \le i < n$）是活跃的，当且仅当存在某个 $j$ 满足 $i < j \le n$ 且 $a_i > a_j$（即后面有一个比自己小的元素）。

我们可以通过在最右侧的一个下降点附近做一次局部交换来构造满足条件的 $q$，具体步骤如下：

1. **处理无解的特例**  
   当 $n < 3$ 时，排列长度过小，任何交换都会改变活跃位置的个数，无法构造出不同的排列，直接输出 `NO`。

2. **寻找最右侧下降点**  
   从右向左扫描排列 $p$，找到第一个满足 $p_i > p_{i+1}$ 的位置 $i$，记为 $\textit{idx}$（下标从 $0$ 开始）。  
   这个位置实际上是原排列中最靠右的一个“活跃位置”，并且它后面的数已经是递增的。  
   如果扫描完整个数组都没有找到这样的下降点，说明排列 $p$ 是完全递增的（$p_1 < p_2 < \dots < p_n$），此时活跃位置个数为 $0$，无法构造出相同活跃位置个数的不同排列，输出 `NO`。

3. **局部交换构造新排列**  
   将 $p$ 复制为 $q$，然后在 $\textit{idx}$ 附近进行交换，使得活跃位置个数保持不变，且 $q \neq p$：
   - 若 $\textit{idx} \le n-3$，交换 $q_{\textit{idx}}$ 与 $q_{\textit{idx}+2}$；
   - 否则（即 $\textit{idx} = n-2$），交换 $q_{\textit{idx}-1}$ 与 $q_{\textit{idx}}$。

   这样只改变了最右侧下降点附近极少几个位置的相对大小，既维持了其他位置的活跃状态，又保证了整个排列的活跃位置总数不变。

4. **输出结果**  
   若成功构造出 $q$，输出 `YES` 并在下一行输出排列 $q$；否则输出 `NO`。

## 复杂度分析

- **时间复杂度**：对于每组数据，找最右侧下降点需要一次从右向左的遍历，交换和复制均为 $O(n)$，因此总时间复杂度为 $O(n)$，所有测试数据的总时间 $O(\sum n) \le O(2\times 10^5)$，可以接受。
- **空间复杂度**：需要额外存储构造出的排列 $q$，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys
input = sys.stdin.readline

def solve():
    T = int(input())                     # 测试用例数量
    for _ in range(T):
        n = int(input())                 # 排列长度
        p = list(map(int, input().split()))   # 读入原排列 p

        # 情况1：n < 3 无解
        if n < 3:
            print("NO")
            continue

        # 寻找最右侧下降点 idx，使得 p[idx] > p[idx+1]
        idx = -1
        for i in range(n - 2, -1, -1):
            if p[i] > p[i + 1]:
                idx = i
                break

        # 如果没有下降点，排列已全增，无解
        if idx < 0:
            print("NO")
            continue

        # 构造 q 并局部交换
        q = p.copy()
        if idx <= n - 3:
            # 交换 q[idx] 与 q[idx+2]
            q[idx], q[idx + 2] = q[idx + 2], q[idx]
        else:
            # idx == n-2，交换 q[idx-1] 与 q[idx]
            q[idx - 1], q[idx] = q[idx], q[idx - 1]

        # 输出结果
        print("YES")
        print(*q)

if __name__ == '__main__':
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());      // 测试用例数量
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine());  // 排列长度
            int[] a = Arrays.stream(br.readLine().split(" "))
                            .mapToInt(Integer::parseInt)
                            .toArray();               // 读入原排列 p

            // 情况1：n < 3 无解
            if (n < 3) {
                System.out.println("NO");
                continue;
            }

            // 寻找最右侧下降点 idx，使得 a[idx] > a[idx+1]
            int idx = -1;
            for (int i = n - 2; i >= 0; --i) {
                if (a[i] > a[i + 1]) {
                    idx = i;
                    break;
                }
            }
            // 如果没有下降点，排列已全增，无解
            if (idx < 0) {
                System.out.println("NO");
                continue;
            }

            // 构造 q 并局部交换
            int[] b = a.clone();
            if (idx <= n - 3) {
                // 交换 b[idx] 与 b[idx+2]
                int tmp = b[idx];
                b[idx] = b[idx + 2];
                b[idx + 2] = tmp;
            } else {
                // idx == n-2，交换 b[idx-1] 与 b[idx]
                int tmp = b[idx - 1];
                b[idx - 1] = b[idx];
                b[idx] = tmp;
            }

            // 输出结果
            System.out.println("YES");
            for (int x : b) {
                System.out.print(x + " ");
            }
            System.out.println();
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;                     // 测试用例数量
    while (T--) {
        int n;
        cin >> n;                 // 排列长度
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];          // 读入原排列 p
        }

        // 情况1：n < 3 无解
        if (n < 3) {
            cout << "NO\n";
            continue;
        }

        // 寻找最右侧下降点 idx，使得 a[idx] > a[idx+1]
        int idx = -1;
        for (int i = n - 2; i >= 0; --i) {
            if (a[i] > a[i + 1]) {
                idx = i;
                break;
            }
        }
        // 如果没有下降点，排列已全增，无解
        if (idx < 0) {
            cout << "NO\n";
            continue;
        }

        // 构造 q 并局部交换
        vector<int> b = a;
        if (idx <= n - 3) {
            // 交换 b[idx] 与 b[idx+2]
            swap(b[idx], b[idx + 2]);
        } else {
            // idx == n-2，交换 b[idx-1] 与 b[idx]
            swap(b[idx - 1], b[idx]);
        }

        // 输出结果
        cout << "YES\n";
        for (int i = 0; i < n; ++i) {
            cout << b[i] << (i + 1 < n ? ' ' : '\n');
        }
    }
    return 0;
}
```
