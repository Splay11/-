## 解题思路

本题可以转化为一个匹配问题：将玩家和关卡进行最优配对，每位玩家至多挑战一个关卡，每个关卡至多被一位玩家挑战，目标是最大化挑战成功的关卡荣誉值总和。题目的匹配规则是：只有当玩家的能量值不低于关卡的能量门槛时，才能进行挑战。

一个高效的贪心策略如下：

1. **排序**：将所有玩家按**能量值从小到大**排序；将所有关卡按**能量门槛从小到大**排序。
2. **贪心匹配**：依次处理排序后的每一位玩家，维护一个**大根堆**（按荣誉值排序）用于存放当前玩家能够挑战的所有关卡的**荣誉值**。
   - 对于当前玩家，将所有“能量门槛 ≤ 当前玩家能量值”的关卡加入大根堆。
   - 从堆中取出荣誉值最大的关卡，作为该玩家挑战的关卡（如果堆非空），并将荣誉值累加到答案中。
   - 该关卡被挑战后即从堆中移除，后续玩家不会再挑战它。
3. **正确性**：能量值小的玩家能挑战的关卡集合是能量值大的玩家能挑战的关卡集合的子集。优先满足能量值小的玩家，不会导致更优的挑战方案被错过。如果当前玩家选择了当前可挑战荣誉值最大的关卡，而在某个最优解中该关卡被后面的玩家挑战，由于后面的玩家能量值 ≥ 当前玩家，两者交换挑战的关卡后方案仍然合法，且总荣誉值不会减少。因此贪心策略能得到全局最优解。

## 复杂度分析

- 排序：对玩家能量值排序的复杂度为 $O(n \log n)$；对关卡按能量门槛排序的复杂度为 $O(m \log m)$。
- 堆操作：每个关卡最多入堆一次、出堆一次，总复杂度为 $O(m \log m)$。
- 总时间复杂度：$O(n \log n + m \log m)$。
- 空间复杂度：需要存储玩家数组、关卡数组以及一个大小为 $O(m)$ 的大根堆，空间复杂度为 $O(n + m)$。

在 $n, m \le 2 \times 10^5$ 的范围内，该复杂度足以通过。

## 代码实现

### Python

```python
import sys
import heapq

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    players = [int(next(it)) for _ in range(n)]
    challenges = []
    for _ in range(m):
        honor = int(next(it))   # 荣誉值
        threshold = int(next(it))  # 能量门槛
        challenges.append((threshold, honor))
    
    # 玩家按能量值从小到大排序
    players.sort()
    # 关卡按能量门槛从小到大排序
    challenges.sort()
    
    max_heap = []  # 大根堆（存入负值）
    ans = 0
    idx = 0
    
    for energy in players:
        # 将当前玩家能挑战的关卡荣誉值全部加入大根堆
        while idx < m and challenges[idx][0] <= energy:
            heapq.heappush(max_heap, -challenges[idx][1])
            idx += 1
        # 取出荣誉值最大的关卡进行挑战
        if max_heap:
            ans += -heapq.heappop(max_heap)
    
    print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        
        int[] players = new int[n];
        for (int i = 0; i < n; i++) {
            players[i] = sc.nextInt();
        }
        
        // 关卡信息：int[0] = 能量门槛, int[1] = 荣誉值
        int[][] challenges = new int[m][2];
        for (int i = 0; i < m; i++) {
            int honor = sc.nextInt();
            int threshold = sc.nextInt();
            challenges[i][0] = threshold;
            challenges[i][1] = honor;
        }
        
        // 玩家按能量值从小到大排序
        Arrays.sort(players);
        // 关卡按能量门槛从小到大排序
        Arrays.sort(challenges, (a, b) -> Integer.compare(a[0], b[0]));
        
        // 大根堆：存储可挑战关卡的荣誉值
        PriorityQueue<Integer> maxHeap = new PriorityQueue<>(Collections.reverseOrder());
        long ans = 0;
        int idx = 0;
        
        for (int energy : players) {
            // 将所有能量门槛 ≤ 当前玩家能量值的关卡加入堆
            while (idx < m && challenges[idx][0] <= energy) {
                maxHeap.offer(challenges[idx][1]);
                idx++;
            }
            // 选择荣誉值最大的关卡进行挑战
            if (!maxHeap.isEmpty()) {
                ans += maxHeap.poll();
            }
        }
        
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n, m;
    cin >> n >> m;
    
    vector<int> players(n);
    for (int i = 0; i < n; i++) {
        cin >> players[i];
    }
    
    // 关卡信息：pair<能量门槛, 荣誉值>
    vector<pair<int, int>> challenges(m);
    for (int i = 0; i < m; i++) {
        int honor, threshold;
        cin >> honor >> threshold;
        challenges[i] = {threshold, honor};
    }
    
    // 玩家按能量值从小到大排序
    sort(players.begin(), players.end());
    // 关卡按能量门槛从小到大排序
    sort(challenges.begin(), challenges.end());
    
    priority_queue<int> maxHeap; // 大根堆存放荣誉值
    long long ans = 0;
    int idx = 0;
    
    for (int energy : players) {
        // 将所有能量门槛 ≤ 当前玩家能量值的关卡荣誉值加入堆
        while (idx < m && challenges[idx].first <= energy) {
            maxHeap.push(challenges[idx].second);
            idx++;
        }
        // 取出荣誉值最大的关卡进行挑战
        if (!maxHeap.empty()) {
            ans += maxHeap.top();
            maxHeap.pop();
        }
    }
    
    cout << ans << '\n';
    return 0;
}
```
