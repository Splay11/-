## 解题思路

本题的目标是：从长度为 $n$ 的序列 $a_1,\dots,a_n$ 中选出若干元素构成非空子序列，使得该子序列所有元素的最大公约数（即**核心值**）恰好等于给定的正整数 $k$。求方案数对 $10^9+7$ 取模的结果。

直接统计核心值恰好为 $k$ 比较困难，我们可以借助**倍数关系**与**容斥/递推**的思想：

1. **统计倍数个数**  
   设 $cnt[x]$ 表示序列中数值等于 $x$ 的元素个数。  
   对于每个 $i$，令 $f[i]$ 表示序列中所有 $i$ 的倍数的元素个数，即  
   $$f[i] = \sum_{i \mid j} cnt[j]$$  
   可以用类似埃氏筛的方法，在 $O(M\log M)$ 时间内求出所有 $f[i]$，其中 $M$ 为数值的最大范围（题目中 $M\le 10^5$）。

2. **计算“公约数是 $i$ 的倍数”的方案数**  
   若只从 $i$ 的倍数中选取非空子序列，则子序列中所有元素都能被 $i$ 整除，因此它们的核心值也必定是 $i$ 的倍数。  
   这样的方案数即为从 $f[i]$ 个元素中任选非空子集的个数：  
   $$g[i] = 2^{f[i]} - 1$$  
   由于 $f[i]$ 可能较大，实际计算时需要对 $10^9+7$ 取模，并可使用快速幂或预处理的幂表。

3. **递推求恰好为核心值 $i$ 的方案数**  
   设 $dp[i]$ 表示核心值恰好等于 $i$ 的子序列个数。  
   由容斥原理，$g[i]$ 包含了所有核心值为 $i$ 的倍数的方案，因此：  
   $$dp[i] = g[i] - \sum_{\substack{j > i \\ i \mid j}} dp[j]$$  
   从大到小倒序枚举 $i$，同样采用埃氏筛的方式累加倍数 $j$ 的 $dp[j]$，即可在 $O(M\log M)$ 时间内求出所有 $dp[i]$。

4. **答案**  
   最终所求即为 $dp[k]$。若序列中没有任何元素是 $k$ 的倍数，则由上述过程 $dp[k]$ 自然为 $0$。

## 复杂度分析

- 时间复杂度：$O(M \log M)$，其中 $M = \max\{k, \max a_i\}$（本题中 $M\le 10^5$）。统计 $f$ 和递推 $dp$ 均需要两重循环，单重循环的复杂度为调和级数级别 $O(M\log M)$。
- 空间复杂度：$O(M)$，需要存储 $cnt, f, g, dp$ 等数组以及预处理 $2$ 的幂次。

## 代码实现

### Python

```python
MOD = 10**9 + 7

def solve():
    import sys
    input = sys.stdin.readline
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    # 数据范围均在10^5以内，设定最大范围
    max_val = max(max(a), k) if a else k
    # 如果 k 大于所有元素，则方案数为0
    if k > max_val:
        print(0)
        return
    M = max_val + 1
    cnt = [0] * M
    for v in a:
        cnt[v] += 1
    
    # 预处理2的幂次，最大次数为n
    pow2 = [1] * (n + 1)
    for i in range(1, n + 1):
        pow2[i] = (pow2[i-1] * 2) % MOD
    
    # f[i] = i 的倍数的个数
    f = [0] * M
    for i in range(1, M):
        for j in range(i, M, i):
            f[i] += cnt[j]
    
    # g[i] = 2^{f[i]} - 1
    g = [0] * M
    for i in range(1, M):
        g[i] = (pow2[f[i]] - 1) % MOD
    
    # dp[i] = gcd恰好为i的方案数
    dp = [0] * M
    for i in range(M - 1, 0, -1):
        dp[i] = g[i]
        for j in range(i * 2, M, i):
            dp[i] = (dp[i] - dp[j]) % MOD
    
    print(dp[k] % MOD)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static final int MOD = 1000000007;
    static final int MAXN = 100005; // 足够覆盖 a_i, k <= 10^5

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] first = br.readLine().split(" ");
        int n = Integer.parseInt(first[0]);
        int k = Integer.parseInt(first[1]);
        String[] parts = br.readLine().split(" ");
        int[] a = new int[n];
        int maxA = 0;
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(parts[i]);
            if (a[i] > maxA) maxA = a[i];
        }
        if (k > maxA) {
            System.out.println(0);
            return;
        }
        int M = Math.max(maxA, k) + 1;
        int[] cnt = new int[M];
        for (int v : a) {
            cnt[v]++;
        }
        // 预处理2的幂，最大到n
        long[] pow2 = new long[n + 1];
        pow2[0] = 1;
        for (int i = 1; i <= n; i++) {
            pow2[i] = (pow2[i-1] * 2) % MOD;
        }
        // f[i] = 倍数个数
        int[] f = new int[M];
        for (int i = 1; i < M; i++) {
            for (int j = i; j < M; j += i) {
                f[i] += cnt[j];
            }
        }
        // g[i] = (2^f[i] - 1) % MOD
        long[] g = new long[M];
        for (int i = 1; i < M; i++) {
            g[i] = (pow2[f[i]] - 1 + MOD) % MOD;
        }
        // dp倒推
        long[] dp = new long[M];
        for (int i = M - 1; i >= 1; i--) {
            dp[i] = g[i];
            for (int j = i * 2; j < M; j += i) {
                dp[i] = (dp[i] - dp[j] + MOD) % MOD;
            }
        }
        System.out.println(dp[k]);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100005;
const int MOD = 1000000007;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    int maxA = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] > maxA) maxA = a[i];
    }
    // 如果 k 大于所有元素，不可能选出任何子序列
    if (k > maxA) {
        cout << 0 << "\n";
        return 0;
    }
    int M = max(maxA, k) + 1;
    vector<int> cnt(M, 0);
    for (int v : a) cnt[v]++;

    // 预处理2的幂次
    vector<long long> pow2(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        pow2[i] = (pow2[i-1] * 2) % MOD;
    }

    // f[i] = i的倍数的个数
    vector<int> f(M, 0);
    for (int i = 1; i < M; i++) {
        for (int j = i; j < M; j += i) {
            f[i] += cnt[j];
        }
    }

    // g[i] = 2^{f[i]} - 1
    vector<long long> g(M, 0);
    for (int i = 1; i < M; i++) {
        g[i] = (pow2[f[i]] - 1 + MOD) % MOD;
    }

    // dp[i] = 核心值恰好为i的方案数
    vector<long long> dp(M, 0);
    for (int i = M - 1; i >= 1; i--) {
        dp[i] = g[i];
        for (int j = i * 2; j < M; j += i) {
            dp[i] = (dp[i] - dp[j] + MOD) % MOD;
        }
    }

    cout << dp[k] << "\n";
    return 0;
}
```
