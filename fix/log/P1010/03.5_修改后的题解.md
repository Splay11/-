## 解题思路

长度为 $g$ 的所有连续工位段读数之和相等，等价于序列以 $g$ 为周期，即下标模 $g$ 相同的工位最终必须取同一个值。一次校准只能把某个读数加 $1$，因此每一组（下标 $\equiv r \pmod g$）只能把所有读数抬到不小于该组当前最大值的某个公共值。

先把每一组齐到该组当前最大值，花费为 $\mathrm{max}\cdot \mathrm{cnt}-\mathrm{sum}$。若总花费超过 $d$，则不可能，输出 $-1$。否则剩余校准次数可以全部加到同一组上（每组工位数为 $\mathrm{cnt}$，该组还能整体再抬 $\lfloor d_{\mathrm{rest}}/\mathrm{cnt}\rfloor$）。对所有组取最大值即可。

## 复杂度分析

时间复杂度 $O(n)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
n, k, x = map(int, input().split())
a = list(map(int, input().split()))
for i in range(k):
    tmp = a[i:n:k]
    x -= max(tmp) * len(tmp) - sum(tmp)
if x >= 0:
    mx = 0
    for i in range(k):
        tmp = a[i:n:k]
        mx = max(mx, max(tmp) + x // len(tmp))
    print(mx)
else:
    print(-1)
```

### Java

```java
import java.util.ArrayList;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int x = scanner.nextInt();
        ArrayList<Integer>[] p = new ArrayList[k];
        for (int i = 0; i < k; i++)
            p[i] = new ArrayList<>();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
            p[i % k].add(a[i]);
        }
        int ans = -1000000000;
        for (int i = 0; i < k; i++) {
            int mxx = -1000000000;
            int sum = 0;
            for (int j = 0; j < p[i].size(); j++) {
                mxx = Math.max(mxx, p[i].get(j));
                sum += p[i].get(j);
            }
            x -= mxx * p[i].size() - sum;
        }
        if (x >= 0) {
            for (int i = 0; i < k; i++) {
                int mxx = -1000000000;
                for (int j = 0; j < p[i].size(); j++) {
                    mxx = Math.max(mxx, p[i].get(j));
                }
                ans = Math.max(ans, mxx + x / p[i].size());
            }
            System.out.println(ans);
        } else
            System.out.println(-1);
    }
}
```

#include <iostream>
#include <vector>
using namespace std;
int a[100005];
int main()
{
    int n, k, x;
    cin >> n >> k >> x;
    vector<int> p[k];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        p[i % k].push_back(a[i]);
    }
    int ans = -1000000000;
    for (int i = 0; i < k; i++) {
        int mxx = -1000000000;
        int sum = 0;
        for (int j = 0; j < p[i].size(); j++) {
            mxx = max(mxx, p[i][j]);
            sum += p[i][j];
        }
        x -= mxx * p[i].size() - sum;
    }
    if (x >= 0) {
        for (int i = 0; i < k; i++) {
            int mxx = -1000000000;
            for (int j = 0; j < p[i].size(); j++) {
                mxx = max(mxx, p[i][j]);
            }
            ans = max(ans, mxx + x / (int)p[i].size());
        }
        cout << ans << endl;
    } else
        cout << -1 << endl;
    return 0;
}
