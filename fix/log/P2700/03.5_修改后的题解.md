## 解题思路

题目要求预处理所有可能的长度 $L$ 对应的区间能量值之和，然后回答 $q$ 次查询。由于 $n$ 与 $q$ 的总和均可达到 $3\times10^5$，必须在线性或接近线性的时间内完成预处理。

核心观察：
- 若区间乘积超过 $10^9$ 则视为 $0$，因此我们只关心乘积 $\le 10^9$ 的区间。
- 包含 $0$ 的区间乘积恒为 $0$（且不可能超过上界），可以直接跳过，不需要额外处理。
- 因此可以将原序列按 $0$ 分隔成多个“非零段”，对每一段单独计算贡献。

对于一段不含 $0$ 的连续子数组 $\text{seg}$（长度为 $m$）：
1. **全为 $1$ 的段**：任何子区间乘积均为 $1$，且不会超过 $10^9$。长度为 $L$ 的子区间个数为 $m - L + 1$，对 $\text{ans}[L]$ 的贡献就是 $m - L + 1$。
2. **非全 $1$ 的段**（即至少含一个 $>1$ 的数）：乘积随长度指数增长，很快超过 $10^9$。对于每个起点 $i$（下标从 $0$ 开始），我们从 $i$ 出发向右延伸，维护当前乘积 $\text{prod}$，直到 $\text{prod} \times \text{seg}[j] > 10^9$ 时停止。对每个合法的终点 $j$，区间长度 $L = j - i + 1$，贡献为 $\text{prod}$，直接累加到 $\text{ans}[L]$ 上。使用除法预判避免溢出：在乘 $\text{seg}[j]$ 前检查 $\text{prod} > 10^9 / \text{seg}[j]$ 则跳出。

预处理完后，对于每个查询 $L$，直接输出 $\text{ans}[L]$ 即可。

该算法保证每个非全 $1$ 段的内层循环总次数为 $O(m \cdot \log_{2} (10^9))$ 实际上非常小，总体时间复杂度可接受。

## 复杂度分析

- 时间复杂度：每个元素被扫描常数次（全 $1$ 段公式计算 $O(m)$，非全 $1$ 段每个起点最多延伸约 $30$ 次），所有段的总处理时间 $O(n)$；查询 $O(1)$ 应答，总复杂度 $O(n + q)$，满足 $n, q$ 总和 $3\times10^5$ 的限制。
- 空间复杂度：需要存储序列 $A$ 和答案数组 $\text{ans}$，大小均为 $O(n)$，总空间 $O(n)$。

## 代码实现

### Python

```python
import sys

LIMIT = 10 ** 9

def solve():
    data = sys.stdin.read().split()
    it = iter(data)
    t = int(next(it))
    out_lines = []
    for _ in range(t):
        n = int(next(it))
        q = int(next(it))
        A = [int(next(it)) for _ in range(n)]
        # ans[L] 表示所有长度为 L 的区间能量值之和，1-indexed
        ans = [0] * (n + 1)

        i = 0
        while i < n:
            if A[i] == 0:
                # 包含 0 的区间能量值均为 0，跳过
                i += 1
                continue

            j = i
            while j < n and A[j] != 0:
                j += 1
            seg = A[i:j]
            m = j - i

            # 判断是否全为 1
            if all(x == 1 for x in seg):
                # 全 1 段：长度为 L 的区间有 m-L+1 个，贡献均为 1
                for L in range(1, m + 1):
                    ans[L] += (m - L + 1)
            else:
                # 非全 1 段：枚举起点，累乘直到超限
                for start in range(m):
                    prod = 1
                    for end in range(start, m):
                        if prod > LIMIT // seg[end]:
                            break
                        prod *= seg[end]
                        if prod > LIMIT:
                            break
                        length = end - start + 1
                        ans[length] += prod
            i = j

        # 回答询问
        for _ in range(q):
            L = int(next(it))
            out_lines.append(str(ans[L]))
        # 多组数据之间不需要额外空行，但按题目要求直接输出
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long LIMIT = 1_000_000_000L;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter pw = new PrintWriter(new BufferedOutputStream(System.out));
        int T = Integer.parseInt(br.readLine());

        while (T-- > 0) {
            String[] first = br.readLine().split(" ");
            int n = Integer.parseInt(first[0]);
            int q = Integer.parseInt(first[1]);

            String[] nums = br.readLine().split(" ");
            long[] A = new long[n];
            for (int i = 0; i < n; i++) {
                A[i] = Long.parseLong(nums[i]);
            }

            long[] ans = new long[n + 1]; // 1-indexed

            int i = 0;
            while (i < n) {
                if (A[i] == 0) {
                    i++;
                    continue;
                }
                int j = i;
                while (j < n && A[j] != 0) j++;
                int m = j - i;
                long[] seg = new long[m];
                for (int k = 0; k < m; k++) seg[k] = A[i + k];

                boolean allOnes = true;
                for (long v : seg) {
                    if (v != 1) {
                        allOnes = false;
                        break;
                    }
                }

                if (allOnes) {
                    for (int L = 1; L <= m; L++) {
                        ans[L] += (m - L + 1);
                    }
                } else {
                    for (int start = 0; start < m; start++) {
                        long prod = 1;
                        for (int end = start; end < m; end++) {
                            if (prod > LIMIT / seg[end]) break;
                            prod *= seg[end];
                            if (prod > LIMIT) break;
                            int length = end - start + 1;
                            ans[length] += prod;
                        }
                    }
                }
                i = j;
            }

            for (int k = 0; k < q; k++) {
                int L = Integer.parseInt(br.readLine());
                pw.println(ans[L]);
            }
        }
        pw.flush();
        pw.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const ll LIMIT = 1'000'000'000LL;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        int n, q;
        cin >> n >> q;
        vector<ll> A(n);
        for (int i = 0; i < n; ++i) cin >> A[i];

        vector<ll> ans(n + 1, 0); // 1-indexed

        int i = 0;
        while (i < n) {
            if (A[i] == 0) {
                ++i;
                continue;
            }
            int j = i;
            while (j < n && A[j] != 0) ++j;
            int m = j - i;
            vector<ll> seg(A.begin() + i, A.begin() + j);

            bool allOnes = true;
            for (ll x : seg) {
                if (x != 1) {
                    allOnes = false;
                    break;
                }
            }

            if (allOnes) {
                for (int L = 1; L <= m; ++L) {
                    ans[L] += (ll)(m - L + 1);
                }
            } else {
                for (int start = 0; start < m; ++start) {
                    ll prod = 1;
                    for (int end = start; end < m; ++end) {
                        if (prod > LIMIT / seg[end]) break;
                        prod *= seg[end];
                        if (prod > LIMIT) break;
                        int length = end - start + 1;
                        ans[length] += prod;
                    }
                }
            }
            i = j;
        }

        for (int k = 0; k < q; ++k) {
            int L;
            cin >> L;
            cout << ans[L] << '\n';
        }
    }
    return 0;
}
```
