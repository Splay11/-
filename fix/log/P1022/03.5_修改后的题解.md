## 解题思路

本题要求求出最少的增加操作次数，使得网格在顺时针旋转 $0^\circ$、$90^\circ$、$180^\circ$、$270^\circ$ 后完全相同。由于每次只能将某个格子的数值增加 $1$，要想使若干格子最终相等，最优的做法是将它们全部增加到当前的最大值，所需操作次数即为 $\sum (\text{最大值} - \text{当前值})$。

关键观察：**旋转四次对应一组格子**。对于网格中的任意位置 $(x, y)$（行列编号从 $0$ 开始），顺时针旋转 $90^\circ$ 后的新坐标为 $(y, n-1-x)$。连续旋转四次会回到起点，因此除了 $n$ 为奇数时的正中心格子单独成一组（大小为 $1$），其余所有格子都可以按旋转轨道分成大小为 $4$ 的组。

算法步骤如下：
1. 读入 $n$ 和 $n \times n$ 的网格数值。
2. 遍历网格的每一个格子 $(i, j)$。
3. 从 $(i, j)$ 出发，旋转四次，找到该轨道内所有格子的最大值 `mxval`。
4. 再次旋转四次，对轨道内的每个格子累加 `mxval - 当前值` 到答案中，并将该格子的值更新为 `mxval`，防止后续遍历到该轨道时重复计算。
5. 对于中心格子（当 $n$ 为奇数时），其轨道只有自身，最大值为自身，累加 $0$，不影响结果。
6. 输出累加的总操作次数。

注意：由于每个格子的初始值最大为 $10^9$，且网格规模可达 $100 \times 100$，总操作次数可能达到 $10^{13}$ 级别，超出 $32$ 位整数范围，因此需要使用 $64$ 位整数（C++ 的 `long long`，Java 的 `long`，Python 自动支持大整数）。

## 复杂度分析

- **时间复杂度**：$O(n^2)$。每个格子被访问常数次（旋转四次寻找最大值再旋转四次更新），总访问量与网格元素总数成正比。
- **空间复杂度**：$O(n^2)$。需要存储整个 $n \times n$ 网格的数值。

## 代码实现

### Python

```python
n = int(input())
mp = []
for i in range(n):
    mp.append(list(map(int, input().split())))

def rotate(x, y):
    # 顺时针旋转90度：(x, y) -> (y, n-1-x)
    return y, n - 1 - x

ans = 0
for i in range(n):
    for j in range(n):
        # 第一步：寻找当前轨道中的最大值
        mxval = mp[i][j]
        x, y = i, j
        for _ in range(4):
            x, y = rotate(x, y)
            mxval = max(mxval, mp[x][y])
        # 第二步：将轨道中所有格子提升到最大值，累加操作次数
        for _ in range(4):
            x, y = rotate(x, y)
            ans += mxval - mp[x][y]
            mp[x][y] = mxval   # 更新为最大值，防止后续重复累加

print(ans)
```

### Java

```java
import java.util.Scanner;

public class Main {
    static int n;

    // 顺时针旋转90度：返回新坐标 [y, n-1-x]
    static int[] rotate(int x, int y) {
        return new int[]{y, n - 1 - x};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        int[][] mp = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                mp[i][j] = sc.nextInt();
            }
        }

        long ans = 0;   // 总操作次数可能很大，使用 long
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                // 寻找轨道最大值
                int mxval = mp[i][j];
                int x = i, y = j;
                for (int t = 0; t < 4; t++) {
                    int[] next = rotate(x, y);
                    x = next[0];
                    y = next[1];
                    mxval = Math.max(mxval, mp[x][y]);
                }
                // 累加操作次数并更新数值
                for (int t = 0; t < 4; t++) {
                    int[] next = rotate(x, y);
                    x = next[0];
                    y = next[1];
                    ans += mxval - mp[x][y];
                    mp[x][y] = mxval;
                }
            }
        }
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

int n;
int mp[105][105];   // n ≤ 100，预留足够空间

// 顺时针旋转90度：将(x, y)变为(y, n-1-x)，使用引用修改
void rotate(int &x, int &y) {
    int tmp = x;
    x = y;
    y = n - tmp - 1;
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> mp[i][j];
        }
    }

    long long ans = 0;   // 可能超出 int 范围，使用 long long
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            // 寻找轨道中的最大值
            int mxval = mp[i][j];
            int x = i, y = j;
            for (int t = 0; t < 4; t++) {
                rotate(x, y);
                mxval = max(mxval, mp[x][y]);
            }
            // 累加操作次数并更新
            for (int t = 0; t < 4; t++) {
                rotate(x, y);
                ans += mxval - mp[x][y];
                mp[x][y] = mxval;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
```
