## 解题思路

需要统计流水号 $t$ 的所有非空有效子序列积分。有效条件为：无前导零（单独的 $0$ 除外），且数字和能被 $5$ 整除。

$|t|$ 可达 $10^5$，不能枚举子序列。从左到右扫描，用长度为 $5$ 的数组维护：

- $dp[r]$：首位不为 $0$ 的子序列中，数字和模 $5$ 等于 $r$ 的个数；
- $dpP[r]$：其中首位属于 $\{2,3,5,7\}$ 的个数。

对每个新数字 $d$：

1. **单字符**：若 $d \equiv 0 \pmod 5$，它本身有效，按首位/末位规则加分。单独的 $0$ 不计后续扩展。
2. **扩展**：每个旧余数 $r$ 的子序列末尾追加 $d$，新余数为 $(r+d) \bmod 5$。若新余数为 $0$，按末位奇偶与首位是否属于 $\{2,3,5,7\}$ 计分。
3. 将非零单字符作为新子序列并入 $dp$。

每个字符只做常数次更新。

## 复杂度分析

时间复杂度 $O(|t|)$，空间复杂度 $O(1)$。

## 代码实现

### Python

```python
MOD = 10**9 + 7

def solve(t: str) -> int:
    dp = [0] * 5
    dpP = [0] * 5
    primes = {'2', '3', '5', '7'}
    ans = 0

    for ch in t:
        d = ord(ch) - ord('0')
        isP = ch in primes
        if d % 5 == 0:
            if ch == '0':
                ans = (ans + 1) % MOD
            else:
                if isP:
                    ans = (ans + (4 if d % 2 == 0 else 3)) % MOD
                else:
                    ans = (ans + 1) % MOD
        add_dp = [0] * 5
        add_dpP = [0] * 5
        for r in range(5):
            cnt = dp[r]
            if cnt == 0:
                continue
            cntP = dpP[r]
            r2 = (r + d) % 5
            if r2 == 0:
                if d % 2 == 0:
                    ans = (ans + cntP * 4) % MOD
                else:
                    ans = (ans + cntP * 3) % MOD
                ans = (ans + (cnt - cntP) * 1) % MOD
            add_dp[r2] = (add_dp[r2] + cnt) % MOD
            add_dpP[r2] = (add_dpP[r2] + cntP) % MOD

        for r in range(5):
            dp[r] = (dp[r] + add_dp[r]) % MOD
            dpP[r] = (dpP[r] + add_dpP[r]) % MOD

        if ch != '0':
            r0 = d % 5
            dp[r0] = (dp[r0] + 1) % MOD
            if isP:
                dpP[r0] = (dpP[r0] + 1) % MOD

    return ans

if __name__ == "__main__":
    print(solve(input().strip()))
```

```java
import java.io.*;

public class Main {
    static final int MOD = 1000000007;
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String t = br.readLine().trim();
        long[] dp = new long[5];
        long[] dpP = new long[5];
        String primes = "2357";
        long ans = 0;

        for (char ch : t.toCharArray()) {
            int d = ch - '0';
            boolean isP = primes.indexOf(ch) >= 0;
            if (d % 5 == 0) {
                if (ch == '0') {
                    ans = (ans + 1) % MOD;
                } else if (isP) {
                    ans = (ans + (d % 2 == 0 ? 4 : 3)) % MOD;
                } else {
                    ans = (ans + 1) % MOD;
                }
            }
            long[] add = new long[5], addP = new long[5];
            for (int r = 0; r < 5; r++) {
                long cnt = dp[r];
                if (cnt == 0) continue;
                long cntP = dpP[r];
                int r2 = (r + d) % 5;
                if (r2 == 0) {
                    if (d % 2 == 0) ans = (ans + cntP * 4) % MOD;
                    else ans = (ans + cntP * 3) % MOD;
                    ans = (ans + (cnt - cntP)) % MOD;
                }
                add[r2] = (add[r2] + cnt) % MOD;
                addP[r2] = (addP[r2] + cntP) % MOD;
            }
            for (int r = 0; r < 5; r++) {
                dp[r] = (dp[r] + add[r]) % MOD;
                dpP[r] = (dpP[r] + addP[r]) % MOD;
            }
            if (ch != '0') {
                int r0 = d % 5;
                dp[r0] = (dp[r0] + 1) % MOD;
                if (isP) dpP[r0] = (dpP[r0] + 1) % MOD;
            }
        }
        System.out.println(ans);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);

    string t;
    cin >> t;
    long long dp[5] = {0}, dpP[5] = {0};
    auto isPrimeDigit = [&](char c) {
        return c == '2' || c == '3' || c == '5' || c == '7';
    };
    long long ans = 0;

    for (char ch : t) {
        int d = ch - '0';
        bool isP = isPrimeDigit(ch);
        if (d % 5 == 0) {
            if (ch == '0') {
                ans = (ans + 1) % MOD;
            } else if (isP) {
                ans = (ans + (d % 2 == 0 ? 4 : 3)) % MOD;
            } else {
                ans = (ans + 1) % MOD;
            }
        }
        long long add[5] = {0}, addP[5] = {0};
        for (int r = 0; r < 5; r++) {
            long long cnt = dp[r];
            if (!cnt) continue;
            long long cntP = dpP[r];
            int r2 = (r + d) % 5;
            if (r2 == 0) {
                ans = (ans + cntP * (d % 2 == 0 ? 4 : 3)) % MOD;
                ans = (ans + (cnt - cntP)) % MOD;
            }
            add[r2] = (add[r2] + cnt) % MOD;
            addP[r2] = (addP[r2] + cntP) % MOD;
        }
        for (int r = 0; r < 5; r++) {
            dp[r] = (dp[r] + add[r]) % MOD;
            dpP[r] = (dpP[r] + addP[r]) % MOD;
        }
        if (ch != '0') {
            int r0 = d % 5;
            dp[r0] = (dp[r0] + 1) % MOD;
            if (isP) dpP[r0] = (dpP[r0] + 1) % MOD;
        }
    }

    cout << ans << "\n";
    return 0;
}
```
