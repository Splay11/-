## 解题思路

1. **问题转化**  
   题目要求最终序列的每个元素必须属于集合 $Y$，且整体序列非递减。为了处理允许值的限制，我们先对集合 $Y$ 进行排序，得到有序序列 $b_1 \le b_2 \le \dots \le b_m$。这样，最终序列中每个元素必然落在某个 $b_j$ 上。

2. **动态规划定义**  
   定义 $dp[i][j]$ 表示考虑前 $i$ 个元素，且将第 $i$ 个元素修改为 $b_j$ 时，满足前 $i$ 个元素均取自 $Y$ 且非递减的最小操作次数。

3. **状态转移**  
   为了保证非递减，第 $i-1$ 个元素只能取 $b_k$ 且 $k \le j$。因此：
   $$dp[i][j] = \min_{1 \le k \le j} \Big( dp[i-1][k] \Big) + |x_i - b_j|$$
   注意到前面部分恰好是 $dp[i-1][1\dots j]$ 的最小值，我们可以在计算 $dp[i]$ 这一层时维护前缀最小值 $pre[j] = \min_{1 \le k \le j} dp[i-1][k]$，从而将转移优化为：
   $$dp[i][j] = pre[j] + |x_i - b_j|$$

4. **边界与答案**  
   - $dp[0][j] = 0$ （前 $0$ 个元素无需操作）  
   - 最终答案为 $\min_{j=1}^{m} dp[n][j]$，即前 $n$ 个元素修改后，最后一个元素为任意 $b_j$ 时的全局最小值。

## 复杂度分析

- **时间复杂度**：状态数为 $O(n \times m)$，每次转移借助前缀最小值实现 $O(1)$ 更新，总时间复杂度为 $O(nm)$。  
- **空间复杂度**：使用了二维 $dp$ 数组和前缀最小值数组，空间复杂度为 $O(nm)$；若使用滚动数组可优化至 $O(m)$，但本题 $n,m \le 2000$，$O(nm)$ 空间在可接受范围内。

## 代码实现

### Python

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    x = list(map(int, input().split()))
    y = list(map(int, input().split()))
    
    y.sort()  # 对允许值集合排序
    INF = 10**18
    
    # dp[i][j] 表示前i个元素，第i个变为y[j]的最小操作次数
    dp = [[INF] * (m + 1) for _ in range(n + 1)]
    dp[0] = [0] * (m + 1)  # 前0个元素的边界情况
    
    for i in range(1, n + 1):
        prefix_min = INF
        for j in range(1, m + 1):
            # prefix_min 维护了 min_{1<=k<=j} dp[i-1][k]
            if dp[i-1][j] < prefix_min:
                prefix_min = dp[i-1][j]
            dp[i][j] = prefix_min + abs(x[i-1] - y[j-1])
    
    ans = min(dp[n][1:])  # 最后一个元素可以是任意y[j]
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] x = new long[n + 1];
        long[] y = new long[m + 1];
        for (int i = 1; i <= n; i++) {
            x[i] = sc.nextLong();
        }
        for (int i = 1; i <= m; i++) {
            y[i] = sc.nextLong();
        }
        Arrays.sort(y, 1, m + 1);  // 对允许值集合排序
        
        long INF = Long.MAX_VALUE / 2;
        long[][] dp = new long[n + 1][m + 1];
        for (long[] row : dp) {
            Arrays.fill(row, INF);
        }
        Arrays.fill(dp[0], 0);  // 前0个元素的边界
        
        for (int i = 1; i <= n; i++) {
            long prefixMin = INF;
            for (int j = 1; j <= m; j++) {
                if (dp[i - 1][j] < prefixMin) {
                    prefixMin = dp[i - 1][j];
                }
                dp[i][j] = prefixMin + Math.abs(x[i] - y[j]);
            }
        }
        
        long ans = INF;
        for (int j = 1; j <= m; j++) {
            if (dp[n][j] < ans) {
                ans = dp[n][j];
            }
        }
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n, m;
    cin >> n >> m;
    vector<ll> x(n + 1), y(m + 1);
    for (int i = 1; i <= n; i++) cin >> x[i];
    for (int i = 1; i <= m; i++) cin >> y[i];
    sort(y.begin() + 1, y.end());  // 对允许值集合排序
    
    const ll INF = 1e18;
    vector<vector<ll>> dp(n + 1, vector<ll>(m + 1, INF));
    fill(dp[0].begin(), dp[0].end(), 0);  // 前0个元素的边界
    
    for (int i = 1; i <= n; i++) {
        ll prefix_min = INF;
        for (int j = 1; j <= m; j++) {
            prefix_min = min(prefix_min, dp[i - 1][j]);
            dp[i][j] = prefix_min + abs(x[i] - y[j]);
        }
    }
    
    ll ans = *min_element(dp[n].begin() + 1, dp[n].end());
    cout << ans << "\n";
    
    return 0;
}
```
