## 解题思路

按题意模拟双门控稀疏加权。设 $Q,K,V\in\mathbb{R}^{L\times d}$。

先计算原始得分

$$
S=\frac{QK^{\top}}{\sqrt{d}}
$$

对每一行，设最大值为 $s_{\max}$，门限 $t_1=\alpha s_{\max}$，$t_2=\beta s_{\max}$。对分数 $x$ 做折线映射：

$$
\tilde{S}_{ij}=
\begin{cases}
0, & S_{ij}<t_1 \\
\dfrac{S_{ij}-t_1}{t_2-t_1}, & t_1\le S_{ij}<t_2 \\
1, & S_{ij}\ge t_2
\end{cases}
$$

若 $t_1$ 与 $t_2$ 几乎重合，则小于 $t_1$ 置 `0`，否则置 `1`。

然后每行只保留最大的 $k_{\mathrm{top}}$ 个位置，其余置 `0`。行归一化：行和为 `0` 则改为均匀分布，否则除以行和得到 $A$。最后 $O=AV$。

$\alpha$、$\beta$ 缺省为 `0.2` 与 `0.6`。输出时所有数值保留 `6` 位小数。

## 复杂度分析

时间复杂度 $O(L^2 d+L^2\log L)$，空间复杂度 $O(L^2+Ld)$。在 $L\le 6$、$d\le 4$ 下完全可行。

## 代码实现

### Python

```python
import sys
import json
import math
import numpy as np


# 实现双门控稀疏加权
def tg_sa(Q, K, V, k_top, alpha=0.2, beta=0.6):
    # 转为 numpy 数组，方便矩阵运算
    Q = np.array(Q, dtype=float)
    K = np.array(K, dtype=float)
    V = np.array(V, dtype=float)

    # 获取序列长度 L 和隐藏维度 d
    L, d = Q.shape

    # 1. 计算初始打分矩阵 S = QK^T / sqrt(d)
    S = np.dot(Q, K.T) / math.sqrt(d)

    # 初始化映射后的分数矩阵
    T = np.zeros((L, L), dtype=float)

    # 2. 对每一行做双门控折线映射
    for i in range(L):
        # 当前行最大值
        s_max = np.max(S[i])

        # 两个门限
        t1 = alpha * s_max
        t2 = beta * s_max

        for j in range(L):
            x = S[i][j]

            # 如果两个门限重合，避免除零
            if abs(t2 - t1) < 1e-12:
                if x < t1:
                    T[i][j] = 0.0
                else:
                    T[i][j] = 1.0
            else:
                # 分段映射
                if x < t1:
                    T[i][j] = 0.0
                elif x < t2:
                    T[i][j] = (x - t1) / (t2 - t1)
                else:
                    T[i][j] = 1.0

    # 3. 每行只保留 Top-k 最大值，其余置 0
    for i in range(L):
        if k_top < L:
            # 取当前行按从大到小排序后的前 k_top 个下标
            idx = np.argsort(-T[i])[:k_top]

            # 构造保留后的新行
            new_row = np.zeros(L, dtype=float)
            for p in idx:
                new_row[p] = T[i][p]
            T[i] = new_row

    # 4. 行归一化，若全 0 则改为均匀分布
    A = np.zeros((L, L), dtype=float)
    for i in range(L):
        row_sum = np.sum(T[i])

        if abs(row_sum) < 1e-12:
            # 全 0 时改为均匀分布
            A[i] = np.ones(L, dtype=float) / L
        else:
            # 否则归一化
            A[i] = T[i] / row_sum

    # 5. 计算输出 O = A V
    O = np.dot(A, V)

    return A, O


def main():
    # 读取整行输入
    s = sys.stdin.readline().strip()

    # 解析 JSON
    data = json.loads(s)

    # 读取输入数据，alpha 和 beta 允许缺省
    Q = data["Q"]
    K = data["K"]
    V = data["V"]
    k_top = data["k_top"]
    alpha = data.get("alpha", 0.2)
    beta = data.get("beta", 0.6)

    # 调用函数计算结果
    A, O = tg_sa(Q, K, V, k_top, alpha, beta)

    # 按题意保留 6 位小数
    A_out = []
    for row in A:
        A_out.append([round(float(x), 6) for x in row])

    O_out = []
    for row in O:
        O_out.append([round(float(x), 6) for x in row])

    # 按要求输出一行 JSON
    ans = {
        "A": A_out,
        "O": O_out
    }
    print(json.dumps(ans, ensure_ascii=False, separators=(",", ":")))


if __name__ == "__main__":
    main()
```

### Java

```java
import com.google.gson.*;
import java.util.*;

public class Main {
    static double[][] toMatrix(JsonArray arr) {
        int n = arr.size();
        int d = arr.get(0).getAsJsonArray().size();
        double[][] a = new double[n][d];
        for (int i = 0; i < n; i++) {
            JsonArray row = arr.get(i).getAsJsonArray();
            for (int j = 0; j < d; j++) a[i][j] = row.get(j).getAsDouble();
        }
        return a;
    }

    static double[][] tgSa(double[][] Q, double[][] K, double[][] V, int kTop, double alpha, double beta) {
        int L = Q.length;
        int d = Q[0].length;
        double[][] S = new double[L][L];
        double inv = 1.0 / Math.sqrt(d);
        for (int i = 0; i < L; i++) {
            for (int j = 0; j < L; j++) {
                double s = 0;
                for (int t = 0; t < d; t++) s += Q[i][t] * K[j][t];
                S[i][j] = s * inv;
            }
        }
        double[][] T = new double[L][L];
        for (int i = 0; i < L; i++) {
            double sMax = S[i][0];
            for (int j = 1; j < L; j++) sMax = Math.max(sMax, S[i][j]);
            double t1 = alpha * sMax;
            double t2 = beta * sMax;
            for (int j = 0; j < L; j++) {
                double x = S[i][j];
                if (Math.abs(t2 - t1) < 1e-12) {
                    T[i][j] = (x < t1) ? 0.0 : 1.0;
                } else if (x < t1) {
                    T[i][j] = 0.0;
                } else if (x < t2) {
                    T[i][j] = (x - t1) / (t2 - t1);
                } else {
                    T[i][j] = 1.0;
                }
            }
        }
        if (kTop < L) {
            for (int i = 0; i < L; i++) {
                Integer[] idx = new Integer[L];
                for (int j = 0; j < L; j++) idx[j] = j;
                final double[] row = T[i];
                Arrays.sort(idx, (p, q) -> Double.compare(row[q], row[p]));
                double[] neu = new double[L];
                for (int t = 0; t < kTop; t++) neu[idx[t]] = T[i][idx[t]];
                T[i] = neu;
            }
        }
        double[][] A = new double[L][L];
        for (int i = 0; i < L; i++) {
            double sum = 0;
            for (int j = 0; j < L; j++) sum += T[i][j];
            if (Math.abs(sum) < 1e-12) {
                for (int j = 0; j < L; j++) A[i][j] = 1.0 / L;
            } else {
                for (int j = 0; j < L; j++) A[i][j] = T[i][j] / sum;
            }
        }
        return A;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String line = sc.nextLine();
        sc.close();
        JsonObject data = JsonParser.parseString(line).getAsJsonObject();
        double[][] Q = toMatrix(data.getAsJsonArray("Q"));
        double[][] K = toMatrix(data.getAsJsonArray("K"));
        double[][] V = toMatrix(data.getAsJsonArray("V"));
        int kTop = data.get("k_top").getAsInt();
        double alpha = data.has("alpha") ? data.get("alpha").getAsDouble() : 0.2;
        double beta = data.has("beta") ? data.get("beta").getAsDouble() : 0.6;
        double[][] A = tgSa(Q, K, V, kTop, alpha, beta);
        int L = Q.length;
        int d = V[0].length;
        double[][] O = new double[L][d];
        for (int i = 0; i < L; i++) {
            for (int j = 0; j < d; j++) {
                double s = 0;
                for (int t = 0; t < L; t++) s += A[i][t] * V[t][j];
                O[i][j] = s;
            }
        }
        JsonArray Aout = new JsonArray();
        for (int i = 0; i < L; i++) {
            JsonArray row = new JsonArray();
            for (int j = 0; j < L; j++) row.add(Math.round(A[i][j] * 1e6) / 1e6);
            Aout.add(row);
        }
        JsonArray Oout = new JsonArray();
        for (int i = 0; i < L; i++) {
            JsonArray row = new JsonArray();
            for (int j = 0; j < d; j++) row.add(Math.round(O[i][j] * 1e6) / 1e6);
            Oout.add(row);
        }
        JsonObject ans = new JsonObject();
        ans.add("A", Aout);
        ans.add("O", Oout);
        System.out.println(ans.toString());
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
#include <numeric>
#include <string>
#include "json.hpp"

using json = nlohmann::json;
using namespace std;

int main() {
    string line;
    getline(cin, line);
    json data = json::parse(line);
    auto Qj = data["Q"], Kj = data["K"], Vj = data["V"];
    int L = (int)Qj.size();
    int d = (int)Qj[0].size();
    int k_top = data["k_top"];
    double alpha = data.value("alpha", 0.2);
    double beta = data.value("beta", 0.6);
    vector<vector<double>> Q(L, vector<double>(d)), K(L, vector<double>(d)), V(L, vector<double>(d));
    for (int i = 0; i < L; i++)
        for (int j = 0; j < d; j++) {
            Q[i][j] = Qj[i][j];
            K[i][j] = Kj[i][j];
            V[i][j] = Vj[i][j];
        }
    vector<vector<double>> S(L, vector<double>(L, 0.0));
    double inv = 1.0 / sqrt((double)d);
    for (int i = 0; i < L; i++)
        for (int j = 0; j < L; j++) {
            double s = 0;
            for (int t = 0; t < d; t++) s += Q[i][t] * K[j][t];
            S[i][j] = s * inv;
        }
    vector<vector<double>> T(L, vector<double>(L, 0.0));
    for (int i = 0; i < L; i++) {
        double smax = *max_element(S[i].begin(), S[i].end());
        double t1 = alpha * smax, t2 = beta * smax;
        for (int j = 0; j < L; j++) {
            double x = S[i][j];
            if (fabs(t2 - t1) < 1e-12) T[i][j] = (x < t1) ? 0.0 : 1.0;
            else if (x < t1) T[i][j] = 0.0;
            else if (x < t2) T[i][j] = (x - t1) / (t2 - t1);
            else T[i][j] = 1.0;
        }
    }
    if (k_top < L) {
        for (int i = 0; i < L; i++) {
            vector<int> idx(L);
            iota(idx.begin(), idx.end(), 0);
            sort(idx.begin(), idx.end(), [&](int a, int b) { return T[i][a] > T[i][b]; });
            vector<double> neu(L, 0.0);
            for (int t = 0; t < k_top; t++) neu[idx[t]] = T[i][idx[t]];
            T[i] = neu;
        }
    }
    vector<vector<double>> A(L, vector<double>(L, 0.0));
    for (int i = 0; i < L; i++) {
        double sum = accumulate(T[i].begin(), T[i].end(), 0.0);
        if (fabs(sum) < 1e-12) {
            for (int j = 0; j < L; j++) A[i][j] = 1.0 / L;
        } else {
            for (int j = 0; j < L; j++) A[i][j] = T[i][j] / sum;
        }
    }
    json Aout = json::array(), Oout = json::array();
    for (int i = 0; i < L; i++) {
        json row = json::array();
        for (int j = 0; j < L; j++) row.push_back(round(A[i][j] * 1e6) / 1e6);
        Aout.push_back(row);
    }
    for (int i = 0; i < L; i++) {
        json row = json::array();
        for (int j = 0; j < d; j++) {
            double s = 0;
            for (int t = 0; t < L; t++) s += A[i][t] * V[t][j];
            row.push_back(round(s * 1e6) / 1e6);
        }
        Oout.push_back(row);
    }
    json ans;
    ans["A"] = Aout;
    ans["O"] = Oout;
    cout << ans.dump() << "\n";
    return 0;
}
```
