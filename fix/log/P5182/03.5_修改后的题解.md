## 解题思路

本题与「工位相位巡检」描述的计算任务一致。按输入格式读入数据后，沿用原题解的算法即可。

使用埃拉托斯特尼筛预处理每个编号是否为质数：

* 质数格子初始为黑色，对应字符 $1$。
* 非质数格子初始为白色，对应字符 $0$。

维护变量 $cur$，表示后续的人执行完当前已经处理的指令后所在的位置。初始时所有人都在格子 $1$，所以 $cur=1$。

处理指令 $s_i$ 时：

1. 从 $cur$ 的右侧寻找第一个初始颜色与 $s_i$ 相同的格子 $p$。
2. 第 $i$ 个人会到达 $p$，并将格子 $p$ 翻转，因此 $p$ 是一个最终颜色发生变化的格子。
3. 对于后续的人，格子 $p$ 已经被翻转，所以他们执行相同指令时会到达第二个初始颜色与 $s_i$ 相同的格子 $q$。
4. 将 $cur$ 更新为 $q$，继续处理下一条指令。

由于每次找到的位置都在 $cur$ 的右侧，因此所有已经翻转的格子都位于当前搜索位置左侧，不会再次被访问。

每个被翻转的格子编号严格递增，所以：

* 不会有格子被重复翻转。
* 最终恰好有 $n$ 个格子的颜色发生变化。
* 得到的答案本身已经按编号从小到大排列。

使用埃拉托斯特尼筛预处理质数。设需要寻找的质数数量为 $k=2N+2$，利用第 $k$ 个质数的上界

$$
p_k<k(\ln k+\ln\ln k)
$$

确定筛法范围。

## 复杂度分析

设所有测试用例中的最大字符串长度为 $N$，筛法上界为 $M$，则

$$
M=O(N\log N)
$$

埃拉托斯特尼筛的时间复杂度为

$$
O(M\log\log M)
$$

对于一个长度为 $n$ 的测试用例，搜索指针只会向右移动，时间复杂度为

$$
O(n\log n)
$$

总空间复杂度为

$$
O(M+n)
$$

其中 $O(M)$ 用于保存质数表，$O(n)$ 用于保存答案。

## 代码实现

### Python

```python
import sys
from math import log


def get_limit(max_n):
    """计算埃拉托斯特尼筛所需的安全上界"""
    k = 2 * max_n + 2

    if k < 6:
        return 20

    return int(k * (log(k) + log(log(k)))) + 20


def build_sieve(limit):
    """使用埃拉托斯特尼筛预处理质数"""
    is_prime = bytearray(b'\x01') * (limit + 1)
    is_prime[0] = 0
    is_prime[1] = 0

    i = 2

    # 不使用 math.isqrt，兼容较低版本的 Python
    while i * i <= limit:
        if is_prime[i]:
            start = i * i
            count = (limit - start) // i + 1

            # 将 i 的所有倍数标记为非质数
            is_prime[start:limit + 1:i] = b'\x00' * count

        i += 1

    return is_prime


def solve_case(s, is_prime):
    """计算当前测试用例中所有被翻转的格子"""
    cur = 1
    answer = []

    for ch in s:
        # 字符为 1 时寻找质数格子，否则寻找非质数格子
        need = 1 if ch == '1' else 0

        # 第一个目标颜色格子是当前被翻转的位置
        pos = cur + 1
        while is_prime[pos] != need:
            pos += 1

        answer.append(pos)

        # 第二个目标颜色格子是后续人的到达位置
        pos += 1
        while is_prime[pos] != need:
            pos += 1

        cur = pos

    return answer


def main():
    data = sys.stdin.buffer.read().split()
    t = int(data[0])
    index = 1

    cases = []
    max_n = 0

    for _ in range(t):
        n = int(data[index])
        s = data[index + 1].decode()
        index += 2

        cases.append(s)
        max_n = max(max_n, n)

    limit = get_limit(max_n)
    is_prime = build_sieve(limit)

    output = []

    for s in cases:
        answer = solve_case(s, is_prime)

        output.append(str(len(answer)))
        output.append(' '.join(map(str, answer)))

    sys.stdout.write('\n'.join(output))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    static int getLimit(int maxN) {
        // 计算埃拉托斯特尼筛所需的安全上界
        long k = 2L * maxN + 2;

        if (k < 6) {
            return 20;
        }

        double limit = k * (Math.log(k) + Math.log(Math.log(k)));
        return (int) limit + 20;
    }

    static boolean[] buildSieve(int limit) {
        // 使用埃拉托斯特尼筛预处理质数
        boolean[] isPrime = new boolean[limit + 1];
        Arrays.fill(isPrime, true);

        isPrime[0] = false;
        isPrime[1] = false;

        for (int i = 2; (long) i * i <= limit; i++) {
            if (isPrime[i]) {
                // 将 i 的所有倍数标记为非质数
                for (int j = i * i; j <= limit; j += i) {
                    isPrime[j] = false;
                }
            }
        }

        return isPrime;
    }

    static int[] solveCase(String s, boolean[] isPrime) {
        // 计算当前测试用例中所有被翻转的格子
        int[] answer = new int[s.length()];
        int cur = 1;

        for (int i = 0; i < s.length(); i++) {
            // 字符为 1 时寻找质数格子，否则寻找非质数格子
            boolean need = s.charAt(i) == '1';

            // 第一个目标颜色格子是当前被翻转的位置
            int pos = cur + 1;
            while (isPrime[pos] != need) {
                pos++;
            }

            answer[i] = pos;

            // 第二个目标颜色格子是后续人的到达位置
            pos++;
            while (isPrime[pos] != need) {
                pos++;
            }

            cur = pos;
        }

        return answer;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader reader =
                new BufferedReader(new InputStreamReader(System.in));

        int t = Integer.parseInt(reader.readLine().trim());

        String[] cases = new String[t];
        int maxN = 0;

        for (int i = 0; i < t; i++) {
            int n = Integer.parseInt(reader.readLine().trim());
            String s = reader.readLine().trim();

            cases[i] = s;
            maxN = Math.max(maxN, n);
        }

        int limit = getLimit(maxN);
        boolean[] isPrime = buildSieve(limit);

        StringBuilder output = new StringBuilder();

        for (String s : cases) {
            int[] answer = solveCase(s, isPrime);

            output.append(answer.length).append('\n');

            for (int i = 0; i < answer.length; i++) {
                if (i > 0) {
                    output.append(' ');
                }

                output.append(answer[i]);
            }

            output.append('\n');
        }

        System.out.print(output);
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <cmath>
#include <algorithm>
using namespace std;

int getLimit(int maxN) {
    // 计算埃拉托斯特尼筛所需的安全上界
    long long k = 2LL * maxN + 2;

    if (k < 6) {
        return 20;
    }

    double limit = k * (log((double)k) + log(log((double)k)));
    return (int)limit + 20;
}

vector<char> buildSieve(int limit) {
    // 使用埃拉托斯特尼筛预处理质数
    vector<char> isPrime(limit + 1, true);

    isPrime[0] = false;
    isPrime[1] = false;

    for (int i = 2; 1LL * i * i <= limit; i++) {
        if (isPrime[i]) {
            // 将 i 的所有倍数标记为非质数
            for (int j = i * i; j <= limit; j += i) {
                isPrime[j] = false;
            }
        }
    }

    return isPrime;
}

vector<int> solveCase(const string& s, const vector<char>& isPrime) {
    // 计算当前测试用例中所有被翻转的格子
    vector<int> answer;
    answer.reserve(s.size());

    int cur = 1;

    for (char ch : s) {
        // 字符为 1 时寻找质数格子，否则寻找非质数格子
        bool need = (ch == '1');

        // 第一个目标颜色格子是当前被翻转的位置
        int pos = cur + 1;
        while ((bool)isPrime[pos] != need) {
            pos++;
        }

        answer.push_back(pos);

        // 第二个目标颜色格子是后续人的到达位置
        pos++;
        while ((bool)isPrime[pos] != need) {
            pos++;
        }

        cur = pos;
    }

    return answer;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    vector<string> cases(t);
    int maxN = 0;

    for (int i = 0; i < t; i++) {
        int n;
        cin >> n >> cases[i];

        maxN = max(maxN, n);
    }

    int limit = getLimit(maxN);
    vector<char> isPrime = buildSieve(limit);

    for (const string& s : cases) {
        vector<int> answer = solveCase(s, isPrime);

        cout << answer.size() << '\n';

        for (int i = 0; i < (int)answer.size(); i++) {
            if (i > 0) {
                cout << ' ';
            }

            cout << answer[i];
        }

        cout << '\n';
    }

    return 0;
}
```
