## 解题思路

本题要求统计连续子段的乘积 $M$ 满足正因子个数 $d(M) \ge k$ 的个数。随着子段长度增加，乘积的质因子指数只增不减，由约数个数定理 $d(M) = \prod (e_i+1)$ 可知因子个数具有单调性。因此可以使用双指针（滑动窗口）高效计算。

具体算法步骤：

1. **预处理每个数的质因数分解**  
   - 限定数值范围 $MAX = 200005$。  
   - 类似埃氏筛的思想，从小到达枚举每个质数 $p$，利用已分解的结果递推其倍数 $x$ 的分解。  
   - 对于每个正整数 $v$，保存其质因数及其指数的映射（或列表），后续可以快速获取。

2. **双指针维护当前窗口的约数个数**  
   - 使用指针 $i$ 和 $j$，初始均指向序列起点 0。  
   - 维护一个全局的质因数指数计数器 `now_f`（字典/映射）以及当前窗口乘积的因子个数 `now_val`（初值为 $1$）。  
   - 逐个移动右指针 $j$ 扩大窗口：
     * 将 $x_j$ 的所有质因数及其指数加入到 `now_f` 中，同时按照约数个数定理更新 `now_val`：每次对于一个质因子 $p$，新增加指数 $e$，先除以原本的贡献 $(cnt_p+1)$，更新 $cnt_p = cnt_p + e$，再乘以新的贡献 $(cnt_p+1)$。
     * 当 `now_val >= k` 时，说明以当前 $j$ 为右端点，窗口的左端点可以向右收缩。不断移动左指针 $i$：
       - 将 $x_i$ 的质因数对应的指数从 `now_f` 中减去，同时同步更新 `now_val`（除法/乘法规则同上）。
       - $i$ 右移一位。
     * 此时对于右端点 $j$，所有满足条件 $d(M) \ge k$ 的子段其左端点 $l$ 的范围为 $[0, i-1]$（因为经过收缩后，$[i, j]$ 已经不满足条件，而 $[0, j], [1, j], \dots, [i-1, j]$ 均满足），所以累加答案 $ans = ans + i$。
   - 右指针继续后移，直至扫描完整个序列。

3. **输出答案**  
   累加的答案即为符合条件的连续子段总数。

## 复杂度分析

- 时间复杂度：  
  预处理每个数的质因数分解的复杂度约为 $O(MAX \log\log MAX)$（实际与调和级数相关）。双指针过程中，每个数被加入和移出窗口各一次，每次操作涉及该数个数的质因子（最多 $O(\log x_i)$ 个），因此这部分为 $O(n \log MAX)$。总时间复杂度 $O((n+MAX) \log MAX)$，在 $n, MAX \le 2\times10^5$ 下可以高效运行。

- 空间复杂度：  
  需要存储每个数的质因数分解，空间 $O(MAX \log MAX)$；维护窗口的质因子计数以及输入序列等需要 $O(n + MAX)$ 空间。总空间复杂度 $O(MAX \log MAX + n)$。

## 代码实现

### Python

```python
from collections import defaultdict

def main():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    MAXA = 200005
    
    # f[v] 存储正整数 v 的质因数分解：质数 -> 指数
    f = [defaultdict(int) for _ in range(MAXA)]
    is_prime = [True] * MAXA
    
    # 类似埃氏筛，递推每个数的质因数分解
    for i in range(2, MAXA):
        if not is_prime[i]:
            continue
        f[i][i] = 1
        for j in range(i + i, MAXA, i):
            # 通过 j // i 的分解得到 j 的指数
            if i in f[j // i]:
                f[j][i] = f[j // i][i] + 1
            else:
                f[j][i] = 1
            is_prime[j] = False
    
    now_f = defaultdict(int)   # 当前窗口质因子指数和
    now_val = 1                # 当前窗口乘积的因子个数
    ans = 0
    i = 0
    
    for j in range(n):
        x = a[j]
        # 加入右端点
        for p, e in f[x].items():
            now_val //= (now_f[p] + 1)
            now_f[p] += e
            now_val *= (now_f[p] + 1)
        
        # 收缩左端点，使窗口不再满足 >= k
        while now_val >= k and i <= j:
            x = a[i]
            for p, e in f[x].items():
                now_val //= (now_f[p] + 1)
                now_f[p] -= e
                now_val *= (now_f[p] + 1)
            i += 1
        
        # 此时左端点从 0 到 i-1 均符合条件
        ans += i
    
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static final int MAXA = 200005;
    static List<Map<Integer, Integer>> f; // 每个数的质因数分解
    static long now_val;
    static Map<Integer, Integer> now_f;
    static long ans;
    static int n, k;
    static int[] a;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = br.readLine().split(" ");
        n = Integer.parseInt(parts[0]);
        k = Integer.parseInt(parts[1]);
        a = new int[n];
        parts = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(parts[i]);
        }

        // 预处理每个数的质因数分解
        f = new ArrayList<>(MAXA);
        for (int i = 0; i < MAXA; i++) {
            f.add(new HashMap<>());
        }
        boolean[] isPrime = new boolean[MAXA];
        Arrays.fill(isPrime, true);
        for (int i = 2; i < MAXA; i++) {
            if (!isPrime[i]) continue;
            f.get(i).put(i, 1);
            for (int j = i + i; j < MAXA; j += i) {
                isPrime[j] = false;
                // 利用 j/i 的分解
                Map<Integer, Integer> prev = f.get(j / i);
                int e = prev.getOrDefault(i, 0) + 1;
                f.get(j).put(i, e);
            }
        }

        now_f = new HashMap<>();
        now_val = 1;
        ans = 0;
        int i = 0;

        for (int j = 0; j < n; j++) {
            // 加入右端点
            for (Map.Entry<Integer, Integer> entry : f.get(a[j]).entrySet()) {
                int p = entry.getKey();
                int e = entry.getValue();
                int cnt = now_f.getOrDefault(p, 0);
                now_val /= (cnt + 1);
                now_f.put(p, cnt + e);
                now_val *= (cnt + e + 1);
            }

            // 收缩左端点
            while (now_val >= k && i <= j) {
                for (Map.Entry<Integer, Integer> entry : f.get(a[i]).entrySet()) {
                    int p = entry.getKey();
                    int e = entry.getValue();
                    int cnt = now_f.get(p);
                    now_val /= (cnt + 1);
                    int newCnt = cnt - e;
                    if (newCnt == 0) {
                        now_f.remove(p);
                    } else {
                        now_f.put(p, newCnt);
                    }
                    now_val *= (newCnt + 1);
                }
                i++;
            }

            ans += i;
        }

        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXA = 200005;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    
    // f[v] 存储 v 的质因数分解 (质数, 指数)
    vector<unordered_map<int, int>> f(MAXA);
    vector<bool> is_prime(MAXA, true);
    
    // 类似埃氏筛预处理每个数的分解
    for (int i = 2; i < MAXA; i++) {
        if (!is_prime[i]) continue;
        f[i][i] = 1;
        for (int j = i + i; j < MAXA; j += i) {
            is_prime[j] = false;
            // 递推：检查 j/i 是否含有质数 i
            if (f[j / i].count(i)) {
                f[j][i] = f[j / i][i] + 1;
            } else {
                f[j][i] = 1;
            }
        }
    }
    
    unordered_map<int, int> now_f;
    long long now_val = 1;
    long long ans = 0;
    int i = 0;
    
    for (int j = 0; j < n; j++) {
        // 加入右端点
        for (auto &entry : f[a[j]]) {
            int p = entry.first;
            int e = entry.second;
            int &cnt = now_f[p];
            now_val /= (cnt + 1);
            cnt += e;
            now_val *= (cnt + 1);
        }
        
        // 收缩左端点
        while (now_val >= k && i <= j) {
            for (auto &entry : f[a[i]]) {
                int p = entry.first;
                int e = entry.second;
                int &cnt = now_f[p];
                now_val /= (cnt + 1);
                cnt -= e;
                if (cnt == 0) {
                    now_f.erase(p);
                }
                now_val *= (cnt + 1);
            }
            i++;
        }
        
        ans += i;
    }
    
    cout << ans << "\n";
    return 0;
}
```
