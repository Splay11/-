## 解题思路

风控任务用 $K$ 近邻从历史任务里迁移正则系数 $C$，再训练逻辑回归。

1. **元特征**：当前任务 $\mathbf{m}_{ctr} = \bigl[n,\ d,\ |n_{\mathrm{pos}}-n_{\mathrm{neg}}|/n\bigr]$。
2. **检索**：计算到每条 history.meta 的 $\ell_2$ 距离，按 $(\text{距离}, \text{行号})$ 升序取前 $K=3$ 条。
3. **选 $C^*$**：对这 `3` 个邻居中出现过的每个 $C$ 求 score 均值（未出现则忽略），取均值最大者；并列取更小的 $C$。
4. **拟合**：$\mathrm{LogisticRegression}(\mathrm{penalty}="l2", C=C^*, \mathrm{solver}="lbfgs", \mathrm{max\_iter}=1000, \mathrm{random\_state}=42)$ 在完整训练集上训练，输出测试标签 `0` / `1`。

JSON 字段名 train_X、train_y、test_X、history、meta、C、score、C_star、pred 必须与输入输出保持一致。

## 复杂度分析

设历史条数为 $H$，训练样本 $n$，维数 $d$，测试样本 $q$。时间复杂度约 $O(H \log H + T \cdot n \cdot d^2)$，空间 $O(H + n d)$。在 $n \le 60$、$d \le 4$ 下开销很小。

```python
import sys, json
import numpy as np
from sklearn.linear_model import LogisticRegression

def solve(data):
    train_X_raw = data["train_X"]
    test_X_raw  = data["test_X"]
    d = min(min(len(r) for r in train_X_raw),
            min(len(r) for r in test_X_raw))
    X_tr = np.array([r[:d] for r in train_X_raw], dtype=float)
    X_te = np.array([r[:d] for r in test_X_raw],  dtype=float)
    y_tr = np.array(data["train_y"], dtype=int)
    history = data["history"]

    n = len(y_tr)
    n_pos = int((y_tr == 1).sum())
    imb = abs(2 * n_pos - n) / n
    m_ctr = np.array([n, d, imb], dtype=float)

    items = []
    for i, h in enumerate(history):
        dist = np.linalg.norm(m_ctr - np.array(h["meta"], dtype=float))
        items.append((dist, i, h["C"], h["score"]))
    items.sort(key=lambda x: (x[0], x[1]))
    top3 = items[:3]

    c_scores = {}
    for _, _, c, s in top3:
        c_scores.setdefault(c, []).append(s)
    best_c, best_avg = None, -1.0
    for c in sorted(c_scores.keys()):
        avg = sum(c_scores[c]) / len(c_scores[c])
        if avg > best_avg:
            best_avg, best_c = avg, c

    model = LogisticRegression(penalty="l2", C=best_c, solver="lbfgs",
                               max_iter=1000, random_state=42)
    model.fit(X_tr, y_tr)
    pred = model.predict(X_te).astype(int).tolist()
    return {"C_star": best_c, "pred": pred}

if __name__ == "__main__":
    data = json.loads(sys.stdin.read())
    print(json.dumps(solve(data)))
```

import java.io.*;
import java.util.*;

public class Main {
    static String src;
    static int pos;

    static void skip() {
        while (pos < src.length() && Character.isWhitespace(src.charAt(pos))) pos++;
    }

    static char peek() { skip(); return src.charAt(pos); }

    static char eat() { skip(); return src.charAt(pos++); }

    static String parseString() {
        skip();
        if (eat() != '"') throw new RuntimeException();
        StringBuilder sb = new StringBuilder();
        while (src.charAt(pos) != '"') {
            char c = src.charAt(pos++);
            if (c == '\\') sb.append(src.charAt(pos++));
            else sb.append(c);
        }
        pos++;
        return sb.toString();
    }

    static double parseNumber() {
        skip();
        int st = pos;
        if (src.charAt(pos) == '-') pos++;
        while (pos < src.length() && (Character.isDigit(src.charAt(pos)) || src.charAt(pos) == '.' || src.charAt(pos) == 'e' || src.charAt(pos) == 'E' || src.charAt(pos) == '+')) pos++;
        return Double.parseDouble(src.substring(st, pos));
    }

    static List<Double> parseNumArray() {
        skip();
        if (eat() != '[') throw new RuntimeException();
        List<Double> a = new ArrayList<>();
        skip();
        if (peek() == ']') { pos++; return a; }
        while (true) {
            a.add(parseNumber());
            skip();
            char c = eat();
            if (c == ']') break;
            if (c != ',') throw new RuntimeException();
        }
        return a;
    }

    static List<List<Double>> parseMat() {
        skip();
        if (eat() != '[') throw new RuntimeException();
        List<List<Double>> a = new ArrayList<>();
        skip();
        if (peek() == ']') { pos++; return a; }
        while (true) {
            a.add(parseNumArray());
            skip();
            char c = eat();
            if (c == ']') break;
            if (c != ',') throw new RuntimeException();
        }
        return a;
    }

    static class Hist {
        double[] meta;
        double C, score;
    }

    static Object parseValue();

    static Map<String, Object> parseObject() {
        skip();
        if (eat() != '{') throw new RuntimeException();
        Map<String, Object> m = new LinkedHashMap<>();
        skip();
        if (peek() == '}') { pos++; return m; }
        while (true) {
            String k = parseString();
            skip();
            if (eat() != ':') throw new RuntimeException();
            m.put(k, parseValue());
            skip();
            char c = eat();
            if (c == '}') break;
            if (c != ',') throw new RuntimeException();
        }
        return m;
    }

    static List<Object> parseArrayAny() {
        skip();
        if (eat() != '[') throw new RuntimeException();
        List<Object> a = new ArrayList<>();
        skip();
        if (peek() == ']') { pos++; return a; }
        while (true) {
            a.add(parseValue());
            skip();
            char c = eat();
            if (c == ']') break;
            if (c != ',') throw new RuntimeException();
        }
        return a;
    }

    static Object parseValue() {
        skip();
        char c = peek();
        if (c == '{') return parseObject();
        if (c == '[') return parseArrayAny();
        if (c == '"') return parseString();
        if (c == '-' || Character.isDigit(c)) return parseNumber();
        throw new RuntimeException("bad json");
    }

    @SuppressWarnings("unchecked")
    static double[][] toMat(Object o) {
        List<Object> rows = (List<Object>) o;
        int n = rows.size();
        int d = n == 0 ? 0 : ((List<Object>) rows.get(0)).size();
        for (Object r : rows) d = Math.min(d, ((List<Object>) r).size());
        double[][] X = new double[n][d];
        for (int i = 0; i < n; i++) {
            List<Object> row = (List<Object>) rows.get(i);
            for (int j = 0; j < d; j++) X[i][j] = ((Number) row.get(j)).doubleValue();
        }
        return X;
    }

    @SuppressWarnings("unchecked")
    static int[] toIntArr(Object o) {
        List<Object> a = (List<Object>) o;
        int[] y = new int[a.size()];
        for (int i = 0; i < a.size(); i++) y[i] = (int) Math.round(((Number) a.get(i)).doubleValue());
        return y;
    }

    static int predictLR(double[][] X, int[] y, double[] x, double C) {
        int n = X.length, d = X[0].length;
        double[] w = new double[d];
        double b = 0;
        double lam = 1.0 / C;
        int T = 2000;
        double eta = 0.1;
        for (int t = 0; t < T; t++) {
            double[] gw = new double[d];
            double gb = 0;
            for (int i = 0; i < n; i++) {
                double z = b;
                for (int j = 0; j < d; j++) z += w[j] * X[i][j];
                double p = 1.0 / (1.0 + Math.exp(-Math.max(-30, Math.min(30, z))));
                double g = p - y[i];
                for (int j = 0; j < d; j++) gw[j] += g * X[i][j];
                gb += g;
            }
            for (int j = 0; j < d; j++) w[j] -= eta * (gw[j] / n + lam * w[j]);
            b -= eta * (gb / n);
        }
        double z = b;
        for (int j = 0; j < d; j++) z += w[j] * x[j];
        return z >= 0 ? 1 : 0;
    }

    @SuppressWarnings("unchecked")
    public static void main(String[] args) throws Exception {
        src = new String(System.in.readAllBytes());
        pos = 0;
        Map<String, Object> data = parseObject();
        double[][] Xtr = toMat(data.get("train_X"));
        double[][] Xte = toMat(data.get("test_X"));
        int d = Math.min(Xtr[0].length, Xte[0].length);
        int[] y = toIntArr(data.get("train_y"));
        int n = y.length;
        int npos = 0;
        for (int v : y) if (v == 1) npos++;
        double imb = Math.abs(2.0 * npos - n) / n;
        double[] mctr = {n, d, imb};

        List<Object> history = (List<Object>) data.get("history");
        class Item implements Comparable<Item> {
            double dist; int idx; double C, score;
            public int compareTo(Item o) {
                if (dist != o.dist) return Double.compare(dist, o.dist);
                return Integer.compare(idx, o.idx);
            }
        }
        List<Item> items = new ArrayList<>();
        for (int i = 0; i < history.size(); i++) {
            Map<String, Object> h = (Map<String, Object>) history.get(i);
            List<Object> meta = (List<Object>) h.get("meta");
            double dist = 0;
            for (int j = 0; j < 3; j++) {
                double diff = mctr[j] - ((Number) meta.get(j)).doubleValue();
                dist += diff * diff;
            }
            dist = Math.sqrt(dist);
            Item it = new Item();
            it.dist = dist; it.idx = i;
            it.C = ((Number) h.get("C")).doubleValue();
            it.score = ((Number) h.get("score")).doubleValue();
            items.add(it);
        }
        Collections.sort(items);
        Map<Double, List<Double>> cScores = new TreeMap<>();
        for (int i = 0; i < Math.min(3, items.size()); i++) {
            Item it = items.get(i);
            cScores.computeIfAbsent(it.C, k -> new ArrayList<>()).add(it.score);
        }
        double bestC = 0, bestAvg = -1;
        for (Map.Entry<Double, List<Double>> e : cScores.entrySet()) {
            double avg = 0;
            for (double s : e.getValue()) avg += s;
            avg /= e.getValue().size();
            if (avg > bestAvg) { bestAvg = avg; bestC = e.getKey(); }
        }

        StringBuilder pred = new StringBuilder();
        pred.append('[');
        for (int i = 0; i < Xte.length; i++) {
            if (i > 0) pred.append(", ");
            double[] x = Arrays.copyOf(Xte[i], d);
            double[][] Xd = new double[n][d];
            for (int r = 0; r < n; r++) Xd[r] = Arrays.copyOf(Xtr[r], d);
            pred.append(predictLR(Xd, y, x, bestC));
        }
        pred.append(']');
        System.out.println("{\"C_star\": " + bestC + ", \"pred\": " + pred + "}");
    }
}

#include <bits/stdc++.h>
using namespace std;

string src; int posi;

void skip() { while (posi < (int)src.size() && isspace((unsigned char)src[posi])) posi++; }
char peekc() { skip(); return src[posi]; }
char eat() { skip(); return src[posi++]; }

string parse_string() {
    skip(); eat();
    string s;
    while (src[posi] != '"') {
        if (src[posi] == '\\') { posi++; s.push_back(src[posi++]); }
        else s.push_back(src[posi++]);
    }
    posi++;
    return s;
}

double parse_number() {
    skip();
    int st = posi;
    if (src[posi] == '-') posi++;
    while (posi < (int)src.size() && (isdigit((unsigned char)src[posi]) || src[posi]=='.' || src[posi]=='e' || src[posi]=='E' || src[posi]=='+')) posi++;
    return stod(src.substr(st, posi - st));
}

struct Val;
using Obj = map<string, Val>;
struct Val {
    int typ = 0; // 0 num 1 str 2 arr 3 obj
    double num = 0;
    string str;
    vector<Val> arr;
    Obj obj;
};

Val parse_val();

Val parse_obj() {
    eat();
    Val v; v.typ = 3;
    skip();
    if (peekc() == '}') { eat(); return v; }
    while (true) {
        string k = parse_string();
        skip(); eat();
        v.obj[k] = parse_val();
        skip();
        char c = eat();
        if (c == '}') break;
    }
    return v;
}

Val parse_arr() {
    eat();
    Val v; v.typ = 2;
    skip();
    if (peekc() == ']') { eat(); return v; }
    while (true) {
        v.arr.push_back(parse_val());
        skip();
        char c = eat();
        if (c == ']') break;
    }
    return v;
}

Val parse_val() {
    skip();
    char c = peekc();
    if (c == '{') return parse_obj();
    if (c == '[') return parse_arr();
    if (c == '"') { Val v; v.typ = 1; v.str = parse_string(); return v; }
    Val v; v.typ = 0; v.num = parse_number(); return v;
}

int predict_lr(const vector<vector<double>>& X, const vector<int>& y, const vector<double>& x, double C) {
    int n = (int)X.size(), d = (int)X[0].size();
    vector<double> w(d, 0);
    double b = 0, lam = 1.0 / C, eta = 0.1;
    int T = 2000;
    for (int t = 0; t < T; t++) {
        vector<double> gw(d, 0); double gb = 0;
        for (int i = 0; i < n; i++) {
            double z = b;
            for (int j = 0; j < d; j++) z += w[j] * X[i][j];
            z = max(-30.0, min(30.0, z));
            double p = 1.0 / (1.0 + exp(-z));
            double g = p - y[i];
            for (int j = 0; j < d; j++) gw[j] += g * X[i][j];
            gb += g;
        }
        for (int j = 0; j < d; j++) w[j] -= eta * (gw[j] / n + lam * w[j]);
        b -= eta * (gb / n);
    }
    double z = b;
    for (int j = 0; j < d; j++) z += w[j] * x[j];
    return z >= 0 ? 1 : 0;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    src.assign((istreambuf_iterator<char>(cin)), istreambuf_iterator<char>());
    posi = 0;
    Val data = parse_obj();
    auto& tr = data.obj["train_X"].arr;
    auto& te = data.obj["test_X"].arr;
    int d = min((int)tr[0].arr.size(), (int)te[0].arr.size());
    int n = (int)tr.size();
    vector<vector<double>> Xtr(n, vector<double>(d));
    for (int i = 0; i < n; i++) for (int j = 0; j < d; j++) Xtr[i][j] = tr[i].arr[j].num;
    vector<int> y(n);
    int npos = 0;
    for (int i = 0; i < n; i++) {
        y[i] = (int)llround(data.obj["train_y"].arr[i].num);
        if (y[i] == 1) npos++;
    }
    double imb = fabs(2.0 * npos - n) / n;
    double mctr[3] = {(double)n, (double)d, imb};

    struct Item { double dist; int idx; double C, score; };
    vector<Item> items;
    auto& hist = data.obj["history"].arr;
    for (int i = 0; i < (int)hist.size(); i++) {
        auto& h = hist[i].obj;
        double dist = 0;
        for (int j = 0; j < 3; j++) {
            double diff = mctr[j] - h["meta"].arr[j].num;
            dist += diff * diff;
        }
        items.push_back({sqrt(dist), i, h["C"].num, h["score"].num});
    }
    sort(items.begin(), items.end(), [](const Item& a, const Item& b) {
        if (a.dist != b.dist) return a.dist < b.dist;
        return a.idx < b.idx;
    });
    map<double, vector<double>> csc;
    for (int i = 0; i < min(3, (int)items.size()); i++) csc[items[i].C].push_back(items[i].score);
    double bestC = 0, bestAvg = -1;
    for (auto& e : csc) {
        double avg = 0;
        for (double s : e.second) avg += s;
        avg /= e.second.size();
        if (avg > bestAvg) { bestAvg = avg; bestC = e.first; }
    }

    cout << "{\"C_star\": " << bestC << ", \"pred\": [";
    int q = (int)te.size();
    for (int i = 0; i < q; i++) {
        if (i) cout << ", ";
        vector<double> x(d);
        for (int j = 0; j < d; j++) x[j] = te[i].arr[j].num;
        cout << predict_lr(Xtr, y, x, bestC);
    }
    cout << "]}\n";
    return 0;
}
