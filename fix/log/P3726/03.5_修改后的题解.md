## 解题思路

货架上至多反转一个子段 $[l,r]$。区间外与区间内的跨段配对在反转后相对位置不变，逆序对数不变；**只会改变区间内部**的两两关系。

设区间长度 $k=r-l+1$，区间内逆序对数为 $\mathrm{invSeg}$，相等对数为 $\mathrm{eqSeg}$，总对数 $C=k(k-1)/2$。反转后内部逆序变成原来的“小于对”，变化量

$$
\Delta = C - \mathrm{eqSeg} - 2\cdot\mathrm{invSeg}.
$$

总答案为 $\mathrm{inv}_{total}+\Delta$。枚举所有 $[l,r]$，最小化该值（含不操作，即 $\Delta=0$）。

对固定左端点 $l$，向右扩展 $r$，用树状数组维护段内值频次以统计大于/等于当前值的个数，并做坐标压缩。$n\le 2000$，复杂度可接受。

## 复杂度分析

时间复杂度 $O(n^2\log n)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys

class BIT:
    def __init__(self, n):
        self.n = n
        self.bit = [0] * (n + 1)

    def add(self, i, delta):
        while i <= self.n:
            self.bit[i] += delta
            i += i & -i

    def sum(self, i):
        s = 0
        while i > 0:
            s += self.bit[i]
            i -= i & -i
        return s

def compress(arr):
    vals = sorted(set(arr))
    idx = {v: i + 1 for i, v in enumerate(vals)}
    return [idx[v] for v in arr], len(vals)

def initial_inversions(rks, m):
    bit = BIT(m)
    inv = 0
    seen = 0
    for rk in rks:
        greater_left = seen - bit.sum(rk)
        inv += greater_left
        bit.add(rk, 1)
        seen += 1
    return inv

def solve():
    data = sys.stdin.read().strip().split()
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))

    rks, m = compress(a)
    inv_total = initial_inversions(rks, m)
    ans = inv_total

    for l in range(n):
        bit = BIT(m)
        freq = [0] * (m + 1)
        invSeg = 0
        eqSeg = 0
        k = 0
        for r in range(l, n):
            rk = rks[r]
            cntGreater = k - bit.sum(rk)
            cntEqual = freq[rk]
            invSeg += cntGreater
            eqSeg += cntEqual
            bit.add(rk, 1)
            freq[rk] += 1
            k += 1
            C = k * (k - 1) // 2
            delta = C - eqSeg - 2 * invSeg
            cand = inv_total + delta
            if cand < ans:
                ans = cand

    print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static class BIT {
        int n;
        long[] bit;
        BIT(int n) {
            this.n = n;
            bit = new long[n + 1];
        }
        void add(int i, long delta) {
            while (i <= n) {
                bit[i] += delta;
                i += i & -i;
            }
        }
        long sum(int i) {
            long s = 0;
            while (i > 0) {
                s += bit[i];
                i -= i & -i;
            }
            return s;
        }
    }

    static int[] compress(int[] a) {
        int n = a.length;
        int[] b = a.clone();
        Arrays.sort(b);
        int m = 0;
        int[] uniq = new int[n];
        for (int i = 0; i < n; i++) {
            if (i == 0 || b[i] != b[i - 1]) uniq[m++] = b[i];
        }
        uniq = Arrays.copyOf(uniq, m);
        int[] rks = new int[n];
        for (int i = 0; i < n; i++) {
            rks[i] = Arrays.binarySearch(uniq, a[i]) + 1;
        }
        return rks;
    }

    static long initialInversions(int[] rks, int m) {
        BIT bit = new BIT(m);
        long inv = 0, seen = 0;
        for (int rk : rks) {
            long greaterLeft = seen - bit.sum(rk);
            inv += greaterLeft;
            bit.add(rk, 1);
            seen++;
        }
        return inv;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        String[] parts = br.readLine().trim().split("\\s+");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = Integer.parseInt(parts[i]);

        int[] rks = compress(a);
        int[] tmp = a.clone();
        Arrays.sort(tmp);
        int m = 0;
        for (int i = 0; i < n; i++) if (i == 0 || tmp[i] != tmp[i - 1]) m++;

        long invTotal = initialInversions(rks, m);
        long ans = invTotal;

        for (int l = 0; l < n; l++) {
            BIT bit = new BIT(m);
            long[] freq = new long[m + 1];
            long invSeg = 0, eqSeg = 0;
            long k = 0;
            for (int r = l; r < n; r++) {
                int rk = rks[r];
                long cntGreater = k - bit.sum(rk);
                long cntEqual = freq[rk];
                invSeg += cntGreater;
                eqSeg += cntEqual;
                bit.add(rk, 1);
                freq[rk] += 1;
                k++;
                long C = k * (k - 1) / 2;
                long delta = C - eqSeg - 2 * invSeg;
                long cand = invTotal + delta;
                if (cand < ans) ans = cand;
            }
        }
        System.out.println(ans);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

struct BIT {
    int n;
    vector<long long> t;
    BIT(int n=0): n(n), t(n+1, 0) {}
    void add(int i, long long delta){
        while(i<=n){ t[i]+=delta; i+=i&-i; }
    }
    long long sum(int i){
        long long s=0;
        while(i>0){ s+=t[i]; i-=i&-i; }
        return s;
    }
};

vector<int> compress(const vector<long long>& a, int &m){
    vector<long long> b=a;
    sort(b.begin(), b.end());
    b.erase(unique(b.begin(), b.end()), b.end());
    m = (int)b.size();
    vector<int> rks(a.size());
    for(size_t i=0;i<a.size();++i){
        int rk = int(lower_bound(b.begin(), b.end(), a[i]) - b.begin()) + 1;
        rks[i]=rk;
    }
    return rks;
}
```
