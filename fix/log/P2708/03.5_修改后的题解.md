## 解题思路

本题的核心在于分析划分段数对总效用的影响，并转化为贪心选取最佳切割点的问题。

设极性系数 $p_i$：若 $t_i = \text{‘1’}$ 则 $p_i = 1$，否则 $p_i = -1$。  
对于第 $j$ 段中的第 $i$ 颗宝石，贡献为 $p_i \times (v_i + j)$。

### 1. 基础效用计算
若不进行任何划分（即所有宝石属于同一段，段编号 $j=1$），总效用为：
$$
\text{base} = \sum_{i=1}^{n} p_i \times (v_i + 1)
$$
这是因为每颗宝石至少获得段编号 $1$。

### 2. 切割带来的额外贡献
如果在某个位置 $i$ 后切一刀（切割点位于第 $i$ 和第 $i+1$ 颗宝石之间），从第 $i+1$ 颗宝石开始的段编号都会增加 $1$。因此该切割点带来的额外效用为：
$$
\text{contrib}(i) = \sum_{j=i+1}^{n} p_j
$$
即从 $i+1$ 到 $n$ 的所有极性系数之和（后缀和）。

### 3. 贪心选取切割点
- 我们至多可以切出 $k$ 段，即最多可以插入 $k-1$ 个切割点。
- 切割点的额外贡献可正可负，我们只选择贡献 **为正** 的切割点。
- 对所有可能的切割点（位置 $1$ 到 $n-1$）计算其贡献，保存正贡献，然后从大到小排序。
- 选取前 $\min(\text{正贡献数量}, k-1)$ 个贡献值加到 $\text{base}$ 上，即可得到最大总效用。

### 4. 算法步骤
1. 读入 $n, k$，数组 $v$ 和字符串 $t$。
2. 根据 $t$ 生成极性数组 $p$。
3. 计算基础效用 $\text{base} = \sum_{i=0}^{n-1} p_i \times (v_i + 1)$。
4. 计算后缀和数组 $\text{suf}[i] = \sum_{j=i}^{n-1} p_j$（从后往前计算）。
5. 遍历 $i = 0$ 到 $n-2$，若 $\text{suf}[i+1] > 0$，将其加入候选列表。
6. 将候选列表降序排序，取前 $k-1$ 个求和得到额外效用 $\text{extra}$。
7. 输出 $\text{base} + \text{extra}$。

## 复杂度分析

- 时间复杂度：$O(n \log n)$，主要由对正贡献值的排序决定，其余步骤均为 $O(n)$。
- 空间复杂度：$O(n)$，用于存储极性数组、后缀和数组以及正贡献列表。

## 代码实现

### Python

```python
import sys

def main():
    input = sys.stdin.readline
    n, k = map(int, input().split())
    v = list(map(int, input().split()))
    t = input().strip()
    
    # 极性系数 p_i
    p = [1 if ch == '1' else -1 for ch in t]
    
    # 基础效用 base = sum(p_i * (v_i + 1))
    base = sum(p[i] * (v[i] + 1) for i in range(n))
    
    # 后缀和 suf[i] = sum_{j=i}^{n-1} p_j
    suf = [0] * n
    suf[-1] = p[-1]
    for i in range(n - 2, -1, -1):
        suf[i] = suf[i + 1] + p[i]
    
    # 收集正贡献切割点
    contributions = []
    for i in range(n - 1):
        if suf[i + 1] > 0:
            contributions.append(suf[i + 1])
    
    # 选取最多 k-1 个最大正贡献
    contributions.sort(reverse=True)
    extra = sum(contributions[:max(0, k - 1)])
    
    print(base + extra)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = br.readLine().trim().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);
        
        String[] vStr = br.readLine().trim().split(" ");
        long[] v = new long[n];
        for (int i = 0; i < n; i++) {
            v[i] = Long.parseLong(vStr[i]);
        }
        
        String t = br.readLine().trim();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = (t.charAt(i) == '1') ? 1 : -1;
        }
        
        // 基础效用 base = sum(p_i * (v_i + 1))
        long base = 0;
        for (int i = 0; i < n; i++) {
            base += (long) p[i] * (v[i] + 1);
        }
        
        // 后缀和
        long[] suf = new long[n];
        suf[n - 1] = p[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            suf[i] = suf[i + 1] + p[i];
        }
        
        // 收集正贡献切割点
        ArrayList<Long> contribs = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            if (suf[i + 1] > 0) {
                contribs.add(suf[i + 1]);
            }
        }
        
        // 降序排序并取前 k-1 个
        Collections.sort(contribs, Collections.reverseOrder());
        long extra = 0;
        int limit = Math.min(contribs.size(), k - 1);
        for (int i = 0; i < limit; i++) {
            extra += contribs.get(i);
        }
        
        System.out.println(base + extra);
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;
typedef long long ll;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n, k;
    cin >> n >> k;
    vector<ll> v(n);
    for (int i = 0; i < n; ++i) {
        cin >> v[i];
    }
    string t;
    cin >> t;
    
    // 极性系数
    vector<int> p(n);
    for (int i = 0; i < n; ++i) {
        p[i] = (t[i] == '1') ? 1 : -1;
    }
    
    // 基础效用 base = sum(p_i * (v_i + 1))
    ll base = 0;
    for (int i = 0; i < n; ++i) {
        base += (ll)p[i] * (v[i] + 1);
    }
    
    // 后缀和
    vector<ll> suf(n);
    suf[n - 1] = p[n - 1];
    for (int i = n - 2; i >= 0; --i) {
        suf[i] = suf[i + 1] + p[i];
    }
    
    // 收集正贡献切割点
    vector<ll> contribs;
    for (int i = 0; i < n - 1; ++i) {
        if (suf[i + 1] > 0) {
            contribs.push_back(suf[i + 1]);
        }
    }
    
    // 降序排序并取前 k-1 个
    sort(contribs.begin(), contribs.end(), greater<ll>());
    ll extra = 0;
    int limit = min((int)contribs.size(), k - 1);
    for (int i = 0; i < limit; ++i) {
        extra += contribs[i];
    }
    
    cout << base + extra << "\n";
    return 0;
}
```
