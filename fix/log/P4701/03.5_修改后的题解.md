## 解题思路

增益档位初值为 $a$，每次调整 $f(x)=\min(x,b)+\gcd(x,b)$。需要求出连续调整 $n$ 次后的结果 $f^{(n)}(a)$。

先看一次变换：

$$
f(x)=\min(x,b)+\gcd(x,b)
$$

分两种情况：

当 $x\ge b$ 时，

$$
f(x)=b+\gcd(x,b)
$$

设 $d=\gcd(x,b)$，因为 $d\mid b$，所以

$$
f(x)=b+d
$$

再算下一次：

$$
\gcd(b+d,b)=\gcd(d,b)=d
$$

于是

$$
f(b+d)=b+d
$$

也就是说，**只要到了 $x\ge b$，再做一次就会进入不动点**。

---

当 $x<b$ 时，

$$
f(x)=x+\gcd(x,b)
$$

设当前

$$
d=\gcd(x,b),\quad x=d\cdot y,\quad b=d\cdot m
$$

因为 $\gcd(y,m)=1$，所以如果接下来若干步 $\gcd$ 不变，那么每一步都只是加上 $d$：

$$
x,\ x+d,\ x+2d,\ \dots
$$

也就是

$$
d\cdot y,\ d\cdot (y+1),\ d\cdot (y+2),\dots
$$

什么时候 $\gcd$ 会变大？
就是最早出现某个正整数 $\Delta$，使得

$$
\gcd(y+\Delta,m)>1
$$

这等价于 $y+\Delta$ 被 $m$ 的某个质因子整除。

所以只要枚举 $m$ 的所有质因子 $p$，就能得到最早的变化步数：

$$
\Delta=\min (p-y\bmod p)
$$

因为当前 $\gcd(y,m)=1$，所以 $y\bmod p\ne 0$，上式一定是 $[1,p-1]$ 内的正数。

这样我们就可以：

1. 先分解 $b$ 的质因子；
2. 每次按“$\gcd$ 不变的一整段”进行跳跃；
3. 一旦 $x\ge b$，再特判不动点。

这就是本题的核心算法：**数论 + 质因子分解 + 分段跳跃模拟**。

## 复杂度分析

先用筛法预处理 $[2,10^6]$ 内的质数，因为

$$
\sqrt{10^{12}}=10^6
$$

对于每组数据：

1. 分解 $b$，时间复杂度为 $O(\pi(\sqrt b))$；
2. 跳跃模拟时，每次都会让当前的 $\gcd(x,b)$ 严格变大，因此循环次数很少，最多是对数级别，记为 $O(\log b)$；
3. 每次跳跃只会枚举 $b$ 的不同质因子个数，记为 $s$，而 $s$ 很小。

所以单组总复杂度可写为：

$$
O(\pi(\sqrt b)+s\log b)
$$

空间复杂度为：

$$
O(\pi(10^6))
$$

用于存质数表，单组额外空间为常数级。

## 代码实现

### Python

```python
import sys
import math


# 筛出 1e6 以内的所有质数
def get_primes(limit):
    is_prime = [True] * (limit + 1)
    is_prime[0] = is_prime[1] = False
    primes = []
    for i in range(2, limit + 1):
        if is_prime[i]:
            primes.append(i)
            if i * i <= limit:
                for j in range(i * i, limit + 1, i):
                    is_prime[j] = False
    return primes


PRIMES = get_primes(10 ** 6)


# 分解 b，返回 b 的不同质因子
def get_prime_factors(x):
    factors = []
    t = x
    for p in PRIMES:
        if p * p > t:
            break
        if t % p == 0:
            factors.append(p)
            while t % p == 0:
                t //= p
    if t > 1:
        factors.append(t)
    return factors


# 计算 f^(n)(a)
def solve_one(a, b, n):
    factors = get_prime_factors(b)
    x = a
    rem = n

    while rem > 0:
        # 情况 1：已经到达 x >= b
        if x >= b:
            d = math.gcd(x, b)
            x = b + d
            rem -= 1
            # 进入不动点后，后面无需再算
            if rem > 0:
                return x
            continue

        # 情况 2：x < b，按 gcd 不变的一整段跳跃
        d = math.gcd(x, b)
        y = x // d
        m = b // d

        # 找到最早让 gcd 变大的步数 delta
        delta = 10 ** 18
        for p in factors:
            if m % p == 0:
                step = p - (y % p)
                if step < delta:
                    delta = step

        # 如果剩余步数不足以走到 gcd 变化的位置，直接返回
        if rem < delta:
            return x + rem * d

        # 否则整段跳过去
        x += delta * d
        rem -= delta

    return x


def main():
    input = sys.stdin.readline
    t = int(input())
    ans = []
    for _ in range(t):
        a, b, n = map(int, input().split())
        ans.append(str(solve_one(a, b, n)))
    print("\n".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {

    // 存储 1e6 以内的所有质数
    static List<Integer> primes = new ArrayList<>();

    // 筛法预处理质数
    static void initPrimes(int limit) {
        boolean[] isPrime = new boolean[limit + 1];
        for (int i = 2; i <= limit; i++) {
            isPrime[i] = true;
        }
        for (int i = 2; i <= limit; i++) {
            if (isPrime[i]) {
                primes.add(i);
                if ((long) i * i <= limit) {
                    for (int j = i * i; j <= limit; j += i) {
                        isPrime[j] = false;
                    }
                }
            }
        }
    }

    // 分解 b，返回不同质因子
    static List<Long> getPrimeFactors(long x) {
        List<Long> factors = new ArrayList<>();
        long t = x;
        for (int p : primes) {
            if ((long) p * p > t) {
                break;
            }
            if (t % p == 0) {
                factors.add((long) p);
                while (t % p == 0) {
                    t /= p;
                }
            }
        }
        if (t > 1) {
            factors.add(t);
        }
        return factors;
    }

    // 求最大公约数
    static long gcd(long a, long b) {
        while (b != 0) {
            long t = a % b;
            a = b;
            b = t;
        }
        return a;
    }

    // 计算 f^(n)(a)
    static long solveOne(long a, long b, long n) {
        List<Long> factors = getPrimeFactors(b);
        long x = a;
        long rem = n;

        while (rem > 0) {
            // 情况 1：x >= b，再做一次就会进入不动点
            if (x >= b) {
                long d = gcd(x, b);
                x = b + d;
                rem--;
                // 进入不动点后，后面结果不再变化
                if (rem > 0) {
                    return x;
                }
                continue;
            }

            // 情况 2：x < b，按 gcd 不变的一段整体跳跃
            long d = gcd(x, b);
            long y = x / d;
            long m = b / d;

            long delta = Long.MAX_VALUE;
            for (long p : factors) {
                if (m % p == 0) {
                    long step = p - (y % p);
                    if (step < delta) {
                        delta = step;
                    }
                }
            }

            // 剩余步数不够走完整段，直接返回
            if (rem < delta) {
                return x + rem * d;
            }

            // 否则整段跳过去
            x += delta * d;
            rem -= delta;
        }

        return x;
    }

    public static void main(String[] args) throws Exception {
        initPrimes(1000000);

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < T; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            long a = Long.parseLong(st.nextToken());
            long b = Long.parseLong(st.nextToken());
            long n = Long.parseLong(st.nextToken());

            sb.append(solveOne(a, b, n)).append('\n');
        }

        System.out.print(sb.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> primes;

// 筛法预处理 1e6 以内的质数
void init_primes(int limit) {
    vector<bool> is_prime(limit + 1, true);
    is_prime[0] = is_prime[1] = false;
    for (int i = 2; i <= limit; i++) {
        if (is_prime[i]) {
            primes.push_back(i);
            if (1LL * i * i <= limit) {
                for (int j = i * i; j <= limit; j += i) {
                    is_prime[j] = false;
                }
            }
        }
    }
}

// 分解 b，返回不同质因子
vector<long long> get_prime_factors(long long x) {
    vector<long long> factors;
    long long t = x;
    for (int p : primes) {
        if (1LL * p * p > t) break;
        if (t % p == 0) {
            factors.push_back(p);
            while (t % p == 0) {
                t /= p;
            }
        }
    }
    if (t > 1) factors.push_back(t);
    return factors;
}

// 计算 f^(n)(a)
long long solve_one(long long a, long long b, long long n) {
    vector<long long> factors = get_prime_factors(b);
    long long x = a;
    long long rem = n;

    while (rem > 0) {
        // 情况 1：x >= b，再做一次进入不动点
        if (x >= b) {
            long long d = __gcd(x, b);
            x = b + d;
            rem--;
            // 进入不动点后，后面不再变化
            if (rem > 0) return x;
            continue;
        }

        // 情况 2：x < b，按 gcd 不变的一段跳跃
        long long d = __gcd(x, b);
        long long y = x / d;
        long long m = b / d;

        long long delta = (long long)4e18;
        for (long long p : factors) {
            if (m % p == 0) {
                long long step = p - (y % p);
                delta = min(delta, step);
            }
        }

        // 剩余步数不够走到 gcd 变化点，直接返回
        if (rem < delta) {
            return x + rem * d;
        }

        // 否则整段跳过去
        x += delta * d;
        rem -= delta;
    }

    return x;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    init_primes(1000000);

    int T;
    cin >> T;
    while (T--) {
        long long a, b, n;
        cin >> a >> b >> n;
        cout << solve_one(a, b, n) << '\n';
    }

    return 0;
}
```
