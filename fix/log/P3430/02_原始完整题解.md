## 题解思路

### 定义回顾

漂亮数（Right-truncatable prime）要求**从右向左逐位删除**时所得所有前缀都仍是素数。

> 例 739 → 73 → 7 → ∅ 全部为素数。

### 总量为何有限

1. **末位限制**
   除了个位本身为 2、3、5、7 的一位素数外，长度 ≥2 的漂亮数末位只能是 1, 3, 7, 9，否则一定是偶数或 5 的倍数。
2. **概率急剧下降**
   枚举第 k 位时只有 4 种扩展，而随机 $10^k$ 内素数密度约为 $1/\ln 10^k$。
   经验上枚举到 8 位以后再无可扩展数。

> **事实**：十进制右截素数仅 **83 个**，最大为 73 939 133（8 位）。

因此漂亮数个数是常数，预处理后可用二分回答任意区间查询。

### 预处理方法——BFS 逐位扩展

```text
队列初始: 2,3,5,7
while 队列非空:
    取出 p
    将 p 记录进答案表
    对 d ∈ {1,3,7,9}:
        q = p*10 + d
        若 q ≤ 1e18 且 q 为素数:
            入队 q
```

素数判定用 64 位确定性的 **Miller-Rabin**（基 2,3,5,7,11,13 即可覆盖 <2^64）或者试除法。

预处理后得到升序数组 `pretty[]`（83 个）。

> 预处理一次 $O(83\log n)$或$O(83 sqrt(n))$ 时间，$O(83)$ 空间。

### 查询

对每组 $[l,r]$

```
ans = upper_bound(pretty, r) - lower_bound(pretty, l)
```

时间 $O(\log 83)≈O(1)$。

## 复杂度分析

* 预处理：$O(83\log n)$，一次性执行。
* 单次询问：$O(\log 83)=O(1)$。
* 总复杂度：$O(T)$（$T≤10^5$ 也轻松通过）。
* 空间：$O(83)$。

## 代码

### Python 3

```python
# ---------------- 预处理 ----------------
from collections import deque
import sys

def is_prime(n: int) -> bool:
    """简单试除法，O(√n)"""
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    if n % 3 == 0:
        return n == 3
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

pretty = []                                     # 右截素数表
q = deque([2, 3, 5, 7])
while q:
    p = q.popleft()
    pretty.append(p)
    if p > 10**18 // 10:                        # 防溢出
        continue
    for d in (1, 3, 7, 9):
        nxt = p * 10 + d
        if nxt <= 10**18 and is_prime(nxt):
            q.append(nxt)
pretty.sort()                                   # 共 83 个

# ---------------- 处理查询 ----------------
import bisect
it = sys.stdin.buffer
t = int(it.readline())
out = []
for _ in range(t):
    l, r = map(int, it.readline().split())
    cnt = bisect.bisect_right(pretty, r) - bisect.bisect_left(pretty, l)
    out.append(str(cnt))
sys.stdout.write("\n".join(out))
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using u128 = __uint128_t;
const long long LIM = (long long)1e18;

bool isPrime(long long n) {
    if (n < 2) return false;
    if (n % 2 == 0) return n == 2;
    if (n % 3 == 0) return n == 3;
    for (long long i = 5; i * i <= n; i += 6)
        if (n % i == 0 || n % (i + 2) == 0) return false;
    return true;
}


/* -------- 预处理右截素数 -------- */
vector<long long> gen_pretty() {
    vector<long long> res;
    queue<long long> q;
    for (long long p : {2, 3, 5, 7}) q.push(p);
    while (!q.empty()) {
        long long p = q.front(); q.pop();
        res.push_back(p);
        if (p > LIM / 10) continue;
        for (int d : {1, 3, 7, 9}) {
            long long nxt = p * 10 + d;
            if (nxt <= LIM && isPrime(nxt)) q.push(nxt);
        }
    }
    sort(res.begin(), res.end());
    return res;         // 大小 83
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    vector<long long> pretty = gen_pretty();

    int T;  cin >> T;
    while (T--) {
        long long l, r; cin >> l >> r;
        auto it1 = lower_bound(pretty.begin(), pretty.end(), l);
        auto it2 = upper_bound(pretty.begin(), pretty.end(), r);
        cout << (it2 - it1) << '\n';
    }
    return 0;
}
```

### Java 17

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.*;

public class Main {

    static final long LIM = (long) 1e18;

    static boolean isPrime(long n) {
        if (n < 2) return false;
        if (n % 2 == 0) return n == 2;
        if (n % 3 == 0) return n == 3;
        for (long i = 5; i * i <= n; i += 6)
            if (n % i == 0 || n % (i + 2) == 0) return false;
        return true;
    }


    /* 生成全部右截素数（83 个） */
    static ArrayList<Long> gen() {
        ArrayDeque<Long> q = new ArrayDeque<>();
        q.addAll(Arrays.asList(2L, 3L, 5L, 7L));
        ArrayList<Long> res = new ArrayList<>();
        while (!q.isEmpty()) {
            long p = q.poll();
            res.add(p);
            if (p > LIM / 10) continue;
            for (int d : new int[]{1, 3, 7, 9}) {
                long nxt = p * 10 + d;
                if (nxt <= LIM && isPrime(nxt)) q.add(nxt);
            }
        }
        Collections.sort(res);
        return res;
    }

    public static void main(String[] args) throws Exception {
        ArrayList<Long> pretty = gen();          // 预处理
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            long l = Long.parseLong(st.nextToken());
            long r = Long.parseLong(st.nextToken());
            int left = Collections.binarySearch(pretty, l);
            int right = Collections.binarySearch(pretty, r);
            if (left < 0) left = -left - 1;          // lower_bound
            if (right < 0) right = -right - 2;       // upper_bound - 1
            sb.append(right - left + 1).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```
