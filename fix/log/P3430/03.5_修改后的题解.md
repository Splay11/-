## 解题思路

残核密钥要求从右向左逐位删去末位后，每一步得到的数都仍是素数。

长度至少为 $2$ 时，末位只能是 $1,3,7,9$，否则必为偶数或 $5$ 的倍数。逐位扩展时每次只有 $4$ 种候选，素数密度随位数下降很快。事实上十进制下这类数一共只有 $83$ 个，最大为 $8$ 位数 $73939133$。

因此可以一次性预处理出全部残核密钥：从一位素数 $2,3,5,7$ 出发做 BFS，每次在右侧接上 $1,3,7,9$，用试除判定是否为素数。得到升序数组后，每次询问 $[l,r]$ 用二分统计落在区间内的个数。

素数判定对不超过 $10^{18}$ 的数做试除即可；实际生成过程中数的位数不超过 $8$，预处理很快。

## 复杂度分析

预处理时间与残核密钥个数及试除代价有关，规模为常数；单次询问 $O(\log 83)=O(1)$。总时间复杂度 $O(T)$，空间复杂度 $O(1)$（表长为常数 $83$）。

## 代码实现

### Python

```python
from collections import deque
import sys
import bisect

def is_prime(n: int) -> bool:
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    if n % 3 == 0:
        return n == 3
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

pretty = []
q = deque([2, 3, 5, 7])
while q:
    p = q.popleft()
    pretty.append(p)
    if p > 10**18 // 10:
        continue
    for d in (1, 3, 7, 9):
        nxt = p * 10 + d
        if nxt <= 10**18 and is_prime(nxt):
            q.append(nxt)
pretty.sort()

it = sys.stdin.buffer
t = int(it.readline())
out = []
for _ in range(t):
    l, r = map(int, it.readline().split())
    cnt = bisect.bisect_right(pretty, r) - bisect.bisect_left(pretty, l)
    out.append(str(cnt))
sys.stdout.write("\n".join(out))
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    static final long LIM = (long) 1e18;

    static boolean isPrime(long n) {
        if (n < 2) return false;
        if (n % 2 == 0) return n == 2;
        if (n % 3 == 0) return n == 3;
        for (long i = 5; i * i <= n; i += 6)
            if (n % i == 0 || n % (i + 2) == 0) return false;
        return true;
    }

    static ArrayList<Long> gen() {
        ArrayDeque<Long> q = new ArrayDeque<>();
        q.addAll(Arrays.asList(2L, 3L, 5L, 7L));
        ArrayList<Long> res = new ArrayList<>();
        while (!q.isEmpty()) {
            long p = q.poll();
            res.add(p);
            if (p > LIM / 10) continue;
            for (int d : new int[]{1, 3, 7, 9}) {
                long nxt = p * 10 + d;
                if (nxt <= LIM && isPrime(nxt)) q.add(nxt);
            }
        }
        Collections.sort(res);
        return res;
    }

    public static void main(String[] args) throws Exception {
        ArrayList<Long> pretty = gen();
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            long l = Long.parseLong(st.nextToken());
            long r = Long.parseLong(st.nextToken());
            int left = Collections.binarySearch(pretty, l);
            int right = Collections.binarySearch(pretty, r);
            if (left < 0) left = -left - 1;
            if (right < 0) right = -right - 2;
            sb.append(right - left + 1).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;
const long long LIM = (long long)1e18;

bool isPrime(long long n) {
    if (n < 2) return false;
    if (n % 2 == 0) return n == 2;
    if (n % 3 == 0) return n == 3;
    for (long long i = 5; i * i <= n; i += 6)
        if (n % i == 0 || n % (i + 2) == 0) return false;
    return true;
}

vector<long long> gen_pretty() {
    vector<long long> res;
    queue<long long> q;
    for (long long p : {2, 3, 5, 7}) q.push(p);
    while (!q.empty()) {
        long long p = q.front();
        q.pop();
        res.push_back(p);
        if (p > LIM / 10) continue;
        for (int d : {1, 3, 7, 9}) {
            long long nxt = p * 10 + d;
            if (nxt <= LIM && isPrime(nxt)) q.push(nxt);
        }
    }
    sort(res.begin(), res.end());
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    vector<long long> pretty = gen_pretty();
    int T;
    cin >> T;
    while (T--) {
        long long l, r;
        cin >> l >> r;
        auto it1 = lower_bound(pretty.begin(), pretty.end(), l);
        auto it2 = upper_bound(pretty.begin(), pretty.end(), r);
        cout << (it2 - it1) << '\n';
    }
    return 0;
}
```
