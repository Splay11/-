## 解题思路

并网会把若干主机并入同一个集群，询问需要知道某台主机当前所在集群的算力之和。这是带权并查集：每个根维护该集群的规模。

询问与并网都带时刻，适合离线处理：

1. 把每次并网记为类型 $0$ 的事件，把每次询问记为类型 $1$ 的事件（并记下原询问编号）。
2. 按时刻从小到大排序；同一时刻类型 $0$ 排在类型 $1$ 前面，从而保证“先并网再询问”。
3. 按顺序扫描：遇到并网就把该批主机依次合并；遇到询问就用并查集找到根，输出其规模。

最后按询问的原始顺序打印答案。

## 复杂度分析

设事件总数为 $m+q$，单次合并涉及的主机个数之和为 $C$。时间复杂度 $O((n+C+m+q)\alpha(n)+(m+q)\log(m+q))$，空间复杂度 $O(n+C+m+q)$。

### Python

```python
def find(x):
    if f[x] == x:
        return x
    f[x] = find(f[x])
    return f[x]


def merge(x, y):
    x = find(x)
    y = find(y)
    if x == y:
        return
    f[x] = y
    sz[y] += sz[x]


n, m, q = map(int, input().split())
f = [i for i in range(n + 1)]
sz = [0] + list(map(int, input().split()))
que = []
for _ in range(m):
    temp = list(map(int, input().split()))
    arr = [0] + temp
    que.append(arr)
ans = [0] * q
for i in range(q):
    t, s = map(int, input().split())
    que.append([1, t, s, i])
que.sort(key=lambda x: (x[1], x[0]))
for arr in que:
    if arr[0] == 0:
        for i in range(4, len(arr)):
            merge(arr[i - 1], arr[i])
    else:
        ans[arr[3]] = sz[find(arr[2])]
for a in ans:
    print(a)
```

### Java

```java
import java.util.*;

public class Main {
    static List<Integer> f = new ArrayList<>();
    static List<Long> sz = new ArrayList<>();

    static int find(int x) {
        if (f.get(x) == x) return x;
        f.set(x, find(f.get(x)));
        return f.get(x);
    }

    static void merge(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) return;
        f.set(x, y);
        sz.set(y, sz.get(y) + sz.get(x));
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();
        f = new ArrayList<>(n + 1);
        sz = new ArrayList<>(n + 1);
        for (int i = 0; i <= n; i++) {
            f.add(i);
            if (i > 0) sz.add(scanner.nextLong());
            else sz.add(0L);
        }
        List<List<Integer>> que = new ArrayList<>();
        while (m-- > 0) {
            int t = scanner.nextInt();
            int c = scanner.nextInt();
            List<Integer> arr = new ArrayList<>();
            arr.add(0);
            arr.add(t);
            for (int i = 0; i < c; i++) arr.add(scanner.nextInt());
            que.add(arr);
        }
        List<Long> ans = new ArrayList<>();
        for (int i = 0; i < q; i++) ans.add(0L);
        for (int i = 0; i < q; i++) {
            int t = scanner.nextInt();
            int s = scanner.nextInt();
            que.add(Arrays.asList(1, t, s, i));
        }
        que.sort((x, y) -> {
            if (!x.get(1).equals(y.get(1))) return x.get(1) - y.get(1);
            return x.get(0) - y.get(0);
        });
        for (List<Integer> arr : que) {
            if (arr.get(0) == 0) {
                for (int i = 3; i < arr.size(); i++) {
                    merge(arr.get(i - 1), arr.get(i));
                }
            } else {
                ans.set(arr.get(3), sz.get(find(arr.get(2))));
            }
        }
        for (long an : ans) System.out.println(an);
    }
}
```

### C++

```cpp
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

vector<int> f;
vector<long long> sz;

int find(int x) {
    if (f[x] == x) return x;
    return f[x] = find(f[x]);
}

void merge(int x, int y) {
    x = find(x);
    y = find(y);
    if (x == y) return;
    f[x] = y;
    sz[y] += sz[x];
}

int main() {
    int n, m, q;
    cin >> n >> m >> q;
    f.resize(n + 1);
    sz.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        f[i] = i;
        cin >> sz[i];
    }
    vector<vector<int>> que;
    while (m-- > 0) {
        int t, c;
        cin >> t >> c;
        vector<int> arr(c + 2);
        arr[1] = t;
        for (int i = 2; i <= c + 1; i++) cin >> arr[i];
        que.push_back(arr);
    }
    vector<long long> ans(q);
    for (int i = 0; i < q; i++) {
        int t, s;
        cin >> t >> s;
        que.push_back({1, t, s, i});
    }
    sort(que.begin(), que.end(), [](const vector<int>& x, const vector<int>& y) {
        if (x[1] != y[1]) return x[1] < y[1];
        return x[0] < y[0];
    });
    for (const auto& arr : que) {
        if (arr[0] == 0) {
            for (int i = 3; i < (int)arr.size(); i++) merge(arr[i - 1], arr[i]);
        } else {
            ans[arr[3]] = sz[find(arr[2])];
        }
    }
    for (int i = 0; i < q; i++) cout << ans[i] << '\n';
    return 0;
}
```
