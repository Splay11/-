## 思路
&emsp;&emsp;此题的解题办法就是使用并查集，因为对于查询我们需要快速找到某个城市对应的城市圈的繁荣度，对于城市圈我们可以用并查集来代替，对城市圈中的每一个城市我们都将其放在同一个并查集中，但是对于城市圈的构建与查询我们需要离线来处理，以方便根据时间的大小来同步构建城市圈以及实时的城市查询，比较考验代码基础

### c++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> f;
vector<long long> sz;

int find(int x) {
    if (f[x] == x) return x;
    return f[x] = find(f[x]);
}

void merge(int x, int y) {
    x = find(x);
    y = find(y);
    if (x == y) return;
    f[x] = y;
    sz[y] += sz[x];
}

int main() {
    int n, m, q;
    cin >> n >> m >> q;
    f.resize(n + 1);
    sz.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        f[i] = i;
        cin >> sz[i];
    }
    vector<vector<int>> que;
    while (m-- > 0) {
        int t, c;
        cin >> t >> c;
        vector<int> arr(c + 2);
        arr[1] = t;
        for (int i = 2; i <= c + 1; i++) {
            cin >> arr[i];
        }
        que.push_back(arr);
    }
    vector<long long> ans(q);
    for (int i = 0; i < q; i++) {
        int t, s;
        cin >> t >> s;
        que.push_back({1, t, s, i});
    }
    sort(que.begin(), que.end(), [](const vector<int>& x, const vector<int>& y) {
        if (x[1] != y[1]) return x[1] < y[1];
        return x[0] < y[0];
    });
    for (const auto& arr : que) {
        if (arr[0] == 0) {
            for (int i = 3; i < arr.size(); i++) {
                merge(arr[i - 1], arr[i]);
            }
        } else {
            ans[arr[3]] = sz[find(arr[2])];
        }
    }
    for (int i = 0; i < q; i++) {
        cout << ans[i] << "\n";
    }
    return 0;
}
```

### java

```java
import java.util.*;

public class Main {
    static List<Integer> f = new ArrayList<>();
    static List<Long> sz = new ArrayList<>();

    static int find(int x) {
        if (f.get(x) == x) return x;
        f.set(x, find(f.get(x)));
        return f.get(x);
    }

    static void merge(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) return;
        f.set(x, y);
        sz.set(y, sz.get(y) + sz.get(x));
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();
        
        f = new ArrayList<>(n + 1);
        sz = new ArrayList<>(n + 1);
        
        for (int i = 0; i <= n; i++) {
            f.add(i);
            if(i>0)
            {
                sz.add(scanner.nextLong());
            }
            else 
            {
                sz.add((long)0);
            }
        }
        
        List<List<Integer>> que = new ArrayList<>();
        
        while (m-- > 0) {
            int t = scanner.nextInt();
            int c = scanner.nextInt();
            List<Integer> arr = new ArrayList<>(c + 2);
            arr.add(0);
            arr.add(t);
            for (int i = 0; i < c; i++) {
                arr.add(scanner.nextInt());
            }
            que.add(arr);
        }
        
        List<Long> ans = new ArrayList<>();
        for(int i=0;i<q;i++)ans.add((long)0);
        for (int i = 0; i < q; i++) {
            int t = scanner.nextInt();
            int s = scanner.nextInt();
            List<Integer> query = new ArrayList<>();
            query.add(1);
            query.add(t);
            query.add(s);
            query.add(i);
            que.add(query);
        }
        
        que.sort(new Comparator<List<Integer>>() {
            @Override
            public int compare(List<Integer> x, List<Integer> y) {
                if (!x.get(1).equals(y.get(1))) return x.get(1) - y.get(1);
                return x.get(0) - y.get(0);
            }
        });
        
        for (List<Integer> arr : que) {
            if (arr.get(0) == 0) {
                for (int i = 3; i < arr.size(); i++) {
                    merge(arr.get(i - 1), arr.get(i));
                }
            } else {
                ans.set(arr.get(3), sz.get(find(arr.get(2))));
            }
        }
        
        for (long an : ans) {
            System.out.println(an);
        }
    }
}
```
