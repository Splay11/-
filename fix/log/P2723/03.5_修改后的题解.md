## 解题思路

本题要求根据员工的出勤记录、抽查情况以及业务考核分数，计算出每位员工的最终绩效得分，并按得分从高到低排序输出。核心步骤如下：

1. **初始化考勤分数**：每位员工的考勤分数初始为 $100$ 分。
2. **模拟抽查扣分**：
   - 依次处理每一天的抽查记录。若当天某位员工**缺勤**（出勤状态为 $0$）且被抽查到，则将其考勤分数扣减 $10$ 分。
   - 考勤分数最低不能低于 $0$，扣分后取 $\max(\text{考勤分数} - 10, 0)$。
3. **计算最终绩效得分**：
   - 最终绩效 = 考勤分数 $\times 0.4$ + 业务考核分数 $\times 0.6$。
4. **排序输出**：
   - 按最终绩效得分从高到低排序，得分相同时按员工编号从小到大排序。
   - 输出员工编号和最终绩效得分，保留两位小数。

输入数据中的员工编号均为 $1$ 到 $N$ 的整数，处理时内部可转为 $0$ 索引以方便数组操作，输出时再转回原编号。

## 复杂度分析

- **时间复杂度**：$O(NM + P + N \log N)$，其中 $N$ 为员工数，$M$ 为工作天数，$P$ 为所有抽查记录中涉及的员工总人次（即全部 $\sum k_i$）。处理出勤和抽查扣分需要遍历所有数据，排序需要 $O(N \log N)$ 时间。由于 $N \le 1000$，$M \le 100$，完全可以接受。
- **空间复杂度**：$O(NM)$，主要用于存储出勤矩阵 `attendance[N][M]`。其他辅助数组为 $O(N)$ 级别。

## 代码实现

### Python

```python
def main():
    n, m = map(int, input().split())
    # 读取业务考核分数
    business_scores = list(map(int, input().split()))
    
    # 读取出勤情况: N行M列，1表示出勤，0表示缺勤
    attendance = []
    for _ in range(n):
        attendance.append(list(map(int, input().split())))
    
    # 读取每天的抽查记录
    roll_call = []
    for _ in range(m):
        data = list(map(int, input().split()))
        k = data[0]                # 当天抽查人数
        roll_call.append(data[1:]) # 存储被抽查的员工编号（1-based）
    
    # 考勤分数初始为100
    regular_scores = [100.0] * n
    
    # 处理每天的抽查扣分
    for day in range(m):
        for stu_id in roll_call[day]:
            stu_idx = stu_id - 1  # 转为0索引
            if attendance[stu_idx][day] == 0:   # 当天缺勤
                regular_scores[stu_idx] = max(regular_scores[stu_idx] - 10, 0.0)
    
    # 计算最终绩效得分
    final_scores = []
    for i in range(n):
        score = regular_scores[i] * 0.4 + business_scores[i] * 0.6
        final_scores.append((i + 1, score))
    
    # 排序：得分从高到低，得分相同时编号从小到大
    final_scores.sort(key=lambda x: (-x[1], x[0]))
    
    # 输出结果
    for emp_id, emp_score in final_scores:
        print(f"{emp_id} {emp_score:.2f}")

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static class Employee {
        int id;
        double score;
        Employee(int id, double score) {
            this.id = id;
            this.score = score;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); // 员工数量
        int m = sc.nextInt(); // 工作天数

        // 读取业务考核分数
        int[] businessScores = new int[n];
        for (int i = 0; i < n; i++) {
            businessScores[i] = sc.nextInt();
        }

        // 读取出勤情况，1表示出勤，0表示缺勤
        int[][] attendance = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                attendance[i][j] = sc.nextInt();
            }
        }

        // 读取每天的抽查记录
        List<List<Integer>> rollCall = new ArrayList<>();
        for (int day = 0; day < m; day++) {
            int k = sc.nextInt();
            List<Integer> checked = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                checked.add(sc.nextInt() - 1); // 转为0索引存储
            }
            rollCall.add(checked);
        }

        // 考勤分数初始化
        double[] regularScores = new double[n];
        Arrays.fill(regularScores, 100.0);

        // 处理抽查扣分
        for (int day = 0; day < m; day++) {
            for (int stuIdx : rollCall.get(day)) {
                if (attendance[stuIdx][day] == 0) { // 缺勤
                    regularScores[stuIdx] = Math.max(regularScores[stuIdx] - 10, 0.0);
                }
            }
        }

        // 计算最终绩效
        List<Employee> list = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            double finalScore = regularScores[i] * 0.4 + businessScores[i] * 0.6;
            list.add(new Employee(i + 1, finalScore));
        }

        // 排序
        list.sort((a, b) -> {
            if (Double.compare(a.score, b.score) == 0) {
                return Integer.compare(a.id, b.id);
            }
            return Double.compare(b.score, a.score);
        });

        // 输出
        for (Employee e : list) {
            System.out.printf("%d %.2f\n", e.id, e.score);
        }

        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <iomanip>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;

    // 读取业务考核分数
    vector<int> business_scores(n);
    for (int i = 0; i < n; ++i) {
        cin >> business_scores[i];
    }

    // 读取出勤情况，1出勤，0缺勤
    vector<vector<int>> attendance(n, vector<int>(m));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            cin >> attendance[i][j];
        }
    }

    // 读取每天的抽查记录
    vector<vector<int>> roll_call(m);
    for (int day = 0; day < m; ++day) {
        int k;
        cin >> k;
        roll_call[day].resize(k);
        for (int i = 0; i < k; ++i) {
            cin >> roll_call[day][i];
            roll_call[day][i]--; // 转为0索引
        }
    }

    // 考勤分数初始化为100
    vector<double> regular_scores(n, 100.0);

    // 处理抽查扣分
    for (int day = 0; day < m; ++day) {
        for (int stu_idx : roll_call[day]) {
            if (attendance[stu_idx][day] == 0) { // 缺勤
                regular_scores[stu_idx] = max(regular_scores[stu_idx] - 10, 0.0);
            }
        }
    }

    // 计算最终绩效并存储
    vector<pair<int, double>> final_scores(n);
    for (int i = 0; i < n; ++i) {
        double score = regular_scores[i] * 0.4 + business_scores[i] * 0.6;
        final_scores[i] = {i + 1, score};
    }

    // 排序
    sort(final_scores.begin(), final_scores.end(),
         [](const pair<int, double>& a, const pair<int, double>& b) {
             if (a.second == b.second)
                 return a.first < b.first;
             return a.second > b.second;
         });

    // 输出
    for (const auto& entry : final_scores) {
        cout << entry.first << " " << fixed << setprecision(2) << entry.second << endl;
    }

    return 0;
}
```
