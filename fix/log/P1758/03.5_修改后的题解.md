## 解题思路

$P = a_1! \times a_2! \times \cdots \times a_n!$ 的因子个数，等于把 $P$ 唯一分解后每个质因数指数加 $1$ 再相乘。

设 $M = \max a_i$。先线性筛出 $[1,M]$ 中的质数。

令 $\mathrm{pre}[t]$ 表示有多少个阶乘包含整数 $t$，即有多少个 $a_i \ge t$。这可以用差分数组 $O(n+M)$ 求出。

对每个质数 $p$，枚举它的倍数 $i$，用递推 $\mathrm{pw}[i] = \mathrm{pw}[i/p] + 1$ 得到 $i$ 中 $p$ 的指数，再累加 $\mathrm{pre}[i] \times \mathrm{pw}[i]$，即得到 $P$ 中 $p$ 的总指数。

最后把每个指数加 $1$ 连乘，并对 `1000000007` 取模。

## 复杂度分析

时间复杂度 $O(M\log\log M + \sum_{p} M/p) = O(M\log\log M)$，空间复杂度 $O(M)$，其中 $M \le 1000000$。

## 代码实现

### Python

```python
mod = 10 ** 9 + 7
ans = 1
n = int(input())
arr = list(map(int, input().split()))
mx = max(arr)

def sieve_of_eratosthenes(n):
    primes = []
    is_prime = [True] * (n + 1)
    for i in range(2, n + 1):
        if is_prime[i]:
            primes.append(i)
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    return primes

primes = sieve_of_eratosthenes(mx)
prefix_sum = [0] * (mx + 2)
for x in arr:
    prefix_sum[1] += 1
    prefix_sum[x + 1] -= 1
for i in range(1, mx + 1):
    prefix_sum[i] += prefix_sum[i - 1]
res = []
power = [0] * (mx + 1)
for p in primes:
    power[p] = 1
    now = 0
    for i in range(p, mx + 1, p):
        power[i] = power[i // p] + 1
        now += prefix_sum[i] * power[i]
    res.append(now)
    for i in range(p, mx + 1, p):
        power[i] = 0
for x in res:
    ans = ans * (x + 1) % mod
print(ans)
```

### Java

```java
import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] arr = new int[n];
        int mx = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = scan.nextInt();
            mx = Math.max(mx, arr[i]);
        }
        boolean[] isPrime = new boolean[mx + 1];
        java.util.Arrays.fill(isPrime, true);
        ArrayList<Integer> primes = new ArrayList<>();
        for (int i = 2; i <= mx; i++) {
            if (isPrime[i]) {
                primes.add(i);
                if ((long) i * i <= mx) {
                    for (int j = i * i; j <= mx; j += i) isPrime[j] = false;
                }
            }
        }
        int[] prefix = new int[mx + 2];
        for (int x : arr) {
            prefix[1]++;
            prefix[x + 1]--;
        }
        for (int i = 1; i <= mx; i++) prefix[i] += prefix[i - 1];
        int[] power = new int[mx + 1];
        long ans = 1;
        int mod = 1000000007;
        for (int p : primes) {
            power[p] = 1;
            long now = 0;
            for (int i = p; i <= mx; i += p) {
                power[i] = power[i / p] + 1;
                now += (long) prefix[i] * power[i];
            }
            ans = ans * ((now + 1) % mod) % mod;
            for (int i = p; i <= mx; i += p) power[i] = 0;
        }
        System.out.println(ans);
        scan.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    const int MOD = 1e9 + 7;
    int n;
    cin >> n;
    vector<int> arr(n);
    int mx = 0;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
        mx = max(mx, arr[i]);
    }
    vector<int> primes;
    vector<bool> is_prime(mx + 1, true);
    for (int i = 2; i <= mx; i++) {
        if (is_prime[i]) {
            primes.push_back(i);
            if (1LL * i * i <= mx) {
                for (int j = i * i; j <= mx; j += i) is_prime[j] = false;
            }
        }
    }
    vector<int> prefix(mx + 2);
    for (int x : arr) {
        prefix[1]++;
        prefix[x + 1]--;
    }
    for (int i = 1; i <= mx; i++) prefix[i] += prefix[i - 1];
    vector<int> power(mx + 1);
    long long ans = 1;
    for (int p : primes) {
        power[p] = 1;
        long long now = 0;
        for (int i = p; i <= mx; i += p) {
            power[i] = power[i / p] + 1;
            now += 1LL * prefix[i] * power[i];
        }
        ans = ans * ((now + 1) % MOD) % MOD;
        for (int i = p; i <= mx; i += p) power[i] = 0;
    }
    cout << ans << '\n';
    return 0;
}
```
