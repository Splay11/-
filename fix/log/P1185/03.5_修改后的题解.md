## 解题思路

将原数字串 $S$ 看作由三部分组成：**前缀** $A$（可为空）、**被删除的非空连续子串** $B$、**后缀** $C$（可为空）。删除 $B$ 后，剩余字符拼接成新的数字串，其数值等于 $V_A \times 10^{len_C} + V_C$，其中 $V_A$ 为 $A$ 的十进制数值，$len_C$ 为 $C$ 的长度，$V_C$ 为 $C$ 的数值。要求该值能被 $15$ 整除。

由于 $15$ 较小，可以采用模 $15$ 的前缀统计与后缀枚举，具体流程如下：

1. **预处理前缀模频数**  
   记 $L = |S|$。定义 `pref[i][m]` 为前 $i$ 个字符（$S[0..i-1]$）的所有前缀（包括空前缀）中，数值模 $15$ 等于 $m$ 的个数。  
   初始时 `pref[0][0] = 1`（空前缀模 $15$ 为 $0$），其余为 $0$。  
   遍历 $i = 1 \dots L$，计算到当前位置的前缀模值 `cur = (cur * 10 + digit) % 15`，并将 `pref[i]` 设置为 `pref[i-1]` 并累加 `pref[i][cur]++`。  
   最终 `pref[j][m]` 即表示了前 $j$ 个字符的所有前缀的模分布。

2. **枚举删除子串的结束位置**  
   设删除子串为 $S[i..j]$（$0 \le i \le j \le L-1$），则前缀 $A$ 为 $S[0..i-1]$（长度 $i$），后缀 $C$ 为 $S[j+1..L-1]$。  
   从后向前遍历 $j = L-1$ 到 $0$，同时维护：  
   - $suf \gets V_C \bmod 15$  
   - $w \gets 10^{len_C} \bmod 15$  
   （初始 $j = L-1$ 时 $suf = 0, w = 1$）  
   对于固定的 $j$，需要寻找所有 $i \le j$ 使得  
   \[
   (m \times w + suf) \bmod 15 = 0,
   \]  
   其中 $m = V_A \bmod 15$。  
   由于 $i \le j$ 的前缀都包含在前 $j$ 个字符的前缀集合中，可直接在 `pref[j]` 里枚举 $m = 0 \dots 14$，若条件成立则将 `pref[j][m]` 累加进答案。

3. **扣除不合法方案**  
   上述枚举会包含一种非法情况：删除整个字符串（$i = 0,\; j = L-1$，前缀和后缀均为空）。此时 $m = 0,\; suf = 0,\; w = 1$，必然满足条件，且 `pref[L-1][0]` 中包含了空前缀的贡献。因此最终答案需减去 $1$。

这样即可在 $O(15 \times L)$ 的时间内求出所有合法删除方案数。

## 复杂度分析

- 时间复杂度：$O(15 \times L)$。枚举 $j$ 的循环中需要遍历 $15$ 种前缀模值，整体与 $L$ 呈线性关系。  
- 空间复杂度：$O(15 \times L)$，用于存储前缀模频数数组 `pref`。当 $L \le 10^5$ 时，内存消耗在合理范围内（约 $12$ MB）。若需要优化，可只保留当前前缀频数，空间降为 $O(15)$，但本题不做要求。

## 代码实现

### Python

```python
import sys

def solve() -> None:
    S = sys.stdin.readline().strip()
    L = len(S)
    MOD = 15

    # pref[i][m]：前i个字符的所有前缀中，模15 = m 的个数
    pref = [[0] * MOD for _ in range(L + 1)]
    pref[0][0] = 1          # 空前缀模15为0
    cur = 0
    for i, ch in enumerate(S, 1):
        cur = (cur * 10 + int(ch)) % MOD
        # 继承前i-1个字符的统计
        for m in range(MOD):
            pref[i][m] = pref[i - 1][m]
        pref[i][cur] += 1

    ans = 0
    suf = 0      # V_C mod 15
    w = 1        # 10^{len_C} mod 15

    # 从后向前枚举删除子串的结束位置 j
    for j in range(L - 1, -1, -1):
        # 枚举前缀模值 m
        for m in range(MOD):
            if (m * w + suf) % MOD == 0:
                ans += pref[j][m]
        # 更新后缀信息，为下一次（j-1）准备
        if j > 0:   # 下一次后缀会多包含 S[j]
            digit = int(S[j])
            suf = (suf + digit * w) % MOD
            w = (w * 10) % MOD

    # 删除整个字符串的方案被计入，需扣除
    ans -= 1
    print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;

public class Main {
    static final int MOD = 15;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String S = br.readLine();
        int L = S.length();

        // pref[i][m]：前i个字符的所有前缀中，模15 = m 的个数
        long[][] pref = new long[L + 1][MOD];
        pref[0][0] = 1;            // 空前缀模15为0
        int cur = 0;
        for (int i = 1; i <= L; i++) {
            cur = (cur * 10 + (S.charAt(i - 1) - '0')) % MOD;
            System.arraycopy(pref[i - 1], 0, pref[i], 0, MOD);
            pref[i][cur]++;
        }

        long ans = 0;
        int suf = 0;   // V_C mod 15
        int w = 1;     // 10^{len_C} mod 15

        // 从后向前枚举删除子串的结束位置 j
        for (int j = L - 1; j >= 0; j--) {
            for (int m = 0; m < MOD; m++) {
                if ((m * w + suf) % MOD == 0) {
                    ans += pref[j][m];
                }
            }
            // 更新后缀信息（为下一轮 j-1 做准备）
            if (j > 0) {
                int digit = S.charAt(j) - '0';
                suf = (suf + digit * w) % MOD;
                w = (w * 10) % MOD;
            }
        }

        ans--;  // 减去删除整个字符串的方案
        System.out.println(ans);
    }
}
```

#include <bits/stdc++.h>
using namespace std;

const int MOD = 15;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string S;
    cin >> S;
    int L = S.size();

    // pref[i][m]：前i个字符的所有前缀中，模15 = m 的个数
    vector<vector<long long>> pref(L + 1, vector<long long>(MOD, 0));
    pref[0][0] = 1;          // 空前缀模15为0
    int cur = 0;
    for (int i = 1; i <= L; i++) {
        cur = (cur * 10 + (S[i - 1] - '0')) % MOD;
        for (int m = 0; m < MOD; m++) {
            pref[i][m] = pref[i - 1][m];
        }
        pref[i][cur]++;
    }

    long long ans = 0;
    int suf = 0;   // V_C mod 15
    int w = 1;     // 10^{len_C} mod 15

    // 从后向前枚举删除子串的结束位置 j
    for (int j = L - 1; j >= 0; j--) {
        for (int m = 0; m < MOD; m++) {
            if ((m * w + suf) % MOD == 0) {
                ans += pref[j][m];
            }
        }
        // 更新后缀信息（为下一轮 j-1 做准备）
        if (j > 0) {
            int digit = S[j] - '0';
            suf = (suf + digit * w) % MOD;
            w = (w * 10) % MOD;
        }
    }

    ans--;  // 减去删除整个字符串的方案
    cout << ans << endl;

    return 0;
}
