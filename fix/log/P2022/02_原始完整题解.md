## 思路
&emsp;&emsp;我们只能够进行一次操作，所以我们可以枚举去交换哪两个位置，在选定了位置之后，观察这两个位置交换之后对总体时间的影响，并计算，设交换位置为i与j，两个位置的时间分别为a与b，可以计算出位置的交换对中间的位置共减少了(j-i-1)*(a-b)的时间，对于交换的两个位置共减少了(a-b)的时间，可以自行证明，这样就可以快速的求出交换任意两个位置后对总体时间的影响
## 代码
### c++

```cpp
#include<iostream>
#include<cstring>
#include<algorithm>
using namespace std;
typedef long long ll;
const int N=1e5+10;
int n;
ll a[N];
int ans,x,y;
int main()
{
    cin>>n;
    for(int i=1;i<=n;i++)cin>>a[i];
    for(int i=1;i<n;i++)
        for(int j=i+1;j<=n;j++)
        {
            ll temp=(j-i)*(a[i]-a[j]);
            if(temp>ans)
            {
                ans=temp;
                x=i;
                y=j;
            }
        }
    if(ans==0)
        cout<<-1<<endl;
    else 
        cout<<x<<' '<<y<<endl;
    return 0;
}
```
### java

```java
import java.util.*;

public class Main {
    static final int N = 100010;
    static int n;
    static long[] a = new long[N];
    static int ans, x, y;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        n = scanner.nextInt();
        for (int i = 1; i <= n; i++) {
            a[i] = scanner.nextLong();
        }

        ans = 0;
        x = 0;
        y = 0;

        for (int i = 1; i < n; i++) {
            for (int j = i + 1; j <= n; j++) {
                long temp = (j - i) * (a[i] - a[j]);
                if (temp > ans) {
                    ans = (int) temp;
                    x = i;
                    y = j;
                }
            }
        }

        if (ans == 0)
            System.out.println("-1");
        else
            System.out.println(x + " " + y);
    }
}
```

n = int(input())
a = list(map(int, input().split()))

ans = 0
x = 0

    for j in range(i + 1, n):

        if temp > ans:
            ans = temp
            x = i+1
            y = j+1

if ans == 0:
    print("-1")
