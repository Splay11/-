# 小歪的数字问题
## 题解思路

1. 题目要求计算所有满足 1 ≤ i < j ≤ n 的二元组 $ (i,j) $ 对应的值  
   $
   \frac{i+j}{\gcd(i,j)}
   $  
   的总和，并对 $10^9+7$ 取模。

2. 注意到令 $ d=gcd(i,j) $，令 $ i=d \cdot a, j=d\cdot b $ 且 $\gcd(a,b)=1$（且 $a<b$），有  
   $
   \frac{i+j}{\gcd(i,j)} = a+b.
   $
   因此整个和可以写成对所有 $d$ 的枚举，再统计在 $1\le a<b\le m$（其中 $ m=\lfloor n/d\rfloor $）且 $\gcd(a,b)=1$ 的对中 $a+b$ 的贡献，即  
   $
   \text{Ans}(n)=$$\sum_{d=1}^{n}$$ f\Bigl(\lfloor n/d\rfloor\Bigr),$
   其中 $f(m)=\sum_{1\le a<b\le m,\;\gcd(a,b)=1} (a+b)$。

3. 如何高效计算 $f(m)$？关键在于利用 Möbius 反演，将筛选出 $\gcd(a,b)=1$ 的对转换为对所有对的求和，再利用莫比乌斯函数做“筛掉”非互质的情况。  
   同时，还会用到欧拉函数（phi）来对部分贡献进行归纳。  

4. 另外，考虑到计算过程中涉及除法，为了模运算下正确计算，需要预处理逆元数组。

5. 最终我们采用预处理的方法：  
   - 利用筛法预处理 1 到 $10^6$ 内的最小质因子、素数、欧拉函数 phi 和莫比乌斯函数 mu。  
   - 预处理逆元数组 inv。  
   - 通过双重循环（外循环变量 i，从 1 到 lim；内循环遍历 i 的倍数 j），利用 Möbius 反演思想累加一个辅助数组 phi_sum[j]，该数组记录了与当前 i 对应的贡献。  
   - 同时再对答案数组 ans[j] 累加来自欧拉函数 phi 和 phi_sum 的两部分贡献。  
   - 最后再对 ans 数组做一次前缀和处理，使得 ans[n] 表示所有 $1\le i<j\le n$ 的答案，便于多组数据查询时直接 O(1) 输出结果。

## 复杂度分析  
   - 预处理最小质因子、phi、mu 的筛法时间复杂度约 O(n)；  
   - 内部双重循环总复杂度大约为 O(n log n)；  
   - 由于 n 的上界为 $10^6$（且所有测试数据中 n 之和不超过 $10^9$），预处理时间可以接受；  
   - 每组测试数据查询只需 O(1) 时间。


## 代码实现


**Python 代码**

```python
# Python 实现
# 注意：由于 Python 循环效率较低，建议使用 PyPy 或其他加速手段

MOD = 10**9 + 7
lim = 10**6

# 预处理数组，初始化最小质因子、欧拉函数和莫比乌斯函数
minfac = [0] * (lim + 1)
phi = [0] * (lim + 1)
mu = [0] * (lim + 1)
primes = []

phi[1] = 1
mu[1] = 1

# 筛法预处理
for i in range(2, lim + 1):
    if minfac[i] == 0:
        minfac[i] = i
        primes.append(i)
        phi[i] = i - 1
        mu[i] = -1
    for p in primes:
        if p > minfac[i] or i * p > lim:
            break
        minfac[i * p] = p
        if i % p == 0:
            phi[i * p] = phi[i] * p
            mu[i * p] = 0
            break
        else:
            phi[i * p] = phi[i] * (p - 1)
            mu[i * p] = -mu[i]

# 计算逆元数组 inv
inv = [0] * (lim + 1)
inv[1] = 1
for i in range(2, lim + 1):
    inv[i] = inv[MOD % i] * (MOD - MOD // i) % MOD

# 预处理 phi_sum 数组
phi_sum = [0] * (lim + 1)
for i in range(1, lim + 1):
    # 枚举 i 的所有倍数 j
    # 此处 now 表示 i*(i+1)/2 * (j/i) 的贡献
    now = (i * (i + 1) // 2) % MOD
    for j in range(i, lim + 1, i):
        phi_sum[j] = (phi_sum[j] + mu[j // i] * now * (j // i)) % MOD

# 预处理答案数组 ans
ans = [0] * (lim + 1)
for i in range(1, lim + 1):
    for j in range(i, lim + 1, i):
        d = phi[i]
        g = j // i
        ans[j] = (ans[j] + d * j % MOD * inv[g]) % MOD
        ans[j] = (ans[j] + phi_sum[i] * g % MOD * inv[g]) % MOD
    # 去掉重复计数部分，调整 ans[i]
    ans[i] = (ans[i] - 2 * i * inv[i]) % MOD
    ans[i] = (ans[i] + MOD) % MOD

# 对 ans 数组做前缀和，便于快速查询答案
for i in range(1, lim + 1):
    ans[i] = (ans[i] + ans[i - 1]) % MOD

# 输入输出部分
import sys
input = sys.stdin.readline
T = int(input())
for _ in range(T):
    n = int(input())
    print(ans[n] % MOD)
```

**Java 代码**

```java
import java.util.*;
import java.io.*;

public class Main {
    // 模数常数
    static final int MOD = (int)1e9 + 7;
    static final int LIM = (int)1e6;
    
    public static void main(String[] args) throws Exception {
        // 使用 BufferedReader 加速输入
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        
        // 预处理数组
        int[] minfac = new int[LIM + 1];   // 最小质因子
        int[] primes = new int[LIM + 1];     // 存储素数
        int[] phi = new int[LIM + 1];        // 欧拉函数
        int[] mu = new int[LIM + 1];         // 莫比乌斯函数
        
        int cnt = 0;
        phi[1] = 1;
        mu[1] = 1;
        
        // 筛法预处理最小质因子、欧拉函数、莫比乌斯函数
        for (int i = 2; i <= LIM; i++) {
            if (minfac[i] == 0) {
                minfac[i] = i;
                primes[++cnt] = i;
                phi[i] = i - 1;
                mu[i] = -1;
            }
            for (int j = 1; j <= cnt; j++) {
                int p = primes[j];
                if (i * p > LIM) break;
                minfac[i * p] = p;
                if (i % p == 0) {
                    phi[i * p] = phi[i] * p;
                    mu[i * p] = 0;
                    break;
                } else {
                    phi[i * p] = phi[i] * (p - 1);
                    mu[i * p] = -mu[i];
                }
            }
        }
        
        // 计算逆元数组 inv
        int[] inv = new int[LIM + 1];
        inv[1] = 1;
        for (int i = 2; i <= LIM; i++) {
            inv[i] = (int)((1L * inv[MOD % i] * (MOD - MOD / i)) % MOD);
        }
        
        // 预处理 phi_sum 数组
        int[] phi_sum = new int[LIM + 1];
        for (int i = 1; i <= LIM; i++) {
            // 计算 now = i*(i+1)/2 的贡献
            int now = (int)((1L * i * (i + 1) / 2) % MOD);
            for (int j = i; j <= LIM; j += i) {
                // j/i 一定为整数
                phi_sum[j] = (int)((phi_sum[j] + 1L * mu[j / i] * now % MOD * (j / i)) % MOD);
            }
        }
        
        // 预处理答案数组 ans
        int[] ans = new int[LIM + 1];
        for (int i = 1; i <= LIM; i++) {
            for (int j = i; j <= LIM; j += i) {
                int d = phi[i];
                int g = j / i;
                ans[j] = (int)((ans[j] + 1L * d * j % MOD * inv[g]) % MOD);
                ans[j] = (int)((ans[j] + 1L * phi_sum[i] * g % MOD * inv[g]) % MOD);
            }
            ans[i] = (int)((ans[i] - 2L * i * inv[i]) % MOD);
            ans[i] = (ans[i] % MOD + MOD) % MOD;
        }
        
        // 对 ans 数组做前缀和，便于每组测试 O(1) 查询答案
        for (int i = 1; i <= LIM; i++) {
            ans[i] = (ans[i] + ans[i - 1]) % MOD;
        }
        
        // 逐组输出答案
        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine().trim());
            sb.append((ans[n] % MOD + MOD) % MOD).append("\n");
        }
        System.out.print(sb.toString());
    }
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;
 
typedef long long ll;
 
const int LIM = 1e6;
const int MOD = 1e9 + 7;
 
int minfac[LIM + 1], primes[LIM + 1], phi[LIM + 1], mu[LIM + 1];
int inv[LIM + 1], phi_sum[LIM + 1], ans[LIM + 1];
 
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
 
    int cnt = 0;
    phi[1] = 1;
    mu[1] = 1;
    // 筛法预处理最小质因子、欧拉函数和莫比乌斯函数
    for (int i = 2; i <= LIM; i++) {
        if (minfac[i] == 0) {
            minfac[i] = i;
            primes[++cnt] = i;
            phi[i] = i - 1;
            mu[i] = -1;
        }
        for (int j = 1; j <= cnt; j++) {
            int p = primes[j];
            if (i * p > LIM) break;
            minfac[i * p] = p;
            if (i % p == 0) {
                phi[i * p] = phi[i] * p;
                mu[i * p] = 0;
                break;
            } else {
                phi[i * p] = phi[i] * (p - 1);
                mu[i * p] = -mu[i];
            }
        }
    }
 
    // 计算逆元数组 inv
    inv[1] = 1;
    for (int i = 2; i <= LIM; i++){
        inv[i] = (int)((1LL * inv[MOD % i] * (MOD - MOD / i)) % MOD);
    }
 
    // 预处理 phi_sum 数组
    for (int i = 1; i <= LIM; i++){
        // 计算 now = i*(i+1)/2 的贡献
        int now = (int)((1LL * i * (i + 1) / 2) % MOD);
        for (int j = i; j <= LIM; j += i){
            int k = j / i; // 注意 j 一定为 i 的倍数
            phi_sum[j] = (phi_sum[j] + 1LL * mu[k] * now % MOD * k) % MOD;
        }
    }
 
    // 预处理答案数组 ans
    for (int i = 1; i <= LIM; i++){
        for (int j = i; j <= LIM; j += i){
            int d = phi[i];
            int g = j / i;
            ans[j] = (ans[j] + 1LL * d * j % MOD * inv[g]) % MOD;
            ans[j] = (ans[j] + 1LL * phi_sum[i] * g % MOD * inv[g]) % MOD;
        }
        ans[i] = (ans[i] - 2LL * i * inv[i]) % MOD;
        ans[i] = (ans[i] % MOD + MOD) % MOD;
    }
 
    // 求前缀和，方便每组测试 O(1) 输出答案
    for (int i = 1; i <= LIM; i++){
        ans[i] = (ans[i] + ans[i - 1]) % MOD;
    }
 
    int T;
    cin >> T;
    while (T--){
        int n;
        cin >> n;
        cout << (ans[n] % MOD + MOD) % MOD << "\n";
    }
    return 0;
}
```
