## 解题思路

1. **代换**：令 $d = \gcd(i,j)$，$i = d \cdot a$，$j = d \cdot b$，则 $\gcd(a,b)=1$ 且 $a < b \le \lfloor n/d \rfloor$，配对权值化为 $a+b$。于是
   $$
   \mathrm{Ans}(n) = \sum_{d=1}^{n} f\bigl(\lfloor n/d \rfloor\bigr),
   $$
   其中 $f(m) = \sum_{1 \le a < b \le m,\; \gcd(a,b)=1} (a+b)$。
2. **莫比乌斯与欧拉函数**：用筛法预处理 $[1, 10^6]$ 的最小质因子、欧拉函数 $\varphi$ 与莫比乌斯函数 $\mu$，并预处理模 $10^9+7$ 下的逆元。
3. **线性筛卷积**：枚举倍数累加 $\varphi$ 与 $\mu$ 的贡献，得到每个 $m$ 上 $f(m)$ 的相关数组，再对答案数组做前缀和。这样 $\mathrm{Ans}(n)$ 可 $O(1)$ 查询。
4. **多测**：先预处理到 $10^6$，再对每组 $n$ 直接输出。

## 复杂度分析

预处理时间复杂度 $O(L \log L)$，其中 $L = 10^6$；每组查询 $O(1)$。空间复杂度 $O(L)$。

## 代码实现

### Python

```python
MOD = 10**9 + 7
lim = 10**6

minfac = [0] * (lim + 1)
phi = [0] * (lim + 1)
mu = [0] * (lim + 1)
primes = []

phi[1] = 1
mu[1] = 1

for i in range(2, lim + 1):
    if minfac[i] == 0:
        minfac[i] = i
        primes.append(i)
        phi[i] = i - 1
        mu[i] = -1
    for p in primes:
        if p > minfac[i] or i * p > lim:
            break
        minfac[i * p] = p
        if i % p == 0:
            phi[i * p] = phi[i] * p
            mu[i * p] = 0
            break
        else:
            phi[i * p] = phi[i] * (p - 1)
            mu[i * p] = -mu[i]

inv = [0] * (lim + 1)
inv[1] = 1
for i in range(2, lim + 1):
    inv[i] = inv[MOD % i] * (MOD - MOD // i) % MOD

phi_sum = [0] * (lim + 1)
for i in range(1, lim + 1):
    now = (i * (i + 1) // 2) % MOD
    for j in range(i, lim + 1, i):
        phi_sum[j] = (phi_sum[j] + mu[j // i] * now * (j // i)) % MOD

ans = [0] * (lim + 1)
for i in range(1, lim + 1):
    for j in range(i, lim + 1, i):
        d = phi[i]
        g = j // i
        ans[j] = (ans[j] + d * j % MOD * inv[g]) % MOD
        ans[j] = (ans[j] + phi_sum[i] * g % MOD * inv[g]) % MOD
    ans[i] = (ans[i] - 2 * i * inv[i]) % MOD
    ans[i] = (ans[i] + MOD) % MOD

for i in range(1, lim + 1):
    ans[i] = (ans[i] + ans[i - 1]) % MOD

import sys
input = sys.stdin.readline
T = int(input())
for _ in range(T):
    n = int(input())
    print(ans[n] % MOD)
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static final int MOD = (int)1e9 + 7;
    static final int LIM = (int)1e6;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());

        int[] minfac = new int[LIM + 1];
        int[] primes = new int[LIM + 1];
        int[] phi = new int[LIM + 1];
        int[] mu = new int[LIM + 1];

        int cnt = 0;
        phi[1] = 1;
        mu[1] = 1;

        for (int i = 2; i <= LIM; i++) {
            if (minfac[i] == 0) {
                minfac[i] = i;
                primes[++cnt] = i;
                phi[i] = i - 1;
                mu[i] = -1;
            }
            for (int j = 1; j <= cnt; j++) {
                int p = primes[j];
                if (i * p > LIM) break;
                minfac[i * p] = p;
                if (i % p == 0) {
                    phi[i * p] = phi[i] * p;
                    mu[i * p] = 0;
                    break;
                } else {
                    phi[i * p] = phi[i] * (p - 1);
                    mu[i * p] = -mu[i];
                }
            }
        }

        int[] inv = new int[LIM + 1];
        inv[1] = 1;
        for (int i = 2; i <= LIM; i++) {
            inv[i] = (int)((1L * inv[MOD % i] * (MOD - MOD / i)) % MOD);
        }

        int[] phi_sum = new int[LIM + 1];
        for (int i = 1; i <= LIM; i++) {
            int now = (int)((1L * i * (i + 1) / 2) % MOD);
            for (int j = i; j <= LIM; j += i) {
                phi_sum[j] = (int)((phi_sum[j] + 1L * mu[j / i] * now % MOD * (j / i)) % MOD);
            }
        }

        int[] ans = new int[LIM + 1];
        for (int i = 1; i <= LIM; i++) {
            for (int j = i; j <= LIM; j += i) {
                int d = phi[i];
                int g = j / i;
                ans[j] = (int)((ans[j] + 1L * d * j % MOD * inv[g]) % MOD);
                ans[j] = (int)((ans[j] + 1L * phi_sum[i] * g % MOD * inv[g]) % MOD);
            }
            ans[i] = (int)((ans[i] - 2L * i * inv[i]) % MOD);
            ans[i] = (ans[i] % MOD + MOD) % MOD;
        }

        for (int i = 1; i <= LIM; i++) {
            ans[i] = (ans[i] + ans[i - 1]) % MOD;
        }

        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine().trim());
            sb.append((ans[n] % MOD + MOD) % MOD).append("\n");
        }
        System.out.print(sb.toString());
    }
}
```
