## 解题思路

本题要求对每个询问区间 $[l, r]$，找出最早一次实验后，使得区间内所有容器属于同一个试剂族群。若至实验结束仍未统一，则输出 $-1$，初始状态视为 $0$ 时刻。

观察实验过程：每次实验选择一个相邻位置对 $(p, p+1)$，若当前两者不在同一族群，则将它们所在的整个族群合并。这本质上是依时间顺序将位置 $p$ 和 $p+1$ 所在的连通块合并，合并时间即为实验编号 $k$。因此，我们可以用 **Kruskal 重构树** 建模这一过程：

1. **初始化重构树**：初始时每个位置 $i$（$1 \le i \le n$）为独立的叶子节点，共 $n$ 个节点。
2. **处理实验**：按顺序遍历 $t$ 次实验，第 $k$ 次实验选择位置 $p$：
   - 利用并查集找到 $p$ 与 $p+1$ 当前所在连通块的代表节点 $f_p, f_{p+1}$。
   - 若 $f_p \neq f_{p+1}$，则新建一个内部节点 $u$（编号从 $n+1$ 递增），并令该节点的权值 $val[u] = k$（记录合并发生的时间）。
   - 将 $f_p$ 和 $f_{p+1}$ 作为 $u$ 的左右儿子，$u$ 成为它们新的父节点，并查集中将 $f_p, f_{p+1}$ 的代表元指向 $u$。
3. **重构树的性质**：对于任意两个位置 $l, r$，它们首次属于同一族群的实验时刻，恰为重构树中它们对应叶节点的 **最近公共祖先（LCA）** 的权值 $val$。

这样，一询问区间 $[l, r]$ 中所有容器属于同一族群的时刻，等价于区间内所有叶子节点在重构树中的 LCA 权值。由树的性质可知，多个节点的 LCA 等于任意一对节点的 LCA（因为一旦某对节点在同一子树，则整棵子树内的节点 LCA 相同，但我们需要确保区间内所有节点都属于该子树）。更具体地，对于 $l = r$ 答案为 $0$（初始已同组）；对于 $l < r$，答案就是叶子 $l$ 与 $r$ 的 LCA 的权值，若它们从未合并（不在同一棵重构树中），则输出 $-1$。

**实现步骤**：
- **并查集 Union-Find**：维护当前连通块代表，支持路径压缩。
- **重构树存储**：数组 `left`, `right` 记录左右孩子，`val` 记录合并时刻，`par` 记录父节点。
- **LCA 预处理**：重构树节点总数最多为 $n + \min(t, n-1) \le 2n$。找到所有根节点（`par[i] == 0`），对每棵树进行 BFS/DFS 计算深度 `depth`，并用倍增法预处理 `up[k][u]`（$2^k$ 级祖先）。同时记录每个节点所属树的根 `root_id`，以便迅速判断两节点是否连通。
- **查询**：对于询问 $[l, r]$，若 $l=r$ 输出 `0`；否则调用 LCA 函数，若两叶子不在同一棵重构树则返回 $-1$，否则输出 `val[lca(l, r)]`。

## 复杂度分析

- **构建重构树**：遍历 $t$ 次操作，每次并查集查找与合并均摊复杂度 $O(\alpha(N))$，总复杂度 $O((n+t)\,\alpha(N))$。
- **LCA 预处理**：节点总数 $N \le 2n$，BFS 与倍增数组构造复杂度 $O(N \log N)$。
- **单次查询**：倍增法求 LCA 复杂度 $O(\log N)$，总查询复杂度 $O(q \log N)$。
- **时间复杂度**：$O((n+t+q) \log n)$，可以处理 $2 \times 10^5$ 的输入。
- **空间复杂度**：存储重构树结构及倍增数组需要 $O(N \log N) \subseteq O(n \log n)$ 空间。

## 代码实现

### Python

```python
import sys
from collections import deque

def main():
    input = sys.stdin.readline
    n, t, q = map(int, input().split())
    ops = [int(input()) for _ in range(t)]
    
    # 初始化并查集
    size = n + t + 5
    dsu_fa = list(range(size))
    left = [0] * size
    right = [0] * size
    val = [0] * size          # 合并时间
    par = [0] * size          # 重构树中的父节点
    
    def dsu_find(x):
        while dsu_fa[x] != x:
            dsu_fa[x] = dsu_fa[dsu_fa[x]]
            x = dsu_fa[x]
        return x
    
    cur = n  # 叶子节点占 1..n，新节点从 n+1 开始
    for k in range(1, t + 1):
        p = ops[k - 1]
        fp = dsu_find(p)
        fp1 = dsu_find(p + 1)
        if fp != fp1:
            cur += 1
            val[cur] = k          # 记录合并发生的实验编号
            left[cur] = fp
            right[cur] = fp1
            par[fp] = cur
            par[fp1] = cur
            # 合并连通块
            dsu_fa[fp] = cur
            dsu_fa[fp1] = cur
    
    # 构建树的有向图（父 -> 子）
    g = [[] for _ in range(cur + 1)]
    for i in range(n + 1, cur + 1):
        g[i].append(left[i])
        g[i].append(right[i])
    
    LOG = 20
    up = [[0] * (cur + 1) for _ in range(LOG)]
    depth = [0] * (cur + 1)
    root_id = [0] * (cur + 1)
    
    # 对所有重构树根节点进行 BFS
    for i in range(1, cur + 1):
        if par[i] == 0:   # 根节点
            root = i
            q_bfs = deque([root])
            root_id[root] = root
            while q_bfs:
                u = q_bfs.popleft()
                up[0][u] = par[u]
                for j in range(1, LOG):
                    up[j][u] = up[j - 1][up[j - 1][u]]
                for v in g[u]:
                    if v != 0:
                        depth[v] = depth[u] + 1
                        root_id[v] = root
                        q_bfs.append(v)
    
    def lca(u, v):
        if root_id[u] != root_id[v]:
            return 0
        if depth[u] < depth[v]:
            u, v = v, u
        for i in range(LOG - 1, -1, -1):
            if depth[u] - (1 << i) >= depth[v]:
                u = up[i][u]
        if u == v:
            return u
        for i in range(LOG - 1, -1, -1):
            if up[i][u] != up[i][v]:
                u = up[i][u]
                v = up[i][v]
        return up[0][u]
    
    out = []
    for _ in range(q):
        l, r = map(int, input().split())
        if l == r:
            out.append("0")
        else:
            anc = lca(l, r)
            if anc == 0:
                out.append("-1")
            else:
                out.append(str(val[anc]))
    
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static int[] dsuFa;
    static int[] left, right, val, par;
    static ArrayList<Integer>[] g;
    static int[][] up;
    static int[] depth, rootId;
    static final int LOG = 20;
    
    static int dsuFind(int x) {
        if (dsuFa[x] != x) {
            dsuFa[x] = dsuFind(dsuFa[x]);
        }
        return dsuFa[x];
    }
    
    static void bfs(int root) {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(root);
        rootId[root] = root;
        depth[root] = 0;
        while (!queue.isEmpty()) {
            int u = queue.poll();
            up[0][u] = par[u];
            for (int i = 1; i < LOG; i++) {
                up[i][u] = up[i - 1][up[i - 1][u]];
            }
            for (int v : g[u]) {
                if (v != 0) {
                    depth[v] = depth[u] + 1;
                    rootId[v] = root;
                    queue.add(v);
                }
            }
        }
    }
    
    static int lca(int u, int v) {
        if (rootId[u] != rootId[v]) return 0;
        if (depth[u] < depth[v]) {
            int tmp = u; u = v; v = tmp;
        }
        for (int i = LOG - 1; i >= 0; i--) {
            if (depth[u] - (1 << i) >= depth[v]) {
                u = up[i][u];
            }
        }
        if (u == v) return u;
        for (int i = LOG - 1; i >= 0; i--) {
            if (up[i][u] != up[i][v]) {
                u = up[i][u];
                v = up[i][v];
            }
        }
        return up[0][u];
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int t = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        
        int[] ops = new int[t];
        for (int i = 0; i < t; i++) {
            ops[i] = Integer.parseInt(br.readLine());
        }
        
        int size = n + t + 5;
        dsuFa = new int[size];
        left = new int[size];
        right = new int[size];
        val = new int[size];
        par = new int[size];
        for (int i = 0; i < size; i++) dsuFa[i] = i;
        
        int cur = n;
        for (int k = 1; k <= t; k++) {
            int p = ops[k - 1];
            int fp = dsuFind(p);
            int fp1 = dsuFind(p + 1);
            if (fp != fp1) {
                cur++;
                val[cur] = k;
                left[cur] = fp;
                right[cur] = fp1;
                par[fp] = cur;
                par[fp1] = cur;
                dsuFa[fp] = cur;
                dsuFa[fp1] = cur;
            }
        }
        
        g = new ArrayList[cur + 1];
        for (int i = 0; i <= cur; i++) g[i] = new ArrayList<>();
        for (int i = n + 1; i <= cur; i++) {
            g[i].add(left[i]);
            g[i].add(right[i]);
        }
        
        up = new int[LOG][cur + 1];
        depth = new int[cur + 1];
        rootId = new int[cur + 1];
        for (int i = 1; i <= cur; i++) {
            if (par[i] == 0) {
                bfs(i);
            }
        }
        
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int l = Integer.parseInt(st.nextToken());
            int r = Integer.parseInt(st.nextToken());
            if (l == r) {
                sb.append("0\n");
            } else {
                int anc = lca(l, r);
                if (anc == 0) sb.append("-1\n");
                else sb.append(val[anc]).append("\n");
            }
        }
        System.out.print(sb);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 4e5 + 10;
const int LOG = 20;

int dsu_fa[MAXN];
int Left[MAXN], Right[MAXN], val[MAXN], par[MAXN];
vector<int> g[MAXN];
int up[LOG][MAXN];
int depth[MAXN], root_id[MAXN];

int dsu_find(int x) {
    if (dsu_fa[x] != x)
        dsu_fa[x] = dsu_find(dsu_fa[x]);
    return dsu_fa[x];
}

void bfs(int root) {
    queue<int> q;
    q.push(root);
    root_id[root] = root;
    depth[root] = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        up[0][u] = par[u];
        for (int i = 1; i < LOG; i++) {
            up[i][u] = up[i - 1][up[i - 1][u]];
        }
        for (int v : g[u]) {
            if (v != 0) {
                depth[v] = depth[u] + 1;
                root_id[v] = root;
                q.push(v);
            }
        }
    }
}

int lca(int u, int v) {
    if (root_id[u] != root_id[v]) return 0;
    if (depth[u] < depth[v]) swap(u, v);
    for (int i = LOG - 1; i >= 0; i--) {
        if (depth[u] - (1 << i) >= depth[v]) {
            u = up[i][u];
        }
    }
    if (u == v) return u;
    for (int i = LOG - 1; i >= 0; i--) {
        if (up[i][u] != up[i][v]) {
            u = up[i][u];
            v = up[i][v];
        }
    }
    return up[0][u];
}

        int fp1 = dsu_find(p + 1);

            Left[cur] = fp;
            Right[cur] = fp1;
            par[fp] = cur;
            par[fp1] = cur;
            dsu_fa[fp] = cur;
            dsu_fa[fp1] = cur;
        }
    }
    
    for (int i = n + 1; i <= cur; i++) {
        g[i].push_back(Left[i]);
        g[i].push_back(Right[i]);
    }
    
    // 对所有根节点进行 BFS
    for (int i = 1; i <= cur; i++) {
        if (par[i] == 0) {

        }
    }

            int anc = lca(l, r);
            if (anc == 0) cout << -1 << '\n';
```
