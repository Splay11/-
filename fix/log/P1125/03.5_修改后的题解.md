## 解题思路

在给定序列中，一个子段平衡意味着该连续子段内所有元素的乘积等于所有元素的按位异或和。分析可知，若子段中包含至少两个大于1的数，乘积会快速增大，必然大于异或和，不可能相等。因此平衡子段只有两种可能：

1. **全为 1 的子段**：  
   乘积恒为 1。异或和取决于 1 的个数：奇数个 1 的异或和为 1，偶数个为 0。  
   因此只有当 1 的个数为奇数时，乘积等于异或和，子段平衡。

2. **恰好包含一个大于 1 的数（记作 $x$），其余元素全为 1 的子段**：  
   设该子段中 $x$ 左侧有 $L$ 个 1，右侧有 $R$ 个 1。乘积为 $x$。  
   异或和为 $(L \bmod 2) \oplus x \oplus (R \bmod 2)$。  
   要求乘积等于异或和，即 $x = (L \bmod 2) \oplus x \oplus (R \bmod 2) \implies L \bmod 2 = R \bmod 2$。  
   因此左右两侧 1 的个数的奇偶性必须相同。

根据上述分析，统计平衡子段数量的方法为：

- 将序列按是否为 1 进行划分，记录每一段连续 1 的长度。大于 1 的数视为分隔点。
- 对于每一段连续 $k$ 个 1，其内部所有长度为奇数的子段都是平衡的。贡献为 $\sum_{j=0}^{k-1}\lceil (k-j)/2 \rceil$，等价于统计所有可能的奇数长度子段。
- 对于每一个大于 1 的数（包括序列首尾的虚拟边界），计算以其为中心（即有且仅有这一个大于 1 的数）时，左右两侧连续 1 的个数奇偶性相同的组合数。  
  具体而言，若左右两侧分别有 $L$ 个和 $R$ 个 1，则需要统计从左侧选 $l$ 个 1（$0 \le l \le L$）且从右侧选 $r$ 个 1（$0 \le r \le R$）时，$l \bmod 2 = r \bmod 2$ 的方案数。  
  用动态规划预处理组合计数：令 $dp[i][0]$ 表示 $i$ 个连续的 1 中，选取前缀（包含空前缀）使得 1 的个数为偶数的方案数；$dp[i][1]$ 表示个数为奇数的方案数。递推关系：  
  $dp[0][0]=1,\,dp[0][1]=0$；  
  $dp[i][0] = dp[i-1][1] + 1$（新增一个 1 翻转为偶数，加空前缀）；  
  $dp[i][1] = dp[i-1][0]$。  
  则右侧与左侧方案数的乘积之和为 $dp[L][0] \times dp[R][0] + dp[L][1] \times dp[R][1]$。

遍历所有连续 1 段和所有分隔点，累加贡献即可得到答案。

## 复杂度分析

- 时间复杂度：预处理 $dp$ 数组与统计连续 1 段均只需一次线性扫描，过程中计算贡献的总操作次数与序列长度成正比，因此时间复杂度为 $O(n)$。
- 空间复杂度：使用了 $dp$ 数组和辅助存储连续 1 段长度的数组，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    n = int(data[0])
    a = list(map(int, data[1:1+n]))

    # 预处理 dp，dp[i][0] 为 i 个1中选前缀得到偶数个1的方案数
    # dp[i][1] 为奇数个1的方案数
    dp = [[0, 0] for _ in range(n + 1)]
    dp[0][0] = 1
    for i in range(1, n + 1):
        dp[i][0] = dp[i-1][1] + 1
        dp[i][1] = dp[i-1][0]

    # 构建连续1段长度列表，非1位置用0占位，首尾补0方便处理边界
    tmp = [0]
    i = 0
    while i < n:
        if a[i] == 1:
            cnt = 0
            while i < n and a[i] == 1:
                cnt += 1
                i += 1
            tmp.append(cnt)
        else:
            tmp.append(0)
            i += 1
    tmp.append(0)

    ans = 0
    m = len(tmp)
    for i in range(1, m - 1):
        if tmp[i] == 0:
            L = tmp[i-1]
            R = tmp[i+1]
            ans += dp[L][0] * dp[R][0] + dp[L][1] * dp[R][1]
        else:
            k = tmp[i]
            for j in range(k):
                ans += (k - j + 1) // 2

    print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        // 预处理 dp
        long[][] dp = new long[n + 1][2];
        dp[0][0] = 1;
        for (int i = 1; i <= n; i++) {
            dp[i][0] = dp[i-1][1] + 1;
            dp[i][1] = dp[i-1][0];
        }

        // 构建连续1段长度列表，非1用0表示
        List<Integer> tmp = new ArrayList<>();
        tmp.add(0);
        int i = 0;
        while (i < n) {
            if (a[i] == 1) {
                int cnt = 0;
                while (i < n && a[i] == 1) {
                    cnt++;
                    i++;
                }
                tmp.add(cnt);
            } else {
                tmp.add(0);
                i++;
            }
        }
        tmp.add(0);

        long ans = 0;
        for (int j = 1; j < tmp.size() - 1; j++) {
            if (tmp.get(j) == 0) {
                int L = tmp.get(j-1);
                int R = tmp.get(j+1);
                ans += dp[L][0] * dp[R][0] + dp[L][1] * dp[R][1];
            } else {
                int k = tmp.get(j);
                for (int t = 0; t < k; t++) {
                    ans += (k - t + 1) / 2;
                }
            }
        }

        System.out.println(ans);
    }
}
```

```cpp
using LL = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    // 预处理 dp，dp[i][0] 偶数方案数，dp[i][1] 奇数方案数
    vector<vector<LL>> dp(n + 1, vector<LL>(2, 0));
    dp[0][0] = 1;
    for (int i = 1; i <= n; i++) {
        dp[i][0] = dp[i-1][1] + 1;
        dp[i][1] = dp[i-1][0];
    }

    // 构建连续1段长度列表，首尾补0
    vector<int> tmp = {0};
    int i = 0;
    while (i < n) {
        if (a[i] == 1) {
            int cnt = 0;
            while (i < n && a[i] == 1) {
                cnt++;
                i++;
            }
            tmp.push_back(cnt);
        } else {
            tmp.push_back(0);
            i++;
        }
    }
    tmp.push_back(0);

    LL ans = 0;
    for (size_t j = 1; j + 1 < tmp.size(); j++) {
        if (tmp[j] == 0) {
            int L = tmp[j-1];
            int R = tmp[j+1];
            ans += dp[L][0] * dp[R][0] + dp[L][1] * dp[R][1];
        } else {
            int k = tmp[j];
            for (int t = 0; t < k; t++) {
                ans += (k - t + 1) / 2;
            }
        }
    }

    cout << ans << "\n";
    return 0;
}
```
