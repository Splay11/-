## 题解

### 题目描述

给定一棵根为节点1的树，每个节点有权值$r_i$和中和成本$c_i$。可以选择切断边（支付代价$d$）或中和节点（支付$c_i$使收益为0）。求根所在连通块的最大净收益（总收益减去操作代价）。

### 思路分析

**树形动态规划**：后序遍历每个节点，计算在保留该节点时，中和或不中和的最大贡献，并处理子节点的切断或保留决策。

1. **状态定义**：`dp[u]`表示节点u在保留时能提供的最大净收益。
2. **中和决策**：中和则收益为$-c_u$，否则为$r_u$。
3. **子节点处理**：对每个子节点v的边，选择切断（代价$-d$）或保留（收益`dp[v]`），取最大值累加。

#### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    
    int n;
    cin >> n;
    vector<int> r(n+1), c(n+1);
    for (int i = 1; i <= n; ++i) cin >> r[i];
    for (int i = 1; i <= n; ++i) cin >> c[i];
    
    vector<vector<pair<int, int>>> adj(n+1);
    for (int i = 0; i < n-1; ++i) {
        int u, v, d;
        cin >> u >> v >> d;
        adj[u].emplace_back(v, d);
        adj[v].emplace_back(u, d);
    }
    
    // 构建树结构，确定父节点和子节点列表
    vector<int> parent(n+1, 0);
    vector<vector<pair<int, int>>> children(n+1);
    queue<int> q;
    q.push(1);
    parent[1] = -1; // 根节点无父节点
    
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (auto& edge : adj[u]) {
            int v = edge.first;
            int d = edge.second;
            if (parent[v] == 0) { // 未被访问过
                parent[v] = u;
                children[u].emplace_back(v, d); // 子节点和边权
                q.push(v);
            }
        }
    }
    
    vector<ll> dp(n+1);
    stack<pair<int, bool>> stk;
    stk.push({1, false});
    
    while (!stk.empty()) {
        auto [u, processed] = stk.top();
        stk.pop();
        if (!processed) {
            stk.push({u, true});
            // 子节点逆序压入，保证处理顺序正确
            for (auto it = children[u].rbegin(); it != children[u].rend(); ++it) {
                stk.push({it->first, false});
            }
        } else {
            ll sum_children = 0;
            for (auto [v, d] : children[u]) {
                sum_children += max(-1LL * d, dp[v]);
            }
            ll option1 = (0LL - c[u]) + sum_children;
            ll option2 = r[u] + sum_children;
            dp[u] = max(option1, option2);
        }
    }
    
    cout << dp[1] << endl;
    
    return 0;
}
```

#### Python

```python
import sys
from collections import deque

def main():
    sys.setrecursionlimit(1 << 25)
    n = int(sys.stdin.readline())
    r = list(map(int, sys.stdin.readline().split()))
    c = list(map(int, sys.stdin.readline().split()))
    r = [0] + r  # 调整为1-based
    c = [0] + c
    
    adj = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v, d = map(int, sys.stdin.readline().split())
        adj[u].append((v, d))
        adj[v].append((u, d))
    
    # 构建树结构，确定父节点和子节点列表
    parent = [0]*(n+1)
    children = [[] for _ in range(n+1)]
    q = deque([1])
    parent[1] = -1  # 根节点无父节点
    
    while q:
        u = q.popleft()
        for v, d in adj[u]:
            if parent[v] == 0 and v != parent[u]:
                parent[v] = u
                children[u].append((v, d))
                q.append(v)
    
    dp = [0]*(n+1)
    stack = [(1, False)]
    
    while stack:
        u, processed = stack.pop()
        if not processed:
            stack.append((u, True))
            # 子节点逆序处理
            for v, d in reversed(children[u]):
                stack.append((v, False))
        else:
            sum_children = 0
            for v, d in children[u]:
                sum_children += max(-d, dp[v])
            option1 = (0 - c[u]) + sum_children
            option2 = r[u] + sum_children
            dp[u] = max(option1, option2)
    
    print(dp[1])

if __name__ == "__main__":
    main()
```

#### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static class Pair {
        int first;
        int second;
        Pair(int f, int s) {
            first = f;
            second = s;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] r = new int[n+1];
        int[] c = new int[n+1];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) r[i] = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) c[i] = Integer.parseInt(st.nextToken());

        List<List<Pair>> adj = new ArrayList<>(n+1);
        for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
        for (int i = 0; i < n-1; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int d = Integer.parseInt(st.nextToken());
            adj.get(u).add(new Pair(v, d));
            adj.get(v).add(new Pair(u, d));
        }

        // 构建树结构，确定父节点和子节点列表
        int[] parent = new int[n+1];
        List<List<Pair>> children = new ArrayList<>(n+1);
        for (int i = 0; i <= n; i++) children.add(new ArrayList<>());
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        parent[1] = -1;

        while (!q.isEmpty()) {
            int u = q.poll();
            for (Pair edge : adj.get(u)) {
                int v = edge.first;
                int d = edge.second;
                if (parent[v] == 0) { // 未被访问过
                    parent[v] = u;
                    children.get(u).add(new Pair(v, d));
                    q.add(v);
                }
            }
        }

        long[] dp = new long[n+1];
        Deque<Pair> stack = new ArrayDeque<>();
        stack.push(new Pair(1, 0)); // 0表示未处理

        while (!stack.isEmpty()) {
            Pair p = stack.pop();
            int u = p.first;
            int processed = p.second;
            if (processed == 0) {
                stack.push(new Pair(u, 1));
                List<Pair> childList = children.get(u);
                for (int i = childList.size() - 1; i >= 0; i--) {
                    Pair child = childList.get(i);
                    stack.push(new Pair(child.first, 0));
                }
            } else {
                long sum = 0;
                for (Pair child : children.get(u)) {
                    int v = child.first;
                    int d = child.second;
                    sum += Math.max(-d, dp[v]);
                }
                long option1 = (0L - c[u]) + sum;
                long option2 = r[u] + sum;
                dp[u] = Math.max(option1, option2);
            }
        }

        System.out.println(dp[1]);
    }
}
```
