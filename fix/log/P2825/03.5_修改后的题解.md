## 解题思路

这是一棵以 $1$ 号总控站为根的树。对每个基站 $u$，需要决策：是否支付 $c_u$ 将其收益置 $0$，以及每条连向子站 $v$ 的边是拆缆（付出 $-d$）还是保留（获得子树贡献）。

令 $dp[u]$ 表示**保留基站 $u$** 时，以 $u$ 为根的子树能提供的最大净收益。后序遍历计算：

- 对每个子站 $v$，边权为 $d$，贡献取 $\max(-d,\ dp[v])$（拆缆或保留）。
- $u$ 自身取 $\max(r_u,\ -c_u)$。
- 两者相加即为 $dp[u]$。

答案为 $dp[1]$。由于 $n$ 可达 $5 \times 10^5$，采用显式栈做后序遍历，避免递归爆栈。

## 复杂度分析

时间复杂度 $O(n)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
from collections import deque

def main():
    sys.setrecursionlimit(1 << 25)
    n = int(sys.stdin.readline())
    r = list(map(int, sys.stdin.readline().split()))
    c = list(map(int, sys.stdin.readline().split()))
    r = [0] + r
    c = [0] + c

    adj = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v, d = map(int, sys.stdin.readline().split())
        adj[u].append((v, d))
        adj[v].append((u, d))

    parent = [0] * (n + 1)
    children = [[] for _ in range(n + 1)]
    q = deque([1])
    parent[1] = -1

    while q:
        u = q.popleft()
        for v, d in adj[u]:
            if parent[v] == 0 and v != parent[u]:
                parent[v] = u
                children[u].append((v, d))
                q.append(v)

    dp = [0] * (n + 1)
    stack = [(1, False)]

    while stack:
        u, processed = stack.pop()
        if not processed:
            stack.append((u, True))
            for v, d in reversed(children[u]):
                stack.append((v, False))
        else:
            sum_children = 0
            for v, d in children[u]:
                sum_children += max(-d, dp[v])
            option1 = (0 - c[u]) + sum_children
            option2 = r[u] + sum_children
            dp[u] = max(option1, option2)

    print(dp[1])

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static class Pair {
        int first;
        int second;
        Pair(int f, int s) {
            first = f;
            second = s;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] r = new int[n + 1];
        int[] c = new int[n + 1];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) r[i] = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) c[i] = Integer.parseInt(st.nextToken());

        List<List<Pair>> adj = new ArrayList<>(n + 1);
        for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int d = Integer.parseInt(st.nextToken());
            adj.get(u).add(new Pair(v, d));
            adj.get(v).add(new Pair(u, d));
        }

        int[] parent = new int[n + 1];
        List<List<Pair>> children = new ArrayList<>(n + 1);
        for (int i = 0; i <= n; i++) children.add(new ArrayList<>());
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        parent[1] = -1;

        while (!q.isEmpty()) {
            int u = q.poll();
            for (Pair edge : adj.get(u)) {
                int v = edge.first;
                int d = edge.second;
                if (parent[v] == 0) {
                    parent[v] = u;
                    children.get(u).add(new Pair(v, d));
                    q.add(v);
                }
            }
        }

        long[] dp = new long[n + 1];
        Deque<Pair> stack = new ArrayDeque<>();
        stack.push(new Pair(1, 0));

        while (!stack.isEmpty()) {
            Pair p = stack.pop();
            int u = p.first;
            int processed = p.second;
            if (processed == 0) {
                stack.push(new Pair(u, 1));
                List<Pair> childList = children.get(u);
                for (int i = childList.size() - 1; i >= 0; i--) {
                    Pair child = childList.get(i);
                    stack.push(new Pair(child.first, 0));
                }
            } else {
                long sum = 0;
                for (Pair child : children.get(u)) {
                    int v = child.first;
                    int d = child.second;
                    sum += Math.max(-d, dp[v]);
                }
                long option1 = (0L - c[u]) + sum;
                long option2 = r[u] + sum;
                dp[u] = Math.max(option1, option2);
            }
        }

        System.out.println(dp[1]);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n;
    cin >> n;
    vector<int> r(n + 1), c(n + 1);
    for (int i = 1; i <= n; ++i) cin >> r[i];
    for (int i = 1; i <= n; ++i) cin >> c[i];

    vector<vector<pair<int, int>>> adj(n + 1);
    for (int i = 0; i < n - 1; ++i) {
        int u, v, d;
        cin >> u >> v >> d;
        adj[u].emplace_back(v, d);
        adj[v].emplace_back(u, d);
    }

    vector<int> parent(n + 1, 0);
    vector<vector<pair<int, int>>> children(n + 1);
    queue<int> q;
    q.push(1);
    parent[1] = -1;

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (auto& edge : adj[u]) {
            int v = edge.first;
            int d = edge.second;
            if (parent[v] == 0) {
                parent[v] = u;
                children[u].emplace_back(v, d);
                q.push(v);
            }
        }
    }

    vector<ll> dp(n + 1);
    stack<pair<int, bool>> stk;
    stk.push({1, false});

    while (!stk.empty()) {
        auto [u, processed] = stk.top();
        stk.pop();
        if (!processed) {
            stk.push({u, true});
            for (auto it = children[u].rbegin(); it != children[u].rend(); ++it) {
                stk.push({it->first, false});
            }
        } else {
            ll sum_children = 0;
            for (auto [v, d] : children[u]) {
                sum_children += max(-1LL * d, dp[v]);
            }
            ll option1 = (0LL - c[u]) + sum_children;
            ll option2 = r[u] + sum_children;
            dp[u] = max(option1, option2);
        }
    }

    cout << dp[1] << endl;
    return 0;
}
```
