## 解题思路

1. **数据读取与解析**  
   从标准输入读取单行 JSON，解析出参考样本集 `train` 和查询样本集 `test`。令样本维度为 $m$，参考样本数为 $n$，查询样本数为 $k$。

2. **初始化中心点**  
   取参考样本集的前 $3$ 个样本作为三个初始中心点 $c_0, c_1, c_2$。

3. **迭代更新（最多 $100$ 轮）**  
   - **分配簇**：对每个参考样本，计算其到当前三个中心点的欧氏距离，将其分配给距离最近的中心点；若距离并列，优先选择编号较小的中心点。  
   - **更新中心点**：对每个中心点，用分配到该中心的所有参考样本的算术平均作为新的中心点；若某中心点未分配到任何样本，则其坐标保持不变。  
   - **收敛判断**：计算本轮与上轮所有中心点各坐标分量的变化量绝对值，若最大值小于 $10^{-6}$，则认为算法已收敛，停止迭代。

4. **中心点排序与重标号**  
   迭代结束后，将三个中心点按坐标字典序升序排列（按照第一维、第二维、……依次比较）。排序后的中心点依次赋予类别编号 $0, 1, 2$。

5. **查询样本归类**  
   用排序后的最终中心点，对每个查询样本计算欧氏距离，选择距离最近的中心点的类别编号作为预测标签（距离并列时取编号最小者）。按输入顺序收集所有预测标签，以 JSON 数组形式输出。

## 复杂度分析

- **时间复杂度**  
  设参考样本数 $n$，特征维度 $m$，查询样本数 $k$，迭代次数 $T \leq 100$。  
  每次迭代中，计算所有参考样本到 $3$ 个中心的欧氏距离需要 $O(n \cdot m \cdot 3) = O(nm)$，更新中心点需要 $O(nm)$，收敛检查 $O(m)$。因此训练阶段复杂度为 $O(Tnm)$。  
  预测阶段计算每个查询样本到 $3$ 个中心的距离为 $O(km)$。  
  总体时间复杂度 $O(Tnm + km)$，实际运行中 $T$ 很小，可近似视为 $O(nm + km)$。

- **空间复杂度**  
  存储参考样本集 $O(nm)$，查询样本集 $O(km)$，中心点 $O(m)$，临时分配标签 $O(n)$。  
  总空间复杂度 $O(nm + km)$。

### Python

```python
import sys
import json
import numpy as np

def solve() -> None:
    data = json.loads(sys.stdin.read())
    train = np.array(data["train"], dtype=float)
    test = np.array(data["test"], dtype=float)
    n, m = train.shape
    k = 3
    max_iter = 100
    tol = 1e-6

    # 1. 初始化：取前3个参考样本作为中心
    cents = train[:k].copy()

    # 2. 迭代
    for _ in range(max_iter):
        # 计算距离并分配簇（并列取下标最小，由argmin保证）
        diff = train[:, None, :] - cents[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        labels = np.argmin(dists, axis=1)

        old_cents = cents.copy()

        # 更新中心点：空簇保持不变
        for i in range(k):
            cluster = train[labels == i]
            if cluster.shape[0] > 0:
                cents[i] = cluster.mean(axis=0)

        # 收敛检查
        if np.max(np.abs(cents - old_cents)) < tol:
            break

    # 3. 中心点排序：字典序升序
    order = np.lexsort(cents.T[::-1])
    cents_sorted = cents[order]

    # 4. 查询样本预测
    diff_test = test[:, None, :] - cents_sorted[None, :, :]
    dists_test = np.linalg.norm(diff_test, axis=2)
    preds = np.argmin(dists_test, axis=1)

    # 输出JSON数组
    sys.stdout.write(json.dumps(preds.tolist()))

if __name__ == "__main__":
    solve()
```

import java.util.*;

 * 需要 org.json 库支持，如 JSONObject, JSONArray

import org.json.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String line = sc.nextLine();
        JSONObject obj = new JSONObject(line);
        JSONArray trainArr = obj.getJSONArray("train");
        JSONArray testArr = obj.getJSONArray("test");

        int n = trainArr.length();
        int m = trainArr.getJSONArray(0).length();
        int k = testArr.length();

        double[][] train = new double[n][m];
        for (int i = 0; i < n; i++) {
            JSONArray row = trainArr.getJSONArray(i);
            for (int j = 0; j < m; j++) {
                train[i][j] = row.getDouble(j);
            }
        }

        double[][] test = new double[k][m];
        for (int i = 0; i < k; i++) {
            JSONArray row = testArr.getJSONArray(i);
            for (int j = 0; j < m; j++) {
                test[i][j] = row.getDouble(j);
            }
        }

        // 初始化中心点：前3个参考样本
        double[][] cents = new double[3][m];
        for (int i = 0; i < 3; i++) {
            System.arraycopy(train[i], 0, cents[i], 0, m);
        }

        int[] labels = new int[n];
        int maxIter = 100;
        double tol = 1e-6;

        // 迭代
        for (int iter = 0; iter < maxIter; iter++) {
            // 2.1 分配簇
            for (int i = 0; i < n; i++) {
                double minDist = Double.MAX_VALUE;
                int best = 0;
                for (int c = 0; c < 3; c++) {
                    double d = 0.0;
                    for (int j = 0; j < m; j++) {
                        double diff = train[i][j] - cents[c][j];
                        d += diff * diff;
                    }
                    d = Math.sqrt(d);
                    if (d < minDist) {
                        minDist = d;
                        best = c;
                    }
                }
                labels[i] = best;
            }

            // 2.2 备份旧中心
            double[][] oldCents = new double[3][m];
            for (int c = 0; c < 3; c++) {
                System.arraycopy(cents[c], 0, oldCents[c], 0, m);
            }

            // 2.3 更新中心点
            int[] cnt = new int[3];
            double[][] sum = new double[3][m];
            for (int i = 0; i < n; i++) {
                int cl = labels[i];
                cnt[cl]++;
                for (int j = 0; j < m; j++) {
                    sum[cl][j] += train[i][j];
                }
            }
            for (int c = 0; c < 3; c++) {
                if (cnt[c] > 0) {
                    for (int j = 0; j < m; j++) {
                        cents[c][j] = sum[c][j] / cnt[c];
                    }
                }
            }

            // 2.4 收敛检查
            double maxDiff = 0.0;
            for (int c = 0; c < 3; c++) {
                for (int j = 0; j < m; j++) {
                    maxDiff = Math.max(maxDiff, Math.abs(cents[c][j] - oldCents[c][j]));
                }
            }
            if (maxDiff < tol) break;
        }

        // 3. 中心点排序：字典序升序
        Integer[] order = {0, 1, 2};
        Arrays.sort(order, (a, b) -> {
            for (int j = 0; j < m; j++) {
                if (cents[a][j] != cents[b][j]) {
                    return Double.compare(cents[a][j], cents[b][j]);
                }
            }
            return 0;
        });
        double[][] sortedCents = new double[3][m];
        for (int i = 0; i < 3; i++) {
            sortedCents[i] = cents[order[i]];
        }

        // 4. 查询样本预测
        int[] preds = new int[k];
        for (int i = 0; i < k; i++) {
            double minDist = Double.MAX_VALUE;
            int best = 0;
            for (int c = 0; c < 3; c++) {
                double d = 0.0;
                for (int j = 0; j < m; j++) {
                    double diff = test[i][j] - sortedCents[c][j];
                    d += diff * diff;
                }
                d = Math.sqrt(d);
                if (d < minDist) {
                    minDist = d;
                    best = c;
                }
            }
            preds[i] = best;
        }

        // 输出JSON数组
        System.out.print("[");
        for (int i = 0; i < k; i++) {
            if (i > 0) System.out.print(",");
            System.out.print(preds[i]);
        }
        System.out.println("]");
    }
}

#include <iostream>
#include <vector>
#include <string>
#include <cmath>
#include <algorithm>
#include "json.hpp"  // nlohmann/json 单头文件

using json = nlohmann::json;

int main() {
    std::string line;
    std::getline(std::cin, line);
    json data = json::parse(line);

    auto train = data["train"].get<std::vector<std::vector<double>>>();
    auto test  = data["test"].get<std::vector<std::vector<double>>>();

    int n = train.size();
    int m = train[0].size();
    int k = test.size();
    const int K = 3;
    const int maxIter = 100;
    const double tol = 1e-6;

    // 初始化中心点：前3个参考样本
    std::vector<std::vector<double>> cents(K, std::vector<double>(m));
    for (int i = 0; i < K; ++i) {
        cents[i] = train[i];
    }

    std::vector<int> labels(n);

    // 迭代
    for (int iter = 0; iter < maxIter; ++iter) {
        // 2.1 分配簇
        for (int i = 0; i < n; ++i) {
            double minDist = 1e100;
            int best = 0;
            for (int c = 0; c < K; ++c) {
                double d = 0.0;
                for (int j = 0; j < m; ++j) {
                    double diff = train[i][j] - cents[c][j];
                    d += diff * diff;
                }
                d = std::sqrt(d);
                if (d < minDist) {
                    minDist = d;
                    best = c;
                }
            }
            labels[i] = best;
        }

        // 2.2 备份旧中心
        auto oldCents = cents;

        // 2.3 更新中心点
        std::vector<int> cnt(K, 0);
        std::vector<std::vector<double>> sum(K, std::vector<double>(m, 0.0));
        for (int i = 0; i < n; ++i) {
            int cl = labels[i];
            cnt[cl]++;
            for (int j = 0; j < m; ++j) {
                sum[cl][j] += train[i][j];
            }
        }
        for (int c = 0; c < K; ++c) {
            if (cnt[c] > 0) {
                for (int j = 0; j < m; ++j) {
                    cents[c][j] = sum[c][j] / cnt[c];
                }
            }
        }

        // 2.4 收敛检查
        double maxDiff = 0.0;
        for (int c = 0; c < K; ++c) {
            for (int j = 0; j < m; ++j) {
                maxDiff = std::max(maxDiff, std::abs(cents[c][j] - oldCents[c][j]));
            }
        }
        if (maxDiff < tol) break;
    }

    // 3. 中心点排序：字典序升序
    std::vector<int> order = {0, 1, 2};
    std::sort(order.begin(), order.end(), [&](int a, int b) {
        for (int j = 0; j < m; ++j) {
            if (cents[a][j] != cents[b][j]) {
                return cents[a][j] < cents[b][j];
            }
        }
        return false;
    });
    std::vector<std::vector<double>> sortedCents(K, std::vector<double>(m));
    for (int i = 0; i < K; ++i) {
        sortedCents[i] = cents[order[i]];
    }

    // 4. 查询样本预测
    std::vector<int> preds(k);
    for (int i = 0; i < k; ++i) {
        double minDist = 1e100;
        int best = 0;
        for (int c = 0; c < K; ++c) {
            double d = 0.0;
            for (int j = 0; j < m; ++j) {
                double diff = test[i][j] - sortedCents[c][j];
                d += diff * diff;
            }
            d = std::sqrt(d);
            if (d < minDist) {
                minDist = d;
                best = c;
            }
        }
        preds[i] = best;
    }

    // 输出JSON数组
    std::cout << "[";
    for (int i = 0; i < k; ++i) {
        if (i > 0) std::cout << ",";
        std::cout << preds[i];
    }
    std::cout << "]" << std::endl;

    return 0;
}
