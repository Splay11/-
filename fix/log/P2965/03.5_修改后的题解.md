## 解题思路

本题需要统计在一棵树上恰好关闭 $k$ 条边后，所有连通块内鸽子数之和均为偶数的方案数。算法核心思路如下：

1. **可行性判断**  
   首先计算所有广场的鸽子总数 $S = \sum_{i=1}^{n} a_i$。若 $S$ 为奇数，则无论怎样划分，至少会有一个连通块的和为奇数，此时对所有 $k$ 方案数均为 $0$，可直接输出。

2. **子树和与可关闭边**  
   若 $S$ 为偶数，选取广场 $1$ 作为根节点进行深度优先遍历。在遍历过程中，维护以每个节点 $u$ 为根的子树中鸽子总数 $sum_u$。  
   如果某个非根节点 $u$ 满足 $sum_u$ 为偶数，则连接 $u$ 与其父节点的边被称为“可关闭边”。  
   **原因**：切断该边后，子树是偶数，剩余部分为 $S - sum_u$，由于 $S$ 和 $sum_u$ 均为偶数，剩余部分也必定是偶数。

3. **可关闭边的独立性**  
   可以证明，所有“可关闭边”之间是相互独立的：任意选择若干条可关闭边同时切断，都能保证产生的所有连通块鸽子总数均为偶数。  
   这是因为关闭任意一条偶数子树对应的边，都不会影响其他子树和的奇偶性。

4. **组合计数**  
   设可关闭边的总数为 $m$。对于每个 $k\ (1 \le k \le n-1)$：  
   - 若 $k \le m$，则方案数为从 $m$ 条边中恰好选择 $k$ 条的组合数 $C(m, k)$；  
   - 若 $k > m$，方案数为 $0$。  
   
   计算组合数时，需要预处理阶乘与阶乘的逆元，利用费马小定理在 $O(1)$ 时间内查询。最终答案对 $10^9+7$ 取模。

## 复杂度分析

- **时间复杂度**：$O(n)$。DFS 遍历一次整棵树计算子树和并统计可关闭边数量，预处理阶乘与逆元也需要 $O(n)$ 时间，输出 $n-1$ 个组合数每次 $O(1)$，总复杂度为线性。
- **空间复杂度**：$O(n)$。需要存储邻接表、权值数组、子树和数组以及阶乘/逆元数组。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(10**7)
MOD = 10**9 + 7

n = int(sys.stdin.readline())
a = [0] + list(map(int, sys.stdin.readline().split()))
g = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    u, v = map(int, sys.stdin.readline().split())
    g[u].append(v)
    g[v].append(u)

subsum = [0] * (n + 1)
m = 0

def dfs(u, fa):
    global m
    subsum[u] = a[u]
    for v in g[u]:
        if v == fa:
            continue
        dfs(v, u)
        subsum[u] += subsum[v]
    # 非根节点且子树和为偶数，则该边可关闭
    if fa != 0 and subsum[u] % 2 == 0:
        m += 1

dfs(1, 0)

# 如果总和为奇数，所有 k 的方案数均为 0
if subsum[1] % 2 == 1:
    print(' '.join(['0'] * (n - 1)))
    sys.exit(0)

# 预处理阶乘和逆元
fact = [1] * (n + 1)
invfact = [1] * (n + 1)
for i in range(1, n + 1):
    fact[i] = fact[i - 1] * i % MOD
invfact[n] = pow(fact[n], MOD - 2, MOD)
for i in range(n, 0, -1):
    invfact[i - 1] = invfact[i] * i % MOD

def C(n_, k_):
    if k_ < 0 or k_ > n_:
        return 0
    return fact[n_] * invfact[k_] % MOD * invfact[n_ - k_] % MOD

res = [str(C(m, k)) for k in range(1, n)]
print(' '.join(res))
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final int MOD = 1000000007;
    static int n;
    static long[] a, subsum;
    static List<Integer>[] g;
    static int m = 0;

    static long modPow(long base, long exp) {
        long res = 1;
        while (exp > 0) {
            if ((exp & 1) == 1) res = res * base % MOD;
            base = base * base % MOD;
            exp >>= 1;
        }
        return res;
    }

    static long[] fact, invfact;
    static void initFact(int N) {
        fact = new long[N + 1];
        invfact = new long[N + 1];
        fact[0] = 1;
        for (int i = 1; i <= N; i++) fact[i] = fact[i - 1] * i % MOD;
        invfact[N] = modPow(fact[N], MOD - 2);
        for (int i = N; i >= 1; i--) invfact[i - 1] = invfact[i] * i % MOD;
    }

    static long C(int n, int k) {
        if (k < 0 || k > n) return 0;
        return fact[n] * invfact[k] % MOD * invfact[n - k] % MOD;
    }

    static void dfs(int u, int fa) {
        subsum[u] = a[u];
        for (int v : g[u]) {
            if (v == fa) continue;
            dfs(v, u);
            subsum[u] += subsum[v];
        }
        if (fa != 0 && subsum[u] % 2 == 0) m++;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        a = new long[n + 1];
        subsum = new long[n + 1];
        g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
        String[] parts = br.readLine().split(" ");
        for (int i = 1; i <= n; i++) a[i] = Long.parseLong(parts[i - 1]);
        for (int i = 1; i < n; i++) {
            parts = br.readLine().split(" ");
            int u = Integer.parseInt(parts[0]);
            int v = Integer.parseInt(parts[1]);
            g[u].add(v);
            g[v].add(u);
        }
        dfs(1, 0);
        if (subsum[1] % 2 == 1) {
            for (int k = 1; k < n; k++) {
                System.out.print(0);
                if (k + 1 < n) System.out.print(" ");
            }
            return;
        }
        initFact(n);
        for (int k = 1; k < n; k++) {
            System.out.print(C(m, k));
            if (k + 1 < n) System.out.print(" ");
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 100000;
const int MOD = 1000000007;

int n;
vector<int> g[MAXN + 1];
ll a[MAXN + 1], subsum[MAXN + 1];
int m = 0;

ll modexp(ll base, ll exp) {
    ll res = 1;
    while (exp) {
        if (exp & 1) res = res * base % MOD;
        base = base * base % MOD;
        exp >>= 1;
    }
    return res;
}

ll fact[MAXN + 1], invfact[MAXN + 1];

void init_fact(int N) {
    fact[0] = 1;
    for (int i = 1; i <= N; ++i) fact[i] = fact[i - 1] * i % MOD;
    invfact[N] = modexp(fact[N], MOD - 2);
    for (int i = N; i > 0; --i) invfact[i - 1] = invfact[i] * i % MOD;
}

ll C(int n, int k) {
    if (k < 0 || k > n) return 0;
    return fact[n] * invfact[k] % MOD * invfact[n - k] % MOD;
}

void dfs(int u, int fa) {
    subsum[u] = a[u];
    for (int v : g[u]) {
        if (v == fa) continue;
        dfs(v, u);
        subsum[u] += subsum[v];
    }
    if (fa != 0 && subsum[u] % 2 == 0) ++m;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 0);
    if (subsum[1] % 2 == 1) {
        for (int k = 1; k < n; ++k) {
            cout << 0;
            if (k + 1 < n) cout << ' ';
        }
        return 0;
    }
    init_fact(n);
    for (int k = 1; k < n; ++k) {
        cout << C(m, k);
        if (k + 1 < n) cout << ' ';
    }
    return 0;
}
```
