## 解题思路

满足条件的一组导览点构成轴对齐直角三角形：直角顶点与另外两个顶点分别同行、同列，且三个格子上的字符恰好是 `t`、`z`、`g` 各一个。

因此可以先统计每个字符在每一行、每一列出现的次数，再枚举直角顶点所在格子：

- 若该格为 `t`，则把「同行 `z` 与同列 `g`」以及「同行 `g` 与同列 `z`」两两配对；
- 若该格为 `z` 或 `g`，对另外两个字母做同样的配对。

每个合法三角形恰有一个直角顶点，因此这样计数不会重复，也不会漏计。

## 复杂度分析

时间复杂度 $O(nm)$，空间复杂度 $O(n+m)$。

## 代码实现

### Python

```python
n, m = map(int, input().split())

s = []
for _ in range(n):
    s.append(input())

sumt = [[0] * n, [0] * m]
sumz = [[0] * n, [0] * m]
sumg = [[0] * n, [0] * m]

for i in range(n):
    for j in range(m):
        if s[i][j] == 't':
            sumt[0][i] += 1
            sumt[1][j] += 1
        if s[i][j] == 'z':
            sumz[0][i] += 1
            sumz[1][j] += 1
        if s[i][j] == 'g':
            sumg[0][i] += 1
            sumg[1][j] += 1

ans = 0
for i in range(n):
    for j in range(m):
        if s[i][j] == 't':
            ans += sumz[0][i] * sumg[1][j] + sumz[1][j] * sumg[0][i]
        if s[i][j] == 'z':
            ans += sumt[0][i] * sumg[1][j] + sumt[1][j] * sumg[0][i]
        if s[i][j] == 'g':
            ans += sumz[0][i] * sumt[1][j] + sumz[1][j] * sumt[0][i]

print(ans)
```

### Java

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine();

        String[] s = new String[n];
        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextLine();
        }

        long[][] sumt = new long[2][1005];
        long[][] sumz = new long[2][1005];
        long[][] sumg = new long[2][1005];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                char c = s[i].charAt(j);
                if (c == 't') {
                    sumt[0][i]++;
                    sumt[1][j]++;
                } else if (c == 'z') {
                    sumz[0][i]++;
                    sumz[1][j]++;
                } else if (c == 'g') {
                    sumg[0][i]++;
                    sumg[1][j]++;
                }
            }
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                char c = s[i].charAt(j);
                if (c == 't') {
                    ans += sumz[0][i] * sumg[1][j] + sumz[1][j] * sumg[0][i];
                } else if (c == 'z') {
                    ans += sumt[0][i] * sumg[1][j] + sumt[1][j] * sumg[0][i];
                } else if (c == 'g') {
                    ans += sumz[0][i] * sumt[1][j] + sumz[1][j] * sumt[0][i];
                }
            }
        }

        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long
const int M = 1005;

string s[M];
ll sumt[2][M], sumz[2][M], sumg[2][M];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (s[i][j] == 't') sumt[0][i]++, sumt[1][j]++;
            if (s[i][j] == 'z') sumz[0][i]++, sumz[1][j]++;
            if (s[i][j] == 'g') sumg[0][i]++, sumg[1][j]++;
        }
    }
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (s[i][j] == 't')
                ans += sumz[0][i] * sumg[1][j] + sumz[1][j] * sumg[0][i];
            if (s[i][j] == 'z')
                ans += sumt[0][i] * sumg[1][j] + sumt[1][j] * sumg[0][i];
            if (s[i][j] == 'g')
                ans += sumz[0][i] * sumt[1][j] + sumz[1][j] * sumt[0][i];
        }
    }
    cout << ans << endl;
    return 0;
}
```
