## 解题思路

题目给出了 $n$ 台仪器，以及 $m$ 条形如“仪器 $i$ 的读数比仪器 $j$ 高 $k$”的偏差关系。我们需要为每台仪器分配一个正整数读数 $x_i$（$1\le x_i\le 10^{18}$），使得所有关系同时成立，或者判断无解。

这类问题本质上是**带权图的连通块遍历与差分约束验证**，具体步骤如下：

1. **建图**  
   将每台仪器看作一个节点。对于每条关系“仪器 $i$ 比仪器 $j$ 高 $k$”，即 $x_i - x_j = k$，可改写为 $x_i = x_j + k$。  
   因此在图上添加两条有向边：  
   - 从 $j$ 到 $i$，边权为 $+k$（表示 $x_i = x_j + k$）  
   - 从 $i$ 到 $j$，边权为 $-k$（表示 $x_j = x_i - k$）  
   这样整张图等价于一个带权无向图。

2. **连通块内计算相对读数**  
   对于每个连通块，任选一个起点 $s$，设其相对读数为 $d_s = 0$。  
   使用 BFS（或 DFS）遍历整个块：  
   - 对于边 $u\to v$，权值为 $w$，若已知 $d_u$，则应有 $d_v = d_u + w$。  
   - 若 $v$ 未被访问过，则赋值 $d_v = d_u + w$ 并入队；  
   - 若 $v$ 已访问，则必须满足已有的 $d_v$ 与 $d_u + w$ 相等，否则说明约束存在矛盾，直接判定无解。

3. **通过平移满足范围限制**  
   在一个连通块中，所有仪器的真实读数可以由相对读数整体平移得到：$x_i = d_i + C$，其中 $C$ 是一个整数。  
   要求 $1 \le x_i \le 10^{18}$ 对所有点成立，等价于对整个块取满足条件的 $C$ 的交集：  
   - 下界：$C \ge 1 - \min\{d_i\}$  
   - 上界：$C \le 10^{18} - \max\{d_i\}$  
   只有当 $1 - \min\{d_i\} \le 10^{18} - \max\{d_i\}$ 时存在合法的 $C$，即  
   $$\max\{d_i\} - \min\{d_i\} \le 10^{18} - 1$$  
   若该不等式成立，我们可以令 $C = 1 - \min\{d_i\}$，这样连通块中的最小读数恰好变为 $1$，同时最大读数也不会超过 $10^{18}$。  
   按此计算该块每台仪器的最终答案 $x_i$；若任何连通块无法找到合法 $C$，则同样输出 `-1`。

4. **多组数据处理**  
   对每组测试数据独立建图、遍历，每个连通块单独处理并验证，一旦发现矛盾或范围不满足就返回 `-1`，否则输出 $n$ 个读数。

## 复杂度分析

设一组数据有 $n$ 台仪器，$m$ 条偏差关系。  
- 时间复杂度：建图耗时 $O(m)$，BFS/DFS 遍历每个节点和每条边各一次，总复杂度 $O(n+m)$。  
  题目保证所有测试数据中 $\sum n, \sum m \le 5\times 10^5$，总复杂度完全可行。  
- 空间复杂度：邻接表、距离数组、访问标记、答案数组均为线性大小，空间复杂度 $O(n+m)$。

## 代码实现

### Python

```python
import sys
from collections import deque

LIMIT = 10 ** 18

def solve_case(n, m, edges):
    # g[u] 中存储 (v, w)，表示 x[v] = x[u] + w
    g = [[] for _ in range(n + 1)]
    for i, j, k in edges:
        g[j].append((i, k))
        g[i].append((j, -k))

    dist = [0] * (n + 1)      # 相对读数
    visited = [False] * (n + 1)
    ans = [0] * (n + 1)

    for start in range(1, n + 1):
        if visited[start]:
            continue

        # BFS 遍历当前连通块
        q = deque([start])
        visited[start] = True
        dist[start] = 0

        comp = []
        mn = 0
        mx = 0

        while q:
            u = q.popleft()
            comp.append(u)

            if dist[u] < mn:
                mn = dist[u]
            if dist[u] > mx:
                mx = dist[u]

            for v, w in g[u]:
                nd = dist[u] + w
                if not visited[v]:
                    visited[v] = True
                    dist[v] = nd
                    q.append(v)
                else:
                    # 已访问的点必须满足相对值一致
                    if dist[v] != nd:
                        return None

        # 平移使最小读数变为 1
        add = 1 - mn
        if mx + add > LIMIT:
            return None

        for u in comp:
            ans[u] = dist[u] + add

    return ans[1:]

def main():
    input = sys.stdin.readline
    T = int(input())
    out = []

    for _ in range(T):
        n, m = map(int, input().split())
        edges = []
        for _ in range(m):
            i, j, k = map(int, input().split())
            edges.append((i, j, k))

        res = solve_case(n, m, edges)
        if res is None:
            out.append("-1")
        else:
            out.append(" ".join(map(str, res)))

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long LIMIT = (long) 1e18;

    static class Edge {
        int to;
        long w;

        Edge(int to, long w) {
            this.to = to;
            this.w = w;
        }
    }

    static class InputReader {
        BufferedReader br;
        StringTokenizer st;

        InputReader() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() throws IOException {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(br.readLine());
            }
            return st.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        long nextLong() throws IOException {
            return Long.parseLong(next());
        }
    }

    static long[] solveCase(int n, List<Edge>[] g) {
        long[] dist = new long[n + 1];
        boolean[] visited = new boolean[n + 1];
        long[] ans = new long[n + 1];

        ArrayDeque<Integer> queue = new ArrayDeque<>();

        for (int start = 1; start <= n; start++) {
            if (visited[start]) {
                continue;
            }

            List<Integer> comp = new ArrayList<>();
            queue.clear();
            queue.offer(start);
            visited[start] = true;
            dist[start] = 0;

            long mn = 0;
            long mx = 0;

            while (!queue.isEmpty()) {
                int u = queue.poll();
                comp.add(u);

                if (dist[u] < mn) {
                    mn = dist[u];
                }
                if (dist[u] > mx) {
                    mx = dist[u];
                }

                for (Edge e : g[u]) {
                    int v = e.to;
                    long nd = dist[u] + e.w;

                    if (!visited[v]) {
                        visited[v] = true;
                        dist[v] = nd;
                        queue.offer(v);
                    } else {
                        // 已访问的点必须满足相对值一致
                        if (dist[v] != nd) {
                            return null;
                        }
                    }
                }
            }

            // 平移使最小读数变为 1
            long add = 1 - mn;
            if (mx + add > LIMIT) {
                return null;
            }

            for (int u : comp) {
                ans[u] = dist[u] + add;
            }
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        InputReader in = new InputReader();
        StringBuilder sb = new StringBuilder();

        int T = in.nextInt();
        while (T-- > 0) {
            int n = in.nextInt();
            int m = in.nextInt();

            List<Edge>[] g = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) {
                g[i] = new ArrayList<>();
            }

            for (int t = 0; t < m; t++) {
                int i = in.nextInt();
                int j = in.nextInt();
                long k = in.nextLong();

                g[j].add(new Edge(i, k));
                g[i].add(new Edge(j, -k));
            }

            long[] res = solveCase(n, g);
            if (res == null) {
                sb.append("-1\n");
            } else {
                for (int i = 1; i <= n; i++) {
                    if (i > 1) {
                        sb.append(' ');
                    }
                    sb.append(res[i]);
                }
                sb.append('\n');
            }
        }

        System.out.print(sb.toString());
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const long long LIMIT = (long long)1e18;

struct Edge {
    int to;
    long long w;
};

// 返回 ans[1..n]，无解返回空 vector
vector<long long> solve_case(int n, const vector<vector<Edge>>& g) {
    vector<long long> dist(n + 1, 0);
    vector<int> visited(n + 1, 0);
    vector<long long> ans(n + 1, 0);

    queue<int> q;

    for (int start = 1; start <= n; start++) {
        if (visited[start]) {
            continue;
        }

        vector<int> comp;
        visited[start] = 1;
        dist[start] = 0;
        q.push(start);

        long long mn = 0;
        long long mx = 0;

        while (!q.empty()) {
            int u = q.front();
            q.pop();
            comp.push_back(u);

            if (dist[u] < mn) {
                mn = dist[u];
            }
            if (dist[u] > mx) {
                mx = dist[u];
            }

            for (const Edge& e : g[u]) {
                int v = e.to;
                long long nd = dist[u] + e.w;

                if (!visited[v]) {
                    visited[v] = 1;
                    dist[v] = nd;
                    q.push(v);
                } else {
                    // 已访问的点必须满足相对值一致
                    if (dist[v] != nd) {
                        return {};
                    }
                }
            }
        }

        // 平移使最小读数变为 1
        long long add = 1 - mn;
        if (mx + add > LIMIT) {
            return {};
        }

        for (int u : comp) {
            ans[u] = dist[u] + add;
        }
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, m;
        cin >> n >> m;

        vector<vector<Edge>> g(n + 1);

        for (int t = 0; t < m; t++) {
            int i, j;
            long long k;
            cin >> i >> j >> k;

            g[j].push_back({i, k});
            g[i].push_back({j, -k});
        }

        vector<long long> res = solve_case(n, g);

        if (res.empty()) {
            cout << -1 << '\n';
        } else {
            for (int i = 1; i <= n; i++) {
                if (i > 1) {
                    cout << ' ';
                }
                cout << res[i];
            }
            cout << '\n';
        }
    }

    return 0;
}
```
