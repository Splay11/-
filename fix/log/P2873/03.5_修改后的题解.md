## 解题思路

1. **统计频次**：  
   使用哈希表（或字典）统计序列 $a$ 中每个整数 $x$ 出现的次数，记为 $\mathrm{cnt}[x]$。

2. **枚举彩虹阶梯的起始值**：  
   一个长度为 $5$ 的子序列要成为“彩虹阶梯”，其五个元素的值经过排序后必须是连续的五个整数，即存在某个整数 $v$，使得这五个元素的值恰好为  
   $$v,\ v+1,\ v+2,\ v+3,\ v+4$$  
   因此，对于每个可能的起始值 $v$，若 $\mathrm{cnt}[v],\ \mathrm{cnt}[v+1],\ \mathrm{cnt}[v+2],\ \mathrm{cnt}[v+3],\ \mathrm{cnt}[v+4]$ 均大于 $0$，则从原序列中选出这五个值的子序列的方案数为它们的乘积：
   $$\mathrm{cnt}[v] \times \mathrm{cnt}[v+1] \times \mathrm{cnt}[v+2] \times \mathrm{cnt}[v+3] \times \mathrm{cnt}[v+4]$$

3. **累加并取模**：  
   将所有满足条件的 $v$ 对应的乘积累加起来，结果对 $10^9+7$ 取模，即可得到答案。

## 复杂度分析

- **时间复杂度**：$O(n)$。仅需一次遍历统计频次，之后遍历哈希表中的所有键，每次进行常数次查询和乘法运算。
- **空间复杂度**：$O(n)$。哈希表在最坏情况下需要存储所有不同元素及其频次。

## 代码实现

### Python

```python
import sys

def main():
    MOD = 10**9 + 7
    data = sys.stdin.read().split()
    n = int(data[0])                     # 序列长度
    a = list(map(int, data[1:1 + n]))   # 读取序列 a
    
    # 统计每个值的出现次数
    cnt = {}
    for x in a:
        cnt[x] = cnt.get(x, 0) + 1
    
    ans = 0
    # 枚举彩虹阶梯可能的起始值 v
    for v, c0 in cnt.items():
        c1 = cnt.get(v + 1, 0)
        c2 = cnt.get(v + 2, 0)
        c3 = cnt.get(v + 3, 0)
        c4 = cnt.get(v + 4, 0)
        # 只有五个值都出现时才有贡献
        if c1 and c2 and c3 and c4:
            prod = c0 * c1 % MOD
            prod = prod * c2 % MOD
            prod = prod * c3 % MOD
            prod = prod * c4 % MOD
            ans = (ans + prod) % MOD
    
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.HashMap;

public class Main {
    static final int MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        String[] parts = br.readLine().split(" ");

        // 统计每个值的出现次数
        HashMap<Long, Long> cnt = new HashMap<>();
        for (int i = 0; i < n; i++) {
            long x = Long.parseLong(parts[i]);
            cnt.put(x, cnt.getOrDefault(x, 0L) + 1);
        }

        long ans = 0;
        // 枚举彩虹阶梯可能的起始值 v
        for (long v : cnt.keySet()) {
            long c0 = cnt.get(v);
            Long c1 = cnt.get(v + 1);
            Long c2 = cnt.get(v + 2);
            Long c3 = cnt.get(v + 3);
            Long c4 = cnt.get(v + 4);
            // 只有五个值都出现时才计算乘积
            if (c1 != null && c2 != null && c3 != null && c4 != null) {
                long prod = c0;
                prod = (prod * c1) % MOD;
                prod = (prod * c2) % MOD;
                prod = (prod * c3) % MOD;
                prod = (prod * c4) % MOD;
                ans = (ans + prod) % MOD;
            }
        }

        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
static const int MOD = 1e9 + 7;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;                              // 序列长度
    unordered_map<long long, long long> cnt;
    cnt.reserve(n * 2);

    // 统计每个值的出现次数
    for (int i = 0; i < n; ++i) {
        long long x;
        cin >> x;
        cnt[x]++;
    }

    long long ans = 0;
    // 枚举彩虹阶梯可能的起始值 v
    for (auto &p : cnt) {
        long long v = p.first;
        // 检查 v, v+1, v+2, v+3, v+4 是否都存在
        if (cnt.count(v + 1) && cnt.count(v + 2) &&
            cnt.count(v + 3) && cnt.count(v + 4)) {
            long long prod = p.second;
            prod = (prod * cnt[v + 1]) % MOD;
            prod = (prod * cnt[v + 2]) % MOD;
            prod = (prod * cnt[v + 3]) % MOD;
            prod = (prod * cnt[v + 4]) % MOD;
            ans = (ans + prod) % MOD;
        }
    }

    cout << ans << "\n";
    return 0;
}
```
