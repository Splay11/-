## 题目思路

本题需要前置知识[最小生成树](https://oi-wiki.org/graph/mst/)。首先判断连通块数量可以通过并查集的方法。考虑连通块为2和1的情况：

1. 连通块数量为2时，此时恰好有两个连通块，所以只需要维护目前的连通块数量，无向图保持连通的最小情况为树，所以等价于求每个连通块的最小生成树，求出生成树后累计边权值，用总边权值减去生成树的边权值即为答案。
2. 连通块数量为1时，此时只有一个连通块，同样考虑构建最小生成树，在最小生成树上去除一条边后即可以分为两个连通块，所以只需要找到最小生成树上的最大边权值减去即可。不存在会比这个方法更优的解法，反证法，假设有两棵树通过一条非常大的边连接，去除这个边之后总的边权会比最小生成树去除最大边权的总边权更小，然而我们考虑最小生成树的过程，我们完全可以改变边的遍历顺序将这两棵树的边先进行合并，所以这种情况的两棵树的权值之和也一定等于总的最小生成树分裂成对应两个子树的权值之和，所以这种情况不会比直接求最小生成树去除最大边更优。

时间复杂度为Kruskal算法的时间复杂度$O(mlogm)$。

**C++**

~~~c++
// ac代码
// kizk
#include<bits/stdc++.h>

using namespace std;

#define int long long
signed main (){
    ios::sync_with_stdio(false);
    std::cin.tie(0);
    int n, m; cin >> n >> m;
    int su = 0;
    vector<array<int, 3>> v;
    for(int a, b, c; cin >> a >> b >> c; )v.push_back({a, b, c}), su += c;
    vector<int> p(n + 1);
    iota(p.begin(), p.end(), 0);
    function<int(int)> find = [&](int x) -> int{
        return p[x] ^ x ? p[x] = find(p[x]) : x;
    };
    sort(v.begin(), v.end(), [&](array<int, 3>& a, array<int, 3>& b){
        return a[2] < b[2];
    });
    int cnt = n;
    int maxv = 0;
    for(auto&t : v)
    {
        int a = t[0], b = t[1], c = t[2];
        a = find(a), b = find(b);
        if(a ^ b)
        {
            p[a] = b;
            su -= c;
            cnt -- ;
            maxv = max(maxv, c);
        }
    }
    if(cnt == 1)
        cout << su + maxv << "\n";
    else if(cnt == 2)
        cout << su << "\n";

    else
        cout << -1 << "\n";
    return 0;
}
~~~

### java
```java
import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        long totalCost = 0;
        List<int[]> edges = new ArrayList<>();

        // 读取边的信息并计算总的边权和
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            int c = scanner.nextInt();
            edges.add(new int[]{a, b, c});
            totalCost += c;
        }

        // 初始化并查集
        int[] parent = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
        }

        // 边按照权重升序排序
        edges.sort(Comparator.comparingInt(e -> e[2]));

        int components = n;
        long maxEdgeWeight = 0;

        for (int[] edge : edges) {
            int a = edge[0];
            int b = edge[1];
            int c = edge[2];

            int rootA = find(parent, a);
            int rootB = find(parent, b);

            if (rootA != rootB) {
                parent[rootA] = rootB;
                totalCost -= c;
                components--;
                maxEdgeWeight = Math.max(maxEdgeWeight, c);
            }
        }

        if (components == 1) {
            System.out.println(totalCost + maxEdgeWeight);
        } else if (components == 2) {
            System.out.println(totalCost);
        } else {
            System.out.println(-1);
        }

        scanner.close();
    }

    private static int find(int[] parent, int x) {
        if (parent[x] != x) {
            parent[x] = find(parent, parent[x]);
        }
        return parent[x];
    }
}
```

### Python
```Python
n, m = map(int, input().split())

def find(p, x):
    if p[x] != x:
        p[x] = find(p, p[x])
    return p[x]

p = list(range(n + 1))
total = 0
cnt = n
roads = []
max_W = 0
for _ in range(m):
    u, v, w = map(int, input().split())
    total += w
    roads.append((u, v, w))

roads.sort(key = lambda x: x[2])
for r in roads:
    u, v, w = r
    u = find(p, u)
    v = find(p, v)
    if u != v:
        p[u] = v
        cnt -= 1
        total -= w
        max_W = max(max_W, w)

if cnt > 2:
    print(-1)
elif cnt == 2:
    print(total)
else:
    print(total + max_W)
    ```
