## 解题思路

给定五个仅含字符 `0` 到 `9` 的字符串 $a,b,c,d,e$。在进制 $x\in[2,10]$ 中，当且仅当五个串的每一位数字都小于 $x$ 时，$x$ 合法。若在该进制下 $A+B=C$（按进制 $x$ 加法，值用十进制表示），则令 $S_x$ 为把 $d$、$e$ 按进制 $x$ 解释后的十进制之和。所有满足条件的 $S_x$ 若完全一致则输出该值，否则输出 `baka`；一个都不存在时也输出 `baka`。

算法：先求五个串中的最大数字 $mx$，合法进制下界为 $\max(2, mx+1)$。对每个候选进制 $x$：逐位把串转成十进制（$v \leftarrow v\cdot x + \mathrm{digit}$）；若 $A+B=C$，计算 $S_x=D+E$。用变量记录首次出现的和，之后若出现不同的和则判定不唯一。

扫描结束：至少有一个进制通过且所有和相同则输出该和，否则输出 `baka`。

## 复杂度分析

单个字符串长度不超过 $9$。每组至多枚举 $9$ 个进制、解析 $5$ 个串，时间复杂度 $O(\mathrm{len})$，空间复杂度 $O(1)$。

## 代码实现

### Python

```python
import sys

def parse_in_base(s, x):
    v = 0
    for ch in s:
        d = ord(ch) - ord('0')
        if d >= x:
            return None
        v = v * x + d
    return v

def solve_case(a, b, c, d, e):
    mx_digit = max(max(ord(ch) - 48 for ch in s) for s in (a, b, c, d, e))
    start_base = max(2, mx_digit + 1)

    ans = None
    ok = False
    for x in range(start_base, 11):
        A = parse_in_base(a, x)
        B = parse_in_base(b, x)
        C = parse_in_base(c, x)
        if A is None or B is None or C is None:
            continue
        if A + B == C:
            D = parse_in_base(d, x)
            E = parse_in_base(e, x)
            if D is None or E is None:
                continue
            S = D + E
            if not ok:
                ans = S
                ok = True
            elif S != ans:
                return "baka"
    return str(ans) if ok else "baka"

def main():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    t = int(next(it))
    out = []
    for _ in range(t):
        a = next(it); b = next(it); c = next(it); d = next(it); e = next(it)
        out.append(solve_case(a, b, c, d, e))
    print("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {

    static long parseInBase(String s, int x) {
        long v = 0L;
        for (int i = 0; i < s.length(); i++) {
            int d = s.charAt(i) - '0';
            if (d >= x) return -1L;
            v = v * x + d;
        }
        return v;
    }

    static String solveCase(String a, String b, String c, String d, String e) {
        int mxDigit = 0;
        String[] arr = {a, b, c, d, e};
        for (String s : arr) {
            for (int i = 0; i < s.length(); i++) {
                mxDigit = Math.max(mxDigit, s.charAt(i) - '0');
            }
        }
        int startBase = Math.max(2, mxDigit + 1);

        boolean ok = false;
        long ans = -1;

        for (int x = startBase; x <= 10; x++) {
            long A = parseInBase(a, x);
            long B = parseInBase(b, x);
            long C = parseInBase(c, x);
            if (A < 0 || B < 0 || C < 0) continue;
            if (A + B == C) {
                long D = parseInBase(d, x);
                long E = parseInBase(e, x);
                if (D < 0 || E < 0) continue;
                long S = D + E;
                if (!ok) {
                    ans = S;
                    ok = true;
                } else if (S != ans) {
                    return "baka";
                }
            }
        }
        return ok ? String.valueOf(ans) : "baka";
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine().trim());
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < t; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            String a = st.nextToken();
            String b = st.nextToken();
            String c = st.nextToken();
            String d = st.nextToken();
            String e = st.nextToken();
            sb.append(solveCase(a, b, c, d, e)).append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

long long parse_in_base(const string& s, int x) {
    long long v = 0;
    for (char ch : s) {
        int d = ch - '0';
        if (d >= x) return -1;
        v = v * x + d;
    }
    return v;
}

string solve_case(const string& a, const string& b, const string& c,
                  const string& d, const string& e) {
    int mxDigit = 0;
    for (char ch : a) mxDigit = max(mxDigit, ch - '0');
    for (char ch : b) mxDigit = max(mxDigit, ch - '0');
    for (char ch : c) mxDigit = max(mxDigit, ch - '0');
    for (char ch : d) mxDigit = max(mxDigit, ch - '0');
    for (char ch : e) mxDigit = max(mxDigit, ch - '0');
    int startBase = max(2, mxDigit + 1);

    bool ok = false;
    long long ans = -1;

    for (int x = startBase; x <= 10; ++x) {
        long long A = parse_in_base(a, x);
        long long B = parse_in_base(b, x);
        long long C = parse_in_base(c, x);
        if (A < 0 || B < 0 || C < 0) continue;
        if (A + B == C) {
            long long D = parse_in_base(d, x);
            long long E = parse_in_base(e, x);
            if (D < 0 || E < 0) continue;
            long long S = D + E;
            if (!ok) {
                ans = S;
                ok = true;
            } else if (S != ans) {
                return "baka";
            }
        }
    }
    return ok ? to_string(ans) : string("baka");
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    if (!(cin >> t)) return 0;
    while (t--) {
        string a, b, c, d, e;
        cin >> a >> b >> c >> d >> e;
        cout << solve_case(a, b, c, d, e) << "\n";
    }
    return 0;
}
```
