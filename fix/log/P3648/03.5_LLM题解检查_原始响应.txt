## 解题思路

本题的核心是基于校准集的重构误差设定阈值，对检测集进行合格/不合格的二分类判别。具体算法步骤如下：

1. **数据解析与拼接**  
   读取输入的单行 JSON 字符串，解析出 `"train"`（校准集）和 `"test"`（检测集）两个二维数值数组。将两者按行拼接成完整的样本矩阵 \(X\)，维度为 \(N \times d\)，其中 \(N = N_{\text{train}} + N_{\text{test}}\)，特征数 \(d \ge 2\)。

2. **特征标准化**  
   对拼接后的全体数据按列进行标准化，使得每一列特征的均值为 0、方差为 1。计算方式为：  
   \[
   \mu_j = \frac{1}{N} \sum_{i=1}^{N} x_{ij}, \quad 
   \sigma_j = \sqrt{\frac{1}{N} \sum_{i=1}^{N} (x_{ij} - \mu_j)^2}
   \]  
   为避免除零，若某个 \(\sigma_j = 0\) 则将其置为 1。标准化后的矩阵 \(Z\) 定义为：
   \[
   z_{ij} = \frac{x_{ij} - \mu_j}{\sigma_j}
   \]

3. **提取第一主成分（等效 SVD）**  
   题目要求使用奇异值分解（SVD）提取第一个主成分，且固定随机种子为 42。我们通过**幂迭代法**求取 \(Z\) 的第一右奇异向量 \(v\)（即第一主成分方向），过程如下：
   - 使用随机种子 42 生成一个 \(d\) 维随机初始向量，并归一化；
   - 重复迭代直至收敛：
     \[
     v_{\text{new}} = \frac{Z^{\top}(Z v)}{\|Z^{\top}(Z v)\|}
     \]
     其中 \(Z v\) 为投影得分，\(Z^{\top}(Z v)\) 将得分方向反映射回原空间。收敛条件为两次迭代的方向变化范数小于 \(10^{-12}\) 或达到预设的最大迭代次数。

4. **投影与重构**  
   得到第一主成分方向 \(v\) 后，将所有标准化样本投影到该方向：
   \[
   s_i = z_i^{\top} v \quad (\text{标量})
   \]
   在标准化空间中的重构值为：
   \[
   \hat{z}_{i} = s_i \cdot v \quad (\text{向量})
   \]

5. **计算重构误差与阈值**  
   对校准集（前 \(N_{\text{train}}\) 个样本）逐样本计算重构误差平方和：
   \[
   e_i = \sum_{j=1}^{d} (z_{ij} - \hat{z}_{ij})^2
   \]
   将所有校准样本的重构误差排序，取第 95 百分位数作为阈值 \(T\)（使用线性插值法，具体索引为 \(0.95 \times (N_{\text{train}} - 1)\)）。

6. **检测集判别**  
   对检测集的每个样本，使用相同的标准化参数和投影方向 \(v\)，计算其在标准化空间中的重构值及重构误差 \(e'\)。若 \(e' \le T\) 则判定为合格（输出 `0`），否则判定为不合格（输出 `1`）。

7. **输出格式**  
   将检测集的所有判定结果按输入顺序组成列表，最后输出单行 JSON 数组（例如 `[0, 1, 0]`）。

## 复杂度分析

- **时间复杂度**：  
  标准化过程需要遍历全矩阵，时间复杂度为 \(O(N d)\)。幂迭代法中，一次矩阵‑向量乘 \(Z v\) 为 \(O(N d)\)，随后 \(Z^{\top}(Z v)\) 也为 \(O(N d)\)，假设迭代次数为 \(k\)（通常 \(k \ll \min(N, d)\)，实际中不超过数百次），投影与重构均为 \(O(N d)\)；误差计算与百分位数求解为 \(O(N d + N_{\text{train}} \log N_{\text{train}})\)。总时间复杂度为 \(O(k N d + N_{\text{train}} \log N_{\text{train}})\)，在 \(N \le 2\times 10^4,\ d \le 100\) 时可高效运行。

- **空间复杂度**：  
  需要存储拼接后的原始矩阵 \(X\)、标准化矩阵 \(Z\)、重构矩阵 \(\hat{Z}\) 以及若干临时向量，总体空间复杂度为 \(O(N d)\)，同样在数据范围内完全可行。

## 代码实现

### Python

```python
import sys
import json
import numpy as np

def main():
    # 读取并解析输入 JSON
    s = sys.stdin.read().strip()
    data = json.loads(s)
    train = np.array(data["train"], dtype=float)
    test = np.array(data["test"], dtype=float)

    assert train.ndim == 2 and test.ndim == 2
    assert train.shape[1] == test.shape[1] and train.shape[1] >= 2

    # 拼接
    X_all = np.vstack([train, test])
    N = X_all.shape[0]
    d = X_all.shape[1]
    n_train = train.shape[0]

    # 标准化（总体方差 ddof=0）
    mean = X_all.mean(axis=0, keepdims=True)
    std = X_all.std(axis=0, ddof=0, keepdims=True)
    std[std == 0] = 1.0
    Z_all = (X_all - mean) / std

    # 幂迭代法求第一右奇异向量（主成分方向）
    np.random.seed(42)
    v = np.random.rand(d)
    v = v / np.linalg.norm(v)
    max_iter = 1000
    for _ in range(max_iter):
        v_old = v.copy()
        v = Z_all.T @ (Z_all @ v)
        norm = np.linalg.norm(v)
        if norm < 1e-12:
            break
        v = v / norm
        if np.linalg.norm(v - v_old) < 1e-12:
            break

    # 投影与重构
    scores = Z_all @ v          # (N,)
    Z_hat = np.outer(scores, v)  # (N, d)

    # 重构误差
    errs = np.sum((Z_all - Z_hat) ** 2, axis=1)

    # 校准集第 95 百分位数（线性插值）
    train_errs = np.sort(errs[:n_train])
    n_train_err = len(train_errs)
    if n_train_err == 1:
        T = train_errs[0]
    else:
        rank = 0.95 * (n_train_err - 1)
        low = int(np.floor(rank))
        high = int(np.ceil(rank))
        fraction = rank - low
        T = train_errs[low] + fraction * (train_errs[high] - train_errs[low])

    # 检测集判定
    test_errs = errs[n_train:]
    labels = (test_errs > T).astype(int).tolist()

    print(json.dumps(labels, ensure_ascii=False))

if __name__ == "__main__":
    main()
```

### Java

```java
import com.google.gson.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String input = scanner.nextLine();
        scanner.close();

        JsonObject root = JsonParser.parseString(input).getAsJsonObject();
        JsonArray trainArr = root.getAsJsonArray("train");
        JsonArray testArr = root.getAsJsonArray("test");

        int nTrain = trainArr.size();
        int nTest = testArr.size();
        int d = trainArr.get(0).getAsJsonArray().size();
        int N = nTrain + nTest;

        double[][] X = new double[N][d];
        // 填充 train
        for (int i = 0; i < nTrain; i++) {
            JsonArray row = trainArr.get(i).getAsJsonArray();
            for (int j = 0; j < d; j++) {
                X[i][j] = row.get(j).getAsDouble();
            }
        }
        // 填充 test
        for (int i = 0; i < nTest; i++) {
            JsonArray row = testArr.get(i).getAsJsonArray();
            for (int j = 0; j < d; j++) {
                X[nTrain + i][j] = row.get(j).getAsDouble();
            }
        }

        // 标准化 (均值为0, 方差为1, ddof=0)
        double[] mean = new double[d];
        double[] std = new double[d];
        for (int j = 0; j < d; j++) {
            double sum = 0.0;
            for (int i = 0; i < N; i++) sum += X[i][j];
            mean[j] = sum / N;
            double sqSum = 0.0;
            for (int i = 0; i < N; i++) {
                double diff = X[i][j] - mean[j];
                sqSum += diff * diff;
            }
            std[j] = Math.sqrt(sqSum / N);
            if (std[j] == 0.0) std[j] = 1.0;
        }

        double[][] Z = new double[N][d];
        for (int i = 0; i < N; i++)
            for (int j = 0; j < d; j++)
                Z[i][j] = (X[i][j] - mean[j]) / std[j];

        // 幂迭代求第一右奇异向量
        Random rng = new Random(42);
        double[] v = new double[d];
        double norm = 0.0;
        for (int j = 0; j < d; j++) {
            v[j] = rng.nextDouble();
            norm += v[j] * v[j];
        }
        norm = Math.sqrt(norm);
        for (int j = 0; j < d; j++) v[j] /= norm;

        int maxIter = 1000;
        double[] vOld = new double[d];
        double[] Zv = new double[N];
        double[] ZT_Zv = new double[d];  // Z^T (Z v)
        for (int iter = 0; iter < maxIter; iter++) {
            System.arraycopy(v, 0, vOld, 0, d);
            // Zv = Z * v
            Arrays.fill(Zv, 0.0);
            for (int i = 0; i < N; i++)
                for (int j = 0; j < d; j++)
                    Zv[i] += Z[i][j] * v[j];
            // ZT_Zv = Z^T * Zv
            Arrays.fill(ZT_Zv, 0.0);
            for (int j = 0; j < d; j++)
                for (int i = 0; i < N; i++)
                    ZT_Zv[j] += Z[i][j] * Zv[i];
            // 归一化
            double newNorm = 0.0;
            for (int j = 0; j < d; j++) newNorm += ZT_Zv[j] * ZT_Zv[j];
            newNorm = Math.sqrt(newNorm);
            if (newNorm < 1e-12) break;
            for (int j = 0; j < d; j++) v[j] = ZT_Zv[j] / newNorm;
            // 收敛判断
            double diff = 0.0;
            for (int j = 0; j < d; j++) {
                double delta = v[j] - vOld[j];
                diff += delta * delta;
            }
            diff = Math.sqrt(diff);
            if (diff < 1e-12) break;
        }

        // 投影得分与重构
        double[] scores = new double[N];
        for (int i = 0; i < N; i++) {
            double s = 0.0;
            for (int j = 0; j < d; j++)
                s += Z[i][j] * v[j];
            scores[i] = s;
        }

        double[][] Z_hat = new double[N][d];
        for (int i = 0; i < N; i++)
            for (int j = 0; j < d; j++)
                Z_hat[i][j] = scores[i] * v[j];

        // 误差
        double[] err = new double[N];
        for (int i = 0; i < N; i++) {
            double e = 0.0;
            for (int j = 0; j < d; j++) {
                double diff = Z[i][j] - Z_hat[i][j];
                e += diff * diff;
            }
            err[i] = e;
        }

        // 训练误差百分位数（线性插值）
        double[] trainErr = Arrays.copyOfRange(err, 0, nTrain);
        Arrays.sort(trainErr);
        double threshold;
        if (nTrain == 1) {
            threshold = trainErr[0];
        } else {
            double rank = 0.95 * (nTrain - 1);
            int low = (int) Math.floor(rank);
            int high = (int) Math.ceil(rank);
            double fraction = rank - low;
            threshold = trainErr[low] + fraction * (trainErr[high] - trainErr[low]);
        }

        // 判定
        List<Integer> labels = new ArrayList<>();
        for (int i = nTrain; i < N; i++) {
            labels.add(err[i] <= threshold ? 0 : 1);
        }

        Gson gson = new Gson();
        System.out.println(gson.toJson(labels));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <random>
#include <algorithm>
#include <numeric>
#include <string>
#include "json.hpp"   // 使用 nlohmann/json 库

using json = nlohmann::json;
using namespace std;

int main() {
    string input;
    getline(cin, input);
    json root = json::parse(input);

    vector<vector<double>> train = root["train"].get<vector<vector<double>>>();
    vector<vector<double>> test = root["test"].get<vector<vector<double>>>();

    int nTrain = train.size();
    int nTest = test.size();
    int d = train[0].size();
    int N = nTrain + nTest;

    vector<vector<double>> X(N, vector<double>(d));
    for (int i = 0; i < nTrain; ++i)
        for (int j = 0; j < d; ++j)
            X[i][j] = train[i][j];
    for (int i = 0; i < nTest; ++i)
        for (int j = 0; j < d; ++j)
            X[nTrain + i][j] = test[i][j];

    // 标准化 (ddof = 0)
    vector<double> mean(d, 0.0), stddev(d, 0.0);
    for (int j = 0; j < d; ++j) {
        double sum = 0.0;
        for (int i = 0; i < N; ++i) sum += X[i][j];
        mean[j] = sum / N;
        double sqSum = 0.0;
        for (int i = 0; i < N; ++i) {
            double diff = X[i][j] - mean[j];
            sqSum += diff * diff;
        }
        stddev[j] = sqrt(sqSum / N);
        if (stddev[j] == 0.0) stddev[j] = 1.0;
    }

    vector<vector<double>> Z(N, vector<double>(d));
    for (int i = 0; i < N; ++i)
        for (int j = 0; j < d; ++j)
            Z[i][j] = (X[i][j] - mean[j]) / stddev[j];

    // 幂迭代求第一右奇异向量，随机种子 42
    mt19937 gen(42);
    uniform_real_distribution<double> dist(0.0, 1.0);
    vector<double> v(d);
    double norm = 0.0;
    for (int j = 0; j < d; ++j) {
        v[j] = dist(gen);
        norm += v[j] * v[j];
    }
    norm = sqrt(norm);
    for (int j = 0; j < d; ++j) v[j] /= norm;

    int maxIter = 1000;
    vector<double> vOld(d);
    vector<double> Zv(N, 0.0);
    vector<double> ZT_Zv(d, 0.0);
    for (int iter = 0; iter < maxIter; ++iter) {
        vOld = v;
        // Zv = Z * v
        fill(Zv.begin(), Zv.end(), 0.0);
        for (int i = 0; i < N; ++i)
            for (int j = 0; j < d; ++j)
                Zv[i] += Z[i][j] * v[j];
        // ZT_Zv = Z^T * Zv
        fill(ZT_Zv.begin(), ZT_Zv.end(), 0.0);
        for (int j = 0; j < d; ++j)
            for (int i = 0; i < N; ++i)
                ZT_Zv[j] += Z[i][j] * Zv[i];
        // 归一化
        double newNorm = 0.0;
        for (int j = 0; j < d; ++j) newNorm += ZT_Zv[j] * ZT_Zv[j];
        newNorm = sqrt(newNorm);
        if (newNorm < 1e-12) break;
        for (int j = 0; j < d; ++j) v[j] = ZT_Zv[j] / newNorm;
        // 收敛判断
        double diff = 0.0;
        for (int j = 0; j < d; ++j) {
            double delta = v[j] - vOld[j];
            diff += delta * delta;
        }
        diff = sqrt(diff);
        if (diff < 1e-12) break;
    }

    // 投影得分与重构
    vector<double> scores(N, 0.0);
    for (int i = 0; i < N; ++i)
        for (int j = 0; j < d; ++j)
            scores[i] += Z[i][j] * v[j];

    vector<vector<double>> Z_hat(N, vector<double>(d));
    for (int i = 0; i < N; ++i)
        for (int j = 0; j < d; ++j)
            Z_hat[i][j] = scores[i] * v[j];

    // 误差
    vector<double> err(N, 0.0);
    for (int i = 0; i < N; ++i) {
        double e = 0.0;
        for (int j = 0; j < d; ++j) {
            double diff = Z[i][j] - Z_hat[i][j];
            e += diff * diff;
        }
        err[i] = e;
    }

    // 校准集百分位数 (线性插值)
    vector<double> trainErr(err.begin(), err.begin() + nTrain);
    sort(trainErr.begin(), trainErr.end());
    double threshold;
    if (nTrain == 1) {
        threshold = trainErr[0];
    } else {
        double rank = 0.95 * (nTrain - 1);
        int low = floor(rank);
        int high = ceil(rank);
        double fraction = rank - low;
        threshold = trainErr[low] + fraction * (trainErr[high] - trainErr[low]);
    }

    // 判定
    vector<int> labels;
    for (int i = nTrain; i < N; ++i) {
        labels.push_back(err[i] <= threshold ? 0 : 1);
    }

    json output = labels;
    cout << output.dump() << endl;
    return 0;
}
```
