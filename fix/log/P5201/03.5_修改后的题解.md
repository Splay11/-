## 解题思路

本题与「逐层消散的嵌套编码」描述的计算任务一致。按输入格式读入数据后，沿用原题解的算法即可。

将每一对括号看成一个节点，括号的嵌套关系形成若干棵树。

设原括号串的最大深度为$D$：

* 第$1$轮删除深度为$D$的括号对。
* 第$2$轮删除深度为$D-1$的括号对。
* 依次类推。

因此，原来深度为$d$的括号对会在第

$D-d+1$

轮消失。

使用线性扫描统计每个字符所属括号对的深度：

* 遇到`(`时，当前深度加$1$，该字符的深度就是当前深度。
* 遇到`)`时，该字符的深度等于当前深度，随后当前深度减$1$。
* 扫描过程中记录最大深度$D$。
* 最后令每个位置的答案为$D-d+1$。

该方法使用括号栈的深度统计思想，但不需要真正维护栈。

## 复杂度分析

对于每组数据：

* 时间复杂度：$O(n)$。
* 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
import sys


# 计算每个字符消失的轮次
def get_rounds(s):
    n = len(s)
    depth = [0] * n

    current_depth = 0
    max_depth = 0

    # 统计每个字符所属括号对的深度
    for i in range(n):
        if s[i] == '(':
            current_depth += 1
            depth[i] = current_depth
            max_depth = max(max_depth, current_depth)
        else:
            depth[i] = current_depth
            current_depth -= 1

    # 根据最大深度计算消失轮次
    for i in range(n):
        depth[i] = max_depth - depth[i] + 1

    return depth


def main():
    data = sys.stdin.read().split()
    t = int(data[0])
    index = 1
    output = []

    for _ in range(t):
        n = int(data[index])
        s = data[index + 1]
        index += 2

        answer = get_rounds(s)
        output.append(' '.join(map(str, answer)))

    print('\n'.join(output))


if __name__ == '__main__':
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;

public class Main {

    // 计算每个字符消失的轮次
    public static int[] getRounds(String s) {
        int n = s.length();
        int[] depth = new int[n];

        int currentDepth = 0;
        int maxDepth = 0;

        // 统计每个字符所属括号对的深度
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                currentDepth++;
                depth[i] = currentDepth;
                maxDepth = Math.max(maxDepth, currentDepth);
            } else {
                depth[i] = currentDepth;
                currentDepth--;
            }
        }

        // 根据最大深度计算消失轮次
        for (int i = 0; i < n; i++) {
            depth[i] = maxDepth - depth[i] + 1;
        }

        return depth;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader reader =
                new BufferedReader(new InputStreamReader(System.in));

        int t = Integer.parseInt(reader.readLine().trim());
        StringBuilder output = new StringBuilder();

        for (int caseIndex = 0; caseIndex < t; caseIndex++) {
            int n = Integer.parseInt(reader.readLine().trim());
            String s = reader.readLine().trim();

            int[] answer = getRounds(s);

            for (int i = 0; i < n; i++) {
                if (i > 0) {
                    output.append(' ');
                }
                output.append(answer[i]);
            }
            output.append('\n');
        }

        System.out.print(output);
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

// 计算每个字符消失的轮次
vector<int> getRounds(const string& s) {
    int n = (int)s.size();
    vector<int> depth(n);

    int currentDepth = 0;
    int maxDepth = 0;

    // 统计每个字符所属括号对的深度
    for (int i = 0; i < n; i++) {
        if (s[i] == '(') {
            currentDepth++;
            depth[i] = currentDepth;
            maxDepth = max(maxDepth, currentDepth);
        } else {
            depth[i] = currentDepth;
            currentDepth--;
        }
    }

    // 根据最大深度计算消失轮次
    for (int i = 0; i < n; i++) {
        depth[i] = maxDepth - depth[i] + 1;
    }

    return depth;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        int n;
        string s;
        cin >> n >> s;

        vector<int> answer = getRounds(s);

        for (int i = 0; i < n; i++) {
            if (i > 0) {
```
