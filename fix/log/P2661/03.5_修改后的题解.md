## 解题思路

1. **识别规则**：每块灯阵是 $5 \times 5$ 网格。字符 `#` 是骨架，数字 `0`--`9` 是干扰填充。识别时把所有数字都看成同一类填充格，只比较 `#` 的位置。
2. **模板**：预先存好数字 $0$ 到 $9$ 的十张骨架模板。模板中 `#` 必须与输入的 `#` 对齐，`.` 对应输入中的数字格。
3. **预处理**：读入 $m$ 块灯阵后，将每一行里的数字字符全部替换为 `.`，`#` 保持不变。
4. **匹配**：把处理后的 $5$ 行与十张模板逐一比较，完全一致的那一张就是该灯阵代表的数字。题目保证一定能唯一匹配。
5. **输出**：按灯阵给出的顺序，把识别出的数字依次拼接成一个数字串输出。

## 复杂度分析

时间复杂度 $O(m)$，每块灯阵大小固定为 $5 \times 5$，与 `10` 张模板比较均为常数时间。空间复杂度 $O(m)$，用于存放全部灯阵。

```python
    for _ in range(m):
        pic = [input().strip() for _ in range(5)]
        pictures.append(pic)

    result = []
    for pic in pictures:
        new_pic = []
        for line in pic:
            new_line = ''.join('.' if ch.isdigit() else ch for ch in line)
            new_pic.append(new_line)

        recognizedDigit = -1
        for d in range(10):
            if new_pic == digitPatterns[d]:
                recognizedDigit = d
                break
        if recognizedDigit == -1:
            recognizedDigit = 0
        result.append(str(recognizedDigit))

    print("".join(result))

if __name__ == '__main__':
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();

        String[][] digitPatterns = new String[10][5];
        digitPatterns[0] = new String[] {"#...#", "#.#.#", "#.#.#", "#.#.#", "#...#"};
        digitPatterns[1] = new String[] {"##.##", "##.##", "##.##", "##.##", "##.##"};
        digitPatterns[2] = new String[] {"#...#", "###.#", "#...#", "#.###", "#...#"};
        digitPatterns[3] = new String[] {"#...#", "###.#", "#...#", "###.#", "#...#"};
        digitPatterns[4] = new String[] {"#.#.#", "#.#.#", "#...#", "###.#", "###.#"};
        digitPatterns[5] = new String[] {"#...#", "#.###", "#...#", "###.#", "#...#"};
        digitPatterns[6] = new String[] {"#...#", "#.###", "#...#", "#.#.#", "#...#"};
        digitPatterns[7] = new String[] {"#...#", "###.#", "###.#", "###.#", "###.#"};
        digitPatterns[8] = new String[] {"#...#", "#.#.#", "#...#", "#.#.#", "#...#"};
        digitPatterns[9] = new String[] {"#...#", "#.#.#", "#...#", "###.#", "#...#"};

        String[][] pictures = new String[m][5];
        for (int i = 0; i < m; i++) {
            for (int r = 0; r < 5; r++) {
                pictures[i][r] = sc.next();
            }
        }

        StringBuilder result = new StringBuilder();
        for (int i = 0; i < m; i++) {
            String[] pic = new String[5];
            for (int r = 0; r < 5; r++) {
                StringBuilder sb = new StringBuilder();
                for (int c = 0; c < 5; c++) {
                    char ch = pictures[i][r].charAt(c);
                    if (Character.isDigit(ch)) {
                        sb.append('.');
                    } else {
                        sb.append(ch);
                    }
                }
                pic[r] = sb.toString();
            }

            int recognizedDigit = -1;
            for (int d = 0; d < 10; d++) {
                boolean match = true;
                for (int r = 0; r < 5; r++) {
                    if (!pic[r].equals(digitPatterns[d][r])) {
                        match = false;
                        break;
                    }
                }
                if (match) {
                    recognizedDigit = d;
                    break;
                }
            }
            if (recognizedDigit == -1) {
                recognizedDigit = 0;
            }
            result.append(recognizedDigit);
        }

        System.out.println(result.toString());
        sc.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int m;
    cin >> m;

    vector<vector<string>> digitPatterns(10, vector<string>(5));
    digitPatterns[0] = {"#...#", "#.#.#", "#.#.#", "#.#.#", "#...#"};
    digitPatterns[1] = {"##.##", "##.##", "##.##", "##.##", "##.##"};
    digitPatterns[2] = {"#...#", "###.#", "#...#", "#.###", "#...#"};
    digitPatterns[3] = {"#...#", "###.#", "#...#", "###.#", "#...#"};
    digitPatterns[4] = {"#.#.#", "#.#.#", "#...#", "###.#", "###.#"};
    digitPatterns[5] = {"#...#", "#.###", "#...#", "###.#", "#...#"};
    digitPatterns[6] = {"#...#", "#.###", "#...#", "#.#.#", "#...#"};
    digitPatterns[7] = {"#...#", "###.#", "###.#", "###.#", "###.#"};
    digitPatterns[8] = {"#...#", "#.#.#", "#...#", "#.#.#", "#...#"};
    digitPatterns[9] = {"#...#", "#.#.#", "#...#", "###.#", "#...#"};

    vector<vector<string>> pictures(m, vector<string>(5));
    for (int i = 0; i < m; i++) {
        for (int r = 0; r < 5; r++) {
            cin >> pictures[i][r];
        }
    }

    string result;
    result.reserve(m);
    for (int i = 0; i < m; i++) {
        for (int r = 0; r < 5; r++) {
            for (int c = 0; c < 5; c++) {
                if (isdigit(pictures[i][r][c])) {
                    pictures[i][r][c] = '.';
                }
            }
        }
        int recognizedDigit = -1;
        for (int d = 0; d < 10; d++) {
            bool match = true;
            for (int r = 0; r < 5; r++) {
                if (pictures[i][r] != digitPatterns[d][r]) {
                    match = false;
                    break;
                }
            }
            if (match) {
                recognizedDigit = d;
                break;
            }
        }
        if (recognizedDigit == -1) {
            recognizedDigit = 0;
        }
        result.push_back(char('0' + recognizedDigit));
    }

    cout << result << "\n";
    return 0;
}
```
