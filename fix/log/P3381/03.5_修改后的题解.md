## 解题思路

灯牌最终每个位置只取决于该行、该列被取反的奇偶次：$a_{i,j}=r_i\oplus c_j$，其中 $r_i$ 为第 $i$ 行取反次数的奇偶，$c_j$ 为第 $j$ 列取反次数的奇偶。因此只需用集合维护被取反奇数次的行集合 $R$ 与列集合 $C$。

每一行只有两种形态：

- 若该行未取反（$r_i=0$），该行作为 $m$ 位二进制的值为 $V_0=\sum_{j\in C} 2^{m-j}\bmod MOD$；
- 若该行被取反（$r_i=1$），则该行是 $V_0$ 的按位取反：$V_1=(2^m-1-V_0)\bmod MOD$。

整块灯牌按行拼接，第 $i$ 行（从 `1` 到 $n$）位于从高到低第 $n-i$ 个长度为 $m$ 的区块，权值为 $2^{m\cdot(n-i)}$。令 $q=2^m\bmod MOD$，则所有行权值之和

$$
S_{\mathrm{all}}=\sum_{t=0}^{n-1} q^t=
\begin{cases}
n\bmod MOD,& q=1,\\
\dfrac{q^n-1}{q-1}\bmod MOD,& q\neq 1.
\end{cases}
$$

再计算被取反行的权值和 $S_R=\sum_{i\in R}2^{m(n-i)}\bmod MOD$，未取反行权值和 $S_{\mathrm{not}}=S_{\mathrm{all}}-S_R$。答案为

$$
(V_0\cdot S_{\mathrm{not}}+V_1\cdot S_R)\bmod (10^9+7).
$$

指数可达 $10^{18}$ 量级，需用快速幂；除法用费马小定理求逆元。

## 复杂度分析

时间复杂度 $O(k+(|R|+|C|)\log(nm))$，空间复杂度 $O(k)$。在 $k\le 2\times 10^5$ 下可以通过。

## 代码实现

### Python

```python
import sys

MOD = 10**9 + 7

def mod_pow(a: int, e: int) -> int:
    res = 1
    a %= MOD
    while e > 0:
        if e & 1:
            res = (res * a) % MOD
        a = (a * a) % MOD
        e >>= 1
    return res

def mod_inv(x: int) -> int:
    return mod_pow(x % MOD, MOD - 2)

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))

    R = set()
    C = set()
    for _ in range(k):
        x = int(next(it)); y = int(next(it))
        if x == 1:
            if y in C: C.remove(y)
            else: C.add(y)
        else:
            if y in R: R.remove(y)
            else: R.add(y)

    V0 = 0
    for j in C:
        V0 = (V0 + mod_pow(2, m - j)) % MOD

    pow2m = mod_pow(2, m)
    V1 = (pow2m - 1 - V0) % MOD

    q = pow2m
    if q == 1:
        S_all = n % MOD
    else:
        qn = mod_pow(q, n)
        S_all = ((qn - 1) % MOD) * mod_inv((q - 1) % MOD) % MOD

    S_R = 0
    for i in R:
        S_R = (S_R + mod_pow(2, m * (n - i))) % MOD

    S_not = (S_all - S_R) % MOD
    ans = ((V0 * S_not) + (V1 * S_R)) % MOD
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1_000_000_007L;

    static long modPow(long a, long e) {
        a %= MOD;
        long res = 1 % MOD;
        while (e > 0) {
            if ((e & 1L) == 1L) res = (res * a) % MOD;
            a = (a * a) % MOD;
            e >>= 1;
        }
        return res;
    }

    static long modInv(long x) {
        x %= MOD;
        if (x < 0) x += MOD;
        return modPow(x, MOD - 2);
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        long n = Long.parseLong(st.nextToken());
        long m = Long.parseLong(st.nextToken());
        int k = Integer.parseInt(st.nextToken());

        HashSet<Long> R = new HashSet<>(k * 2);
        HashSet<Long> C = new HashSet<>(k * 2);
        for (int t = 0; t < k; t++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            long y = Long.parseLong(st.nextToken());
            if (x == 1) {
                if (C.contains(y)) C.remove(y);
                else C.add(y);
            } else {
                if (R.contains(y)) R.remove(y);
                else R.add(y);
            }
        }

        long V0 = 0;
        for (long j : C) {
            V0 += modPow(2, m - j);
            if (V0 >= MOD) V0 -= MOD;
        }

        long pow2m = modPow(2, m);
        long V1 = (pow2m - 1 - V0) % MOD;
        if (V1 < 0) V1 += MOD;

        long q = pow2m;
        long S_all;
        if (q == 1) {
            S_all = n % MOD;
        } else {
            long qn = modPow(q, n);
            long numerator = (qn - 1) % MOD;
            if (numerator < 0) numerator += MOD;
            long denom = (q - 1 + MOD) % MOD;
            S_all = (numerator * modInv(denom)) % MOD;
        }

        long S_R = 0;
        for (long i : R) {
            S_R += modPow(2, m * (n - i));
            if (S_R >= MOD) S_R -= MOD;
        }

        long S_not = (S_all - S_R) % MOD;
        if (S_not < 0) S_not += MOD;
        long ans = (((V0 * S_not) % MOD) + ((V1 * S_R) % MOD)) % MOD;
        if (ans < 0) ans += MOD;
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

static const long long MOD = 1000000007LL;

long long mod_pow(long long a, unsigned long long e) {
    long long res = 1 % MOD;
    a %= MOD;
    while (e > 0) {
        if (e & 1ULL) res = (__int128)res * a % MOD;
        a = (__int128)a * a % MOD;
        e >>= 1ULL;
    }
    return res;
}

long long mod_inv(long long x) {
    return mod_pow((x % MOD + MOD) % MOD, MOD - 2);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long n, m;
    int k;
    if (!(cin >> n >> m >> k)) return 0;

    unordered_set<long long> R, C;
    R.reserve(k * 2);
    C.reserve(k * 2);
    for (int t = 0; t < k; ++t) {
        int x;
        long long y;
        cin >> x >> y;
        if (x == 1) {
            if (C.count(y)) C.erase(y);
            else C.insert(y);
        } else {
            if (R.count(y)) R.erase(y);
            else R.insert(y);
        }
    }

    long long V0 = 0;
    for (auto j : C) {
        V0 += mod_pow(2, (unsigned long long)(m - j));
        if (V0 >= MOD) V0 -= MOD;
    }

    long long pow2m = mod_pow(2, (unsigned long long)m);
    long long V1 = (pow2m - 1 - V0) % MOD;
    if (V1 < 0) V1 += MOD;

    long long q = pow2m;
    long long S_all;
    if (q == 1) {
        S_all = n % MOD;
    } else {
        long long qn = mod_pow(q, (unsigned long long)n);
        long long numerator = (qn - 1) % MOD;
        if (numerator < 0) numerator += MOD;
        long long denom = (q - 1 + MOD) % MOD;
        S_all = (__int128)numerator * mod_inv(denom) % MOD;
    }

    long long S_R = 0;
    for (auto i : R) {
        unsigned long long exp = (unsigned long long)m * (unsigned long long)(n - i);
        S_R += mod_pow(2, exp);
        if (S_R >= MOD) S_R -= MOD;
    }

    long long S_not = (S_all - S_R) % MOD;
    if (S_not < 0) S_not += MOD;
    long long ans = (((__int128)V0 * S_not) + ((__int128)V1 * S_R)) % MOD;
    if (ans < 0) ans += MOD;
    cout << ans << '\n';
    return 0;
}
```
