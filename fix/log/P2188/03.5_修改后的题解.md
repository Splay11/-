## 解题思路

1. 一次交错调和把长度为 $k$ 的序列变成 $k-1$，直接模拟需要 $O(n^2)$ 时间，在 $n\le 10^5$ 时不可行。需要求出每个初始浓度 $v_i$ 对最终结果的线性系数。
2. 当 $n$ 为奇数时，按题意先以 $\sigma=+1$ 做一次调和：
   $$v_i \leftarrow v_i + (-1)^{i-1} v_{i+1}\qquad (1\le i\le n-1),$$
   长度变为偶数 $n-1$。此后只需处理偶数长度。
3. 对偶数长度 $m$，打表可知：把相邻两项配成一对 $(v_{2j-1},v_{2j})$，第 $j$ 对的公共系数是组合数 $\binom{m/2-1}{j-1}$；对内第二项再乘上符号 $\sigma_0$，其中
   $$\sigma_0=\begin{cases}-1,& m\equiv 0\pmod{4},\\ +1,& m\not\equiv 0\pmod{4}.\end{cases}$$
   因此答案为
   $$\sum_{j=0}^{m/2-1}\binom{m/2-1}{j}\bigl(v_{2j+1}+\sigma_0 v_{2j+2}\bigr).$$
   这对系数来自杨辉三角（即反复相邻合成后的二项式展开）。
4. 当 $m\le 2$ 时，结果就是 $v_1+v_2$（若 $n=1$，则视为 $v_2=0$）。
5. 预处理阶乘与阶乘逆元，即可 $O(1)$ 计算模 $10^9+7$ 下的组合数。所有运算在模意义下进行，中间结果加上模数再取模，以避免负数。

## 复杂度分析

设试剂份数为 $n$，预处理上限为 $N=10^5$。

- 时间复杂度：$O(N\log MOD + n)$，其中求逆元使用快速幂。
- 空间复杂度：$O(N)$。

## 代码实现

### Python

```python
MOD = 10**9 + 7
N = 10**5 + 10
a = [0] * N
fact = [0] * N
inv_fact = [0] * N


def qmi(base, exp):
    res = 1
    while exp:
        if exp & 1:
            res = res * base % MOD
        exp >>= 1
        base = base * base % MOD
    return res


def init():
    fact[0] = inv_fact[0] = 1
    for i in range(1, N):
        fact[i] = fact[i - 1] * i % MOD
        inv_fact[i] = inv_fact[i - 1] * qmi(i, MOD - 2) % MOD


def comb(n, m):
    if n < m:
        return 0
    return fact[n] * inv_fact[n - m] % MOD * inv_fact[m] % MOD


def main():
    init()
    n = int(input())
    vals = list(map(int, input().split()))
    for i in range(1, n + 1):
        a[i] = vals[i - 1]

    if n <= 2:
        print((a[1] + a[2]) % MOD)
        return

    # 奇数长度先按起始符号 +1 调和一次
    if n & 1:
        sign = 1
        for i in range(1, n):
            a[i] = a[i] + a[i + 1] * sign
            sign = -sign
        n -= 1

    sign = 1 if (n % 4) else -1
    res = 0
    for i in range(1, n + 1, 2):
        coeff = comb(n // 2 - 1, i // 2)
        res = (res + (a[i] + sign * a[i + 1]) * coeff + MOD) % MOD

    print((res + MOD) % MOD)


if __name__ == '__main__':
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    static final int N = 100010;
    static final int MOD = 1000000007;
    static long[] a = new long[N];
    static long[] fact = new long[N];
    static long[] invFact = new long[N];

    static long qmi(long base, long exp) {
        long res = 1;
        while (exp != 0) {
            if ((exp & 1) != 0) {
                res = res * base % MOD;
            }
            exp >>= 1;
            base = base * base % MOD;
        }
        return res;
    }

    static void init() {
        fact[0] = invFact[0] = 1;
        for (int i = 1; i < N; i++) {
            fact[i] = fact[i - 1] * i % MOD;
            invFact[i] = invFact[i - 1] * qmi(i, MOD - 2) % MOD;
        }
    }

    static long comb(long n, long m) {
        if (n < m) {
            return 0;
        }
        return fact[(int) n] * invFact[(int) (n - m)] % MOD * invFact[(int) m] % MOD;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        init();
        int n = sc.nextInt();
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextLong();
        }

        if (n <= 2) {
            System.out.println((a[1] + a[2]) % MOD);
            return;
        }

        if ((n & 1) != 0) {
            int sign = 1;
            for (int i = 1; i < n; i++) {
                a[i] = a[i] + a[i + 1] * sign;
                sign = -sign;
            }
            n--;
        }

        int sign = (n % 4 != 0) ? 1 : -1;
        long res = 0;
        for (int i = 1; i <= n; i += 2) {
            long coeff = comb(n / 2 - 1, i / 2);
            res = (res + (a[i] + sign * a[i + 1]) * coeff + MOD) % MOD;
        }
        System.out.println((res + MOD) % MOD);
    }
}
```
