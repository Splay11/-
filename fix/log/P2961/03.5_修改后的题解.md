## 解题思路

本题的核心在于分析交换操作的免费条件，并根据不同的 $k$ 取值，将最少体力消耗转化为不同的逆序对统计问题。货箱的优先级编号恰好是 $1$ 到 $n$ 的一个排列，记为 $a_1, a_2, \dots, a_n$。

- **情况一：$k \ge 3$**
  由于交换时优先级差的绝对值大于 $1$ 时不消耗体力，且允许交换的两个货箱优先级差值最大为 $k$（至少为 $3$）。借助差值为 $2$ 或以上的免费交换，可以在不消耗任何体力的情况下连通任意两个货箱，并完成整个排列的排序。因此答案为 $0$。

- **情况二：$k = 1$**
  此时只允许交换优先级差值为 $1$ 的货箱，每次交换恰好消耗 $1$ 单位体力。这等价于只能进行相邻元素的交换。将一个排列通过相邻交换变为升序所需的最少交换次数，等于原排列的逆序对总数。每消除一个逆序对至少需要一次相邻交换，而存在一种交换顺序使得总消耗恰好等于逆序对数。因此答案为 $\sum_{1 \le i < j \le n} [a_i > a_j]$，可以使用归并排序在 $O(n \log n)$ 时间内求出。

- **情况三：$k = 2$**
  优先级差值为 $2$ 的交换免费，这意味着**同奇偶性**的货箱之间可以通过免费交换任意重排（例如：两个奇数或两个偶数的差均为偶数，必然可以通过若干次差值为 $2$ 的免费交换完成内部排序）。而奇偶性不同的两个货箱若需要交换相对顺序，则必须进行至少一次差值为 $1$ 的交换，每次消耗 $1$ 点体力。因此，最少体力消耗等于原序列的所有逆序对中，**两个元素奇偶性不同的逆序对总数**。
  
  具体统计方法：
  1. 维护两个树状数组（Fenwick Tree），分别记录右侧剩余的**奇数**和**偶数**优先级编号的分布。
  2. 初始时将所有货箱的优先级按奇偶性插入对应的树状数组。
  3. 从左到右扫描每一个货箱 $a_i$：
     - 将 $a_i$ 从它所属的树状数组中删除（表示它已离开“右侧剩余集合”）；
     - 查询另一棵（奇偶性不同的）树状数组中值小于 $a_i$ 的元素个数，累加到答案中。
  这样每个逆序对 $(i,j)$（$a_i > a_j$ 且奇偶性不同）恰好被统计一次。
  时间复杂度 $O(n \log n)$。

## 复杂度分析

- 时间复杂度：对于 $k = 1$，归并排序求全逆序对 $O(n \log n)$；对于 $k = 2$，使用两个树状数组进行扫描与查询，每次操作 $O(\log n)$，总复杂度 $O(n \log n)$；对于 $k \ge 3$，直接输出 $O(1)$。整体时间复杂度为 $O(n \log n)$。
- 空间复杂度：需要存储货箱数组 $O(n)$，临时数组或树状数组 $O(n)$。整体空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(10**7)

def merge_all(a, tmp, l, r):
    """归并排序统计全部逆序对（k=1）"""
    if r - l <= 1:
        return 0
    m = (l + r) // 2
    cnt = merge_all(a, tmp, l, m) + merge_all(a, tmp, m, r)
    i, j, p = l, m, l
    while i < m or j < r:
        if j == r or (i < m and a[i] <= a[j]):
            tmp[p] = a[i]
            i += 1
        else:
            cnt += (m - i)
            tmp[p] = a[j]
            j += 1
        p += 1
    a[l:r] = tmp[l:r]
    return cnt

class BIT:
    """树状数组"""
    def __init__(self, n):
        self.n = n
        self.t = [0] * (n + 1)
    def update(self, i, v=1):
        while i <= self.n:
            self.t[i] += v
            i += i & -i
    def query(self, i):
        s = 0
        while i > 0:
            s += self.t[i]
            i -= i & -i
        return s

def main():
    data = list(map(int, sys.stdin.read().split()))
    n, k = data[0], data[1]
    a = data[2:]

    if k >= 3:
        print(0)
        return

    if k == 1:
        tmp = [0] * n
        print(merge_all(a, tmp, 0, n))
    else:
        # k == 2：免费交换同奇偶，只计算奇偶不同的逆序对
        bit_even = BIT(n)
        bit_odd = BIT(n)
        for x in a:
            if x & 1:
                bit_odd.update(x, 1)
            else:
                bit_even.update(x, 1)

        ans = 0
        for x in a:
            if x & 1:
                bit_odd.update(x, -1)
                ans += bit_even.query(x - 1)
            else:
                bit_even.update(x, -1)
                ans += bit_odd.query(x - 1)
        print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 归并排序统计全部逆序对（k=1）
    static long mergeAll(int[] a, int l, int r, int[] tmp) {
        if (r - l <= 1) return 0;
        int m = (l + r) >> 1;
        long cnt = mergeAll(a, l, m, tmp) + mergeAll(a, m, r, tmp);
        int i = l, j = m, p = l;
        while (i < m || j < r) {
            if (j == r || (i < m && a[i] <= a[j]))
                tmp[p++] = a[i++];
            else {
                cnt += (m - i);
                tmp[p++] = a[j++];
            }
        }
        System.arraycopy(tmp, l, a, l, r - l);
        return cnt;
    }

    // 树状数组
    static class BIT {
        int n;
        int[] t;
        BIT(int n) {
            this.n = n;
            t = new int[n + 1];
        }
        void update(int i, int v) {
            while (i <= n) {
                t[i] += v;
                i += i & -i;
            }
        }
        int query(int i) {
            int s = 0;
            while (i > 0) {
                s += t[i];
                i -= i & -i;
            }
            return s;
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        int[] a = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }

        if (k >= 3) {
            System.out.println(0);
            return;
        }

        if (k == 1) {
            System.out.println(mergeAll(a, 0, n, new int[n]));
        } else { // k == 2
            BIT bitEven = new BIT(n);
            BIT bitOdd = new BIT(n);
            for (int x : a) {
                if ((x & 1) == 1)
                    bitOdd.update(x, 1);
                else
                    bitEven.update(x, 1);
            }
            long ans = 0;
            for (int x : a) {
                if ((x & 1) == 1) {
                    bitOdd.update(x, -1);
                    ans += bitEven.query(x - 1);
                } else {
                    bitEven.update(x, -1);
                    ans += bitOdd.query(x - 1);
                }
            }
            System.out.println(ans);
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 归并排序统计全部逆序对（k=1）
ll count_all_inv(vector<int>& a) {
    int n = a.size();
    vector<int> tmp(n);
    function<ll(int,int)> dfs = [&](int l, int r) -> ll {
        if (r - l <= 1) return 0;
        int m = (l + r) >> 1;
        ll cnt = dfs(l, m) + dfs(m, r);
        int i = l, j = m, p = l;
        while (i < m || j < r) {
            if (j == r || (i < m && a[i] <= a[j]))
                tmp[p++] = a[i++];
            else {
                cnt += (m - i);
                tmp[p++] = a[j++];
            }
        }
        for (int k = l; k < r; ++k) a[k] = tmp[k];
        return cnt;
    };
    return dfs(0, n);
}

// 树状数组
struct BIT {
    int n;
    vector<int> t;
    BIT(int _n) : n(_n), t(_n + 1, 0) {}
    void update(int i, int v = 1) {
        for (; i <= n; i += i & -i) t[i] += v;
    }
    int query(int i) {
        int s = 0;
        for (; i > 0; i -= i & -i) s += t[i];
        return s;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    if (k >= 3) {
        cout << 0 << '\n';
        return 0;
    }

    if (k == 1) {
        cout << count_all_inv(a) << '\n';
    } else { // k == 2
        BIT bit_even(n), bit_odd(n);
        for (int x : a) {
            if (x & 1)
                bit_odd.update(x, 1);
            else
                bit_even.update(x, 1);
        }
        ll ans = 0;
        for (int x : a) {
            if (x & 1) {
                bit_odd.update(x, -1);
                ans += bit_even.query(x - 1);
            } else {
                bit_even.update(x, -1);
                ans += bit_odd.query(x - 1);
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
```
