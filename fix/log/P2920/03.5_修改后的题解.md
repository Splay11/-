## 解题思路

我们记原序列为 $b_1, b_2, \dots, b_n$，问题需要回答 $m$ 个询问，每个询问给定一个区间 $[l, r]$，要求输出

$$
\sum_{\substack{p\in [l, r] \\ p\ \text{是素数}}} W(p) \pmod{998244353},
$$

其中

$$
W(k)=\sum_{i=1}^{n-k+1}\sum_{j=i}^{i+k-1}b_j.
$$

为了在 $O(1)$ 时间内回答每个询问，我们需要预处理出所有长度 $k$ 对应的 $W(k)$（记为 $G[k]$），然后对素数长度做前缀和。

### 1. 前缀和与二次前缀和

定义一维前缀和 $P[i]=\sum_{j=1}^i b_j$（$1\le i\le n$），以及二次前缀和 $Q[i]=\sum_{j=1}^i P[j]$。

对于固定的长度 $k$，有

$$
W(k)=\sum_{i=1}^{n-k+1} (P[i+k-1]-P[i-1]).
$$

展开并交换求和次序，可以化简为

$$
W(k)=Q[n]-Q[k-1]-Q[n-k].
$$

该公式可在 $O(1)$ 时间内计算任意一个 $W(k)$，且只需要 $P$ 和 $Q$ 数组。

### 2. 预处理所有 $G[k]=W(k) \bmod 998244353$

利用上述公式：

$$
G[k] = (Q[n] - Q[k-1] - Q[n-k]) \bmod 998244353, \quad 1\le k\le n.
$$

计算时注意取模保正。

### 3. 筛法判定素数长度

因为 $n\le 10^4$，可以使用埃氏筛在 $O(n\log\log n)$ 时间内标记 $1\sim n$ 中哪些数是素数。素数定义从 $2$ 开始（即 $1$ 不是素数）。

### 4. 针对素数长度构造答案前缀和

定义数组 $h[i]$ 表示长度在 $1\sim i$ 中所有素数长度对应的 $G$ 值之和：

$$
h[i] = h[i-1] + 
\begin{cases}
G[i], & \text{若 } i \text{ 是素数}\\
0, & \text{否则}
\end{cases}\pmod{998244353}.
$$

### 5. 处理查询

对于询问 $[l, r]$，答案可直接得到：

$$
\mathrm{ans} = (h[r] - h[l-1]) \bmod 998244353.
$$

保证结果非负输出即可。

## 复杂度分析

- 计算前缀和 $P, Q$ 以及所有 $G[k]$：$O(n)$  
- 埃氏筛素数：$O(n\log\log n)$  
- 构建前缀和 $h$：$O(n)$  
- 每个询问：$O(1)$，共 $m$ 次，总计 $O(m)$

总时间复杂度为 $O(n\log\log n + m)$，空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys
import math

def solve() -> None:
    input = sys.stdin.readline
    MOD = 998244353
    n, m = map(int, input().split())
    b = list(map(int, input().split()))
    
    # 1. 前缀和 P 与二次前缀和 Q
    P = [0] * (n + 1)
    Q = [0] * (n + 1)
    for i in range(1, n + 1):
        P[i] = P[i-1] + b[i-1]
        Q[i] = Q[i-1] + P[i]
    
    # 2. 计算 G[k] = W(k) mod MOD
    G = [0] * (n + 1)
    for k in range(1, n + 1):
        G[k] = (Q[n] - Q[k-1] - Q[n-k]) % MOD
    
    # 3. 埃氏筛，标记素数长度
    is_prime = [True] * (n + 1)
    if n >= 0:
        is_prime[0] = False
    if n >= 1:
        is_prime[1] = False
    for i in range(2, int(math.isqrt(n)) + 1):
        if is_prime[i]:
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    
    # 4. 前缀和 h[i]：前 i 个长度中素数长度对应的 G 值之和
    h = [0] * (n + 1)
    for i in range(1, n + 1):
        h[i] = h[i-1] + (G[i] if is_prime[i] else 0)
        if h[i] >= MOD:
            h[i] -= MOD
    
    # 5. 回答询问
    out_lines = []
    for _ in range(m):
        l, r = map(int, input().split())
        ans = (h[r] - h[l-1]) % MOD
        out_lines.append(str(ans))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final int MOD = 998244353;
    
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        
        long[] b = new long[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            b[i] = Long.parseLong(st.nextToken());
        }
        
        // 前缀和 P 和二次前缀和 Q
        long[] P = new long[n + 1];
        long[] Q = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            P[i] = P[i-1] + b[i];
            Q[i] = Q[i-1] + P[i];
        }
        
        // G[k] = W(k) mod MOD
        long[] G = new long[n + 1];
        for (int k = 1; k <= n; k++) {
            long val = Q[n] - Q[k-1] - Q[n-k];
            val %= MOD;
            if (val < 0) val += MOD;
            G[k] = val;
        }
        
        // 埃氏筛
        boolean[] isPrime = new boolean[n + 1];
        Arrays.fill(isPrime, true);
        if (n >= 0) isPrime[0] = false;
        if (n >= 1) isPrime[1] = false;
        for (int i = 2; i * i <= n; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= n; j += i) {
                    isPrime[j] = false;
                }
            }
        }
        
        // 前缀和 h
        long[] h = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            h[i] = h[i-1] + (isPrime[i] ? G[i] : 0);
            if (h[i] >= MOD) h[i] -= MOD;
        }
        
        // 回答查询
        PrintWriter out = new PrintWriter(System.out);
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int l = Integer.parseInt(st.nextToken());
            int r = Integer.parseInt(st.nextToken());
            long ans = (h[r] - h[l-1]) % MOD;
            if (ans < 0) ans += MOD;
            out.println(ans);
        }
        out.flush();
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
using ll = long long;

const int MOD = 998244353;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;
    vector<ll> b(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> b[i];
    }

    // 前缀和 P 和二次前缀和 Q
    vector<ll> P(n + 1), Q(n + 1);
    for (int i = 1; i <= n; ++i) {
        P[i] = P[i-1] + b[i];
        Q[i] = Q[i-1] + P[i];
    }

    // G[k] = W(k) mod MOD
    vector<ll> G(n + 1);
    for (int k = 1; k <= n; ++k) {
        ll val = Q[n] - Q[k-1] - Q[n-k];
        val %= MOD;
        if (val < 0) val += MOD;
        G[k] = val;
    }

    // 埃氏筛
    vector<bool> isPrime(n + 1, true);
    if (n >= 0) isPrime[0] = false;
    if (n >= 1) isPrime[1] = false;
    for (int i = 2; i * i <= n; ++i) {
        if (isPrime[i]) {
            for (int j = i * i; j <= n; j += i) {
                isPrime[j] = false;
            }
        }
    }

    // 前缀和 h
    vector<ll> h(n + 1);
    for (int i = 1; i <= n; ++i) {
        h[i] = h[i-1] + (isPrime[i] ? G[i] : 0);
        if (h[i] >= MOD) h[i] -= MOD;
    }

    // 回答查询
    while (m--) {
        int l, r;
        cin >> l >> r;
        ll ans = (h[r] - h[l-1]) % MOD;
        if (ans < 0) ans += MOD;
        cout << ans << '\n';
    }

    return 0;
}
```
