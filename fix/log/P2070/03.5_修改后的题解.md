## 解题思路

我们需要统计满足 $1 \le i < j \le n$ 且 $L(i) > R(j)$ 的下标对 $(i,j)$ 的数量，其中：
- $L(i)$ 是 $a_i$ 在前缀 $a_1 \dots a_i$ 中出现的次数；
- $R(j)$ 是 $a_j$ 在后缀 $a_j \dots a_n$ 中出现的次数。

**核心思路：树状数组 + 频数统计**

1. **预处理右累积频数 $R(j)$**  
   从右向左遍历序列，用一个哈希表维护每个元素已经出现的次数。对于位置 $j$，$R(j)$ 即为遍历到 $j$ 时哈希表中 $a_j$ 对应的计数值。将每个 $R(j)$ 加入树状数组（对应频数值的位置 $+1$），表示这个右累积频数“可用”。

2. **从左向右计算答案**  
   清空哈希表后，从左到右枚举 $i$（作为左端点），对于当前位置 $i$：
   - 先更新哈希表中 $a_i$ 的计数，得到 $L(i)$。
   - 由于 $j$ 必须严格大于 $i$，当前 $R(i)$ 不能作为右端点的频数参与统计，因此从树状数组中将 $R(i)$ 移除（对应位置 $-1$）。
   - 在树状数组中查询所有小于 $L(i)$ 的右累积频数的总数，即查询前缀和 $\text{query}(L(i)-1)$，并将结果累加至答案。这一步对应所有满足 $R(j) < L(i)$ 且 $j > i$ 的位置 $j$。

3. **答案输出**  
   最终累加的结果即为满足条件的对数。

**注意**：序列中的数值范围较大（$\le 10^9$），需要用哈希表（如 C++ 的 `map` / Python 的 `dict` / Java 的 `HashMap`）进行频数统计；树状数组的大小设置为 $2 \times 10^5 + 5$ 足够覆盖所有可能的频数值（$n \le 10^5$）。

## 复杂度分析

- **时间复杂度**：$O(n \log n)$  
  预处理后缀频数并更新树状数组需要 $O(n \log n)$；从左到右扫描时每个位置进行一次树状数组的删除和查询，也是 $O(n \log n)$。哈希表操作单次 $O(\log n)$ 或均摊 $O(1)$，不影响整体复杂度。
- **空间复杂度**：$O(n)$  
  需要存储原数组、后缀频数数组、树状数组以及哈希表，均与 $n$ 同阶。

## 代码实现

### Python

```python
import sys

def lowbit(x):
    return x & -x

def add(x, y, t1, N):
    while x < N:
        t1[x] += y
        x += lowbit(x)

def query(r, t1):
    res = 0
    while r > 0:
        res += t1[r]
        r -= lowbit(r)
    return res

def main():
    input = sys.stdin.read
    data = input().split()
    n = int(data[0])
    a = list(map(int, data[1:n+1]))
    
    N = 200005
    t1 = [0] * N
    suf = [0] * (n + 1)   # 1-indexed，suf[i] 对应 R(i)
    mp = {}
    
    # 预处理右累积频数 R(j) 并加入树状数组
    for i in range(n - 1, -1, -1):
        mp[a[i]] = mp.get(a[i], 0) + 1
        suf[i + 1] = mp[a[i]]
        add(suf[i + 1], 1, t1, N)
    
    mp.clear()
    ans = 0
    
    # 从左向右枚举 i
    for i in range(1, n + 1):
        mp[a[i - 1]] = mp.get(a[i - 1], 0) + 1   # L(i)
        add(suf[i], -1, t1, N)                   # 移除 R(i)
        ans += query(mp[a[i - 1]] - 1, t1)       # 累加 L(i) > R(j) 的数量
    
    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static int N = 200005;
    static long[] t1;

    static int lowbit(int x) {
        return x & -x;
    }

    static void add(int x, int y) {
        while (x < N) {
            t1[x] += y;
            x += lowbit(x);
        }
    }

    static long query(int r) {
        long res = 0;
        while (r > 0) {
            res += t1[r];
            r -= lowbit(r);
        }
        return res;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        
        int[] a = new int[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        
        t1 = new long[N];
        int[] suf = new int[n + 1];
        Map<Integer, Integer> mp = new HashMap<>();
        
        // 预处理右累积频数 R(j) 并加入树状数组
        for (int i = n; i >= 1; i--) {
            int cnt = mp.getOrDefault(a[i], 0) + 1;
            mp.put(a[i], cnt);
            suf[i] = cnt;
            add(suf[i], 1);
        }
        
        mp.clear();
        long ans = 0;
        
        // 从左向右枚举 i
        for (int i = 1; i <= n; i++) {
            int cnt = mp.getOrDefault(a[i], 0) + 1;
            mp.put(a[i], cnt);          // L(i)
            add(suf[i], -1);            // 移除 R(i)
            ans += query(cnt - 1);      // 累加 L(i) > R(j) 的数量
        }
        
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int N = 200005;
ll t1[N];

int lowbit(int x) {
    return x & -x;
}

void add(int x, int y) {
    for (int i = x; i < N; i += lowbit(i)) {
        t1[i] += y;
    }
}

ll query(int r) {
    ll res = 0;
    for (int i = r; i > 0; i -= lowbit(i)) {
        res += t1[i];
    }
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    
    int n;
    cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    
    vector<int> suf(n + 1);
    unordered_map<int, int> mp;
    
    // 预处理右累积频数 R(j) 并加入树状数组
    for (int i = n; i >= 1; i--) {
        mp[a[i]]++;
        suf[i] = mp[a[i]];
        add(suf[i], 1);
    }
    
    mp.clear();
    ll ans = 0;
    
    // 从左向右枚举 i
    for (int i = 1; i <= n; i++) {
        mp[a[i]]++;               // L(i)
        add(suf[i], -1);          // 移除 R(i)
        ans += query(mp[a[i]] - 1);   // 累加 L(i) > R(j) 的数量
    }
    
    cout << ans << '\n';
    return 0;
}
```
