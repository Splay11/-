## 解题思路

1. **问题分析**  
   信号序列中每个元素均为 $0\sim 9$ 的整数，变换规则为 $(x \times t) \bmod 10$。  
   由于模 $10$ 运算只关心乘积的个位数，因此真正影响结果的只有 $t \bmod 10$ 的值。  
   若我们能快速知道任意区间 $[l,r]$ 内原始数字 $0\sim 9$ 各出现多少次，就能根据映射直接得到变换后的分布。

2. **前缀统计**  
   构造二维前缀数组 `pre`，其中 `pre[i][d]` 表示前 $i$ 个信号中数字 $d$ 出现的次数。  
   状态转移：`pre[i] = pre[i-1]`，然后将当前信号 `S[i]` 对应的计数加 $1$。  
   由此，区间 $[l,r]$ 中数字 $d$ 的个数可以通过差分得到：
   $$cnt_d = pre[r][d] - pre[l-1][d]$$

3. **映射预处理**  
   预计算一个 $10\times 10$ 的映射表 `trans[t][d] = (d \times t) \bmod 10`，其中 $t\in[0,9], d\in[0,9]$。  
   这样对于任何询问的转置因子 $t$，只需取其个位 $t' = t \bmod 10$，然后查表即可获得转换关系。

4. **回答询问**  
   对于每个询问 $(l,r,t)$：
   - 取 $t' = t \bmod 10$；
   - 对每个原始数字 $d\in[0,9]$，计算其在 $[l,r]$ 内的出现次数 $cnt_d$；
   - 将 $cnt_d$ 累加到答案数组的第 $\text{trans}[t'][d]$ 个位置上；
   - 输出最终的 $10$ 个整数，分别对应变换后值 $0,1,\dots,9$ 的出现次数。

## 复杂度分析

- 时间复杂度：前缀统计需要遍历一次序列，维护 $10$ 个计数，为 $O(10n)$；每次询问需遍历 $10$ 个原始数字，为 $O(10)$。总时间复杂度 $O(10n+10q)$，由于常数很小，可视为 $O(n+q)$。
- 空间复杂度：主要是前缀数组 `pre`，大小为 $(n+1)\times 10$，故空间复杂度 $O(10n)$，在 $n\le 10^5$ 时完全可行。

## 代码实现

### Python

```python
def solve(n, arr, queries):
    # pre[i][d] 表示前 i 个数中数字 d 出现的次数
    pre = [[0] * 10 for _ in range(n + 1)]

    for i in range(1, n + 1):
        # 继承前一个位置的统计结果
        for d in range(10):
            pre[i][d] = pre[i - 1][d]
        # 当前数字出现次数加一
        pre[i][arr[i - 1]] += 1

    # 预处理个位映射：trans[t][d] = (d * t) % 10
    trans = [[0] * 10 for _ in range(10)]
    for t in range(10):
        for d in range(10):
            trans[t][d] = (d * t) % 10

    ans_list = []

    for l, r, t in queries:
        t_mod = t % 10
        ans = [0] * 10

        # 统计区间内原始数字，再映射到结果中
        for d in range(10):
            cnt = pre[r][d] - pre[l - 1][d]
            ans[trans[t_mod][d]] += cnt

        ans_list.append(ans)

    return ans_list


def main():
    n = int(input())
    arr = list(map(int, input().split()))
    q = int(input())

    queries = []
    for _ in range(q):
        l, r, t = map(int, input().split())
        queries.append((l, r, t))

    res = solve(n, arr, queries)

    for row in res:
        print(*row)


if __name__ == "__main__":
    main()
```

import java.util.*;

public class Main {

    public static List<int[]> solve(int n, int[] arr, int[][] queries) {
        // pre[i][d] 表示前 i 个数中数字 d 出现的次数
        int[][] pre = new int[n + 1][10];

        for (int i = 1; i <= n; i++) {
            // 继承前一个位置的统计结果
            for (int d = 0; d < 10; d++) {
                pre[i][d] = pre[i - 1][d];
            }
            // 当前数字出现次数加一
            pre[i][arr[i - 1]]++;
        }

        // 预处理个位映射：trans[t][d] = (d * t) % 10
        int[][] trans = new int[10][10];
        for (int t = 0; t < 10; t++) {
            for (int d = 0; d < 10; d++) {
                trans[t][d] = (d * t) % 10;
            }
        }

        List<int[]> res = new ArrayList<>();

        for (int[] query : queries) {
            int l = query[0];
            int r = query[1];
            int t_mod = query[2] % 10;

            int[] ans = new int[10];

            // 统计区间内原始数字，再映射到结果中
            for (int d = 0; d < 10; d++) {
                int cnt = pre[r][d] - pre[l - 1][d];
                ans[trans[t_mod][d]] += cnt;
            }

            res.add(ans);
        }

        return res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int q = sc.nextInt();
        int[][] queries = new int[q][3];
        for (int i = 0; i < q; i++) {
            queries[i][0] = sc.nextInt();
            queries[i][1] = sc.nextInt();
            queries[i][2] = sc.nextInt();
        }

        List<int[]> res = solve(n, arr, queries);

        StringBuilder sb = new StringBuilder();
        for (int[] row : res) {
            for (int i = 0; i < 10; i++) {
                if (i > 0) sb.append(" ");
                sb.append(row[i]);
            }
            sb.append("\n");
        }

        System.out.print(sb.toString());
        sc.close();
    }
}

#include <iostream>
#include <vector>
using namespace std;

vector<vector<int>> solve(int n, const vector<int>& arr, const vector<vector<int>>& queries) {
    // pre[i][d] 表示前 i 个数中数字 d 出现的次数
    vector<vector<int>> pre(n + 1, vector<int>(10, 0));

    for (int i = 1; i <= n; i++) {
        // 继承前一个位置的统计结果
        for (int d = 0; d < 10; d++) {
            pre[i][d] = pre[i - 1][d];
        }
        // 当前数字出现次数加一
        pre[i][arr[i - 1]]++;
    }

    // 预处理个位映射：trans[t][d] = (d * t) % 10
    int trans[10][10];
    for (int t = 0; t < 10; t++) {
        for (int d = 0; d < 10; d++) {
            trans[t][d] = (d * t) % 10;
        }
    }

    vector<vector<int>> res;

    for (const auto& query : queries) {
        int l = query[0];
        int r = query[1];
        int t_mod = query[2] % 10;

        vector<int> ans(10, 0);

        // 统计区间内原始数字，再映射到结果中
        for (int d = 0; d < 10; d++) {
            int cnt = pre[r][d] - pre[l - 1][d];
            ans[trans[t_mod][d]] += cnt;
        }

        res.push_back(ans);
    }

    return res;
}

int main() {
    int n;
    cin >> n;

    vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }

    int q;
    cin >> q;

    vector<vector<int>> queries(q, vector<int>(3));
    for (int i = 0; i < q; i++) {
        cin >> queries[i][0] >> queries[i][1] >> queries[i][2];
    }

    vector<vector<int>> res = solve(n, arr, queries);

    for (const auto& row : res) {
        for (int i = 0; i < 10; i++) {
            if (i > 0) cout << " ";
            cout << row[i];
        }
        cout << "\n";
    }

    return 0;
}
