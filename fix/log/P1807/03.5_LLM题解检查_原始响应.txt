## 解题思路

1. **预处理纯数（素数）表**  
   因为给定正整数 $x \le 10^9$，任何一个合数的最小的大于 $1$ 的因数（即最小素因子）不会超过 $\sqrt{10^9} \approx 31623$。我们只需要筛出 $32000$ 以内的所有纯数（素数），用于后续判断素数和分解最小因子。可使用线性筛或埃氏筛预处理。

2. **模拟蜕变过程**  
   从输入的 $x$ 出发，反复执行操作，直到 $x$ 是一个整方数（完全平方数）为止：
   - 若当前 $x$ 是纯数（素数）：执行 $x \leftarrow x - 1$，操作次数加 $1$；
   - 否则，$x$ 是合数，在预先筛出的素数表中寻找第一个能整除 $x$ 的素数 $p$（即 $x$ 的最小素因子），执行 $x \leftarrow x / p$，操作次数加 $1$。

3. **判断整方数**  
   检查一个数是否为整方数，只需计算其平方根向下取整后平方是否等于原数即可。

4. **结果输出**  
   当 $x$ 成为整方数时停止，输出累计操作次数。若初始 $x$ 已经是整方数，则直接输出 $0$。

## 复杂度分析

- **时间复杂度**：预处理素数表需要 $O(\sqrt{N} \log\log \sqrt{N})$，其中 $N=10^9$。  
  蜕变过程中，每次操作要么减 $1$，要么至少除以 $2$，总操作次数不超过约 $60$ 次。每次判断素数和查找最小素因子仅需遍历不超过 $3400$ 个素数，可视为常数时间。因此整体时间复杂度接近于 $O(\sqrt{N} \log\log \sqrt{N})$，实际运行极快。

- **空间复杂度**：存储 $\sqrt{10^9}$ 以内的素数表和标记数组，空间为 $O(\sqrt{N})$。

## 代码实现

### Python

```python
import math

MAXN = 32000
is_composite = [False] * MAXN
primes = []

# 线性筛预处理素数
for i in range(2, MAXN):
    if not is_composite[i]:
        primes.append(i)
    for p in primes:
        if p * i >= MAXN:
            break
        is_composite[p * i] = True
        if i % p == 0:
            break

def is_prime(x: int) -> bool:
    """判断 x 是否为纯数（素数）"""
    if x < 2:
        return False
    if x < MAXN:
        return not is_composite[x]
    # 大于 MAXN 的数，试除到 sqrt(x)
    for i in range(2, int(math.isqrt(x)) + 1):
        if x % i == 0:
            return False
    return True

def is_perfect_square(x: int) -> bool:
    """判断 x 是否为整方数（完全平方数）"""
    r = int(math.isqrt(x))
    return r * r == x

x = int(input())
ans = 0

while not is_perfect_square(x):
    if is_prime(x):          # 纯数 -> 减 1
        x -= 1
    else:                    # 合数 -> 除以最小素因子
        for p in primes:
            if x % p == 0:
                x //= p
                break
    ans += 1

print(ans)
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static final int MAXN = 32000;
    static List<Integer> primes = new ArrayList<>();
    static boolean[] isComposite = new boolean[MAXN];

    static void sieve() {
        for (int i = 2; i < MAXN; i++) {
            if (!isComposite[i]) {
                primes.add(i);
            }
            for (int p : primes) {
                if (p * i >= MAXN) break;
                isComposite[p * i] = true;
                if (i % p == 0) break;
            }
        }
    }

    static boolean isPrime(int x) {
        if (x < 2) return false;
        if (x < MAXN) return !isComposite[x];
        for (int i = 2; i * i <= x; i++) {
            if (x % i == 0) return false;
        }
        return true;
    }

    static boolean isPerfectSquare(int x) {
        int r = (int) Math.sqrt(x);
        return r * r == x;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        sieve();
        int ans = 0;
        while (!isPerfectSquare(x)) {
            if (isPrime(x)) {
                x -= 1;
            } else {
                for (int p : primes) {
                    if (x % p == 0) {
                        x /= p;
                        break;
                    }
                }
            }
            ans++;
        }
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <iostream>
#include <cmath>
#include <vector>

using namespace std;

const int MAXN = 32000;
vector<int> primes;
bool is_composite[MAXN] = {false};

void sieve() {
    for (int i = 2; i < MAXN; ++i) {
        if (!is_composite[i]) {
            primes.push_back(i);
        }
        for (int p : primes) {
            if (p * i >= MAXN) break;
            is_composite[p * i] = true;
            if (i % p == 0) break;
        }
    }
}

bool is_prime(int x) {
    if (x < 2) return false;
    if (x < MAXN) return !is_composite[x];
    for (int i = 2; i * i <= x; ++i) {
        if (x % i == 0) return false;
    }
    return true;
}

bool is_perfect_square(int x) {
    int r = sqrt(x);
    return r * r == x;
}

int main() {
    sieve();
    int x;
    cin >> x;
    int ans = 0;
    while (!is_perfect_square(x)) {
        if (is_prime(x)) {
            x -= 1;
        } else {
            for (int p : primes) {
                if (x % p == 0) {
                    x /= p;
                    break;
                }
            }
        }
        ++ans;
    }
    cout << ans << endl;
    return 0;
}
```
