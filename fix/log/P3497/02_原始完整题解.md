### 思路

* 设 $x$ 的十进制位为 $d_0d_1\ldots d_{n-1}$（从高位到低位）。总数位和为 $S$。
* 想要 $y<x$ 且数位和等于 $S$，一种稳定构造是：在某个位置 $i$（$0 \le i \le n-2$）把高位 $d_i$ 减去 $1$（要求 $d_i \ge 1$），然后把后缀位置 $i+1\ldots n-1$ 的数位重新分配，使后缀的和增加 $1$。这样整体和不变，且由于第 $i$ 位变小，必有 $y<x$。
* 若记原后缀和为 $R=\sum_{j=i+1}^{n-1} d_j$，新后缀目标和为 $R+1$。只要后缀长度 $L=n-i-1$ 满足

  $$
  R+1 \le 9L,
  $$

  就能把和 $R+1$ 分配到 $L$ 个 $[0,9]$ 的数位中。
* 注意 $R \le 9L$ 恒成立，所以上式不成立当且仅当后缀全为 $9$（即 $R=9L$）。因此只需找到一个位置 $i$ 满足：

  * $d_i \ge 1$，且
  * 位置 $i$ 之后的后缀并非全 $9$。
* 若存在这样的 $i$，构造可行；否则无解。无解的情形等价于：对每个 $d_i>0$ 的位置，其后缀全为 $9$。这说明 $x$ 的形态为：若干个前导 $0$，接着一个非零数字，随后全是 $9$（例如 $9, 99, 1999, 00999$ 等）。单个数字 $x\in[1,9]$ 也显然无解；而 $x=0$ 也无解。

### C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	string s;
	if (!(cin >> s)) return 0;

	// 特判：x == 0 或单个数字均无解
	if (s == "0" || s.size() == 1) {
		cout << -1 << '\n';
		return 0;
	}

	int n = (int)s.size();
	vector<int> d(n);
	for (int i = 0; i < n; ++i) d[i] = s[i] - '0';

	// 后缀是否全为9：all9[i] 表示从 i 到 n-1 全为9
	vector<char> all9(n + 1, 1); // all9[n] 视作 true（空后缀），便于写法
	for (int i = n - 1; i >= 0; --i) {
		all9[i] = (d[i] == 9) && all9[i + 1];
	}

	// 后缀和：suf[i] = sum of d[i..n-1]
	vector<int> suf(n + 1, 0);
	for (int i = n - 1; i >= 0; --i) {
		suf[i] = suf[i + 1] + d[i];
	}

	// 找位置 i
	int pos = -1;
	for (int i = 0; i <= n - 2; ++i) {
		if (d[i] >= 1 && !all9[i + 1]) {
			pos = i;
			break;
		}
	}
	if (pos == -1) {
		cout << -1 << '\n';
		return 0;
	}

	// 构造 y
	vector<int> y = d;
	y[pos] -= 1;
	int T = suf[pos + 1] + 1; // 新后缀目标和
	for (int j = n - 1; j >= pos + 1; --j) {
		int put = min(9, T);
		y[j] = put;
		T -= put;
	}
	// 组装字符串并去前导0
	string out;
	out.reserve(n);
	for (int i = 0; i < n; ++i) out.push_back(char('0' + y[i]));
	// 去掉前导0
	int k = 0;
	while (k + 1 < (int)out.size() && out[k] == '0') ++k;
	out.erase(0, k);

	cout << out << '\n';
	return 0;
}
```

```python
import sys

def solve():
    s = sys.stdin.readline().strip()
    # 特判：x == 0 或单个数字均无解
    if s == "0" or len(s) == 1:
        print(-1)
        return

    n = len(s)
    d = [ord(c) - 48 for c in s]

    # 后缀是否全为9
    all9 = [True] * (n + 1)  # all9[n] = True
    for i in range(n - 1, -1, -1):
        all9[i] = (d[i] == 9) and all9[i + 1]

    # 后缀和
    suf = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suf[i] = suf[i + 1] + d[i]

    # 找位置 i
    pos = -1
    for i in range(0, n - 1):
        if d[i] >= 1 and not all9[i + 1]:
            pos = i
            break
    if pos == -1:
        print(-1)
        return

    # 构造 y
    y = d[:]
    y[pos] -= 1
    T = suf[pos + 1] + 1  # 新后缀目标和

    for j in range(n - 1, pos, -1):
        put = min(9, T)
        y[j] = put
        T -= put

    # 组装并去前导0
    out = ''.join(chr(v + 48) for v in y)
    k = 0
    while k + 1 < len(out) and out[k] == '0':
        k += 1
    out = out[k:]
    print(out)

if __name__ == "__main__":
    solve()
```

```java
import java.io.*;
import java.util.*;

public class Main {
	public static void main(String[] args) throws Exception {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		String s = br.readLine();
		if (s == null) return;
		s = s.trim();

		// 特判：x == 0 或单个数字均无解
		if (s.equals("0") || s.length() == 1) {
			System.out.println(-1);
			return;
		}

		int n = s.length();
		int[] d = new int[n];
		for (int i = 0; i < n; i++) d[i] = s.charAt(i) - '0';

		// 后缀是否全为9
		boolean[] all9 = new boolean[n + 1];
		Arrays.fill(all9, true); // all9[n] = true
		for (int i = n - 1; i >= 0; --i) {
			all9[i] = (d[i] == 9) && all9[i + 1];
		}

		// 后缀和
		int[] suf = new int[n + 1];
		for (int i = n - 1; i >= 0; --i) {
			suf[i] = suf[i + 1] + d[i];
		}

		// 找位置 i
		int pos = -1;
		for (int i = 0; i <= n - 2; ++i) {
			if (d[i] >= 1 && !all9[i + 1]) {
				pos = i;
				break;
			}
		}
		if (pos == -1) {
			System.out.println(-1);
			return;
		}

		// 构造 y
		int[] y = Arrays.copyOf(d, n);
		y[pos] -= 1;
		int T = suf[pos + 1] + 1; // 新后缀目标和

		for (int j = n - 1; j >= pos + 1; --j) {
			int put = Math.min(9, T);
			y[j] = put;
			T -= put;
		}

		// 组装并去前导0
		StringBuilder out = new StringBuilder(n);
		for (int v : y) out.append((char) (v + '0'));
		int k = 0;
		while (k + 1 < out.length() && out.charAt(k) == '0') k++;
		System.out.println(out.substring(k));
	}
}
```
