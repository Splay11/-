## 解题思路

把 $x$ 看成十进制数位序列 $d_0d_1\ldots d_{n-1}$（从高位到低位），数位和为 $S$。要找 $y<x$ 且数位和仍为 $S$。

一种稳定构造：在某个位置 $i$（$0 \le i \le n-2$）把 $d_i$ 减 `1`（要求 $d_i \ge 1$），再把后缀 $i+1,\ldots,n-1$ 重新分配，使后缀数位和增加 `1`。整体和不变，且第 $i$ 位变小，必有 $y<x$。

记原后缀和为 $R$，新后缀目标和为 $R+1$，后缀长度为 $L=n-i-1$。只要 $R+1 \le 9L$ 就能把和 $R+1$ 分配到 $L$ 个 $[0,9]$ 的数位中。$R \le 9L$ 恒成立，上式不成立当且仅当后缀全为 `9`。因此只需找到位置 $i$ 满足：

- $d_i \ge 1$；
- 位置 $i$ 之后的后缀并非全 `9`。

若存在这样的 $i$，从右往左贪心把后缀填成目标和 $R+1$；否则无解。无解形态包括：$x=0$、单个数字，以及“若干前导 `0` + 一个非零数字 + 后面全 `9`”（如 `9`、`99`、`1999`）。

最后去掉前导零后输出。

## 复杂度分析

- **时间复杂度**：$O(n)$，其中 $n$ 为 $x$ 的位数。
- **空间复杂度**：$O(n)$，需要存放数位数组与后缀信息。

## 代码实现

### Python

```python
import sys

def solve():
    s = sys.stdin.readline().strip()
    if s == "0" or len(s) == 1:
        print(-1)
        return

    n = len(s)
    d = [ord(c) - 48 for c in s]

    all9 = [True] * (n + 1)
    for i in range(n - 1, -1, -1):
        all9[i] = (d[i] == 9) and all9[i + 1]

    suf = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suf[i] = suf[i + 1] + d[i]

    pos = -1
    for i in range(0, n - 1):
        if d[i] >= 1 and not all9[i + 1]:
            pos = i
            break
    if pos == -1:
        print(-1)
        return

    y = d[:]
    y[pos] -= 1
    T = suf[pos + 1] + 1
    for j in range(n - 1, pos, -1):
        put = min(9, T)
        y[j] = put
        T -= put

    out = ''.join(chr(v + 48) for v in y)
    k = 0
    while k + 1 < len(out) and out[k] == '0':
        k += 1
    out = out[k:]
    print(out)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        if (s == null) return;
        s = s.trim();

        if (s.equals("0") || s.length() == 1) {
            System.out.println(-1);
            return;
        }

        int n = s.length();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) d[i] = s.charAt(i) - '0';

        boolean[] all9 = new boolean[n + 1];
        Arrays.fill(all9, true);
        for (int i = n - 1; i >= 0; --i) {
            all9[i] = (d[i] == 9) && all9[i + 1];
        }

        int[] suf = new int[n + 1];
        for (int i = n - 1; i >= 0; --i) {
            suf[i] = suf[i + 1] + d[i];
        }

        int pos = -1;
        for (int i = 0; i <= n - 2; ++i) {
            if (d[i] >= 1 && !all9[i + 1]) {
                pos = i;
                break;
            }
        }
        if (pos == -1) {
            System.out.println(-1);
            return;
        }

        int[] y = Arrays.copyOf(d, n);
        y[pos] -= 1;
        int T = suf[pos + 1] + 1;
        for (int j = n - 1; j >= pos + 1; --j) {
            int put = Math.min(9, T);
            y[j] = put;
            T -= put;
        }

        StringBuilder out = new StringBuilder(n);
        for (int v : y) out.append((char) (v + '0'));
        int k = 0;
        while (k + 1 < out.length() && out.charAt(k) == '0') k++;
        System.out.println(out.substring(k));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string s;
    if (!(cin >> s)) return 0;

    if (s == "0" || s.size() == 1) {
        cout << -1 << '\n';
        return 0;
    }

    int n = (int)s.size();
    vector<int> d(n);
    for (int i = 0; i < n; ++i) d[i] = s[i] - '0';

    vector<char> all9(n + 1, 1);
    for (int i = n - 1; i >= 0; --i) {
        all9[i] = (d[i] == 9) && all9[i + 1];
    }

    vector<int> suf(n + 1, 0);
    for (int i = n - 1; i >= 0; --i) {
        suf[i] = suf[i + 1] + d[i];
    }

    int pos = -1;
    for (int i = 0; i <= n - 2; ++i) {
        if (d[i] >= 1 && !all9[i + 1]) {
            pos = i;
            break;
        }
    }
    if (pos == -1) {
        cout << -1 << '\n';
        return 0;
    }

    vector<int> y = d;
    y[pos] -= 1;
    int T = suf[pos + 1] + 1;
    for (int j = n - 1; j >= pos + 1; --j) {
        int put = min(9, T);
        y[j] = put;
        T -= put;
    }

    string out;
    out.reserve(n);
    for (int i = 0; i < n; ++i) out.push_back(char('0' + y[i]));
    int k = 0;
    while (k + 1 < (int)out.size() && out[k] == '0') ++k;
    out.erase(0, k);
    cout << out << '\n';
    return 0;
}
```
