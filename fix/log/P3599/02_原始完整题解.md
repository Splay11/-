## **关键观察**

1. 下标的 **$i \mod m$** 不会变化，所以每个模类是彼此独立的。
2. 数组被自然分成了 **$n/m$** 个区块：

   * 第 $1$ 个区块：$a_1, a_2, \dots, a_m$
   * 第 $2$ 个区块：$a_{m+1}, \dots, a_{2m}$
   * 以此类推...
3. 在每个模类内部，我们可以自由交换元素的位置。

---

## **核心思路**

为了让**每组的最大值总和最小**，我们希望**尽量把大的数分散到不同的区块，而不是集中到同一个区块**。

换句话说：

* 假设 $m = 3$, $n = 6$
  下标取模分组：

  * 组 0: $a_1, a_4$
  * 组 1: $a_2, a_5$
  * 组 2: $a_3, a_6$

如果我们在组内排序，将大的数尽量往不同区块的相同位置分配，就能最小化每个区块最大值的和。
## 代码

## C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    int T;
    cin >> T;
    while(T--){
        int n, m;
        cin >> n >> m;
        vector<long long> a(n);
        for(int i = 0; i < n; i++) cin >> a[i];

        // m 个模类
        vector<vector<long long>> groups(m);
        for(int i = 0; i < n; i++){
            groups[i % m].push_back(a[i]);
        }

        // 每个模类排序（升序）
        for(int r = 0; r < m; r++){
            sort(groups[r].begin(), groups[r].end());
        }

        // 按块重新分配
        vector<long long> res(n);
        for(int block = 0; block < n/m; block++){
            for(int r = 0; r < m; r++){
                // 取出当前模类的第 block 个元素
                res[block * m + r] = groups[r][block];
            }
        }

        for(int i = 0; i < n; i++){
            cout << res[i] << (i + 1 == n ? '\n' : ' ');
        }
    }
    return 0;
}
```

## Python 

```python
import sys
input = sys.stdin.read
data = input().strip().split()
t = int(data[0])
index = 1

for _ in range(t):
    n = int(data[index]); m = int(data[index+1])
    index += 2
    a = list(map(int, data[index:index+n]))
    index += n

    # m 个模类
    groups = [[] for _ in range(m)]
    for i in range(n):
        groups[i % m].append(a[i])

    # 每个模类升序排序
    for r in range(m):
        groups[r].sort()

    # 按块重新排列
    res = [0] * n
    for block in range(n // m):
        for r in range(m):
            res[block * m + r] = groups[r][block]

    print(' '.join(map(str, res)))
```

## Java 

```java
import java.util.*;
public class Main {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        while(T-- > 0){
            int n = sc.nextInt();
            int m = sc.nextInt();
            long[] a = new long[n];
            for(int i = 0; i < n; i++) a[i] = sc.nextLong();

            // m 个模类
            List<List<Long>> groups = new ArrayList<>();
            for(int i = 0; i < m; i++) groups.add(new ArrayList<>());
            for(int i = 0; i < n; i++){
                groups.get(i % m).add(a[i]);
            }

            // 每个模类升序排序
            for(int r = 0; r < m; r++){
                Collections.sort(groups.get(r));
            }

            long[] res = new long[n];
            for(int block = 0; block < n/m; block++){
                for(int r = 0; r < m; r++){
                    res[block * m + r] = groups.get(r).get(block);
                }
            }

            for(int i = 0; i < n; i++){
                System.out.print(res[i] + (i + 1 == n ? "\n" : " "));
            }
        }
    }
}
```
