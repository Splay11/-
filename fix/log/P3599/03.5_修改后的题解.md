## 解题思路

下标的 $i \bmod m$ 在同桩换车中不变，因此 $m$ 个同余桩位彼此独立，类内可以任意重排。整列被切成 $n/m$ 段，每段长度为 $m$。

为使各段最大值之和最小，应把每个同余类里的重量按升序排好，再按“第 $k$ 段取该类的第 $k$ 小”回填。这样较重的车被分散到靠后的段，而不会挤在同一段里抬高多个最大值。

## 复杂度分析

时间复杂度 $O(n\log n)$（各同余类排序），空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
data = sys.stdin.read().strip().split()
t = int(data[0])
index = 1

for _ in range(t):
    n = int(data[index]); m = int(data[index + 1])
    index += 2
    a = list(map(int, data[index:index + n]))
    index += n

    groups = [[] for _ in range(m)]
    for i in range(n):
        groups[i % m].append(a[i])
    for r in range(m):
        groups[r].sort()

    res = [0] * n
    for block in range(n // m):
        for r in range(m):
            res[block * m + r] = groups[r][block]

    print(' '.join(map(str, res)))
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        while (T-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            long[] a = new long[n];
            for (int i = 0; i < n; i++) a[i] = sc.nextLong();

            List<List<Long>> groups = new ArrayList<>();
            for (int i = 0; i < m; i++) groups.add(new ArrayList<>());
            for (int i = 0; i < n; i++) {
                groups.get(i % m).add(a[i]);
            }
            for (int r = 0; r < m; r++) {
                Collections.sort(groups.get(r));
            }

            long[] res = new long[n];
            for (int block = 0; block < n / m; block++) {
                for (int r = 0; r < m; r++) {
                    res[block * m + r] = groups.get(r).get(block);
                }
            }
            for (int i = 0; i < n; i++) {
                System.out.print(res[i] + (i + 1 == n ? "\n" : " "));
            }
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    int T;
    cin >> T;
    while(T--){
        int n, m;
        cin >> n >> m;
        vector<long long> a(n);
        for(int i = 0; i < n; i++) cin >> a[i];

        vector<vector<long long>> groups(m);
        for(int i = 0; i < n; i++){
            groups[i % m].push_back(a[i]);
        }
        for(int r = 0; r < m; r++){
            sort(groups[r].begin(), groups[r].end());
        }

        vector<long long> res(n);
        for(int block = 0; block < n/m; block++){
            for(int r = 0; r < m; r++){
                res[block * m + r] = groups[r][block];
            }
        }
        for(int i = 0; i < n; i++){
            cout << res[i] << (i + 1 == n ? '\n' : ' ');
        }
    }
    return 0;
}
```
