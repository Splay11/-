## 解题思路

1. **读取输入数据**：输入为一行字符串，表示一个 Python 风格的二维列表。每个内层列表的最后一个元素是字符串 `'0'` 或 `'1'` 的类别标签，其余元素为整数特征。在 Python 中可直接使用 `eval()` 将其转换为列表对象；在 Java 和 C++ 中需要对字符串进行解析，提取每个样本的标签。
2. **提取标签并统计**：遍历所有样本，获取每个样本的最后一个元素（即标签），并使用计数器（字典或数组）记录标签 `'0'` 和 `'1'` 出现的次数。
3. **计算比例**：设样本总数为 $N$，标签 $k$ 的样本数为 $c_k$，则比例 $p_k = c_k / N$。由于标签只有 `0` 和 `1` 两类，当一类未出现时，其 $p_k=0$，对应的熵贡献为 $0$。
4. **套用混杂度公式**：根据公式 $H = -\sum_{k} p_k \log_2 p_k$ 计算。若 $p_k > 0$，累加 $-p_k \cdot \log_2(p_k)$；若为 $0$ 则直接跳过。
5. **格式化输出**：最终结果保留三位小数输出，可使用格式化字符串（如 `printf("%.3f", H)` 或 `f"{H:.3f}"`）。

## 复杂度分析

- **时间复杂度**：$O(N)$，其中 $N$ 为样本总数。需要遍历一遍所有样本提取标签，每个样本的解析和统计操作在常数时间内完成。
- **空间复杂度**：$O(1)$。只需要常数个变量存储样本总数、两类标签的计数以及最终的熵值，与输入规模无关。

## 代码实现

### Python

```python
import math

# 读取输入，eval 直接解析为 Python 列表
data = eval(input().strip())

# 总样本数
total = len(data)

# 统计两个标签的出现次数
cnt0 = 0
cnt1 = 0
for sample in data:
    label = sample[-1]   # 最后一列为标签
    if label == '0':
        cnt0 += 1
    else:
        cnt1 += 1

# 计算混杂度
entropy = 0.0
if cnt0 > 0:
    p0 = cnt0 / total
    entropy -= p0 * math.log2(p0)
if cnt1 > 0:
    p1 = cnt1 / total
    entropy -= p1 * math.log2(p1)

# 输出保留三位小数
print(f"{entropy:.3f}")
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String line = sc.nextLine().trim();
        sc.close();

        // 去掉最外层方括号
        line = line.substring(1, line.length() - 1);
        int total = 0;
        int cnt0 = 0, cnt1 = 0;

        // 分割每个样本列表：以 "],[" 或 "], [" 作为分隔
        String[] samples = line.split("\\],\\s*\\[");
        total = samples.length;

        for (String sampleStr : samples) {
            // 去掉可能残留的 '[' 或 ']'
            sampleStr = sampleStr.replace("[", "").replace("]", "");
            // 按逗号分割特征和标签
            String[] parts = sampleStr.split(",");
            // 最后一个是标签，形如 '0' 或 '1'
            String labelStr = parts[parts.length - 1].trim();
            // 取出引号内的字符
            char label = labelStr.charAt(1);
            if (label == '0') {
                cnt0++;
            } else {
                cnt1++;
            }
        }

        double entropy = 0.0;
        if (cnt0 > 0) {
            double p0 = (double) cnt0 / total;
            entropy -= p0 * (Math.log(p0) / Math.log(2));
        }
        if (cnt1 > 0) {
            double p1 = (double) cnt1 / total;
            entropy -= p1 * (Math.log(p1) / Math.log(2));
        }

        System.out.printf("%.3f\n", entropy);
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <cmath>
#include <sstream>
using namespace std;

// 字符串分割函数
vector<string> split(const string &s, char delim) {
    vector<string> res;
    stringstream ss(s);
    string item;
    while (getline(ss, item, delim)) {
        res.push_back(item);
    }
    return res;
}

int main() {
    string line;
    getline(cin, line);
    // 去掉首尾空格
    while (!line.empty() && isspace(line.front())) line.erase(line.begin());
    while (!line.empty() && isspace(line.back())) line.pop_back();

    // 去除最外层 '[' 和 ']'
    line = line.substr(1, line.size() - 2);

    int total = 0;
    int cnt0 = 0, cnt1 = 0;

    // 以 "],[" 或 "], [" 分割每个样本，先统一分隔符
    string delim = "],[";
    size_t pos = 0;
    vector<string> samples;
    while ((pos = line.find(delim)) != string::npos) {
        samples.push_back(line.substr(0, pos));
        line.erase(0, pos + delim.length());
        // 跳过可能存在的空格
        if (!line.empty() && line.front() == ' ') line.erase(0, 1);
    }
    if (!line.empty()) samples.push_back(line);

    total = samples.size();

    for (string &sample : samples) {
        // 去掉可能残留的 '[' 或 ']'
        while (!sample.empty() && sample.front() == '[') sample.erase(sample.begin());
        while (!sample.empty() && sample.back() == ']') sample.pop_back();

        // 按逗号分割
        vector<string> parts = split(sample, ',');
        // 最后一个元素为标签
        string labelStr = parts.back();
        // 去除引号和空格
        char label = labelStr[1];  // 格式为 '0' 或 '1'
        if (label == '0') cnt0++;
        else cnt1++;
    }

    double entropy = 0.0;
    if (cnt0 > 0) {
        double p0 = (double)cnt0 / total;
        entropy -= p0 * log2(p0);
    }
    if (cnt1 > 0) {
        double p1 = (double)cnt1 / total;
        entropy -= p1 * log2(p1);
    }

    printf("%.3f\n", entropy);
    return 0;
}
```
