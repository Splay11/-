## 解题思路

每一项校准任务的失败次数只能是 $0,1,2,3$ 之一：成功即停止；同一任务连续失败三次后下一次强制成功，因此不会超过三次失败。

设第 $i$ 项任务单次失败概率为 $p_i$，则

$$
\begin{aligned}
P_i(0) &= 1-p_i,\\
P_i(1) &= p_i(1-p_i),\\
P_i(2) &= p_i^2(1-p_i),\\
P_i(3) &= p_i^3.
\end{aligned}
$$

其中 $P_i(3)=1-P_i(0)-P_i(1)-P_i(2)$，与直接写 $p_i^3$ 一致。

任务彼此独立。令 $f[i][j]$ 表示做完前 $i$ 项任务、失败总次数恰好为 $j$ 的概率。初始 $f[0][0]=1$，转移为

$$
f[i][j+t] \mathrel{+}= f[i-1][j]\cdot P_i(t),\qquad t\in\{0,1,2,3\}.
$$

$j$ 的上界为 $3i$。全部任务结束后，熔断次数是总失败次数 $S$ 整除 `3` 的商，故期望为

$$
\sum_{j=0}^{3n} f[n][j]\cdot \left\lfloor\frac{j}{3}\right\rfloor.
$$

$n\le 500$，二维 DP 规模可接受。

## 复杂度分析

时间复杂度 $O(n^2)$，空间复杂度 $O(n^2)$。

## 代码实现

### Python

```python
n = int(input())
w = [0] + list(map(float, input().split()))
p = [[0.0] * 4 for _ in range(n + 1)]
f = [[0.0] * (n * 3 + 1) for _ in range(n + 1)]

for i in range(1, n + 1):
    p[i][0] = 1 - w[i]
    p[i][1] = w[i] * p[i][0]
    p[i][2] = w[i] * p[i][1]
    p[i][3] = 1 - p[i][0] - p[i][1] - p[i][2]

f[0][0] = 1.0
for i in range(1, n + 1):
    for j in range(i * 3 + 1):
        for t in range(4):
            if j + t <= i * 3:
                f[i][j + t] += f[i - 1][j] * p[i][t]

res = 0.0
for j in range(n * 3 + 1):
    res += f[n][j] * (j // 3)

print("{:.9f}".format(res))
```

### Java

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        double[] w = new double[n + 1];
        for (int i = 1; i <= n; i++) w[i] = in.nextDouble();
        double[][] p = new double[n + 1][4];
        double[][] f = new double[n + 1][n * 3 + 1];
        for (int i = 1; i <= n; i++) {
            p[i][0] = 1 - w[i];
            p[i][1] = w[i] * p[i][0];
            p[i][2] = w[i] * p[i][1];
            p[i][3] = 1 - p[i][0] - p[i][1] - p[i][2];
        }
        f[0][0] = 1.0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= i * 3; j++) {
                for (int t = 0; t <= 3; t++) {
                    if (j + t <= i * 3) {
                        f[i][j + t] += f[i - 1][j] * p[i][t];
                    }
                }
            }
        }
        double res = 0;
        for (int j = 0; j <= n * 3; j++) {
            res += f[n][j] * (j / 3); // 整除，对应熔断次数
        }
        System.out.printf("%.9f%n", res);
    }
}
```

### C++

```cpp
#include <iomanip>
#include <iostream>
#include <vector>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<double> w(n + 1);
    for (int i = 1; i <= n; i++) cin >> w[i];
    vector<vector<double>> p(n + 1, vector<double>(4));
    vector<vector<double>> f(n + 1, vector<double>(n * 3 + 1, 0.0));
    for (int i = 1; i <= n; i++) {
        p[i][0] = 1.0 - w[i];
        p[i][1] = w[i] * p[i][0];
        p[i][2] = w[i] * p[i][1];
        p[i][3] = 1.0 - p[i][0] - p[i][1] - p[i][2];
    }
    f[0][0] = 1.0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= i * 3; j++) {
            for (int t = 0; t <= 3; t++) {
                if (j + t <= i * 3) {
                    f[i][j + t] += f[i - 1][j] * p[i][t];
                }
            }
        }
    }
    double res = 0;
    for (int j = 0; j <= n * 3; j++) {
        res += f[n][j] * (j / 3); // 整除
    }
    cout << fixed << setprecision(9) << res << '\n';
    return 0;
}
```
