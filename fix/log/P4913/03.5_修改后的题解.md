## 解题思路

1. 由于可以任意重新排列零件顺序，为了最大程度减少批次波动，首先将所有零件按加工难度从小到大排序。可以证明，在排序后的序列上划分连续段一定不劣于划分非连续段。
2. 设排序后的难度数组为 $a[0 \dots n-1]$。令 $dp[i]$ 表示前 $i$ 个零件（下标 $0$ 到 $i-1$）所需的最少批次数，初始 $dp[0]=0$。  
   考虑最后一段覆盖区间 $[j, i-1]$，其中 $j \le i-1$。该段长度为 $len = i-j$，需要满足：
   - 长度限制：$len \le K$；
   - 波动限制：$a[i-1] - a[j] \le A + B \times (len-1)$。
3. 将波动条件变形：  
   令 $b[k] = a[k] - B \times k$，则上述条件等价于 $b[j] \ge b[i-1] - A$。  
   因此转移方程为：
   $$
   dp[i] = 1 + \min_{\substack{j \in [\, \max(0, i-K),\; i-1 \,] \\ b[j] \ge b[i-1] - A}} dp[j]
   $$
4. 使用滑动窗口 + 线段树维护区间最小 $dp$ 值。
   - 将所有的 $b$ 值离散化，得到一个有序数组 $vals$。
   - 从左到右枚举右端点 $r = i-1$（即 $i$ 从 $1$ 到 $n$），维护当前窗口内所有的左端点 $j \in [r-K+1, r]$。
   - 对每个离散化后的 $b$ 值所在桶，使用一个优先队列（堆）存放窗口中该 $b$ 值对应的 $dp[j]$，方便支持删除。
   - 整棵线段树维护每个离散值对应的最小 $dp$，支持查询后缀最小值（即所有 $b_j \ge$ 给定阈值的 $dp$ 最小值）。
5. 具体操作：
   - 处理右端点 $r$ 时，先将左端点 $j = r$ 作为新的可转移点加入窗口，在线段树上更新对应桶的最小值；
   - 若窗口长度超过 $K$，移除左端点 $j = r-K$，并更新对应桶和线段树；
   - 然后查询 $b_j \ge b[r] - A$ 的范围内最小的 $dp$ 值，加 $1$ 得到 $dp[r+1]$。
6. 最终答案为 $dp[n]$。

## 复杂度分析

- 排序数组：$O(n \log n)$；
- 构造辅助数组 $b$ 并进行离散化：$O(n \log n)$；
- 枚举 $n$ 个右端点，每个左端点至多加入和删除一次，涉及堆操作和线段树单点更新与区间查询，总复杂度 $O(n \log n)$；
- 空间复杂度：离散化数组、线段树、堆等辅助结构均与 $n$ 同阶，为 $O(n)$；
- 单组数据时间复杂度 $O(n \log n)$，所有测试数据 $n$ 总和不超过 $2\times 10^5$，满足时限要求。

```python
import sys
from bisect import bisect_left
from heapq import heappop, heappush

INF = 10 ** 9


def build_tree(count):
    size = 1
    while size < count:
        size *= 2
    tree = [INF] * (size * 2)
    return tree, size


def update(tree, size, index, value):
    pos = size + index
    tree[pos] = value
    pos //= 2
    while pos:
        tree[pos] = min(tree[pos * 2], tree[pos * 2 + 1])
        pos //= 2


def range_min(tree, size, left, right):
    if left > right:
        return INF
    left += size
    right += size
    answer = INF
    while left <= right:
        if left % 2 == 1:
            answer = min(answer, tree[left])
            left += 1
        if right % 2 == 0:
            answer = min(answer, tree[right])
            right -= 1
        left //= 2
        right //= 2
    return answer


def refresh(bucket_id, add_heaps, remove_heaps, tree, size):
    add_heap = add_heaps[bucket_id]
    remove_heap = remove_heaps[bucket_id]
    while add_heap and remove_heap and add_heap[0] == remove_heap[0]:
        heappop(add_heap)
        heappop(remove_heap)
    value = add_heap[0][0] if add_heap else INF
    update(tree, size, bucket_id, value)


def min_batches(numbers, A, B, K):
    numbers.sort()
    n = len(numbers)
    # b[i] = numbers[i] - B * i
    transformed = [numbers[i] - B * i for i in range(n)]
    # 离散化
    values = sorted(set(transformed))
    value_id = {v: i for i, v in enumerate(values)}

    tree, size = build_tree(len(values))
    add_heaps = [[] for _ in values]
    remove_heaps = [[] for _ in values]

    dp = [INF] * (n + 1)
    dp[0] = 0

    for right in range(n):
        # 将 right 作为新段左端点加入窗口，值为 dp[right]
        bucket_id = value_id[transformed[right]]
        heappush(add_heaps[bucket_id], (dp[right], right))
        refresh(bucket_id, add_heaps, remove_heaps, tree, size)

        # 移除超出窗口的左端点
        expired = right - K
        if expired >= 0:
            bucket_id = value_id[transformed[expired]]
            heappush(remove_heaps[bucket_id], (dp[expired], expired))
            refresh(bucket_id, add_heaps, remove_heaps, tree, size)

        # 查询满足 b_j >= b[right] - A 的最小 dp
        threshold = transformed[right] - A
        first = bisect_left(values, threshold)
        dp[right + 1] = range_min(tree, size, first, len(values) - 1) + 1

    return dp[n]


def main():
    data = sys.stdin.read().strip().split()
    t = int(data[0])
    idx = 1
    out = []
    for _ in range(t):
        n = int(data[idx]); A = int(data[idx+1]); B = int(data[idx+2]); K = int(data[idx+3])
        idx += 4
        numbers = list(map(int, data[idx:idx+n]))
        idx += n
        out.append(str(min_batches(numbers, A, B, K)))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static class SegmentTree {
        private final int size;
        private final int[] tree;
        private final int INF;

        SegmentTree(int count, int inf) {
            this.INF = inf;
            int s = 1;
            while (s < count) s <<= 1;
            this.size = s;
            tree = new int[size * 2];
            Arrays.fill(tree, INF);
        }

        void update(int index, int value) {
            int pos = size + index;
            tree[pos] = value;
            for (pos >>= 1; pos >= 1; pos >>= 1) {
                tree[pos] = Math.min(tree[pos * 2], tree[pos * 2 + 1]);
            }
        }

        int query(int left, int right) {
            if (left > right) return INF;
            left += size;
            right += size;
            int ans = INF;
            while (left <= right) {
                if ((left & 1) == 1) ans = Math.min(ans, tree[left++]);
                if ((right & 1) == 0) ans = Math.min(ans, tree[right--]);
                left >>= 1;
                right >>= 1;
            }
            return ans;
        }
    }

    static int lowerBound(long[] values, long target) {
        int lo = 0, hi = values.length;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (values[mid] < target) lo = mid + 1;
            else hi = mid;
        }
        return lo;
    }

    static void refresh(int id, PriorityQueue<int[]>[] addHeaps,
                        PriorityQueue<int[]>[] removeHeaps, SegmentTree seg, int INF) {
        PriorityQueue<int[]> add = addHeaps[id];
        PriorityQueue<int[]> rem = removeHeaps[id];
        while (!add.isEmpty() && !rem.isEmpty()
                && add.peek()[0] == rem.peek()[0] && add.peek()[1] == rem.peek()[1]) {
            add.poll();
            rem.poll();
        }
        int val = add.isEmpty() ? INF : add.peek()[0];
        seg.update(id, val);
    }

    static int minBatches(long[] a, long A, long B, int K) {
        Arrays.sort(a);
        int n = a.length;
        long[] trans = new long[n];
        for (int i = 0; i < n; i++) {
            trans[i] = a[i] - B * i;
        }

        long[] vals = trans.clone();
        Arrays.sort(vals);
        int m = 0;
        for (int i = 0; i < vals.length; i++) {
            if (i == 0 || vals[i] != vals[m - 1]) {
                vals[m++] = vals[i];
            }
        }
        vals = Arrays.copyOf(vals, m);

        int INF = n + 5;
        SegmentTree seg = new SegmentTree(m, INF);
        Comparator<int[]> cmp = (x, y) -> x[0] != y[0] ? Integer.compare(x[0], y[0])
                                                       : Integer.compare(x[1], y[1]);
        PriorityQueue<int[]>[] addHeaps = new PriorityQueue[m];
        PriorityQueue<int[]>[] remHeaps = new PriorityQueue[m];
        for (int i = 0; i < m; i++) {
            addHeaps[i] = new PriorityQueue<>(cmp);
            remHeaps[i] = new PriorityQueue<>(cmp);
        }

        int[] coord = new int[n];
        int[] dp = new int[n + 1];
        Arrays.fill(dp, INF);
        dp[0] = 0;

        for (int right = 0; right < n; right++) {
            coord[right] = lowerBound(vals, trans[right]);
            // 加入新左端点 right
            addHeaps[coord[right]].offer(new int[]{dp[right], right});
            refresh(coord[right], addHeaps, remHeaps, seg, INF);

            // 删除超窗的左端点
            int expired = right - K;
            if (expired >= 0) {
                remHeaps[coord[expired]].offer(new int[]{dp[expired], expired});
                refresh(coord[expired], addHeaps, remHeaps, seg, INF);
            }

            long threshold = trans[right] - A;
            int first = lowerBound(vals, threshold);
            dp[right + 1] = seg.query(first, m - 1) + 1;
        }
        return dp[n];
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        int T = Integer.parseInt(br.readLine());
        while (T-- > 0) {
            String[] line = br.readLine().split(" ");
            int n = Integer.parseInt(line[0]);
            long A = Long.parseLong(line[1]);
            long B = Long.parseLong(line[2]);
            int K = Integer.parseInt(line[3]);
            long[] a = new long[n];
            String[] parts = br.readLine().split(" ");
            for (int i = 0; i < n; i++) {
                a[i] = Long.parseLong(parts[i]);
            }
            sb.append(minBatches(a, A, B, K)).append('\n');
        }
        System.out.print(sb);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

class SegmentTree {
    int size, INF;
    vector<int> tree;
public:
    SegmentTree(int n, int inf) : INF(inf) {
        size = 1;
        while (size < n) size <<= 1;
        tree.assign(size * 2, INF);
    }
    void update(int idx, int val) {
        int pos = size + idx;
        tree[pos] = val;
        for (pos >>= 1; pos; pos >>= 1)
            tree[pos] = min(tree[pos * 2], tree[pos * 2 + 1]);
    }
    int query(int l, int r) {
        if (l > r) return INF;
        l += size; r += size;
        int ans = INF;
        while (l <= r) {
            if (l & 1) ans = min(ans, tree[l++]);
            if (!(r & 1)) ans = min(ans, tree[r--]);
            l >>= 1; r >>= 1;
        }
        return ans;
    }
};

int minBatches(vector<long long>& a, long long A, long long B, int K) {
    sort(a.begin(), a.end());
    int n = a.size();
    vector<long long> trans(n);
    for (int i = 0; i < n; ++i)
        trans[i] = a[i] - B * i;

    vector<long long> vals = trans;
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());
    int m = vals.size();

    const int INF = n + 5;
    SegmentTree seg(m, INF);
    vector<multiset<int>> buckets(m);

    auto refresh = [&](int id) {
        int val = buckets[id].empty() ? INF : *buckets[id].begin();
        seg.update(id, val);
    };

    vector<int> dp(n + 1, INF);
    dp[0] = 0;
    vector<int> coord(n);
    for (int right = 0; right < n; ++right) {
        coord[right] = lower_bound(vals.begin(), vals.end(), trans[right]) - vals.begin();
        // 加入左端点 right
        buckets[coord[right]].insert(dp[right]);
        refresh(coord[right]);

        // 删除超窗左端点
        int expired = right - K;
        if (expired >= 0) {
            buckets[coord[expired]].erase(buckets[coord[expired]].find(dp[expired]));
            refresh(coord[expired]);
        }

        long long threshold = trans[right] - A;
        int first = lower_bound(vals.begin(), vals.end(), threshold) - vals.begin();
        dp[right + 1] = seg.query(first, m - 1) + 1;
    }
    return dp[n];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;
    while (T--) {
        int n, K;
        long long A, B;
        cin >> n >> A >> B >> K;
        vector<long long> a(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        cout << minBatches(a, A, B, K) << '\n';
    }
    return 0;
}
```
