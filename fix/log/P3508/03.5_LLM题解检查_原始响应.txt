## 解题思路

1. 一个大于 $1$ 的整数恰好有三个不同的正因子，当且仅当它是某个素数的平方。  
   - 设 $n = p^2$（$p$ 为素数），其因子为 $1, p, p^2$ 共三个。  
   - 反之，若 $n$ 恰有三个因子，其素因数分解必须为 $p^2$ 的形式。  
   因此，题目中的“三叶数”等价于**素数的平方**。

2. 给定查询区间 $[L, R]$，三叶数的数量等于满足 $p^2 \in [L, R]$ 的素数 $p$ 的个数。

3. 令 $A = \left\lfloor\sqrt{R}\right\rfloor$，$B = \left\lfloor\sqrt{L-1}\right\rfloor$（当 $L=1$ 时 $B=0$）。  
   则区间内三叶数的个数为  
   $$ \pi(A) - \pi(B) $$  
   其中 $\pi(x)$ 表示不超过 $x$ 的素数个数。

4. 由于 $R \leq 10^{14}$，有 $A \leq \sqrt{10^{14}} = 10^7$。  
   预处理时使用埃氏筛生成 $[2, 10^7]$ 内的所有素数，并存放在有序列表中。  
   对于每个查询，通过二分查找（`upper_bound`）快速求出 $\pi(A)$ 与 $\pi(B)$，相减即得答案。

## 复杂度分析

- **时间复杂度**：  
  埃氏筛预处理 $10^7$ 范围内的素数，时间复杂度 $O(N \log \log N)$，其中 $N = 10^7$。  
  共有 $Q$ 个查询，每次查询在素数列表中进行两次二分查找，单次 $O(\log \pi(N))$，总查询复杂度 $O(Q \log \log N)$。  
  整体可以在时限内轻松完成。

- **空间复杂度**：  
  筛法标记数组占 $O(N)$ 空间，素数列表占 $O(\pi(N))$ 空间，总体为 $O(N)$，其中 $N = 10^7$。

## 代码实现

### Python

```python
import sys
from math import isqrt
from bisect import bisect_right

def sieve_primes(limit: int):
    """埃氏筛，返回 [2, limit] 内的所有素数"""
    comp = bytearray(limit + 1)
    primes = []
    if limit >= 2:
        primes.append(2)
    # 只筛奇数
    for i in range(3, limit + 1, 2):
        if not comp[i]:
            primes.append(i)
            if i * i <= limit:
                step = 2 * i
                comp[i * i : limit + 1 : step] = b'\x01' * ((limit - i * i) // step + 1)
    return primes

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    Q = int(next(it))
    LIMIT = 10_000_000          # sqrt(1e14)
    primes = sieve_primes(LIMIT)

    out = []
    for _ in range(Q):
        L = int(next(it))
        R = int(next(it))
        A = isqrt(R)
        B = isqrt(L - 1) if L > 1 else 0
        cntA = bisect_right(primes, A)
        cntB = bisect_right(primes, B)
        out.append(str(cntA - cntB))

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static final int LIMIT = 10_000_000; // sqrt(1e14)
    static int[] primes;
    static int primeCount;

    public static void main(String[] args) throws IOException {
        sieve(); // 预处理素数表

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int Q = Integer.parseInt(br.readLine());
        StringBuilder sb = new StringBuilder();

        for (int t = 0; t < Q; t++) {
            String[] parts = br.readLine().split(" ");
            long L = Long.parseLong(parts[0]);
            long R = Long.parseLong(parts[1]);

            long A = isqrt(R);
            long B = (L > 1) ? isqrt(L - 1) : 0L;

            int cntA = upperBound(primes, primeCount, (int) A);
            int cntB = upperBound(primes, primeCount, (int) B);
            sb.append(cntA - cntB).append('\n');
        }
        System.out.print(sb);
    }

    /** 整数平方根（向下取整） */
    static long isqrt(long n) {
        long x = (long) Math.sqrt(n);
        while ((x + 1) * (x + 1) <= n) x++;
        while (x * x > n) x--;
        return x;
    }

    /** 埃氏筛，生成 <= LIMIT 的所有素数 */
    static void sieve() {
        boolean[] comp = new boolean[LIMIT + 1];
        primes = new int[664579];   // π(1e7) ≈ 664579
        primeCount = 0;
        if (LIMIT >= 2) primes[primeCount++] = 2;
        for (int i = 3; i <= LIMIT; i += 2) {
            if (!comp[i]) {
                primes[primeCount++] = i;
                if ((long) i * i <= LIMIT) {
                    for (long j = (long) i * i; j <= LIMIT; j += 2L * i) {
                        comp[(int) j] = true;
                    }
                }
            }
        }
    }

    /** 二分查找，返回数组 a 中 <= key 的元素个数 */
    static int upperBound(int[] a, int n, int key) {
        int lo = 0, hi = n;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (a[mid] <= key) lo = mid + 1;
            else hi = mid;
        }
        return lo;
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using ull = unsigned long long;

/** 整数平方根（向下取整） */
static inline ull isqrt(ull n) {
    long double d = sqrtl((long double)n);
    ull x = (ull)d;
    // 处理浮点误差
    while ((__int128)(x + 1) * (x + 1) <= n) ++x;
    while ((__int128)x * x > n) --x;
    return x;
}

/** 埃氏筛，返回 <= limit 的所有素数 */
static vector<int> sieve_primes(int limit) {
    vector<char> comp(limit + 1, 0);
    vector<int> primes;
    if (limit >= 2) primes.push_back(2);
    for (int i = 3; i <= limit; i += 2) {
        if (!comp[i]) {
            primes.push_back(i);
            if ((ll)i * i <= limit) {
                for (ll j = 1LL * i * i; j <= limit; j += 2LL * i) {
                    comp[(size_t)j] = 1;
                }
            }
        }
    }
    return primes;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    const int LIMIT = 10000000;  // sqrt(1e14)
    vector<int> primes = sieve_primes(LIMIT);

    int Q;
    cin >> Q;
    while (Q--) {
        ull L, R;
        cin >> L >> R;

        ull A = isqrt(R);
        ull B = (L > 1) ? isqrt(L - 1) : 0;

        int cntA = upper_bound(primes.begin(), primes.end(), (int)A) - primes.begin();
        int cntB = upper_bound(primes.begin(), primes.end(), (int)B) - primes.begin();
        cout << cntA - cntB << "\n";
    }
    return 0;
}
```
