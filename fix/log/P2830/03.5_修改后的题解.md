## 解题思路

本题要求计算有向无环图中每个节点的“波动幅度”，本质是对于每个起点 $s$，求出从 $s$ 出发沿有向边可以到达的所有节点（包括 $s$ 自身）上可能被记录到的信号强度的最大值与最小值之差。

1. **信号强度的计算**  
   从起点 $s$ 出发，沿路径经过 $d$ 条边到达节点 $v$ 时，记录值等于 $\max(0,\ a_v - d \cdot k)$。  
   因此，对固定的起点 $s$，所有可达记录值就是该起点能产生的“信号集合”，我们需要求该集合的极差。

2. **动态规划定义**  
   记 $dp[u][0]$ 表示从基站 $u$ 出发能获得的最小信号值，$dp[u][1]$ 表示最大信号值。  
   初始时，基站 $u$ 自身的记录值为 $a_u$，所以 $dp[u][0] = dp[u][1] = a_u$。  
   对于 $u$ 的每条出边 $u \to v$，从 $v$ 出发的所有信号值在传递到 $u$ 时都要减去 $k$（相当于多走了一条边），且不能低于 $0$。因此：
   $$dp[u][0] = \min\Bigl(dp[u][0],\ \max(0, dp[v][0] - k)\Bigr)$$
   $$dp[u][1] = \max\Bigl(dp[u][1],\ \max(0, dp[v][1] - k)\Bigr)$$

3. **计算顺序**  
   由于图是有向无环图（DAG），可以采用**记忆化搜索（DFS）**自底向上计算。  
   从所有入度为 $0$ 的基站开始递归，对于每个未访问过的基站，先递归计算其所有后继，再使用后继的 $dp$ 值更新自身。  
   一个基站可能被多个前驱访问，但只需计算一次，之后其他前驱可以直接使用已算好的 $dp$ 值进行更新。

4. **答案计算**  
   基站 $u$ 的波动幅度即为 $dp[u][1] - dp[u][0]$。按顺序输出所有基站的结果即可。

## 复杂度分析

- **时间复杂度**：$O(n + m)$。每个基站和每条光纤在 DFS 中均被访问常数次。
- **空间复杂度**：$O(n + m)$。需要存储邻接表、入度数组、访问标记和 $dp$ 数组，均与输入规模线性相关。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(300000)

def dfs(u):
    vis[u] = True
    dp[u][0] = a[u]   # 最小信号初始化为自身
    dp[u][1] = a[u]   # 最大信号初始化为自身

    for v in adj[u]:
        if not vis[v]:
            dfs(v)
        # 通过边 u->v 传递，衰减 k，且不低于 0
        dp[u][0] = min(dp[u][0], max(0, dp[v][0] - k))
        dp[u][1] = max(dp[u][1], max(0, dp[v][1] - k))

if __name__ == "__main__":
    n, m, k = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    adj = [[] for _ in range(n)]
    du = [0] * n
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1; v -= 1
        adj[u].append(v)
        du[v] += 1

    dp = [[0, 0] for _ in range(n)]
    vis = [False] * n

    # 从所有入度为 0 的基站开始 DFS
    for i in range(n):
        if du[i] == 0 and not vis[i]:
            dfs(i)

    # 输出每个基站的波动幅度
    res = [str(dp[i][1] - dp[i][0]) for i in range(n)]
    print(" ".join(res))
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static int n, m, k;
    static int[] a;
    static List<Integer>[] adj;
    static int[][] dp;
    static boolean[] vis;
    static int[] du;

    public static void dfs(int u) {
        vis[u] = true;
        dp[u][0] = a[u];
        dp[u][1] = a[u];

        for (int v : adj[u]) {
            if (!vis[v]) {
                dfs(v);
            }
            dp[u][0] = Math.min(dp[u][0], Math.max(0, dp[v][0] - k));
            dp[u][1] = Math.max(dp[u][1], Math.max(0, dp[v][1] - k));
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = br.readLine().split(" ");
        n = Integer.parseInt(parts[0]);
        m = Integer.parseInt(parts[1]);
        k = Integer.parseInt(parts[2]);

        a = new int[n];
        parts = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(parts[i]);
        }

        adj = new ArrayList[n];
        for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
        du = new int[n];

        for (int i = 0; i < m; i++) {
            parts = br.readLine().split(" ");
            int u = Integer.parseInt(parts[0]) - 1;
            int v = Integer.parseInt(parts[1]) - 1;
            adj[u].add(v);
            du[v]++;
        }

        dp = new int[n][2];
        vis = new boolean[n];

        for (int i = 0; i < n; i++) {
            if (du[i] == 0 && !vis[i]) {
                dfs(i);
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append(dp[i][1] - dp[i][0]).append(" ");
        }
        System.out.println(sb.toString().trim());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, m, k;
vector<int> a;
vector<vector<int>> adj;
vector<pair<int, int>> dp;
vector<bool> vis;
vector<int> du;

void dfs(int u) {
    vis[u] = true;
    dp[u].first = a[u];   // 最小信号
    dp[u].second = a[u];  // 最大信号

    for (int v : adj[u]) {
        if (!vis[v]) {
            dfs(v);
        }
        dp[u].first = min(dp[u].first, max(0, dp[v].first - k));
        dp[u].second = max(dp[u].second, max(0, dp[v].second - k));
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m >> k;
    a.resize(n);
    adj.resize(n);
    dp.resize(n);
    vis.assign(n, false);
    du.assign(n, 0);

    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        --u; --v;
        adj[u].push_back(v);
        du[v]++;
    }

    for (int i = 0; i < n; ++i) {
        if (du[i] == 0 && !vis[i]) {
            dfs(i);
        }
    }

    for (int i = 0; i < n; ++i) {
        cout << (dp[i].second - dp[i].first) << (i + 1 == n ? '\n' : ' ');
    }

    return 0;
}
```
