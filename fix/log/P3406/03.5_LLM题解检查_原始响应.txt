## 解题思路

本题需要模拟一个带有全局衰减机制的能量核心处理过程。由于每次处理前所有激活类型的能量值都减少相同的量（都是 1），我们可以将“能量值”转化为“能量衰减截止时间”，从而避免显式地对所有类型进行衰减操作。

核心观察：  
设当前时间为 `time`，某个类型的能量值为 $E$，那么它将在未来第 $E$ 个时间步结束时能量值变为 0，即它在时刻 `time + E` 衰减为 0。因此，若我们记录每个激活类型的能量衰减截止时间 `expire`，则只要在某个时间步 `time` 检查 `expire == time` 即可判定该类型刚刚变为未激活状态。

具体算法步骤：

1. **数据结构准备**  
   - 使用字典 `cnt` 记录每种类型出现的累计次数 $k$。  
   - 使用字典 `expire` 记录当前处于激活状态的类型的能量衰减截止时间。  
   - 使用一个小根堆（优先队列）`pq` 存储 `(expire_time, type)`，以便快速找出刚刚衰减到 0 的类型。  
   - 维护变量 `active` 表示当前激活类型的数量，`max_active` 记录历史最大值。  
   - 维护当前时间步 `time`，初始为 0。

2. **按顺序处理每个核心**  
   对输入序列中的每个类型 $c$ 执行以下操作：

   - **时间推进**：`time = time + 1`  
   - **衰减清理**：不断检查堆顶元素，若堆顶的截止时间等于 `time`，说明该类型的能量值恰好在当前步衰减后变为 0。将其从堆中弹出，若该类型确实在 `expire` 字典中（防止重复弹出的情况），则从字典中删除并令 `active = active - 1`。  
   - **出现次数计算**：取得类型 $c$ 的出现次数，令 $k = cnt.get(c, 0) + 1$，并更新 `cnt[c] = k`。增量 $delta = k^2$。  
   - **能量值更新**：  
     * 若类型 $c$ 当前不在 `expire` 中（即衰减后能量值为 0）：  
       * 将其截止时间设置为 `time + delta`，并插入堆中；  
       * `active = active + 1`。  
     * 若类型 $c$ 当前已在 `expire` 中（即衰减后能量值 > 0）：  
       * 其原截止时间为 $old\_expire$，剩余能量为 $old\_expire - time$。  
       * 增加 $delta$ 后新剩余能量为 $(old\_expire - time) + delta$，因此新的截止时间为 $time + 新剩余能量 = old\_expire + delta$。  
       * 更新 `expire[c]` 并将新截止时间入堆（旧记录保留在堆中但会在弹出时被判断丢弃）。  
   - **更新答案**：`max_active = max(max_active, active)`

3. **输出**  
   对每组测试数据输出 `max_active` 的值。

## 复杂度分析

- **时间复杂度**：每个核心最多导致一次堆的插入和一次堆的弹出，堆操作复杂度为 $O(\log k)$，其中 $k$ 是当前激活类型的数量，不超过 $n$。因此总体复杂度为 $O(n \log n)$。所有测试数据 $n$ 之和不超过 $2\times 10^5$，所以总运行时间在可接受范围内。
- **空间复杂度**：需要存储计数字典、截止时间字典和优先队列，最多存储 $n$ 个元素，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
import heapq

def solve() -> None:
    input_data = sys.stdin.read().split()
    it = iter(input_data)
    t = int(next(it))
    out_lines = []
    for _ in range(t):
        n = int(next(it))
        # 读取核心类型序列
        cores = [int(next(it)) for _ in range(n)]
        
        expire = {}          # 记录当前激活类型的衰减截止时间
        cnt = {}             # 记录每种类型的累计出现次数
        pq = []              # 小根堆，存储 (截止时间, 类型)
        active = 0
        max_active = 0
        time = 0
        
        for c in cores:
            time += 1
            
            # 移除刚好在当前时刻衰减到0的类型
            while pq and pq[0][0] == time:
                et, typ = heapq.heappop(pq)
                if expire.get(typ) == et:
                    del expire[typ]
                    active -= 1
            
            # 更新当前类型的出现次数
            k = cnt.get(c, 0) + 1
            cnt[c] = k
            delta = k * k
            
            if c in expire:
                # 原本处于激活状态，直接累加能量
                expire[c] += delta
            else:
                # 原本未激活，重新激活
                expire[c] = time + delta
                active += 1
                
            heapq.heappush(pq, (expire[c], c))
            
            if active > max_active:
                max_active = active
        
        out_lines.append(str(max_active))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        StringBuilder sb = new StringBuilder();
        
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine());
            String[] parts = br.readLine().split(" ");
            int[] cores = new int[n];
            for (int i = 0; i < n; i++) {
                cores[i] = Integer.parseInt(parts[i]);
            }
            
            Map<Integer, Integer> expire = new HashMap<>();       // 截止时间
            Map<Integer, Integer> cnt = new HashMap<>();         // 出现次数
            PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
            
            int active = 0, maxActive = 0, time = 0;
            
            for (int c : cores) {
                time++;
                
                // 衰减清理
                while (!pq.isEmpty() && pq.peek()[0] == time) {
                    int[] top = pq.poll();
                    int et = top[0], typ = top[1];
                    if (expire.getOrDefault(typ, -1) == et) {
                        expire.remove(typ);
                        active--;
                    }
                }
                
                // 计算增量
                int k = cnt.getOrDefault(c, 0) + 1;
                cnt.put(c, k);
                int delta = k * k;
                
                if (expire.containsKey(c)) {
                    // 已经激活，直接增加能量
                    expire.put(c, expire.get(c) + delta);
                } else {
                    // 重新激活
                    expire.put(c, time + delta);
                    active++;
                }
                
                pq.offer(new int[]{expire.get(c), c});
                maxActive = Math.max(maxActive, active);
            }
            
            sb.append(maxActive).append("\n");
        }
        System.out.print(sb);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <unordered_map>
#include <queue>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        vector<int> cores(n);
        for (int i = 0; i < n; ++i) {
            cin >> cores[i];
        }
        
        unordered_map<int, int> expire;   // 截止时间
        unordered_map<int, int> cnt;      // 出现次数
        // 小根堆
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
        
        int active = 0, max_active = 0, time = 0;
        
        for (int c : cores) {
            ++time;
            
            // 衰减清理
            while (!pq.empty() && pq.top().first == time) {
                auto [et, typ] = pq.top();
                pq.pop();
                auto it = expire.find(typ);
                if (it != expire.end() && it->second == et) {
                    expire.erase(it);
                    --active;
                }
            }
            
            // 更新出现次数
            int k = cnt[c] + 1;
            cnt[c] = k;
            int delta = k * k;
            
            auto it = expire.find(c);
            if (it != expire.end()) {
                // 处于激活状态，延长截止时间
                it->second += delta;
            } else {
                // 重新激活
                expire[c] = time + delta;
                ++active;
            }
            
            pq.emplace(expire[c], c);
            if (active > max_active) max_active = active;
        }
        
        cout << max_active << "\n";
    }
    return 0;
}
```
