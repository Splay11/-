## 解题思路

产线切点只能落在相邻元素之间，即班组 $b = a[1..i]$，班组 $c = a[i+1..n]$（$1 \le i < n$）。

预处理前缀和 $S_i = a_1 + \cdots + a_i$，以及后缀按位或 $O_i = a_i | a_{i+1} | \cdots | a_n$。枚举切点 $i$，取 $|S_i - O_{i+1}|$ 的最小值。

## 复杂度分析

时间复杂度 $O(n)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
input = sys.stdin.readline

def solve():
    T = int(input())
    for _ in range(T):
        n = int(input())
        a = list(map(int, input().split()))
        pre = [0] * (n + 1)
        for i in range(n):
            pre[i + 1] = pre[i] + a[i]
        suf = [0] * (n + 2)
        for i in range(n - 1, -1, -1):
            suf[i + 1] = suf[i + 2] | a[i]
        ans = float('inf')
        for i in range(1, n):
            diff = abs(pre[i] - suf[i + 1])
            if diff < ans:
                ans = diff
        print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine().trim());
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine().trim());
            String[] parts = br.readLine().split(" ");
            long[] a = new long[n + 2];
            for (int i = 1; i <= n; i++) a[i] = Long.parseLong(parts[i - 1]);
            long[] pre = new long[n + 2];
            for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
            long[] suf = new long[n + 3];
            for (int i = n; i >= 1; i--) suf[i] = suf[i + 1] | a[i];
            long ans = Long.MAX_VALUE;
            for (int i = 1; i < n; i++) {
                long diff = Math.abs(pre[i] - suf[i + 1]);
                ans = Math.min(ans, diff);
            }
            System.out.println(ans);
        }
    }
}
```

#include <bits/stdc++.h>

int main() {
    ios::sync_with_stdio(false);

    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        vector<long long> a(n + 2), pre(n + 2), suf(n + 2);
        for (int i = 1; i <= n; i++) cin >> a[i];
        for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
        suf[n + 1] = 0;
        for (int i = n; i >= 1; i--) suf[i] = suf[i + 1] | a[i];
        long long ans = LLONG_MAX;
        for (int i = 1; i < n; i++) {
            ans = min(ans, llabs(pre[i] - suf[i + 1]));
        }
        cout << ans << "\n";
    }
    return 0;
}
