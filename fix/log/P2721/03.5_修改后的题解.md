## 解题思路

1. **距离形式**：两座测站之间的距离为曼哈顿距离 $|x_i-x_j|+|y_i-y_j|$。
2. **绝对值拆分**：该式可拆成四种符号组合，等价于比较 $x+y$ 与 $x-y$ 两类线性函数的极值差：
   - $|x_i-x_j|+|y_i-y_j| = \max\bigl(|(x_i+y_i)-(x_j+y_j)|,\ |(x_i-y_i)-(x_j-y_j)|\bigr)$。
3. **预处理极值**：一遍扫描求出所有点的 $x+y$ 的最大值、最小值，以及 $x-y$ 的最大值、最小值。
4. **逐点作答**：对每个点 $(x,y)$，它到其余点的最大距离即为
   $$\max\bigl(|x+y-\max(x+y)|,\ |x+y-\min(x+y)|,\ |x-y-\max(x-y)|,\ |x-y-\min(x-y)|\bigr).$$
   该值一定能在某个点上取到（极值点自身与自己的差为 $0$，不会影响“到其余点”的最大值，因为 $n\ge 2$ 时至少还有另一个点）。

## 复杂度分析

- **时间复杂度**：$O(n)$。两遍线性扫描。
- **空间复杂度**：$O(n)$。存储全部坐标及答案。

## 代码实现

### Python

```python
def manhattan_furthest_points(points):
    max_xy = min_xy = points[0][0] + points[0][1]
    max_x_y = min_x_y = points[0][0] - points[0][1]
    for x, y in points:
        max_xy = max(max_xy, x + y)
        min_xy = min(min_xy, x + y)
        max_x_y = max(max_x_y, x - y)
        min_x_y = min(min_x_y, x - y)
    result = []
    for x, y in points:
        dist = max(abs(x + y - max_xy), abs(x + y - min_xy),
                   abs(x - y - max_x_y), abs(x - y - min_x_y))
        result.append(dist)
    return result

n = int(input())
points = [tuple(map(int, input().split())) for _ in range(n)]
print(*manhattan_furthest_points(points))
```

### Java

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] x = new long[n];
        long[] y = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextLong();
            y[i] = sc.nextLong();
        }
        long maxXY = Long.MIN_VALUE, minXY = Long.MAX_VALUE;
        long maxXminusY = Long.MIN_VALUE, minXminusY = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            maxXY = Math.max(maxXY, x[i] + y[i]);
            minXY = Math.min(minXY, x[i] + y[i]);
            maxXminusY = Math.max(maxXminusY, x[i] - y[i]);
            minXminusY = Math.min(minXminusY, x[i] - y[i]);
        }
        for (int i = 0; i < n; i++) {
            long dist = Math.max(Math.abs(x[i] + y[i] - maxXY), Math.abs(x[i] + y[i] - minXY));
            dist = Math.max(dist, Math.abs(x[i] - y[i] - maxXminusY));
            dist = Math.max(dist, Math.abs(x[i] - y[i] - minXminusY));
            System.out.print(dist + " ");
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<long long> manhattan_furthest_points(const vector<pair<long long, long long>>& points) {
    long long max_xy = LLONG_MIN, min_xy = LLONG_MAX;
    long long max_x_minus_y = LLONG_MIN, min_x_minus_y = LLONG_MAX;
    for (auto &[x, y] : points) {
        max_xy = max(max_xy, x + y);
        min_xy = min(min_xy, x + y);
        max_x_minus_y = max(max_x_minus_y, x - y);
        min_x_minus_y = min(min_x_minus_y, x - y);
    }
    vector<long long> result;
    for (auto &[x, y] : points) {
        long long dist = max({abs(x + y - max_xy), abs(x + y - min_xy),
                              abs(x - y - max_x_minus_y), abs(x - y - min_x_minus_y)});
        result.push_back(dist);
    }
    return result;
}

int main() {
    int n;
    cin >> n;
    vector<pair<long long, long long>> points(n);
    for (int i = 0; i < n; ++i) {
        cin >> points[i].first >> points[i].second;
    }
    vector<long long> result = manhattan_furthest_points(points);
    for (long long dist : result) {
        cout << dist << " ";
    }
    return 0;
}
```
