## 解题思路

题目描述了一棵以城市 $1$ 为根的有根树，旅人在城市 $u$ 时会遵循以下规则：
- 若以 $u$ 为根的子树中存在其他城市（即 $u$ 不是叶子），则前往这些城市中编号最小的那一座；
- 若 $u$ 是叶子，则不再移动。
我们需要求出从每座城市出发能够访问的不同城市总数（包含起点）。

上述过程等价于在树上进行路径压缩：从 $u$ 出发，每次跳到子树（不含 $u$）中编号最小的城市，直到叶子为止。因此，可以定义 $f[u]$ 表示从城市 $u$ 出发能够访问的城市数量。转移关系如下：
- 若 $u$ 是叶子，则 $f[u] = 1$；
- 否则，设 $j$ 为 $u$ 的子树中（不含 $u$）编号最小的城市，则 $f[u] = f[j] + 1$。

为了高效计算每个 $f[u]$，可以采用**树形 DP** 自底向上进行一遍 DFS。具体做法：
1. 令 DFS 函数返回以当前节点 $u$ 为根的子树中（包含 $u$）的最小编号。
2. 对于 $u$，遍历其所有子节点 $v$，递归调用 DFS 并获得子树的整体最小编号，这些编号都不含 $u$ 自身。取其中的最小值记为 $\textit{min\_son}$。
3. 若 $u$ 没有子节点（即为叶子），则 $f[u] = 1$，并向父节点返回 $u$（子树最小编号就是 $u$ 本身）。
4. 否则，根据转移得到 $f[u] = f[\textit{min\_son}] + 1$，并向父节点返回 $\min(u, \textit{min\_son})$。
5. 最终 $f[1], f[2], \dots, f[n]$ 即为答案。

## 复杂度分析

- **时间复杂度**：每个城市和每条边都被访问常数次，整个 DFS 过程为 $O(n)$，输出也为 $O(n)$，总时间复杂度为 $O(n)$。
- **空间复杂度**：邻接表存储树需要 $O(n)$ 空间，递归栈最坏深度为 $O(n)$，额外使用 $f$ 数组 $O(n)$，总空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys
sys.setrecursionlimit(200000)

def solve():
    n = int(sys.stdin.readline())
    g = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v = map(int, sys.stdin.readline().split())
        g[u].append(v)
        g[v].append(u)
    
    f = [0] * (n + 1)
    
    def dfs(u, fa):
        # min_son记录所有子节点的子树最小编号（不含u自身）
        min_son = n + 1
        for v in g[u]:
            if v == fa:
                continue
            min_son = min(min_son, dfs(v, u))
        if min_son == n + 1:        # u是叶子，没有子节点
            f[u] = 1
            return u
        else:
            f[u] = f[min_son] + 1
            return min(u, min_son)
    
    dfs(1, 0)
    print(' '.join(str(f[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.util.*;

public class Main {
    static List<Integer>[] g;
    static int[] f;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            g[u].add(v);
            g[v].add(u);
        }
        f = new int[n + 1];
        dfs(1, 0);
        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            sb.append(f[i]).append(i == n ? "" : " ");
        }
        System.out.println(sb.toString());
    }
    
    // 返回以u为根的子树中（包含u）的最小编号
    static int dfs(int u, int fa) {
        int minSon = Integer.MAX_VALUE;
        for (int v : g[u]) {
            if (v == fa) continue;
            minSon = Math.min(minSon, dfs(v, u));
        }
        if (minSon == Integer.MAX_VALUE) { // 叶子节点
            f[u] = 1;
            return u;
        } else {
            f[u] = f[minSon] + 1;
            return Math.min(u, minSon);
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;
vector<int> g[N];
int f[N];

int dfs(int u, int fa) {
    int minSon = N;      // N 大于所有城市编号，方便取 min
    for (int v : g[u]) {
        if (v == fa) continue;
        minSon = min(minSon, dfs(v, u));
    }
    if (minSon == N) {   // u 是叶子
        f[u] = 1;
        return u;
    } else {
        f[u] = f[minSon] + 1;
        return min(u, minSon);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n;
    cin >> n;
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 0);
    for (int i = 1; i <= n; i++) {
        cout << f[i] << (i == n ? '\n' : ' ');
    }
    return 0;
}
```
