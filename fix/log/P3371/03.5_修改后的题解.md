## 解题思路

正 $k$ 边形要求 $k\ge 3$ 且所有边等长。因此只需统计每种长度出现的次数 $c$，对每个边数 $k=3,4,\ldots,n$，方案数为所有满足 $c\ge k$ 的长度对应的组合数 $C(c,k)$ 之和。

预处理阶乘与阶乘逆元后，每次 $C(c,k)$ 可 $O(1)$ 计算。答案对 $998244353$ 取模。

## 复杂度分析

时间复杂度 $O(n^2)$（每种边数需遍历所有出现过的长度，长度种类不超过 $n$），空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
from collections import Counter

MOD = 998244353

def main():
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    freq = list(Counter(a).values())
    fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = fact[i - 1] * i % MOD
    inv_fact = [1] * (n + 1)
    inv_fact[n] = pow(fact[n], MOD - 2, MOD)
    for i in range(n - 1, -1, -1):
        inv_fact[i] = inv_fact[i + 1] * (i + 1) % MOD
    res = []
    for m in range(3, n + 1):
        total = 0
        for c in freq:
            if c >= m:
                comb = fact[c] * inv_fact[m] % MOD
                comb = comb * inv_fact[c - m] % MOD
                total = (total + comb) % MOD
        res.append(total)
    print(' '.join(map(str, res)))

if __name__ == '__main__':
    main()
```

### Java

```java
import java.util.*;
import java.io.*;

public class Main {
    static final int MOD = 998244353;

    static long pow(long a, long b, long mod) {
        long res = 1;
        a %= mod;
        while (b > 0) {
            if ((b & 1) == 1) res = res * a % mod;
            a = a * a % mod;
            b >>= 1;
        }
        return res;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        StringTokenizer st = new StringTokenizer(br.readLine());
        Map<Integer, Integer> counts = new HashMap<>();
        for (int i = 0; i < n; ++i) {
            int num = Integer.parseInt(st.nextToken());
            counts.put(num, counts.getOrDefault(num, 0) + 1);
        }
        List<Integer> freq = new ArrayList<>(counts.values());
        long[] fact = new long[n + 1];
        fact[0] = 1;
        for (int i = 1; i <= n; ++i) fact[i] = fact[i - 1] * i % MOD;
        long[] invFact = new long[n + 1];
        invFact[n] = pow(fact[n], MOD - 2, MOD);
        for (int i = n - 1; i >= 0; --i) invFact[i] = invFact[i + 1] * (i + 1) % MOD;
        StringBuilder sb = new StringBuilder();
        for (int m = 3; m <= n; ++m) {
            long total = 0;
            for (int c : freq) {
                if (c >= m) {
                    long comb = fact[c] * invFact[m] % MOD;
                    comb = comb * invFact[c - m] % MOD;
                    total = (total + comb) % MOD;
                }
            }
            if (m > 3) sb.append(' ');
            sb.append(total);
        }
        System.out.print(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;

long long qpow(long long a, long long b, long long mod) {
    long long res = 1;
    a %= mod;
    while (b > 0) {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    map<int, int> counts;
    for (int num : a) ++counts[num];
    vector<int> freq;
    for (auto &p : counts) freq.push_back(p.second);
    vector<long long> fact(n + 1, 1), invFact(n + 1, 1);
    for (int i = 1; i <= n; ++i) fact[i] = fact[i - 1] * i % MOD;
    invFact[n] = qpow(fact[n], MOD - 2, MOD);
    for (int i = n - 1; i >= 0; --i) invFact[i] = invFact[i + 1] * (i + 1) % MOD;
    for (int m = 3; m <= n; ++m) {
        long long total = 0;
        for (int c : freq) {
            if (c >= m) {
                long long comb = fact[c] * invFact[m] % MOD;
                comb = comb * invFact[c - m] % MOD;
                total += comb;
                if (total >= MOD) total -= MOD;
            }
        }
        if (m > 3) cout << ' ';
        cout << total;
    }
    return 0;
}
```
