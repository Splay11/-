## 解题思路

本题要求用给定的木棍拼出一个**正多边形**的方案数。根据正多边形的定义，每个正多边形的边数 $k$ 必须满足所有边的长度相等，因此需要统计每个长度出现的次数，并计算对于每个 $k$ 能够组成正 $k$ 边形的方案数。方案数等于所有长度中出现次数 $c \geq k$ 的组合数 $C(c, k)$ 之和。



## 解题步骤

1. **统计频率**：使用哈希表统计每个长度出现的次数。
2. **预处理阶乘和逆元**：为了快速计算组合数 $C(n, k)$，预处理阶乘数组 `fact` 和其逆元数组 `inv_fact`。
3. **计算方案数**：对于每个边数 $k$ 从 3 到 $n$，遍历所有长度的频率，计算对应的组合数之和。



## 复杂度分析

- **时间复杂度**：$O(n^2)$。预处理阶乘和逆元的时间是 $O(n)$，每个边数 $k$ 需要遍历所有频率，每个频率的组合数计算是 $O(1)$。
- **空间复杂度**：$O(n)$，用于存储阶乘、逆元和频率信息。



## Python 代码

```python
import sys
from collections import Counter

MOD = 998244353

def main():
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    counts = Counter(a)
    freq = list(counts.values())
    max_n = n
    
    # 预处理阶乘
    fact = [1] * (max_n + 1)
    for i in range(1, max_n + 1):
        fact[i] = fact[i - 1] * i % MOD
    
    # 预处理阶乘逆元
    inv_fact = [1] * (max_n + 1)
    inv_fact[max_n] = pow(fact[max_n], MOD - 2, MOD)
    for i in range(max_n - 1, -1, -1):
        inv_fact[i] = inv_fact[i + 1] * (i + 1) % MOD
    
    # 计算每个边数的方案数
    res = []
    for m in range(3, n + 1):
        total = 0
        for c in freq:
            if c >= m:
                comb = fact[c] * inv_fact[m] % MOD
                comb = comb * inv_fact[c - m] % MOD
                total = (total + comb) % MOD
        res.append(total)
    
    print(' '.join(map(str, res)))

if __name__ == '__main__':
    main()
```



## Java 代码

```java
import java.util.*;
import java.io.*;

public class Main {
    static final int MOD = 998244353;
    
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; ++i) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        
        // 统计频率
        Map<Integer, Integer> counts = new HashMap<>();
        for (int num : a) {
            counts.put(num, counts.getOrDefault(num, 0) + 1);
        }
        List<Integer> freq = new ArrayList<>(counts.values());
        
        // 预处理阶乘
        int maxN = n;
        long[] fact = new long[maxN + 1];
        fact[0] = 1;
        for (int i = 1; i <= maxN; ++i) {
            fact[i] = fact[i - 1] * i % MOD;
        }
        
        // 预处理阶乘逆元
        long[] invFact = new long[maxN + 1];
        invFact[maxN] = pow(fact[maxN], MOD - 2, MOD);
        for (int i = maxN - 1; i >= 0; --i) {
            invFact[i] = invFact[i + 1] * (i + 1) % MOD;
        }
        
        // 计算每个边数的方案数
        List<Long> resList = new ArrayList<>();
        for (int m = 3; m <= n; ++m) {
            long total = 0;
            for (int c : freq) {
                if (c >= m) {
                    long comb = fact[c] * invFact[m] % MOD;
                    comb = comb * invFact[c - m] % MOD;
                    total = (total + comb) % MOD;
                }
            }
            resList.add(total);
        }
        
        // 输出结果
        for (int i = 0; i < resList.size(); ++i) {
            System.out.print(resList.get(i));
            if (i != resList.size() - 1) System.out.print(" ");
        }
    }
    
    // 快速幂
    static long pow(long a, long b, long mod) {
        long res = 1;
        a %= mod;
        while (b > 0) {
            if ((b & 1) == 1) {
                res = res * a % mod;
            }
            a = a * a % mod;
            b >>= 1;
        }
        return res;
    }
}
```



## C++ 代码

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;

// 快速幂
long long qpow(long long a, long long b, long long mod) {
	long long res = 1;
	a %= mod;
	while (b > 0) {
		if (b & 1) res = res * a % mod;
		a = a * a % mod;
		b >>= 1;
	}
	return res;
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	cin >> n;
	vector<int> a(n);
	for (int i = 0; i < n; ++i) cin >> a[i];

	// 统计频率
	map<int, int> counts;
	for (int num : a) ++counts[num];
	vector<int> freq;
	freq.reserve(counts.size());
	for (auto &p : counts) freq.push_back(p.second);

	// 预处理阶乘和逆元
	int maxN = n;
	vector<long long> fact(maxN + 1, 1), invFact(maxN + 1, 1);
	for (int i = 1; i <= maxN; ++i) fact[i] = fact[i - 1] * i % MOD;
	invFact[maxN] = qpow(fact[maxN], MOD - 2, MOD);
	for (int i = maxN - 1; i >= 0; --i) invFact[i] = invFact[i + 1] * (i + 1) % MOD;

	// 计算每个边数的方案数
	vector<long long> res;
	res.reserve(max(0, n - 2));
	for (int m = 3; m <= n; ++m) {
		long long total = 0;
		for (int c : freq) {
			if (c >= m) {
				long long comb = fact[c] * invFact[m] % MOD;
				comb = comb * invFact[c - m] % MOD;
				total += comb;
				if (total >= MOD) total -= MOD;
			}
		}
		res.push_back(total);
	}

	// 输出结果
	for (size_t i = 0; i < res.size(); ++i) {
		if (i) cout << ' ';
		cout << res[i];
	}
	return 0;
}
```
