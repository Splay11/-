## 解题思路

记 $N_k$ 为“包含 $0,1,\ldots,k-1$ 全部元素的连续片段个数”。某个片段的空号若为 $m$，则它对 $k=1,2,\ldots,m$ 各贡献 `1`，因此

$$
\sum \mathrm{vac} = \sum_{k=1}^{n} N_k.
$$

最大化空号之和，等价于对每个 $k$ 最大化 $N_k$。设 $0,1,\ldots,k-1$ 所在下标区间为 $[L_k,R_k]$（下标从 `1` 到 $n$），则覆盖该区间的片段数为 $N_k = L_k \cdot (n-R_k+1)$。

令 $m=\left\lfloor \dfrac{n+1}{2} \right\rfloor$，把 `0` 放在第 $m$ 个位置，可使 $N_1$ 最大。之后每次把下一个更小的数贴在当前块的某一侧扩张，始终让 $\{0,1,\ldots,k\}$ 占据以中位为中心、长度为 $k+1$ 的连续块。由此得到显式构造：

- 左侧放置 $n-2,n-4,\ldots$（正数），中间放 `0`，右侧按升序补上尚未出现的数。
- $n$ 为奇数时左侧是奇数降序；$n$ 为偶数时左侧是偶数降序。

最大值有闭式：

- $n$ 为偶数：$n(n+2)(2n+5)/24$；
- $n$ 为奇数：$(n+1)(n+3)(2n+1)/24$。

注意中间过程要用 $64$ 位整数，避免乘法溢出。

## 复杂度分析

每组数据用闭式计算答案，再用线性扫描构造序列。

- 时间复杂度：$O(n)$。
- 空间复杂度：$O(n)$。

在 $\sum n \le 2 \times 10^5$ 的限制下可以通过。

## 代码实现

### Python

```python
import sys

def max_sum(n: int) -> int:
    if n % 2 == 0:
        return n * (n + 2) * (2 * n + 5) // 24
    else:
        return (n + 1) * (n + 3) * (2 * n + 1) // 24

def build_perm(n: int):
    a = []
    x = n - 2
    while x > 0:
        a.append(x)
        x -= 2
    a.append(0)
    used = [False] * n
    for v in a:
        used[v] = True
    for v in range(1, n):
        if not used[v]:
            a.append(v)
    return a

def main():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    T = int(next(it))
    out_lines = []
    for _ in range(T):
        n = int(next(it))
        out_lines.append(str(max_sum(n)))
        out_lines.append(' '.join(map(str, build_perm(n))))
    print('\n'.join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static long maxSum(long n) {
        if ((n & 1) == 0) {
            return n * (n + 2) * (2 * n + 5) / 24;
        } else {
            return (n + 1) * (n + 3) * (2 * n + 1) / 24;
        }
    }

    static int[] buildPerm(int n) {
        ArrayList<Integer> list = new ArrayList<>();
        for (int x = n - 2; x > 0; x -= 2) list.add(x);
        list.add(0);
        boolean[] used = new boolean[n];
        for (int v : list) used[v] = true;
        for (int v = 1; v <= n - 1; ++v) if (!used[v]) list.add(v);
        int[] a = new int[n];
        for (int i = 0; i < n; ++i) a[i] = list.get(i);
        return a;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        int T = Integer.parseInt(br.readLine().trim());
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine().trim());
            sb.append(maxSum(n)).append('\n');
            int[] a = buildPerm(n);
            for (int i = 0; i < n; ++i) {
                if (i > 0) sb.append(' ');
                sb.append(a[i]);
            }
            sb.append('\n');
        }
        System.out.print(sb.toString());
    }
}
```

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    if (!(cin >> T)) return 0;
    while (T--) {
        long long n;
        cin >> n;
        long long ans;
        if (n % 2 == 0) {
            ans = n * (n + 2) * (2 * n + 5) / 24;
        } else {
            ans = (n + 1) * (n + 3) * (2 * n + 1) / 24;
        }
        cout << ans << "\n";

        vector<int> a;
        for (long long x = n - 2; x > 0; x -= 2) a.push_back((int)x);
        a.push_back(0);
        vector<char> used(n, 0);
        for (int v : a) used[v] = 1;
        for (int v = 1; v <= (int)n - 1; ++v) if (!used[v]) a.push_back(v);

        for (int i = 0; i < (int)a.size(); ++i) {
            if (i) cout << ' ';
            cout << a[i];
        }
        cout << "\n";
    }
    return 0;
}
