## 解题思路

设 $dp[i][j]$ 表示前 $i$ 段岩芯分成 $j$ 组时，$\mathrm{gcd}$ 之和的最大值。转移枚举第 $j$ 组的起点 $k$：

$$
dp[i][j]=\max_{j\le k\le i}\bigl(dp[k-1][j-1]+\mathrm{gcd}(a_k,\ldots,a_i)\bigr)
$$

其中 $j=1$ 且 $k=1$ 时直接取整段 $\mathrm{gcd}$。

区间 $\mathrm{gcd}$ 用稀疏表预处理，查询 $O(1)$。初始 $dp[0][0]=0$，其余为 $-\infty$。

## 复杂度分析

时间复杂度 $O(n^2 m)$，空间复杂度 $O(nm+n\log n)$。

## 代码实现

### Python

```python
import sys

def gcd(a, b):
    return a if b == 0 else gcd(b, a % b)

def get_gcd(l, r):
    k = Log[r - l + 1]
    return gcd(st[l][k], st[r - (1 << k) + 1][k])

data = sys.stdin.read().split()
n = int(data[0])
m = int(data[1])
a = list(map(int, data[2:n + 2]))

dp = [[-float('inf')] * (m + 1) for _ in range(n + 1)]
st = [[0] * 20 for _ in range(n + 1)]
Log = [0] * (n + 1)

for i in range(1, n + 1):
    st[i][0] = a[i - 1]

Log[1] = 0
for i in range(2, n + 1):
    Log[i] = Log[i // 2] + 1

for j in range(1, Log[n] + 1):
    for i in range(1, n - (1 << j) + 2):
        st[i][j] = gcd(st[i][j - 1], st[i + (1 << (j - 1))][j - 1])

dp[0][0] = 0
for i in range(1, n + 1):
    for j in range(1, min(i, m) + 1):
        for k in range(j, i + 1):
            w = get_gcd(k, i)
            if k == 1 and j == 1:
                dp[i][j] = max(dp[i][j], w)
            elif j > 1:
                dp[i][j] = max(dp[i][j], dp[k - 1][j - 1] + w)

print(dp[n][m])
```

### Java

```java
import java.util.Scanner;

public class Main {
    static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    static int getGCD(int l, int r, int[][] st, int[] Log) {
        int k = Log[r - l + 1];
        return gcd(st[l][k], st[r - (1 << k) + 1][k]);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n + 1];
        int[][] dp = new int[n + 1][m + 1];
        int[][] st = new int[n + 1][20];
        int[] Log = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                dp[i][j] = Integer.MIN_VALUE;
            }
        }
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
            st[i][0] = a[i];
        }
        sc.close();
        Log[1] = 0;
        for (int i = 2; i <= n; i++) {
            Log[i] = Log[i / 2] + 1;
        }
        for (int j = 1; j <= Log[n]; j++) {
            for (int i = 1; i + (1 << j) - 1 <= n; i++) {
                st[i][j] = gcd(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);
            }
        }
        dp[0][0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= Math.min(i, m); j++) {
                for (int k = j; k <= i; k++) {
                    int w = getGCD(k, i, st, Log);
                    if (k == 1 && j == 1) {
                        dp[i][j] = Math.max(dp[i][j], w);
                    } else if (j > 1) {
                        dp[i][j] = Math.max(dp[i][j], dp[k - 1][j - 1] + w);
                    }
                }
            }
        }
        System.out.println(dp[n][m]);
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int gcd(int a, int b) {
    return b ? gcd(b, a % b) : a;
}

int n, m;
int a[505];
int dp[505][505];
int st[505][20];
int Log[505];

int get_gcd(int l, int r) {
    int k = Log[r - l + 1];
    return gcd(st[l][k], st[r - (1 << k) + 1][k]);
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        st[i][0] = a[i];
    }
    Log[1] = 0;
    for (int i = 2; i <= n; i++) {
        Log[i] = Log[i / 2] + 1;
    }
    for (int j = 1; j <= Log[n]; j++) {
        for (int i = 1; i + (1 << j) - 1 <= n; i++) {
            st[i][j] = gcd(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);
        }
    }
    dp[0][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= min(i, m); j++) {
            for (int k = j; k <= i; k++) {
                int w = get_gcd(k, i);
                if (k == 1 && j == 1) {
                    dp[i][j] = max(dp[i][j], w);
                } else if (j > 1) {
                    dp[i][j] = max(dp[i][j], dp[k - 1][j - 1] + w);
                }
            }
        }
    }
    cout << dp[n][m] << '\n';
    return 0;
}
```
