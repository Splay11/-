#思路
考虑动态规划设计状态为dp[i][j],前i个元素分成j个组的最大gcd之和，使用st表预处理方便o(logn)查询区间gcd，初始状态 dp[0][0] = 0，表示不使用任何元素且不分组的情况，对于 dp[i][j]，我们需要找到第 j 组的起始位置 k，并将前 k-1 个元素分成 j-1 组，使得 GCD 之和最大化，将 dp[i][j]更新为dp[k-1][j-1]+get_gcd(k, i)和dp[i][j]的最大值即可.
## 代码如下
### cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

// 辗转相除法求 GCD
int gcd(int a, int b) {
    return b ? gcd(b, a % b) : a;
}

int n, m;
int a[505];
int dp[505][505]; // dp[i][j] 表示前 i 个元素分成 j 个组的最大 GCD 之和
int st[505][20]; // 稀疏表，st[i][j] 表示从 i 开始长度为 2^j 的区间的 GCD 值
int Log[505]; // 预处理 log 值，Log[i] 表示 i 的 log2 值

// 使用稀疏表快速获取区间 [l, r] 的 GCD
int get_gcd(int l, int r) {
    int k = Log[r - l + 1];
    return gcd(st[l][k], st[r - (1 << k) + 1][k]);
}

int main() {
    // 输入数组大小 n 和组数 m
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        st[i][0] = a[i]; // 初始化稀疏表的第一列
    }

    // 预处理 log 值
    Log[1] = 0;
    for (int i = 2; i <= n; i++) {
        Log[i] = Log[i / 2] + 1;
    }

    // 构建稀疏表，用于快速计算区间 GCD
    for (int j = 1; j <= Log[n]; j++) {
        for (int i = 1; i + (1 << j) - 1 <= n; i++) {
            st[i][j] = gcd(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);
        }
    }

    // 动态规划初始化，表示不选任何元素的情况
    dp[0][0] = 0;

    // 进行动态规划求解
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= min(i, m); j++) {
            for (int k = j; k <= i; k++) {
                int w = get_gcd(k, i); // 获取区间 [k, i] 的 GCD 值
                if (k == 1 && j == 1) {
                    // 特殊情况：当 k 和 j 都为 1 时，表示第一组从第一个元素开始
                    dp[i][j] = max(dp[i][j], w);
                } else if (j > 1) {
                    // 更新 dp 值，取最大值
                    dp[i][j] = max(dp[i][j], dp[k - 1][j - 1] + w);
                }
            }
        }
    }

    // 输出最终答案，即前 n 个元素分成 m 个组的最大 GCD 之和
    cout << dp[n][m] << '\n';
    return 0;
}
```

```python
    return a if b == 0 else gcd(b, a % b)

# 读取输入
input = sys.stdin.read
data = input().split()
n = int(data[0])
m = int(data[1])
a = list(map(int, data[2:n+2]))

# 初始化稀疏表和 dp 数组
dp = [[-float('inf')] * (m + 1) for _ in range(n + 1)]
st = [[0] * 20 for _ in range(n + 1)]
Log = [0] * (n + 1)

# 初始化稀疏表的第一列
for i in range(1, n + 1):
    st[i][0] = a[i - 1]

# 预处理 log 值
Log[1] = 0
for i in range(2, n + 1):
    Log[i] = Log[i // 2] + 1

# 构建稀疏表，用于快速计算区间 GCD
for j in range(1, Log[n] + 1):
    for i in range(1, n - (1 << j) + 2):
        st[i][j] = gcd(st[i][j - 1], st[i + (1 << (j - 1))][j - 1])

# 动态规划初始化，表示不选任何元素的情况
```

### java
```java
import java.util.Scanner;

public class Main {
    
    // 辗转相除法求 GCD
    static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    // 使用稀疏表快速获取区间 [l, r] 的 GCD
    static int getGCD(int l, int r, int[][] st, int[] Log) {
        int k = Log[r - l + 1];
        return gcd(st[l][k], st[r - (1 << k) + 1][k]);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        // 输入数组大小 n 和组数 m
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n + 1]; // 1-based indexing
        int[][] dp = new int[n + 1][m + 1]; // dp[i][j] 表示前 i 个元素分成 j 个组的最大 GCD 之和
        int[][] st = new int[n + 1][20]; // 稀疏表，st[i][j] 表示从 i 开始长度为 2^j 的区间的 GCD 值
        int[] Log = new int[n + 1]; // 预处理 log 值，Log[i] 表示 i 的 log2 值

        // 初始化dp数组为负无穷大，以便求最大值
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                dp[i][j] = Integer.MIN_VALUE;
            }
        }

        // 读取数组 a
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
            st[i][0] = a[i]; // 初始化稀疏表的第一列
        }
        sc.close();

        // 预处理 log 值
        Log[1] = 0;
        for (int i = 2; i <= n; i++) {
            Log[i] = Log[i / 2] + 1;
        }

        // 构建稀疏表，用于快速计算区间 GCD
        for (int j = 1; j <= Log[n]; j++) {
            for (int i = 1; i + (1 << j) - 1 <= n; i++) {
                st[i][j] = gcd(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);
            }
        }

        // 动态规划初始化，表示不选任何元素的情况
        dp[0][0] = 0;

        // 进行动态规划求解
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= Math.min(i, m); j++) {
                for (int k = j; k <= i; k++) {
                    int w = getGCD(k, i, st, Log); // 获取区间 [k, i] 的 GCD 值
                    if (k == 1 && j == 1) {
                        // 特殊情况：当 k 和 j 都为 1 时，表示第一组从第一个元素开始
                        dp[i][j] = Math.max(dp[i][j], w);
                    } else if (j > 1) {
                        // 更新 dp 值，取最大值
                        dp[i][j] = Math.max(dp[i][j], dp[k - 1][j - 1] + w);
                    }
                }
            }
        }

        // 输出最终答案，即前 n 个元素分成 m 个组的最大 GCD 之和
        System.out.println(dp[n][m]);
    }
}
```
