## 解题思路

设整条流水线总节拍的质因数分解为 $\mathrm{LCM}(a_1,\dots,a_n)=\prod_p p^{E_p}$，其中 $E_p=\max_i v_p(a_i)$。一段连续工序的 LCM 等于总节拍，当且仅当对每个质数 $p$，该段中至少有一个位置的 $p$ 指数达到 $E_p$。

于是转化为：每个质数 $p$ 对应一类“必须命中”的位置，求最短窗口覆盖全部类别。做法如下：

1. 筛出不超过 $\sqrt{10^9}=31623$ 的质数，对每个 $a_i$ 做质因数分解（相同数值缓存）。
2. 统计每个质数的最大指数 $E_p$。
3. 给每个位置打标签：若该位置上某质数 $p$ 的指数恰好等于 $E_p$，则覆盖了需求 $p$。
4. 双指针维护当前窗口覆盖了多少种质数需求，右扩左缩求最短窗口。

特判：若所有数都是 `1`，总节拍为 `1`，没有任何质因数需求，答案为 `1`。

不直接维护 LCM 乘法：乘法易溢出，且滑窗左端无法“除掉”一个数对 LCM 的贡献。

## 复杂度分析

时间复杂度 $O\left(\sum n\cdot\dfrac{\sqrt{A}}{\log\sqrt{A}}\right)$ 量级（分解为主，滑窗线性），空间复杂度 $O(n+P)$，其中 $A\le 10^9$，$P$ 为筛出的质数个数。在 $\sum n\le 10^5$ 下可以通过。

## 代码实现

### Python

```python
import sys

def sieve(limit=31623):
    bs = [True] * (limit + 1)
    ps = []
    for i in range(2, limit + 1):
        if bs[i]:
            ps.append(i)
            step = i
            start = i * i
            if start <= limit:
                bs[start:limit + 1:step] = [False] * (((limit - start) // step) + 1)
    return ps

PRIMES = sieve()

def factorize(x, cache):
    if x in cache:
        return cache[x]
    y = x
    res = {}
    for p in PRIMES:
        if p * p > y:
            break
        if y % p == 0:
            e = 0
            while y % p == 0:
                y //= p
                e += 1
            res[p] = e
    if y > 1:
        res[y] = res.get(y, 0) + 1
    cache[x] = res
    return res

def solve():
    input_data = sys.stdin.read().strip().split()
    t = int(input_data[0])
    idx = 1
    out_lines = []
    for _ in range(t):
        n = int(input_data[idx]); idx += 1
        arr = list(map(int, input_data[idx:idx + n])); idx += n

        if all(a == 1 for a in arr):
            out_lines.append("1")
            continue

        fcache = {}
        need = {}
        facts = [None] * n
        for i, a in enumerate(arr):
            fac = factorize(a, fcache)
            facts[i] = fac
            for p, e in fac.items():
                if e > need.get(p, 0):
                    need[p] = e

        primes_needed = list(need.keys())
        m = len(primes_needed)
        if m == 0:
            out_lines.append("1")
            continue

        tags = [[] for _ in range(n)]
        pid = {p: k for k, p in enumerate(primes_needed)}
        for i in range(n):
            fac = facts[i]
            for p, E in need.items():
                if fac.get(p, 0) == E:
                    tags[i].append(pid[p])

        cnt = [0] * m
        have = 0
        need_all = m
        ans = n + 1
        l = 0
        for r in range(n):
            for k in tags[r]:
                cnt[k] += 1
                if cnt[k] == 1:
                    have += 1
            while have == need_all:
                ans = min(ans, r - l + 1)
                for k in tags[l]:
                    cnt[k] -= 1
                    if cnt[k] == 0:
                        have -= 1
                l += 1
        out_lines.append(str(ans if ans <= n else n))
    print("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static ArrayList<Integer> primes = new ArrayList<>();

    static void sieve(int limit) {
        boolean[] bs = new boolean[limit + 1];
        Arrays.fill(bs, true);
        for (int i = 2; i <= limit; i++) {
            if (bs[i]) {
                primes.add(i);
                long start = 1L * i * i;
                if (start <= limit) {
                    for (int j = (int) start; j <= limit; j += i) bs[j] = false;
                }
            }
        }
    }

    static Map<Integer, Integer> factorize(int x, HashMap<Integer, Map<Integer, Integer>> cache) {
        Map<Integer, Integer> res = cache.get(x);
        if (res != null) return res;
        int y = x;
        HashMap<Integer, Integer> f = new HashMap<>();
        for (int p : primes) {
            if (1L * p * p > y) break;
            if (y % p == 0) {
                int e = 0;
                while (y % p == 0) {
                    y /= p; e++;
                }
                f.put(p, e);
            }
        }
        if (y > 1) f.put(y, f.getOrDefault(y, 0) + 1);
        cache.put(x, f);
        return f;
    }

    public static void main(String[] args) throws Exception {
        sieve(31623);
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder out = new StringBuilder();
        int T = Integer.parseInt(br.readLine().trim());
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine().trim());
            String[] ss = br.readLine().trim().split("\\s+");
            int[] a = new int[n];
            boolean allOne = true;
            for (int i = 0; i < n; i++) {
                a[i] = Integer.parseInt(ss[i]);
                if (a[i] != 1) allOne = false;
            }
            if (allOne) {
                out.append(1).append('\n');
                continue;
            }
            HashMap<Integer, Map<Integer, Integer>> cache = new HashMap<>();
            HashMap<Integer, Integer> need = new HashMap<>();
            @SuppressWarnings("unchecked")
            Map<Integer, Integer>[] facts = new Map[n];
            for (int i = 0; i < n; i++) {
                Map<Integer, Integer> fac = factorize(a[i], cache);
                facts[i] = fac;
                for (Map.Entry<Integer, Integer> e : fac.entrySet()) {
                    int p = e.getKey(), v = e.getValue();
                    need.put(p, Math.max(need.getOrDefault(p, 0), v));
                }
            }
            if (need.isEmpty()) {
                out.append(1).append('\n');
                continue;
            }
            int m = need.size();
            HashMap<Integer, Integer> pid = new HashMap<>();
            int id = 0;
            for (int p : need.keySet()) pid.put(p, id++);
            ArrayList<Integer>[] tags = new ArrayList[n];
            for (int i = 0; i < n; i++) tags[i] = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                Map<Integer, Integer> fac = facts[i];
                for (Map.Entry<Integer, Integer> e : need.entrySet()) {
                    int p = e.getKey(), E = e.getValue();
                    if (fac.getOrDefault(p, 0) == E) tags[i].add(pid.get(p));
                }
            }
            int[] cnt = new int[m];
            int have = 0, needAll = m;
            int ans = n + 1, l = 0;
            for (int r = 0; r < n; r++) {
                for (int k : tags[r]) {
                    if (cnt[k] == 0) have++;
                    cnt[k]++;
                }
                while (have == needAll) {
                    ans = Math.min(ans, r - l + 1);
                    for (int k : tags[l]) {
                        cnt[k]--;
                        if (cnt[k] == 0) have--;
                    }
                    l++;
                }
            }
            out.append(ans <= n ? ans : n).append('\n');
        }
        System.out.print(out.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> primes;
void sieve(int limit = 31623) {
    vector<bool> bs(limit + 1, true);
    for (int i = 2; i <= limit; ++i) {
        if (bs[i]) {
            primes.push_back(i);
            long long start = 1LL * i * i;
            if (start <= limit) {
                for (int j = (int)start; j <= limit; j += i) bs[j] = false;
            }
        }
    }
}

vector<pair<int,int>> factorize(int x, unordered_map<int, vector<pair<int,int>>> &cache) {
    auto it = cache.find(x);
    if (it != cache.end()) return it->second;
    int y = x;
    vector<pair<int,int>> res;
    for (int p : primes) {
        if (1LL * p * p > y) break;
        if (y % p == 0) {
            int e = 0;
            while (y % p == 0) { y /= p; ++e; }
            res.push_back({p, e});
        }
    }
    if (y > 1) res.push_back({y, 1});
    cache.emplace(x, res);
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    sieve();
    int T;
    if (!(cin >> T)) return 0;
    while (T--) {
        int n; cin >> n;
        vector<int> a(n);
        bool allOne = true;
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
            if (a[i] != 1) allOne = false;
        }
        if (allOne) {
            cout << 1 << "\n";
            continue;
        }
        unordered_map<int, vector<pair<int,int>>> cache;
        vector<vector<pair<int,int>>> facts(n);
        unordered_map<int,int> need;
        for (int i = 0; i < n; ++i) {
            auto fac = factorize(a[i], cache);
            facts[i] = fac;
            for (auto &pe : fac) {
                int p = pe.first, e = pe.second;
                auto it = need.find(p);
                if (it == need.end()) need[p] = e;
                else if (e > it->second) it->second = e;
            }
        }
        if (need.empty()) {
            cout << 1 << "\n";
            continue;
        }
        int m = (int)need.size();
        unordered_map<int,int> pid;
        pid.reserve(m * 2);
        int id = 0;
        for (auto &kv : need) pid[kv.first] = id++;
        vector<vector<int>> tags(n);
        for (int i = 0; i < n; ++i) {
            for (auto &kv : need) {
                int p = kv.first, E = kv.second;
                int ehere = 0;
                for (auto &pe : facts[i]) if (pe.first == p) { ehere = pe.second; break; }
                if (ehere == E) tags[i].push_back(pid[p]);
            }
        }
        vector<int> cnt(m, 0);
        int have = 0, needAll = m;
        int ans = n + 1, l = 0;
        for (int r = 0; r < n; ++r) {
            for (int k : tags[r]) {
                if (cnt[k] == 0) ++have;
                ++cnt[k];
            }
            while (have == needAll) {
                ans = min(ans, r - l + 1);
                for (int k : tags[l]) {
                    --cnt[k];
                    if (cnt[k] == 0) --have;
                }
                ++l;
            }
        }
        cout << (ans <= n ? ans : n) << "\n";
    }
    return 0;
}
```
