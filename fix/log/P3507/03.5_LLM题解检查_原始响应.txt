## 解题思路

1. **核心性质**：对于正整数 $x$，定义其“平方因子核” $k(x)$ 为 $x$ 的质因数分解中次数为奇数的所有质因子的乘积（可通过不断除去平方因子得到）。可以证明，两个正整数 $a$ 和 $b$ 的乘积是完全平方数，当且仅当 $k(a) = k(b)$。

2. **问题转化**：在序列 $a_1, a_2, \dots, a_n$ 中，一对相邻元素 $(a_i, a_{i+1})$ 能贡献得分（乘积为完全平方数）等价于它们的平方因子核相同。不同核的元素相邻不可能得分。

3. **上界分析**：将 $1$ 到 $n$ 的所有整数按照 $k(x)$ 的值划分为若干组。若构造序列时将同一组的元素连续排列，则每组内部会产生 $(|G|-1)$ 个得分位置；不同组的邻接处不会得分。因此总得分最大可能值（上界）为
   $$\sum_{G} (|G|-1) = n - (\text{不同 } k(x) \text{ 的个数})$$
   这个上界是可以达到的。

4. **构造方案**：只需将 $1 \sim n$ 按 $k(x)$ 分组，然后将所有组拼接起来，组内可以保持数字本身的升序或其他任意顺序。最终输出这个排列即可得到最大得分。由于存在多种合法方案，任意一种均可通过。

5. **计算平方因子核**：使用最小质因子（SPF）筛法预处理 $1$ 到 $\max n$ 的 SPF 数组。对于每个 $x$，利用 SPF 不断提取质因子并记录每个质因子的次数奇偶性，将所有奇数次质因子相乘即得 $k(x)$。

6. **实现要点**：所有测试数据的 $n$ 之和不超过 $2\times 10^5$，因此可以一次性预处理到全局最大 $n$，然后对每组 $n$ 计算 $1 \sim n$ 的 $k(x)$，按 $k(x)$ 排序输出即可。

## 复杂度分析

- **时间复杂度**：预处理最小质因子数组需要 $O(\max N \log \log \max N)$，其中 $\max N \le 2\times 10^5$。对于每组 $n$，计算 $n$ 个数的平方因子核并对它们排序，总复杂度 $O(\sum n \log n)$。综合时间复杂度为 $O(\max N \log \log \max N + \sum n \log n)$，在数据范围内非常高效。

- **空间复杂度**：需要存储所有 $n$ 仅需 $O(T)$；SPF 数组 $O(\max N)$；排序过程中临时数组 $O(n)$。整体空间复杂度 $O(\max N + \max n)$，满足题目限制。

## 代码实现

### Python

```python
import sys

# 构建到 maxN 的最小质因子表
def build_spf(maxN: int):
    spf = [0] * (maxN + 1)
    if maxN >= 1:
        spf[1] = 1
    for i in range(2, maxN + 1):
        if spf[i] == 0:
            for j in range(i, maxN + 1, i):
                if spf[j] == 0:
                    spf[j] = i
    return spf

# 计算平方因子核 k(x)：质因子中次数为奇数的乘积
def squarefree_kernel(x: int, spf):
    res = 1
    while x > 1:
        p = spf[x]
        parity = 0
        while x % p == 0:
            x //= p
            parity ^= 1   # 只关心奇偶性
        if parity:
            res *= p
    return res

def main():
    data = list(map(int, sys.stdin.read().strip().split()))
    if not data:
        return
    it = iter(data)
    T = next(it)
    ns = [next(it) for _ in range(T)]
    maxN = max(ns) if ns else 0
    spf = build_spf(maxN)

    out_lines = []
    for n in ns:
        pairs = []
        for i in range(1, n + 1):
            k = squarefree_kernel(i, spf)
            pairs.append((k, i))
        # 按平方因子核排序，核相同时按数值排序（顺序无所谓）
        pairs.sort()
        out_lines.append(' '.join(str(b) for _, b in pairs))
    sys.stdout.write('\n'.join(out_lines))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 快速读入
    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        FastScanner(InputStream is) { in = is; }
        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }
        int nextInt() throws IOException {
            int c, sgn = 1, x = 0;
            do { c = read(); } while (c <= 32);
            if (c == '-') { sgn = -1; c = read(); }
            for (; c > 32; c = read()) x = x * 10 + (c - '0');
            return x * sgn;
        }
    }

    // 构建到 maxN 的最小质因子表
    static int[] buildSpf(int maxN) {
        int[] spf = new int[maxN + 1];
        if (maxN >= 1) spf[1] = 1;
        for (int i = 2; i <= maxN; i++) {
            if (spf[i] == 0) {
                for (int j = i; j <= maxN; j += i) {
                    if (spf[j] == 0) spf[j] = i;
                }
            }
        }
        return spf;
    }

    // 计算平方因子核 k(x)
    static int squarefreeKernel(int x, int[] spf) {
        int res = 1;
        while (x > 1) {
            int p = spf[x];
            int parity = 0;
            while (x % p == 0) {
                x /= p;
                parity ^= 1;
            }
            if (parity == 1) res *= p;
        }
        return res;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        int T = fs.nextInt();
        int[] ns = new int[T];
        int maxN = 0;
        for (int i = 0; i < T; i++) {
            ns[i] = fs.nextInt();
            if (ns[i] > maxN) maxN = ns[i];
        }
        int[] spf = buildSpf(maxN);
        StringBuilder out = new StringBuilder();
        for (int tc = 0; tc < T; tc++) {
            int n = ns[tc];
            int[][] arr = new int[n][2]; // [核, 数值]
            for (int i = 1; i <= n; i++) {
                int k = squarefreeKernel(i, spf);
                arr[i - 1][0] = k;
                arr[i - 1][1] = i;
            }
            // 按核排序，核相同时按数值排序
            Arrays.sort(arr, (a, b) -> {
                if (a[0] != b[0]) return Integer.compare(a[0], b[0]);
                return Integer.compare(a[1], b[1]);
            });
            for (int i = 0; i < n; i++) {
                if (i > 0) out.append(' ');
                out.append(arr[i][1]);
            }
            out.append('\n');
        }
        System.out.print(out.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 计算到 maxN 的最小质因子表
static vector<int> build_spf(int maxN) {
    vector<int> spf(maxN + 1);
    for (int i = 2; i <= maxN; ++i) {
        if (spf[i] == 0) {
            for (long long j = i; j <= maxN; j += i) {
                if (spf[j] == 0) spf[j] = i;
            }
        }
    }
    spf[0] = 0; spf[1] = 1;
    return spf;
}

// 计算平方因子核 k(x)：质因子中幂次为奇数的乘积
static int squarefree_kernel(int x, const vector<int>& spf) {
    int res = 1;
    while (x > 1) {
        int p = spf[x];
        int parity = 0;
        while (x % p == 0) {
            x /= p;
            parity ^= 1;  // 只关心奇偶
        }
        if (parity) res *= p;
    }
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    if (!(cin >> T)) return 0;
    vector<int> ns(T);
    int maxN = 0;
    for (int i = 0; i < T; ++i) {
        cin >> ns[i];
        maxN = max(maxN, ns[i]);
    }

    // 预处理 SPF
    vector<int> spf = build_spf(maxN);

    for (int tc = 0; tc < T; ++tc) {
        int n = ns[tc];
        vector<pair<int, int>> v;
        v.reserve(n);
        for (int i = 1; i <= n; ++i) {
            int k = squarefree_kernel(i, spf);
            v.emplace_back(k, i);
        }
        // 按平方因子核分组，组内顺序任意
        sort(v.begin(), v.end(), [](const auto& a, const auto& b) {
            if (a.first != b.first) return a.first < b.first;
            return a.second < b.second;
        });
        for (int i = 0; i < n; ++i) {
            if (i) cout << ' ';
            cout << v[i].second;
        }
        cout << '\n';
    }
    return 0;
}
```
