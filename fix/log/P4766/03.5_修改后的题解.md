## 解题思路

按给定公式用归一化梯度下降拟合权重 $\mathbf{w}$。目标为

$$
f(\mathbf{w})=\frac{1}{2}\|X\mathbf{w}-\mathbf{y}\|_2^2+\lambda\|\mathbf{w}\|_2^2
$$

梯度为

$$
\nabla f=X^{\top}(X\mathbf{w}-\mathbf{y})+2\lambda\mathbf{w}
$$

先把梯度归一化：

$$
\hat{\mathbf{g}}=\frac{\nabla f}{\max(\|\nabla f\|_2,\varepsilon)}
$$

再按衰减学习率更新：

$$
\eta_t=\frac{\eta_0}{\sqrt{t}},\qquad \mathbf{w}_{t+1}=\mathbf{w}_t-\eta_t\hat{\mathbf{g}}
$$

实现细节：

1. 权重初值为全 `0` 向量。
2. 固定迭代 $T=60$ 步，参数 $\eta_0=0.2$，$\lambda=0.01$，$\varepsilon=10^{-10}$。
3. 若梯度含非有限值，或 $\|\nabla f\|_2<\varepsilon$，则跳过本步更新，但仍计一次迭代。
4. 训练结束后计算 $\hat{y}=X_{\mathrm{test}}\mathbf{w}$，结果 $\ge 0$ 输出 `1`，否则输出 `0`。
5. 权重按四舍五入保留 `6` 位小数，并与预测一起输出 JSON。

## 复杂度分析

设训练集规模为 $n\times d$，测试集为 $m\times d$，迭代次数 $T=60$。每次迭代主要是 $O(nd)$ 的矩阵向量乘，总时间复杂度 $O(Tnd)$，空间复杂度 $O(nd+md)$。

## 代码实现

### Python

```python
import sys
import json
import numpy as np


# 按公式执行归一化梯度下降
def train_ngd(train_X, train_y):
    # 题目给定参数
    eta0 = 0.2
    lam = 0.01
    T = 60
    eps = 1e-10

    # 获取特征维度 d
    d = train_X.shape[1]

    # 权重初始化为全 0 向量
    w = np.zeros(d, dtype=float)

    # 严格按题意迭代 60 步
    for t in range(1, T + 1):
        # 计算梯度：X^T(Xw - y) + 2λw
        grad = train_X.T.dot(train_X.dot(w) - train_y) + 2 * lam * w

        # 若梯度中存在 NaN 或 Inf，则跳过本次更新
        if not np.all(np.isfinite(grad)):
            continue

        # 计算梯度的 L2 范数
        norm = np.linalg.norm(grad, 2)

        # 若梯度范数过小，则跳过本次更新
        if norm < eps:
            continue

        # 梯度归一化
        g_hat = grad / max(norm, eps)

        # 计算当前学习率
        eta = eta0 / np.sqrt(t)

        # 更新权重
        w = w - eta * g_hat

    return w


# 线性输出后按题意取符号
def predict(test_X, w):
    # 计算测试集线性输出
    scores = test_X.dot(w)

    # 结果 >= 0 输出 1，否则输出 0
    pred = (scores >= 0).astype(int)

    return pred.tolist()


def main():
    # 读取单行 JSON 输入
    s = sys.stdin.readline().strip()
    data = json.loads(s)

    # 读取并转为 numpy 数组
    train_X = np.array(data["train_X"], dtype=float)
    train_y = np.array(data["train_y"], dtype=float)
    test_X = np.array(data["test_X"], dtype=float)

    # 调用训练函数得到最终权重
    w = train_ngd(train_X, train_y)

    # 调用预测函数得到测试集结果
    test_pred = predict(test_X, w)

    # 权重按题意保留 6 位小数
    weights = [round(float(x), 6) for x in w]

    # 按要求输出 JSON
    ans = {
        "weights": weights,
        "test_pred": test_pred
    }
    print(json.dumps(ans, ensure_ascii=False))


if __name__ == "__main__":
    main()
```

### Java

```java
import com.google.gson.*;
import java.util.*;

public class Main {
    static final double ETA0 = 0.2;
    static final double LAM = 0.01;
    static final int T = 60;
    static final double EPS = 1e-10;

    static double[][] toMatrix(JsonArray arr) {
        int n = arr.size();
        int d = arr.get(0).getAsJsonArray().size();
        double[][] a = new double[n][d];
        for (int i = 0; i < n; i++) {
            JsonArray row = arr.get(i).getAsJsonArray();
            for (int j = 0; j < d; j++) {
                a[i][j] = row.get(j).getAsDouble();
            }
        }
        return a;
    }

    static double[] toVec(JsonArray arr) {
        double[] v = new double[arr.size()];
        for (int i = 0; i < arr.size(); i++) {
            v[i] = arr.get(i).getAsDouble();
        }
        return v;
    }

    static boolean finite(double[] g) {
        for (double x : g) {
            if (Double.isNaN(x) || Double.isInfinite(x)) return false;
        }
        return true;
    }

    static double[] train(double[][] X, double[] y) {
        int n = X.length;
        int d = X[0].length;
        double[] w = new double[d];
        for (int t = 1; t <= T; t++) {
            double[] xw = new double[n];
            for (int i = 0; i < n; i++) {
                double s = 0;
                for (int j = 0; j < d; j++) s += X[i][j] * w[j];
                xw[i] = s - y[i];
            }
            double[] grad = new double[d];
            for (int j = 0; j < d; j++) {
                double s = 0;
                for (int i = 0; i < n; i++) s += X[i][j] * xw[i];
                grad[j] = s + 2 * LAM * w[j];
            }
            if (!finite(grad)) continue;
            double norm = 0;
            for (double g : grad) norm += g * g;
            norm = Math.sqrt(norm);
            if (norm < EPS) continue;
            double den = Math.max(norm, EPS);
            double eta = ETA0 / Math.sqrt(t);
            for (int j = 0; j < d; j++) {
                w[j] -= eta * (grad[j] / den);
            }
        }
        return w;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String line = sc.nextLine();
        sc.close();

        JsonObject data = JsonParser.parseString(line).getAsJsonObject();
        double[][] trainX = toMatrix(data.getAsJsonArray("train_X"));
        double[] trainY = toVec(data.getAsJsonArray("train_y"));
        double[][] testX = toMatrix(data.getAsJsonArray("test_X"));

        double[] w = train(trainX, trainY);
        int m = testX.length;
        int d = w.length;
        JsonArray weights = new JsonArray();
        for (int j = 0; j < d; j++) {
            double r = Math.round(w[j] * 1e6) / 1e6;
            weights.add(r);
        }
        JsonArray pred = new JsonArray();
        for (int i = 0; i < m; i++) {
            double s = 0;
            for (int j = 0; j < d; j++) s += testX[i][j] * w[j];
            pred.add(s >= 0 ? 1 : 0);
        }
        JsonObject ans = new JsonObject();
        ans.add("weights", weights);
        ans.add("test_pred", pred);
        System.out.println(ans.toString());
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <string>
#include "json.hpp"

using json = nlohmann::json;
using namespace std;

const double ETA0 = 0.2;
const double LAM = 0.01;
const int TT = 60;
const double EPS = 1e-10;

vector<double> train_ngd(const vector<vector<double>>& X, const vector<double>& y) {
    int n = (int)X.size();
    int d = (int)X[0].size();
    vector<double> w(d, 0.0);
    for (int t = 1; t <= TT; t++) {
        vector<double> resid(n, 0.0);
        for (int i = 0; i < n; i++) {
            double s = 0;
            for (int j = 0; j < d; j++) s += X[i][j] * w[j];
            resid[i] = s - y[i];
        }
        vector<double> grad(d, 0.0);
        bool ok = true;
        for (int j = 0; j < d; j++) {
            double s = 0;
            for (int i = 0; i < n; i++) s += X[i][j] * resid[i];
            grad[j] = s + 2 * LAM * w[j];
            if (!isfinite(grad[j])) ok = false;
        }
        if (!ok) continue;
        double norm = 0;
        for (int j = 0; j < d; j++) norm += grad[j] * grad[j];
        norm = sqrt(norm);
        if (norm < EPS) continue;
        double den = max(norm, EPS);
        double eta = ETA0 / sqrt((double)t);
        for (int j = 0; j < d; j++) w[j] -= eta * (grad[j] / den);
    }
    return w;
}

int main() {
    string line;
    getline(cin, line);
    json data = json::parse(line);
    auto trainXj = data["train_X"];
    auto trainYj = data["train_y"];
    auto testXj = data["test_X"];
    int n = (int)trainXj.size();
    int d = (int)trainXj[0].size();
    int m = (int)testXj.size();
    vector<vector<double>> X(n, vector<double>(d));
    vector<double> y(n);
    vector<vector<double>> Xt(m, vector<double>(d));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < d; j++) X[i][j] = trainXj[i][j];
        y[i] = trainYj[i];
    }
    for (int i = 0; i < m; i++)
        for (int j = 0; j < d; j++) Xt[i][j] = testXj[i][j];

    vector<double> w = train_ngd(X, y);
    json weights = json::array();
    for (int j = 0; j < d; j++) {
        double r = round(w[j] * 1e6) / 1e6;
        weights.push_back(r);
    }
    json pred = json::array();
    for (int i = 0; i < m; i++) {
        double s = 0;
        for (int j = 0; j < d; j++) s += Xt[i][j] * w[j];
        pred.push_back(s >= 0 ? 1 : 0);
    }
    json ans;
    ans["weights"] = weights;
    ans["test_pred"] = pred;
    cout << ans.dump() << "\n";
    return 0;
}
```
