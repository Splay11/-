## 解题思路

若火险点数 $m\le 2$，两座瞭望台可直接放在火险点上，答案为 `0`。

否则先把非火险叶子不断剥掉，得到连接所有火险点的核心树。在核心树上求加权直径端点 $a,b$ 及路径 $P$。最优的两座瞭望台可以假定都在 $P$ 上。

每个火险点投影到 $P$ 上某点 $p$，记垂距 $h$，同一 $p$ 取最大垂距 $H(p)$。二分答案 $D$：对每个投影点得到 $P$ 上允许放置瞭望台的区间，用“按右端点排序、贪心放置不超过两个离散点”判定能否覆盖全部区间。二分上界取 $\lceil\mathrm{diam}/2\rceil$。

## 复杂度分析

时间复杂度 $O(n\log W)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
from collections import deque

sys.setrecursionlimit(1 << 25)
input = sys.stdin.readline

n, m = map(int, input().split())
alarms = list(map(int, input().split()))
is_alarm = [0] * (n + 1)
for r in alarms:
	is_alarm[r] = 1

g = [[] for _ in range(n + 1)]
for _ in range(n - 1):
	u, v, w = map(int, input().split())
	g[u].append((v, w))
	g[v].append((u, w))

# 特判
if m <= 2:
	print(0)
	sys.exit(0)

# 核心化：剥离非告警节点叶子
deg = [len(g[i]) for i in range(n + 1)]
alive = [True] * (n + 1)
q = deque()
for i in range(1, n + 1):
	if deg[i] == 1 and is_alarm[i] == 0:
		q.append(i)

while q:
	u = q.popleft()
	if not alive[u]:
		continue
	alive[u] = False
	for v, _ in g[u]:
		if not alive[v]:
			continue
		deg[v] -= 1
		if deg[v] == 1 and is_alarm[v] == 0:
			q.append(v)

start = -1
for r in alarms:
	if alive[r]:
		start = r
		break
if start == -1:
	print(0)
	sys.exit(0)

def farthest(s, need_parent=False):
	parent = [-1] * (n + 1)
	dist = [-1] * (n + 1)
	st = [s]
	parent[s] = -1
	dist[s] = 0
	best_node, best_dist = s, 0
	while st:
		u = st.pop()
		for v, w in g[u]:
			if not alive[v] or v == parent[u]:
				continue
			parent[v] = u
			dist[v] = dist[u] + w
			st.append(v)
			if dist[v] > best_dist:
				best_dist = dist[v]
				best_node = v
	return (best_node, best_dist, parent) if need_parent else (best_node, best_dist)

# 直径两次 DFS
a, _ = farthest(start)
b, diam, parent = farthest(a, need_parent=True)

# 恢复路径 a->b
path = []
x = b
while x != -1:
	path.append(x)
	x = parent[x]
path.reverse()

# onPath & pos
on_path = [False] * (n + 1)
for u in path:
	on_path[u] = True

def get_w(u, v):
	for to, w in g[u]:
		if to == v:
			return w
	return 0

L = len(path)
pos = [0] * L
for i in range(1, L):
	pos[i] = pos[i - 1] + get_w(path[i - 1], path[i])

# 计算 H 与约束点
constraints = []  # (pos, H)
NEG = -1

for i in range(L):
	u = path[i]
	H = NEG
	if is_alarm[u]:
		H = max(H, 0)
	for v, w in g[u]:
		if (not alive[v]) or on_path[v]:
			continue
		# 分支深度：以 u-v 进入分支
		best = NEG
		st = [(v, u, w)]
		while st:
			x, p, d = st.pop()
			if is_alarm[x]:
				best = max(best, d)
			for y, w2 in g[x]:
				if (not alive[y]) or y == p or on_path[y]:
					continue
				st.append((y, x, d + w2))
		if best != NEG:
			H = max(H, best)
	if H != NEG:
		constraints.append((pos[i], H))

cand = pos[:]  # 候选中心位置

def feasible(D):
	segs = []
	for p, h in constraints:
		if D < h:
			return False
		r = D - h
		segs.append((p - r, p + r))
	segs.sort(key=lambda x: (x[1], x[0]))
	used = 0
	placed = -10**30
	from bisect import bisect_right
	for Lb, Rb in segs:
		if Lb <= placed <= Rb:
			continue
		idx = bisect_right(cand, Rb) - 1
		if idx < 0:
			return False
		if cand[idx] < Lb:
			return False
		placed = cand[idx]
		used += 1
		if used > 2:
			return False
	return True

lo, hi = 0, (diam + 1) // 2
while lo < hi:
	mid = (lo + hi) // 2
	if feasible(mid):
		hi = mid
	else:
		lo = mid + 1
print(lo)
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
	static class Edge {
		int to, w;
		Edge(int t, int ww) { to = t; w = ww; }
	}
	public static void main(String[] args) throws Exception {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(st.nextToken());
		int[] alarmList = new int[m];
		boolean[] isAlarm = new boolean[n + 1];
		st = new StringTokenizer(br.readLine());
		for (int i = 0; i < m; ++i) {
			alarmList[i] = Integer.parseInt(st.nextToken());
			isAlarm[alarmList[i]] = true;
		}
		ArrayList<Edge>[] g = new ArrayList[n + 1];
		for (int i = 1; i <= n; ++i) g[i] = new ArrayList<>();
		for (int i = 0; i < n - 1; ++i) {
			st = new StringTokenizer(br.readLine());
			int u = Integer.parseInt(st.nextToken());
			int v = Integer.parseInt(st.nextToken());
			int w = Integer.parseInt(st.nextToken());
			g[u].add(new Edge(v, w));
			g[v].add(new Edge(u, w));
		}

		// 特判
		if (m <= 2) {
			System.out.println(0);
			return;
		}

		// 核心化：剥离非告警节点叶子
		int[] deg = new int[n + 1];
		boolean[] alive = new boolean[n + 1];
		for (int i = 1; i <= n; ++i) {
			deg[i] = g[i].size();
			alive[i] = true;
		}
		ArrayDeque<Integer> q = new ArrayDeque<>();
		for (int i = 1; i <= n; ++i) {
			if (deg[i] == 1 && !isAlarm[i]) q.add(i);
		}
		while (!q.isEmpty()) {
			int u = q.poll();
			if (!alive[u]) continue;
			alive[u] = false;
			for (Edge e : g[u]) {
				int v = e.to;
				if (!alive[v]) continue;
				if (--deg[v] == 1 && !isAlarm[v]) q.add(v);
			}
		}

		int start = -1;
		for (int r : alarmList) if (alive[r]) { start = r; break; }
		if (start == -1) {
			System.out.println(0);
			return;
		}

		// 直径两次 DFS
		class Pair { int node; long dist; Pair(int a,long b){node=a;dist=b;} }
		java.util.function.Function<Boolean, Pair> dummy = null;

		// 返回最远点与距离；needParent=true 时记录父亲
		class FarResult { int node; long dist; int[] parent; FarResult(int a,long b,int[] p){node=a;dist=b;parent=p;} }
		java.util.function.BiFunction<Integer, Boolean, FarResult> farthest = (s, needParent) -> {
			int[] parent = new int[n + 1];
			Arrays.fill(parent, -1);
			long[] dist = new long[n + 1];
			Arrays.fill(dist, -1);
			int[] stn = new int[n];
			int[] stp = new int[n];
			int sp = 0;
			stn[sp] = s; stp[sp] = -1; sp++;
			parent[s] = -1; dist[s] = 0;
			int bestNode = s; long bestDist = 0;
			while (sp > 0) {
				sp--;
				int u = stn[sp], p = stp[sp];
				for (Edge e : g[u]) {
					int v = e.to;
					if (!alive[v] || v == p) continue;
					parent[v] = u;
					dist[v] = dist[u] + e.w;
					stn[sp] = v; stp[sp] = u; sp++;
					if (dist[v] > bestDist) { bestDist = dist[v]; bestNode = v; }
				}
			}
			return new FarResult(bestNode, bestDist, needParent ? parent : null);
		};

		FarResult fa = farthest.apply(start, false);
		int a = fa.node;
		FarResult fb = farthest.apply(a, true);
		int b = fb.node;
		long diam = fb.dist;
		int[] parent = fb.parent;

		// 恢复路径 a->b
		ArrayList<Integer> path = new ArrayList<>();
		for (int x = b; x != -1; x = parent[x]) path.add(x);
		Collections.reverse(path);

		boolean[] onPath = new boolean[n + 1];
		for (int u : path) onPath[u] = true;

		java.util.function.BiFunction<Integer, Integer, Integer> getW = (u, v) -> {
			for (Edge e : g[u]) if (e.to == v) return e.w;
			return 0;
		};

		int L = path.size();
		long[] pos = new long[L];
		for (int i = 1; i < L; ++i) pos[i] = pos[i - 1] + getW.apply(path.get(i - 1), path.get(i));

		// 计算约束 (pos, H)
		ArrayList<long[]> constraints = new ArrayList<>();
		final long NEG = -1;

		for (int i = 0; i < L; ++i) {
			int u = path.get(i);
			long H = NEG;
			if (isAlarm[u]) H = Math.max(H, 0L);
			for (Edge e : g[u]) {
				int v = e.to;
				if (!alive[v] || onPath[v]) continue;
				// 分支深度
				long best = NEG;
				int[] stx = new int[n];
				int[] stp2 = new int[n];
				long[] std = new long[n];
				int sp2 = 0;
				stx[sp2] = v; stp2[sp2] = u; std[sp2] = e.w; sp2++;
				while (sp2 > 0) {
					sp2--;
					int x = stx[sp2], p2 = stp2[sp2];
					long d = std[sp2];
					if (isAlarm[x]) best = Math.max(best, d);
					for (Edge ee : g[x]) {
						int y = ee.to;
						if (!alive[y] || y == p2 || onPath[y]) continue;
						stx[sp2] = y; stp2[sp2] = x; std[sp2] = d + ee.w; sp2++;
					}
				}
				if (best != NEG) H = Math.max(H, best);
			}
			if (H != NEG) constraints.add(new long[]{pos[i], H});
		}

		long[] cand = Arrays.copyOf(pos, pos.length);

		java.util.function.LongPredicate feasible = (D) -> {
			long[][] segs = new long[constraints.size()][2];
			for (int i = 0; i < constraints.size(); ++i) {
				long p = constraints.get(i)[0], h = constraints.get(i)[1];
				if (D < h) return false;
				long r = D - h;
				segs[i][0] = p - r;
				segs[i][1] = p + r;
			}
			Arrays.sort(segs, (s1, s2) -> {
				if (s1[1] != s2[1]) return Long.compare(s1[1], s2[1]);
				return Long.compare(s1[0], s2[0]);
			});
			int used = 0;
			long placed = Long.MIN_VALUE / 4;
			for (long[] sg : segs) {
				long Lb = sg[0], Rb = sg[1];
				if (placed >= Lb && placed <= Rb) continue;
				int idx = upperBound(cand, Rb) - 1;
				if (idx < 0) return false;
				if (cand[idx] < Lb) return false;
				placed = cand[idx];
				if (++used > 2) return false;
			}
			return true;
		};

		long lo = 0, hi = (diam + 1) / 2;
		while (lo < hi) {
			long mid = (lo + hi) >>> 1;
			if (feasible.test(mid)) hi = mid;
			else lo = mid + 1;
		}
		System.out.println(lo);
	}

	static int upperBound(long[] a, long x) {
		int l = 0, r = a.length;
		while (l < r) {
			int md = (l + r) >>> 1;
			if (a[md] <= x) l = md + 1;
			else r = md;
		}
		return l;
	}
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Edge { int to; int w; };

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int n, m;
	if (!(cin >> n >> m)) return 0;
	vector<int> alarms(m);
	vector<int> isAlarm(n + 1, 0);
	for (int i = 0; i < m; ++i) { cin >> alarms[i]; isAlarm[alarms[i]] = 1; }
	vector<vector<Edge>> g(n + 1);
	for (int i = 0; i < n - 1; ++i) {
		int u, v, w; cin >> u >> v >> w;
		g[u].push_back({v, w});
		g[v].push_back({u, w});
	}

	// 特判：若告警节点数 <= 2，两个中心可放在告警节点处，答案为 0
	if (m <= 2) { cout << 0 << '\n'; return 0; }

	// 1) 核心化：剥离非告警节点叶子
	vector<int> deg(n + 1, 0);
	vector<char> alive(n + 1, 1);
	for (int i = 1; i <= n; ++i) deg[i] = (int)g[i].size();
	deque<int> q;
	for (int i = 1; i <= n; ++i) if (deg[i] == 1 && !isAlarm[i]) q.push_back(i);
	while (!q.empty()) {
		int u = q.front(); q.pop_front();
		if (!alive[u]) continue;
		alive[u] = 0;
		for (auto &e : g[u]) {
			int v = e.to;
			if (!alive[v]) continue;
			if (--deg[v] == 1 && !isAlarm[v]) q.push_back(v);
		}
	}

	// 找一个仍然存活的告警节点作为起点
	int start = -1;
	for (int r : alarms) if (alive[r]) { start = r; break; }
	if (start == -1) { cout << 0 << '\n'; return 0; }

	// 2) 直径两次 DFS（迭代）: 返回最远结点及距离，并可记录父亲
	auto farthest = [&](int s, vector<int>* outParent)->pair<int, long long> {
		vector<int> parent(n + 1, -1);
		vector<long long> dist(n + 1, -1);
		vector<int> st; st.reserve(n);
		st.push_back(s);
		parent[s] = -1; dist[s] = 0;
		pair<int, long long> best = {s, 0};
		while (!st.empty()) {
			int u = st.back(); st.pop_back();
			for (auto &e : g[u]) {
				int v = e.to;
				if (!alive[v] || v == parent[u]) continue;
				parent[v] = u;
				dist[v] = dist[u] + e.w;
				st.push_back(v);
				if (dist[v] > best.second) best = {v, dist[v]};
			}
		}
		if (outParent) *outParent = move(parent);
		return best;
	};

	// 第一次：从 start 找 a
	auto pa = farthest(start, nullptr);
	int a = pa.first;
	// 第二次：从 a 找 b，并记录父亲以恢复路径
	vector<int> parent;
	auto pb = farthest(a, &parent);
	int b = pb.first;
	long long diam = pb.second;

	// 恢复 a->b 的路径
	vector<int> path;
	for (int x = b; x != -1; x = parent[x]) path.push_back(x);
	reverse(path.begin(), path.end()); // path[0]=a, path.back()=b

	// 预处理：onPath 标记，路径上位置 pos（从 a 起累计距离）
	vector<char> onPath(n + 1, 0);
	for (int u : path) onPath[u] = 1;

	auto getW = [&](int u, int v)->int {
		for (auto &e : g[u]) if (e.to == v) return e.w;
		return 0;
	};

	int L = (int)path.size();
	vector<long long> pos(L, 0);
	for (int i = 1; i < L; ++i) pos[i] = pos[i - 1] + getW(path[i - 1], path[i]);

	// 3) 计算每个路径结点的 H（其所属“侧枝/自身”内到告警节点的最大距离），并仅保留有告警节点投影的结点
	const long long NEG = -1; // 无告警节点
	vector<pair<long long,long long>> constraints; // (pos[u], H[u])
	constraints.reserve(L);

	for (int i = 0; i < L; ++i) {
		int u = path[i];
		long long H = NEG;
		if (isAlarm[u]) H = max(H, 0LL);
		for (auto &e : g[u]) {
			int v = e.to;
			if (!alive[v] || onPath[v]) continue;
			// 以 u-v 进入该分支，求该分支内到告警节点的最大距离
			long long best = NEG;
			vector<tuple<int,int,long long>> st;
			st.emplace_back(v, u, (long long)e.w);
			while (!st.empty()) {
				auto [x, p, d] = st.back(); st.pop_back();
				if (isAlarm[x]) best = max(best, d);
				for (auto &ee : g[x]) {
					int y = ee.to;
					if (!alive[y] || y == p || onPath[y]) continue;
					st.emplace_back(y, x, d + (long long)ee.w);
				}
			}
			if (best != NEG) H = max(H, best);
		}
		if (H != NEG) constraints.emplace_back(pos[i], H);
	}
	// 候选中心只能选在路径结点（离散坐标集合）
	vector<long long> cand = pos;

	// 4) 判定函数：给定 D，是否能用至多 2 个路径结点命中所有可行区间
	auto feasible = [&](long long D)->bool {
		vector<pair<long long,long long>> segs;
		segs.reserve(constraints.size());
		for (auto &pr : constraints) {
			long long p = pr.first, h = pr.second;
			if (D < h) return false;
			long long r = D - h;
			segs.emplace_back(p - r, p + r);
		}
		sort(segs.begin(), segs.end(), [](auto &a, auto &b){
			if (a.second != b.second) return a.second < b.second;
			return a.first < b.first;
		});
		int used = 0;
		long long placed = LLONG_MIN/4;
		for (auto &sg : segs) {
			long long Lb = sg.first, Rb = sg.second;
			if (placed >= Lb && placed <= Rb) continue; // 已覆盖
			// 在候选离散点中选 <= Rb 的最靠右者
			auto it = upper_bound(cand.begin(), cand.end(), Rb);
			if (it == cand.begin()) return false; // 无可选点
			--it;
			if (*it < Lb) return false; // 可选点都太靠左
			placed = *it;
			if (++used > 2) return false;
		}
		return true;
	};

	// 5) 二分答案
	long long lo = 0, hi = (diam + 1) / 2;
	while (lo < hi) {
		long long mid = (lo + hi) >> 1;
		if (feasible(mid)) hi = mid;
		else lo = mid + 1;
	}
	cout << lo << '\n';
	return 0;
}
```
