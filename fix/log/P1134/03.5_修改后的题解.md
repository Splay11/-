## 解题思路

一个正整数末尾 `0` 的个数等于其质因数分解中 $2$ 的个数与 $5$ 的个数的较小值。因此只需维护每个节点权值中因子 $2$ 和因子 $5$ 的指数。

每次操作把以 $t$ 为根的子树都乘上 $g$，等价于给该子树每个节点加上 $g$ 中 $2$ 和 $5$ 的指数。若每次暴力遍历子树，复杂度无法接受。

采用懒标记：操作时只在节点 $t$ 上累加指数，全部操作结束后做一次自顶向下的 DFS，把祖先的标记加到子孙上。这样每个节点得到自己被所有祖先操作影响后的指数。

然后再把每个节点初始权值中的 $2$、$5$ 指数加进去。最后做一次自底向上的 DFS，把子树内所有节点的指数求和。对每个节点 $i$，答案就是其子树中 $2$ 的指数和与 $5$ 的指数和的较小值。

树以节点 `1` 为根。

## 复杂度分析

时间复杂度 $O(n+q)$，空间复杂度 $O(n)$。

## 代码实现

### Python

```python
import sys
from collections import defaultdict

sys.setrecursionlimit(300000)

def get(x, y):
    cnt = 0
    while x % y == 0:
        cnt += 1
        x //= y
    return cnt

e = defaultdict(list)
lazy = defaultdict(list)

n = int(input())
a = [0] + list(map(int, input().split()))
for i in range(n - 1):
    u, v = map(int, input().split())
    e[u].append(v)
    e[v].append(u)
for i in range(1, n + 1):
    lazy[i].append(0)
    lazy[i].append(0)
q = int(input())
for i in range(q):
    t, g = map(int, input().split())
    x = get(g, 2)
    y = get(g, 5)
    lazy[t][0] += x
    lazy[t][1] += y

res = [[0, 0] for _ in range(n + 1)]

def dfs1(u, fa):
    for v in e[u]:
        if v == fa:
            continue
        lazy[v][0] += lazy[u][0]
        lazy[v][1] += lazy[u][1]
        dfs1(v, u)

def dfs2(u, fa):
    for v in e[u]:
        if v == fa:
            continue
        dfs2(v, u)
        lazy[u][0] += lazy[v][0]
        lazy[u][1] += lazy[v][1]
    res[u] = lazy[u][:]

dfs1(1, -1)
for i in range(1, n + 1):
    lazy[i][0] += get(a[i], 2)
    lazy[i][1] += get(a[i], 5)
dfs2(1, -1)

print(" ".join(str(min(res[i][0], res[i][1])) for i in range(1, n + 1)))
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static int[][] vals;
    static List<Integer>[] edges;
    static int[][] lazy;
    static boolean[] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        vals = new int[N + 1][2];
        for (int i = 1; i <= N; i++) {
            int num = sc.nextInt();
            vals[i][0] = getN(num, 2);
            vals[i][1] = getN(num, 5);
        }
        lazy = new int[N + 1][2];
        edges = new List[N + 1];
        vis = new boolean[N + 1];
        for (int i = 1; i <= N; i++) {
            edges[i] = new ArrayList<>();
        }
        for (int i = 0; i < N - 1; i++) {
            int from = sc.nextInt();
            int to = sc.nextInt();
            edges[from].add(to);
            edges[to].add(from);
        }
        int Q = sc.nextInt();
        for (int i = 0; i < Q; i++) {
            int node = sc.nextInt();
            int mulNum = sc.nextInt();
            lazy[node][0] += getN(mulNum, 2);
            lazy[node][1] += getN(mulNum, 5);
        }
        dfs1(1);
        for (int i = 0; i <= N; i++) vis[i] = false;
        dfs2(1);
        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= N; i++) {
            if (i > 1) sb.append(' ');
            sb.append(Math.min(vals[i][0], vals[i][1]));
        }
        System.out.println(sb.toString());
    }

    private static void dfs2(int root) {
        vis[root] = true;
        for (int nxt : edges[root]) {
            if (!vis[nxt]) {
                dfs2(nxt);
                vals[root][0] += vals[nxt][0];
                vals[root][1] += vals[nxt][1];
            }
        }
    }

    private static void dfs1(int root) {
        int i2 = lazy[root][0];
        int i5 = lazy[root][1];
        vals[root][0] += i2;
        vals[root][1] += i5;
        vis[root] = true;
        for (int nxt : edges[root]) {
            if (!vis[nxt]) {
                lazy[nxt][0] += i2;
                lazy[nxt][1] += i5;
                dfs1(nxt);
            }
        }
    }

    private static int getN(int a, int b) {
        int ans = 0;
        while (a % b == 0) {
            ans++;
            a /= b;
        }
        return ans;
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int N = 100010;
vector<int> e[N];
long long lazy2[N], lazy5[N];
long long res2[N], res5[N];
int val[N];

int get(int x, int y) {
    int cnt = 0;
    while (x % y == 0) {
        cnt++;
        x /= y;
    }
    return cnt;
}

void dfs1(int u, int fa) {
    for (int v : e[u]) {
        if (v == fa) continue;
        lazy2[v] += lazy2[u];
        lazy5[v] += lazy5[u];
        dfs1(v, u);
    }
}

void dfs2(int u, int fa) {
    for (int v : e[u]) {
        if (v == fa) continue;
        dfs2(v, u);
        lazy2[u] += lazy2[v];
        lazy5[u] += lazy5[v];
    }
    res2[u] = lazy2[u];
    res5[u] = lazy5[u];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> val[i];
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    int q;
    cin >> q;
    while (q--) {
        int t, g;
        cin >> t >> g;
        lazy2[t] += get(g, 2);
        lazy5[t] += get(g, 5);
    }
    dfs1(1, 0);
    for (int i = 1; i <= n; i++) {
        lazy2[i] += get(val[i], 2);
        lazy5[i] += get(val[i], 5);
    }
    dfs2(1, 0);
    for (int i = 1; i <= n; i++) {
        if (i > 1) cout << ' ';
        cout << min(res2[i], res5[i]);
    }
    cout << '\n';
    return 0;
}
```
