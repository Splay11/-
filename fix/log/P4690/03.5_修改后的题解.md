## 解题思路

本题需要模拟一个非负整数 $E$ 的变化过程，并在每一步输出 $E$ 的二进制表示中 $1$ 的个数（即 popcount）。由于操作次数可达 $10^6$，若直接使用大整数并每次重新计算 popcount，会超时。因此必须维护 $E$ 的二进制结构，并高效更新 $1$ 的个数。

核心思路：采用**二进制运行长度编码（RLE）**，将二进制从高位到低位按连续相同位压缩成若干“段”，每段记录该段是 $0$ 还是 $1$，以及这一段的长度。同时用一个变量 $cnt$ 维护当前二进制中 $1$ 的总个数。

对于四种指令，它们只会影响二进制末尾附近的若干段，具体模拟方法如下：

- **`'+'`（$E$ 增加 $1$）**  
  相当于在最低位加 $1$，分三种情况处理：
  - 若 $E=0$（栈为空），变为 $1$，新增一个 `[1,1]` 段，$cnt$ 加 $1$。
  - 若最低位为 $0$，则将该位改为 $1$：弹出末尾的 $0$ 段并缩短，然后追加 `[1,1]` 段，$cnt$ 加 $1$。
  - 若最低位为 $1$（可能多个连续的 $1$），会发生进位：弹出末尾整个 $1$ 段，设其长度为 $k$，$cnt$ 减少 $k$。此时下一段必为 $0$（若存在），将该 $0$ 段缩短一位，并在该位写入 $1$，随后补上 $k$ 个 $0$。若栈变空（例如 `111 + 1`），则新建 `[1,1]` 段和 `[0, k]` 段，$cnt$ 加 $1$。

- **`'-'`（$E$ 减少 $1$）**  
  相当于减 $1$，若 $E=0$ 则忽略该指令：
  - 若最低位为 $1$，直接删除这个 $1$ 位，$cnt$ 减 $1$。若结果大于 $0$，则最低位补一个 $0$。
  - 若最低位为 $0$（可能有多个连续 $0$），需要向高位借位：弹出末尾整个 $0$ 段，其长度为 $k$；紧接着弹出一个 $1$ 位（借位），$cnt$ 减 $1$。若借位后还有更高位，保留一个 $0$。最后将原来的 $k$ 个 $0$ 变成 $k$ 个 $1$，$cnt$ 增加 $k$。

- **`'*'`（$E$ 乘以 $2$）**  
  相当于在二进制末尾添加一个 $0$。直接向栈尾追加一个 $0$ 段（若 $E=0$ 则忽略），$cnt$ 不变。

- **`'/'`（$E$ 除以 $2$ 向下取整）**  
  相当于删除二进制最低位。若最低位为 $1$，$cnt$ 减 $1$；然后删除该位。若 $E=0$ 则忽略。

每次操作后，将当前 $cnt$ 存入答案数组，最后按行输出。

通过这种方式，每次操作只需修改常数个连续段，整体复杂度线性。

## 复杂度分析

设指令数量为 $n$。

- 时间复杂度：$O(n)$。每次操作只影响栈尾的常数个段，每个段至多被创建和销毁一次，总操作次数与 $n$ 同级。
- 空间复杂度：$O(n)$。最坏情况下二进制表示会有 $O(n)$ 个连续段（例如交替的 $0$ 和 $1$），需要存储它们。

以上复杂度满足 $n \le 10^6$ 的要求。

## 代码实现

### Python

```python
import sys

def add_seg(st, bit, length):
    """向栈末尾添加一段，若与栈顶相同则合并"""
    if length <= 0:
        return
    if st and st[-1][0] == bit:
        st[-1][1] += length
    else:
        st.append([bit, length])

def remove_last_bit(st):
    """删除二进制末尾一位"""
    if not st:
        return
    st[-1][1] -= 1
    if st[-1][1] == 0:
        st.pop()

def solve_ops(s):
    st = []       # 二进制连续段，每段为 [bit, len]
    cnt = 0       # 当前二进制中 1 的个数
    ans = []

    for ch in s:
        if ch == '+':
            if not st:               # E = 0
                add_seg(st, 1, 1)
                cnt += 1
            elif st[-1][0] == 0:     # 最低位是 0
                remove_last_bit(st)
                add_seg(st, 1, 1)
                cnt += 1
            else:                    # 最低位是 1，进位
                k = st[-1][1]
                st.pop()
                cnt -= k
                if not st:           # 原数全是 1
                    add_seg(st, 1, 1)
                    cnt += 1
                    add_seg(st, 0, k)
                else:
                    remove_last_bit(st)   # 前面的 0 变 1
                    add_seg(st, 1, 1)
                    cnt += 1
                    add_seg(st, 0, k)
        elif ch == '-':
            if not st:               # E = 0，忽略
                pass
            elif st[-1][0] == 1:     # 最低位是 1
                remove_last_bit(st)
                cnt -= 1
                if st:               # 若结果大于 0，补 0
                    add_seg(st, 0, 1)
            else:                    # 最低位是 0，借位
                k = st[-1][1]
                st.pop()
                remove_last_bit(st)  # 弹出一个 1
                cnt -= 1
                if st:
                    add_seg(st, 0, 1)
                add_seg(st, 1, k)
                cnt += k
        elif ch == '*':
            if st:                   # 乘以 2，末尾补 0
                add_seg(st, 0, 1)
        else:  # ch == '/'
            if st:                   # 除以 2，删除最低位
                if st[-1][0] == 1:
                    cnt -= 1
                remove_last_bit(st)

        ans.append(str(cnt))

    return ans

def main():
    input = sys.stdin.readline
    n = int(input().strip())
    s = input().strip()
    ans = solve_ops(s)
    sys.stdout.write('\n'.join(ans))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;

public class Main {

    static class Node {
        int bit; // 该段是 0 还是 1
        int len; // 该段长度

        Node(int bit, int len) {
            this.bit = bit;
            this.len = len;
        }
    }

    // 向末尾加入一段，若和栈顶相同则合并
    static void addSeg(ArrayList<Node> st, int bit, int len) {
        if (len <= 0) return;
        if (!st.isEmpty() && st.get(st.size() - 1).bit == bit) {
            st.get(st.size() - 1).len += len;
        } else {
            st.add(new Node(bit, len));
        }
    }

    // 删除末尾一位
    static void removeLastBit(ArrayList<Node> st) {
        if (st.isEmpty()) return;
        Node last = st.get(st.size() - 1);
        last.len--;
        if (last.len == 0) {
            st.remove(st.size() - 1);
        }
    }

    static StringBuilder solveOps(String s) {
        ArrayList<Node> st = new ArrayList<>(); // 二进制连续段
        int cnt = 0;                            // 当前二进制中 1 的个数
        StringBuilder ans = new StringBuilder();

        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            if (ch == '+') {
                if (st.isEmpty()) {
                    addSeg(st, 1, 1);
                    cnt++;
                } else if (st.get(st.size() - 1).bit == 0) {
                    removeLastBit(st);
                    addSeg(st, 1, 1);
                    cnt++;
                } else {
                    int k = st.get(st.size() - 1).len;
                    st.remove(st.size() - 1);
                    cnt -= k;
                    if (st.isEmpty()) {
                        addSeg(st, 1, 1);
                        cnt++;
                        addSeg(st, 0, k);
                    } else {
                        removeLastBit(st);
                        addSeg(st, 1, 1);
                        cnt++;
                        addSeg(st, 0, k);
                    }
                }
            } else if (ch == '-') {
                if (st.isEmpty()) {
                    // E = 0，忽略
                } else if (st.get(st.size() - 1).bit == 1) {
                    removeLastBit(st);
                    cnt--;
                    if (!st.isEmpty()) {
                        addSeg(st, 0, 1);
                    }
                } else {
                    int k = st.get(st.size() - 1).len;
                    st.remove(st.size() - 1);
                    removeLastBit(st);
                    cnt--;
                    if (!st.isEmpty()) {
                        addSeg(st, 0, 1);
                    }
                    addSeg(st, 1, k);
                    cnt += k;
                }
            } else if (ch == '*') {
                if (!st.isEmpty()) {
                    addSeg(st, 0, 1);
                }
            } else { // ch == '/'
                if (!st.isEmpty()) {
                    if (st.get(st.size() - 1).bit == 1) {
                        cnt--;
                    }
                    removeLastBit(st);
                }
            }
            ans.append(cnt).append('\n');
        }
        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        String s = br.readLine().trim();
        System.out.print(solveOps(s).toString());
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

struct Node {
    int bit;   // 这一段是 0 还是 1
    int len;   // 这一段长度
};

// 向末尾加入一段，若和栈顶相同则合并
void addSeg(vector<Node>& st, int bit, int len) {
    if (len <= 0) return;
    if (!st.empty() && st.back().bit == bit) {
        st.back().len += len;
    } else {
        st.push_back({bit, len});
    }
}

// 删除末尾一位
void removeLastBit(vector<Node>& st) {
    if (st.empty()) return;
    st.back().len--;
    if (st.back().len == 0) st.pop_back();
}

// 处理操作串
vector<int> solveOps(const string& s) {
    vector<Node> st;   // 二进制连续段
    int cnt = 0;       // 当前二进制中 1 的个数
    vector<int> ans;
    ans.reserve(s.size());

    for (char ch : s) {
        if (ch == '+') {
            if (st.empty()) {
                addSeg(st, 1, 1);
                cnt++;
            } else if (st.back().bit == 0) {
                removeLastBit(st);
                addSeg(st, 1, 1);
                cnt++;
            } else {
                int k = st.back().len;
                st.pop_back();
                cnt -= k;
                if (st.empty()) {
                    addSeg(st, 1, 1);
                    cnt++;
                    addSeg(st, 0, k);
                } else {
                    removeLastBit(st);
                    addSeg(st, 1, 1);
                    cnt++;
                    addSeg(st, 0, k);
                }
            }
        } else if (ch == '-') {
            if (st.empty()) {
                // E = 0，忽略
            } else if (st.back().bit == 1) {
                removeLastBit(st);
                cnt--;
                if (!st.empty()) {
                    addSeg(st, 0, 1);
                }
            } else {
                int k = st.back().len;
                st.pop_back();
                removeLastBit(st);
                cnt--;
                if (!st.empty()) {
                    addSeg(st, 0, 1);
                }
                addSeg(st, 1, k);
                cnt += k;
            }
        } else if (ch == '*') {
            if (!st.empty()) {
                addSeg(st, 0, 1);
            }
        } else { // ch == '/'
            if (!st.empty()) {
                if (st.back().bit == 1) cnt--;
                removeLastBit(st);
            }
        }
        ans.push_back(cnt);
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    string s;
    cin >> n >> s;

    vector<int> ans = solveOps(s);
    for (int x : ans) {
        cout << x << '\n';
    }
    return 0;
}
```
