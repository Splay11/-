## 解题思路

本题要求将 $n$ 个二维采样点通过自底向上的层次聚类合并成恰好 $C$ 个簇，并在合并过程中按照距离和平局规则选择合并目标，最后按簇中心的坐标分配连续编号。

### 1. 距离定义与预处理
定义两个点 $p_i, p_j$ 的欧氏距离为  
$$d(p_i,p_j)=\sqrt{(x_i-x_j)^2+(y_i-y_j)^2}$$  
由于在比较距离时平方运算不影响大小关系，可以预先计算所有点对之间的**平方距离** $d^2_{ij} = (x_i-x_j)^2+(y_i-y_j)^2$，避免反复开方。  
将所有平方距离存入矩阵 `dist2`，其中 `dist2[i][j]` 表示点 $i$ 与点 $j$ 的平方距离。

### 2. 初始簇与合并终止条件
初始时每个点自成一个簇，共 $n$ 个簇。每次合并两个簇，当簇总数等于 $C$ 时停止。

### 3. 簇间距离与合并目标选择
对于任意两个簇 $A,B$，定义其距离为  
$$D(A,B)=\min_{p_i\in A,\,p_j\in B} d(p_i,p_j)$$  
该距离等价于两个簇中最近点对的欧氏距离。实际计算时，直接使用预存的平方距离：
$$D^2(A,B)=\min_{p_i\in A,\,p_j\in B} d^2_{ij}$$
每一轮合并执行以下步骤：
- 遍历所有簇对 $(A,B)$，计算 $D^2(A,B)$；
- 找到平方距离最小的簇对；若有多个簇对的距离相等（考虑浮点误差，使用 $10^{-9}$ 阈值判断相等），则按照题目规则进一步选择：
  - 记簇 $A$ 和 $B$ 中包含的点的最小编号分别为 $a$ 和 $b$，且令 $a<b$，构造有序对 $(a,b)$；
  - 选择有序对**字典序最小**的那对簇作为合并目标。

### 4. 合并操作
将选出的两个簇合并为一个新簇，其成员为原来两个簇中所有点的并集，同时保持内部点编号的升序排列，便于后续提取最小编号。  
重复该过程直至剩余簇数量等于 $C$。

### 5. 最终编号分配
合并结束后得到 $C$ 个簇。为每个簇计算中心坐标（算术平均）：
$$(C_x,C_y)=\left(\frac{1}{|A|}\sum_{p\in A}x_p,\;\frac{1}{|A|}\sum_{p\in A}y_p\right)$$
按以下优先级对簇进行升序排序：
1. 中心 $x$ 坐标：若两中心 $x$ 坐标之差的绝对值 $\ge 10^{-12}$，按 $x$ 坐标升序；
2. 中心 $y$ 坐标：若 $x$ 坐标无法区分，且 $y$ 坐标之差的绝对值 $\ge 10^{-12}$，按 $y$ 坐标升序；
3. 簇内最小编号：若 $x,y$ 皆无法区分，按簇内所含点的最小编号升序。

排序后，依序将编号 $0,1,\dots,C-1$ 分配给各个簇。最后输出每个初始点所属簇的编号。

## 复杂度分析

- **时间复杂度**：共需要合并 $n-C$ 次，每次枚举所有簇对并计算簇间最小距离，簇大小总和为 $n$。在最坏情况下单次合并的复杂度为 $O(n^2)$，因此总时间复杂度为 $O(n^3)$，当 $n\le 100$ 时可以在时限内完成。
- **空间复杂度**：存储点坐标与平方距离矩阵需要 $O(n^2)$ 额外空间，其余辅助数组均为 $O(n)$。总空间复杂度 $O(n^2)$。

## 代码实现

### Python

```python
import sys
import json
from functools import cmp_to_key

def solve():
    data = json.loads(sys.stdin.readline())
    C = data['C']
    points = data['points']
    n = len(points)

    xs = [v[0] for v in points]
    ys = [v[1] for v in points]

    # 预计算平方距离
    dist2 = [[0.0] * n for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            d2 = dx * dx + dy * dy
            dist2[i][j] = dist2[j][i] = d2

    # 初始簇：每个点自成一簇
    clusters = [[i] for i in range(n)]

    EPS = 1e-9               # 用于判断平方距离相等
    while len(clusters) > C:
        m = len(clusters)
        best_d2 = float('inf')
        best_i = best_j = -1
        best_key = (0, 0)

        for i in range(m):
            for j in range(i + 1, m):
                A = clusters[i]
                B = clusters[j]
                # 簇间最小平方距离
                min_d2 = float('inf')
                for a in A:
                    for b in B:
                        d2 = dist2[a][b]
                        if d2 < min_d2:
                            min_d2 = d2

                min_a, min_b = A[0], B[0]   # 簇内已有序，首元素即最小编号
                if min_a > min_b:
                    min_a, min_b = min_b, min_a
                key = (min_a, min_b)

                if min_d2 < best_d2 - EPS:
                    best_d2 = min_d2
                    best_i, best_j = i, j
                    best_key = key
                elif abs(min_d2 - best_d2) <= EPS and key < best_key:
                    best_i, best_j = i, j
                    best_key = key

        # 合并两个簇，保持点编号有序
        merged = sorted(clusters[best_i] + clusters[best_j])
        if best_i < best_j:
            clusters.pop(best_j)
            clusters.pop(best_i)
        else:
            clusters.pop(best_i)
            clusters.pop(best_j)
        clusters.append(merged)

    # 计算中心及排序
    infos = []
    for cl in clusters:
        sx = sum(xs[i] for i in cl) / len(cl)
        sy = sum(ys[i] for i in cl) / len(cl)
        infos.append((sx, sy, cl[0], cl))   # (中心x, 中心y, 最小编号, 点列表)

    SORT_EPS = 1e-12
    def cmp(a, b):
        if abs(a[0] - b[0]) >= SORT_EPS:
            return -1 if a[0] < b[0] else 1
        if abs(a[1] - b[1]) >= SORT_EPS:
            return -1 if a[1] < b[1] else 1
        return -1 if a[2] < b[2] else 1

    infos.sort(key=cmp_to_key(cmp))

    # 分配最终编号
    ans = [0] * n
    for label, info in enumerate(infos):
        for idx in info[3]:
            ans[idx] = label

    print(json.dumps(ans))


if __name__ == '__main__':
    solve()
```

### Java

```java
import java.util.*;
import java.util.regex.*;

public class Main {
    static final double EPS = 1e-9;
    static final double SORT_EPS = 1e-12;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String line = sc.nextLine().trim();
        sc.close();

        // 提取所有数字（含整数和浮点数）
        Pattern pattern = Pattern.compile("-?\\d+(\\.\\d+)?");
        Matcher matcher = pattern.matcher(line);
        List<String> numbers = new ArrayList<>();
        while (matcher.find()) {
            numbers.add(matcher.group());
        }

        int K = Integer.parseInt(numbers.get(0));
        int n = (numbers.size() - 1) / 2;
        double[] xs = new double[n];
        double[] ys = new double[n];
        for (int i = 0; i < n; i++) {
            xs[i] = Double.parseDouble(numbers.get(1 + 2 * i));
            ys[i] = Double.parseDouble(numbers.get(1 + 2 * i + 1));
        }

        // 预计算平方距离
        double[][] dist2 = new double[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                double dx = xs[i] - xs[j];
                double dy = ys[i] - ys[j];
                dist2[i][j] = dist2[j][i] = dx * dx + dy * dy;
            }
        }

        // 初始簇
        List<List<Integer>> clusters = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            List<Integer> cl = new ArrayList<>();
            cl.add(i);
            clusters.add(cl);
        }

        while (clusters.size() > K) {
            int m = clusters.size();
            double bestD2 = Double.POSITIVE_INFINITY;
            int bestI = -1, bestJ = -1;
            int bestSmall = -1, bestLarge = -1;

            for (int i = 0; i < m; i++) {
                for (int j = i + 1; j < m; j++) {
                    List<Integer> A = clusters.get(i);
                    List<Integer> B = clusters.get(j);
                    double minD2 = Double.POSITIVE_INFINITY;
                    for (int a : A)
                        for (int b : B)
                            if (dist2[a][b] < minD2)
                                minD2 = dist2[a][b];

                    int minA = A.get(0);
                    int minB = B.get(0);
                    if (minA > minB) {
                        int t = minA; minA = minB; minB = t;
                    }

                    boolean better = false;
                    if (minD2 < bestD2 - EPS) {
                        better = true;
                    } else if (Math.abs(minD2 - bestD2) <= EPS) {
                        if (minA < bestSmall || (minA == bestSmall && minB < bestLarge)) {
                            better = true;
                        }
                    }

                    if (better) {
                        bestD2 = minD2;
                        bestI = i;
                        bestJ = j;
                        bestSmall = minA;
                        bestLarge = minB;
                    }
                }
            }

            // 合并
            List<Integer> merged = new ArrayList<>();
            merged.addAll(clusters.get(bestI));
            merged.addAll(clusters.get(bestJ));
            Collections.sort(merged);
            if (bestI < bestJ) {
                clusters.remove(bestJ);
                clusters.remove(bestI);
            } else {
                clusters.remove(bestI);
                clusters.remove(bestJ);
            }
            clusters.add(merged);
        }

        // 计算中心并排序
        int C = clusters.size();
        double[] cx = new double[C], cy = new double[C];
        int[] minId = new int[C];
        for (int i = 0; i < C; i++) {
            double sx = 0, sy = 0;
            for (int idx : clusters.get(i)) {
                sx += xs[idx];
                sy += ys[idx];
            }
            cx[i] = sx / clusters.get(i).size();
            cy[i] = sy / clusters.get(i).size();
            minId[i] = clusters.get(i).get(0);
        }

        Integer[] order = new Integer[C];
        for (int i = 0; i < C; i++) order[i] = i;
        Arrays.sort(order, (a, b) -> {
            if (Math.abs(cx[a] - cx[b]) >= SORT_EPS)
                return Double.compare(cx[a], cx[b]);
            if (Math.abs(cy[a] - cy[b]) >= SORT_EPS)
                return Double.compare(cy[a], cy[b]);
            return Integer.compare(minId[a], minId[b]);
        });

        // 分配最终编号
        int[] ans = new int[n];
        for (int label = 0; label < C; label++) {
            int orig = order[label];
            for (int idx : clusters.get(orig))
                ans[idx] = label;
        }

        // 输出 JSON 数组
        StringBuilder sb = new StringBuilder();
        sb.append("[");
        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(", ");
            sb.append(ans[i]);
        }
        sb.append("]");
        System.out.println(sb.toString());
    }
}
```

### C++

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <cmath>
#include <algorithm>
#include <limits>
#include <regex>

using namespace std;

const double EPS = 1e-9;
const double SORT_EPS = 1e-12;

int main() {
    string line;
    getline(cin, line);

    // 提取所有数字
    regex num_re("-?\\d+(\\.\\d+)?");
    sregex_iterator iter(line.begin(), line.end(), num_re), end;
    vector<string> numbers;
    while (iter != end) {
        numbers.push_back(iter->str());
        ++iter;
    }

    int K = stoi(numbers[0]);
    int n = ((int)numbers.size() - 1) / 2;
    vector<double> xs(n), ys(n);
    for (int i = 0; i < n; i++) {
        xs[i] = stod(numbers[1 + 2 * i]);
        ys[i] = stod(numbers[1 + 2 * i + 1]);
    }

    // 平方距离矩阵
    vector<vector<double>> dist2(n, vector<double>(n, 0.0));
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            double dx = xs[i] - xs[j];
            double dy = ys[i] - ys[j];
            dist2[i][j] = dist2[j][i] = dx * dx + dy * dy;
        }
    }

    // 初始簇
    vector<vector<int>> clusters(n);
    for (int i = 0; i < n; i++) clusters[i].push_back(i);

    while ((int)clusters.size() > K) {
        int m = clusters.size();
        double bestD2 = numeric_limits<double>::infinity();
        int bestI = -1, bestJ = -1;
        int bestSmall = -1, bestLarge = -1;

        for (int i = 0; i < m; i++) {
            for (int j = i + 1; j < m; j++) {
                const auto& A = clusters[i];
                const auto& B = clusters[j];
                double minD2 = numeric_limits<double>::infinity();
                for (int a : A)
                    for (int b : B)
                        if (dist2[a][b] < minD2)
                            minD2 = dist2[a][b];

                int minA = A[0], minB = B[0];
                if (minA > minB) swap(minA, minB);

                bool better = false;
                if (minD2 < bestD2 - EPS) {
                    better = true;
                } else if (fabs(minD2 - bestD2) <= EPS) {
                    if (minA < bestSmall || (minA == bestSmall && minB < bestLarge))
                        better = true;
                }

                if (better) {
                    bestD2 = minD2;
                    bestI = i; bestJ = j;
                    bestSmall = minA; bestLarge = minB;
                }
            }
        }

        // 合并
        vector<int> merged = clusters[bestI];
        merged.insert(merged.end(), clusters[bestJ].begin(), clusters[bestJ].end());
        sort(merged.begin(), merged.end());
        if (bestI < bestJ) {
            clusters.erase(clusters.begin() + bestJ);
            clusters.erase(clusters.begin() + bestI);
        } else {
            clusters.erase(clusters.begin() + bestI);
            clusters.erase(clusters.begin() + bestJ);
        }
        clusters.push_back(move(merged));
    }

    // 计算中心并排序
    int C = clusters.size();
    vector<double> cx(C), cy(C);
    vector<int> minId(C);
    for (int i = 0; i < C; i++) {
        double sx = 0, sy = 0;
        for (int idx : clusters[i]) {
            sx += xs[idx];
            sy += ys[idx];
        }
        cx[i] = sx / clusters[i].size();
        cy[i] = sy / clusters[i].size();
        minId[i] = clusters[i][0];
    }

    vector<int> order(C);
    for (int i = 0; i < C; i++) order[i] = i;
    sort(order.begin(), order.end(), [&](int a, int b) {
        if (fabs(cx[a] - cx[b]) >= SORT_EPS)
            return cx[a] < cx[b];
        if (fabs(cy[a] - cy[b]) >= SORT_EPS)
            return cy[a] < cy[b];
        return minId[a] < minId[b];
    });

    // 分配编号
    vector<int> ans(n);
    for (int label = 0; label < C; label++) {
        int orig = order[label];
        for (int idx : clusters[orig])
            ans[idx] = label;
    }

    // 输出 JSON 数组
    cout << "[";
    for (int i = 0; i < n; i++) {
        if (i > 0) cout << ", ";
        cout << ans[i];
    }
    cout << "]" << endl;

    return 0;
}
```
