## 解题思路

我们需要找出最小的非负整数秒 $t$，使得 $t$ 不被任何一个预约区间 $[l, r]$ 覆盖。

由于 $n$ 最大可达 $2 \times 10^5$，直接检查每一个整数秒是不可行的。我们可以采用排序加“合并”覆盖区间的方法：

1. 将所有预约区间按左端点 $l$ 从小到大排序。
2. 维护一个变量 `ans`，表示当前最早可能空闲的时刻，初始值为 $0$。
3. 依次遍历排序后的每个区间 $[l, r]$：
   - 若 $l > \text{ans}$：说明区间从 $\text{ans}$ 之后才开始，$\text{ans}$ 没有被当前及之前的任何预约覆盖。此时 $\text{ans}$ 就是答案，可以直接结束遍历。
   - 若 $l \le \text{ans} \le r$：说明 $\text{ans}$ 被当前区间覆盖，那么最早空闲时刻至少需要向后推迟到 $r + 1$，更新 $\text{ans} = r + 1$。
   - 若 $r < \text{ans}$：当前区间完全位于 $\text{ans}$ 之前，不会影响结果，直接跳过。
4. 遍历完所有区间后，$\text{ans}$ 即为第一个未被覆盖的整数秒。

该算法的本质是：扫描排序后的区间，动态维护从 $0$ 开始已经被连续覆盖的区间的右边界，从而得到第一个空闲秒。

## 复杂度分析

- 时间复杂度：排序需要 $O(n \log n)$，遍历所有区间需要 $O(n)$，因此总时间复杂度为 $O(n \log n)$。可以满足 $n \le 2 \times 10^5$ 的要求。
- 空间复杂度：存储所有区间需要 $O(n)$ 的额外空间，因此空间复杂度为 $O(n)$。

## 代码实现

### Python

```python
import sys

def find_first_free(appointments):
    # 按区间左端点从小到大排序
    appointments.sort()
    
    # ans 表示当前最早可能空闲的时刻
    ans = 0
    
    for l, r in appointments:
        # 如果当前区间的左端点大于 ans，说明 ans 未被占用，即为答案
        if l > ans:
            break
        # 如果 ans 落在当前区间内，则将 ans 推进到区间右端点之后
        if r >= ans:
            ans = r + 1
            
    return ans

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    
    appointments = []
    idx = 1
    for _ in range(n):
        l = int(data[idx])
        r = int(data[idx + 1])
        idx += 2
        appointments.append((l, r))
        
    print(find_first_free(appointments))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static long findFirstFree(long[][] appointments) {
        // 按区间左端点从小到大排序
        Arrays.sort(appointments, (a, b) -> {
            if (a[0] == b[0]) {
                return Long.compare(a[1], b[1]);
            }
            return Long.compare(a[0], b[0]);
        });

        long ans = 0;  // 当前最早可能空闲的时刻

        for (long[] interval : appointments) {
            long l = interval[0];
            long r = interval[1];

            // 如果当前区间的左端点大于 ans，说明 ans 未被占用，即为答案
            if (l > ans) {
                break;
            }
            // 如果 ans 落在当前区间内，则将 ans 推进到区间右端点之后
            if (r >= ans) {
                ans = r + 1;
            }
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        long[][] appointments = new long[n][2];

        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            appointments[i][0] = Long.parseLong(st.nextToken());
            appointments[i][1] = Long.parseLong(st.nextToken());
        }

        System.out.println(findFirstFree(appointments));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

long long findFirstFree(vector<pair<long long, long long>>& appointments) {
    // 按区间左端点从小到大排序
    sort(appointments.begin(), appointments.end());

    long long ans = 0;  // 当前最早可能空闲的时刻

    for (auto& interval : appointments) {
        long long l = interval.first;
        long long r = interval.second;

        // 如果当前区间的左端点大于 ans，说明 ans 未被占用，即为答案
        if (l > ans) {
            break;
        }
        // 如果 ans 落在当前区间内，则将 ans 推进到区间右端点之后
        if (r >= ans) {
            ans = r + 1;
        }
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<pair<long long, long long>> appointments(n);
    for (int i = 0; i < n; i++) {
        cin >> appointments[i].first >> appointments[i].second;
    }

    cout << findFirstFree(appointments) << '\n';

    return 0;
}
```
