## 解题思路

探险家可以执行 **清除** 和 **收集** 操作各至多一次，且清除操作如果在收集之后执行则没有意义，因此最优策略一定是**先清除一部分宝箱，再进行一次收集**（也可能不执行任何操作，或只执行其中一个操作）。

我们把宝箱按收集顺序 $p$ 重新排列，得到序列  
$$s_i = a_{p_i}\quad (1 \le i \le n)$$  
那么“选择一个 $t$ ，按 $p_1,\dots,p_t$ 收集”就等价于取序列 $s$ 的一个前缀和 $\sum_{j=1}^t s_j$ 。若从未进行清除，则收益就是 $\max_{1\le t\le n} \sum_{j=1}^t s_j$（以及 $0$）。

现在考虑先进行清除。清除操作按清除顺序 $q$ 进行，依次将 $q_1,q_2,\dots,q_u$ 这些宝箱的当前价值变为 $0$ 。在序列 $s$ 中，宝箱 $x$ 出现的位置是它在 $p$ 中的下标，记为 $\mathrm{pos}[x]$（即 $p_{\mathrm{pos}[x]}=x$）。将宝箱 $x$ 的价值清零，相当于把 $s_{\mathrm{pos}[x]}$ 变为 $0$ ，这会使得所有包含该位置的前缀和都减少 $a_x$ 。具体来说，前缀和数组  
$$\mathrm{pre}_i = \sum_{j=1}^i s_j$$  
在将 $s_{\mathrm{pos}[x]}$ 清零后，区间 $[\mathrm{pos}[x],\,n]$ 内的所有 $\mathrm{pre}$ 值都会减少 $a_x$。

因此，我们可以用线段树维护前缀和数组 $\mathrm{pre}$ ，支持区间加法（相当于减少宝箱价值）和查询全局最大值。算法步骤如下：

1. 构造数组 $s$ 和 $\mathrm{pos}$：对于 $i=1\sim n$，令 $s_i = a_{p_i}$，并记录 $\mathrm{pos}[p_i]=i$。
2. 计算初始前缀和 $\mathrm{pre}_i = \sum_{j=1}^i s_j$，并用线段树维护，初始答案为  
   $$\mathrm{ans} = \max\bigl(0,\ \max_{1\le i\le n} \mathrm{pre}_i\bigr)$$
3. 按 $u=1,2,\dots,n$ 的顺序枚举清除操作：设当前清除的宝箱为 $x = q_u$，它在 $s$ 中的位置为 $p' = \mathrm{pos}[x]$，价值为 $v = a_x$。在线段树上对区间 $[p', n]$ 执行 **加 $-v$** 操作，表示该宝箱价值变为 $0$。然后查询线段树的全局最大值，更新 $\mathrm{ans}$。
4. 最终的 $\mathrm{ans}$ 即为所求的最大收益。

## 复杂度分析

- 时间复杂度：每组数据建线段树 $O(n)$，依次处理 $n$ 次清除操作，每次区间加和全局查询均为 $O(\log n)$，总时间复杂度 $O(n \log n)$。所有测试数据的 $n$ 之和不超过 $2\times 10^5$，因此整体运行时间充裕。
- 空间复杂度：需要存储 $s$、$\mathrm{pos}$、$\mathrm{pre}$ 以及线段树节点信息，均为 $O(n)$。

### Python

```python
import sys


class SegmentTree:
    def __init__(self, arr):
        self.n = len(arr) - 1
        self.mx = [0] * (4 * self.n)
        self.lazy = [0] * (4 * self.n)
        self._build(1, 1, self.n, arr)

    def _build(self, node, l, r, arr):
        if l == r:
            self.mx[node] = arr[l]
            return
        mid = (l + r) // 2
        self._build(node * 2, l, mid, arr)
        self._build(node * 2 + 1, mid + 1, r, arr)
        self.mx[node] = max(self.mx[node * 2], self.mx[node * 2 + 1])

    def _push_down(self, node):
        if self.lazy[node]:
            tag = self.lazy[node]
            self.mx[node * 2] += tag
            self.lazy[node * 2] += tag
            self.mx[node * 2 + 1] += tag
            self.lazy[node * 2 + 1] += tag
            self.lazy[node] = 0

    def range_add(self, node, l, r, ql, qr, val):
        if ql <= l and r <= qr:
            self.mx[node] += val
            self.lazy[node] += val
            return
        self._push_down(node)
        mid = (l + r) // 2
        if ql <= mid:
            self.range_add(node * 2, l, mid, ql, qr, val)
        if qr > mid:
            self.range_add(node * 2 + 1, mid + 1, r, ql, qr, val)
        self.mx[node] = max(self.mx[node * 2], self.mx[node * 2 + 1])

    def query_max(self):
        return self.mx[1]


def solve_case(n, a, p, q):
    # 收集序列 s_i = a_{p_i}
    s = [0] * (n + 1)
    pos = [0] * (n + 1)          # pos[x] = i 表示宝箱x在s中的下标
    for i in range(1, n + 1):
        s[i] = a[p[i]]
        pos[p[i]] = i

    # 初始前缀和
    pre = [0] * (n + 1)
    for i in range(1, n + 1):
        pre[i] = pre[i - 1] + s[i]

    seg = SegmentTree(pre)
    ans = max(0, seg.query_max())   # 不执行清除或只执行收集

    # 枚举清除操作的步数
    for i in range(1, n + 1):
        x = q[i]                    # 当前清除的宝箱编号
        idx = pos[x]                # 该宝箱在s中的下标
        seg.range_add(1, 1, n, idx, n, -a[x])
        ans = max(ans, seg.query_max())

    return ans


def main():
    input = sys.stdin.readline
    T = int(input().strip())
    out_lines = []
    for _ in range(T):
        n = int(input().strip())
        a = [0] + list(map(int, input().split()))
        p = [0] + list(map(int, input().split()))
        q = [0] + list(map(int, input().split()))
        out_lines.append(str(solve_case(n, a, p, q)))
    print("\n".join(out_lines))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static class SegTree {
        long[] mx, lazy;
        int n;

        SegTree(long[] arr, int n) {
            this.n = n;
            mx = new long[4 * n];
            lazy = new long[4 * n];
            build(1, 1, n, arr);
        }

        void build(int node, int l, int r, long[] arr) {
            if (l == r) {
                mx[node] = arr[l];
                return;
            }
            int mid = (l + r) >> 1;
            build(node << 1, l, mid, arr);
            build(node << 1 | 1, mid + 1, r, arr);
            mx[node] = Math.max(mx[node << 1], mx[node << 1 | 1]);
        }

        void push(int node) {
            if (lazy[node] != 0) {
                long tag = lazy[node];
                mx[node << 1] += tag;
                mx[node << 1 | 1] += tag;
                lazy[node << 1] += tag;
                lazy[node << 1 | 1] += tag;
                lazy[node] = 0;
            }
        }

        void rangeAdd(int node, int l, int r, int ql, int qr, long val) {
            if (ql <= l && r <= qr) {
                mx[node] += val;
                lazy[node] += val;
                return;
            }
            push(node);
            int mid = (l + r) >> 1;
            if (ql <= mid) rangeAdd(node << 1, l, mid, ql, qr, val);
            if (qr > mid) rangeAdd(node << 1 | 1, mid + 1, r, ql, qr, val);
            mx[node] = Math.max(mx[node << 1], mx[node << 1 | 1]);
        }

        long queryMax() {
            return mx[1];
        }
    }

    static long solveCase(int n, long[] a, int[] p, int[] q) {
        long[] s = new long[n + 1];
        int[] pos = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            s[i] = a[p[i]];
            pos[p[i]] = i;
        }

        long[] pre = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            pre[i] = pre[i - 1] + s[i];
        }

        SegTree seg = new SegTree(pre, n);
        long ans = Math.max(0L, seg.queryMax());

        for (int i = 1; i <= n; i++) {
            int x = q[i];
            int idx = pos[x];
            seg.rangeAdd(1, 1, n, idx, n, -a[x]);
            ans = Math.max(ans, seg.queryMax());
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        StringBuilder sb = new StringBuilder();
        while (T-- > 0) {
            int n = Integer.parseInt(br.readLine());
            long[] a = new long[n + 1];
            int[] p = new int[n + 1];
            int[] q = new int[n + 1];

            String[] parts = br.readLine().split(" ");
            for (int i = 1; i <= n; i++) a[i] = Long.parseLong(parts[i - 1]);
            parts = br.readLine().split(" ");
            for (int i = 1; i <= n; i++) p[i] = Integer.parseInt(parts[i - 1]);
            parts = br.readLine().split(" ");
            for (int i = 1; i <= n; i++) q[i] = Integer.parseInt(parts[i - 1]);

            sb.append(solveCase(n, a, p, q)).append('\n');
        }
        System.out.print(sb);
    }
}
```
