## 解题思路

本题要求实现一个基于用户的协同筛选习题推荐系统，具体步骤如下：

1. **计算学生平均评分**  
   对每位学生 $u$，统计其已作答习题（评分 $>0$）的平均分 $\bar{r}_u$。若学生未作答任何习题，令 $\bar{r}_u = 0$。

2. **计算学生相似度**  
   对于目标学生 $U$ 和另一名学生 $v$，找出两人共同作答的习题集合 $I_{Uv}$（即评分均 $>0$ 的习题）。  
   使用皮尔逊相关系数计算相似度：
   $$\text{sim}(U, v) = \frac{\sum_{i \in I_{Uv}} (r_{U,i} - \bar{r}_U)(r_{v,i} - \bar{r}_v)}{\sqrt{\sum_{i \in I_{Uv}} (r_{U,i} - \bar{r}_U)^2} \sqrt{\sum_{i \in I_{Uv}} (r_{v,i} - \bar{r}_v)^2}}$$
   若 $I_{Uv}$ 为空或分母为 $0$，令 $\text{sim}(U, v) = 0$。

3. **选取最相似的学生**  
   将其他所有学生按与 $U$ 的相似度从高到低排序，选取前 $K = 3$ 名学生作为邻居集合 $K_U$。

4. **预测未作答习题的评分**  
   对于目标学生 $U$ 未作答的每一道习题 $j$，利用邻居集合 $K_U$ 进行预测：
   $$\hat{r}_{U,j} = \bar{r}_U + \frac{\sum_{v \in K_U} \text{sim}(U, v) \cdot (r_{v,j} - \bar{r}_v)}{\sum_{v \in K_U} |\text{sim}(U, v)|}$$
   若分母为 $0$（即邻居均未作答习题 $j$），则令 $\hat{r}_{U,j} = \bar{r}_U$。

5. **生成推荐列表**  
   将所有未作答习题按预测评分从高到低排序，若预测评分相同则按习题编号从小到大排序，输出前 $T$ 道习题的编号。

## 复杂度分析

- **时间复杂度**：计算所有学生对相似度需要 $O(N^2 M)$，预测评分需遍历目标学生的 $M$ 道习题及 $K=3$ 个邻居，为 $O(MK)=O(M)$，排序复杂度 $O(M \log M)$。整体时间复杂度为 $O(N^2 M)$，当 $N,M \le 500$ 时可以高效运行。
- **空间复杂度**：存储评分矩阵需要 $O(NM)$，存储学生平均分和相似度等辅助数组需要 $O(N)$，预测评分数组需要 $O(M)$，总空间复杂度 $O(NM)$。

## 代码实现

### Python

```python
def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    M = int(next(it))
    ratings = []
    for _ in range(N):
        row = [int(next(it)) for _ in range(M)]
        ratings.append(row)
    U = int(next(it))
    T = int(next(it))
    
    K = 3
    
    # 计算每位学生的平均评分
    user_avg = [0.0] * N
    for u in range(N):
        cnt = 0
        total = 0
        for j in range(M):
            if ratings[u][j] > 0:
                cnt += 1
                total += ratings[u][j]
        if cnt > 0:
            user_avg[u] = total / cnt
    
    # 计算目标学生U与其他学生的相似度
    sims = [0.0] * N
    for v in range(N):
        if v == U:
            continue
        # 寻找共同作答的习题
        num = den1 = den2 = 0.0
        for j in range(M):
            if ratings[U][j] > 0 and ratings[v][j] > 0:
                r1 = ratings[U][j] - user_avg[U]
                r2 = ratings[v][j] - user_avg[v]
                num += r1 * r2
                den1 += r1 * r1
                den2 += r2 * r2
        den = (den1 ** 0.5) * (den2 ** 0.5)
        if den1 == 0 or den2 == 0 or den == 0:
            sims[v] = 0.0
        else:
            sims[v] = num / den
    
    # 选取最相似的前K个学生（排除自己）
    indices = list(range(N))
    indices.remove(U)
    # 按相似度降序排序
    top_k = sorted(indices, key=lambda x: -sims[x])[:K]
    
    # 预测未作答习题的评分
    preds = []
    for j in range(M):
        if ratings[U][j] > 0:
            continue  # 跳过已作答的习题
        num = 0.0
        den = 0.0
        for v in top_k:
            if sims[v] == 0:
                continue
            if ratings[v][j] > 0:
                num += sims[v] * (ratings[v][j] - user_avg[v])
                den += abs(sims[v])
        if den == 0:
            pred = user_avg[U]
        else:
            pred = user_avg[U] + num / den
        preds.append((pred, j))
    
    # 排序：预测评分降序，相同则习题编号升序
    preds.sort(key=lambda x: (-x[0], x[1]))
    
    # 取前T个
    result = [str(preds[i][1]) for i in range(T)]
    print(' '.join(result))

if __name__ == '__main__':
    solve()
```

### Java

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[][] ratings = new int[N][M];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < M; j++) {
                ratings[i][j] = sc.nextInt();
            }
        }
        int U = sc.nextInt();
        int T = sc.nextInt();
        int K = 3;

        // 计算每位学生的平均评分
        double[] userAvg = new double[N];
        for (int u = 0; u < N; u++) {
            int cnt = 0;
            double sum = 0.0;
            for (int j = 0; j < M; j++) {
                if (ratings[u][j] > 0) {
                    cnt++;
                    sum += ratings[u][j];
                }
            }
            userAvg[u] = (cnt > 0) ? sum / cnt : 0.0;
        }

        // 计算目标学生U与其他学生的相似度
        double[] sims = new double[N];
        for (int v = 0; v < N; v++) {
            if (v == U) continue;
            double num = 0.0, den1 = 0.0, den2 = 0.0;
            for (int j = 0; j < M; j++) {
                if (ratings[U][j] > 0 && ratings[v][j] > 0) {
                    double r1 = ratings[U][j] - userAvg[U];
                    double r2 = ratings[v][j] - userAvg[v];
                    num += r1 * r2;
                    den1 += r1 * r1;
                    den2 += r2 * r2;
                }
            }
            if (den1 == 0 || den2 == 0) {
                sims[v] = 0.0;
            } else {
                sims[v] = num / (Math.sqrt(den1) * Math.sqrt(den2));
            }
        }

        // 选取最相似的前K个学生（排除U自身）
        List<Integer> indices = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            if (i != U) indices.add(i);
        }
        indices.sort((a, b) -> Double.compare(sims[b], sims[a]));
        List<Integer> topK = new ArrayList<>();
        for (int i = 0; i < K && i < indices.size(); i++) {
            topK.add(indices.get(i));
        }

        // 预测未作答习题的评分
        List<int[]> predList = new ArrayList<>(); // [pred, index]
        for (int j = 0; j < M; j++) {
            if (ratings[U][j] > 0) continue;
            double num = 0.0, den = 0.0;
            for (int v : topK) {
                if (sims[v] == 0) continue;
                if (ratings[v][j] > 0) {
                    num += sims[v] * (ratings[v][j] - userAvg[v]);
                    den += Math.abs(sims[v]);
                }
            }
            double pred;
            if (den == 0.0) {
                pred = userAvg[U];
            } else {
                pred = userAvg[U] + num / den;
            }
            predList.add(new int[]{(int)Math.round(pred * 1000), j}); // 使用扩展系数便于排序
        }

        // 排序：预测评分降序，相同则按习题编号升序
        predList.sort((a, b) -> {
            // a[0]/1000 相当于原始预测值，但直接用整数比较可能导致精度问题，这里用Double比较
            double p1 = Double.parseDouble(String.format("%.4f", (double)a[0] / 1000)); // 简单转换
            double p2 = Double.parseDouble(String.format("%.4f", (double)b[0] / 1000));
            if (p1 != p2) return Double.compare(p2, p1);
            return Integer.compare(a[1], b[1]);
        });

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < T; i++) {
            sb.append(predList.get(i)[1]);
            if (i < T - 1) sb.append(" ");
        }
        System.out.println(sb.toString());
    }
}
```

```java
import java.util.*;

public class Main {

        for (int u = 0; u < N; u++) {

            for (int j = 0; j < M; j++) {
                if (ratings[u][j] > 0) { cnt++; sum += ratings[u][j]; }
            }
            avg[u] = cnt > 0 ? sum / cnt : 0.0;

        double[] sim = new double[N];
        for (int v = 0; v < N; v++) {
            if (v == U) continue;
            double num = 0, den1 = 0, den2 = 0;
            for (int j = 0; j < M; j++) {
                if (ratings[U][j] > 0 && ratings[v][j] > 0) {
                    double r1 = ratings[U][j] - avg[U];
                    double r2 = ratings[v][j] - avg[v];
                    num += r1 * r2;
                    den1 += r1 * r1;
                    den2 += r2 * r2;
                }
            }
            sim[v] = (den1 == 0 || den2 == 0) ? 0.0 : num / (Math.sqrt(den1) * Math.sqrt(den2));
        }

        List<Integer> idx = new ArrayList<>();
        for (int i = 0; i < N; i++) if (i != U) idx.add(i);
        idx.sort((a, b) -> Double.compare(sim[b], sim[a]));
        List<Integer> topK = new ArrayList<>();
        for (int i = 0; i < K && i < idx.size(); i++) topK.add(idx.get(i));

        List<double[]> preds = new ArrayList<>();
        for (int j = 0; j < M; j++) {
            if (ratings[U][j] > 0) continue;
            double num = 0, den = 0;
            for (int v : topK) {
                if (sim[v] == 0) continue;
                if (ratings[v][j] > 0) {
                    num += sim[v] * (ratings[v][j] - avg[v]);
                    den += Math.abs(sim[v]);
                }
            }
            double p = den == 0 ? avg[U] : avg[U] + num / den;
            preds.add(new double[]{p, j});
        }

        preds.sort((a, b) -> {
            if (b[0] != a[0]) return Double.compare(b[0], a[0]);
            return Integer.compare((int)a[1], (int)b[1]);
        });

        for (int i = 0; i < T; i++) {
            System.out.print((int)preds.get(i)[1]);
            if (i < T - 1) System.out.print(" ");
        }
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

int main() {
    int N, M;
    cin >> N >> M;
    vector<vector<int>> ratings(N, vector<int>(M));
    for (int i = 0; i < N; ++i)
        for (int j = 0; j < M; ++j)
            cin >> ratings[i][j];
    
    int U, T;
    cin >> U >> T;
    const int K = 3;
    
    // 计算每位学生的平均评分
    vector<double> userAvg(N, 0.0);
    for (int u = 0; u < N; ++u) {
        int cnt = 0;
        double sum = 0.0;
        for (int j = 0; j < M; ++j) {
            if (ratings[u][j] > 0) {
                cnt++;
                sum += ratings[u][j];
            }
        }
        if (cnt > 0) userAvg[u] = sum / cnt;
    }
    
    // 计算目标学生U与其他学生的相似度
    vector<double> sim(N, 0.0);
    for (int v = 0; v < N; ++v) {
        if (v == U) continue;
        double num = 0.0, den1 = 0.0, den2 = 0.0;
        for (int j = 0; j < M; ++j) {
            if (ratings[U][j] > 0 && ratings[v][j] > 0) {
                double r1 = ratings[U][j] - userAvg[U];
                double r2 = ratings[v][j] - userAvg[v];
                num += r1 * r2;
                den1 += r1 * r1;
                den2 += r2 * r2;
            }
        }
        if (den1 == 0 || den2 == 0)
            sim[v] = 0.0;
        else
            sim[v] = num / (sqrt(den1) * sqrt(den2));
    }
    
    // 选取最相似的前K个学生
    vector<int> idx;
    for (int i = 0; i < N; ++i)
        if (i != U) idx.push_back(i);
    
    sort(idx.begin(), idx.end(), [&](int a, int b) {
        return sim[a] > sim[b];
    });
    vector<int> topK;
    for (int i = 0; i < K && i < (int)idx.size(); ++i)
        topK.push_back(idx[i]);
    
    // 预测未作答习题的评分
    vector<pair<double, int>> preds; // (预测评分, 习题编号)
    for (int j = 0; j < M; ++j) {
        if (ratings[U][j] > 0) continue;
        double num = 0.0, den = 0.0;
        for (int v : topK) {
            if (sim[v] == 0) continue;
            if (ratings[v][j] > 0) {
                num += sim[v] * (ratings[v][j] - userAvg[v]);
                den += fabs(sim[v]);
            }
        }
        double pred;
        if (den == 0.0) pred = userAvg[U];
        else pred = userAvg[U] + num / den;
        preds.push_back({pred, j});
    }
    
    // 排序：预测评分降序，相同则习题编号升序
    sort(preds.begin(), preds.end(), [](const pair<double,int>& a, const pair<double,int>& b) {
        if (a.first != b.first) return a.first > b.first;
        return a.second < b.second;
    });
    
    // 输出前T个习题编号
    for (int i = 0; i < T; ++i) {
        cout << preds[i].second;
        if (i < T - 1) cout << ' ';
    }
    cout << endl;
    
    return 0;
}
```
