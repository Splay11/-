## 思路：构造

判断实际送达时间 $t_3$ 是否晚于预计送达时间 $t_2$ 。

需要注意的是，这里可能会存在跨过凌晨 $0$ 点的时间，对于这部分时间，我们将小时数加 $24$ ，这样一定可以保证 $t_3$ 和 $t_2$ 是大于 $t_1$ 的，然后再去比较 $t_3$ 是否大于 $t_2$ 即可。

时间复杂度：$O(1)$

## 代码

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 判断 h1:m1 是否晚于 h2:m2
bool gt(int h1, int m1, int h2, int m2) {
    return h1 > h2 || (h1 == h2 && m1 > m2);
}

int main() {
    int n;
    cin >> n;

    for (int i = 0; i < n; i++) {
        int h1, m1, h2, m2, h3, m3;
        char dummy;
        
        cin >> h1 >> dummy >> m1;
        cin >> h2 >> dummy >> m2;
        cin >> h3 >> dummy >> m3;

        // 如果 h1:m1 晚于 h2:m2 ，则说明过了凌晨 0 点，h2 加 24 即可
        if (gt(h1, m1, h2, m2)) {
            h2 += 24;
        }
        // 如果 h1:m1 晚于 h3:m3 ，则说明过了凌晨 0 点，h3 加 24 即可
        if (gt(h1, m1, h3, m3)) {
            h3 += 24;
        }

        // 判断 h3:m3 是否晚于 h2:m2，晚于则是超时，否则就是未超时
        if (gt(h3, m3, h2, m2)) {
            cout << "Yes" << endl;
        } else {
            cout << "No" << endl;
        }
    }

    return 0;
}
```

### Java

```java
import java.util.Scanner;

public class Main {
    // 判断 h1:m1 是否晚于 h2:m2
    public static boolean gt(int h1, int m1, int h2, int m2) {
        return h1 > h2 || (h1 == h2 && m1 > m2);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            int h1, m1, h2, m2, h3, m3;
            String time1, time2, time3;

            time1 = scanner.next();
            time2 = scanner.next();
            time3 = scanner.next();

            String[] parts1 = time1.split(":");
            String[] parts2 = time2.split(":");
            String[] parts3 = time3.split(":");

            h1 = Integer.parseInt(parts1[0]);
            m1 = Integer.parseInt(parts1[1]);
            h2 = Integer.parseInt(parts2[0]);
            m2 = Integer.parseInt(parts2[1]);
            h3 = Integer.parseInt(parts3[0]);
            m3 = Integer.parseInt(parts3[1]);

            // 如果 h1:m1 晚于 h2:m2 ，则说明过了凌晨 0 点，h2 加 24 即可
            if (gt(h1, m1, h2, m2)) {
                h2 += 24;
            }
            // 如果 h1:m1 晚于 h3:m3 ，则说明过了凌晨 0 点，h3 加 24 即可
            if (gt(h1, m1, h3, m3)) {
                h3 += 24;
            }

            // 判断 h3:m3 是否晚于 h2:m2，晚于则是超时，否则就是未超时
            if (gt(h3, m3, h2, m2)) {
                System.out.println("Yes");
            } else {
                System.out.println("No");
            }
        }
    }
}
```
