### 思路

* **关键结论**：对前缀 $a_1,\dots,a_i$ 的元素集合 $S_i$，令

  * $M_i=\max(S_i)$
  * $D_i=|S_i|$（不同元素个数）
* 因为 $m$ 必须满足 $m\ge M_i$，且集合中所有元素都不超过 $m$，所以在任意 $m\ge M_i$ 下，已有的属于 $1..m$ 的不同元素个数就是 $D_i$。要补全为 $1..m$ 的排列，需要插入的数量为 $m-D_i$。当取最小可行的 $m=M_i$ 时插入数最少。
* 因而答案为：

$$
\text{ans}_i = M_i - D_i
$$

* 实现上，单次线性扫描，维护当前前缀的最大值 $M_i$ 与去重计数 $D_i$（用集合判断新元素），即可在线输出。

### 正确性与复杂度

* **正确性**：当 $m>M_i$ 时，$\text{插入}=m-D_i$ 单调增大；最小取 $m=M_i$，得到 $\text{ans}_i=M_i-D_i$。
* **复杂度**：总时间 $O(n)$，空间 $O(n)$。
## 代码

### C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int n;
	if (!(cin >> n)) return 0;
	vector<long long> a(n);
	for (int i = 0; i < n; ++i) cin >> a[i];

	unordered_set<long long> seen;
	seen.reserve((size_t)2 * n + 1);
	long long curMax = 0;
	long long distinct = 0;

	for (int i = 0; i < n; ++i) {
		long long x = a[i];
		if (x > curMax) curMax = x;
		if (seen.insert(x).second) {
			// 新的不同元素
			++distinct;
		}
		long long ans = curMax - distinct;
		if (i) cout << ' ';
		cout << ans;
	}
	cout << '\n';
	return 0;
}
```

### Python 

```python
import sys

def main():
	# 读入
	data = sys.stdin.read().strip().split()
	if not data:
		return
	it = iter(data)
	n = int(next(it))
	a = [int(next(it)) for _ in range(n)]

	seen = set()
	cur_max = 0
	distinct = 0
	out = []

	for x in a:
		if x > cur_max:
			cur_max = x
		if x not in seen:
			seen.add(x)
			distinct += 1
		out.append(str(cur_max - distinct))

	print(" ".join(out))

if __name__ == "__main__":
	main()
```

```java
import java.io.*;
import java.util.*;

public class Main {
	static class FastScanner {
		private final InputStream in;
		private final byte[] buffer = new byte[1 << 16];
		private int ptr = 0, len = 0;
		FastScanner(InputStream is) { in = is; }
		private int read() throws IOException {
			if (ptr >= len) {
				len = in.read(buffer);
				ptr = 0;
				if (len <= 0) return -1;
			}
			return buffer[ptr++];
		}
		int nextInt() throws IOException {
			int c, sgn = 1, val = 0;
			do { c = read(); } while (c <= 32);
			if (c == '-') { sgn = -1; c = read(); }
			while (c > 32) {
				val = val * 10 + (c - '0');
				c = read();
			}
			return val * sgn;
		}
	}

	public static void main(String[] args) throws Exception {
		FastScanner fs = new FastScanner(System.in);
		int n = fs.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; i++) a[i] = fs.nextInt();

		HashSet<Integer> seen = new HashSet<>(Math.max(16, n * 2));
		long curMax = 0;
		long distinct = 0;
		StringBuilder sb = new StringBuilder();

		for (int i = 0; i < n; i++) {
			int x = a[i];
			if (x > curMax) curMax = x;
			if (seen.add(x)) {
				// 新的不同元素
				distinct++;
			}
			long ans = curMax - distinct;
			if (i > 0) sb.append(' ');
			sb.append(ans);
		}
		System.out.println(sb.toString());
	}
}
```
