### 题面描述

给定 $n$ 名用户及其初始等级分 $r_i$。每位用户可能有若干条发言，每条发言给出一个等级分 $s$。对每个用户 $i$，计算其所有发言的算术平均等级分与其初始等级分的差的绝对值，如果该差值严格大于阈值 $x$，则判定为“不一致”。未发言用户不计入统计。输出“不一致”的用户数量。


### 思路

* **核心**：对每个用户统计发言总分 `sum` 与发言次数 `cnt`。不一致条件为

  * `|(sum / cnt) − r| > x`
  * 为避免浮点误差，改写为整数比较：`|sum − r * cnt| > x * cnt`
* **未发言**：`cnt = 0` 的用户不参与统计。
* **复杂度**：`O(n + m)`，只需一次遍历与哈希映射查询。
* **数据范围安全**：`sum、r*cnt、x*cnt` 均 ≤ $5e8$，使用 $64$ 位整型更稳妥。


### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	int n, m;
	long long x;
	if (!(cin >> n >> m >> x)) return 0;

	unordered_map<string, int> id;
	id.reserve(n * 2);
	vector<long long> r(n, 0), sum(n, 0), cnt(n, 0);

	for (int i = 0; i < n; ++i) {
		string name;
		long long ri;
		cin >> name >> ri;
		id[name] = i;
		r[i] = ri;
	}

	for (int j = 0; j < m; ++j) {
		string name;
		long long s;
		cin >> name >> s;
		int i = id[name];
		sum[i] += s;
		cnt[i] += 1;
	}

	int ans = 0;
	for (int i = 0; i < n; ++i) {
		if (cnt[i] == 0) continue; // 未发言不计入
		long long left = llabs(sum[i] - r[i] * cnt[i]);
		long long right = x * cnt[i];
		if (left > right) ++ans;   // 严格大于阈值
	}
	cout << ans << '\n';
	return 0;
}
```

### Python 

```python
import sys

def main():
	data = sys.stdin.read().strip().split()
	if not data:
		return
	it = iter(data)

	n = int(next(it)); m = int(next(it)); x = int(next(it))

	# 映射用户到索引
	idx = {}
	r = []
	sumv = []
	cnt = []

	for i in range(n):
		name = next(it)
		ri = int(next(it))
		idx[name] = i
		r.append(ri)
		sumv.append(0)
		cnt.append(0)

	for _ in range(m):
		name = next(it)
		s = int(next(it))
		i = idx[name]
		sumv[i] += s
		cnt[i] += 1

	ans = 0
	for i in range(n):
		if cnt[i] == 0:
			continue  # 未发言不计入
		left = abs(sumv[i] - r[i] * cnt[i])
		right = x * cnt[i]
		if left > right:  # 严格大于
			ans += 1

	print(ans)

if __name__ == "__main__":
	import threading
	threading.Thread(target=main,).start()
```
### Java

```java
import java.io.*;
import java.util.*;

public class Main {
	public static void main(String[] args) throws Exception {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st;

		st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(st.nextToken());
		long x = Long.parseLong(st.nextToken());

		HashMap<String, Integer> idx = new HashMap<>(n * 2);
		long[] r = new long[n];
		long[] sum = new long[n];
		long[] cnt = new long[n];

		for (int i = 0; i < n; i++) {
			st = new StringTokenizer(br.readLine());
			String name = st.nextToken();
			long ri = Long.parseLong(st.nextToken());
			idx.put(name, i);
			r[i] = ri;
		}

		for (int j = 0; j < m; j++) {
			st = new StringTokenizer(br.readLine());
			String name = st.nextToken();
			long s = Long.parseLong(st.nextToken());
			int i = idx.get(name);
			sum[i] += s;
			cnt[i] += 1;
		}

		int ans = 0;
		for (int i = 0; i < n; i++) {
			if (cnt[i] == 0) continue; // 未发言不计入
			long left = Math.abs(sum[i] - r[i] * cnt[i]);
			long right = x * cnt[i];
			if (left > right) ans++;   // 严格大于阈值
		}

		System.out.println(ans);
	}
}
```
