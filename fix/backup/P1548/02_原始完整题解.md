### 思路

对于每个格子如果出现了tzg三个字母中的一个，都有可能会出现符合条件的三角形。那么我们枚举三角形的其中一个顶点，那么另外两个顶点就会存在于该点同一行和同一列，所以我们可以统计出三个单词每行每列的个数，当出现符合条件的单词时，将另外的两个单词分别在同行和同列，同列和同行的情况两两组合。

### 代码

**C++**

```cpp
#include <bits/stdc++.h>

using namespace std;

#define ll long long
const int M = 1005;

string s[M];
ll sumt[2][M], sumz[2][M], sumg[2][M]; // 第一维为0时表示行，1表示列

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (s[i][j] == 't') sumt[0][i]++, sumt[1][j]++; // 统计每行每列的tzg的个数
            if (s[i][j] == 'z') sumz[0][i]++, sumz[1][j]++;
            if (s[i][j] == 'g') sumg[0][i]++, sumg[1][j]++;
        }
    }
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (s[i][j] == 't') 
                ans += sumz[0][i] * sumg[1][j] + sumz[1][j] * sumg[0][i]; // 当前位置为t时，让行为z列为g两两组合和行为g列为z两两组合
            if (s[i][j] == 'z') 
                ans += sumt[0][i] * sumg[1][j] + sumt[1][j] * sumg[0][i]; // 同理
            if (s[i][j] == 'g') 
                ans += sumz[0][i] * sumt[1][j] + sumz[1][j] * sumt[0][i];  // 同理
        }
    }
    cout << ans << endl;
    return 0;
}
```

            s[i] = scanner.nextLine();
        }

        long[][] sumg = new long[2][1005];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                char c = s[i].charAt(j); // 统计每行每列的tzg的个数
                if (c == 't') {
                    sumt[0][i]++;
                    sumt[1][j]++;
                } else if (c == 'z') {
                    sumz[0][i]++;
                    sumz[1][j]++;
                } else if (c == 'g') {
                    sumg[0][i]++;
                    sumg[1][j]++;
                }
            }
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                char c = s[i].charAt(j);
                if (c == 't') {
                    ans += sumz[0][i] * sumg[1][j] + sumz[1][j] * sumg[0][i]; // 当前位置为t时，让行为z列为g两两组合和行为g列为z两两组合
                } else if (c == 'z') {
                    ans += sumt[0][i] * sumg[1][j] + sumt[1][j] * sumg[0][i]; // 同理
                } else if (c == 'g') {
                    ans += sumz[0][i] * sumt[1][j] + sumz[1][j] * sumt[0][i]; // 同理
                }
            }
        }

        System.out.println(ans);
    }
}
