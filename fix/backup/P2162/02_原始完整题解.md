# 思路
观察一下可以知道一个数最大操作数也就是logn次 ，发现n最大只有10^12，暴力判断一次素数则需要10^6，且判断的越来越快，o(sqrt(n)*logn)满足时间复杂度
## 代码如下
### cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

// 判断是否为素数的函数
bool isPrime(long long n) {
    // 处理小于等于1的情况，因为1及以下的数不是素数
    if (n <= 1) {
        return false;
    }
    // 从2开始检查到 sqrt(n)，如果存在因子，则不是素数
    for (long long i = 2; i * i <= n; ++i) {
        if (n % i == 0) {
            return false; // 如果有因子，说明不是素数
        }
    }
    return true; // 没有因子，说明是素数
}

int main() {
    long long n;
    // 输入糖果数量
    cin >> n;
    int ans = 0; // 记录天数的变量

    // 模拟每天吃糖果的过程，直到糖果吃完
    while (n > 0) {
        ans += 1; // 计数一天

        // 判断糖果数量是否为素数
        if (isPrime(n)) {
            // 如果是素数，吃 n/3 + 1 颗糖果
            n = n - (n / 3 + 1);
        } else {
            // 如果不是素数，吃 n/2 + 1 颗糖果
            n = n - (n / 2 + 1);
        }
    }

    // 输出吃糖果的天数
    cout << ans << '\n';
}
```
### python
```python
import math

# 判断是否为素数的函数
def is_prime(n):
    if n <= 1:
        return False  # 1及以下的数不是素数
    for i in range(2, int(math.sqrt(n)) + 1):
        if n % i == 0:
            return False  # 如果有因子，说明不是素数
    return True  # 没有因子，说明是素数

# 从输入读取糖果数量
n = int(input())
days = 0  # 记录天数

# 循环直到糖果数量为0
while n > 0:
    days += 1  # 计数一天
    if is_prime(n):
        # 如果糖果数量是素数，吃 n//3 + 1 颗
        n -= (n // 3 + 1)
    else:
        # 如果不是素数，吃 n//2 + 1 颗
        n -= (n // 2 + 1)

# 输出吃糖果的总天数
print(days)
```
### java
```java
import java.util.Scanner;

public class Main {

    // 判断是否为素数的函数
    public static boolean isPrime(long n) {
        if (n <= 1) {
            return false; // 1及以下的数不是素数
        }
        for (long i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                return false; // 如果有因子，说明不是素数
            }
        }
        return true; // 没有因子，说明是素数
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong(); // 从输入读取糖果数量
        int days = 0; // 记录天数

        // 循环直到糖果数量为0
        while (n > 0) {
            days++; // 计数一天
            if (isPrime(n)) {
                // 如果糖果数量是素数，吃 n/3 + 1 颗
                n -= (n / 3 + 1);
            } else {
                // 如果不是素数，吃 n/2 + 1 颗
                n -= (n / 2 + 1);
            }
        }

        // 输出吃糖果的总天数
        System.out.println(days);
    }
}
```
