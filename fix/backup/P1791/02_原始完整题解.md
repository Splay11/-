## gcd

题目要求将绳子都切成同样的长度，细想可以发现，如果要是该长度能够切出来，那么就一定是切成所有数的公约数。进一步，切成最大公约数，切的次数就是最少的。

所以求出最大公约数，判断每个$a_i$需要切多少次，累计次数并与$k$比较即可。

时间复杂度为$O(TNlogN)$。题目保证$\sum{n}\le1e5$。

## 代码

**C++**

```c
#include<bits/stdc++.h>
using namespace std;
#define ll long long

const int maxn=1e5+10;
int a[maxn];
int n;
ll k;

int gcd(int x,int y){
	return y==0 ? x:gcd(y,x%y);
}

int main(){
	std::ios::sync_with_stdio(false);
	int T;
	cin>>T;
	while(T--){
		cin>>n>>k;
		int tmp=0;
		for(int i=1;i<=n;++i){
			cin>>a[i];
			if(tmp==0){
				tmp=a[i];
			}else{
				tmp=gcd(tmp,a[i]);
			}
		}
		for(int i=1;i<=n;++i){
			k-=(a[i]/tmp-1);
		}
		if(k<0){
			cout<<"NO"<<endl;
		}else{
			cout<<"YES"<<endl;
		}
	}
	
	return 0;
} 
```

### java
``` java
import java.util.*;
// 注意类名必须为Main
class Main{
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        for (int i = 0; i < n; i++) {
            long k = in.nextLong();
            long m = in.nextLong();
            long[] arr = new long[(int) k];
            long tmp = 0;
            for (int j = 0; j < k; j++) {
                arr[j] = in.nextLong();
                if(tmp == 0){
                    tmp = arr[j];
                }else{
                    tmp = gcd(tmp,arr[j]);
                }
            }
            long res = 0;
            for (int j = 0; j < k; j++) {
                res += (arr[j] / tmp) -1;
            }
            if(res > m){
                System.out.println("NO");
            }else{
                System.out.println("YES");
            }
        }
    }
    private static long gcd(long a,long b){
        return b == 0 ? a : gcd(b,a%b);
    }
}
```
