### 思路

* **区间判断**：分别判断 $v \in [250, 350]$ 输出 $G$，$v \in [160, 250]$ 输出 $D$，$v \in [200, 300]$ 输出 $C$。
* **输出顺序**：严格按 $G \rightarrow D \rightarrow C$ 的顺序检查与输出。
* **边界**：端点 $160, 200, 250, 300, 350$ 均属于对应闭区间。
* **复杂度**：每组 $O(1)$，总计 $O(T)$。

## 代码

### C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	int T; 
	if (!(cin >> T)) return 0;
	while (T--) {
		int v; 
		cin >> v;
		bool first = true;

		// 高铁 [250, 350]
		if (v >= 250 && v <= 350) {
			cout << "G";
			first = false;
		}
		// 动车 [160, 250]
		if (v >= 160 && v <= 250) {
			if (!first) cout << ' ';
			cout << "D";
			first = false;
		}
		// 城际列车 [200, 300]
		if (v >= 200 && v <= 300) {
			if (!first) cout << ' ';
			cout << "C";
			first = false;
		}
		if (first) cout << "other";
		cout << '\n';
	}
	return 0;
}
```

### Python 

```python
import sys

data = sys.stdin.buffer.read().split()
if not data:
    sys.exit(0)

t = int(data[0])
ans = []
idx = 1

for _ in range(t):
    v = int(data[idx]); idx += 1
    parts = []
    # 高铁 [250, 350]
    if 250 <= v <= 350:
        parts.append('G')
    # 动车 [160, 250]
    if 160 <= v <= 250:
        parts.append('D')
    # 城际列车 [200, 300]
    if 200 <= v <= 300:
        parts.append('C')
    if parts:
        ans.append(' '.join(parts))
    else:
        ans.append('other')

sys.stdout.write('\n'.join(ans))
```

```java
import java.io.*;
import java.util.*;

public class Main {
	public static void main(String[] args) throws Exception {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder out = new StringBuilder();
		int T;
		{
			String s = br.readLine();
			while (s != null && s.trim().isEmpty()) s = br.readLine();
			if (s == null) {
				System.out.print("");
				return;
			}
			T = Integer.parseInt(s.trim());
		}
		for (int i = 0; i < T; i++) {
			String line = br.readLine();
			while (line != null && line.trim().isEmpty()) line = br.readLine();
			int v = Integer.parseInt(line.trim());

			boolean first = true;
			// 高铁 [250, 350]
			if (v >= 250 && v <= 350) {
				out.append("G");
				first = false;
			}
			// 动车 [160, 250]
			if (v >= 160 && v <= 250) {
				if (!first) out.append(' ');
				out.append("D");
				first = false;
			}
			// 城际列车 [200, 300]
			if (v >= 200 && v <= 300) {
				if (!first) out.append(' ');
				out.append("C");
				first = false;
			}
			if (first) out.append("other");
			if (i + 1 < T) out.append('\n');
		}
		System.out.print(out.toString());
	}
}
```
