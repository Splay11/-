# 思路
简单的小模拟，可以遍历每一个字符，碰到'_'跳过，开头和下划线后面一个都变成大写字母输出即可
也可以利用自带的函数简易实现
## 代码如下
### cpp
```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; // 声明变量t，表示测试组数
    cin >> t; // 读取输入的t值
    while (t--) { // 循环处理每组输入
        string s; // 声明一个字符串s，用于存储变量名
        cin >> s; // 读取变量名
        for (int i = 0; i < s.length(); i++) { // 遍历字符串s的每个字符
            if (s[i] == '_') continue; // 如果当前字符是下划线，跳过
            if (i == 0 || s[i - 1] == '_') { // 如果是首字母或下划线后的第一个字母
                s[i] += 'A' - 'a'; // 将字母转为大写
            }
            cout << s[i]; // 输出字符
        }
        cout << '\n'; // 换行输出
    }
}
```
### python
```python
# 读取测试组数T
T = int(input())

# 处理每组输入
for _ in range(T):
    s = input().strip()  # 读取变量名并去掉多余的空白字符
    result = []  # 用于存储转换后的字符串
    i = 0
    
    while i < len(s):  # 遍历输入字符串s
        if s[i] == '_':  # 跳过下划线
            i += 1
            continue
        
        # 如果是第一个字符或者下划线后的字符，转为大写
        if i == 0 or s[i - 1] == '_':
            result.append(s[i].upper())  # 转换为大写字母
        else:
            result.append(s[i])  # 保留原来的字符
        
        i += 1  # 移动到下一个字符
    
    # 输出转换后的字符串
    print(''.join(result))
```
### java
```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);  // 创建输入流
        int t = scanner.nextInt();  // 读取测试组数T
        scanner.nextLine();  // 清除换行符

        // 处理每组输入
        while (t-- > 0) {
            String s = scanner.nextLine();  // 读取变量名
            StringBuilder result = new StringBuilder();  // 用于构建转换后的字符串

            for (int i = 0; i < s.length(); i++) {
                char c = s.charAt(i);  // 获取当前字符
                if (c == '_') continue;  // 跳过下划线

                // 如果是第一个字符或者下划线后的字符，转为大写
                if (i == 0 || s.charAt(i - 1) == '_') {
                    result.append(Character.toUpperCase(c));  // 转换为大写字母
                } else {
                    result.append(c);  // 保留原来的字符
                }
            }

            System.out.println(result.toString());  // 输出转换后的字符串
        }

        scanner.close();  // 关闭输入流
    }
}
```
