### 思路

由于n不大，可以直接暴力，直接枚举所有排列。然后判断相邻的数的和是否为素数，20以内素数只有8个，复杂度为$10*n!*8$。

## 代码

**C++**

```C++
#include<bits/stdc++.h>

using namespace std;

int a[15];
int pri[] = {2, 3, 5, 7, 11, 13, 17, 19}; // 20以内所有素数

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        a[i] = i;
    }
    int ans = 0;
    do {
        bool ok = 0; // 是否有相邻加起来为素数
        for (int i = 2; i <= n; i++) {
            for (int j = 0; j < 8; j++) { // 枚举所有素数
                if (a[i - 1] + a[i] == pri[j]) ok = 1;
            }
        }
        ans += !ok;
    }while (next_permutation(a + 1, a + 1 + n)); // 枚举所有情况
    cout << ans << endl;
    return 0;
}
```

**java**

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[15];
        int[] pri = {2, 3, 5, 7, 11, 13, 17, 19}; // 20以内所有素数

        for (int i = 1; i <= n; i++) {
            a[i] = i;
        }
        
        int ans = 0;
        do {
            boolean ok = false; // 是否有相邻加起来为素数
            for (int i = 2; i <= n; i++) {
                for (int j = 0; j < 8; j++) { // 枚举所有素数
                    if (a[i - 1] + a[i] == pri[j]) {
                        ok = true;
                    }
                }
            }
            ans += ok ? 0 : 1;
        } while (nextPermutation(a, 1, n)); // 枚举所有情况
        
        System.out.println(ans);
    }

    public static boolean nextPermutation(int[] a, int left, int right) {
        int i = right;
        while (i > left && a[i - 1] >= a[i]) {
            i--;
        }
        if (i <= left) {
            return false;
        }

        int j = right;
        while (a[j] <= a[i - 1]) {
            j--;
        }

        int temp = a[i - 1];
        a[i - 1] = a[j];
        a[j] = temp;

        j = right;
        while (i < j) {
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
            i++;
            j--;
        }

        return true;
    }
}

```

**python**

```python
from itertools import permutations

n = int(input())
a = list(range(1, n + 1))
pri = [2, 3, 5, 7, 11, 13, 17, 19]  # 20以内所有素数

ans = 0
for perm in permutations(a):
    ok = False  # 是否有相邻加起来为素数
    for i in range(1, n):
        if perm[i - 1] + perm[i] in pri:
            ok = True
            break
    ans += not ok

print(ans)
```
