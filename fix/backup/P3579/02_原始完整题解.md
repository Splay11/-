## 思路总览

### 等价变形

允许交换的限制是：只在相同的 $a_i$ 分组内任意交换 $s$ 的字符。
因此，**每个分组内 $s$ 的字符可以任意重排**。要让重排后的 $s$ 与 $t$ 同构（即 $t_i$ 是 $s_i$ 的大写），等价于：

* 对于每个分组 $g\in\{0,1\}$，**把该组里 $s$ 的小写字母转成大写后**得到的多重集合，必须与该组里 $t$ 的多重集合完全相同。

### 算法与方法

对每个测试用例：

1. 把位置按 $a_i$ 分成两组：0 组与 1 组。
2. 分别统计两组里 26 个字母的出现次数：

   * $S$ 侧：统计 $s$ 在各组的字母频次（小写直接映射到 0..25，相当于其对应大写）。
   * $T$ 侧：统计 $t$ 在各组的字母频次（大写映射到 0..25）。
3. 对两组逐一比较 26 维频次数组，若两组都完全一致，则输出 `Yes`，否则输出 `No`。


### 复杂度分析

* 统计与比较均为线性：每个用例 $O(n + 26)$，总复杂度 $O(\sum n)$，满足 $\sum n \le 2\times 10^5$ 的数据范围。
* 额外空间：常数级（两个 $2\times 26$ 的计数数组）。


## 代码实现

### Python

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    it = iter(data)

    T = int(next(it))
    out = []
    for _ in range(T):
        n = int(next(it))
        a = [int(next(it)) for _ in range(n)]
        s = next(it).decode()
        t = next(it).decode()

        # 计数数组：两个分组，每组 26 个字母
        cntS = [[0]*26 for _ in range(2)]
        cntT = [[0]*26 for _ in range(2)]

        # 遍历位置，分别统计
        for i in range(n):
            g = a[i]
            # s 是小写，映射到 0..25
            si = ord(s[i]) - ord('a')
            # t 是大写，映射到 0..25
            ti = ord(t[i]) - ord('A')
            cntS[g][si] += 1
            cntT[g][ti] += 1

        ok = (cntS[0] == cntT[0]) and (cntS[1] == cntT[1])
        out.append("Yes" if ok else "No")

    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

### Java 实现

```java
import java.io.*;
import java.util.*;

public class Main {
    static class FastIn {
        BufferedReader br;
        StringTokenizer st;
        FastIn(InputStream in) { br = new BufferedReader(new InputStreamReader(in)); }
        String next() throws IOException {
            while (st == null || !st.hasMoreElements()) {
                String line = br.readLine();
                if (line == null) return null;
                st = new StringTokenizer(line);
            }
            return st.nextToken();
        }
        int nextInt() throws IOException { return Integer.parseInt(next()); }
    }

    public static void main(String[] args) throws Exception {
        FastIn in = new FastIn(System.in);
        String tok = in.next();
        if (tok == null) return;
        int T = Integer.parseInt(tok);
        StringBuilder sb = new StringBuilder();

        while (T-- > 0) {
            int n = Integer.parseInt(in.next());
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = Integer.parseInt(in.next());

            // 读 s 和 t
            String s = in.next();
            String t = in.next();

            int[][] cntS = new int[2][26];
            int[][] cntT = new int[2][26];

            // 统计各组频次
            for (int i = 0; i < n; i++) {
                int g = a[i];
                int si = s.charAt(i) - 'a'; // 小写映射
                int ti = t.charAt(i) - 'A'; // 大写映射
                cntS[g][si]++;
                cntT[g][ti]++;
            }

            boolean ok = true;
            for (int g = 0; g < 2 && ok; g++) {
                for (int c = 0; c < 26; c++) {
                    if (cntS[g][c] != cntT[g][c]) { ok = false; break; }
                }
            }
            sb.append(ok ? "Yes" : "No").append('\n');
        }

        System.out.print(sb.toString());
    }
}
```

```cpp
// 思路：按 a_i 分组，统计 s(小写) 与 t(大写) 各组的 26 字母频次，比较即得
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    if (!(cin >> T)) return 0;
    while (T--) {
        int n; 
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        string s, t;
        cin >> s >> t;

        int cntS[2][26] = {}; // 初始化为 0
        int cntT[2][26] = {};

        for (int i = 0; i < n; ++i) {
            int g = a[i];
            int si = s[i] - 'a'; // 小写映射
            int ti = t[i] - 'A'; // 大写映射
            cntS[g][si]++;
            cntT[g][ti]++;
        }

        bool ok = true;
        for (int g = 0; g < 2 && ok; ++g) {
            for (int c = 0; c < 26; ++c) {
                if (cntS[g][c] != cntT[g][c]) { ok = false; break; }
            }
        }
        cout << (ok ? "Yes" : "No") << '\n';
    }
    return 0;
}
```
