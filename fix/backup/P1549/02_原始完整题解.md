## 思路

其实这道题考的就是思维，目的要让所有数尽量平均，其实自己拿两个数模拟一下就能知道，一个数加1一个数减1是可以让所有都尽量平均，所以直接通过所有数之和判断能否使所有数都在[L,R]之间，要让操作最小，那就让操作不浪费，每一次操作都是使必须要加的数加1和要减的数减1，直接在必须要加的次数与必须要减的次数中取最大值


## 代码
###  java
```java
import java.util.*;
class Main {
    static Scanner sc = new Scanner(System.in);
    static int N=200010;
    static long T,n,l,r;
    static long[] arr=new long[N];
    public static void main(String[] args) {
        T=sc.nextLong();
        for(int t=1;t<=T;t++)
        {
            n=sc.nextLong();
            l=sc.nextLong();
            r=sc.nextLong();
            long sum=0,minn=0,maxn=0;
            for(int i=1;i<=n;i++)
            {
                arr[i]=sc.nextLong();
                sum+=arr[i];
                if(arr[i]<l)minn+=l-arr[i];
                else if(arr[i]>r)maxn+=arr[i]-r;
            }
            if(sum<n*l||sum>n*r)System.out.println(-1);
            else System.out.println(Math.max(minn,maxn));
        }
    }
}
```
### C++
```cpp
#include <iostream>
using namespace std;

const int N = 200010;
long long arr[N];

int main() {
    long long T, n, l, r;
    cin >> T;
    
    for (long long t = 1; t <= T; t++) {
        cin >> n >> l >> r;
        long long sum = 0, minn = 0, maxn = 0;
        
        for (long long i = 1; i <= n; i++) {
            cin >> arr[i];
            sum += arr[i];
            
            if (arr[i] < l)
                minn += l - arr[i];
            else if (arr[i] > r)
                maxn += arr[i] - r;
        }
        
        if (sum < n * l || sum > n * r)
            cout << -1 << endl;
        else
            cout << max(minn, maxn) << endl;
    }
    
    return 0;
}
```

```python
T = int(input())

for t in range(1, T + 1):
    n, l, r = map(int, input().split())
    arr = list(map(int, input().split()))
    sum_val = sum(arr)
    minn = 0
    maxn = 0

    for i in range(n):
        if arr[i] < l:
            minn += l - arr[i]
        elif arr[i] > r:
            maxn += arr[i] - r

    if sum_val < n * l or sum_val > n * r:
        print(-1)
    else:
        print(max(minn, maxn))
```
