## 题面描述

小红有一个长度为 $3$ 且仅由小写字母组成的字符串 $s(1)$。她想基于这个字符串，构造一个总共包含 $n$ 个字符串的序列 $s(1), s(2), \dots, s(n)$，要求如下：

1. **同质性**：序列中每个字符串都与 $s(1)$ 的字母种类和对应数量相同（可视为 $s(1)$ 的任意排列）。
2. **接龙性**：对于任意 $i$（$2 \leq i \leq n$），需满足 $s(i)$ 的首字母等于 $s(i-1)$ 的末字母。

请你计算，在给定初始字符串 $s(1)$ 和整数 $n$ 的条件下，满足上述要求的不同序列方案有多少种。若两种方案中存在某个位置的字符串不同，则视为两种不同方案。最终答案可能很大，请对 $10^9+7$ 取模后输出。

### 思路

本题使用**动态规划**结合**记忆化搜索**来解决问题。通过定义状态 `dp[i][x][y][z]` 表示长度为 `i` 的字符串，最后三个字符分别为 `x`, `y`, `z` 时的合法字符串个数，并通过递归枚举所有可能的字符组合，避免连续三个字符相同的情况。利用记忆化存储已计算的状态，避免重复计算，最终输出满足条件的字符串个数，结果对 `10^9 + 7` 取模。

1. **状态表示**：我们可以使用动态规划来解决这个问题。定义 `dp[i][x][y][z]` 表示长度为 `i` 的字符串，且最后三个字符分别为 `x`, `y`, `z` 时的合法字符串个数。

2. **状态转移**：对于长度为 `i` 的字符串，我们考虑它的最后三个字符 `x`, `y`, `z`：
   - 如果 `x`, `y`, `z` 不全相同，那么我们可以通过在长度为 `i-1` 的字符串后面添加一个字符来得到长度为 `i` 的字符串。
   - 如果 `x`, `y`, `z` 全相同，那么这种情况是不合法的，直接跳过。

3. **初始条件**：当 `i = 1` 时，只有一种情况，即字符串长度为 1，直接返回 1。

4. **记忆化搜索**：为了避免重复计算，我们使用记忆化搜索来存储已经计算过的状态。

### C++ 代码

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;
int T;
string s;
int n;
unordered_map<char, int> mp;
vector<unordered_map<int, int>> dp;

int dfs(int i, int x, int y, int z) {
    if (i == 1) return 1; // 初始条件
    int k = x + y * 3 + z * 9; // 将状态编码为一个整数
    if (dp[i].count(k)) return dp[i][k]; // 如果已经计算过，直接返回
    if (x != y) {
        dp[i][k] = (dfs(i - 1, z, x, y) + dfs(i - 1, z, y, x)) % MOD; // 状态转移
    } else {
        dp[i][k] = dfs(i - 1, z, x, y); // 状态转移
    }
    return dp[i][k];
}

int main() {
    cin >> T;
    while (T--) {
        cin >> s >> n;
        mp.clear();
        int p = 0;
        for (char c : s) {
            if (!mp.count(c)) {
                mp[c] = p++;
            }
        }
        dp.assign(n + 1, unordered_map<int, int>());
        cout << dfs(n, mp[s[0]], mp[s[1]], mp[s[2]]) % MOD << endl;
    }
    return 0;
}
```

### Java 代码

```java
import java.util.*;

public class Main {
    static final int MOD = (int)1e9 + 7;
    static int T;
    static String s;
    static int n;
    static Map<Character, Integer> mp;
    static List<Map<Integer, Integer>> dp;

    static int dfs(int i, int x, int y, int z) {
        if (i == 1) return 1; // 初始条件
        int k = x + y * 3 + z * 9; // 将状态编码为一个整数
        if (dp.get(i).containsKey(k)) return dp.get(i).get(k); // 如果已经计算过，直接返回
        if (x != y) {
            dp.get(i).put(k, (dfs(i - 1, z, x, y) + dfs(i - 1, z, y, x)) % MOD); // 状态转移
        } else {
            dp.get(i).put(k, dfs(i - 1, z, x, y)); // 状态转移
        }
        return dp.get(i).get(k);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        T = sc.nextInt();
        while (T-- > 0) {
            s = sc.next();
            n = sc.nextInt();
            mp = new HashMap<>();
            int p = 0;
            for (char c : s.toCharArray()) {
                if (!mp.containsKey(c)) {
                    mp.put(c, p++);
                }
            }
            dp = new ArrayList<>();
            for (int i = 0; i <= n; i++) {
                dp.add(new HashMap<>());
            }
            System.out.println(dfs(n, mp.get(s.charAt(0)), mp.get(s.charAt(1)), mp.get(s.charAt(2))) % MOD);
        }
    }
}
```
## python
```python
import sys
sys.setrecursionlimit(2000000)  # 设置递归深度，避免递归过深导致栈溢出

mod = 10**9 + 7  # 定义模数
T = int(input())  # 读取测试用例的数量
dp = [{} for i in range(100003)]  # 初始化动态规划数组，dp[i]用于存储长度为i时的状态

while T > 0:
    T -= 1
    s, n = input().split()  # 读取字符串s和整数n
    n = int(n)
    mp = {}  # 用于将字符映射为整数
    p = 0  # 映射的整数值
    for i in s:
        if i not in mp:
            mp[i] = p  # 将字符映射为整数
            p += 1

    # 定义递归函数dfs，计算长度为i，最后三个字符为x, y, z时的合法字符串个数
    def dfs(i, x, y, z):
        if i == 1:  # 如果长度为1，只有一种情况
            return 1
        k = x + y * 3 + z * 9  # 将状态编码为一个整数，方便存储
        if k in dp[i]:  # 如果状态已经计算过，直接返回结果
            return dp[i][k]
        if x != y:  # 如果前两个字符不相同，可以交换后两个字符
            dp[i][k] = (dfs(i - 1, z, x, y) + dfs(i - 1, z, y, x)) % mod
        else:  # 如果前两个字符相同，只能保持顺序
            dp[i][k] = dfs(i - 1, z, x, y)
        return dp[i][k]  # 返回当前状态的结果

    # 输出结果，初始状态为s的前三个字符对应的映射值
    print(dfs(n, mp[s[0]], mp[s[1]], mp[s[2]]) % mod)
```
