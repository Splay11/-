## 解题思路

我们只允许**至多一次**把一个子段 `[l,r]` 反转。逆序对只与**相对顺序**有关。把 `[l,r]` 反转时：

* 区间外与区间内的“跨区间对”在反转前后依然满足“左下标 < 右下标”，比较的两值也没变，因此这部分逆序对数量**不变**。
* **唯一会改变**的是区间内部两两成对的关系。

设区间长度为 `k = r-l+1`，区间内

* `invSeg`：大于对数（`ai > aj`）；
* `eqSeg`：相等对数（`ai = aj`，不计入逆序）；
* 总对数 `C = k*(k-1)/2`。

反转后，区间内的新逆序对数变为“原来的小于对数”，即 `C - invSeg - eqSeg`。
因此区间内部的变化量

```
Δ_in = 新内部逆序 - 旧内部逆序
     = (C - invSeg - eqSeg) - invSeg
     = C - eqSeg - 2*invSeg
```

总逆序对变为 `ans = inv_total + Δ_in`。所以问题变成：在所有 `[l,r]` 中**最小化** `C - eqSeg - 2*invSeg`（或等价地最大化收益 `2*invSeg + eqSeg - C`）。

如何高效求每个 `[l,r]` 的 `invSeg` 和 `eqSeg`？
对固定的 `l`，从左到右扩展 `r`，维护：

* 一个树状数组（Fenwick / BIT）记录当前段内每个值出现次数，用于在加入 `a[r]` 时求“已有元素中 > a\[r] 的个数”，即增量 `cntGreater`，累加到 `invSeg`；
* 一个频次数组 `freq`，使得加入 `a[r]` 时 `cntEqual = freq[a[r]]`，累加到 `eqSeg`；
* 每次更新 `C = k*(k-1)/2`，据式子更新答案。

为支持比较与 BIT，下标需**坐标压缩**。
整体算法：先用 BIT 在 `O(n log n)` 求原始逆序对 `inv_total`；然后枚举所有 `l`，线性扩 `r`，每次 `O(log n)` 更新，复杂度 `O(n^2 log n)`，在 `n ≤ 2000` 下充裕。

**算法范畴**：枚举 + 数据结构（树状数组 / Fenwick Tree）+ 坐标压缩。

## 复杂度分析

* 时间复杂度：`O(n log n + n^2 log n)`（求初始逆序对一次 + 枚举所有子段并用 BIT 维护）。
* 空间复杂度：`O(n)`（坐标压缩数组、BIT、频次数组）。

## 代码实现

### Python

```python
import sys

class BIT:
    # 树状数组：支持前缀和
    def __init__(self, n):
        self.n = n
        self.bit = [0]*(n+1)

    def add(self, i, delta):
        while i <= self.n:
            self.bit[i] += delta
            i += i & -i

    def sum(self, i):
        s = 0
        while i > 0:
            s += self.bit[i]
            i -= i & -i
        return s

def compress(arr):
    # 坐标压缩，返回 rank 数组（从 1 开始）
    vals = sorted(set(arr))
    idx = {v:i+1 for i,v in enumerate(vals)}
    return [idx[v] for v in arr], len(vals)

def initial_inversions(rks, m):
    # 用 BIT 计算原始逆序对
    bit = BIT(m)
    inv = 0
    seen = 0
    for rk in rks:
        # 左边已出现的 > 当前 的个数
        greater_left = seen - bit.sum(rk)
        inv += greater_left
        bit.add(rk, 1)
        seen += 1
    return inv

def solve():
    data = sys.stdin.read().strip().split()
    n = int(data[0])
    a = list(map(int, data[1:1+n]))

    rks, m = compress(a)
    inv_total = initial_inversions(rks, m)

    ans = inv_total  # 可以不操作

    # 枚举左端点 l
    for l in range(n):
        bit = BIT(m)           # 区间 [l, r] 内的值频次，用于求 > 的个数
        freq = [0]*(m+1)       # 用于统计相等对数增量
        invSeg = 0             # 区间内逆序对数
        eqSeg = 0              # 区间内相等对数
        k = 0                  # 当前区间长度
        for r in range(l, n):
            rk = rks[r]
            # 大于 a[r] 的已有个数
            cntGreater = k - bit.sum(rk)
            # 等于 a[r] 的已有个数
            cntEqual = freq[rk]

            invSeg += cntGreater
            eqSeg += cntEqual

            bit.add(rk, 1)
            freq[rk] += 1
            k += 1

            C = k*(k-1)//2
            delta = C - eqSeg - 2*invSeg
            cand = inv_total + delta
            if cand < ans:
                ans = cand

    print(ans)

if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 树状数组（Fenwick）
    static class BIT {
        int n;
        long[] bit;
        BIT(int n) {
            this.n = n;
            bit = new long[n+1];
        }
        void add(int i, long delta) {
            while (i <= n) {
                bit[i] += delta;
                i += i & -i;
            }
        }
        long sum(int i) {
            long s = 0;
            while (i > 0) {
                s += bit[i];
                i -= i & -i;
            }
            return s;
        }
    }

    // 坐标压缩
    static int[] compress(int[] a) {
        int n = a.length;
        int[] b = a.clone();
        Arrays.sort(b);
        int m = 0;
        int[] uniq = new int[n];
        for (int i = 0; i < n; i++) {
            if (i == 0 || b[i] != b[i-1]) uniq[m++] = b[i];
        }
        uniq = Arrays.copyOf(uniq, m);
        int[] rks = new int[n];
        for (int i = 0; i < n; i++) {
            int rk = Arrays.binarySearch(uniq, a[i]) + 1; // 1-based
            rks[i] = rk;
        }
        return rks; // m = uniq.length
    }

    // 初始逆序对
    static long initialInversions(int[] rks, int m) {
        BIT bit = new BIT(m);
        long inv = 0, seen = 0;
        for (int rk : rks) {
            long greaterLeft = seen - bit.sum(rk);
            inv += greaterLeft;
            bit.add(rk, 1);
            seen++;
        }
        return inv;
    }

    public static void main(String[] args) throws Exception {
        // 标准输入（ACM 风格）
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int n = Integer.parseInt(s.trim());
        String[] parts = br.readLine().trim().split("\\s+");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = Integer.parseInt(parts[i]);

        int[] rks = compress(a);
        // 获取唯一值个数 m
        int[] tmp = a.clone();
        Arrays.sort(tmp);
        int m = 0;
        for (int i = 0; i < n; i++) if (i == 0 || tmp[i] != tmp[i-1]) m++;

        long invTotal = initialInversions(rks, m);
        long ans = invTotal;

        // 枚举左端点
        for (int l = 0; l < n; l++) {
            BIT bit = new BIT(m);           // 维护区间内频次（用于求 > 的个数）
            long[] freq = new long[m+1];    // 维护相等个数
            long invSeg = 0, eqSeg = 0;
            long k = 0;                     // 当前区间长度
            for (int r = l; r < n; r++) {
                int rk = rks[r];
                long cntGreater = k - bit.sum(rk);
                long cntEqual = freq[rk];

                invSeg += cntGreater;
                eqSeg += cntEqual;

                bit.add(rk, 1);
                freq[rk] += 1;
                k++;

                long C = k*(k-1)/2;
                long delta = C - eqSeg - 2*invSeg;
                long cand = invTotal + delta;
                if (cand < ans) ans = cand;
            }
        }

        System.out.println(ans);
    }
}
```

```cpp
#include <bits/stdc++.h>

struct BIT {
    int n;
    vector<long long> t;
    BIT(int n=0): n(n), t(n+1, 0) {}
    void reset(int n_) { n = n_; t.assign(n+1, 0); }
    void add(int i, long long delta){
        while(i<=n){ t[i]+=delta; i+=i&-i; }
    }
    long long sum(int i){
        long long s=0;
        while(i>0){ s+=t[i]; i-=i&-i; }
        return s;
    }
};

vector<int> compress(const vector<long long>& a, int &m){
    vector<long long> b=a;
    sort(b.begin(), b.end());
    b.erase(unique(b.begin(), b.end()), b.end());
    m = (int)b.size();
    vector<int> rks(a.size());
    for(size_t i=0;i<a.size();++i){
        int rk = int(lower_bound(b.begin(), b.end(), a[i]) - b.begin()) + 1;
        rks[i]=rk;
    }
    return rks;
}

long long initial_inversions(const vector<int>& rks, int m){

    for(int rk: rks){
        long long greaterLeft = seen - bit.sum(rk);
        inv += greaterLeft;
        bit.add(rk, 1);

    }
    return inv;
}

            long long C = k*(k-1)/2;
            long long delta = C - eqSeg - 2*invSeg;

            if(cand < ans) ans = cand;
        }
    }

    cout << ans << "\n";
```
