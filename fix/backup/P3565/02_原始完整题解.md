### 思路

* 设不小于 $l$ 的第一个满足同余的数为 $f$。可用如下任一等价形式求 $f$：

  * $f = l + \big((x - l) \bmod k\big)$，其中取值为 $0 \ldots k-1$ 的非负余数；
  * $f$ = $\left\lceil \dfrac{l - x}{k} \right\rceil \cdot k$ + $x$。
* 若 $f > r$，答案为 $0$，否则

$\text{ans}$ = $\left\lfloor \dfrac{r - f}{k} \right\rfloor$ + $1$

* 复杂度：每组 $O(1)$ 时间、$O(1)$ 额外空间，总体 $O(T)$。

## 代码

### C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

long long mod_nonneg(long long a, long long m) {
	// 返回非负余数 [0, m-1]
	long long r = a % m;
	if (r < 0) r += m;
	return r;
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T;
	if (!(cin >> T)) return 0;
	while (T--) {
		long long l, r, k, x;
		cin >> l >> r >> k >> x;
		// 计算不小于 l 的第一个满足 i % k == x 的数 f
		long long f = l + mod_nonneg(x - l, k);
		long long ans = 0;
		if (f <= r) {
			ans = (r - f) / k + 1;
		}
		cout << ans << '\n';
	}
	return 0;
}
```

### Python 

```python
import sys

def mod_nonneg(a: int, m: int) -> int:
    # 返回非负余数 [0, m-1]
    r = a % m
    if r < 0:
        r += m
    return r

data = sys.stdin.read().strip().split()
it = iter(data)
try:
    T = int(next(it))
except StopIteration:
    sys.exit(0)

out_lines = []
for _ in range(T):
    l = int(next(it)); r = int(next(it)); k = int(next(it)); x = int(next(it))
    f = l + mod_nonneg(x - l, k)  # 不小于 l 的第一个满足同余的数
    if f <= r:
        ans = (r - f) // k + 1
    else:
        ans = 0
    out_lines.append(str(ans))

sys.stdout.write("\n".join(out_lines))
```

### Java 

```java
import java.io.*;
import java.util.*;

public class Main {
	static class FastScanner {
		private final InputStream in;
		private final byte[] buffer = new byte[1 << 16];
		private int ptr = 0, len = 0;
		FastScanner(InputStream is) { in = is; }
		private int read() throws IOException {
			if (ptr >= len) {
				len = in.read(buffer);
				ptr = 0;
				if (len <= 0) return -1;
			}
			return buffer[ptr++];
		}
		long nextLong() throws IOException {
			int c; do { c = read(); } while (c <= ' ' && c != -1);
			boolean neg = false;
			if (c == '-') { neg = true; c = read(); }
			long x = 0;
			while (c > ' ') {
				x = x * 10 + (c - '0');
				c = read();
			}
			return neg ? -x : x;
		}
		int nextInt() throws IOException { return (int) nextLong(); }
	}
	static long modNonneg(long a, long m) {
		// 返回非负余数 [0, m-1]
		long r = a % m;
		if (r < 0) r += m;
		return r;
	}
	public static void main(String[] args) throws Exception {
		FastScanner fs = new FastScanner(System.in);
		StringBuilder sb = new StringBuilder();
		int T;
		try {
			T = fs.nextInt();
		} catch (Exception e) {
			return;
		}
		while (T-- > 0) {
			long l = fs.nextLong();
			long r = fs.nextLong();
			long k = fs.nextLong();
			long x = fs.nextLong();
			long f = l + modNonneg(x - l, k); // 不小于 l 的第一个满足同余的数
			long ans = 0;
			if (f <= r) {
				ans = (r - f) / k + 1;
			}
			sb.append(ans).append('\n');
		}
		System.out.print(sb.toString());
	}
}
```
