# 思路与证明

设给定多项式

$$
P(x)=\prod_{i=1}^{n}\bigl(x+a_i\bigr),
$$

其中每个 $a_i \in \{\pm1,\pm2,\dots,\pm9\}$（将 $(x-d)$ 视为 $x+(-d)$）。

我们要的是展开后 $x^1$ 的系数，记为 $c_1$。有一个非常简洁的结论：

$$
c_1 = P'(0).
$$

理由：若 $P(x)=c_0+c_1x+c_2x^2+\cdots$，则 $P'(x)=c_1+2c_2x+\cdots$，从而 $P'(0)=c_1$。

## 线性时间做法（前后缀积）

令

* 前缀积：$L[i]=\prod_{k=0}^{i-1} a_k$（$L[0]=1$）
* 后缀积：$R[i]=\prod_{k=i+1}^{n-1} a_k$（$R[n-1]=1$）

则

$$
\prod_{j\neq i} a_j=L[i]\cdot R[i].
$$

于是答案为

![](/file/2/K5puT38m8Y0RG4ccVxjoR.png)


时间复杂度 $O(n)$，空间复杂度 $O(n)$。
由于每个 $a_i\in\{\pm1,\dots,\pm9\}$，模 $10007$（几乎肯定是质数）下不会为 0，但我们根本不需要求逆，前后缀积即可稳妥解决。

> 备注：也可用 $\left(\prod a_i\right)\cdot \sum_i a_i^{-1}$ 的形式（需要求逆），但前后缀积更直接安全。

## 解析与输入格式

字符串严格由若干段 `"(x±d)"` 依次拼接，长度为 5 的倍数。第 `5k+2` 位是符号 `+` 或 `-`，第 `5k+3` 位是数字字符 `'1'..'9'`。
解析得到 $a_k = \pm d$，然后在模数下做前后缀积累加即可。

# C++ 

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    const int MOD = 10007;

    string s;
    if (!getline(cin, s)) return 0;

    int n = (int)s.size() / 5;           // 因为每段都是 "(x+d)"，长度为5
    vector<int> a(n);

    // 解析每个因子 (x±d) -> a_k = ±d
    for (int k = 0; k < n; ++k) {
        int base = 5 * k;
        char op = s[base + 2];           // '+' 或 '-'
        int d = s[base + 3] - '0';       // '1'..'9'
        int val = (op == '-') ? -d : d;
        // 归一化到 [0, MOD)
        int aval = (val % MOD + MOD) % MOD;
        a[k] = aval;
    }

    // 前缀积 L[i] = a[0]*...*a[i-1]，L[0]=1
    vector<int> L(n, 1), R(n, 1);
    for (int i = 1; i < n; ++i) {
        L[i] = (int)((1LL * L[i - 1] * a[i - 1]) % MOD);
    }
    // 后缀积 R[i] = a[i+1]*...*a[n-1]，R[n-1]=1
    for (int i = n - 2; i >= 0; --i) {
        R[i] = (int)((1LL * R[i + 1] * a[i + 1]) % MOD);
    }

    // 答案：sum L[i]*R[i]
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += 1LL * L[i] * R[i] % MOD;
        if (ans >= (1LL<<60)) ans %= MOD; // 防长整溢出的小优化（可有可无）
    }
    cout << (ans % MOD + MOD) % MOD << "\n";
    return 0;
}
```

# Python 

```python
import sys

MOD = 10007

def main():
    s = sys.stdin.readline().strip()
    n = len(s) // 5

    a = []
    for k in range(n):
        base = 5 * k
        op = s[base + 2]       # '+' or '-'
        d = ord(s[base + 3]) - ord('0')  # 1..9
        val = -d if op == '-' else d
        a.append((val % MOD + MOD) % MOD)

    # prefix L[i] = product a[0..i-1], L[0]=1
    L = [1] * n
    for i in range(1, n):
        L[i] = (L[i - 1] * a[i - 1]) % MOD

    # suffix R[i] = product a[i+1..n-1], R[n-1]=1
    R = [1] * n
    for i in range(n - 2, -1, -1):
        R[i] = (R[i + 1] * a[i + 1]) % MOD

    # sum L[i]*R[i]
    ans = 0
    for i in range(n):
        ans = (ans + L[i] * R[i]) % MOD

    print(ans)

if __name__ == "__main__":
    main()
```

# Java 

```java
import java.io.*;
import java.util.*;

public class Main {
    static final int MOD = 10007;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int n = s.length() / 5;

        int[] a = new int[n];
        for (int k = 0; k < n; k++) {
            int base = 5 * k;
            char op = s.charAt(base + 2);        // '+' or '-'
            int d = s.charAt(base + 3) - '0';    // 1..9
            int val = (op == '-') ? -d : d;
            int aval = ((val % MOD) + MOD) % MOD;
            a[k] = aval;
        }

        // prefix products L[i] = a[0]*...*a[i-1], L[0]=1
        int[] L = new int[n];
        int[] R = new int[n];
        L[0] = 1;
        for (int i = 1; i < n; i++) {
            L[i] = (int)((1L * L[i - 1] * a[i - 1]) % MOD);
        }

        // suffix products R[i] = a[i+1]*...*a[n-1], R[n-1]=1
        R[n - 1] = 1;
        for (int i = n - 2; i >= 0; i--) {
            R[i] = (int)((1L * R[i + 1] * a[i + 1]) % MOD);
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += (1L * L[i] * R[i]) % MOD;
            if (ans >= (1L<<60)) ans %= MOD; // 防止极端情况下的长整溢出（可选）
        }
        ans %= MOD;
        if (ans < 0) ans += MOD;

        System.out.println(ans);
    }
}
```
