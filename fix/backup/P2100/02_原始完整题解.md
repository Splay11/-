# 二分答案
只能修改连续区间的人，考虑二分答案，也就是一个最大值x使得数组中的所有数值经过修改后都大于等于x，对于小于mid的值找到下标最小的开始连续修改，看最后剩下还有没有修改次数。
## 代码如下
### cpp
```cpp
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define N 1000005
int a[N];
int n,k,ll;
bool check(int x){
	int num=k;
	for(int i=1;i<=n;i++){
		if(a[i]<x){
			if(num){
				i+=ll-1;
				num--;
			}
			else{
				return 0;
			}		
		}
	}
	return 1;
}
signed main() {
	ios::sync_with_stdio(0);
	cin.tie(0);
	cin>>n>>k>>ll;
	for(int i=1;i<=n;i++){
		cin>>a[i];
	}
	int l=1,r=1e9;
	while(l<=r){
		int mid=(l+r)/2;
		if(check(mid)){
			l=mid+1;
		}
		else{
			r=mid-1;
		}
	}
	cout<<l-1<<'\n';
	return 0;
}

```
### java

import java.io.*;
import java.util.*;

public class Main {
    static long[] a;
    static int n, k, ll;
    
    static boolean check(int x){
        int num = k;
        int i = 1;
        while(i <= n){
            if(a[i] < x){
                if(num > 0){
                    num--;
                    i += ll;
                }
                else{
                    return false;
                }
            }
            else{
                i +=1;
            }
        }
        return true;
    }
    
    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        ll = Integer.parseInt(st.nextToken());
        a = new long[n +1];
        st = new StringTokenizer(br.readLine());
        for(int i=1; i <=n; i++){
            a[i] = Long.parseLong(st.nextToken());
        }
        int l =1;
        int r = 1000000000;
        while(l <= r){
            int mid = l + (r - l) /2;
            if(check(mid)){
                l = mid +1;
            }
            else{
                r = mid -1;
            }
        }
        System.out.println(l -1);
    }
}

def main():
    import sys
    n, k, ll = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    a = [0] + a
    def check(x):
        num = k
        i = 1
        while i <= n:
            if a[i] < x:
                if num > 0:
                    num -= 1
                    i += ll
                else:
                    return False
            else:
                i += 1
        return True
    l = 1
    r = 10**9
    while l <= r:
        mid = (l + r) // 2
        if check(mid):
            l = mid + 1
        else:
            r = mid - 1
    print(l - 1)

if __name__ == "__main__":
    main()
