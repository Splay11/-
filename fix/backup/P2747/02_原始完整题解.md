## 思路（修改后的版本）

对于合法出入栈序列，如果某个相邻位置被交换，我们可以根据对应的相反数去判断每个相邻位置是否合法。

## 算法步骤
1. **字典 `mp` 的构建**：  
   首先通过字典 `mp` 将每个数值的索引保存下来，索引值是从0开始的（后面需要转换为1-indexed）。这是用来帮助快速查找对应出栈数的索引。

2. **四种情况的判断**：  
   - **正数相邻**：如果有两个正数相邻，并且其对应的负数出栈位置不符合顺序，即后入栈的正数应该先出栈，那么就输出它们的交换位置。
   - **负数相邻**：如果有两个负数相邻，同样也要检查对应的正数出栈位置是否符合顺序。
   - **负数后跟正数**：当负数后面跟正数时，应该检查是否形成了匹配的入栈出栈对。如果匹配，则不需要交换。
   - **正数后跟负数**：如果正数后面跟负数，必须验证它们是否是合法的出栈配对，如果不是且负数对应的数字在之前入栈，则说明需要交换它们。

## 代码实现

### Python 代码

```python
n = int(input())  # 输入序列长度
ls = list(map(int, input().split()))  # 输入序列

mp = {}  # 用于记录每个元素的位置
for i in range(n):
    mp[ls[i]] = i  # 记录元素和其索引的对应关系

# 遍历序列，检查出入栈是否不合法
for i in range(n - 1):
    if ls[i] > 0 and ls[i + 1] > 0:
        # 如果相邻两个元素都是入栈操作，检查它们对应的出栈顺序
        if mp[-ls[i]] < mp[-ls[i + 1]]:
            print(i + 1, i + 2)  # 输出交换的两个位置
            break
    elif ls[i] < 0 and ls[i + 1] < 0:
        # 如果相邻两个元素都是出栈操作，检查它们对应的入栈顺序
        if mp[-ls[i]] < mp[-ls[i + 1]]:
            print(i + 1, i + 2)  # 输出交换的两个位置
            break
    elif ls[i] < 0 and ls[i + 1] > 0:
        # 如果第一个元素是出栈，第二个元素是入栈，检查是否符合入栈出栈顺序
        if -ls[i] == ls[i + 1]:
            print(i + 1, i + 2)  # 输出交换的两个位置
            break
    elif ls[i] > 0 and ls[i + 1] < 0:
        # 如果第一个元素是入栈，第二个元素是出栈，检查是否符合入栈出栈顺序
        if ls[i] != -ls[i + 1] and mp[-ls[i + 1]] < i:
            print(i + 1, i + 2)  # 输出交换的两个位置
            break
```

### Java 代码

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();  // 输入序列长度
        int[] ls = new int[n];  // 序列数组
        Map<Integer, Integer> mp = new HashMap<>();  // 存储每个元素的位置
        
        // 输入序列
        for (int i = 0; i < n; i++) {
            ls[i] = sc.nextInt();
            mp.put(ls[i], i);
        }
        
        // 遍历序列，检查不合法的地方
        for (int i = 0; i < n - 1; i++) {
            if (ls[i] > 0 && ls[i + 1] > 0) {
                if (mp.get(-ls[i]) < mp.get(-ls[i + 1])) {
                    System.out.println((i + 1) + " " + (i + 2));  // 输出交换的两个位置
                    return;
                }
            } else if (ls[i] < 0 && ls[i + 1] < 0) {
                if (mp.get(-ls[i]) < mp.get(-ls[i + 1])) {
                    System.out.println((i + 1) + " " + (i + 2));  // 输出交换的两个位置
                    return;
                }
            } else if (ls[i] < 0 && ls[i + 1] > 0) {
                if (-ls[i] == ls[i + 1]) {
                    System.out.println((i + 1) + " " + (i + 2));  // 输出交换的两个位置
                    return;
                }
            } else if (ls[i] > 0 && ls[i + 1] < 0) {
                if (ls[i] != -ls[i + 1] && mp.get(-ls[i + 1]) < i) {
                    System.out.println((i + 1) + " " + (i + 2));  // 输出交换的两个位置
                    return;
                }
            }
        }
    }
}
```

### C++ 代码

```cpp
#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> ls(n);
    unordered_map<int, int> mp;

    // 输入序列并记录位置
    for (int i = 0; i < n; i++) {
        cin >> ls[i];
        mp[ls[i]] = i;
    }

    // 遍历序列，检查出入栈是否不合法
    for (int i = 0; i < n - 1; i++) {
        if (ls[i] > 0 && ls[i + 1] > 0) {
            if (mp[-ls[i]] < mp[-ls[i + 1]]) {
                cout << i + 1 << " " << i + 2 << endl;  // 输出交换的两个位置
                return 0;
            }
        } else if (ls[i] < 0 && ls[i + 1] < 0) {
            if (mp[-ls[i]] < mp[-ls[i + 1]]) {
                cout << i + 1 << " " << i + 2 << endl;  // 输出交换的两个位置
                return 0;
            }
        } else if (ls[i] < 0 && ls[i + 1] > 0) {
            if (-ls[i] == ls[i + 1]) {
                cout << i + 1 << " " << i + 2 << endl;  // 输出交换的两个位置
                return 0;
            }
        } else if (ls[i] > 0 && ls[i + 1] < 0) {
            if (ls[i] != -ls[i + 1] && mp[-ls[i + 1]] < i) {
                cout << i + 1 << " " << i + 2 << endl;  // 输出交换的两个位置
                return 0;
            }
        }
    }

    return 0;
}
```
