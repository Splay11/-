## 解题思路

设按排列 $b$ 取出的顺序对应的数列为

$$
s_i=a_{b_i}
$$

那么第一种操作“选择一个整数 $x$，获得权值 $\sum_{i=1}^{x} a_{b_i}$”，本质上就是：

* 在数列 $s$ 中选择一个前缀
* 获得这个前缀和

所以如果只执行第一种操作，答案就是数列 $s$ 的最大前缀和。

现在考虑第二种操作。

第二种操作“选择一个整数 $y$，将 $a_{c_1},a_{c_2},\dots,a_{c_y}$ 修改为 $0$”，等价于：

* 按照排列 $c$ 的顺序，依次把某些位置上的值删掉（变成 $0$）

如果第二种操作在第一种操作之前执行，那么之后再按 $b$ 取前缀时，取到的其实是一个“部分位置被改成 $0$ 的新数列”。

因此整个问题可以转化为：

* 先把数列按 $b$ 映射成 $s$
* 再按 $c$ 的顺序，依次把对应元素在 $s$ 中的位置改成 $0$
* 每改一次，都求一次当前数列的最大前缀和
* 所有过程中取最大值，同时和 $0$ 比较（因为两种操作都可以不做）



进一步思考如何高效维护。

### 核心转化

设

$$
s_i = a_{b_i}
$$

定义前缀和

$$
pre_i=\sum_{j=1}^{i} s_j
$$

则当前数列的最大前缀和就是

$$
\max_{1\le i\le n} pre_i
$$

假设现在把 $s_p$ 改成 $0$。

若原来 $s_p$ 的值为 $v$，那么从位置 $p$ 开始的所有前缀和都会减少 $v$，即：

* 若 $v>0$，后面的前缀和都会变小
* 若 $v<0$，后面的前缀和都会变大

也就是说，这次修改对前缀和数组的影响是：

* 对区间 $[p,n]$ 整体加上 $-v$

所以问题变成了：

* 初始有一个前缀和数组 $pre$
* 每次进行一次“后缀区间加”
* 每次查询整个数组的最大值

这正是线段树可以高效解决的问题。



### 实现方法

#### 1. 构造按 $b$ 顺序的数列

先令

$$
s_i=a_{b_i}
$$

并记录每个原下标在 $b$ 中出现的位置：

$$
posB[x] = x \text{ 在排列 } b \text{ 中的位置}
$$

这样当我们要把 $a_{c_k}$ 变成 $0$ 时，就能立刻知道它对应的是 $s$ 中哪个位置。



#### 2. 建立前缀和数组

构造

$$
pre_i = pre_{i-1}+s_i
$$

线段树维护整个 $pre$ 数组区间最大值，并支持区间加懒标记。



#### 3. 按 $c$ 的顺序依次删除

设当前要删除的是下标 $idx=c_k$。

它在 $s$ 中的位置为

$$
p=posB[idx]
$$

原值为

$$
v=a_{idx}=s_p
$$

删除后相当于把 $s_p$ 从 $v$ 变成 $0$，因此对所有前缀和 $pre_p,pre_{p+1},\dots,pre_n$ 都要加上

$$
-v
$$

即在线段树上执行区间加：

$$
[p,n] \text{ 加 } -v
$$

然后查询当前线段树维护的最大值，就是当前状态下的最大前缀和。

枚举所有删除步数，取最大值即可。



#### 4. 为什么这样一定正确

因为题目中两种操作都至多执行一次，顺序任意，所以只有以下几种情况：

1. 两种都不做，权值为 $0$
2. 只做第一种操作，等价于初始数列的最大前缀和
3. 先做第二种，再做第一种，等价于删掉 $c$ 的前若干项后，求新数列的最大前缀和
4. 先做第一种，再做第二种，第二种不会影响已经得到的权值，等价于只做第一种

所以只需要枚举“删了前 $0,1,2,\dots,n$ 个元素”这 $n+1$ 种状态，并求每种状态下的最大前缀和即可，答案取这些值与 $0$ 的最大值。



## 复杂度分析

设单组数据长度为 $n$。

* 构造 $s$、$posB$、$pre$ 的时间复杂度为 $O(n)$
* 线段树建树复杂度为 $O(n)$
* 每次删除进行一次区间加、一次查询，复杂度为 $O(\log n)$
* 共删除 $n$ 次，总复杂度为 $O(n\log n)$

因此单组数据总时间复杂度为：

$$
O(n\log n)
$$

空间复杂度：

* 数组 $a,b,c,s,posB,pre$ 需要 $O(n)$
* 线段树需要 $O(n)$

总空间复杂度为：

$$
O(n)
$$

题目保证所有测试数据的 $n$ 之和不超过 $2\times 10^5$，该复杂度完全可行。

```python
import sys


# 线段树：维护区间最大值，支持区间加
class SegmentTree:
    def __init__(self, arr):
        # arr 是 1 下标数组，这里只使用 arr[1..n]
        self.n = len(arr) - 1
        self.mx = [0] * (self.n * 4)
        self.lazy = [0] * (self.n * 4)
        self._build(1, 1, self.n, arr)

    # 建树
    def _build(self, node, l, r, arr):
        if l == r:
            self.mx[node] = arr[l]
            return
        mid = (l + r) // 2
        self._build(node * 2, l, mid, arr)
        self._build(node * 2 + 1, mid + 1, r, arr)
        self.mx[node] = max(self.mx[node * 2], self.mx[node * 2 + 1])

    # 下传懒标记
    def _push_down(self, node):
        if self.lazy[node] != 0:
            tag = self.lazy[node]
            self.mx[node * 2] += tag
            self.mx[node * 2 + 1] += tag
            self.lazy[node * 2] += tag
            self.lazy[node * 2 + 1] += tag
            self.lazy[node] = 0

    # 区间加
    def range_add(self, node, l, r, ql, qr, val):
        if ql <= l and r <= qr:
            self.mx[node] += val
            self.lazy[node] += val
            return
        self._push_down(node)
        mid = (l + r) // 2
        if ql <= mid:
            self.range_add(node * 2, l, mid, ql, qr, val)
        if qr > mid:
            self.range_add(node * 2 + 1, mid + 1, r, ql, qr, val)
        self.mx[node] = max(self.mx[node * 2], self.mx[node * 2 + 1])

    # 查询整个数组最大值
    def query_max(self):
        return self.mx[1]


# 计算单组测试数据答案
def solve_case(n, a, b, c):
    # s[i] = a[b[i]]
    s = [0] * (n + 1)

    # posB[x] 表示原下标 x 在排列 b 中的位置
    posB = [0] * (n + 1)

    for i in range(1, n + 1):
        s[i] = a[b[i]]
        posB[b[i]] = i

    # 前缀和数组
    pre = [0] * (n + 1)
    for i in range(1, n + 1):
        pre[i] = pre[i - 1] + s[i]

    # 线段树维护前缀和数组 pre[1..n] 的最大值
    seg = SegmentTree(pre)

    # 可以什么都不做，所以初始答案至少为 0
    ans = max(0, seg.query_max())

    # 按 c 的顺序依次删除
    for i in range(1, n + 1):
        idx = c[i]
        p = posB[idx]      # 该元素在 s 中的位置
        v = a[idx]         # 原值

        # 把 s[p] 从 v 变为 0
        # 等价于前缀和数组 pre[p..n] 全部加上 -v
        seg.range_add(1, 1, n, p, n, -v)

        # 当前状态下的最大前缀和
        ans = max(ans, seg.query_max())

    return ans


def main():
    input = sys.stdin.readline
    t = int(input().strip())
    res = []

    for _ in range(t):
        n = int(input().strip())

        arr = list(map(int, input().split()))
        a = [0] + arr

        arr = list(map(int, input().split()))
        b = [0] + arr

        arr = list(map(int, input().split()))
        c = [0] + arr

        res.append(str(solve_case(n, a, b, c)))

    print("\n".join(res))


if __name__ == "__main__":
    main()
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {

    // 线段树：维护区间最大值，支持区间加
    static class SegmentTree {
        long[] max;
        long[] lazy;
        int n;

        SegmentTree(long[] arr, int n) {
            this.n = n;
            max = new long[n * 4];
            lazy = new long[n * 4];
            build(1, 1, n, arr);
        }

        // 建树
        void build(int node, int l, int r, long[] arr) {
            if (l == r) {
                max[node] = arr[l];
                return;
            }
            int mid = (l + r) >> 1;
            build(node << 1, l, mid, arr);
            build(node << 1 | 1, mid + 1, r, arr);
            max[node] = Math.max(max[node << 1], max[node << 1 | 1]);
        }

        // 下传懒标记
        void pushDown(int node) {
            if (lazy[node] != 0) {
                long tag = lazy[node];
                max[node << 1] += tag;
                max[node << 1 | 1] += tag;
                lazy[node << 1] += tag;
                lazy[node << 1 | 1] += tag;
                lazy[node] = 0;
            }
        }

        // 区间加
        void rangeAdd(int node, int l, int r, int ql, int qr, long val) {
            if (ql <= l && r <= qr) {
                max[node] += val;
                lazy[node] += val;
                return;
            }
            pushDown(node);
            int mid = (l + r) >> 1;
            if (ql <= mid) {
                rangeAdd(node << 1, l, mid, ql, qr, val);
            }
            if (qr > mid) {
                rangeAdd(node << 1 | 1, mid + 1, r, ql, qr, val);
            }
            max[node] = Math.max(max[node << 1], max[node << 1 | 1]);
        }

        // 查询整个数组最大值
        long queryMax() {
            return max[1];
        }
    }

    // 计算单组测试数据答案
    static long solveCase(int n, long[] a, int[] b, int[] c) {
        // s[i] = a[b[i]]
        long[] s = new long[n + 1];

        // posB[x] 表示原下标 x 在排列 b 中的位置
        int[] posB = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            s[i] = a[b[i]];
            posB[b[i]] = i;
        }

        // 前缀和数组
        long[] pre = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            pre[i] = pre[i - 1] + s[i];
        }

        // 线段树维护前缀和数组 pre[1..n]
        SegmentTree seg = new SegmentTree(pre, n);

        // 可以什么都不做，所以答案至少为 0
        long ans = Math.max(0L, seg.queryMax());

        // 按 c 的顺序依次删除
        for (int i = 1; i <= n; i++) {
            int idx = c[i];
            int p = posB[idx];   // 该元素在 s 中的位置
            long v = a[idx];     // 原值

            // 把 s[p] 从 v 变为 0
            // 等价于前缀和数组 pre[p..n] 全部加上 -v
            seg.rangeAdd(1, 1, n, p, n, -v);

            // 当前状态下的最大前缀和
            ans = Math.max(ans, seg.queryMax());
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();

        int T = fs.nextInt();
        StringBuilder sb = new StringBuilder();

        while (T-- > 0) {
            int n = fs.nextInt();

            long[] a = new long[n + 1];
            int[] b = new int[n + 1];
            int[] c = new int[n + 1];

            for (int i = 1; i <= n; i++) {
                a[i] = fs.nextLong();
            }
            for (int i = 1; i <= n; i++) {
                b[i] = fs.nextInt();
            }
            for (int i = 1; i <= n; i++) {
                c[i] = fs.nextInt();
            }

            sb.append(solveCase(n, a, b, c)).append('\n');
        }

        System.out.print(sb.toString());
    }

    // 朴素输入工具，适合本题数据范围
    static class FastScanner {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;

        String next() throws IOException {
            while (st == null || !st.hasMoreElements()) {
                st = new StringTokenizer(br.readLine());
            }
            return st.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        long nextLong() throws IOException {
            return Long.parseLong(next());
        }
    }
}
```

```cpp
public:
    vector<long long> mx;
    vector<long long> lazy;
    int n;

    SegmentTree(const vector<long long>& arr, int n) {
        this->n = n;
        mx.assign(n * 4 + 5, 0);
        lazy.assign(n * 4 + 5, 0);
        build(1, 1, n, arr);
    }

            mx[node] = arr[l];
            return;

        build(node << 1, l, mid, arr);
        build(node << 1 | 1, mid + 1, r, arr);
```
