## 解题思路

记 $\textit{modSet}$ 的最小值为 $\textit{mn}$。

链式折减性质：

- 若当前 $\textit{cur} < \textit{order}_i$，则 $\textit{cur} \bmod \textit{order}_i = \textit{cur}$；
- 若 $\textit{cur} \ge \textit{order}_i$，则结果严格变小。

**情形一**：$\textit{seed} < \textit{mn}$。所有模数均大于当前值，折减无效，答案为 $\textit{seed}$。

**情形二**：$\textit{seed} \ge \textit{mn}$。最终 $\textit{cur}_{\textit{cnt}}$ 必满足 $\textit{cur}_{\textit{cnt}} < \textit{mn}$（否则存在某步会把值压到 $< \textit{mn}$）。

在值域 $[0, 2000]$ 内 BFS：

1. 以所有 $\textit{seed} \bmod \textit{order}_i$（$\textit{order}_i \le \textit{seed}$）为初始可达状态；
2. 从状态 $v$ 可转移到 $v \bmod \textit{order}_i$（当 $\textit{order}_i \le v$）；
3. 在可达状态中取小于 $\textit{mn}$ 的最大值。

## 复杂度分析

设模数上界为 $M$（$M \le 2000$），单组 $\textit{cnt} \le 2000$。

- 时间复杂度：$O(M \cdot \textit{cnt})$
- 空间复杂度：$O(M)$

所有测试的 $\textit{cnt}$ 之和不超过 $5000$，总复杂度可接受。

## 代码实现

### Python

```python
import sys
from collections import deque

def solve_one(cnt, seed, mod_set):
    mn = min(mod_set)
    if seed < mn:
        return seed
    mods = list(set(mod_set))
    max_mod = max(mod_set)
    vis = [False] * (max_mod + 1)
    q = deque()
    for a in mods:
        if a <= seed:
            r = seed % a
            if not vis[r]:
                vis[r] = True
                q.append(r)
    while q:
        cur = q.popleft()
        for a in mods:
            if a <= cur:
                nxt = cur % a
                if not vis[nxt]:
                    vis[nxt] = True
                    q.append(nxt)
    ans = 0
    for i in range(mn):
        if vis[i]:
            ans = i
    return ans

data = list(map(int, sys.stdin.buffer.read().split()))
idx = 0
T = data[idx]; idx += 1
res = []
for _ in range(T):
    cnt, seed = data[idx], data[idx+1]; idx += 2
    mod_set = data[idx:idx+cnt]; idx += cnt
    res.append(str(solve_one(cnt, seed, mod_set)))
print("\n".join(res))
```

### Java

```java
import java.util.*;

public class Main {
    static long solveOne(int cnt, long seed, int[] modSet) {
        int mn = modSet[0], maxMod = modSet[0];
        for (int v : modSet) { mn = Math.min(mn, v); maxMod = Math.max(maxMod, v); }
        if (seed < mn) return seed;
        boolean[] exist = new boolean[maxMod + 1];
        ArrayList<Integer> mods = new ArrayList<>();
        for (int v : modSet) if (!exist[v]) { exist[v] = true; mods.add(v); }
        boolean[] vis = new boolean[maxMod + 1];
        Queue<Integer> q = new LinkedList<>();
        for (int a : mods) {
            if (a <= seed) {
                int r = (int)(seed % a);
                if (!vis[r]) { vis[r] = true; q.offer(r); }
            }
        }
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int a : mods) {
                if (a <= cur) {
                    int nxt = cur % a;
                    if (!vis[nxt]) { vis[nxt] = true; q.offer(nxt); }
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < mn; i++) if (vis[i]) ans = i;
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        StringBuilder sb = new StringBuilder();
        for (int tc = 0; tc < T; tc++) {
            int cnt = sc.nextInt();
            long seed = sc.nextLong();
            int[] modSet = new int[cnt];
            for (int i = 0; i < cnt; i++) modSet[i] = sc.nextInt();
            sb.append(solveOne(cnt, seed, modSet)).append('\n');
        }
        System.out.print(sb);
        sc.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

long long solveOne(int cnt, long long seed, vector<int>& modSet) {
    int mn = *min_element(modSet.begin(), modSet.end());
    int maxMod = *max_element(modSet.begin(), modSet.end());
    if (seed < mn) return seed;
    vector<int> mods;
    vector<int> exist(maxMod + 1, 0);
    for (int v : modSet) if (!exist[v]) { exist[v] = 1; mods.push_back(v); }
    vector<int> vis(maxMod + 1, 0);
    queue<int> q;
    for (int a : mods) {
        if (a <= seed) {
            int r = (int)(seed % a);
            if (!vis[r]) { vis[r] = 1; q.push(r); }
        }
    }
    while (!q.empty()) {
        int cur = q.front(); q.pop();
        for (int a : mods) {
            if (a <= cur) {
                int nxt = cur % a;
                if (!vis[nxt]) { vis[nxt] = 1; q.push(nxt); }
            }
        }
    }
    int ans = 0;
    for (int i = 0; i < mn; i++) if (vis[i]) ans = i;
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T; cin >> T;
    while (T--) {
        int cnt; long long seed;
        cin >> cnt >> seed;
        vector<int> modSet(cnt);
        for (int i = 0; i < cnt; i++) cin >> modSet[i];
        cout << solveOne(cnt, seed, modSet) << '\n';
    }
    return 0;
}
```
