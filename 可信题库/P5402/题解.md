## 解题思路

本题是在大字符矩阵里找小矩形是否作为**子矩阵**出现，要求返回行优先的第一个左上角。数据到 $300 \times 300$，逐格对比子块会超时，用 **二维字符串哈希 + 二维前缀** 做到 $O(1)$ 询问。

给矩阵 $g$ 定义：

$$
h[i][j]=\sum_{x=0}^{i-1}\sum_{y=0}^{j-1} g[x][y]\cdot P^{i-1-x}\cdot Q^{j-1-y}
$$

递推（自然溢出或 $2^{64}$ 截断均可）：

$$
h[i][j]=h[i-1][j]\cdot P + h[i][j-1]\cdot Q - h[i-1][j-1]\cdot P\cdot Q + g[i-1][j-1]
$$

左上角 $(r,c)$、高 $a$ 宽 $b$ 的子块哈希为：

$$
h[r+a][c+b] - h[r][c+b]\cdot P^{a} - h[r+a][c]\cdot Q^{b} + h[r][c]\cdot P^{a}Q^{b}
$$

先算出模具整块哈希，再按行、列扫描烤盘每个合法左上角，第一个哈希相等的位置就是答案。字母映射成 `ord - 64`，避免出现 $0$。

常见假解：当成「四向走路径拼单词」、允许旋转模具、多个匹配时没取最左上、暴力四重循环在大数据超时。

## 复杂度分析

- 时间：$O(mn)$ 建表并扫描所有窗口。
- 空间：$O(mn)$ 前缀哈希。

## 代码实现

### Python

```python
from typing import List, Tuple

MASK = (1 << 64) - 1
P = 131
Q = 13331


def _build(g: List[List[str]]) -> Tuple[List[List[int]], int, int]:
    m, n = len(g), len(g[0])
    h = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m):
        for j in range(n):
            v = ord(g[i][j]) - 64
            # 二维前缀哈希递推
            h[i + 1][j + 1] = (h[i][j + 1] * P + h[i + 1][j] * Q - h[i][j] * P * Q + v) & MASK
    return h, m, n


def _pw(k: int, base: int) -> List[int]:
    a = [1] * (k + 1)
    for i in range(1, k + 1):
        a[i] = (a[i - 1] * base) & MASK
    return a


def _rect(h, r, c, a, b, pP, pQ) -> int:
    v = h[r + a][c + b]
    v = (v - h[r][c + b] * pP[a]) & MASK
    v = (v - h[r + a][c] * pQ[b]) & MASK
    v = (v + h[r][c] * pP[a] * pQ[b]) & MASK
    return v


class Solution:
    def findStampPos(self, tray: List[List[str]], stamp: List[List[str]]) -> List[int]:
        ht, m, n = _build(tray)
        hs, a, b = _build(stamp)
        pP, pQ = _pw(m, P), _pw(n, Q)
        need = _rect(hs, 0, 0, a, b, pP, pQ)
        for i in range(m - a + 1):
            for j in range(n - b + 1):
                if _rect(ht, i, j, a, b, pP, pQ) == need:
                    return [i, j]
        return [-1, -1]
```

### Java

```java
class Solution {
    static final long P = 131L, Q = 13331L;

    public int[] findStampPos(String[][] tray, String[][] stamp) {
        int m = tray.length, n = tray[0].length;
        int a = stamp.length, b = stamp[0].length;
        long[][] ht = build(tray), hs = build(stamp);
        long[] pP = pow(m, P), pQ = pow(n, Q);
        long need = rect(hs, 0, 0, a, b, pP, pQ);
        for (int i = 0; i <= m - a; i++) {
            for (int j = 0; j <= n - b; j++) {
                if (rect(ht, i, j, a, b, pP, pQ) == need) return new int[] {i, j};
            }
        }
        return new int[] {-1, -1};
    }

    private long[][] build(String[][] g) {
        int m = g.length, n = g[0].length;
        long[][] h = new long[m + 1][n + 1];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                long v = g[i][j].charAt(0) - 64;
                h[i + 1][j + 1] = h[i][j + 1] * P + h[i + 1][j] * Q - h[i][j] * P * Q + v;
            }
        }
        return h;
    }

    private long[] pow(int k, long base) {
        long[] a = new long[k + 1];
        a[0] = 1;
        for (int i = 1; i <= k; i++) a[i] = a[i - 1] * base;
        return a;
    }

    private long rect(long[][] h, int r, int c, int a, int b, long[] pP, long[] pQ) {
        long v = h[r + a][c + b];
        v -= h[r][c + b] * pP[a];
        v -= h[r + a][c] * pQ[b];
        v += h[r][c] * pP[a] * pQ[b];
        return v;
    }
}
```

### JavaScript

```javascript
class Solution {
    findStampPos(tray, stamp) {
        const MASK = 1n << 64n;
        const P = 131n, Q = 13331n;
        const toU = (x) => ((x % MASK) + MASK) % MASK;

        const build = (g) => {
            const m = g.length, n = g[0].length;
            const h = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0n));
            for (let i = 0; i < m; i++) {
                for (let j = 0; j < n; j++) {
                    const v = BigInt(g[i][j].charCodeAt(0) - 64);
                    h[i + 1][j + 1] = toU(h[i][j + 1] * P + h[i + 1][j] * Q - h[i][j] * P * Q + v);
                }
            }
            return [h, m, n];
        };

        const pw = (k, base) => {
            const a = Array(k + 1).fill(1n);
            for (let i = 1; i <= k; i++) a[i] = toU(a[i - 1] * base);
            return a;
        };

        const rect = (h, r, c, a, b, pP, pQ) => {
            let v = h[r + a][c + b];
            v = toU(v - h[r][c + b] * pP[a]);
            v = toU(v - h[r + a][c] * pQ[b]);
            v = toU(v + h[r][c] * pP[a] * pQ[b]);
            return v;
        };

        const [ht, m, n] = build(tray);
        const [hs, a, b] = build(stamp);
        const pP = pw(m, P), pQ = pw(n, Q);
        const need = rect(hs, 0, 0, a, b, pP, pQ);
        for (let i = 0; i <= m - a; i++) {
            for (let j = 0; j <= n - b; j++) {
                if (rect(ht, i, j, a, b, pP, pQ) === need) return [i, j];
            }
        }
        return [-1, -1];
    }
}
```

### C

```c
#define MAXN 305
static unsigned long long HT[MAXN][MAXN], HS[MAXN][MAXN];
static unsigned long long PP[MAXN], PQ[MAXN];
static const unsigned long long P = 131, Q = 13331;

static void build(char** g, int m, int n, unsigned long long h[MAXN][MAXN]) {
    for (int i = 0; i <= m; i++)
        for (int j = 0; j <= n; j++) h[i][j] = 0;
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            unsigned long long v = (unsigned char)g[i][j] - 64;
            h[i + 1][j + 1] = h[i][j + 1] * P + h[i + 1][j] * Q - h[i][j] * P * Q + v;
        }
    }
}

static unsigned long long rect(unsigned long long h[MAXN][MAXN], int r, int c, int a, int b) {
    unsigned long long v = h[r + a][c + b];
    v -= h[r][c + b] * PP[a];
    v -= h[r + a][c] * PQ[b];
    v += h[r][c] * PP[a] * PQ[b];
    return v;
}

void findStampPos(char** tray, int traySize, int trayColSize,
                  char** stamp, int stampSize, int stampColSize, int* out) {
    int m = traySize, n = trayColSize, a = stampSize, b = stampColSize;
    build(tray, m, n, HT);
    build(stamp, a, b, HS);
    PP[0] = PQ[0] = 1;
    for (int i = 1; i <= m; i++) PP[i] = PP[i - 1] * P;  // 行方向幂
    for (int j = 1; j <= n; j++) PQ[j] = PQ[j - 1] * Q;
    unsigned long long need = rect(HS, 0, 0, a, b);
    for (int i = 0; i <= m - a; i++) {
        for (int j = 0; j <= n - b; j++) {
            if (rect(HT, i, j, a, b) == need) {
                out[0] = i;
                out[1] = j;
                return;
            }
        }
    }
    out[0] = out[1] = -1;
}
```
