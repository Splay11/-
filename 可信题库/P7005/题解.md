## 解题思路

题目要求：对所有非空连续子数组，把各自的**最大值**加总，再对 $10^9+7$ 取模。

直接枚举所有区间是 $O(n^2)$，在 $n=10^5$ 时不可行。换贡献视角：对每个下标 $i$，统计有多少个子数组以 $\texttt{loads}[i]$ 为最大值，再累加 $\texttt{loads}[i]\times\text{次数}$。

用**单调栈**求左右跨度：

- $\textit{left}[i]$：从 $i$ 向左延伸，直到遇到**严格大于** $\texttt{loads}[i]$ 的位置（不含该位置），跨度含自身。
- $\textit{right}[i]$：从 $i$ 向右延伸，直到遇到**大于等于** $\texttt{loads}[i]$ 的位置（不含该位置），跨度含自身。

于是以 $i$ 为峰值的子数组个数为 $\textit{left}[i]\times\textit{right}[i]$。左右一侧用严格、一侧用非严格，是为了处理相等元素时每个区间只归属一个峰值，避免重复或漏计。

## 复杂度分析

- 时间复杂度：$O(n)$，每个下标进栈出栈至多一次。
- 空间复杂度：$O(n)$，存放左右跨度与栈。

## 代码实现

### Python

```python
from typing import List

MOD = 10**9 + 7

class Solution:
    def sumPeakRisk(self, loads: List[int]) -> int:
        n = len(loads)
        left = [0] * n
        right = [0] * n
        st = []
        # 左侧最近严格更大：弹出 <= 当前值
        for i in range(n):
            while st and loads[st[-1]] <= loads[i]:
                st.pop()
            left[i] = i - (st[-1] if st else -1)
            st.append(i)
        st.clear()
        # 右侧最近 >=：弹出严格更小，相等则切开
        for i in range(n - 1, -1, -1):
            while st and loads[st[-1]] < loads[i]:
                st.pop()
            right[i] = (st[-1] if st else n) - i
            st.append(i)
        ans = 0
        for i in range(n):
            # 贡献 = 峰值 × 左跨度 × 右跨度
            ans = (ans + loads[i] * left[i] * right[i]) % MOD
        return ans
```

### Java

```java
public class Solution {
    private static final int MOD = 1_000_000_007;

    public int sumPeakRisk(int[] loads) {
        int n = loads.length;
        int[] left = new int[n];
        int[] right = new int[n];
        int[] st = new int[n];
        int top = 0;
        // 左侧严格更大
        for (int i = 0; i < n; i++) {
            while (top > 0 && loads[st[top - 1]] <= loads[i]) top--;
            left[i] = i - (top > 0 ? st[top - 1] : -1);
            st[top++] = i;
        }
        top = 0;
        // 右侧大于等于，避免相等峰值重复计
        for (int i = n - 1; i >= 0; i--) {
            while (top > 0 && loads[st[top - 1]] < loads[i]) top--;
            right[i] = (top > 0 ? st[top - 1] : n) - i;
            st[top++] = i;
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            // 用 long 防止乘法溢出
            ans = (ans + (long) loads[i] * left[i] % MOD * right[i]) % MOD;
        }
        return (int) ans;
    }
}
```

### C++

```cpp
#include <vector>
using namespace std;

class Solution {
 public:
  int sumPeakRisk(vector<int>& loads) {
    const int MOD = 1e9 + 7;
    int n = (int)loads.size();
    vector<int> left(n), right(n), st;
    // 左侧：直到 prev > loads[i]
    for (int i = 0; i < n; i++) {
      while (!st.empty() && loads[st.back()] <= loads[i]) st.pop_back();
      left[i] = i - (st.empty() ? -1 : st.back());
      st.push_back(i);
    }
    st.clear();
    // 右侧：直到 next >= loads[i]
    for (int i = n - 1; i >= 0; i--) {
      while (!st.empty() && loads[st.back()] < loads[i]) st.pop_back();
      right[i] = (st.empty() ? n : st.back()) - i;
      st.push_back(i);
    }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
      // 累计每个采样点作为峰值的贡献
      ans = (ans + 1LL * loads[i] * left[i] % MOD * right[i]) % MOD;
    }
    return (int)ans;
  }
};
```

```c
#include <stdlib.h>

int sumPeakRisk(int* loads, int loadsSize) {
    const int MOD = 1000000007;
    int n = loadsSize;
    int* left = (int*)malloc(n * sizeof(int));
    int* right = (int*)malloc(n * sizeof(int));
    int* st = (int*)malloc(n * sizeof(int));
    int top = 0;
    // 左侧严格更大跨度
    for (int i = 0; i < n; i++) {
        while (top > 0 && loads[st[top - 1]] <= loads[i]) top--;
        left[i] = i - (top > 0 ? st[top - 1] : -1);
        st[top++] = i;
    }
    top = 0;
    // 右侧 >= 跨度，相等只计一次
    for (int i = n - 1; i >= 0; i--) {
        while (top > 0 && loads[st[top - 1]] < loads[i]) top--;
        right[i] = (top > 0 ? st[top - 1] : n) - i;
        st[top++] = i;
    }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        // 贡献取模累加
        ans = (ans + (long long)loads[i] * left[i] % MOD * right[i]) % MOD;
    }
    free(left);
    free(right);
    free(st);
    return (int)ans;
}
```
