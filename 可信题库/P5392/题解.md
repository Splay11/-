## 解题思路

本题是哈希表上的**接口模拟**。用一张映射 `fileId -> ownerId` 表示「当前谁锁着这份文件」即可。

- `lock`：键已存在则失败；否则写入并返回成功。注意：**同一 `ownerId` 再锁同一文件也要失败**（不可重入）。
- `unlock`：仅当键存在且值等于 `ownerId` 时删除并成功；未锁或持有者不匹配都失败且不改表。
- `holder`：键不存在返回 `-1`，否则返回持有者。
- `lockedCount`：映射大小。

不需要超时、容量或级联；状态只随这四个方法变化。

常见假解：

- 允许同一持有者重复 `lock` 成功（可重入）。
- `unlock` 不校验持有者，谁来都能解开。
- `lock` 已存在时覆盖成新的 `ownerId`。
- 未锁时 `holder` 返回 `0` 而不是 `-1`。

## 复杂度分析

- 时间：每次操作平均 $O(1)$（哈希表）；累计调用 $q$ 则总 $O(q)$。
- 空间：$O(q)$，最坏每个 `lock` 都新增一份文件。

## 代码实现

### Python

```python
class FileLockBoard:
    def __init__(self):
        self.own = {}  # fileId -> ownerId

    def lock(self, fileId: int, ownerId: int) -> bool:
        # 已锁则失败，包括同一持有者再次 lock
        if fileId in self.own:
            return False
        self.own[fileId] = ownerId
        return True

    def unlock(self, fileId: int, ownerId: int) -> bool:
        # 必须由当前持有者来解
        if self.own.get(fileId) != ownerId:
            return False
        del self.own[fileId]
        return True

    def holder(self, fileId: int) -> int:
        return self.own.get(fileId, -1)

    def lockedCount(self) -> int:
        return len(self.own)
```

### Java

```java
import java.util.HashMap;
import java.util.Map;

class FileLockBoard {
    private Map<Integer, Integer> own = new HashMap<>();

    public FileLockBoard() {}

    public boolean lock(int fileId, int ownerId) {
        // 已锁则失败，不可重入
        if (own.containsKey(fileId)) return false;
        own.put(fileId, ownerId);
        return true;
    }

    public boolean unlock(int fileId, int ownerId) {
        Integer h = own.get(fileId);
        if (h == null || h != ownerId) return false;
        own.remove(fileId);
        return true;
    }

    public int holder(int fileId) {
        Integer h = own.get(fileId);
        return h == null ? -1 : h;
    }

    public int lockedCount() {
        return own.size();
    }
}
```

### C++

```cpp
#include <unordered_map>
using namespace std;

class FileLockBoard {
    unordered_map<int, int> own;
public:
    FileLockBoard() {}

    bool lock(int fileId, int ownerId) {
        // 已锁则失败，不可重入
        if (own.count(fileId)) return false;
        own[fileId] = ownerId;
        return true;
    }

    bool unlock(int fileId, int ownerId) {
        auto it = own.find(fileId);
        if (it == own.end() || it->second != ownerId) return false;
        own.erase(it);
        return true;
    }

    int holder(int fileId) {
        auto it = own.find(fileId);
        return it == own.end() ? -1 : it->second;
    }

    int lockedCount() { return (int)own.size(); }
};
```

### JavaScript

```javascript
class FileLockBoard {
    constructor() {
        this.own = new Map();
    }

    lock(fileId, ownerId) {
        // 已锁则失败，不可重入
        if (this.own.has(fileId)) return false;
        this.own.set(fileId, ownerId);
        return true;
    }

    unlock(fileId, ownerId) {
        if (this.own.get(fileId) !== ownerId) return false;
        this.own.delete(fileId);
        return true;
    }

    holder(fileId) {
        return this.own.has(fileId) ? this.own.get(fileId) : -1;
    }

    lockedCount() {
        return this.own.size;
    }
}
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>

#define MAXN 10050

typedef struct {
    int fileId;
    int ownerId;
    int used;
} Slot;

typedef struct FileLockBoard {
    Slot a[MAXN];
} FileLockBoard;

static int findSlot(FileLockBoard* obj, int fileId) {
    for (int i = 0; i < MAXN; i++) {
        if (obj->a[i].used && obj->a[i].fileId == fileId) return i;
    }
    return -1;
}

FileLockBoard* fileLockBoardCreate(void) {
    FileLockBoard* obj = (FileLockBoard*)calloc(1, sizeof(FileLockBoard));
    return obj;
}

void fileLockBoardFree(FileLockBoard* obj) { free(obj); }

bool fileLockBoardLock(FileLockBoard* obj, int fileId, int ownerId) {
    if (findSlot(obj, fileId) >= 0) return false;  // 已锁
    for (int i = 0; i < MAXN; i++) {
        if (!obj->a[i].used) {
            obj->a[i].used = 1;
            obj->a[i].fileId = fileId;
            obj->a[i].ownerId = ownerId;
            return true;
        }
    }
    return false;
}

bool fileLockBoardUnlock(FileLockBoard* obj, int fileId, int ownerId) {
    int i = findSlot(obj, fileId);
    if (i < 0 || obj->a[i].ownerId != ownerId) return false;
    obj->a[i].used = 0;
    return true;
}

int fileLockBoardHolder(FileLockBoard* obj, int fileId) {
    int i = findSlot(obj, fileId);
    return i < 0 ? -1 : obj->a[i].ownerId;
}

int fileLockBoardLockedCount(FileLockBoard* obj) {
    int c = 0;
    for (int i = 0; i < MAXN; i++) if (obj->a[i].used) c++;
    return c;
}
```
