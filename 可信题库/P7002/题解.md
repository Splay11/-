## 解题思路

用**优先队列 + 懒删除**模拟打印调度：

- 堆中存 `(-priority, jobId)`：优先级大者先出；同优先级编号小者先出。
- 用集合 `active` 记录当前仍在队列中的任务编号。
- `submit`：若编号已在 `active` 返回 `false`，否则入堆并登记。
- `cancel`：仅从 `active` 删除（堆中残留条目在 `peek`/`pop` 时惰性清掉）。
- `peekJob` / `popJob`：先弹出堆顶中已取消的脏数据，再读/取真正的下一任务。

调用量 $\le 1000$，也可用线性扫描维护“当前最优任务”。

易错点：取消后仍弹出；重复 `submit` 未拒绝；同优先级未按编号比较；弹出后未移出 `active`。

## 复杂度分析

- 单次 `submit` / `pop`：$O(\log n)$（堆）；懒删除摊还良好。
- 空间：$O(n)$，$n$ 为历史提交次数上界。

## 代码实现

### Python

```python
import heapq
from typing import List, Set


class JobQueueSys:
    def __init__(self):
        self.heap: List[tuple] = []  # (-priority, jobId)
        self.active: Set[int] = set()

    def submit(self, jobId: int, priority: int) -> bool:
        if jobId in self.active:
            return False
        self.active.add(jobId)
        heapq.heappush(self.heap, (-priority, jobId))
        return True

    def cancel(self, jobId: int) -> bool:
        if jobId not in self.active:
            return False
        self.active.remove(jobId)
        return True

    def _clean(self) -> None:
        # 清掉已取消但仍留在堆中的脏项
        while self.heap and self.heap[0][1] not in self.active:
            heapq.heappop(self.heap)

    def peekJob(self) -> int:
        self._clean()
        if not self.heap:
            return -1
        return self.heap[0][1]

    def popJob(self) -> int:
        self._clean()
        if not self.heap:
            return -1
        _, job_id = heapq.heappop(self.heap)
        self.active.discard(job_id)
        return job_id
```

### Java

```java
import java.util.*;

class JobQueueSys {
    private PriorityQueue<int[]> heap; // {priority, jobId}，自定义比较
    private Set<Integer> active;

    public JobQueueSys() {
        // 优先级大优先；同优先级编号小优先
        heap = new PriorityQueue<>((a, b) -> {
            if (a[0] != b[0]) return Integer.compare(b[0], a[0]);
            return Integer.compare(a[1], b[1]);
        });
        active = new HashSet<>();
    }

    public boolean submit(int jobId, int priority) {
        if (active.contains(jobId)) return false;
        active.add(jobId);
        heap.offer(new int[]{priority, jobId});
        return true;
    }

    public boolean cancel(int jobId) {
        if (!active.contains(jobId)) return false;
        active.remove(jobId);
        return true;
    }

    private void clean() {
        while (!heap.isEmpty() && !active.contains(heap.peek()[1])) heap.poll();
    }

    public int peekJob() {
        clean();
        return heap.isEmpty() ? -1 : heap.peek()[1];
    }

    public int popJob() {
        clean();
        if (heap.isEmpty()) return -1;
        int id = heap.poll()[1];
        active.remove(id);
        return id;
    }
}
```

### C++

```cpp
#include <queue>
#include <unordered_set>
#include <vector>
using namespace std;

class JobQueueSys {
    // pair: {-priority, jobId}，小顶堆即高优先级优先
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> heap;
    unordered_set<int> active;

    void clean() {
        while (!heap.empty() && !active.count(heap.top().second)) heap.pop();
    }

public:
    JobQueueSys() {}

    bool submit(int jobId, int priority) {
        if (active.count(jobId)) return false;
        active.insert(jobId);
        heap.push({-priority, jobId});
        return true;
    }

    bool cancel(int jobId) {
        if (!active.count(jobId)) return false;
        active.erase(jobId);
        return true;
    }

    int peekJob() {
        clean();
        return heap.empty() ? -1 : heap.top().second;
    }

    int popJob() {
        clean();
        if (heap.empty()) return -1;
        int id = heap.top().second;
        heap.pop();
        active.erase(id);
        return id;
    }
};
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

typedef struct JobQueueSys JobQueueSys;

typedef struct {
    int jobId, priority;
    char used;
} Node;

struct JobQueueSys {
    Node* a;
    int n, cap;
};

JobQueueSys* jobQueueSysCreate(void) {
    JobQueueSys* obj = (JobQueueSys*)calloc(1, sizeof(JobQueueSys));
    obj->cap = 16;
    obj->a = (Node*)calloc((size_t)obj->cap, sizeof(Node));
    return obj;
}

void jobQueueSysFree(JobQueueSys* obj) {
    if (!obj) return;
    free(obj->a);
    free(obj);
}

static int findIdx(JobQueueSys* obj, int jobId) {
    for (int i = 0; i < obj->n; i++)
        if (obj->a[i].used && obj->a[i].jobId == jobId) return i;
    return -1;
}

bool jobQueueSysSubmit(JobQueueSys* obj, int jobId, int priority) {
    if (findIdx(obj, jobId) >= 0) return false;
    if (obj->n == obj->cap) {
        obj->cap *= 2;
        obj->a = (Node*)realloc(obj->a, sizeof(Node) * (size_t)obj->cap);
    }
    obj->a[obj->n].jobId = jobId;
    obj->a[obj->n].priority = priority;
    obj->a[obj->n].used = 1;
    obj->n++;
    return true;
}

bool jobQueueSysCancel(JobQueueSys* obj, int jobId) {
    int i = findIdx(obj, jobId);
    if (i < 0) return false;
    obj->a[i].used = 0;
    return true;
}

static int bestIdx(JobQueueSys* obj) {
    int best = -1;
    for (int i = 0; i < obj->n; i++) {
        if (!obj->a[i].used) continue;
        if (best < 0 ||
            obj->a[i].priority > obj->a[best].priority ||
            (obj->a[i].priority == obj->a[best].priority &&
             obj->a[i].jobId < obj->a[best].jobId))
            best = i;
    }
    return best;
}

int jobQueueSysPeekJob(JobQueueSys* obj) {
    int i = bestIdx(obj);
    return i < 0 ? -1 : obj->a[i].jobId;
}

int jobQueueSysPopJob(JobQueueSys* obj) {
    int i = bestIdx(obj);
    if (i < 0) return -1;
    int id = obj->a[i].jobId;
    obj->a[i].used = 0;
    return id;
}
```
