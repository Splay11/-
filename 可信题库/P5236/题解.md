## 解题思路

下标按 **从 1 开始** 理解：

1. 列数 $C=\max(colNum)$；行只取输入里出现过的 `rowNum`（排序后依次输出），**跳过空行**。
2. 将每个 `Cell` 填入对应行的第 $colNum$ 列（数组下标 $colNum-1$），后写覆盖先写。
3. 宽度 $w=\max(3,\max|content|)$，左对齐补空格。
4. 输出顶边、各内容行、底边；行间不加 `+---+`。

样例 `[[1,3,""],[3,2,""]]`：$C=3$，行 $\{1,3\}$，得到：

```text
+---+---+---+
|   |   |   |
|   |   |   |
+---+---+---+
```

**注意**：空壳 `return []` 只是模板占位，不是正解。

## 复杂度分析

- 时间：$O(R\cdot C\cdot w)$，$R$ 为出现行数。
- 空间：$O(R\cdot C\cdot w)$。

## 代码实现

### Python

```python
class Cell:
    def __init__(self, rowNum: int, colNum: int, content: str):
        self.rowNum = rowNum
        self.colNum = colNum
        self.content = content


class Solution:
    def transformTable(self, table: list) -> list:
        cells = []
        for c in table:
            if isinstance(c, Cell):
                cells.append(c)
            else:
                cells.append(Cell(int(c[0]), int(c[1]), str(c[2])))
        if not cells:
            return []

        # 只保留出现过的行；列数为 max(col)
        rows = sorted({c.rowNum for c in cells})
        C = max(c.colNum for c in cells)
        grid = {r: [""] * C for r in rows}
        for cell in cells:
            grid[cell.rowNum][cell.colNum - 1] = cell.content

        w = 3
        for r in rows:
            for content in grid[r]:
                w = max(w, len(content))

        def pad(s: str) -> str:
            return s + " " * (w - len(s))

        sep = "+" + "+".join(["-" * w] * C) + "+"
        lines = [sep]
        for r in rows:
            lines.append("|" + "|".join(pad(grid[r][c]) for c in range(C)) + "|")
        lines.append(sep)
        return lines
```

### Java

```java
import java.util.*;

class Cell {
    public int rowNum;
    public int colNum;
    public String content;
    public Cell(int rowNum, int colNum, String content) {
        this.rowNum = rowNum;
        this.colNum = colNum;
        this.content = content;
    }
}

public class Solution {
    public String[] transformTable(Cell[] table) {
        if (table == null || table.length == 0) return new String[0];
        TreeSet<Integer> rowSet = new TreeSet<>();
        int C = 0;
        for (Cell cell : table) {
            rowSet.add(cell.rowNum);
            C = Math.max(C, cell.colNum);
        }
        Map<Integer, String[]> grid = new HashMap<>();
        for (int r : rowSet) {
            String[] row = new String[C];
            Arrays.fill(row, "");
            grid.put(r, row);
        }
        for (Cell cell : table) grid.get(cell.rowNum)[cell.colNum - 1] = cell.content;
        int w = 3;
        for (int r : rowSet)
            for (String s : grid.get(r)) w = Math.max(w, s.length());
        String sep = "+";
        for (int c = 0; c < C; c++) {
            for (int k = 0; k < w; k++) sep += "-";
            sep += "+";
        }
        ArrayList<String> lines = new ArrayList<>();
        lines.add(sep);
        for (int r : rowSet) {
            StringBuilder sb = new StringBuilder("|");
            for (int c = 0; c < C; c++) {
                String s = grid.get(r)[c];
                sb.append(s);
                for (int k = s.length(); k < w; k++) sb.append(' ');
                sb.append('|');
            }
            lines.add(sb.toString());
        }
        lines.add(sep);
        return lines.toArray(new String[0]);
    }
}
```

### C++

```cpp
#include <algorithm>
#include <map>
#include <set>
#include <string>
#include <vector>

using namespace std;

struct Cell {
    int rowNum;
    int colNum;
    string content;
    Cell() : rowNum(0), colNum(0) {}
    Cell(int r, int c, string s) : rowNum(r), colNum(c), content(std::move(s)) {}
};

class Solution {
   public:
    vector<string> transformTable(vector<Cell>& table) {
        if (table.empty()) return {};
        set<int> rowSet;
        int C = 0;
        for (auto& cell : table) {
            rowSet.insert(cell.rowNum);
            C = max(C, cell.colNum);
        }
        vector<int> rows(rowSet.begin(), rowSet.end());
        map<int, vector<string>> grid;
        for (int r : rows) grid[r] = vector<string>(C);
        for (auto& cell : table) grid[cell.rowNum][cell.colNum - 1] = cell.content;
        int w = 3;
        for (int r : rows)
            for (auto& s : grid[r]) w = max(w, (int)s.size());
        auto pad = [&](const string& s) { return s + string(w - (int)s.size(), ' '); };
        string sep = "+";
        for (int c = 0; c < C; c++) {
            sep += string(w, '-');
            sep.push_back('+');
        }
        vector<string> lines;
        lines.push_back(sep);
        for (int r : rows) {
            string line = "|";
            for (int c = 0; c < C; c++) {
                line += pad(grid[r][c]);
                line.push_back('|');
            }
            lines.push_back(line);
        }
        lines.push_back(sep);
        return lines;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

typedef struct {
    int rowNum;
    int colNum;
    char* content;
} Cell;

static int cmpInt(const void* a, const void* b) {
    return *(const int*)a - *(const int*)b;
}

char** transformTable(Cell* table, int tableSize, int* returnSize) {
    if (tableSize == 0) {
        *returnSize = 0;
        return NULL;
    }
    // 收集出现过的行号，跳过空行
    int* rows = (int*)malloc((size_t)tableSize * sizeof(int));
    int rowCnt = 0;
    int C = 0;
    for (int i = 0; i < tableSize; i++) {
        int r = table[i].rowNum;
        int found = 0;
        for (int j = 0; j < rowCnt; j++)
            if (rows[j] == r) {
                found = 1;
                break;
            }
        if (!found) rows[rowCnt++] = r;
        if (table[i].colNum > C) C = table[i].colNum;
    }
    qsort(rows, (size_t)rowCnt, sizeof(int), cmpInt);

    char*** grid = (char***)malloc((size_t)rowCnt * sizeof(char**));
    for (int i = 0; i < rowCnt; i++) {
        grid[i] = (char**)malloc((size_t)C * sizeof(char*));
        for (int c = 0; c < C; c++) grid[i][c] = "";
    }
    for (int i = 0; i < tableSize; i++) {
        int ri = -1;
        for (int j = 0; j < rowCnt; j++)
            if (rows[j] == table[i].rowNum) {
                ri = j;
                break;
            }
        grid[ri][table[i].colNum - 1] = table[i].content;
    }

    int w = 3;
    for (int i = 0; i < rowCnt; i++)
        for (int c = 0; c < C; c++) {
            int len = (int)strlen(grid[i][c]);
            if (len > w) w = len;
        }

    // 构造分隔线 +---+
    int sepLen = 1 + C * (w + 1);
    char* sep = (char*)malloc((size_t)sepLen + 1);
    int p = 0;
    sep[p++] = '+';
    for (int c = 0; c < C; c++) {
        for (int k = 0; k < w; k++) sep[p++] = '-';
        sep[p++] = '+';
    }
    sep[p] = '\0';

    *returnSize = rowCnt + 2;
    char** lines = (char**)malloc((size_t)(*returnSize) * sizeof(char*));
    lines[0] = (char*)malloc((size_t)sepLen + 1);
    strcpy(lines[0], sep);
    for (int i = 0; i < rowCnt; i++) {
        char* line = (char*)malloc((size_t)sepLen + 1);
        int q = 0;
        line[q++] = '|';
        for (int c = 0; c < C; c++) {
            const char* s = grid[i][c];
            int len = (int)strlen(s);
            memcpy(line + q, s, (size_t)len);
            q += len;
            for (int k = len; k < w; k++) line[q++] = ' ';
            line[q++] = '|';
        }
        line[q] = '\0';
        lines[i + 1] = line;
    }
    lines[rowCnt + 1] = (char*)malloc((size_t)sepLen + 1);
    strcpy(lines[rowCnt + 1], sep);

    free(sep);
    for (int i = 0; i < rowCnt; i++) free(grid[i]);
    free(grid);
    free(rows);
    return lines;
}
```
