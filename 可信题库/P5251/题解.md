## 解题思路

本题用**哈希表（集合）**做一次从左到右的扫描。

- 维护集合 `seen`，存放已经出现过的用户编号。
- 扫描到某个编号时，若它已在 `seen` 中，说明这是**第一次发生重复**的位置，直接返回该编号。
- 否则把编号加入 `seen`。
- 若扫完仍无重复，返回 `-1`。

关键点：答案取的是时间线上**最早**触发重复的那一次，所以不能先统计全部再找。

## 复杂度分析

- 时间：$O(n)$，每个元素最多查询/插入哈希表一次。
- 空间：$O(n)$，最坏全部互异时集合大小为 $n$。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def firstDuplicate(self, users: List[int]) -> int:
        seen = set()
        for u in users:
            # 第二次见到该用户即为首次重复
            if u in seen:
                return u
            seen.add(u)
        return -1
```

import java.util.*;

class Solution {
    public int firstDuplicate(int[] users) {
        HashSet<Integer> seen = new HashSet<>();
        for (int u : users) {
            // 第二次见到该用户即为首次重复
            if (seen.contains(u)) return u;
            seen.add(u);
        }
        return -1;
    }
}

### C++

```cpp
#include <unordered_set>
#include <vector>
using namespace std;

class Solution {
public:
    int firstDuplicate(vector<int>& users) {
        unordered_set<int> seen;
        for (int u : users) {
            // 第二次见到该用户即为首次重复
            if (seen.count(u)) return u;
            seen.insert(u);
        }
        return -1;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

/* 开放寻址哈希集合，用于检测首次重复 */
typedef struct {
    int* key;
    char* used;
    int cap;
} HashSet;

static HashSet* hs_create(int n) {
    HashSet* h = (HashSet*)malloc(sizeof(HashSet));
    h->cap = 1;
    while (h->cap < n * 2 + 16) h->cap <<= 1;
    h->key = (int*)malloc(h->cap * sizeof(int));
    h->used = (char*)calloc(h->cap, 1);
    return h;
}

static int hs_insert_or_seen(HashSet* h, int x) {
    unsigned int idx = (unsigned int)x * 2654435761u;
    idx &= (unsigned int)(h->cap - 1);
    while (h->used[idx]) {
        if (h->key[idx] == x) return 1; /* 已存在 */
        idx = (idx + 1) & (unsigned int)(h->cap - 1);
    }
    h->used[idx] = 1;
    h->key[idx] = x;
    return 0;
}

static void hs_free(HashSet* h) {
    free(h->key);
    free(h->used);
    free(h);
}

int firstDuplicate(int* users, int usersSize) {
    HashSet* seen = hs_create(usersSize > 0 ? usersSize : 1);
    for (int i = 0; i < usersSize; i++) {
        // 第二次见到该用户即为首次重复
        if (hs_insert_or_seen(seen, users[i])) {
            int ans = users[i];
            hs_free(seen);
            return ans;
        }
    }
    hs_free(seen);
    return -1;
}
```
