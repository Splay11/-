## 解题思路

这是经典的**环形区间动态规划**（石子合并）。

线性情形：设 $dp[i][j]$ 为把下标 $i..j$ 合并成一盘的最小代价，$sum(i,j)$ 为该区间份量之和，则

$dp[i][j]=\min_{i\le k<j}(dp[i][k]+dp[k+1][j])+\mathrm{sum}(i,j)$

边界 $dp[i][i]=0$。

环形情形：把数组复制一份得到长度 $2n$ 的序列，在上面做同样的区间 DP；答案取所有长度为 $n$ 的区间 $dp[i][i+n-1]$（$0\le i<n$）的最小值。等价于枚举「最后一次合并」切开的位置。

注意：朴素对每个切口各跑一遍线性 DP 是 $O(n^4)$，$n=400$ 会超时；必须在倍长数组上一次性 DP。石子合并满足四边形不等式，可用 Knuth 优化把枚举分点摊还到 $O(n^2)$，Python 建议加上。

## 复杂度分析

- 时间复杂度：朴素 $O(n^3)$；Knuth 优化后 $O(n^2)$
- 空间复杂度：$O(n^2)$

## 代码实现

### Python

```python
from typing import List

class Solution:
    def minCircleMerge(self, weights: List[int]) -> int:
        n = len(weights)
        if n <= 1:
            return 0
        if n == 2:
            return weights[0] + weights[1]

        # 断环：复制数组，在 2n 上做区间 DP
        m = 2 * n
        a = weights + weights
        pref = [0] * (m + 1)
        for i in range(m):
            pref[i + 1] = pref[i] + a[i]

        INF = 10 ** 18
        dp = [[0] * m for _ in range(m)]
        opt = [[0] * m for _ in range(m)]
        for i in range(m):
            opt[i][i] = i
        for length in range(2, n + 1):
            for i in range(0, m - length + 1):
                j = i + length - 1
                lo = opt[i][j - 1]
                hi = min(opt[i + 1][j], j - 1)
                best = INF
                bk = lo
                # Knuth：最优分点单调，缩小枚举区间
                for k in range(lo, hi + 1):
                    cur = dp[i][k] + dp[k + 1][j]
                    if cur < best:
                        best = cur
                        bk = k
                dp[i][j] = best + (pref[j + 1] - pref[i])
                opt[i][j] = bk

        ans = INF
        for i in range(n):
            ans = min(ans, dp[i][i + n - 1])
        return int(ans)
```

### Java

```java
class Solution {
    public long minCircleMerge(int[] weights) {
        int n = weights.length;
        if (n <= 1) return 0;
        if (n == 2) return (long) weights[0] + weights[1];

        int m = 2 * n;
        int[] a = new int[m];
        for (int i = 0; i < n; i++) {
            a[i] = weights[i];
            a[i + n] = weights[i];
        }
        long[] pref = new long[m + 1];
        for (int i = 0; i < m; i++) pref[i + 1] = pref[i] + a[i];

        final long INF = Long.MAX_VALUE / 4;
        long[][] dp = new long[m][m];
        for (int len = 2; len <= n; len++) {
            for (int i = 0; i + len - 1 < m; i++) {
                int j = i + len - 1;
                long best = INF;
                for (int k = i; k < j; k++) {
                    best = Math.min(best, dp[i][k] + dp[k + 1][j]);
                }
                // 合并整段的代价加上区间份量和
                dp[i][j] = best + (pref[j + 1] - pref[i]);
            }
        }
        long ans = INF;
        for (int i = 0; i < n; i++) ans = Math.min(ans, dp[i][i + n - 1]);
        return ans;
    }
}
```

```cpp
class Solution {
```

### C

```c
#include <stdlib.h>

long long minCircleMerge(int* weights, int n) {
    if (n <= 1) return 0;
    if (n == 2) return (long long)weights[0] + weights[1];

    int m = 2 * n;
    int* a = (int*)malloc(sizeof(int) * m);
    for (int i = 0; i < n; i++) {
        a[i] = weights[i];
        a[i + n] = weights[i];
    }
    long long* pref = (long long*)calloc((unsigned)(m + 1), sizeof(long long));
    for (int i = 0; i < m; i++) pref[i + 1] = pref[i] + a[i];

    /* 评测会先 include 选手代码再带标准头，必须自行包含 stdlib.h */
    const long long INF = (1LL << 62);
    long long* dp = (long long*)calloc((unsigned)m * (unsigned)m, sizeof(long long));
    #define DP(i, j) dp[(long long)(i) * m + (j)]

    for (int len = 2; len <= n; len++) {
        for (int i = 0; i + len - 1 < m; i++) {
            int j = i + len - 1;
            long long best = INF;
            for (int k = i; k < j; k++) {
                long long cur = DP(i, k) + DP(k + 1, j);
                if (cur < best) best = cur;
            }
            DP(i, j) = best + (pref[j + 1] - pref[i]);
        }
    }
    long long ans = INF;
    for (int i = 0; i < n; i++) {
        if (DP(i, i + n - 1) < ans) ans = DP(i, i + n - 1);
    }
    free(a);
    free(pref);
    free(dp);
    return ans;
    #undef DP
}
```
