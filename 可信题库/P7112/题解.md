## 解题思路

本题是带**懒过期**的分片互斥租约模拟。维护两张表：

- `lease[shard] = (clientId, expireAt)`
- `by_client[clientId] = {shard: expireAt}`

每次读写前，用当前 `now` 清理相关分片/客户端上已过期的租约（$now \ge expireAt$ 即失效）。

关键语义：

1. `acquire`：分片空闲、客户端持有数 $< maxHold$、参数合法才成功；已持有同一分片应走 `renew`，再次 `acquire` 返回 `false`。
2. `renew`：仅当前未过期持有者可把过期时刻改为 $now+ttl$。
3. `release`：仅持有者可删租约。
4. `owner` / `heldCount`：查询前必须先过期清理。

**常见假解**

- 不做懒过期，导致 `maxHold` 被僵尸租约占满。
- 过期比较写成 `now > expireAt`（题意是 $now\ge expireAt$ 失效）。
- 允许非持有者 `renew`/`release`。

## 复杂度分析

- 单次操作摊还 $O(\textit{maxHold})$（清理该客户端名下分片）；累计调用 $\le 8000$。
- 空间：$O(\textit{shardCount})$。

```python
from typing import Dict, Tuple


class ShardLeaseManager:
    def __init__(self, shardCount: int, maxHold: int) -> None:
        self.shard_count = shardCount
        self.max_hold = maxHold
        self.lease: Dict[int, Tuple[int, int]] = {}
        self.by_client: Dict[int, Dict[int, int]] = {}

    def _valid_shard(self, shard_id: int) -> bool:
        return 0 <= shard_id < self.shard_count

    def _expire_shard(self, shard_id: int, now: int) -> None:
        # 懒过期：到期则双向删掉
        if shard_id not in self.lease:
            return
        cid, exp = self.lease[shard_id]
        if now >= exp:
            del self.lease[shard_id]
            mp = self.by_client.get(cid)
            if mp is not None and shard_id in mp:
                del mp[shard_id]
                if not mp:
                    del self.by_client[cid]

    def _expire_client(self, client_id: int, now: int) -> None:
        mp = self.by_client.get(client_id)
        if not mp:
            return
        for sid in [s for s, e in mp.items() if now >= e]:
            self._expire_shard(sid, now)

    def _active_count(self, client_id: int, now: int) -> int:
        self._expire_client(client_id, now)
        mp = self.by_client.get(client_id)
        return 0 if not mp else len(mp)

    def acquire(self, clientId: int, shardId: int, now: int, ttl: int) -> bool:
        if clientId <= 0 or ttl <= 0 or not self._valid_shard(shardId):
            return False
        self._expire_shard(shardId, now)
        self._expire_client(clientId, now)
        if shardId in self.lease:
            return False  # 已被占用或应走 renew
        if self._active_count(clientId, now) >= self.max_hold:
            return False
        exp = now + ttl
        self.lease[shardId] = (clientId, exp)
        self.by_client.setdefault(clientId, {})[shardId] = exp
        return True

    def renew(self, clientId: int, shardId: int, now: int, ttl: int) -> bool:
        if clientId <= 0 or ttl <= 0 or not self._valid_shard(shardId):
            return False
        self._expire_shard(shardId, now)
        if shardId not in self.lease or self.lease[shardId][0] != clientId:
            return False
        exp = now + ttl
        self.lease[shardId] = (clientId, exp)
        self.by_client.setdefault(clientId, {})[shardId] = exp
        return True

    def release(self, clientId: int, shardId: int, now: int) -> bool:
        if clientId <= 0 or not self._valid_shard(shardId):
            return False
        self._expire_shard(shardId, now)
        if shardId not in self.lease or self.lease[shardId][0] != clientId:
            return False
        del self.lease[shardId]
        mp = self.by_client.get(clientId)
        if mp is not None and shardId in mp:
            del mp[shardId]
            if not mp:
                del self.by_client[clientId]
        return True

    def owner(self, shardId: int, now: int) -> int:
        if not self._valid_shard(shardId):
            return -1
        self._expire_shard(shardId, now)
        return -1 if shardId not in self.lease else self.lease[shardId][0]

    def heldCount(self, clientId: int, now: int) -> int:
        if clientId <= 0:
            return -1
        return self._active_count(clientId, now)
```

### Java

```java
import java.util.*;

class ShardLeaseManager {
    private int shardCount, maxHold;
    private Map<Integer, int[]> lease = new HashMap<>(); // shard -> {cid, exp}
    private Map<Integer, Map<Integer, Integer>> byClient = new HashMap<>();

    public ShardLeaseManager(int shardCount, int maxHold) {
        this.shardCount = shardCount;
        this.maxHold = maxHold;
    }

    private boolean validShard(int sid) {
        return sid >= 0 && sid < shardCount;
    }

    private void expireShard(int sid, int now) {
        int[] info = lease.get(sid);
        if (info == null) return;
        if (now >= info[1]) { // 懒过期
            lease.remove(sid);
            Map<Integer, Integer> mp = byClient.get(info[0]);
            if (mp != null) {
                mp.remove(sid);
                if (mp.isEmpty()) byClient.remove(info[0]);
            }
        }
    }

    private void expireClient(int cid, int now) {
        Map<Integer, Integer> mp = byClient.get(cid);
        if (mp == null) return;
        List<Integer> dead = new ArrayList<>();
        for (Map.Entry<Integer, Integer> e : mp.entrySet())
            if (now >= e.getValue()) dead.add(e.getKey());
        for (int sid : dead) expireShard(sid, now);
    }

    private int activeCount(int cid, int now) {
        expireClient(cid, now);
        Map<Integer, Integer> mp = byClient.get(cid);
        return mp == null ? 0 : mp.size();
    }

    public boolean acquire(int clientId, int shardId, int now, int ttl) {
        if (clientId <= 0 || ttl <= 0 || !validShard(shardId)) return false;
        expireShard(shardId, now);
        expireClient(clientId, now);
        if (lease.containsKey(shardId)) return false;
        if (activeCount(clientId, now) >= maxHold) return false;
        int exp = now + ttl;
        lease.put(shardId, new int[]{clientId, exp});
        byClient.computeIfAbsent(clientId, k -> new HashMap<>()).put(shardId, exp);
        return true;
    }

    public boolean renew(int clientId, int shardId, int now, int ttl) {
        if (clientId <= 0 || ttl <= 0 || !validShard(shardId)) return false;
        expireShard(shardId, now);
        int[] info = lease.get(shardId);
        if (info == null || info[0] != clientId) return false;
        int exp = now + ttl;
        lease.put(shardId, new int[]{clientId, exp});
        byClient.computeIfAbsent(clientId, k -> new HashMap<>()).put(shardId, exp);
        return true;
    }

    public boolean release(int clientId, int shardId, int now) {
        if (clientId <= 0 || !validShard(shardId)) return false;
        expireShard(shardId, now);
        int[] info = lease.get(shardId);
        if (info == null || info[0] != clientId) return false;
        lease.remove(shardId);
        Map<Integer, Integer> mp = byClient.get(clientId);
        if (mp != null) {
            mp.remove(shardId);
            if (mp.isEmpty()) byClient.remove(clientId);
        }
        return true;
    }

    public int owner(int shardId, int now) {
        if (!validShard(shardId)) return -1;
        expireShard(shardId, now);
        int[] info = lease.get(shardId);
        return info == null ? -1 : info[0];
    }

    public int heldCount(int clientId, int now) {
        if (clientId <= 0) return -1;
        return activeCount(clientId, now);
    }
}
```

### C++

```cpp
#include <unordered_map>
using namespace std;

class ShardLeaseManager {
    int shardCount, maxHold;
    unordered_map<int, pair<int,int>> lease; // shard -> {cid, exp}
    unordered_map<int, unordered_map<int,int>> byClient;

    bool validShard(int sid) const { return sid >= 0 && sid < shardCount; }

    void expireShard(int sid, int now) {
        auto it = lease.find(sid);
        if (it == lease.end()) return;
        if (now >= it->second.second) {
            int cid = it->second.first;
            lease.erase(it);
            auto& mp = byClient[cid];
            mp.erase(sid);
            if (mp.empty()) byClient.erase(cid);
        }
    }

    void expireClient(int cid, int now) {
        auto it = byClient.find(cid);
        if (it == byClient.end()) return;
        vector<int> dead;
        for (auto& kv : it->second)
            if (now >= kv.second) dead.push_back(kv.first);
        for (int sid : dead) expireShard(sid, now);
    }

    int activeCount(int cid, int now) {
        expireClient(cid, now);
        auto it = byClient.find(cid);
        return it == byClient.end() ? 0 : (int)it->second.size();
    }

public:
    ShardLeaseManager(int shardCount, int maxHold)
        : shardCount(shardCount), maxHold(maxHold) {}

    bool acquire(int clientId, int shardId, int now, int ttl) {
        if (clientId <= 0 || ttl <= 0 || !validShard(shardId)) return false;
        expireShard(shardId, now);
        expireClient(clientId, now);
        if (lease.count(shardId)) return false;
        if (activeCount(clientId, now) >= maxHold) return false;
        int exp = now + ttl;
        lease[shardId] = {clientId, exp};
        byClient[clientId][shardId] = exp;
        return true;
    }

    bool renew(int clientId, int shardId, int now, int ttl) {
        if (clientId <= 0 || ttl <= 0 || !validShard(shardId)) return false;
        expireShard(shardId, now);
        auto it = lease.find(shardId);
        if (it == lease.end() || it->second.first != clientId) return false;
        int exp = now + ttl;
        it->second = {clientId, exp};
        byClient[clientId][shardId] = exp;
        return true;
    }

    bool release(int clientId, int shardId, int now) {
        if (clientId <= 0 || !validShard(shardId)) return false;
        expireShard(shardId, now);
        auto it = lease.find(shardId);
        if (it == lease.end() || it->second.first != clientId) return false;
        lease.erase(it);
        auto& mp = byClient[clientId];
        mp.erase(shardId);
        if (mp.empty()) byClient.erase(clientId);
        return true;
    }

    int owner(int shardId, int now) {
        if (!validShard(shardId)) return -1;
        expireShard(shardId, now);
        auto it = lease.find(shardId);
        return it == lease.end() ? -1 : it->second.first;
    }

    int heldCount(int clientId, int now) {
        if (clientId <= 0) return -1;
        return activeCount(clientId, now);
    }
};
```

### JavaScript

```javascript
class ShardLeaseManager {
  constructor(shardCount, maxHold) {
    this.shardCount = shardCount;
    this.maxHold = maxHold;
    this.lease = new Map(); // shard -> [cid, exp]
    this.byClient = new Map();
  }

  validShard(sid) {
    return sid >= 0 && sid < this.shardCount;
  }

  expireShard(sid, now) {
    const info = this.lease.get(sid);
    if (!info) return;
    if (now >= info[1]) {
      this.lease.delete(sid);
      const mp = this.byClient.get(info[0]);
      if (mp) {
        mp.delete(sid);
        if (mp.size === 0) this.byClient.delete(info[0]);
      }
    }
  }

  expireClient(cid, now) {
    const mp = this.byClient.get(cid);
    if (!mp) return;
    [...mp.entries()].forEach(([sid, exp]) => {
      if (now >= exp) this.expireShard(sid, now);
    });
  }

  activeCount(cid, now) {
    this.expireClient(cid, now);
    const mp = this.byClient.get(cid);
    return mp ? mp.size : 0;
  }

  acquire(clientId, shardId, now, ttl) {
    if (clientId <= 0 || ttl <= 0 || !this.validShard(shardId)) return false;
    this.expireShard(shardId, now);
    this.expireClient(clientId, now);
    if (this.lease.has(shardId)) return false;
    if (this.activeCount(clientId, now) >= this.maxHold) return false;
    const exp = now + ttl;
    this.lease.set(shardId, [clientId, exp]);
    if (!this.byClient.has(clientId)) this.byClient.set(clientId, new Map());
    this.byClient.get(clientId).set(shardId, exp);
    return true;
  }

  renew(clientId, shardId, now, ttl) {
    if (clientId <= 0 || ttl <= 0 || !this.validShard(shardId)) return false;
    this.expireShard(shardId, now);
    const info = this.lease.get(shardId);
    if (!info || info[0] !== clientId) return false;
    const exp = now + ttl;
    this.lease.set(shardId, [clientId, exp]);
    if (!this.byClient.has(clientId)) this.byClient.set(clientId, new Map());
    this.byClient.get(clientId).set(shardId, exp);
    return true;
  }

  release(clientId, shardId, now) {
    if (clientId <= 0 || !this.validShard(shardId)) return false;
    this.expireShard(shardId, now);
    const info = this.lease.get(shardId);
    if (!info || info[0] !== clientId) return false;
    this.lease.delete(shardId);
    const mp = this.byClient.get(clientId);
    if (mp) {
      mp.delete(shardId);
      if (mp.size === 0) this.byClient.delete(clientId);
    }
    return true;
  }

  owner(shardId, now) {
    if (!this.validShard(shardId)) return -1;
    this.expireShard(shardId, now);
    const info = this.lease.get(shardId);
    return info ? info[0] : -1;
  }

  heldCount(clientId, now) {
    if (clientId <= 0) return -1;
    return this.activeCount(clientId, now);
  }
}
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

#define MAX_SHARD 1000
#define MAX_CLIENT_SLOT 64

typedef struct {
    int shardCount, maxHold;
    int ownerId[MAX_SHARD];
    int expireAt[MAX_SHARD];
    int used[MAX_SHARD];
} shardLeaseManager;

static void expireShard(shardLeaseManager* obj, int sid, int now) {
    if (!obj->used[sid]) return;
    if (now >= obj->expireAt[sid]) { // 懒过期
        obj->used[sid] = 0;
        obj->ownerId[sid] = 0;
    }
}

static int activeCount(shardLeaseManager* obj, int cid, int now) {
    int cnt = 0;
    for (int s = 0; s < obj->shardCount; ++s) {
        expireShard(obj, s, now);
        if (obj->used[s] && obj->ownerId[s] == cid) ++cnt;
    }
    return cnt;
}

shardLeaseManager* shardLeaseManagerCreate(int shardCount, int maxHold) {
    shardLeaseManager* obj = (shardLeaseManager*)calloc(1, sizeof(shardLeaseManager));
    obj->shardCount = shardCount;
    obj->maxHold = maxHold;
    return obj;
}

void shardLeaseManagerFree(shardLeaseManager* obj) { free(obj); }

bool shardLeaseManagerAcquire(shardLeaseManager* obj, int clientId, int shardId, int now, int ttl) {
    if (clientId <= 0 || ttl <= 0 || shardId < 0 || shardId >= obj->shardCount) return false;
    expireShard(obj, shardId, now);
    if (obj->used[shardId]) return false;
    if (activeCount(obj, clientId, now) >= obj->maxHold) return false;
    obj->used[shardId] = 1;
    obj->ownerId[shardId] = clientId;
    obj->expireAt[shardId] = now + ttl;
    return true;
}

bool shardLeaseManagerRenew(shardLeaseManager* obj, int clientId, int shardId, int now, int ttl) {
    if (clientId <= 0 || ttl <= 0 || shardId < 0 || shardId >= obj->shardCount) return false;
    expireShard(obj, shardId, now);
    if (!obj->used[shardId] || obj->ownerId[shardId] != clientId) return false;
    obj->expireAt[shardId] = now + ttl;
    return true;
}

bool shardLeaseManagerRelease(shardLeaseManager* obj, int clientId, int shardId, int now) {
    if (clientId <= 0 || shardId < 0 || shardId >= obj->shardCount) return false;
    expireShard(obj, shardId, now);
    if (!obj->used[shardId] || obj->ownerId[shardId] != clientId) return false;
    obj->used[shardId] = 0;
    return true;
}

int shardLeaseManagerOwner(shardLeaseManager* obj, int shardId, int now) {
    if (shardId < 0 || shardId >= obj->shardCount) return -1;
    expireShard(obj, shardId, now);
    return obj->used[shardId] ? obj->ownerId[shardId] : -1;
}

int shardLeaseManagerHeldCount(shardLeaseManager* obj, int clientId, int now) {
    if (clientId <= 0) return -1;
    return activeCount(obj, clientId, now);
}
```
