## 解题思路

坐标均为整数，矩形可视为半开区域 $[x_1,x_2)\times[y_1,y_2)$。查询区域内每个单位格 $(x,y)$（$x\in[qx_1,qx_2)$，$y\in[qy_1,qy_2)$）至多对应一个**最上层**控件：从大编号到小编号扫描，第一个覆盖该格的矩形即为该格可见控件。

收集所有出现过的顶层编号，再按降序输出。相切（无单位格）自然不会被计入，符合「相切不算相交」。

**常见假解**：

- 只判断与 query 相交、忽略遮挡；
- 遮挡时用「完全被包含」代替「相交区内是否仍有暴露像素」；
- 输出升序。

## 复杂度分析

- 时间：$O((x_2-x_1)(y_2-y_1)\cdot n)$，坐标差 $\le 100$，$n\le 100$，可接受；
- 空间：$O(n)$。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def queryVisibleRects(self, rects: List[List[int]], queryRect: List[int]) -> List[int]:
        qx1, qy1, qx2, qy2 = queryRect
        visible = set()
        # 枚举查询区内每个单位格，找最上层控件
        for x in range(qx1, qx2):
            for y in range(qy1, qy2):
                top = -1
                for i in range(len(rects) - 1, -1, -1):
                    x1, y1, x2, y2 = rects[i]
                    if x1 <= x < x2 and y1 <= y < y2:
                        top = i
                        break
                if top >= 0:
                    visible.add(top)
        return sorted(visible, reverse=True)
```

### Java

```java
import java.util.*;

public class Solution {
    public int[] queryVisibleRects(int[][] rects, int[] queryRect) {
        int qx1 = queryRect[0], qy1 = queryRect[1], qx2 = queryRect[2], qy2 = queryRect[3];
        TreeSet<Integer> visible = new TreeSet<>(Collections.reverseOrder());
        int n = rects.length;
        for (int x = qx1; x < qx2; x++) {
            for (int y = qy1; y < qy2; y++) {
                int top = -1;
                // 从上层往下找第一个覆盖该格的控件
                for (int i = n - 1; i >= 0; i--) {
                    int x1 = rects[i][0], y1 = rects[i][1], x2 = rects[i][2], y2 = rects[i][3];
                    if (x1 <= x && x < x2 && y1 <= y && y < y2) {
                        top = i;
                        break;
                    }
                }
                if (top >= 0) visible.add(top);
            }
        }
        int[] ans = new int[visible.size()];
        int k = 0;
        for (int v : visible) ans[k++] = v;
        return ans;
    }
}
```

### C++

```cpp
#include <algorithm>
#include <functional>
#include <set>
#include <vector>

using namespace std;

class Solution {
   public:
    vector<int> queryVisibleRects(vector<vector<int>>& rects, vector<int>& queryRect) {
        int qx1 = queryRect[0], qy1 = queryRect[1], qx2 = queryRect[2], qy2 = queryRect[3];
        set<int> visible;
        int n = (int)rects.size();
        for (int x = qx1; x < qx2; x++) {
            for (int y = qy1; y < qy2; y++) {
                int top = -1;
                for (int i = n - 1; i >= 0; i--) {
                    int x1 = rects[i][0], y1 = rects[i][1], x2 = rects[i][2], y2 = rects[i][3];
                    if (x1 <= x && x < x2 && y1 <= y && y < y2) {
                        top = i;
                        break;
                    }
                }
                if (top >= 0) visible.insert(top);
            }
        }
        vector<int> ans(visible.begin(), visible.end());
        sort(ans.begin(), ans.end(), greater<int>());
        return ans;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

static int cmpDesc(const void* a, const void* b) {
    return *(const int*)b - *(const int*)a;
}

int* queryVisibleRects(int** rects, int rectsSize, int* rectsColSize, int* queryRect, int queryRectSize,
                       int* returnSize) {
    (void)rectsColSize;
    (void)queryRectSize;
    int qx1 = queryRect[0], qy1 = queryRect[1], qx2 = queryRect[2], qy2 = queryRect[3];
    // 标记出现过的顶层编号
    char* seen = (char*)calloc((size_t)rectsSize, 1);
    for (int x = qx1; x < qx2; x++) {
        for (int y = qy1; y < qy2; y++) {
            int top = -1;
            // 从上层往下找第一个覆盖该格的控件
            for (int i = rectsSize - 1; i >= 0; i--) {
                int x1 = rects[i][0], y1 = rects[i][1], x2 = rects[i][2], y2 = rects[i][3];
                if (x1 <= x && x < x2 && y1 <= y && y < y2) {
                    top = i;
                    break;
                }
            }
            if (top >= 0) seen[top] = 1;
        }
    }
    int* ans = (int*)malloc((size_t)rectsSize * sizeof(int));
    int k = 0;
    for (int i = 0; i < rectsSize; i++)
        if (seen[i]) ans[k++] = i;
    qsort(ans, (size_t)k, sizeof(int), cmpDesc);
    free(seen);
    *returnSize = k;
    return ans;
}
```
