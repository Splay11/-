## 解题思路

把每个 URI 按 `/` 切成子段序列。每次在**尚未入组**的下标集合上：

1. 枚举每个 URI 的所有非空前缀（子段元组），用哈希表统计「前缀 → 覆盖的未入组下标列表」；
2. 在覆盖数 $\ge 2$ 的前缀中取**长度最大**者 $P$；
3. 将这些下标全部划入同一组并移除；组数加一。

若没有任何长度 $\ge 1$ 且覆盖 $\ge 2$ 的前缀，则结束。

同一最大长度下，不同前缀的成员集合互不相交（URI 的前 $L$ 段唯一），故任取其一不影响最终组数。

**常见假解**：按两两最长公共前缀配对（会把样例1的三元组拆错）；只比字符串字典序前缀而非子段；允许空公共前缀成组。

## 复杂度分析

- 时间：设 $n\le 200$，子段数 $\le 20$，每轮 $O(n\cdot 20)$ 建表，轮数 $\le n/2$，总体约 $O(n^2\cdot 20)$，可接受。
- 空间：$O(n\cdot 20)$。

## 代码实现

### Python

```python
from collections import defaultdict
from typing import List


class Solution:
    def countSimilarGroups(self, uriReqs: List[str]) -> int:
        # 去掉开头 '/' 再按段切开
        segs = [u[1:].split("/") for u in uriReqs]
        remaining = set(range(len(segs)))
        groups = 0
        while True:
            prefix_members = defaultdict(list)
            for i in remaining:
                s = segs[i]
                for L in range(1, len(s) + 1):
                    prefix_members[tuple(s[:L])].append(i)
            best_L = 0
            best_key = None
            for pref, mems in prefix_members.items():
                # 至少两人共享，且前缀尽量长
                if len(mems) >= 2 and len(pref) > best_L:
                    best_L = len(pref)
                    best_key = pref
            if best_L == 0:
                break
            for i in prefix_members[best_key]:
                remaining.discard(i)
            groups += 1
        return groups
```

### Java

```java
import java.util.*;

public class Solution {
    public int countSimilarGroups(String[] uriReqs) {
        int n = uriReqs.length;
        List<String[]> segs = new ArrayList<>();
        for (String u : uriReqs) segs.add(u.substring(1).split("/"));
        Set<Integer> remaining = new HashSet<>();
        for (int i = 0; i < n; i++) remaining.add(i);
        int groups = 0;
        while (true) {
            Map<String, List<Integer>> mp = new HashMap<>();
            for (int i : remaining) {
                String[] s = segs.get(i);
                StringBuilder key = new StringBuilder();
                for (int L = 0; L < s.length; L++) {
                    if (L > 0) key.append('\u001f');
                    key.append(s[L]);
                    mp.computeIfAbsent(key.toString(), k -> new ArrayList<>()).add(i);
                }
            }
            int bestL = 0;
            String bestKey = null;
            for (Map.Entry<String, List<Integer>> e : mp.entrySet()) {
                if (e.getValue().size() < 2) continue;
                int L = 1;
                for (int k = 0; k < e.getKey().length(); k++)
                    if (e.getKey().charAt(k) == '\u001f') L++;
                if (L > bestL) {
                    bestL = L;
                    bestKey = e.getKey();
                }
            }
            if (bestL == 0) break;
            for (int i : mp.get(bestKey)) remaining.remove(i);
            groups++;
        }
        return groups;
    }
}
```

### C++

```cpp
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <vector>

using namespace std;

class Solution {
   public:
    int countSimilarGroups(vector<string>& uriReqs) {
        int n = (int)uriReqs.size();
        vector<vector<string>> segs(n);
        for (int i = 0; i < n; i++) {
            string s = uriReqs[i].substr(1), cur;
            for (char c : s) {
                if (c == '/') {
                    segs[i].push_back(cur);
                    cur.clear();
                } else
                    cur.push_back(c);
            }
            segs[i].push_back(cur);
        }
        unordered_set<int> remaining;
        for (int i = 0; i < n; i++) remaining.insert(i);
        int groups = 0;
        while (true) {
            unordered_map<string, vector<int>> mp;
            for (int i : remaining) {
                string key;
                for (size_t L = 0; L < segs[i].size(); L++) {
                    if (L) key.push_back('\x1f');
                    key += segs[i][L];
                    mp[key].push_back(i);
                }
            }
            int bestL = 0;
            string bestKey;
            for (auto& e : mp) {
                if ((int)e.second.size() < 2) continue;
                int L = 1;
                for (char c : e.first)
                    if (c == '\x1f') L++;
                if (L > bestL) {
                    bestL = L;
                    bestKey = e.first;
                }
            }
            if (bestL == 0) break;
            for (int i : mp[bestKey]) remaining.erase(i);
            groups++;
        }
        return groups;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

#define MAXN 220
#define MAXSEG 25
#define MAXSEGLEN 64
#define KEYMAX 512

int countSimilarGroups(char** uriReqs, int uriReqsSize) {
    int n = uriReqsSize;
    char segs[MAXN][MAXSEG][MAXSEGLEN];
    int segCnt[MAXN];
    memset(segs, 0, sizeof(segs));
    for (int i = 0; i < n; i++) {
        segCnt[i] = 0;
        const char* s = uriReqs[i];
        if (*s == '/') s++;
        char cur[MAXSEGLEN];
        int cl = 0;
        for (int k = 0; s[k]; k++) {
            if (s[k] == '/') {
                cur[cl] = '\0';
                strcpy(segs[i][segCnt[i]++], cur);
                cl = 0;
            } else {
                if (cl + 1 < MAXSEGLEN) cur[cl++] = s[k];
            }
        }
        cur[cl] = '\0';
        strcpy(segs[i][segCnt[i]++], cur);
    }

    char remaining[MAXN];
    memset(remaining, 1, n);
    int groups = 0;
    while (1) {
        /* 暴力：对每个前缀 key 统计 remaining 中的命中 */
        int bestL = 0;
        char bestKey[KEYMAX];
        bestKey[0] = '\0';
        int bestIds[MAXN];
        int bestCnt = 0;

        for (int i = 0; i < n; i++) {
            if (!remaining[i]) continue;
            char key[KEYMAX];
            key[0] = '\0';
            for (int L = 0; L < segCnt[i]; L++) {
                if (L) {
                    int kl = (int)strlen(key);
                    if (kl + 1 < KEYMAX) { key[kl] = '\x1f'; key[kl + 1] = '\0'; }
                }
                strcat(key, segs[i][L]);
                /* 统计该 key */
                int ids[MAXN], cnt = 0;
                for (int j = 0; j < n; j++) {
                    if (!remaining[j]) continue;
                    if (L >= segCnt[j]) continue;
                    char kj[KEYMAX];
                    kj[0] = '\0';
                    int ok = 1;
                    for (int t = 0; t <= L; t++) {
                        if (t) {
                            int kl = (int)strlen(kj);
                            if (kl + 1 < KEYMAX) { kj[kl] = '\x1f'; kj[kl + 1] = '\0'; }
                        }
                        if (strcmp(segs[j][t], segs[i][t]) != 0) { ok = 0; break; }
                        strcat(kj, segs[j][t]);
                    }
                    if (ok) ids[cnt++] = j;
                }
                if (cnt >= 2) {
                    int curL = L + 1;
                    if (curL > bestL) {
                        bestL = curL;
                        strcpy(bestKey, key);
                        bestCnt = cnt;
                        for (int t = 0; t < cnt; t++) bestIds[t] = ids[t];
                    }
                }
            }
        }
        if (bestL == 0) break;
        for (int t = 0; t < bestCnt; t++) remaining[bestIds[t]] = 0;
        groups++;
    }
    return groups;
}
```
