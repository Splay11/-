## 解题思路

零信任通道同时带**时延**和**风险消耗**，求额度内的最小时延。这是带一维资源约束的最短路：状态必须是 $(\textit{节点},\textit{已用风险})$，而不能只对每个节点保留一个最短时延。

原因：先走到某点的「更短路径」可能已经把额度花光，另一条较慢但更省额度的路径才能继续走向终点。样例 3 就是这种情况。

边权时延为正，在状态图上做 **Dijkstra**：

- 建有向邻接表
- `dist[u][used]` 表示到达 $u$ 且已消耗 `used` 额度时的最小时延
- 松弛：走边 $(u,v,\textit{delay},\textit{risk})$ 后 `used' = used + risk`，要求 `used' <= riskBudget`
- `src == dst` 直接返回 $0$

状态数 $O(n\cdot B)$，$B=\textit{riskBudget}$。

## 复杂度分析

- 时间复杂度：$O((nB + mB)\log(nB))$，堆优化 Dijkstra。
- 空间复杂度：$O(nB + m)$。

## 代码实现

### Python

```python
from typing import List
import heapq

class Solution:
    def minTrustDelay(self, n: int, edges: List[List[int]], src: int, dst: int, riskBudget: int) -> int:
        if src == dst:
            return 0
        g = [[] for _ in range(n)]
        for u, v, d, r in edges:
            g[u].append((v, d, r))
        inf = 10 ** 18
        # dist[u][used]：到达 u、已用额度 used 的最小时延
        dist = [[inf] * (riskBudget + 1) for _ in range(n)]
        dist[src][0] = 0
        pq = [(0, src, 0)]
        while pq:
            delay, u, used = heapq.heappop(pq)
            if delay != dist[u][used]:
                continue
            if u == dst:
                return delay
            for v, w, r in g[u]:
                nu = used + r
                if nu > riskBudget:
                    continue
                nd = delay + w
                if nd < dist[v][nu]:
                    dist[v][nu] = nd
                    heapq.heappush(pq, (nd, v, nu))
        return -1
```

### Java

```java
import java.util.*;

public class Solution {
    public int minTrustDelay(int n, int[][] edges, int src, int dst, int riskBudget) {
        if (src == dst) return 0;
        List<int[]>[] g = new ArrayList[n];
        for (int i = 0; i < n; i++) g[i] = new ArrayList<>();
        for (int[] e : edges) g[e[0]].add(new int[] {e[1], e[2], e[3]});
        long inf = (1L << 62);
        long[][] dist = new long[n][riskBudget + 1];
        for (int i = 0; i < n; i++) Arrays.fill(dist[i], inf);
        dist[src][0] = 0;
        PriorityQueue<long[]> pq = new PriorityQueue<>(Comparator.comparingLong(a -> a[0]));
        pq.add(new long[] {0, src, 0});
        while (!pq.isEmpty()) {
            long[] cur = pq.poll();
            long delay = cur[0];
            int u = (int) cur[1], used = (int) cur[2];
            if (delay != dist[u][used]) continue;
            if (u == dst) return (int) delay;
            for (int[] e : g[u]) {
                int nu = used + e[2];
                if (nu > riskBudget) continue;
                long nd = delay + e[1];
                if (nd < dist[e[0]][nu]) {
                    dist[e[0]][nu] = nd;
                    pq.add(new long[] {nd, e[0], nu});
                }
            }
        }
        return -1;
    }
}
```

### C++

```cpp
#include <queue>
#include <vector>
using namespace std;

class Solution {
public:
    int minTrustDelay(int n, vector<vector<int>>& edges, int src, int dst, int riskBudget) {
        if (src == dst) return 0;
        vector<vector<vector<int>>> g(n);
        for (size_t i = 0; i < edges.size(); i++) {
            vector<int>& e = edges[i];
            g[e[0]].push_back({e[1], e[2], e[3]});
        }
        const long long INF = (1LL << 62);
        vector<vector<long long>> dist(n, vector<long long>(riskBudget + 1, INF));
        dist[src][0] = 0;
        priority_queue<vector<long long>, vector<vector<long long>>, greater<vector<long long>>> pq;
        pq.push({0, (long long)src, 0});
        while (!pq.empty()) {
            vector<long long> cur = pq.top();
            pq.pop();
            long long delay = cur[0];
            int u = (int)cur[1], used = (int)cur[2];
            if (delay != dist[u][used]) continue;
            if (u == dst) return (int)delay;
            for (size_t i = 0; i < g[u].size(); i++) {
                int v = g[u][i][0], w = g[u][i][1], r = g[u][i][2];
                int nu = used + r;
                if (nu > riskBudget) continue;
                long long nd = delay + w;
                if (nd < dist[v][nu]) {
                    dist[v][nu] = nd;
                    pq.push({nd, (long long)v, (long long)nu});
                }
            }
        }
        return -1;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

typedef struct {
    long long delay;
    int u, used;
} Node;

typedef struct {
    Node* a;
    int size, cap;
} Heap;

static void heap_push(Heap* h, Node x) {
    if (h->size >= h->cap) {
        h->cap = h->cap ? h->cap * 2 : 64;
        h->a = (Node*)realloc(h->a, h->cap * sizeof(Node));
    }
    int i = h->size++;
    h->a[i] = x;
    while (i > 0) {
        int p = (i - 1) / 2;
        if (h->a[p].delay <= h->a[i].delay) break;
        Node t = h->a[p]; h->a[p] = h->a[i]; h->a[i] = t;
        i = p;
    }
}

static Node heap_pop(Heap* h) {
    Node r = h->a[0];
    h->a[0] = h->a[--h->size];
    int i = 0;
    while (1) {
        int l = i * 2 + 1, rgt = l + 1, m = i;
        if (l < h->size && h->a[l].delay < h->a[m].delay) m = l;
        if (rgt < h->size && h->a[rgt].delay < h->a[m].delay) m = rgt;
        if (m == i) break;
        Node t = h->a[i]; h->a[i] = h->a[m]; h->a[m] = t;
        i = m;
    }
    return r;
}

int minTrustDelay(int n, int** edges, int edgesSize, int* edgesColSize, int src, int dst, int riskBudget) {
    (void)edgesColSize;
    if (src == dst) return 0;
    int* head = (int*)malloc(n * sizeof(int));
    memset(head, -1, n * sizeof(int));
    int* to = (int*)malloc((edgesSize + 1) * sizeof(int));
    int* w = (int*)malloc((edgesSize + 1) * sizeof(int));
    int* rk = (int*)malloc((edgesSize + 1) * sizeof(int));
    int* nxt = (int*)malloc((edgesSize + 1) * sizeof(int));
    int ec = 0;
    for (int i = 0; i < edgesSize; i++) {
        int u = edges[i][0], v = edges[i][1];
        to[ec] = v; w[ec] = edges[i][2]; rk[ec] = edges[i][3];
        nxt[ec] = head[u]; head[u] = ec++;
    }
    const long long INF = (1LL << 62);
    long long** dist = (long long**)malloc(n * sizeof(long long*));
    for (int i = 0; i < n; i++) {
        dist[i] = (long long*)malloc((riskBudget + 1) * sizeof(long long));
        for (int j = 0; j <= riskBudget; j++) dist[i][j] = INF;
    }
    dist[src][0] = 0;
    Heap h = {NULL, 0, 0};
    Node s = {0, src, 0};
    heap_push(&h, s);
    int ans = -1;
    while (h.size) {
        Node cur = heap_pop(&h);
        if (cur.delay != dist[cur.u][cur.used]) continue;
        if (cur.u == dst) { ans = (int)cur.delay; break; }
        for (int e = head[cur.u]; e != -1; e = nxt[e]) {
            int nu = cur.used + rk[e];
            if (nu > riskBudget) continue;
            long long nd = cur.delay + w[e];
            if (nd < dist[to[e]][nu]) {
                dist[to[e]][nu] = nd;
                Node nx = {nd, to[e], nu};
                heap_push(&h, nx);
            }
        }
    }
    free(h.a); free(head); free(to); free(w); free(rk); free(nxt);
    for (int i = 0; i < n; i++) free(dist[i]);
    free(dist);
    return ans;
}
```
