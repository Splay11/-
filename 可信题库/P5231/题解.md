## 解题思路

本题是**位运算拼装**：把若干定宽无符号整数按给定位宽依次接到同一个比特流中，再按字节对齐后转大写十六进制。

维护累加器 `acc` 与已写入位数 `total`。对每个字段 $(w,v)$ 执行：

$$
\texttt{acc} \leftarrow (\texttt{acc}\ll w)\mid v,\quad \texttt{total}\leftarrow \texttt{total}+w
$$

这等价于「字段值左侧高位补 0 到 $w$ 位后再拼接」。全部字段写完后，令

$$
\texttt{pad}=(8-\texttt{total}\bmod 8)\bmod 8
$$

将 `acc` 再左移 `pad` 位（在整段**右侧**补 0）。此时总位数是 $8$ 的倍数且 $\le 48$，再从高到低按字节取出并格式化为两位大写十六进制即可。

**常见假解**：

- 字段内右侧补 0（低位补）而非高位补；
- 整段在左侧补齐到字节，而不是右侧；
- 十六进制字母用小写；
- 用字符串逐位拼接导致实现冗长且易错（逻辑上可对，但需注意补齐方向）。

## 复杂度分析

- 时间：$O(n)$，$n\le 10$ 为字段个数；位宽总和 $\le 48$，常数极小。
- 空间：$O(1)$ 额外空间（不计输出串）。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def packFields(self, sectionWidth: List[int], sectionValues: List[int]) -> str:
        acc = 0
        total = 0
        for w, v in zip(sectionWidth, sectionValues):
            # 高位对齐：先腾出 w 位，再写入字段值
            acc = (acc << w) | v
            total += w
        # 整段右侧补 0 到 8 的倍数
        pad = (8 - total % 8) % 8
        acc <<= pad
        total += pad
        nbytes = total // 8
        return f"{acc:0{nbytes * 2}X}"
```

### Java

```java
public class Solution {
    public String packFields(int[] sectionWidth, int[] sectionValues) {
        long acc = 0;
        int total = 0;
        for (int i = 0; i < sectionWidth.length; i++) {
            int w = sectionWidth[i];
            // 依次左移位宽后并入字段值
            acc = (acc << w) | (sectionValues[i] & 0xFFFFFFFFL);
            total += w;
        }
        int pad = (8 - total % 8) % 8;
        acc <<= pad;
        total += pad;
        int nbytes = total / 8;
        char[] hex = "0123456789ABCDEF".toCharArray();
        char[] ans = new char[nbytes * 2];
        // 从低字节往前回填，保证输出高位在前
        for (int i = nbytes - 1; i >= 0; i--) {
            int b = (int) (acc & 0xFF);
            acc >>= 8;
            ans[i * 2] = hex[b >> 4];
            ans[i * 2 + 1] = hex[b & 0xF];
        }
        return new String(ans);
    }
}
```

### C++

```cpp
#include <string>
#include <vector>

using namespace std;

class Solution {
   public:
    string packFields(vector<int>& sectionWidth, vector<int>& sectionValues) {
        unsigned long long acc = 0;
        int total = 0;
        for (size_t i = 0; i < sectionWidth.size(); i++) {
            int w = sectionWidth[i];
            // 高位对齐拼接当前字段
            acc = (acc << w) | (unsigned long long)sectionValues[i];
            total += w;
        }
        int pad = (8 - total % 8) % 8;
        acc <<= pad;
        total += pad;
        int nbytes = total / 8;
        static const char* HEX = "0123456789ABCDEF";
        string ans(nbytes * 2, '0');
        // 按字节从低到高弹出，再写入结果串的右侧
        for (int i = nbytes - 1; i >= 0; i--) {
            unsigned char b = (unsigned char)(acc & 0xFF);
            acc >>= 8;
            ans[i * 2] = HEX[b >> 4];
            ans[i * 2 + 1] = HEX[b & 0xF];
        }
        return ans;
    }
};
```

char* packFields(int* sectionWidth, int sectionWidthSize, int* sectionValues, int sectionValuesSize) {
