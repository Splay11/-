## 解题思路

目标是让数组中存在两个位置，它们的值有大于 $1$ 的公因数，且每次只能把某个位置加 $1$，求最少次数。

关键观察：任意两个数都可以分别加到偶数，从而共享质因子 $2$，因此答案不会超过 $2$。于是只需依次判断答案能否为 $0$ 或 $1$。

1. **答案为 $0$**：统计每个质因子出现在多少个不同位置。若某个质因子覆盖至少两个位置，说明已有一对不互质，返回 $0$。
2. **答案为 $1$**：枚举只给位置 $i$ 加一次，检查 $pieces[i]+1$ 的任一质因子是否已出现在**其它**位置。若是，返回 $1$。
3. 否则返回 $2$。

质因子用线性筛预处理最小质因子，再对每个值做分解即可。

## 复杂度分析

- 时间复杂度：$O(V\log\log V + n\cdot\omega)$，其中 $V$ 为值域上限，$\omega$ 为单个数的不同质因子个数。
- 空间复杂度：$O(V+n)$。

## 代码实现

### Python

```python
from typing import List

class Solution:
    def minShareOps(self, pieces: List[int]) -> int:
        n = len(pieces)
        if n < 2:
            return 0
        mx = max(pieces) + 1
        spf = list(range(mx + 1))
        # 最小质因子筛
        for i in range(2, int(mx ** 0.5) + 1):
            if spf[i] == i:
                for j in range(i * i, mx + 1, i):
                    if spf[j] == j:
                        spf[j] = i

        def factors(x: int):
            s = set()
            while x > 1:
                p = spf[x]
                s.add(p)
                while x % p == 0:
                    x //= p
            return s

        cnt = {}
        facs = []
        for v in pieces:
            fs = factors(v)
            facs.append(fs)
            for p in fs:
                cnt[p] = cnt.get(p, 0) + 1
        # 已有公共质因子
        for c in cnt.values():
            if c >= 2:
                return 0
        # 只加一次
        for i, v in enumerate(pieces):
            for p in factors(v + 1):
                if p in facs[i]:
                    if cnt.get(p, 0) >= 2:
                        return 1
                elif cnt.get(p, 0) >= 1:
                    return 1
        return 2
```

### Java

```java
import java.util.*;

public class Solution {
    public int minShareOps(int[] pieces) {
        int n = pieces.length;
        if (n < 2) return 0;
        int mx = 0;
        for (int v : pieces) mx = Math.max(mx, v);
        mx += 1;
        int[] spf = new int[mx + 1];
        for (int i = 0; i <= mx; i++) spf[i] = i;
        // 筛最小质因子
        for (int i = 2; (long) i * i <= mx; i++) {
            if (spf[i] == i) {
                for (int j = i * i; j <= mx; j += i) {
                    if (spf[j] == j) spf[j] = i;
                }
            }
        }
        List<Set<Integer>> facs = new ArrayList<>();
        Map<Integer, Integer> cnt = new HashMap<>();
        for (int v : pieces) {
            Set<Integer> s = new HashSet<>();
            int x = v;
            while (x > 1) {
                int p = spf[x];
                s.add(p);
                while (x % p == 0) x /= p;
            }
            facs.add(s);
            for (int p : s) cnt.put(p, cnt.getOrDefault(p, 0) + 1);
        }
        for (int c : cnt.values()) if (c >= 2) return 0;
        for (int i = 0; i < n; i++) {
            int x = pieces[i] + 1;
            while (x > 1) {
                int p = spf[x];
                // 加一次后撞上其它位置的质因子
                if (facs.get(i).contains(p)) {
                    if (cnt.getOrDefault(p, 0) >= 2) return 1;
                } else if (cnt.getOrDefault(p, 0) >= 1) {
                    return 1;
                }
                while (x % p == 0) x /= p;
            }
        }
        return 2;
    }
}
```

### C++

```cpp
class Solution {
public:
    int minShareOps(vector<int>& pieces) {
        int n = (int)pieces.size();
        if (n < 2) return 0;
        int mx = *max_element(pieces.begin(), pieces.end()) + 1;
        vector<int> spf(mx + 1);
        iota(spf.begin(), spf.end(), 0);
        // 筛最小质因子
        for (int i = 2; i * i <= mx; i++) if (spf[i] == i)
            for (int j = i * i; j <= mx; j += i) if (spf[j] == j) spf[j] = i;
        auto factors = [&](int x) {
            unordered_set<int> s;
            while (x > 1) {
                int p = spf[x];
                s.insert(p);
                while (x % p == 0) x /= p;
            }
            return s;
        };
        unordered_map<int, int> cnt;
        vector<unordered_set<int>> facs(n);
        for (int i = 0; i < n; i++) {
            facs[i] = factors(pieces[i]);
            for (int p : facs[i]) cnt[p]++;
        }
        for (auto &kv : cnt) if (kv.second >= 2) return 0;
        for (int i = 0; i < n; i++) {
            for (int p : factors(pieces[i] + 1)) {
                // 只操作一次能否形成公共因子
                if (facs[i].count(p)) {
                    if (cnt[p] >= 2) return 1;
                } else if (cnt[p] >= 1) return 1;
            }
        }
        return 2;
    }
};
```

```javascript
class Solution {
    minShareOps(pieces) {
        const n = pieces.length;
        if (n < 2) return 0;
        let mx = Math.max(...pieces) + 1;
        const spf = Array.from({length: mx + 1}, (_, i) => i);
        // 筛最小质因子
        for (let i = 2; i * i <= mx; i++) {
            if (spf[i] === i) {
                for (let j = i * i; j <= mx; j += i) {
                    if (spf[j] === j) spf[j] = i;
                }
            }
        }
        const factors = (x) => {
            const s = new Set();
            while (x > 1) {
                const p = spf[x];
                s.add(p);
                while (x % p === 0) x = Math.floor(x / p);
            }
            return s;
        };
        const cnt = new Map();
        const facs = [];
        for (const v of pieces) {
            const fs = factors(v);
            facs.push(fs);
            for (const p of fs) cnt.set(p, (cnt.get(p) || 0) + 1);
        }
        for (const c of cnt.values()) if (c >= 2) return 0;
        for (let i = 0; i < n; i++) {
            for (const p of factors(pieces[i] + 1)) {
                // 加一次后是否与其它模具共享质因子
                if (facs[i].has(p)) {
                    if ((cnt.get(p) || 0) >= 2) return 1;
                } else if ((cnt.get(p) || 0) >= 1) return 1;
            }
        }
        return 2;
    }
}
```

### C

```c
#include <stdlib.h>

static void factorize(int x, int* spf, int* buf, int* sz) {
    *sz = 0;
    while (x > 1) {
        int p = spf[x];
        buf[(*sz)++] = p;
        while (x % p == 0) x /= p;
    }
}

int minShareOps(int* pieces, int n) {
    if (n < 2) return 0;
    int mx = pieces[0];
    for (int i = 1; i < n; i++) if (pieces[i] > mx) mx = pieces[i];
    mx += 1;
    int* spf = (int*)malloc((mx + 1) * sizeof(int));
    for (int i = 0; i <= mx; i++) spf[i] = i;
    // 筛最小质因子
    for (int i = 2; (long long)i * i <= mx; i++) {
        if (spf[i] == i) {
            for (int j = i * i; j <= mx; j += i)
                if (spf[j] == j) spf[j] = i;
        }
    }
    int* cnt = (int*)calloc(mx + 1, sizeof(int));
    int* selfMark = (int*)calloc(mx + 1, sizeof(int));
    int stamp = 1;
    int buf[64], sz;
    for (int i = 0; i < n; i++) {
        factorize(pieces[i], spf, buf, &sz);
        for (int k = 0; k < sz; k++) cnt[buf[k]]++;
    }
    for (int p = 2; p <= mx; p++) if (cnt[p] >= 2) {
        free(spf); free(cnt); free(selfMark);
        return 0;
    }
    for (int i = 0; i < n; i++) {
        stamp++;
        factorize(pieces[i], spf, buf, &sz);
        for (int k = 0; k < sz; k++) selfMark[buf[k]] = stamp;
        factorize(pieces[i] + 1, spf, buf, &sz);
        for (int k = 0; k < sz; k++) {
            int p = buf[k];
            // 加一次后是否与其它位置共享质因子
            if (selfMark[p] == stamp) {
                if (cnt[p] >= 2) {
                    free(spf); free(cnt); free(selfMark);
                    return 1;
                }
            } else if (cnt[p] >= 1) {
                free(spf); free(cnt); free(selfMark);
                return 1;
            }
        }
    }
    free(spf); free(cnt); free(selfMark);
    return 2;
}
```
