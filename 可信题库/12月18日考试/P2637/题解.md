## 解题思路

本题是典型的**图上状态查询**：微服务之间的依赖构成一张有向无环图（题面保证无环），每个服务还带有「在哪些服务器上运行」的实例集合。

维护三份信息：

1. `running[service]`：该服务当前有实例的服务器集合；
2. `onServer[server]`：该服务器上正在运行的服务集合（便于重启时批量下线）；
3. `deps[from]`：`from` 的直接依赖集合。

各操作对应：

- `startService`：若 `(server, service)` 已存在则返回 `false`，否则两端集合各插入一次；
- `rebootServers`：对每台服务器，把它上的所有服务实例从 `running` 中删掉；
- `addDependency`：依赖边去重后插入；
- `isServiceAvailable`：用记忆化 DFS——服务可提供服务，当且仅当自身实例非空，且所有直接依赖也都可提供服务（递归自然覆盖传递依赖）。

常见错解：只判断依赖是否「在运行」而不递归判断「可提供服务」；或重启一台服务器后忽略其它实例仍在运行。

## 复杂度分析

设调用次数为 $Q$（$Q \le 1000$），涉及服务数为 $S$、依赖边数为 $E$。

- 单次 `startService` / `addDependency` / `rebootServers`：$O(1)$ 到 $O(\text{该服务器上实例数})$。
- 单次 `isServiceAvailable`：记忆化后均摊 $O(S + E)$。
- 总体时间 $O(Q \cdot (S + E))$，空间 $O(S + E + \text{实例数})$，在本题约束下充足。

## 代码实现

### Python

```python
from typing import List, Dict, Set


class ServiceMgrSys:
    def __init__(self):
        # 服务 -> 运行中的服务器集合；服务器 -> 其上服务；依赖边
        self.running: Dict[str, Set[int]] = {}
        self.on_server: Dict[int, Set[str]] = {}
        self.deps: Dict[str, Set[str]] = {}

    def rebootServers(self, serverIds: List[int]) -> None:
        for sid in serverIds:
            # 将该服务器上全部实例下线
            for name in list(self.on_server.get(sid, set())):
                self.running[name].discard(sid)
                if not self.running[name]:
                    del self.running[name]
            self.on_server[sid] = set()

    def startService(self, serverId: int, serviceName: str) -> bool:
        # 同一服务器上同一服务只能有一个实例
        if serviceName in self.on_server.get(serverId, set()):
            return False
        self.on_server.setdefault(serverId, set()).add(serviceName)
        self.running.setdefault(serviceName, set()).add(serverId)
        return True

    def addDependency(self, fromServiceName: str, toServiceName: str) -> bool:
        if toServiceName in self.deps.get(fromServiceName, set()):
            return False
        self.deps.setdefault(fromServiceName, set()).add(toServiceName)
        return True

    def isServiceAvailable(self, serviceName: str) -> bool:
        memo: Dict[str, bool] = {}

        def dfs(name: str) -> bool:
            if name in memo:
                return memo[name]
            # 关键：自身必须在跑，且每个直接依赖都「可提供服务」
            if name not in self.running or not self.running[name]:
                memo[name] = False
                return False
            for dep in self.deps.get(name, set()):
                if not dfs(dep):
                    memo[name] = False
                    return False
            memo[name] = True
            return True

        return dfs(serviceName)
```

### Java

```java
import java.util.*;

class ServiceMgrSys {
    Map<String, Set<Integer>> running = new HashMap<>();
    Map<Integer, Set<String>> onServer = new HashMap<>();
    Map<String, Set<String>> deps = new HashMap<>();

    ServiceMgrSys() {}

    void rebootServers(int[] serverIds) {
        for (int sid : serverIds) {
            // 批量清除该服务器上的全部微服务实例
            Set<String> names = onServer.getOrDefault(sid, Collections.emptySet());
            for (String name : new ArrayList<>(names)) {
                Set<Integer> servers = running.get(name);
                if (servers != null) {
                    servers.remove(sid);
                    if (servers.isEmpty()) running.remove(name);
                }
            }
            onServer.put(sid, new HashSet<>());
        }
    }

    boolean startService(int serverId, String serviceName) {
        Set<String> set = onServer.computeIfAbsent(serverId, k -> new HashSet<>());
        if (set.contains(serviceName)) return false; // 已在运行
        set.add(serviceName);
        running.computeIfAbsent(serviceName, k -> new HashSet<>()).add(serverId);
        return true;
    }

    boolean addDependency(String fromServiceName, String toServiceName) {
        Set<String> set = deps.computeIfAbsent(fromServiceName, k -> new HashSet<>());
        if (set.contains(toServiceName)) return false; // 依赖已存在
        set.add(toServiceName);
        return true;
    }

    boolean isServiceAvailable(String serviceName) {
        Map<String, Boolean> memo = new HashMap<>();
        return dfs(serviceName, memo);
    }

    private boolean dfs(String name, Map<String, Boolean> memo) {
        if (memo.containsKey(name)) return memo.get(name);
        Set<Integer> servers = running.get(name);
        // 无实例则不可用；有实例则递归检查全部直接依赖
        if (servers == null || servers.isEmpty()) {
            memo.put(name, false);
            return false;
        }
        for (String dep : deps.getOrDefault(name, Collections.emptySet())) {
            if (!dfs(dep, memo)) {
                memo.put(name, false);
                return false;
            }
        }
        memo.put(name, true);
        return true;
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

class ServiceMgrSys {
public:
    unordered_map<string, unordered_set<int>> running;
    unordered_map<int, unordered_set<string>> onServer;
    unordered_map<string, unordered_set<string>> deps;

    ServiceMgrSys() {}

    void rebootServers(vector<int>& serverIds) {
        for (int sid : serverIds) {
            // 服务器重启：其上实例全部停止且不自动拉起
            for (const string& name : onServer[sid]) {
                running[name].erase(sid);
                if (running[name].empty()) running.erase(name);
            }
            onServer[sid].clear();
        }
    }

    bool startService(int serverId, string serviceName) {
        if (onServer[serverId].count(serviceName)) return false;
        onServer[serverId].insert(serviceName);
        running[serviceName].insert(serverId);
        return true;
    }

    bool addDependency(string fromServiceName, string toServiceName) {
        if (deps[fromServiceName].count(toServiceName)) return false;
        deps[fromServiceName].insert(toServiceName);
        return true;
    }

    bool isServiceAvailable(string serviceName) {
        unordered_map<string, bool> memo;
        function<bool(const string&)> dfs = [&](const string& name) -> bool {
            if (memo.count(name)) return memo[name];
            // 自身无实例 → false；否则要求所有直接依赖均可提供服务
            if (!running.count(name) || running[name].empty()) {
                return memo[name] = false;
            }
            for (const string& dep : deps[name]) {
                if (!dfs(dep)) return memo[name] = false;
            }
            return memo[name] = true;
        };
        return dfs(serviceName);
    }
};
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

typedef struct ServiceMgrSys ServiceMgrSys;

#define MAX_NAME 64
#define MAX_SVC 512
#define MAX_SRV 512

struct ServiceMgrSys {
    char names[MAX_SVC][MAX_NAME];
    int svcCnt;
    // running[svc][k] = serverId
    int running[MAX_SVC][MAX_SRV];
    int runCnt[MAX_SVC];
    // onServer: list of (serverId, svcIdx) pairs stored as per-server lists
    int servers[MAX_SRV];
    int srvCnt;
    int onServer[MAX_SRV][MAX_SVC];
    int onCnt[MAX_SRV];
    // deps[from][k] = to
    int deps[MAX_SVC][MAX_SVC];
    int depCnt[MAX_SVC];
};

static int findSvc(ServiceMgrSys* obj, const char* name) {
    for (int i = 0; i < obj->svcCnt; i++)
        if (strcmp(obj->names[i], name) == 0) return i;
    return -1;
}

static int getOrAddSvc(ServiceMgrSys* obj, const char* name) {
    int id = findSvc(obj, name);
    if (id >= 0) return id;
    id = obj->svcCnt++;
    strncpy(obj->names[id], name, MAX_NAME - 1);
    obj->names[id][MAX_NAME - 1] = '\0';
    return id;
}

static int findSrv(ServiceMgrSys* obj, int serverId) {
    for (int i = 0; i < obj->srvCnt; i++)
        if (obj->servers[i] == serverId) return i;
    return -1;
}

static int getOrAddSrv(ServiceMgrSys* obj, int serverId) {
    int id = findSrv(obj, serverId);
    if (id >= 0) return id;
    id = obj->srvCnt++;
    obj->servers[id] = serverId;
    return id;
}

ServiceMgrSys* serviceMgrSysCreate(void) {
    return (ServiceMgrSys*)calloc(1, sizeof(ServiceMgrSys));
}

void serviceMgrSysFree(ServiceMgrSys* obj) {
    free(obj);
}

void serviceMgrSysRebootServers(ServiceMgrSys* obj, int* serverIds, int serverIdsSize) {
    for (int t = 0; t < serverIdsSize; t++) {
        int sid = serverIds[t];
        int si = findSrv(obj, sid);
        if (si < 0) continue;
        // 服务器重启：其上实例全部停止
        for (int k = 0; k < obj->onCnt[si]; k++) {
            int svc = obj->onServer[si][k];
            int w = 0;
            for (int j = 0; j < obj->runCnt[svc]; j++)
                if (obj->running[svc][j] != sid) obj->running[svc][w++] = obj->running[svc][j];
            obj->runCnt[svc] = w;
        }
        obj->onCnt[si] = 0;
    }
}

bool serviceMgrSysStartService(ServiceMgrSys* obj, int serverId, char* serviceName) {
    int svc = getOrAddSvc(obj, serviceName);
    int si = getOrAddSrv(obj, serverId);
    for (int k = 0; k < obj->onCnt[si]; k++)
        if (obj->onServer[si][k] == svc) return false;
    obj->onServer[si][obj->onCnt[si]++] = svc;
    obj->running[svc][obj->runCnt[svc]++] = serverId;
    return true;
}

bool serviceMgrSysAddDependency(ServiceMgrSys* obj, char* fromServiceName, char* toServiceName) {
    int from = getOrAddSvc(obj, fromServiceName);
    int to = getOrAddSvc(obj, toServiceName);
    for (int k = 0; k < obj->depCnt[from]; k++)
        if (obj->deps[from][k] == to) return false;
    obj->deps[from][obj->depCnt[from]++] = to;
    return true;
}

static bool dfsAvail(ServiceMgrSys* obj, int nameIdx, char* memo, char* vis) {
    if (vis[nameIdx]) return memo[nameIdx];
    vis[nameIdx] = 1;
    if (obj->runCnt[nameIdx] == 0) {
        memo[nameIdx] = 0;
        return false;
    }
    for (int k = 0; k < obj->depCnt[nameIdx]; k++) {
        if (!dfsAvail(obj, obj->deps[nameIdx][k], memo, vis)) {
            memo[nameIdx] = 0;
            return false;
        }
    }
    memo[nameIdx] = 1;
    return true;
}

bool serviceMgrSysIsServiceAvailable(ServiceMgrSys* obj, char* serviceName) {
    int idx = findSvc(obj, serviceName);
    if (idx < 0) return false;
    char memo[MAX_SVC], vis[MAX_SVC];
    memset(memo, 0, sizeof(memo));
    memset(vis, 0, sizeof(vis));
    return dfsAvail(obj, idx, memo, vis);
}
```
