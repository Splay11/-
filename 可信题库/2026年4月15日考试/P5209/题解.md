## 解题思路

$n\le 100$，可枚举所有连续子区间 $[i,j]$。对每个固定左端点 $i$，从左到右扩展右端点，并用有序插入维护当前区间的排序数组；每次扩展后按奇偶长度取中位数并累加。

偶数长度 $k=2m$ 时，中间两数为有序数组下标 $m-1$ 与 $m$，取较小者即 `cur[k/2-1]`；奇数取 `cur[k/2]`。

**常见假解**：偶数长度取较大者或取平均；只统计奇数长度子数组。

## 复杂度分析

- 时间：$O(n^3)$（插入 $O(n)$，共 $O(n^2)$ 个区间）；
- 空间：$O(n)$。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def sumOfSubarrayMedians(self, numbers: List[int]) -> int:
        n = len(numbers)
        ans = 0
        for i in range(n):
            cur = []
            for j in range(i, n):
                # 有序插入当前元素
                x = numbers[j]
                lo, hi = 0, len(cur)
                while lo < hi:
                    mid = (lo + hi) // 2
                    if cur[mid] < x:
                        lo = mid + 1
                    else:
                        hi = mid
                cur.insert(lo, x)
                k = len(cur)
                # 偶数取中间较小者
                if k % 2 == 1:
                    ans += cur[k // 2]
                else:
                    ans += cur[k // 2 - 1]
        return ans
```

### Java

```java
import java.util.*;

public class Solution {
    public long sumOfSubarrayMedians(int[] numbers) {
        int n = numbers.length;
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ArrayList<Integer> cur = new ArrayList<>();
            for (int j = i; j < n; j++) {
                int x = numbers[j];
                int lo = 0, hi = cur.size();
                while (lo < hi) {
                    int mid = (lo + hi) >>> 1;
                    if (cur.get(mid) < x) lo = mid + 1;
                    else hi = mid;
                }
                cur.add(lo, x);
                int k = cur.size();
                if (k % 2 == 1) ans += cur.get(k / 2);
                else ans += cur.get(k / 2 - 1);
            }
        }
        return ans;
    }
}
```

```cpp
            vector<int> cur;
            for (int j = i; j < n; j++) {
                auto it = lower_bound(cur.begin(), cur.end(), numbers[j]);
                cur.insert(it, numbers[j]);
                int k = (int)cur.size();
                if (k % 2 == 1)
                    ans += cur[k / 2];
                else
                    ans += cur[k / 2 - 1];
            }
        }
        return ans;
    }
};
```

### C

```c
#include <stdlib.h>

long long sumOfSubarrayMedians(int* numbers, int numbersSize) {
    long long ans = 0;
    // 枚举左端点，维护有序数组
    for (int i = 0; i < numbersSize; i++) {
        int* cur = (int*)malloc((size_t)(numbersSize - i) * sizeof(int));
        int k = 0;
        for (int j = i; j < numbersSize; j++) {
            int x = numbers[j];
            // 二分找插入位置
            int lo = 0, hi = k;
            while (lo < hi) {
                int mid = (lo + hi) / 2;
                if (cur[mid] < x)
                    lo = mid + 1;
                else
                    hi = mid;
            }
            for (int t = k; t > lo; t--) cur[t] = cur[t - 1];
            cur[lo] = x;
            k++;
            // 偶数取中间较小者
            if (k % 2 == 1)
                ans += cur[k / 2];
            else
                ans += cur[k / 2 - 1];
        }
        free(cur);
    }
    return ans;
}
```
