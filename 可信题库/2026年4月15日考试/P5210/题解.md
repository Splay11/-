## 解题思路

六种对齐均为根据全体卡片的边界（或中心）计算目标线，再只改 `x` 或只改 `y`：

- `left/right/top/bottom`：对齐到极值边缘；
- `horizontal_center` / `vertical_center`：先算包围盒中点（向下取整），再令 `new = mid - size//2`。

`width`、`height` 不变，卡片顺序不变。

**常见假解**：居中时用四舍五入；`right` 写成对齐左边缘；垂直/水平轴搞反。

## 复杂度分析

- 时间：$O(n)$；
- 空间：$O(n)$（输出）。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def alignCards(self, cards: List[List[int]], alignment: str) -> List[List[int]]:
        ans = [c[:] for c in cards]
        if alignment == "top":
            ty = min(c[1] for c in cards)
            for a in ans:
                a[1] = ty
        elif alignment == "bottom":
            by = max(c[1] + c[3] for c in cards)
            for a in ans:
                a[1] = by - a[3]
        elif alignment == "left":
            lx = min(c[0] for c in cards)
            for a in ans:
                a[0] = lx
        elif alignment == "right":
            rx = max(c[0] + c[2] for c in cards)
            for a in ans:
                a[0] = rx - a[2]
        elif alignment == "horizontal_center":
            x1 = min(c[0] for c in cards)
            xr = max(c[0] + c[2] for c in cards)
            mid = (x1 + xr) // 2
            for a in ans:
                a[0] = mid - a[2] // 2
        elif alignment == "vertical_center":
            y1 = min(c[1] for c in cards)
            yb = max(c[1] + c[3] for c in cards)
            mid = (y1 + yb) // 2
            for a in ans:
                a[1] = mid - a[3] // 2
        return ans
```

### Java

```java
public class Solution {
    public int[][] alignCards(int[][] cards, String alignment) {
        int n = cards.length;
        int[][] ans = new int[n][4];
        for (int i = 0; i < n; i++) ans[i] = cards[i].clone();
        if (alignment.equals("top")) {
            int ty = cards[0][1];
            for (int[] c : cards) ty = Math.min(ty, c[1]);
            for (int[] a : ans) a[1] = ty;
        } else if (alignment.equals("bottom")) {
            int by = cards[0][1] + cards[0][3];
            for (int[] c : cards) by = Math.max(by, c[1] + c[3]);
            for (int[] a : ans) a[1] = by - a[3];
        } else if (alignment.equals("left")) {
            int lx = cards[0][0];
            for (int[] c : cards) lx = Math.min(lx, c[0]);
            for (int[] a : ans) a[0] = lx;
        } else if (alignment.equals("right")) {
            int rx = cards[0][0] + cards[0][2];
            for (int[] c : cards) rx = Math.max(rx, c[0] + c[2]);
            for (int[] a : ans) a[0] = rx - a[2];
        } else if (alignment.equals("horizontal_center")) {
            int x1 = cards[0][0], xr = cards[0][0] + cards[0][2];
            for (int[] c : cards) {
                x1 = Math.min(x1, c[0]);
                xr = Math.max(xr, c[0] + c[2]);
            }
            int mid = (x1 + xr) / 2;
            for (int[] a : ans) a[0] = mid - a[2] / 2;
        } else if (alignment.equals("vertical_center")) {
            int y1 = cards[0][1], yb = cards[0][1] + cards[0][3];
            for (int[] c : cards) {
                y1 = Math.min(y1, c[1]);
                yb = Math.max(yb, c[1] + c[3]);
            }
            int mid = (y1 + yb) / 2;
            for (int[] a : ans) a[1] = mid - a[3] / 2;
        }
        return ans;
    }
}
```

### C

```c
#include <stdlib.h>
#include <string.h>

int** alignCards(int** cards, int cardsSize, int* cardsColSize, char* alignment, int* returnSize,
                 int** returnColumnSizes) {
    (void)cardsColSize;
    *returnSize = cardsSize;
    *returnColumnSizes = (int*)malloc((size_t)cardsSize * sizeof(int));
    int** ans = (int**)malloc((size_t)cardsSize * sizeof(int*));
    for (int i = 0; i < cardsSize; i++) {
        (*returnColumnSizes)[i] = 4;
        ans[i] = (int*)malloc(4 * sizeof(int));
        // 先复制，再只改 x 或 y
        memcpy(ans[i], cards[i], 4 * sizeof(int));
    }
    if (cardsSize == 0) return ans;

    if (strcmp(alignment, "top") == 0) {
        int ty = cards[0][1];
        for (int i = 0; i < cardsSize; i++)
            if (cards[i][1] < ty) ty = cards[i][1];
        for (int i = 0; i < cardsSize; i++) ans[i][1] = ty;
    } else if (strcmp(alignment, "bottom") == 0) {
        int by = cards[0][1] + cards[0][3];
        for (int i = 0; i < cardsSize; i++) {
            int v = cards[i][1] + cards[i][3];
            if (v > by) by = v;
        }
        for (int i = 0; i < cardsSize; i++) ans[i][1] = by - ans[i][3];
    } else if (strcmp(alignment, "left") == 0) {
        int lx = cards[0][0];
        for (int i = 0; i < cardsSize; i++)
            if (cards[i][0] < lx) lx = cards[i][0];
        for (int i = 0; i < cardsSize; i++) ans[i][0] = lx;
    } else if (strcmp(alignment, "right") == 0) {
        int rx = cards[0][0] + cards[0][2];
        for (int i = 0; i < cardsSize; i++) {
            int v = cards[i][0] + cards[i][2];
            if (v > rx) rx = v;
        }
        for (int i = 0; i < cardsSize; i++) ans[i][0] = rx - ans[i][2];
    } else if (strcmp(alignment, "horizontal_center") == 0) {
        int x1 = cards[0][0], xr = cards[0][0] + cards[0][2];
        for (int i = 0; i < cardsSize; i++) {
            if (cards[i][0] < x1) x1 = cards[i][0];
            int r = cards[i][0] + cards[i][2];
            if (r > xr) xr = r;
        }
        int mid = (x1 + xr) / 2;
        for (int i = 0; i < cardsSize; i++) ans[i][0] = mid - ans[i][2] / 2;
    } else if (strcmp(alignment, "vertical_center") == 0) {
        int y1 = cards[0][1], yb = cards[0][1] + cards[0][3];
        for (int i = 0; i < cardsSize; i++) {
            if (cards[i][1] < y1) y1 = cards[i][1];
            int b = cards[i][1] + cards[i][3];
            if (b > yb) yb = b;
        }
        int mid = (y1 + yb) / 2;
        for (int i = 0; i < cardsSize; i++) ans[i][1] = mid - ans[i][3] / 2;
    }
    return ans;
}
```
