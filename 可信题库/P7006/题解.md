## 解题思路

本题是**哈希表模拟**题：维护集群节点与任务的状态，按 **best-fit** 规则调度。

用两张表即可：

- `cap[nodeId]` / `used[nodeId]` / `jobCnt[nodeId]`：节点容量、已用额度、其上未结束任务数
- `jobs[jobId] -> (nodeId, size)`：运行中任务的落点与占用

各接口语义：

1. **addNode**：`nodeId` 已存在或 `capacity <= 0` 则失败；否则写入三张节点表，`used=0`。
2. **removeNode**：节点不存在，或 `jobCnt > 0`，失败；否则删除节点记录。成功后同一 `nodeId` 可再 `addNode`（容量可不同）。
3. **submit**：`jobId` 已在跑或 `size <= 0` 返回 `-1`。否则扫描所有节点，在 `rem = cap - used >= size` 的候选中，选 **剩余最小**；并列选 **nodeId 更小**。找不到返回 `-1`，找到则累加 `used` / `jobCnt` 并登记任务。
4. **kill**：查不到任务失败；否则释放额度、递减计数、删掉任务记录。结束后同一 `jobId` 可再次提交。
5. **usedOf / freeOf / jobNode**：存在性查询，不存在返回 `-1`。

易错点：

- 并列时必须用更小 `nodeId`，不能用插入顺序或 first-fit
- 有任务时禁止下线；杀光后再删
- 单次占用之和不超过 `capacity`，用 32 位整数即可

## 复杂度分析

- 时间复杂度：每次 `submit` 扫描全部节点，设节点数为 $N$、调用次数为 $Q$，则总复杂度 $O(QN)$。本题 $Q\le 4000$，足够。
- 空间复杂度：$O(N+J)$，其中 $J$ 为同时运行的任务数。

### Python

```python
from typing import Dict, Tuple


class ClusterPool:
    def __init__(self):
        self.cap: Dict[int, int] = {}
        self.used: Dict[int, int] = {}
        self.job_cnt: Dict[int, int] = {}
        self.jobs: Dict[int, Tuple[int, int]] = {}  # jobId -> (nodeId, size)

    def addNode(self, nodeId: int, capacity: int) -> bool:
        # 已存在或非法容量
        if nodeId in self.cap or capacity <= 0:
            return False
        self.cap[nodeId] = capacity
        self.used[nodeId] = 0
        self.job_cnt[nodeId] = 0
        return True

    def removeNode(self, nodeId: int) -> bool:
        if nodeId not in self.cap:
            return False
        # 仍有任务不可下线
        if self.job_cnt[nodeId] > 0:
            return False
        del self.cap[nodeId]
        del self.used[nodeId]
        del self.job_cnt[nodeId]
        return True

    def submit(self, jobId: int, size: int) -> int:
        if jobId in self.jobs or size <= 0:
            return -1
        best_id, best_rem = -1, None
        for nid, capacity in self.cap.items():
            rem = capacity - self.used[nid]
            if rem < size:
                continue
            # best-fit：更小剩余优先；并列取更小 nodeId
            if best_rem is None or rem < best_rem or (rem == best_rem and nid < best_id):
                best_rem, best_id = rem, nid
        if best_id < 0:
            return -1
        self.used[best_id] += size
        self.job_cnt[best_id] += 1
        self.jobs[jobId] = (best_id, size)
        return best_id

    def kill(self, jobId: int) -> bool:
        info = self.jobs.pop(jobId, None)
        if info is None:
            return False
        nid, size = info
        self.used[nid] -= size
        self.job_cnt[nid] -= 1
        return True

    def usedOf(self, nodeId: int) -> int:
        return -1 if nodeId not in self.cap else self.used[nodeId]

    def freeOf(self, nodeId: int) -> int:
        if nodeId not in self.cap:
            return -1
        return self.cap[nodeId] - self.used[nodeId]

    def jobNode(self, jobId: int) -> int:
        info = self.jobs.get(jobId)
        return -1 if info is None else info[0]
```

### Java

```java
import java.util.HashMap;
import java.util.Map;

class ClusterPool {
    private Map<Integer, Integer> cap = new HashMap<>();
    private Map<Integer, Integer> used = new HashMap<>();
    private Map<Integer, Integer> jobCnt = new HashMap<>();
    private Map<Integer, int[]> jobs = new HashMap<>(); // jobId -> {nodeId, size}

    public ClusterPool() {}

    public boolean addNode(int nodeId, int capacity) {
        // 已存在或非法容量
        if (cap.containsKey(nodeId) || capacity <= 0) return false;
        cap.put(nodeId, capacity);
        used.put(nodeId, 0);
        jobCnt.put(nodeId, 0);
        return true;
    }

    public boolean removeNode(int nodeId) {
        if (!cap.containsKey(nodeId)) return false;
        // 仍有任务不可下线
        if (jobCnt.get(nodeId) > 0) return false;
        cap.remove(nodeId);
        used.remove(nodeId);
        jobCnt.remove(nodeId);
        return true;
    }

    public int submit(int jobId, int size) {
        if (jobs.containsKey(jobId) || size <= 0) return -1;
        int bestId = -1;
        Integer bestRem = null;
        for (Map.Entry<Integer, Integer> e : cap.entrySet()) {
            int nid = e.getKey();
            int rem = e.getValue() - used.get(nid);
            if (rem < size) continue;
            // best-fit：更小剩余优先；并列取更小 nodeId
            if (bestRem == null || rem < bestRem || (rem == bestRem && nid < bestId)) {
                bestRem = rem;
                bestId = nid;
            }
        }
        if (bestId < 0) return -1;
        used.put(bestId, used.get(bestId) + size);
        jobCnt.put(bestId, jobCnt.get(bestId) + 1);
        jobs.put(jobId, new int[] {bestId, size});
        return bestId;
    }

    public boolean kill(int jobId) {
        int[] info = jobs.remove(jobId);
        if (info == null) return false;
        int nid = info[0], size = info[1];
        used.put(nid, used.get(nid) - size);
        jobCnt.put(nid, jobCnt.get(nid) - 1);
        return true;
    }

    public int usedOf(int nodeId) {
        return cap.containsKey(nodeId) ? used.get(nodeId) : -1;
    }

    public int freeOf(int nodeId) {
        if (!cap.containsKey(nodeId)) return -1;
        return cap.get(nodeId) - used.get(nodeId);
    }

    public int jobNode(int jobId) {
        int[] info = jobs.get(jobId);
        return info == null ? -1 : info[0];
    }
}
```

### C++

```cpp
#include <unordered_map>
#include <utility>
using namespace std;

class ClusterPool {
    unordered_map<int, int> cap, used, jobCnt;
    unordered_map<int, pair<int, int>> jobs;  // jobId -> {nodeId, size}

public:
    ClusterPool() {}

    bool addNode(int nodeId, int capacity) {
        // 已存在或非法容量
        if (cap.count(nodeId) || capacity <= 0) return false;
        cap[nodeId] = capacity;
        used[nodeId] = 0;
        jobCnt[nodeId] = 0;
        return true;
    }

    bool removeNode(int nodeId) {
        if (!cap.count(nodeId)) return false;
        // 仍有任务不可下线
        if (jobCnt[nodeId] > 0) return false;
        cap.erase(nodeId);
        used.erase(nodeId);
        jobCnt.erase(nodeId);
        return true;
    }

    int submit(int jobId, int size) {
        if (jobs.count(jobId) || size <= 0) return -1;
        int bestId = -1;
        int bestRem = 0;
        bool found = false;
        for (auto& e : cap) {
            int nid = e.first;
            int rem = e.second - used[nid];
            if (rem < size) continue;
            // best-fit：更小剩余优先；并列取更小 nodeId
            if (!found || rem < bestRem || (rem == bestRem && nid < bestId)) {
                found = true;
                bestRem = rem;
                bestId = nid;
            }
        }
        if (!found) return -1;
        used[bestId] += size;
        jobCnt[bestId] += 1;
        jobs[jobId] = {bestId, size};
        return bestId;
    }

    bool kill(int jobId) {
        auto it = jobs.find(jobId);
        if (it == jobs.end()) return false;
        int nid = it->second.first;
        int sz = it->second.second;
        used[nid] -= sz;
        jobCnt[nid] -= 1;
        jobs.erase(it);
        return true;
    }

    int usedOf(int nodeId) {
        return cap.count(nodeId) ? used[nodeId] : -1;
    }

    int freeOf(int nodeId) {
        if (!cap.count(nodeId)) return -1;
        return cap[nodeId] - used[nodeId];
    }

    int jobNode(int jobId) {
        auto it = jobs.find(jobId);
        return it == jobs.end() ? -1 : it->second.first;
    }
};
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

/* 开放寻址哈希表：存节点 / 任务 */
typedef struct {
    int key;
    int a, b, c; /* 节点: cap,used,jobCnt；任务: nodeId,size,unused */
    int occupied; /* 槽是否占用 */
} Slot;

typedef struct {
    Slot* tab;
    int n, cap;
} Map;

typedef struct clusterPool {
    Map nodes; /* key=nodeId, a=cap, b=used, c=jobCnt */
    Map jobs;  /* key=jobId, a=nodeId, b=size */
} clusterPool;

static unsigned hash_i(int k) { return (unsigned)k * 2654435761u; }

static void map_init(Map* m, int cap) {
    m->cap = cap;
    m->n = 0;
    m->tab = (Slot*)calloc((size_t)cap, sizeof(Slot));
}

static void map_free(Map* m) {
    free(m->tab);
    m->tab = NULL;
}

static Slot* map_find(Map* m, int key) {
    unsigned h = hash_i(key) % (unsigned)m->cap;
    for (int i = 0; i < m->cap; i++) {
        unsigned idx = (h + (unsigned)i) % (unsigned)m->cap;
        if (!m->tab[idx].occupied) return NULL;
        if (m->tab[idx].key == key) return &m->tab[idx];
    }
    return NULL;
}

static void map_grow(Map* m);

static Slot* map_put(Map* m, int key) {
    if (m->n * 2 >= m->cap) map_grow(m);
    unsigned h = hash_i(key) % (unsigned)m->cap;
    for (int i = 0; i < m->cap; i++) {
        unsigned idx = (h + (unsigned)i) % (unsigned)m->cap;
        if (!m->tab[idx].occupied) {
            m->tab[idx].occupied = 1;
            m->tab[idx].key = key;
            m->tab[idx].a = m->tab[idx].b = m->tab[idx].c = 0;
            m->n++;
            return &m->tab[idx];
        }
        if (m->tab[idx].key == key) return &m->tab[idx];
    }
    return NULL;
}

static void map_grow(Map* m) {
    Slot* old = m->tab;
    int oc = m->cap;
    map_init(m, oc * 2);
    for (int i = 0; i < oc; i++) {
        if (!old[i].occupied) continue;
        Slot* s = map_put(m, old[i].key);
        s->a = old[i].a;
        s->b = old[i].b;
        s->c = old[i].c;
    }
    free(old);
}

/* 删除后把后续冲突链前移，避免探测断裂 */
static void map_erase(Map* m, int key) {
    unsigned h = hash_i(key) % (unsigned)m->cap;
    for (int i = 0; i < m->cap; i++) {
        unsigned idx = (h + (unsigned)i) % (unsigned)m->cap;
        if (!m->tab[idx].occupied) return;
        if (m->tab[idx].key != key) continue;
        m->tab[idx].occupied = 0;
        m->n--;
        unsigned j = (idx + 1) % (unsigned)m->cap;
        while (m->tab[j].occupied) {
            Slot tmp = m->tab[j];
            m->tab[j].occupied = 0;
            m->n--;
            Slot* s = map_put(m, tmp.key);
            s->a = tmp.a;
            s->b = tmp.b;
            s->c = tmp.c;
            j = (j + 1) % (unsigned)m->cap;
        }
        return;
    }
}

clusterPool* clusterPoolCreate(void) {
    clusterPool* obj = (clusterPool*)calloc(1, sizeof(clusterPool));
    map_init(&obj->nodes, 16);
    map_init(&obj->jobs, 16);
    return obj;
}

void clusterPoolFree(clusterPool* obj) {
    if (!obj) return;
    map_free(&obj->nodes);
    map_free(&obj->jobs);
    free(obj);
}

bool clusterPoolAddNode(clusterPool* obj, int nodeId, int capacity) {
    if (capacity <= 0 || map_find(&obj->nodes, nodeId)) return false;
    Slot* s = map_put(&obj->nodes, nodeId);
    s->a = capacity;
    s->b = 0;
    s->c = 0;
    return true;
}

bool clusterPoolRemoveNode(clusterPool* obj, int nodeId) {
    Slot* s = map_find(&obj->nodes, nodeId);
    if (!s) return false;
    if (s->c > 0) return false; /* 仍有任务不可下线 */
    map_erase(&obj->nodes, nodeId);
    return true;
}

int clusterPoolSubmit(clusterPool* obj, int jobId, int size) {
    if (size <= 0 || map_find(&obj->jobs, jobId)) return -1;
    int bestId = -1, bestRem = 0, found = 0;
    for (int i = 0; i < obj->nodes.cap; i++) {
        Slot* s = &obj->nodes.tab[i];
        if (!s->occupied) continue;
        int rem = s->a - s->b;
        if (rem < size) continue;
        /* best-fit：更小剩余优先；并列取更小 nodeId */
        if (!found || rem < bestRem || (rem == bestRem && s->key < bestId)) {
            found = 1;
            bestRem = rem;
            bestId = s->key;
        }
    }
    if (!found) return -1;
    Slot* ns = map_find(&obj->nodes, bestId);
    ns->b += size;
    ns->c += 1;
    Slot* js = map_put(&obj->jobs, jobId);
    js->a = bestId;
    js->b = size;
    return bestId;
}

bool clusterPoolKill(clusterPool* obj, int jobId) {
    Slot* js = map_find(&obj->jobs, jobId);
    if (!js) return false;
    Slot* ns = map_find(&obj->nodes, js->a);
    ns->b -= js->b;
    ns->c -= 1;
    map_erase(&obj->jobs, jobId);
    return true;
}

int clusterPoolUsedOf(clusterPool* obj, int nodeId) {
    Slot* s = map_find(&obj->nodes, nodeId);
    return s ? s->b : -1;
}

int clusterPoolFreeOf(clusterPool* obj, int nodeId) {
    Slot* s = map_find(&obj->nodes, nodeId);
    return s ? (s->a - s->b) : -1;
}

int clusterPoolJobNode(clusterPool* obj, int jobId) {
    Slot* s = map_find(&obj->jobs, jobId);
    return s ? s->a : -1;
}
```
