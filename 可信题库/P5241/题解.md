## 解题思路

单核**可抢占**调度：用事件推进模拟。

- 按到达时间扫描任务；就绪集合用优先队列，键为 `(-priority, index)`（优先级高、下标小者优先）。
- 维护当前占用 CPU 的任务及其剩余时间。
- 时间跳到「下一到达」或「当前任务完成」中较早者：若新到达者更优则抢占；否则继续跑完。

不要按单位时间模拟（到达可达 $10^9$），必须事件跳转。

## 复杂度分析

- 时间：$O(n\log n)$，每个任务进出堆常数次量级。
- 空间：$O(n)$。

## 代码实现

### Python

```python
import heapq
from typing import List


class Solution:
    def finishTimes(
        self, arrival: List[int], duration: List[int], priority: List[int]
    ) -> List[int]:
        n = len(arrival)
        rem = duration[:]
        ans = [-1] * n
        order = sorted(range(n), key=lambda i: (arrival[i], i))
        ready = []  # (-priority, index)
        time = 0
        i = 0
        cur = None

        while i < n or ready or cur is not None:
            # 空闲则跳到下一到达
            if cur is None and not ready:
                if i >= n:
                    break
                time = max(time, arrival[order[i]])

            while i < n and arrival[order[i]] <= time:
                idx = order[i]
                heapq.heappush(ready, (-priority[idx], idx))
                i += 1

            # 更高优先级（或同优更小下标）可抢占
            if cur is not None:
                if ready and (ready[0][0], ready[0][1]) < (-priority[cur], cur):
                    heapq.heappush(ready, (-priority[cur], cur))
                    cur = heapq.heappop(ready)[1]
            else:
                if not ready:
                    continue
                cur = heapq.heappop(ready)[1]

            next_arr = arrival[order[i]] if i < n else None
            finish = time + rem[cur]
            if next_arr is not None and next_arr < finish:
                rem[cur] -= next_arr - time
                time = next_arr
            else:
                time = finish
                ans[cur] = time
                rem[cur] = 0
                cur = None

        return ans
```

### Java

```java
import java.util.*;

class Solution {
    public int[] finishTimes(int[] arrival, int[] duration, int[] priority) {
        int n = arrival.length;
        int[] rem = duration.clone();
        int[] ans = new int[n];
        Arrays.fill(ans, -1);
        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) order[i] = i;
        Arrays.sort(order, (a, b) -> {
            if (arrival[a] != arrival[b]) return Integer.compare(arrival[a], arrival[b]);
            return Integer.compare(a, b);
        });
        // [-priority, index]
        PriorityQueue<int[]> ready = new PriorityQueue<>((x, y) -> {
            if (x[0] != y[0]) return Integer.compare(x[0], y[0]);
            return Integer.compare(x[1], y[1]);
        });
        long time = 0;
        int i = 0;
        int cur = -1;

        while (i < n || !ready.isEmpty() || cur != -1) {
            if (cur == -1 && ready.isEmpty()) {
                if (i >= n) break;
                time = Math.max(time, arrival[order[i]]);
            }
            while (i < n && arrival[order[i]] <= time) {
                int idx = order[i++];
                ready.offer(new int[]{-priority[idx], idx});
            }
            if (cur != -1) {
                if (!ready.isEmpty()) {
                    int[] top = ready.peek();
                    if (top[0] < -priority[cur] || (top[0] == -priority[cur] && top[1] < cur)) {
                        ready.offer(new int[]{-priority[cur], cur});
                        cur = ready.poll()[1];
                    }
                }
            } else {
                if (ready.isEmpty()) continue;
                cur = ready.poll()[1];
            }
            long finish = time + rem[cur];
            if (i < n && arrival[order[i]] < finish) {
                rem[cur] -= (int)(arrival[order[i]] - time);
                time = arrival[order[i]];
            } else {
                time = finish;
                ans[cur] = (int) time;
                rem[cur] = 0;
                cur = -1;
            }
        }
        return ans;
    }
}
```

### C++

```cpp
#include <algorithm>
#include <queue>
#include <vector>
using namespace std;

class Solution {
public:
    vector<int> finishTimes(vector<int>& arrival, vector<int>& duration,
                            vector<int>& priority) {
        int n = (int)arrival.size();
        vector<int> rem = duration, ans(n, -1), order(n);
        for (int i = 0; i < n; i++) order[i] = i;
        sort(order.begin(), order.end(), [&](int a, int b) {
            if (arrival[a] != arrival[b]) return arrival[a] < arrival[b];
            return a < b;
        });
        // min-heap on (-priority, index)
        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> ready;
        long long time = 0;
        int i = 0, cur = -1;
        while (i < n || !ready.empty() || cur != -1) {
            if (cur == -1 && ready.empty()) {
                if (i >= n) break;
                time = max(time, (long long)arrival[order[i]]);
            }
            while (i < n && arrival[order[i]] <= time) {
                int idx = order[i++];
                ready.push({-priority[idx], idx});
            }
            if (cur != -1) {
                if (!ready.empty()) {
                    auto top = ready.top();
                    if (make_pair(top.first, top.second) < make_pair(-priority[cur], cur)) {
                        ready.push({-priority[cur], cur});
                        ready.pop();
                        cur = top.second;
                    }
                }
            } else {
                if (ready.empty()) continue;
                cur = ready.top().second;
                ready.pop();
            }
            long long finish = time + rem[cur];
            if (i < n && arrival[order[i]] < finish) {
                rem[cur] -= (int)(arrival[order[i]] - time);
                time = arrival[order[i]];
            } else {
                time = finish;
                ans[cur] = (int)time;
                rem[cur] = 0;
                cur = -1;
            }
        }
        return ans;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

typedef struct { int neg_pri, idx; } Node;

static void heap_swap(Node* a, Node* b) { Node t = *a; *a = *b; *b = t; }

static int node_less(Node a, Node b) {
    if (a.neg_pri != b.neg_pri) return a.neg_pri < b.neg_pri;
    return a.idx < b.idx;
}

static void heap_push(Node* h, int* sz, Node v) {
    int i = (*sz)++;
    h[i] = v;
    while (i > 0) {
        int p = (i - 1) / 2;
        if (!node_less(h[i], h[p])) break;
        heap_swap(&h[i], &h[p]);
        i = p;
    }
}

static Node heap_pop(Node* h, int* sz) {
    Node top = h[0];
    h[0] = h[--(*sz)];
    int i = 0;
    while (1) {
        int l = i * 2 + 1, r = l + 1, best = i;
        if (l < *sz && node_less(h[l], h[best])) best = l;
        if (r < *sz && node_less(h[r], h[best])) best = r;
        if (best == i) break;
        heap_swap(&h[i], &h[best]);
        i = best;
    }
    return top;
}

typedef struct { int* arr; int* arrival; } SortCtx;
static int cmp_order(const void* a, const void* b, void* ctx) {
    SortCtx* c = (SortCtx*)ctx;
    int ia = *(const int*)a, ib = *(const int*)b;
    if (c->arrival[ia] != c->arrival[ib]) return c->arrival[ia] - c->arrival[ib];
    return ia - ib;
}

/* MSVC/qsort_s 不可用时用包装：把 arrival 放全局临时 */
static int* g_arrival;
static int cmp_order2(const void* a, const void* b) {
    int ia = *(const int*)a, ib = *(const int*)b;
    if (g_arrival[ia] != g_arrival[ib]) return g_arrival[ia] - g_arrival[ib];
    return ia - ib;
}

int* finishTimes(int* arrival, int arrivalSize, int* duration, int durationSize,
                 int* priority, int prioritySize, int* returnSize) {
    (void)durationSize; (void)prioritySize;
    int n = arrivalSize;
    int* rem = (int*)malloc(n * sizeof(int));
    int* ans = (int*)malloc(n * sizeof(int));
    int* order = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) { rem[i] = duration[i]; ans[i] = -1; order[i] = i; }
    g_arrival = arrival;
    qsort(order, n, sizeof(int), cmp_order2);

    Node* ready = (Node*)malloc(n * sizeof(Node));
    int rsz = 0;
    long long time = 0;
    int i = 0, cur = -1;
    while (i < n || rsz > 0 || cur != -1) {
        if (cur == -1 && rsz == 0) {
            if (i >= n) break;
            if (time < arrival[order[i]]) time = arrival[order[i]];
        }
        while (i < n && arrival[order[i]] <= time) {
            int idx = order[i++];
            Node nd; nd.neg_pri = -priority[idx]; nd.idx = idx;
            heap_push(ready, &rsz, nd);
        }
        if (cur != -1) {
            if (rsz > 0) {
                Node top = ready[0];
                Node curN; curN.neg_pri = -priority[cur]; curN.idx = cur;
                if (node_less(top, curN)) {
                    heap_push(ready, &rsz, curN);
                    heap_pop(ready, &rsz);
                    cur = top.idx;
                }
            }
        } else {
            if (rsz == 0) continue;
            cur = heap_pop(ready, &rsz).idx;
        }
        long long finish = time + rem[cur];
        if (i < n && arrival[order[i]] < finish) {
            rem[cur] -= (int)(arrival[order[i]] - time);
            time = arrival[order[i]];
        } else {
            time = finish;
            ans[cur] = (int)time;
            rem[cur] = 0;
            cur = -1;
        }
    }
    free(rem); free(order); free(ready);
    *returnSize = n;
    return ans;
}
```
