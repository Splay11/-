## 解题思路

用**栈**模拟单车道（出口在栈顶），用**双端队列**模拟入口外等待队列，再用**集合**维护车号是否已在车道/队列中。另用历史栈记录每次成功的 $arrive$ / $admit$ / $depart$，以支持 $undo$。

- $arrive$：未满则压栈，否则入队尾；重复车号失败
- $admit$：队首出队并压栈（车道须有空位）
- $depart$：弹出挡路车辆到临时栈，放行目标车后再压回；返回挪开数量
- $undo$：按历史逆操作；失败则把记录压回历史（与题面约定一致）

## 复杂度分析

- 时间复杂度：单次 $depart$ / 中间删除最坏 $O(capacity)$，总操作 $Q\le 10^4$，整体可接受
- 空间复杂度：$O(Q)$

```python
from collections import deque
from typing import Deque, List, Set

class ParkingLane:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.lane: List[int] = []  # 栈顶为出口侧
        self.wait: Deque[int] = deque()
        self.in_lane: Set[int] = set()
        self.in_wait: Set[int] = set()
        self.hist: List[tuple] = []

    def arrive(self, carId: int) -> bool:
        if carId in self.in_lane or carId in self.in_wait:
            return False
        if len(self.lane) < self.cap:
            self.lane.append(carId)
            self.in_lane.add(carId)
            self.hist.append(("arrive", carId, "lane"))
        else:
            self.wait.append(carId)
            self.in_wait.add(carId)
            self.hist.append(("arrive", carId, "wait"))
        return True

    def admit(self) -> bool:
        if not self.wait or len(self.lane) >= self.cap:
            return False
        carId = self.wait.popleft()
        self.in_wait.remove(carId)
        self.lane.append(carId)
        self.in_lane.add(carId)
        self.hist.append(("admit", carId))
        return True

    def depart(self, carId: int) -> int:
        if carId not in self.in_lane:
            return -1
        temp: List[int] = []
        moved = 0
        while self.lane and self.lane[-1] != carId:
            temp.append(self.lane.pop())
            moved += 1
        self.lane.pop()
        self.in_lane.remove(carId)
        while temp:
            self.lane.append(temp.pop())
        self.hist.append(("depart", carId))
        return moved

    def undo(self) -> bool:
        if not self.hist:
            return False
        rec = self.hist.pop()
        kind = rec[0]
        if kind == "arrive":
            carId, where = rec[1], rec[2]
            if where == "lane":
                if not self.lane or self.lane[-1] != carId:
                    self.hist.append(rec)
                    return False
                self.lane.pop()
                self.in_lane.remove(carId)
            else:
                if carId not in self.in_wait:
                    self.hist.append(rec)
                    return False
                self.wait.remove(carId)
                self.in_wait.remove(carId)
            return True
        if kind == "admit":
            carId = rec[1]
            if not self.lane or self.lane[-1] != carId:
                self.hist.append(rec)
                return False
            self.lane.pop()
            self.in_lane.remove(carId)
            self.wait.appendleft(carId)
            self.in_wait.add(carId)
            return True
        if kind == "depart":
            carId = rec[1]
            if carId in self.in_lane or carId in self.in_wait or len(self.lane) >= self.cap:
                self.hist.append(rec)
                return False
            self.lane.append(carId)
            self.in_lane.add(carId)
            return True
        return False

    def front(self) -> int:
        return self.lane[-1] if self.lane else -1

    def waiting(self) -> int:
        return len(self.wait)

    def size(self) -> int:
        return len(self.lane)
```

### Java

```java
import java.util.*;

public class ParkingLane {
    private int cap;
    private Deque<Integer> lane = new ArrayDeque<>(); // 栈：末尾为出口
    private Deque<Integer> wait = new ArrayDeque<>();
    private Set<Integer> inLane = new HashSet<>();
    private Set<Integer> inWait = new HashSet<>();
    private List<int[]> hist = new ArrayList<>(); // {kind, carId, where} kind:0 arrive,1 admit,2 depart; where:0 lane,1 wait

    public ParkingLane(int capacity) { this.cap = capacity; }

    public boolean arrive(int carId) {
        if (inLane.contains(carId) || inWait.contains(carId)) return false;
        if (lane.size() < cap) {
            lane.addLast(carId);
            inLane.add(carId);
            hist.add(new int[]{0, carId, 0});
        } else {
            wait.addLast(carId);
            inWait.add(carId);
            hist.add(new int[]{0, carId, 1});
        }
        return true;
    }

    public boolean admit() {
        if (wait.isEmpty() || lane.size() >= cap) return false;
        int carId = wait.pollFirst();
        inWait.remove(carId);
        lane.addLast(carId);
        inLane.add(carId);
        hist.add(new int[]{1, carId, 0});
        return true;
    }

    public int depart(int carId) {
        if (!inLane.contains(carId)) return -1;
        List<Integer> temp = new ArrayList<>();
        int moved = 0;
        while (!lane.isEmpty() && lane.peekLast() != carId) {
            temp.add(lane.pollLast());
            moved++;
        }
        lane.pollLast();
        inLane.remove(carId);
        for (int i = temp.size() - 1; i >= 0; i--) lane.addLast(temp.get(i));
        hist.add(new int[]{2, carId, 0});
        return moved;
    }

    public boolean undo() {
        if (hist.isEmpty()) return false;
        int[] rec = hist.remove(hist.size() - 1);
        int kind = rec[0], carId = rec[1], where = rec[2];
        if (kind == 0) { // arrive
            if (where == 0) {
                if (lane.isEmpty() || lane.peekLast() != carId) {
                    hist.add(rec);
                    return false;
                }
                lane.pollLast();
                inLane.remove(carId);
            } else {
                if (!inWait.contains(carId)) {
                    hist.add(rec);
                    return false;
                }
                wait.remove(carId);
                inWait.remove(carId);
            }
            return true;
        }
        if (kind == 1) { // admit
            if (lane.isEmpty() || lane.peekLast() != carId) {
                hist.add(rec);
                return false;
            }
            lane.pollLast();
            inLane.remove(carId);
            wait.addFirst(carId);
            inWait.add(carId);
            return true;
        }
        // depart
        if (inLane.contains(carId) || inWait.contains(carId) || lane.size() >= cap) {
            hist.add(rec);
            return false;
        }
        lane.addLast(carId);
        inLane.add(carId);
        return true;
    }

    public int front() { return lane.isEmpty() ? -1 : lane.peekLast(); }
    public int waiting() { return wait.size(); }
    public int size() { return lane.size(); }
}
```

```cpp
class ParkingLane {
    int cap;
    vector<int> lane; // back = exit
    deque<int> wait;
    unordered_set<int> inLane, inWait;
    // hist: kind 0/1/2 = arrive/admit/depart; where 0/1 = lane/wait
    vector<array<int,3>> hist;
public:
    ParkingLane(int capacity) : cap(capacity) {}
    bool arrive(int carId) {
        if (inLane.count(carId) || inWait.count(carId)) return false;
        if ((int)lane.size() < cap) {
            lane.push_back(carId); inLane.insert(carId);
            hist.push_back({0, carId, 0});
        } else {
            wait.push_back(carId); inWait.insert(carId);
            hist.push_back({0, carId, 1});
        }
        return true;
    }
    bool admit() {
        if (wait.empty() || (int)lane.size() >= cap) return false;
        int carId = wait.front(); wait.pop_front(); inWait.erase(carId);
        lane.push_back(carId); inLane.insert(carId);
        hist.push_back({1, carId, 0});
        return true;
    }
    int depart(int carId) {
        if (!inLane.count(carId)) return -1;
        vector<int> temp; int moved = 0;
        while (!lane.empty() && lane.back() != carId) {
            temp.push_back(lane.back()); lane.pop_back(); moved++;
        }
        lane.pop_back(); inLane.erase(carId);
        while (!temp.empty()) { lane.push_back(temp.back()); temp.pop_back(); }
        hist.push_back({2, carId, 0});
        return moved;
    }
    bool undo() {
        if (hist.empty()) return false;
        auto rec = hist.back(); hist.pop_back();
        int kind = rec[0], carId = rec[1], where = rec[2];
        if (kind == 0) {
            if (where == 0) {
                if (lane.empty() || lane.back() != carId) { hist.push_back(rec); return false; }
                lane.pop_back(); inLane.erase(carId);
            } else {
                if (!inWait.count(carId)) { hist.push_back(rec); return false; }
                auto it = find(wait.begin(), wait.end(), carId);
                wait.erase(it); inWait.erase(carId);
            }
            return true;
        }
        if (kind == 1) {
            if (lane.empty() || lane.back() != carId) { hist.push_back(rec); return false; }
            lane.pop_back(); inLane.erase(carId);
            wait.push_front(carId); inWait.insert(carId);
            return true;
        }
        if (inLane.count(carId) || inWait.count(carId) || (int)lane.size() >= cap) {
            hist.push_back(rec); return false;
        }
        lane.push_back(carId); inLane.insert(carId);
        return true;
    }
    int front() { return lane.empty() ? -1 : lane.back(); }
    int waiting() { return (int)wait.size(); }
    int size() { return (int)lane.size(); }
};
```

### JavaScript

```javascript
class ParkingLane {
    constructor(capacity) {
        this.cap = capacity;
        this.lane = []; // 末尾为出口
        this.wait = [];
        this.inLane = new Set();
        this.inWait = new Set();
        this.hist = [];
    }
    arrive(carId) {
        if (this.inLane.has(carId) || this.inWait.has(carId)) return false;
        if (this.lane.length < this.cap) {
            this.lane.push(carId);
            this.inLane.add(carId);
            this.hist.push({ kind: "arrive", carId, where: "lane" });
        } else {
            this.wait.push(carId);
            this.inWait.add(carId);
            this.hist.push({ kind: "arrive", carId, where: "wait" });
        }
        return true;
    }
    admit() {
        if (!this.wait.length || this.lane.length >= this.cap) return false;
        const carId = this.wait.shift();
        this.inWait.delete(carId);
        this.lane.push(carId);
        this.inLane.add(carId);
        this.hist.push({ kind: "admit", carId });
        return true;
    }
    depart(carId) {
        if (!this.inLane.has(carId)) return -1;
        const temp = [];
        let moved = 0;
        while (this.lane.length && this.lane[this.lane.length - 1] !== carId) {
            temp.push(this.lane.pop());
            moved++;
        }
        this.lane.pop();
        this.inLane.delete(carId);
        while (temp.length) this.lane.push(temp.pop());
        this.hist.push({ kind: "depart", carId });
        return moved;
    }
    undo() {
        if (!this.hist.length) return false;
        const rec = this.hist.pop();
        if (rec.kind === "arrive") {
            if (rec.where === "lane") {
                if (!this.lane.length || this.lane[this.lane.length - 1] !== rec.carId) {
                    this.hist.push(rec); return false;
                }
                this.lane.pop();
                this.inLane.delete(rec.carId);
            } else {
                const idx = this.wait.indexOf(rec.carId);
                if (idx < 0) { this.hist.push(rec); return false; }
                this.wait.splice(idx, 1);
                this.inWait.delete(rec.carId);
            }
            return true;
        }
        if (rec.kind === "admit") {
            if (!this.lane.length || this.lane[this.lane.length - 1] !== rec.carId) {
                this.hist.push(rec); return false;
            }
            this.lane.pop();
            this.inLane.delete(rec.carId);
            this.wait.unshift(rec.carId);
            this.inWait.add(rec.carId);
            return true;
        }
        if (this.inLane.has(rec.carId) || this.inWait.has(rec.carId) || this.lane.length >= this.cap) {
            this.hist.push(rec); return false;
        }
        this.lane.push(rec.carId);
        this.inLane.add(rec.carId);
        return true;
    }
    front() { return this.lane.length ? this.lane[this.lane.length - 1] : -1; }
    waiting() { return this.wait.length; }
    size() { return this.lane.length; }
}
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

#define MAXN 10005
#define HCAP 20023

typedef struct {
    int cap;
    int lane[MAXN], laneSz;
    int wait[MAXN], waitSz;
    int histKind[MAXN]; /* 0 arrive_lane, 1 arrive_wait, 2 admit, 3 depart */
    int histCar[MAXN];
    int histSz;
    int hkey[HCAP];
    char hloc[HCAP]; /* 0 empty, 1 lane, 2 wait, 3 tomb */
} ParkingLane;

static int hashProbe(ParkingLane* o, int key, int forInsert) {
    unsigned h = (unsigned)key % HCAP;
    int firstDel = -1;
    for (int t = 0; t < HCAP; t++) {
        int i = (int)((h + (unsigned)t) % HCAP);
        if (o->hloc[i] == 0) return forInsert ? (firstDel >= 0 ? firstDel : i) : -1;
        if (o->hloc[i] == 3) {
            if (firstDel < 0) firstDel = i;
            continue;
        }
        if (o->hkey[i] == key) return i;
    }
    return forInsert ? firstDel : -1;
}

static int getLoc(ParkingLane* o, int key) {
    int i = hashProbe(o, key, 0);
    return i < 0 ? 0 : (int)o->hloc[i];
}

static void setLoc(ParkingLane* o, int key, int loc) {
    if (loc == 0) {
        int i = hashProbe(o, key, 0);
        if (i >= 0) o->hloc[i] = 3;
        return;
    }
    int i = hashProbe(o, key, 1);
    o->hkey[i] = key;
    o->hloc[i] = (char)loc;
}

ParkingLane* parkingLaneCreate(int capacity) {
    ParkingLane* o = (ParkingLane*)calloc(1, sizeof(ParkingLane));
    o->cap = capacity;
    return o;
}

void parkingLaneFree(ParkingLane* obj) { free(obj); }

bool parkingLaneArrive(ParkingLane* obj, int carId) {
    if (getLoc(obj, carId) != 0) return false;
    if (obj->laneSz < obj->cap) {
        obj->lane[obj->laneSz++] = carId;
        setLoc(obj, carId, 1);
        obj->histKind[obj->histSz] = 0;
        obj->histCar[obj->histSz++] = carId;
    } else {
        obj->wait[obj->waitSz++] = carId;
        setLoc(obj, carId, 2);
        obj->histKind[obj->histSz] = 1;
        obj->histCar[obj->histSz++] = carId;
    }
    return true;
}

bool parkingLaneAdmit(ParkingLane* obj) {
    if (obj->waitSz == 0 || obj->laneSz >= obj->cap) return false;
    int carId = obj->wait[0];
    memmove(obj->wait, obj->wait + 1, (size_t)(obj->waitSz - 1) * sizeof(int));
    obj->waitSz--;
    setLoc(obj, carId, 0);
    obj->lane[obj->laneSz++] = carId;
    setLoc(obj, carId, 1);
    obj->histKind[obj->histSz] = 2;
    obj->histCar[obj->histSz++] = carId;
    return true;
}

int parkingLaneDepart(ParkingLane* obj, int carId) {
    if (getLoc(obj, carId) != 1) return -1;
    int temp[MAXN], tempSz = 0, moved = 0;
    while (obj->laneSz > 0 && obj->lane[obj->laneSz - 1] != carId) {
        temp[tempSz++] = obj->lane[--obj->laneSz];
        moved++;
    }
    obj->laneSz--; /* pop target */
    setLoc(obj, carId, 0);
    while (tempSz > 0) obj->lane[obj->laneSz++] = temp[--tempSz];
    obj->histKind[obj->histSz] = 3;
    obj->histCar[obj->histSz++] = carId;
    return moved;
}

bool parkingLaneUndo(ParkingLane* obj) {
    if (obj->histSz == 0) return false;
    int kind = obj->histKind[--obj->histSz];
    int carId = obj->histCar[obj->histSz];
    if (kind == 0) { /* arrive lane */
        if (obj->laneSz == 0 || obj->lane[obj->laneSz - 1] != carId) {
            obj->histSz++;
            return false;
        }
        obj->laneSz--;
        setLoc(obj, carId, 0);
        return true;
    }
    if (kind == 1) { /* arrive wait：从队列中间删掉 */
        int idx = -1;
        for (int i = 0; i < obj->waitSz; i++) if (obj->wait[i] == carId) { idx = i; break; }
        if (idx < 0) { obj->histSz++; return false; }
        memmove(obj->wait + idx, obj->wait + idx + 1, (size_t)(obj->waitSz - idx - 1) * sizeof(int));
        obj->waitSz--;
        setLoc(obj, carId, 0);
        return true;
    }
    if (kind == 2) { /* admit */
        if (obj->laneSz == 0 || obj->lane[obj->laneSz - 1] != carId) {
            obj->histSz++;
            return false;
        }
        obj->laneSz--;
        setLoc(obj, carId, 0);
        memmove(obj->wait + 1, obj->wait, (size_t)obj->waitSz * sizeof(int));
        obj->wait[0] = carId;
        obj->waitSz++;
        setLoc(obj, carId, 2);
        return true;
    }
    /* depart */
    if (getLoc(obj, carId) != 0 || obj->laneSz >= obj->cap) {
        obj->histSz++;
        return false;
    }
    obj->lane[obj->laneSz++] = carId;
    setLoc(obj, carId, 1);
    return true;
}

int parkingLaneFront(ParkingLane* obj) {
    return obj->laneSz ? obj->lane[obj->laneSz - 1] : -1;
}
int parkingLaneWaiting(ParkingLane* obj) { return obj->waitSz; }
int parkingLaneSize(ParkingLane* obj) { return obj->laneSz; }
```
