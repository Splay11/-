## 解题思路

任务依赖构成有向图。无环时，最短完工时间等于 **DAG 上按依赖约束的最长路径**（并行时取前置最大值）：

$$
\textit{finish}[v] = \textit{time}[v] + \max_{u\to v}\textit{finish}[u]
$$

（无前置时即为 `time[v]`）。答案取所有 `finish` 的最大值。

用**拓扑排序（Kahn）**推进：入度为 0 的点入队；处理边时更新后继完工时间。若最终处理点数 `< n`（含自环/重边导致的环），返回 `-1`。

接口 `minFinishTime(n, prev, next, time)` 中第 $i$ 条边为 `prev[i] → next[i]`。

易错点：把答案算成总和而非最长路；有环仍返回正数；自环未识别；并行理解成串行。

## 复杂度分析

- 时间：$O(n+m)$，拓扑一遍。
- 空间：$O(n+m)$，邻接表与入度。

## 代码实现

### Python

```python
from collections import deque
from typing import List


class Solution:
    def minFinishTime(self, n: int, prev: List[int], next: List[int], time: List[int]) -> int:
        g = [[] for _ in range(n)]
        indeg = [0] * n
        for u, v in zip(prev, next):
            g[u].append(v)
            indeg[v] += 1
        finish = [0] * n
        q = deque()
        for i in range(n):
            if indeg[i] == 0:
                finish[i] = time[i]
                q.append(i)
        seen = 0
        while q:
            u = q.popleft()
            seen += 1
            for v in g[u]:
                # 前置最大完工 + 自身耗时
                if finish[u] + time[v] > finish[v]:
                    finish[v] = finish[u] + time[v]
                indeg[v] -= 1
                if indeg[v] == 0:
                    q.append(v)
        if seen != n:
            return -1  # 有环
        return max(finish)
```

### Java

```java
import java.util.*;

class Solution {
    public int minFinishTime(int n, int[] prev, int[] next, int[] time) {
        List<List<Integer>> g = new ArrayList<>();
        for (int i = 0; i < n; i++) g.add(new ArrayList<>());
        int[] indeg = new int[n];
        int[] finish = new int[n];
        for (int i = 0; i < prev.length; i++) {
            g.get(prev[i]).add(next[i]);
            indeg[next[i]]++;
        }
        Queue<Integer> q = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            if (indeg[i] == 0) {
                finish[i] = time[i];
                q.offer(i);
            }
        }
        int seen = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            seen++;
            for (int v : g.get(u)) {
                // 前置最大完工 + 自身耗时
                finish[v] = Math.max(finish[v], finish[u] + time[v]);
                if (--indeg[v] == 0) q.offer(v);
            }
        }
        if (seen != n) return -1; // 有环
        int ans = 0;
        for (int x : finish) ans = Math.max(ans, x);
        return ans;
    }
}
```

### C++

```cpp
#include <algorithm>
#include <queue>
#include <vector>
using namespace std;

class Solution {
public:
    int minFinishTime(int n, vector<int>& prev, vector<int>& next, vector<int>& time) {
        vector<vector<int>> g(n);
        vector<int> indeg(n, 0), finish(n, 0);
        for (size_t i = 0; i < prev.size(); i++) {
            g[prev[i]].push_back(next[i]);
            indeg[next[i]]++;
        }
        queue<int> q;
        for (int i = 0; i < n; i++) {
            if (indeg[i] == 0) {
                finish[i] = time[i];
                q.push(i);
            }
        }
        int seen = 0;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            seen++;
            for (int v : g[u]) {
                // 前置最大完工 + 自身耗时
                finish[v] = max(finish[v], finish[u] + time[v]);
                if (--indeg[v] == 0) q.push(v);
            }
        }
        if (seen != n) return -1; // 有环
        return *max_element(finish.begin(), finish.end());
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

int minFinishTime(int n, int* prev, int prevSize, int* next, int nextSize, int* time, int timeSize) {
    (void)nextSize;
    (void)timeSize;
    int* head = (int*)malloc(sizeof(int) * (size_t)n);
    int* to = (int*)malloc(sizeof(int) * (size_t)(prevSize + 1));
    int* enxt = (int*)malloc(sizeof(int) * (size_t)(prevSize + 1));
    int* indeg = (int*)calloc((size_t)n, sizeof(int));
    int* finish = (int*)calloc((size_t)n, sizeof(int));
    int* q = (int*)malloc(sizeof(int) * (size_t)n);
    for (int i = 0; i < n; i++) head[i] = -1;
    int ec = 0;
    for (int i = 0; i < prevSize; i++) {
        int u = prev[i], v = next[i];
        to[ec] = v;
        enxt[ec] = head[u];
        head[u] = ec++;
        indeg[v]++;
    }
    int qh = 0, qt = 0, seen = 0;
    for (int i = 0; i < n; i++) {
        if (indeg[i] == 0) {
            finish[i] = time[i];
            q[qt++] = i;
        }
    }
    while (qh < qt) {
        int u = q[qh++];
        seen++;
        for (int e = head[u]; e != -1; e = enxt[e]) {
            int v = to[e];
            // 前置最大完工 + 自身耗时
            if (finish[u] + time[v] > finish[v]) finish[v] = finish[u] + time[v];
            if (--indeg[v] == 0) q[qt++] = v;
        }
    }
    int ans = -1;
    if (seen == n) {
        ans = 0;
        for (int i = 0; i < n; i++)
            if (finish[i] > ans) ans = finish[i];
    }
    free(head); free(to); free(enxt); free(indeg); free(finish); free(q);
    return ans;
}
```
