## 解题思路

这是带**变更次数上限**的多天算力规划 **DP**。

第 $i$ 天实例数 $x$ 须满足 $x \ge \texttt{load}[i]$。相对前一天 $p$（第 $0$ 天前 $p=0$）的费用为
$$
x\cdot \textit{runCost} + |x-p|\cdot \textit{changeCost}
$$
且整个规划中 $x\neq p$ 的天数至多为 `maxChanges`。

**状态**：`dp[c][p]` 表示已用 $c$ 次变更、当前实例数为 $p$ 的最小费用。初始 `dp[0][0]=0`。

**转移**：对第 $i$ 天，枚举旧状态 $(c,p)$ 与合法 $x\in[\texttt{load}[i], M]$（$M=\max load$），令
$$
c' = c + [x\neq p],\quad
\textit{cost}' = \textit{cost} + x\cdot\textit{runCost} + |x-p|\cdot\textit{changeCost}
$$
若 $c'\le \texttt{maxChanges}$ 则更新 `dp'[c'][x]`。

取所有合法终态最小值。$x$ 不必超过 $M$：更高只会增加运行费且不降低后续需求。

常见假解：忽略 `maxChanges`、把「变更次数」算成 $|x-p|$ 的总量、第 $0$ 天忘记相对 $0$ 计一次变更、或错误认为必须每天恰好 `load[i]`。

## 复杂度分析

- 时间：$O(n\cdot \textit{maxChanges}\cdot M^2)$，$n\le 80$，$M\le 40$，约 $10^7$ 量级。
- 空间：$O(\textit{maxChanges}\cdot M)$（滚动两天）。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def minComputeCost(
        self, load: List[int], runCost: int, changeCost: int, maxChanges: int
    ) -> int:
        n = len(load)
        M = max(load) if load else 0
        INF = 10**18
        # prev[c][p]：已用 c 次变更、当前实例 p
        prev = [{} for _ in range(maxChanges + 1)]
        prev[0][0] = 0
        for i in range(n):
            cur = [{} for _ in range(maxChanges + 1)]
            for c in range(maxChanges + 1):
                for p, cost in prev[c].items():
                    for x in range(load[i], M + 1):
                        nc = c + (0 if x == p else 1)  # x!=p 才消耗一次变更
                        if nc > maxChanges:
                            continue
                        nc_cost = cost + x * runCost + abs(x - p) * changeCost
                        if x not in cur[nc] or nc_cost < cur[nc][x]:
                            cur[nc][x] = nc_cost
            prev = cur
        ans = INF
        for c in range(maxChanges + 1):
            if prev[c]:
                ans = min(ans, min(prev[c].values()))
        return ans
```

### Java

```java
import java.util.*;

class Solution {
    public int minComputeCost(int[] load, int runCost, int changeCost, int maxChanges) {
        int n = load.length;
        int M = 0;
        for (int v : load) M = Math.max(M, v);
        final long INF = (long) 1e18;
        // dp[c]：Map<p, minCost>
        Map<Integer, Long>[] prev = new Map[maxChanges + 1];
        for (int c = 0; c <= maxChanges; c++) prev[c] = new HashMap<>();
        prev[0].put(0, 0L);

        for (int i = 0; i < n; i++) {
            Map<Integer, Long>[] cur = new Map[maxChanges + 1];
            for (int c = 0; c <= maxChanges; c++) cur[c] = new HashMap<>();
            for (int c = 0; c <= maxChanges; c++) {
                for (Map.Entry<Integer, Long> e : prev[c].entrySet()) {
                    int p = e.getKey();
                    long cost = e.getValue();
                    for (int x = load[i]; x <= M; x++) {
                        int nc = c + (x == p ? 0 : 1); // 变更天数计数
                        if (nc > maxChanges) continue;
                        long ncCost = cost + (long) x * runCost + (long) Math.abs(x - p) * changeCost;
                        Long old = cur[nc].get(x);
                        if (old == null || ncCost < old) cur[nc].put(x, ncCost);
                    }
                }
            }
            prev = cur;
        }
        long ans = INF;
        for (int c = 0; c <= maxChanges; c++) {
            for (long v : prev[c].values()) ans = Math.min(ans, v);
        }
        return (int) ans;
    }
}
```

### C++

```cpp
#include <vector>
#include <unordered_map>
#include <algorithm>
#include <climits>
using namespace std;

class Solution {
public:
    int minComputeCost(vector<int>& load, int runCost, int changeCost, int maxChanges) {
        int n = (int)load.size();
        int M = 0;
        for (int v : load) M = max(M, v);
        const long long INF = (long long)1e18;
        // prev[c][p] = min cost
        vector<unordered_map<int, long long>> prev(maxChanges + 1);
        prev[0][0] = 0;
        for (int i = 0; i < n; i++) {
            vector<unordered_map<int, long long>> cur(maxChanges + 1);
            for (int c = 0; c <= maxChanges; c++) {
                for (auto& [p, cost] : prev[c]) {
                    for (int x = load[i]; x <= M; x++) {
                        int nc = c + (x == p ? 0 : 1); // 仅实例变化计一次
                        if (nc > maxChanges) continue;
                        long long ncCost = cost + 1LL * x * runCost + 1LL * abs(x - p) * changeCost;
                        auto it = cur[nc].find(x);
                        if (it == cur[nc].end() || ncCost < it->second) cur[nc][x] = ncCost;
                    }
                }
            }
            prev.swap(cur);
        }
        long long ans = INF;
        for (int c = 0; c <= maxChanges; c++)
            for (auto& [_, v] : prev[c]) ans = min(ans, v);
        return (int)ans;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

#define INF ((long long)1e18)

int minComputeCost(int* load, int loadSize, int runCost, int changeCost, int maxChanges) {
    int n = loadSize;
    int M = 0;
    for (int i = 0; i < n; i++) if (load[i] > M) M = load[i];
    /* prev[c][p]：用到 c 次变更、前一天实例数为 p 的最小费用；-1 表示不可达 */
    long long* prev = (long long*)malloc((maxChanges + 1) * 41 * sizeof(long long));
    long long* cur = (long long*)malloc((maxChanges + 1) * 41 * sizeof(long long));
    for (int i = 0; i < (maxChanges + 1) * 41; i++) prev[i] = -1;
    prev[0 * 41 + 0] = 0; /* 第 -1 天实例为 0 */
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < (maxChanges + 1) * 41; j++) cur[j] = -1;
        for (int c = 0; c <= maxChanges; c++) {
            for (int p = 0; p <= M; p++) {
                long long cost = prev[c * 41 + p];
                if (cost < 0) continue;
                for (int x = load[i]; x <= M; x++) {
                    int nc = c + (x == p ? 0 : 1); /* 相对前一天发生变化计 1 次 */
                    if (nc > maxChanges) continue;
                    long long ncCost = cost + 1LL * x * runCost + 1LL * (x > p ? x - p : p - x) * changeCost;
                    int idx = nc * 41 + x;
                    if (cur[idx] < 0 || ncCost < cur[idx]) cur[idx] = ncCost;
                }
            }
        }
        long long* tmp = prev; prev = cur; cur = tmp;
    }
    long long ans = INF;
    for (int c = 0; c <= maxChanges; c++)
        for (int p = 0; p <= M; p++) {
            long long v = prev[c * 41 + p];
            if (v >= 0 && v < ans) ans = v;
        }
    free(prev); free(cur);
    return (int)ans;
}
```
