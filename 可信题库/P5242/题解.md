## 解题思路

用数组维护每个车位上的占用区间列表，再用哈希表记录 `carId -> (spot, start, end)`。

- `reserve`：若该车已有预约或 `start >= end` 则失败；否则按车位编号从小到大，找第一个与 `[start,end)` **半开不相交**（`start < e && s < end` 为相交）的车位分配。
- `cancel` / `spotOf`：查表删除或返回车位。
- `busyCount`：统计满足 `start <= time < end` 的预约数。

调用量 $\le 3000$、$n\le 200$，直接扫描区间即可。

## 复杂度分析

- 单次 `reserve`：$O(n\cdot k)$，$k$ 为单车位区间数；总调用可控。
- 空间：$O(\text{预约数})$。

## 代码实现

### Python

```python
from typing import Dict, List, Tuple


class ParkingLot:
    def __init__(self, n: int):
        self.n = n
        self.spots: List[List[Tuple[int, int]]] = [[] for _ in range(n)]
        self.car: Dict[int, Tuple[int, int, int]] = {}

    def _conflict(self, intervals: List[Tuple[int, int]], start: int, end: int) -> bool:
        # 半开区间相交判定
        for s, e in intervals:
            if start < e and s < end:
                return True
        return False

    def reserve(self, carId: int, start: int, end: int) -> int:
        if carId in self.car or start >= end:
            return -1
        for spot in range(self.n):
            if not self._conflict(self.spots[spot], start, end):
                self.spots[spot].append((start, end))
                self.car[carId] = (spot, start, end)
                return spot
        return -1

    def cancel(self, carId: int) -> bool:
        if carId not in self.car:
            return False
        spot, start, end = self.car.pop(carId)
        self.spots[spot].remove((start, end))
        return True

    def spotOf(self, carId: int) -> int:
        if carId not in self.car:
            return -1
        return self.car[carId][0]

    def busyCount(self, time: int) -> int:
        cnt = 0
        for _, start, end in self.car.values():
            if start <= time < end:
                cnt += 1
        return cnt
```

### Java

```java
import java.util.*;

class ParkingLot {
    private final int n;
    private final List<List<int[]>> spots;
    private final Map<Integer, int[]> car; // carId -> {spot,start,end}

    public ParkingLot(int n) {
        this.n = n;
        spots = new ArrayList<>();
        for (int i = 0; i < n; i++) spots.add(new ArrayList<>());
        car = new HashMap<>();
    }

    private boolean conflict(List<int[]> intervals, int start, int end) {
        for (int[] it : intervals) {
            // 半开相交
            if (start < it[1] && it[0] < end) return true;
        }
        return false;
    }

    public int reserve(int carId, int start, int end) {
        if (car.containsKey(carId) || start >= end) return -1;
        for (int spot = 0; spot < n; spot++) {
            if (!conflict(spots.get(spot), start, end)) {
                spots.get(spot).add(new int[]{start, end});
                car.put(carId, new int[]{spot, start, end});
                return spot;
            }
        }
        return -1;
    }

    public boolean cancel(int carId) {
        int[] info = car.remove(carId);
        if (info == null) return false;
        List<int[]> vec = spots.get(info[0]);
        for (int i = 0; i < vec.size(); i++) {
            if (vec.get(i)[0] == info[1] && vec.get(i)[1] == info[2]) {
                vec.remove(i);
                break;
            }
        }
        return true;
    }

    public int spotOf(int carId) {
        int[] info = car.get(carId);
        return info == null ? -1 : info[0];
    }

    public int busyCount(int time) {
        int cnt = 0;
        for (int[] info : car.values()) {
            if (info[1] <= time && time < info[2]) cnt++;
        }
        return cnt;
    }
}
```

### C++

```cpp
#include <cstddef>
#include <tuple>
#include <unordered_map>
#include <utility>
#include <vector>
using namespace std;

class ParkingLot {
    int n;
    vector<vector<pair<int, int> > > spots;
    unordered_map<int, tuple<int, int, int> > car;

    bool conflict(const vector<pair<int, int> >& intervals, int start, int end) {
        for (size_t i = 0; i < intervals.size(); i++) {
            int s = intervals[i].first, e = intervals[i].second;
            if (start < e && s < end) return true; // 半开相交
        }
        return false;
    }

public:
    ParkingLot(int n_) : n(n_), spots(n_) {}

    int reserve(int carId, int start, int end) {
        if (car.count(carId) || start >= end) return -1;
        for (int spot = 0; spot < n; spot++) {
            if (!conflict(spots[spot], start, end)) {
                spots[spot].push_back(make_pair(start, end));
                car[carId] = make_tuple(spot, start, end);
                return spot;
            }
        }
        return -1;
    }

    bool cancel(int carId) {
        auto it = car.find(carId);
        if (it == car.end()) return false;
        int spot = get<0>(it->second), start = get<1>(it->second), end = get<2>(it->second);
        car.erase(it);
        auto& vec = spots[spot];
        for (size_t i = 0; i < vec.size(); i++) {
            if (vec[i].first == start && vec[i].second == end) {
                vec.erase(vec.begin() + (ptrdiff_t)i);
                break;
            }
        }
        return true;
    }

    int spotOf(int carId) {
        auto it = car.find(carId);
        return it == car.end() ? -1 : get<0>(it->second);
    }

    int busyCount(int time) {
        int cnt = 0;
        for (auto it = car.begin(); it != car.end(); ++it) {
            int start = get<1>(it->second), end = get<2>(it->second);
            if (start <= time && time < end) cnt++;
        }
        return cnt;
    }
};
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

typedef struct ParkingLot ParkingLot;

typedef struct {
    int start, end;
} Interval;

struct ParkingLot {
    int n;
    Interval** spots;
    int* spotCnt;
    int* spotCap;
    int* carSpot;
    int* carStart;
    int* carEnd;
    char* hasCar;
};

static int conflict(Interval* intervals, int cnt, int start, int end) {
    for (int i = 0; i < cnt; i++) {
        if (start < intervals[i].end && intervals[i].start < end) return 1;
    }
    return 0;
}

ParkingLot* parkingLotCreate(int n) {
    ParkingLot* obj = (ParkingLot*)calloc(1, sizeof(ParkingLot));
    obj->n = n;
    obj->spots = (Interval**)calloc((size_t)n, sizeof(Interval*));
    obj->spotCnt = (int*)calloc((size_t)n, sizeof(int));
    obj->spotCap = (int*)calloc((size_t)n, sizeof(int));
    // carId in [1, 10000]
    obj->carSpot = (int*)malloc(sizeof(int) * 10001);
    obj->carStart = (int*)malloc(sizeof(int) * 10001);
    obj->carEnd = (int*)malloc(sizeof(int) * 10001);
    obj->hasCar = (char*)calloc(10001, 1);
    return obj;
}

void parkingLotFree(ParkingLot* obj) {
    if (!obj) return;
    for (int i = 0; i < obj->n; i++) free(obj->spots[i]);
    free(obj->spots);
    free(obj->spotCnt);
    free(obj->spotCap);
    free(obj->carSpot);
    free(obj->carStart);
    free(obj->carEnd);
    free(obj->hasCar);
    free(obj);
}

int parkingLotReserve(ParkingLot* obj, int carId, int start, int end) {
    if (obj->hasCar[carId] || start >= end) return -1;
    for (int spot = 0; spot < obj->n; spot++) {
        if (!conflict(obj->spots[spot], obj->spotCnt[spot], start, end)) {
            if (obj->spotCnt[spot] == obj->spotCap[spot]) {
                int nc = obj->spotCap[spot] ? obj->spotCap[spot] * 2 : 4;
                obj->spots[spot] = (Interval*)realloc(obj->spots[spot], sizeof(Interval) * (size_t)nc);
                obj->spotCap[spot] = nc;
            }
            obj->spots[spot][obj->spotCnt[spot]].start = start;
            obj->spots[spot][obj->spotCnt[spot]].end = end;
            obj->spotCnt[spot]++;
            obj->hasCar[carId] = 1;
            obj->carSpot[carId] = spot;
            obj->carStart[carId] = start;
            obj->carEnd[carId] = end;
            return spot;
        }
    }
    return -1;
}

bool parkingLotCancel(ParkingLot* obj, int carId) {
    if (!obj->hasCar[carId]) return false;
    int spot = obj->carSpot[carId], start = obj->carStart[carId], end = obj->carEnd[carId];
    obj->hasCar[carId] = 0;
    Interval* vec = obj->spots[spot];
    for (int i = 0; i < obj->spotCnt[spot]; i++) {
        if (vec[i].start == start && vec[i].end == end) {
            for (int j = i + 1; j < obj->spotCnt[spot]; j++) vec[j - 1] = vec[j];
            obj->spotCnt[spot]--;
            break;
        }
    }
    return true;
}

int parkingLotSpotOf(ParkingLot* obj, int carId) {
    return obj->hasCar[carId] ? obj->carSpot[carId] : -1;
}

int parkingLotBusyCount(ParkingLot* obj, int time) {
    int cnt = 0;
    for (int id = 1; id <= 10000; id++) {
        if (!obj->hasCar[id]) continue;
        if (obj->carStart[id] <= time && time < obj->carEnd[id]) cnt++;
    }
    return cnt;
}
```
