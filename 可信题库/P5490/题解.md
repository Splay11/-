## 解题思路

本题用长度为 $n$ 的数组模拟格口状态，下标 $0$ 对应 $1$ 号格。数组值 $0$ 表示空，正整数表示餐的重量。

各接口都先做边界检查，再改状态：

- $put$：编号合法、重量为正、该格为空才写入。
- $take$：合法且非空才清空并返回原重量，否则返回 $0$。
- $moveRight$：源格非空且不是最右格、右侧为空才搬家。
- $occupied$：统计非 $0$ 的格数。也可以在 $put$/$take$/$moveRight$ 时维护计数，避免每次扫描。

失败路径必须**不改动**已有状态，这是入门接口题最常见的扣分点。

## 复杂度分析

- 单次 $put$、$take$、$moveRight$：$O(1)$。
- 单次 $occupied$：扫描为 $O(n)$；若维护计数则为 $O(1)$。
- 累计调用次数记为 $q$，整体时间不超过 $O(q n)$，在 $n \le 200$、$q \le 2000$ 下足够。
- 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
class ParcelSlots:
    def __init__(self, n: int):
        self.n = n
        # 0 表示空格
        self.slot = [0] * n
        self.cnt = 0

    def put(self, i: int, w: int) -> bool:
        # 越界、非正重量、已占用都不改动
        if i < 1 or i > self.n or w <= 0 or self.slot[i - 1] != 0:
            return False
        self.slot[i - 1] = w
        self.cnt += 1
        return True

    def take(self, i: int) -> int:
        if i < 1 or i > self.n or self.slot[i - 1] == 0:
            return 0
        w = self.slot[i - 1]
        self.slot[i - 1] = 0
        self.cnt -= 1
        return w

    def moveRight(self, i: int) -> bool:
        # 最右格、源空、右侧占用都不能挪
        if i < 1 or i >= self.n or self.slot[i - 1] == 0 or self.slot[i] != 0:
            return False
        self.slot[i] = self.slot[i - 1]
        self.slot[i - 1] = 0
        return True

    def occupied(self) -> int:
        return self.cnt
```

        cnt++;

class ParcelSlots {
    constructor(n) {
        this.n = n;
        this.slot = Array(n).fill(0);
        this.cnt = 0;
    }

    put(i, w) {
        // 越界、非正重量、已占用都不改动
        if (i < 1 || i > this.n || w <= 0 || this.slot[i - 1] !== 0) {
            return false;
        }
        this.slot[i - 1] = w;
        this.cnt++;
        return true;
    }

    take(i) {
        if (i < 1 || i > this.n || this.slot[i - 1] === 0) {
            return 0;
        }
        const w = this.slot[i - 1];
        this.slot[i - 1] = 0;
        this.cnt--;
        return w;
    }

    moveRight(i) {
        // 最右格、源空、右侧占用都不能挪
        if (i < 1 || i >= this.n || this.slot[i - 1] === 0 || this.slot[i] !== 0) {
            return false;
        }
        this.slot[i] = this.slot[i - 1];
        this.slot[i - 1] = 0;
        return true;
    }

    occupied() {
        return this.cnt;
    }
}

#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int n;
    int* slot;
    int cnt;

parcelSlots* parcelSlotsCreate(int n) {
    parcelSlots* obj = (parcelSlots*)calloc(1, sizeof(parcelSlots));
    obj->n = n;
    obj->slot = (int*)calloc(n, sizeof(int));
    obj->cnt = 0;
    return obj;
}

void parcelSlotsFree(parcelSlots* obj) {
    if (!obj) return;
    free(obj->slot);
    free(obj);
}

bool parcelSlotsPut(parcelSlots* obj, int i, int w) {

    obj->slot[i - 1] = w;
    obj->cnt++;
    return true;
}

int parcelSlotsTake(parcelSlots* obj, int i) {
