## 解题思路

对每种 `alarmType` 独立做**滑动窗口**限流：按时间顺序扫描告警，用队列保存该类型**已被保留**的时间戳。

处理时刻 $t$ 的告警时：

1. 弹出不在半开区间 $(t-\textit{window},\, t]$ 内的历史保留（`window=0` 时同刻已保留仍计入，故只弹出严格早于 $t$ 且 $\le t-\textit{window}$ 的项）。
2. 若队列长度已达 `limit`，抑制；否则入队并计入保留。

关键点：限流按类型隔离；统计对象是**已保留**而非原始上报；`window=0` 时同刻仍互相挤占额度。

## 复杂度分析

- 时间：$O(n)$，每条告警至多入队、出队一次。
- 空间：$O(n)$，最坏全部保留。

## 代码实现

### Python

```python
from collections import defaultdict, deque
from typing import List


class Solution:
    def countKeptAlarms(self, events: List[List[int]], window: int, limit: int) -> int:
        # 每种类型一条单调时间队列
        dq = defaultdict(deque)
        kept = 0
        for t, typ in events:
            q = dq[typ]
            # 维护窗口 (t-window, t]；window=0 时保留同刻
            while q and q[0] <= t - window and q[0] < t:
                q.popleft()
            if len(q) < limit:
                q.append(t)
                kept += 1
        return kept
```

### Java

```java
import java.util.*;

class Solution {
    public int countKeptAlarms(int[][] events, int window, int limit) {
        Map<Integer, ArrayDeque<Integer>> dq = new HashMap<>();
        int kept = 0;
        for (int[] e : events) {
            int t = e[0], typ = e[1];
            ArrayDeque<Integer> q = dq.computeIfAbsent(typ, k -> new ArrayDeque<>());
            // 弹出窗口外的已保留时间戳
            while (!q.isEmpty() && q.peekFirst() <= t - window && q.peekFirst() < t) {
                q.pollFirst();
            }
            if (q.size() < limit) {
                q.addLast(t);
                kept++;
            }
        }
        return kept;
    }
}
```

### C++

```cpp
#include <deque>
#include <unordered_map>
#include <vector>
using namespace std;

class Solution {
public:
    int countKeptAlarms(vector<vector<int>>& events, int window, int limit) {
        unordered_map<int, deque<int>> dq;
        int kept = 0;
        for (auto& e : events) {
            int t = e[0], typ = e[1];
            auto& q = dq[typ];
            // 按类型维护 (t-window, t] 内已保留
            while (!q.empty() && q.front() <= t - window && q.front() < t) {
                q.pop_front();
            }
            if ((int)q.size() < limit) {
                q.push_back(t);
                kept++;
            }
        }
        return kept;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

/* 按 alarmType 哈希到动态双端队列（数组 + 头指针） */
typedef struct {
    int* data;
    int head, size, cap;
} Deque;

typedef struct {
    int* keys;
    Deque* vals;
    char* used;
    int cap;
} TypeMap;

static TypeMap* tm_create(int n) {
    TypeMap* m = (TypeMap*)malloc(sizeof(TypeMap));
    m->cap = 1;
    while (m->cap < n * 2 + 16) m->cap <<= 1;
    m->keys = (int*)malloc(m->cap * sizeof(int));
    m->vals = (Deque*)calloc(m->cap, sizeof(Deque));
    m->used = (char*)calloc(m->cap, 1);
    return m;
}

static Deque* tm_get(TypeMap* m, int typ) {
    unsigned int idx = (unsigned int)typ * 2654435761u;
    idx &= (unsigned int)(m->cap - 1);
    while (m->used[idx]) {
        if (m->keys[idx] == typ) return &m->vals[idx];
        idx = (idx + 1) & (unsigned int)(m->cap - 1);
    }
    m->used[idx] = 1;
    m->keys[idx] = typ;
    m->vals[idx].data = NULL;
    m->vals[idx].head = m->vals[idx].size = m->vals[idx].cap = 0;
    return &m->vals[idx];
}

static void dq_push(Deque* q, int t) {
    /* head 前移后可能 size < cap 但 head+size 越界，需先压紧 */
    if (q->size >= q->cap || (q->cap > 0 && q->head + q->size >= q->cap)) {
        int ncap = q->cap ? q->cap * 2 : 4;
        if (q->size >= ncap) ncap = q->size * 2 + 4;
        int* nd = (int*)malloc(ncap * sizeof(int));
        for (int i = 0; i < q->size; i++) nd[i] = q->data[q->head + i];
        free(q->data);
        q->data = nd; q->head = 0; q->cap = ncap;
    }
    q->data[q->head + q->size] = t;
    q->size++;
}

static void tm_free(TypeMap* m) {
    for (int i = 0; i < m->cap; i++) if (m->used[i]) free(m->vals[i].data);
    free(m->keys); free(m->vals); free(m->used); free(m);
}

int countKeptAlarms(int** events, int eventsSize, int* eventsColSize, int window, int limit) {
    (void)eventsColSize;
    TypeMap* dq = tm_create(eventsSize > 0 ? eventsSize : 1);
    int kept = 0;
    for (int i = 0; i < eventsSize; i++) {
        int t = events[i][0], typ = events[i][1];
        Deque* q = tm_get(dq, typ);
        // 滑动窗口维护 (t-window, t] 内已保留
        while (q->size > 0 && q->data[q->head] <= t - window && q->data[q->head] < t) {
            q->head++;
            q->size--;
        }
        if (q->size < limit) {
            dq_push(q, t);
            kept++;
        }
    }
    tm_free(dq);
    return kept;
}
```
