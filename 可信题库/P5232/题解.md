## 解题思路

本题是**数组上的内存分配模拟**：用长度为 `num` 的数组记录每页归属进程（`-1` 表示空闲），并用哈希表维护 `processId -> (start, size)`。

- **分配**：从左到右找第一段长度 $\ge size$ 的连续空闲（首次适应）。找不到则触发整理后再找；仍失败返回 `-1`。
- **释放**：把该进程占据的页全部置空闲。
- **查询**：直接返回表中起始下标。
- **整理**：收集存活进程，按 `(size, processId)` 升序排序，从下标 `num - totalUsed` 起依次紧密写回，使空闲集中在左侧。注意：即使总空闲 `< size` 也必须整理（样例2）。

**常见假解**：整理时按原地址顺序或按进程号单独排序；整理靠左而非靠右；总空闲不足时跳过整理；分配时用最差适应/最佳适应。

## 复杂度分析

- 单次操作：$O(\textit{num}\log \textit{num})$ 量级（整理时排序），`num\le 256`，调用 $\le 1000$，充足。
- 空间：$O(\textit{num})$。

## 代码实现

### Python

```python
from typing import Dict, List, Tuple


class MemMgmtSys:
    def __init__(self, num: int):
        self.n = num
        self.page: List[int] = [-1] * num  # 每页归属，-1 空闲
        self.info: Dict[int, Tuple[int, int]] = {}

    def _find(self, size: int) -> int:
        i = 0
        while i < self.n:
            if self.page[i] != -1:
                i += 1
                continue
            j = i
            while j < self.n and self.page[j] == -1:
                j += 1
            # 首次适应：第一段够长的空闲
            if j - i >= size:
                return i
            i = j
        return -1

    def _defrag(self) -> None:
        items = [(sz, pid) for pid, (_st, sz) in self.info.items()]
        items.sort()  # 页数升序，再 processId 升序
        self.page = [-1] * self.n
        self.info.clear()
        used = sum(sz for sz, _ in items)
        pos = self.n - used  # 靠右紧排
        for sz, pid in items:
            for k in range(sz):
                self.page[pos + k] = pid
            self.info[pid] = (pos, sz)
            pos += sz

    def processMemAlloc(self, processId: int, size: int) -> int:
        st = self._find(size)
        if st < 0:
            self._defrag()
            st = self._find(size)
            if st < 0:
                return -1
        for k in range(size):
            self.page[st + k] = processId
        self.info[processId] = (st, size)
        return st

    def processMemFree(self, processId: int) -> None:
        st, sz = self.info.pop(processId)
        for k in range(sz):
            self.page[st + k] = -1

    def processMemQuery(self, processId: int) -> int:
        return self.info[processId][0]
```

### Java

```java
import java.util.*;

class MemMgmtSys {
    private int n;
    private int[] page;
    private Map<Integer, int[]> info = new HashMap<>(); // pid -> {start,size}

    public MemMgmtSys(int num) {
        n = num;
        page = new int[num];
        Arrays.fill(page, -1);
    }

    private int findSlot(int size) {
        int i = 0;
        while (i < n) {
            if (page[i] != -1) { i++; continue; }
            int j = i;
            while (j < n && page[j] == -1) j++;
            if (j - i >= size) return i; // 首次适应
            i = j;
        }
        return -1;
    }

    private void defrag() {
        List<int[]> items = new ArrayList<>(); // {size, pid}
        for (Map.Entry<Integer, int[]> e : info.entrySet())
            items.add(new int[]{e.getValue()[1], e.getKey()});
        items.sort((a, b) -> a[0] != b[0] ? a[0] - b[0] : a[1] - b[1]);
        Arrays.fill(page, -1);
        info.clear();
        int used = 0;
        for (int[] it : items) used += it[0];
        int pos = n - used; // 靠右
        for (int[] it : items) {
            int sz = it[0], pid = it[1];
            for (int k = 0; k < sz; k++) page[pos + k] = pid;
            info.put(pid, new int[]{pos, sz});
            pos += sz;
        }
    }

    public int processMemAlloc(int processId, int size) {
        int st = findSlot(size);
        if (st < 0) {
            defrag();
            st = findSlot(size);
            if (st < 0) return -1;
        }
        for (int k = 0; k < size; k++) page[st + k] = processId;
        info.put(processId, new int[]{st, size});
        return st;
    }

    public void processMemFree(int processId) {
        int[] t = info.remove(processId);
        for (int k = 0; k < t[1]; k++) page[t[0] + k] = -1;
    }

    public int processMemQuery(int processId) {
        return info.get(processId)[0];
    }
}
```

### C++

```cpp
#include <algorithm>
#include <unordered_map>
#include <utility>
#include <vector>

using namespace std;

class MemMgmtSys {
    int n;
    vector<int> page;
    unordered_map<int, pair<int, int>> info;

    int findSlot(int size) {
        int i = 0;
        while (i < n) {
            if (page[i] != -1) { i++; continue; }
            int j = i;
            while (j < n && page[j] == -1) j++;
            if (j - i >= size) return i;
            i = j;
        }
        return -1;
    }

    void defrag() {
        vector<pair<int, int>> items;  // size, pid
        for (auto& e : info) items.push_back({e.second.second, e.first});
        sort(items.begin(), items.end());
        fill(page.begin(), page.end(), -1);
        info.clear();
        int used = 0;
        for (auto& e : items) used += e.first;
        int pos = n - used;
        for (auto& e : items) {
            int sz = e.first, pid = e.second;
            for (int k = 0; k < sz; k++) page[pos + k] = pid;
            info[pid] = {pos, sz};
            pos += sz;
        }
    }

   public:
    MemMgmtSys(int num) : n(num), page(num, -1) {}

    int processMemAlloc(int processId, int size) {
        int st = findSlot(size);
        if (st < 0) {
            defrag();
            st = findSlot(size);
            if (st < 0) return -1;
        }
        for (int k = 0; k < size; k++) page[st + k] = processId;
        info[processId] = {st, size};
        return st;
    }

    void processMemFree(int processId) {
        auto t = info[processId];
        info.erase(processId);
        for (int k = 0; k < t.second; k++) page[t.first + k] = -1;
    }

    int processMemQuery(int processId) { return info[processId].first; }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

typedef struct MemMgmtSys MemMgmtSys;

struct MemMgmtSys {
    int n;
    int* page;
    int* start;
    int* size;
    char* has;
};

static int findSlot(MemMgmtSys* obj, int size) {
    int i = 0;
    while (i < obj->n) {
        if (obj->page[i] != -1) {
            i++;
            continue;
        }
        int j = i;
        while (j < obj->n && obj->page[j] == -1) j++;
        if (j - i >= size) return i;
        i = j;
    }
    return -1;
}

typedef struct {
    int size, pid;
} Item;

static int cmpItem(const void* a, const void* b) {
    const Item* x = (const Item*)a;
    const Item* y = (const Item*)b;
    if (x->size != y->size) return x->size - y->size;
    return x->pid - y->pid;
}

static void defrag(MemMgmtSys* obj) {
    Item items[256];
    int m = 0;
    for (int pid = 0; pid <= 10000; pid++) {
        if (obj->has[pid]) {
            items[m].size = obj->size[pid];
            items[m].pid = pid;
            m++;
        }
    }
    qsort(items, (size_t)m, sizeof(Item), cmpItem);
    for (int i = 0; i < obj->n; i++) obj->page[i] = -1;
    for (int pid = 0; pid <= 10000; pid++) obj->has[pid] = 0;
    int used = 0;
    for (int i = 0; i < m; i++) used += items[i].size;
    int pos = obj->n - used;
    for (int i = 0; i < m; i++) {
        int sz = items[i].size, pid = items[i].pid;
        for (int k = 0; k < sz; k++) obj->page[pos + k] = pid;
        obj->start[pid] = pos;
        obj->size[pid] = sz;
        obj->has[pid] = 1;
        pos += sz;
    }
}

MemMgmtSys* memMgmtSysCreate(int num) {
    MemMgmtSys* obj = (MemMgmtSys*)calloc(1, sizeof(MemMgmtSys));
    obj->n = num;
    obj->page = (int*)malloc(sizeof(int) * (size_t)num);
    for (int i = 0; i < num; i++) obj->page[i] = -1;
    obj->start = (int*)calloc(10001, sizeof(int));
    obj->size = (int*)calloc(10001, sizeof(int));
    obj->has = (char*)calloc(10001, 1);
    return obj;
}

void memMgmtSysFree(MemMgmtSys* obj) {
    if (!obj) return;
    free(obj->page);
    free(obj->start);
    free(obj->size);
    free(obj->has);
    free(obj);
}

int memMgmtSysProcessMemAlloc(MemMgmtSys* obj, int processId, int size) {
    int st = findSlot(obj, size);
    if (st < 0) {
        defrag(obj);
        st = findSlot(obj, size);
        if (st < 0) return -1;
    }
    for (int k = 0; k < size; k++) obj->page[st + k] = processId;
    obj->start[processId] = st;
    obj->size[processId] = size;
    obj->has[processId] = 1;
    return st;
}

void memMgmtSysProcessMemFree(MemMgmtSys* obj, int processId) {
    int st = obj->start[processId], sz = obj->size[processId];
    obj->has[processId] = 0;
    for (int k = 0; k < sz; k++) obj->page[st + k] = -1;
}

int memMgmtSysProcessMemQuery(MemMgmtSys* obj, int processId) {
    return obj->start[processId];
}
```
