## 解题思路

在「占用天数」上二分答案；判定函数是「末次考试日 + 复习额度贪心」，而不是简单的段和阈值判断。

目标：最小化前缀长度 $D$，使前 $D$ 天内 $m$ 门科目全部考完。

**单调性**：前 $D$ 天可行 ⇒ 前 $D+1$ 天可行，故二分最小 $D$；若 $D=n$ 仍不可行则返回 `-1`。

**判定 `ok(D)`**

1. 在下标 $[0,D)$ 内，对每门科目记录**最后一次**可考日 `last[t]`；若某门未出现，失败。
2. 每天至多一门可考 ⇒ 各 `last[t]` 互异。
3. 按考试日从早到晚扫描：维护空闲天 `free`（可当复习）。走到某门考试日时，若 `free < prep[t]` 则失败，否则扣减；考试日本身不计入复习。其它天 `free++`。

为何用末次可考日：在固定前缀内，把考试尽量靠后，才能把前面的天尽量留给复习，这是该模型下的最优安排。

**常见假解**

- 仍做成切分数组 / 最小化最大段和。
- 考试日也计入复习，或要求复习必须连续。
- 每门随便挑一个可考日，而不是该前缀内的最后一次。

## 复杂度分析

- 时间：$O((n+m\log m)\log n)$（二分 × 每次扫天 + 按考试日排序）。
- 空间：$O(m)$。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def minPassDays(self, slots: List[int], prep: List[int]) -> int:
        n, m = len(slots), len(prep)

        def ok(days: int) -> bool:
            # 前 days 天：每门取最后可考日，再按考试日顺序扣复习
            last = [-1] * (m + 1)
            for i in range(days):
                t = slots[i]
                if t > 0:
                    last[t] = i
            for t in range(1, m + 1):
                if last[t] < 0:
                    return False
            order = sorted(range(1, m + 1), key=lambda t: last[t])
            free = 0
            j = 0
            for day in range(days):
                if j < m and last[order[j]] == day:
                    need = prep[order[j] - 1]
                    if free < need:
                        return False
                    free -= need
                    j += 1  # 考试日不增加 free
                else:
                    free += 1
            return j == m

        if not ok(n):
            return -1
        lo, hi = 1, n
        while lo < hi:
            mid = (lo + hi) // 2
            if ok(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```

### Java

```java
import java.util.*;

class Solution {
    public int minPassDays(int[] slots, int[] prep) {
        int n = slots.length;
        if (!ok(slots, prep, n)) return -1;
        int lo = 1, hi = n;
        while (lo < hi) {
            int mid = (lo + hi) / 2;
            if (ok(slots, prep, mid)) hi = mid;
            else lo = mid + 1;
        }
        return lo;
    }

    private boolean ok(int[] slots, int[] prep, int days) {
        int m = prep.length;
        int[] last = new int[m + 1];
        Arrays.fill(last, -1);
        for (int i = 0; i < days; i++) if (slots[i] > 0) last[slots[i]] = i;
        for (int t = 1; t <= m; t++) if (last[t] < 0) return false;
        Integer[] order = new Integer[m];
        for (int t = 1; t <= m; t++) order[t - 1] = t;
        Arrays.sort(order, Comparator.comparingInt(a -> last[a]));
        int free = 0, j = 0;
        for (int day = 0; day < days; day++) {
            if (j < m && last[order[j]] == day) {
                if (free < prep[order[j] - 1]) return false;
                free -= prep[order[j] - 1];
                j++;
            } else free++;
        }
        return j == m;
    }
}
```

```cpp
class Solution {
public:
    int minPassDays(vector<int>& slots, vector<int>& prep) {
        int n = (int)slots.size();
        int m = (int)prep.size();
        auto ok = [&](int days) {
            vector<int> last(m + 1, -1);
            for (int i = 0; i < days; ++i)
                if (slots[i] > 0) last[slots[i]] = i;
            for (int t = 1; t <= m; ++t)
                if (last[t] < 0) return false;
            vector<int> order(m);
            for (int t = 1; t <= m; ++t) order[t - 1] = t;
            sort(order.begin(), order.end(),
                 [&](int a, int b) { return last[a] < last[b]; });
            int freeDays = 0, j = 0;
            for (int day = 0; day < days; ++day) {
                if (j < m && last[order[j]] == day) {
                    if (freeDays < prep[order[j] - 1]) return false;
                    freeDays -= prep[order[j] - 1];
                    ++j;
                } else ++freeDays;
            }
            return j == m;
        };
        if (!ok(n)) return -1;
        int lo = 1, hi = n;
        while (lo < hi) {
            int mid = (lo + hi) / 2;
            if (ok(mid)) hi = mid;
            else lo = mid + 1;
        }
        return lo;
    }
```

```javascript
class Solution {
  minPassDays(slots, prep) {
    const n = slots.length, m = prep.length;
    const ok = (days) => {
      const last = Array(m + 1).fill(-1);
      for (let i = 0; i < days; i++) if (slots[i] > 0) last[slots[i]] = i;
      for (let t = 1; t <= m; t++) if (last[t] < 0) return false;
      const order = [...Array(m).keys()].map((x) => x + 1);
      order.sort((a, b) => last[a] - last[b]);
      let free = 0, j = 0;
      for (let day = 0; day < days; day++) {
        if (j < m && last[order[j]] === day) {
          if (free < prep[order[j] - 1]) return false;
          free -= prep[order[j] - 1];
          j++;
        } else free++;
      }
      return j === m;
    };
    if (!ok(n)) return -1;
    let lo = 1, hi = n;
    while (lo < hi) {
      const mid = (lo + hi) >> 1;
      if (ok(mid)) hi = mid;
      else lo = mid + 1;
    }
    return lo;
  }
}
```

### C

```c
#include <stdlib.h>
#include <string.h>

static int check_ok(int* slots, int* prep, int m, int days, int* last, int* order) {
    for (int t = 1; t <= m; ++t) last[t] = -1;
    for (int i = 0; i < days; ++i)
        if (slots[i] > 0) last[slots[i]] = i;
    for (int t = 1; t <= m; ++t)
        if (last[t] < 0) return 0;
    for (int t = 1; t <= m; ++t) order[t - 1] = t;
    for (int i = 0; i < m; ++i) {
        for (int j = i + 1; j < m; ++j) {
            if (last[order[j]] < last[order[i]]) {
                int tmp = order[i];
                order[i] = order[j];
                order[j] = tmp;
            }
        }
    }
    int freeDays = 0, j = 0;
    for (int day = 0; day < days; ++day) {
        if (j < m && last[order[j]] == day) {
            if (freeDays < prep[order[j] - 1]) return 0;
            freeDays -= prep[order[j] - 1];
            ++j;
        } else {
            ++freeDays;
        }
    }
    return j == m;
}

int minPassDays(int* slots, int slotsSize, int* prep, int prepSize) {
    int n = slotsSize, m = prepSize;
    int* last = (int*)malloc(sizeof(int) * (m + 1));
    int* order = (int*)malloc(sizeof(int) * m);
    int ans = -1;
    if (check_ok(slots, prep, m, n, last, order)) {
        int lo = 1, hi = n;
        while (lo < hi) {
            int mid = (lo + hi) / 2;
            if (check_ok(slots, prep, m, mid, last, order)) hi = mid;
            else lo = mid + 1;
        }
        ans = lo;
    }
    free(last);
    free(order);
    return ans;
}
```
