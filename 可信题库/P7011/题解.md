## 解题思路

把证书存成森林：每张证书记录父节点、过期时刻、是否被吊销。

关键约定是**懒吊销**：`revoke` 只打当前节点的标记，不扫子孙。有效性在查询时沿父链检查——路径上任一节点被吊销或已过期，则整条链无效。这样 `revoke` 为 $O(1)$，`isValid` / `ttl` 为 $O(\textit{高度})$。

`issue` 时除了 id 冲突、非法过期时刻外，还要：

- 父节点存在且父链上没有吊销标记（不能挂在已失效链下）
- 子证书 `expireAt` 不超过父证书，从而时间上也不会比祖先活得更久

`ttl` 在有效的前提下取路径上最小 `expireAt` 再减 `now`。`issuerOf` / `rootOf` 只看结构，吊销后仍可查询。

## 复杂度分析

- 单次 `issue` / `revoke` / 查询：$O(h)$，$h$ 为树高；累计 $4000$ 次调用即可。
- 空间复杂度：$O(\textit{证书数})$。

## 代码实现

### Python

```python
class CertAuthority:
    def __init__(self):
        self.parent = {}
        self.expire = {}
        self.revoked = {}

    def _chain_clean(self, cert_id: int) -> bool:
        # 父链上不能有已吊销节点
        cur = cert_id
        seen = set()
        while cur != 0:
            if cur in seen or cur not in self.parent or self.revoked[cur]:
                return False
            seen.add(cur)
            cur = self.parent[cur]
        return True

    def issue(self, certId: int, parentId: int, expireAt: int) -> bool:
        if certId <= 0 or expireAt <= 0 or certId in self.parent:
            return False
        if parentId == 0:
            self.parent[certId] = 0
            self.expire[certId] = expireAt
            self.revoked[certId] = False
            return True
        if parentId not in self.parent or not self._chain_clean(parentId):
            return False
        if expireAt > self.expire[parentId]:
            return False
        self.parent[certId] = parentId
        self.expire[certId] = expireAt
        self.revoked[certId] = False
        return True

    def revoke(self, certId: int) -> bool:
        if certId not in self.parent or self.revoked[certId]:
            return False
        self.revoked[certId] = True
        return True

    def isValid(self, certId: int, now: int) -> bool:
        cur = certId
        seen = set()
        while cur != 0:
            if cur in seen or cur not in self.parent:
                return False
            if self.revoked[cur] or now >= self.expire[cur]:
                return False
            seen.add(cur)
            cur = self.parent[cur]
        return certId in self.parent

    def ttl(self, certId: int, now: int) -> int:
        if not self.isValid(certId, now):
            return -1
        cur = certId
        mn = self.expire[cur]
        while cur != 0:
            mn = min(mn, self.expire[cur])
            cur = self.parent[cur]
        return mn - now

    def issuerOf(self, certId: int) -> int:
        return self.parent.get(certId, -1)

    def rootOf(self, certId: int) -> int:
        if certId not in self.parent:
            return -1
        cur = certId
        while self.parent[cur] != 0:
            cur = self.parent[cur]
        return cur
```

### Java

```java
import java.util.*;

class CertAuthority {
    private Map<Integer, Integer> parent = new HashMap<>();
    private Map<Integer, Integer> expire = new HashMap<>();
    private Map<Integer, Boolean> revoked = new HashMap<>();

    private boolean chainClean(int certId) {
        int cur = certId;
        Set<Integer> seen = new HashSet<>();
        while (cur != 0) {
            if (seen.contains(cur) || !parent.containsKey(cur) || revoked.get(cur)) return false;
            seen.add(cur);
            cur = parent.get(cur);
        }
        return true;
    }

    public boolean issue(int certId, int parentId, int expireAt) {
        if (certId <= 0 || expireAt <= 0 || parent.containsKey(certId)) return false;
        if (parentId == 0) {
            parent.put(certId, 0);
            expire.put(certId, expireAt);
            revoked.put(certId, false);
            return true;
        }
        if (!parent.containsKey(parentId) || !chainClean(parentId)) return false;
        if (expireAt > expire.get(parentId)) return false;
        parent.put(certId, parentId);
        expire.put(certId, expireAt);
        revoked.put(certId, false);
        return true;
    }

    public boolean revoke(int certId) {
        if (!parent.containsKey(certId) || revoked.get(certId)) return false;
        revoked.put(certId, true);
        return true;
    }

    public boolean isValid(int certId, int now) {
        int cur = certId;
        Set<Integer> seen = new HashSet<>();
        while (cur != 0) {
            if (seen.contains(cur) || !parent.containsKey(cur)) return false;
            if (revoked.get(cur) || now >= expire.get(cur)) return false;
            seen.add(cur);
            cur = parent.get(cur);
        }
        return parent.containsKey(certId);
    }

    public int ttl(int certId, int now) {
        if (!isValid(certId, now)) return -1;
        int cur = certId, mn = expire.get(cur);
        while (cur != 0) {
            mn = Math.min(mn, expire.get(cur));
            cur = parent.get(cur);
        }
        return mn - now;
    }

    public int issuerOf(int certId) {
        return parent.getOrDefault(certId, -1);
    }

    public int rootOf(int certId) {
        if (!parent.containsKey(certId)) return -1;
        int cur = certId;
        while (parent.get(cur) != 0) cur = parent.get(cur);
        return cur;
    }
}
```

### C++

```cpp
#include <unordered_map>
using namespace std;

class CertAuthority {
    unordered_map<int, int> parent, expire, revoked;

    bool chainClean(int certId) {
        int cur = certId;
        unordered_map<int, int> seen;
        while (cur != 0) {
            if (seen.count(cur) || !parent.count(cur) || revoked[cur]) return false;
            seen[cur] = 1;
            cur = parent[cur];
        }
        return true;
    }

public:
    bool issue(int certId, int parentId, int expireAt) {
        if (certId <= 0 || expireAt <= 0 || parent.count(certId)) return false;
        if (parentId == 0) {
            parent[certId] = 0;
            expire[certId] = expireAt;
            revoked[certId] = 0;
            return true;
        }
        if (!parent.count(parentId) || !chainClean(parentId)) return false;
        if (expireAt > expire[parentId]) return false;
        parent[certId] = parentId;
        expire[certId] = expireAt;
        revoked[certId] = 0;
        return true;
    }

    bool revoke(int certId) {
        if (!parent.count(certId) || revoked[certId]) return false;
        revoked[certId] = 1;
        return true;
    }

    bool isValid(int certId, int now) {
        int cur = certId;
        unordered_map<int, int> seen;
        while (cur != 0) {
            if (seen.count(cur) || !parent.count(cur)) return false;
            if (revoked[cur] || now >= expire[cur]) return false;
            seen[cur] = 1;
            cur = parent[cur];
        }
        return parent.count(certId);
    }

    int ttl(int certId, int now) {
        if (!isValid(certId, now)) return -1;
        int cur = certId, mn = expire[cur];
        while (cur != 0) {
            mn = min(mn, expire[cur]);
            cur = parent[cur];
        }
        return mn - now;
    }

    int issuerOf(int certId) {
        return parent.count(certId) ? parent[certId] : -1;
    }

    int rootOf(int certId) {
        if (!parent.count(certId)) return -1;
        int cur = certId;
        while (parent[cur] != 0) cur = parent[cur];
        return cur;
    }
};
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id, parent, expire, revoked, used;
} Cert;

typedef struct certAuthority {
    Cert* a;
    int n, cap;
} certAuthority;

static Cert* findCert(certAuthority* obj, int id) {
    for (int i = 0; i < obj->n; i++) if (obj->a[i].id == id) return &obj->a[i];
    return NULL;
}

static bool chainClean(certAuthority* obj, int id) {
    int guard = 0;
    while (id != 0) {
        Cert* c = findCert(obj, id);
        if (!c || c->revoked) return false;
        id = c->parent;
        if (++guard > 5000) return false;
    }
    return true;
}

certAuthority* certAuthorityCreate(void) {
    certAuthority* obj = (certAuthority*)calloc(1, sizeof(certAuthority));
    obj->cap = 16;
    obj->a = (Cert*)calloc(obj->cap, sizeof(Cert));
    return obj;
}

void certAuthorityFree(certAuthority* obj) {
    if (!obj) return;
    free(obj->a);
    free(obj);
}

bool certAuthorityIssue(certAuthority* obj, int certId, int parentId, int expireAt) {
    if (certId <= 0 || expireAt <= 0 || findCert(obj, certId)) return false;
    if (parentId == 0) {
        /* ok */
    } else {
        Cert* p = findCert(obj, parentId);
        if (!p || !chainClean(obj, parentId) || expireAt > p->expire) return false;
    }
    if (obj->n >= obj->cap) {
        obj->cap *= 2;
        obj->a = (Cert*)realloc(obj->a, obj->cap * sizeof(Cert));
    }
    Cert* c = &obj->a[obj->n++];
    c->id = certId;
    c->parent = parentId;
    c->expire = expireAt;
    c->revoked = 0;
    return true;
}

bool certAuthorityRevoke(certAuthority* obj, int certId) {
    Cert* c = findCert(obj, certId);
    if (!c || c->revoked) return false;
    c->revoked = 1;
    return true;
}

bool certAuthorityIsValid(certAuthority* obj, int certId, int now) {
    if (!findCert(obj, certId)) return false;
    int id = certId, guard = 0;
    while (id != 0) {
        Cert* c = findCert(obj, id);
        if (!c || c->revoked || now >= c->expire) return false;
        id = c->parent;
        if (++guard > 5000) return false;
    }
    return true;
}

int certAuthorityTtl(certAuthority* obj, int certId, int now) {
    if (!certAuthorityIsValid(obj, certId, now)) return -1;
    int id = certId, mn = findCert(obj, certId)->expire, guard = 0;
    while (id != 0) {
        Cert* c = findCert(obj, id);
        if (c->expire < mn) mn = c->expire;
        id = c->parent;
        if (++guard > 5000) break;
    }
    return mn - now;
}

int certAuthorityIssuerOf(certAuthority* obj, int certId) {
    Cert* c = findCert(obj, certId);
    return c ? c->parent : -1;
}

int certAuthorityRootOf(certAuthority* obj, int certId) {
    Cert* c = findCert(obj, certId);
    if (!c) return -1;
    int id = certId, guard = 0;
    while (1) {
        Cert* x = findCert(obj, id);
        if (!x) return -1;
        if (x->parent == 0) return id;
        id = x->parent;
        if (++guard > 5000) return -1;
    }
}
```
