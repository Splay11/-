## 解题思路

本题是**依赖图上的影响面扩散 + DAG 并行调度**。

1. **重建集合**：`deps` 中 `[a, b]` 表示 `a` 依赖 `b`。变更沿**反向边**传播：若 `a` 依赖 `b`，则建边 `b → a`。从 `changed` 做 BFS/DFS，得到所有直接或间接受影响的模块集合 `S`。
2. **并行完成时刻**：只在 `S` 内考虑依赖。对 `u ∈ S`，设其在 `S` 内的依赖为 `D(u)`，则
   $$
   \textit{finish}[u] = \textit{buildTime}[u] + \max_{v \in D(u)} \textit{finish}[v]
   $$
   （若 `D(u)` 为空，取 $\max=0$）。这正是 DAG 上按拓扑序递推最长路。
3. 答案为 $\max_{u \in S} \textit{finish}[u]$；`S` 为空则返回 $0$。

常见假解：只统计受影响模块个数、把依赖方向建反、串行累加 `buildTime`、或把不在 `S` 内的依赖也当作必须等待。

## 复杂度分析

- 时间：$O(n + m)$，扩散与拓扑各扫一遍边。
- 空间：$O(n + m)$，邻接表与入度/完成时刻数组。

$n \le 10^4$，$m \le 2\cdot 10^4$，线性算法足够。

## 代码实现

### Python

```python
from typing import List
from collections import defaultdict, deque


class Solution:
    def minRebuildTime(
        self, n: int, deps: List[List[int]], buildTime: List[int], changed: List[int]
    ) -> int:
        dependents = defaultdict(list)  # b → 依赖 b 的模块
        depends_on = defaultdict(list)  # a → a 依赖的模块
        for a, b in deps:
            dependents[b].append(a)
            depends_on[a].append(b)

        # 从变更点沿反向边扩散重建集合
        need = set()
        q = deque()
        for x in changed:
            if x not in need:
                need.add(x)
                q.append(x)
        while q:
            u = q.popleft()
            for v in dependents[u]:
                if v not in need:
                    need.add(v)
                    q.append(v)
        if not need:
            return 0

        # 仅统计「重建集合内」的依赖作为入度
        indeg = {u: 0 for u in need}
        for u in need:
            for b in depends_on[u]:
                if b in need:
                    indeg[u] += 1

        qq = deque([u for u in need if indeg[u] == 0])
        finish = {}
        while qq:
            u = qq.popleft()
            mx = 0
            for b in depends_on[u]:
                if b in need:
                    mx = max(mx, finish[b])  # 等集合内依赖全部完成
            finish[u] = mx + buildTime[u]
            for v in dependents[u]:
                if v in need:
                    indeg[v] -= 1
                    if indeg[v] == 0:
                        qq.append(v)
        return max(finish.values())
```

### Java

```java
import java.util.*;

class Solution {
    public int minRebuildTime(int n, int[][] deps, int[] buildTime, int[] changed) {
        List<List<Integer>> dependents = new ArrayList<>();
        List<List<Integer>> dependsOn = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            dependents.add(new ArrayList<>());
            dependsOn.add(new ArrayList<>());
        }
        for (int[] e : deps) {
            int a = e[0], b = e[1];
            dependents.get(b).add(a); // 反向：谁依赖 b
            dependsOn.get(a).add(b);
        }

        boolean[] need = new boolean[n];
        Deque<Integer> q = new ArrayDeque<>();
        for (int x : changed) {
            if (!need[x]) {
                need[x] = true;
                q.add(x);
            }
        }
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : dependents.get(u)) {
                if (!need[v]) {
                    need[v] = true;
                    q.add(v);
                }
            }
        }

        int[] indeg = new int[n];
        int cnt = 0;
        for (int u = 0; u < n; u++) {
            if (!need[u]) continue;
            cnt++;
            for (int b : dependsOn.get(u)) {
                if (need[b]) indeg[u]++; // 只计集合内依赖
            }
        }
        if (cnt == 0) return 0;

        Deque<Integer> qq = new ArrayDeque<>();
        long[] finish = new long[n];
        for (int u = 0; u < n; u++) {
            if (need[u] && indeg[u] == 0) qq.add(u);
        }
        long ans = 0;
        while (!qq.isEmpty()) {
            int u = qq.poll();
            long mx = 0;
            for (int b : dependsOn.get(u)) {
                if (need[b]) mx = Math.max(mx, finish[b]);
            }
            finish[u] = mx + buildTime[u];
            ans = Math.max(ans, finish[u]);
            for (int v : dependents.get(u)) {
                if (!need[v]) continue;
                if (--indeg[v] == 0) qq.add(v);
            }
        }
        return (int) ans;
    }
}
```

### C++

```cpp
#include <vector>
#include <queue>
#include <algorithm>
using namespace std;

class Solution {
public:
    int minRebuildTime(int n, vector<vector<int>>& deps, vector<int>& buildTime, vector<int>& changed) {
        vector<vector<int>> dependents(n), dependsOn(n);
        for (auto& e : deps) {
            int a = e[0], b = e[1];
            dependents[b].push_back(a); // 反向边：变更传播方向
            dependsOn[a].push_back(b);
        }

        vector<char> need(n, 0);
        queue<int> q;
        for (int x : changed) {
            if (!need[x]) {
                need[x] = 1;
                q.push(x);
            }
        }
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int v : dependents[u]) {
                if (!need[v]) {
                    need[v] = 1;
                    q.push(v);
                }
            }
        }

        vector<int> indeg(n, 0);
        int cnt = 0;
        for (int u = 0; u < n; u++) {
            if (!need[u]) continue;
            cnt++;
            for (int b : dependsOn[u])
                if (need[b]) indeg[u]++;
        }
        if (!cnt) return 0;

        queue<int> qq;
        vector<long long> finish(n, 0);
        for (int u = 0; u < n; u++)
            if (need[u] && indeg[u] == 0) qq.push(u);

        long long ans = 0;
        while (!qq.empty()) {
            int u = qq.front();
            qq.pop();
            long long mx = 0;
            for (int b : dependsOn[u])
                if (need[b]) mx = max(mx, finish[b]); // 并行：取依赖完成时刻最大值
            finish[u] = mx + buildTime[u];
            ans = max(ans, finish[u]);
            for (int v : dependents[u]) {
                if (!need[v]) continue;
                if (--indeg[v] == 0) qq.push(v);
            }
        }
        return (int)ans;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

int minRebuildTime(int n, int** deps, int depsSize, int* depsColSize,
                   int* buildTime, int buildTimeSize, int* changed, int changedSize) {
    (void)depsColSize; (void)buildTimeSize;
    /* 邻接：dependents[b] 含依赖 b 的模块；dependsOn[a] 含 a 依赖的模块 */
    int* deg_out = (int*)calloc(n, sizeof(int));
    int* deg_in = (int*)calloc(n, sizeof(int));
    for (int i = 0; i < depsSize; i++) {
        int a = deps[i][0], b = deps[i][1];
        deg_out[b]++;
        deg_in[a]++;
    }
    int** dependents = (int**)malloc(n * sizeof(int*));
    int** dependsOn = (int**)malloc(n * sizeof(int*));
    int* out_cnt = (int*)calloc(n, sizeof(int));
    int* in_cnt = (int*)calloc(n, sizeof(int));
    for (int i = 0; i < n; i++) {
        dependents[i] = (int*)malloc((deg_out[i] > 0 ? deg_out[i] : 1) * sizeof(int));
        dependsOn[i] = (int*)malloc((deg_in[i] > 0 ? deg_in[i] : 1) * sizeof(int));
    }
    for (int i = 0; i < depsSize; i++) {
        int a = deps[i][0], b = deps[i][1];
        dependents[b][out_cnt[b]++] = a;
        dependsOn[a][in_cnt[a]++] = b;
    }

    char* need = (char*)calloc(n, 1);
    int* q = (int*)malloc(n * sizeof(int));
    int qh = 0, qt = 0;
    for (int i = 0; i < changedSize; i++) {
        int x = changed[i];
        if (!need[x]) { need[x] = 1; q[qt++] = x; }
    }
    while (qh < qt) {
        int u = q[qh++];
        for (int i = 0; i < out_cnt[u]; i++) {
            int v = dependents[u][i];
            if (!need[v]) { need[v] = 1; q[qt++] = v; }
        }
    }

    int* indeg = (int*)calloc(n, sizeof(int));
    int cnt = 0;
    for (int u = 0; u < n; u++) {
        if (!need[u]) continue;
        cnt++;
        for (int i = 0; i < in_cnt[u]; i++)
            if (need[dependsOn[u][i]]) indeg[u]++;
    }
    if (!cnt) {
        for (int i = 0; i < n; i++) { free(dependents[i]); free(dependsOn[i]); }
        free(dependents); free(dependsOn); free(deg_out); free(deg_in);
        free(out_cnt); free(in_cnt); free(need); free(q); free(indeg);
        return 0;
    }

    qh = qt = 0;
    long long* finish = (long long*)calloc(n, sizeof(long long));
    for (int u = 0; u < n; u++)
        if (need[u] && indeg[u] == 0) q[qt++] = u;

    long long ans = 0;
    while (qh < qt) {
        int u = q[qh++];
        long long mx = 0;
        for (int i = 0; i < in_cnt[u]; i++) {
            int b = dependsOn[u][i];
            if (need[b] && finish[b] > mx) mx = finish[b]; /* 并行：取依赖完成时间最大值 */
        }
        finish[u] = mx + buildTime[u];
        if (finish[u] > ans) ans = finish[u];
        for (int i = 0; i < out_cnt[u]; i++) {
            int v = dependents[u][i];
            if (!need[v]) continue;
            if (--indeg[v] == 0) q[qt++] = v;
        }
    }

    for (int i = 0; i < n; i++) { free(dependents[i]); free(dependsOn[i]); }
    free(dependents); free(dependsOn); free(deg_out); free(deg_in);
    free(out_cnt); free(in_cnt); free(need); free(q); free(indeg); free(finish);
    return (int)ans;
}
```
