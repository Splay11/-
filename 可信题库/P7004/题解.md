## 解题思路

合规归档场景下，专线按日计费、包不可拆且次序固定。本题求「在不超过 `limitDays` 个归档日内传完」的**最小日容量** $cap$。

关键性质：容量越大，需要的天数越少（单调），因此可以对答案做**二分**。

判定函数：给定容量 $cap$，按固定顺序贪心装箱——同一天尽量连续装下若干完整包，当前包装不下就新开一天。统计所需天数，若 $\le$ `limitDays` 则可行。

二分下界为 $\max(\texttt{logs})$（单包不能拆），上界为 $\sum\texttt{logs}$（一天装完全部）。

## 复杂度分析

- 时间复杂度：$O(n\log S)$，其中 $S=\sum\texttt{logs}$，每次判定扫描一遍数组。
- 空间复杂度：$O(1)$ 额外空间（不计输入）。

## 代码实现

### Python

```python
from typing import List

class Solution:
    def minArchiveCap(self, logs: List[int], limitDays: int) -> int:
        def need(cap: int) -> int:
            # 贪心连续装箱：装不下就新开一天
            days, cur = 1, 0
            for x in logs:
                if cur + x > cap:
                    days += 1
                    cur = 0
                cur += x
            return days

        lo, hi = max(logs), sum(logs)
        while lo < hi:
            mid = (lo + hi) // 2
            # 可行则尝试更小容量
            if need(mid) <= limitDays:
                hi = mid
            else:
                lo = mid + 1
        return lo
```

### Java

```java
public class Solution {
    public long minArchiveCap(int[] logs, int limitDays) {
        long lo = 0, hi = 0;
        for (int x : logs) {
            lo = Math.max(lo, x); // 下界：最大单包
            hi += x;              // 上界：总和
        }
        while (lo < hi) {
            long mid = (lo + hi) >>> 1;
            if (need(logs, mid) <= limitDays) {
                hi = mid; // mid 可行，向左收缩
            } else {
                lo = mid + 1;
            }
        }
        return lo;
    }

    private int need(int[] logs, long cap) {
        int days = 1;
        long cur = 0;
        for (int x : logs) {
            // 当天剩余额度装不下当前整包
            if (cur + x > cap) {
                days++;
                cur = 0;
            }
            cur += x;
        }
        return days;
    }
}
```

### C++

```cpp
#include <vector>
#include <numeric>
#include <algorithm>
using namespace std;

class Solution {
 public:
  long long minArchiveCap(vector<int>& logs, int limitDays) {
    long long lo = *max_element(logs.begin(), logs.end());
    long long hi = accumulate(logs.begin(), logs.end(), 0LL);
    while (lo < hi) {
      long long mid = lo + (hi - lo) / 2;
      // 判定：容量 mid 是否能在 limitDays 天内传完
      if (need(logs, mid) <= limitDays)
        hi = mid;
      else
        lo = mid + 1;
    }
    return lo;
  }

 private:
  int need(vector<int>& logs, long long cap) {
    int days = 1;
    long long cur = 0;
    for (int x : logs) {
      if (cur + x > cap) { // 新开归档日
        days++;
        cur = 0;
      }
      cur += x;
    }
    return days;
  }
};
```

### C

```c
long long minArchiveCap(int* logs, int logsSize, int limitDays) {
    long long lo = 0, hi = 0;
    for (int i = 0; i < logsSize; i++) {
        if (logs[i] > lo) lo = logs[i]; // 下界
        hi += logs[i];                   // 上界
    }
    while (lo < hi) {
        long long mid = lo + (hi - lo) / 2;
        int days = 1;
        long long cur = 0;
        for (int i = 0; i < logsSize; i++) {
            // 连续整包装箱
            if (cur + logs[i] > mid) {
                days++;
                cur = 0;
            }
            cur += logs[i];
        }
        if (days <= limitDays)
            hi = mid; // 可行缩小容量
        else
            lo = mid + 1;
    }
    return lo;
}
```
