## 解题思路

这道题只需要两个最基础的容器：一个**队列**负责排队，一个**集合（哈希表）**负责判重。

- 队列按顺序存下还在等待的号，**队尾进、队首出**，正好就是「先取号先叫号」。
- 集合里放当前还在等待的号，用它 $O(1)$ 地判断某个号是不是已经排着了。

四个方法怎么实现：

- `PickupDesk()`：队列和集合都建成空的，表示还没有人取号。
- `order(ticketId)`：先看集合里有没有这个号。有，说明它还在等待，直接返回 `false`，队列和集合都不动；没有，就把它放到**队尾**，同时加进集合，返回 `true`。
- `serve()`：队列为空就返回 `-1`，什么都不改；否则弹出**队首**的号，返回它。注意弹队首的同时要把它**从集合里删掉**，这一点很容易漏，漏了以后这个号就永远被当成「还在等待」，再也取不上。
- `waiting()`：还在等待的号数，就是集合的大小，它和队列里剩余元素的个数始终相等。

还要留意一句：只有**还在队列里**的号才算重复。号被叫走之后就从集合删掉了，这时再取同一个号是允许的，会重新排到队尾。

## 复杂度分析

- 时间：每个方法只做一次入队或出队、加上常数次哈希表查找，平均 $O(1)$。累计调用 $q$ 次，总时间 $O(q)$。
- 空间：队列和集合最多同时保存 $q$ 个号，$O(q)$。

## 代码实现

### Python

```python
class PickupDesk:
    def __init__(self):
        # queue 按取号先后存等待的号，head 是队首下标
        self.queue = []
        self.head = 0
        # waitingIds 记录当前还在等待的号，用于 O(1) 判重
        self.waitingIds = set()

    def order(self, ticketId: int) -> bool:
        # 号已经在等待队列里，重复取号不改变任何状态
        if ticketId in self.waitingIds:
            return False
        self.queue.append(ticketId)      # 新号排到队尾
        self.waitingIds.add(ticketId)
        return True

    def serve(self) -> int:
        # 队列为空时叫号失败，返回 -1
        if self.head == len(self.queue):
            return -1
        ticketId = self.queue[self.head]
        self.head += 1                   # 队首出队，先来先叫
        # 叫走之后这个号不再等待，才能被重新取
        self.waitingIds.remove(ticketId)
        return ticketId

    def waiting(self) -> int:
        # 集合大小就是还在等待的号数
        return len(self.waitingIds)
```

### Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.HashSet;
import java.util.Set;

class PickupDesk {
    // q 按取号先后存等待的号，pollFirst 取队首保证先来先叫
    private Deque<Integer> q = new ArrayDeque<>();
    // waitingIds 记录当前还在等待的号，用于判重
    private Set<Integer> waitingIds = new HashSet<>();

    public PickupDesk() {}

    public boolean order(int ticketId) {
        // 已经在等待队列里，重复取号失败
        if (waitingIds.contains(ticketId)) return false;
        q.addLast(ticketId);              // 新号排到队尾
        waitingIds.add(ticketId);
        return true;
    }

    public int serve() {
        // 队列为空返回 -1
        if (q.isEmpty()) return -1;
        int ticketId = q.pollFirst();     // 叫走队首的号
        // 出队后同步从等待集合中删掉，号才能再次被取
        waitingIds.remove(ticketId);
        return ticketId;
    }

    public int waiting() {
        return waitingIds.size();
    }
}
```

### C++

```cpp
#include <queue>
#include <unordered_set>

using namespace std;

class PickupDesk {
    // q 按取号先后存等待的号，front/pop 取队首
    queue<int> q;
    // waitingIds 记录当前还在等待的号，用于判重
    unordered_set<int> waitingIds;

public:
    PickupDesk() {}

    bool order(int ticketId) {
        // 已经在等待队列里，判重失败
        if (waitingIds.count(ticketId)) return false;
        q.push(ticketId);                 // 新号排到队尾
        waitingIds.insert(ticketId);
        return true;
    }

    int serve() {
        // 队列为空返回 -1
        if (q.empty()) return -1;
        int ticketId = q.front();
        q.pop();                          // 先来先叫
        // 出队后同步从等待集合中删掉
        waitingIds.erase(ticketId);
        return ticketId;
    }

    int waiting() { return (int)waitingIds.size(); }
};
```

### JavaScript

```javascript
class PickupDesk {
    constructor() {
        // q 当队列用，head 指向队首，避免频繁移动整个数组
        this.q = [];
        this.head = 0;
        // wait 记录当前还在等待的号，用于判重
        this.wait = new Set();
    }

    order(ticketId) {
        // 已经在等待队列里，重复取号失败
        if (this.wait.has(ticketId)) return false;
        this.q.push(ticketId);            // 新号排到队尾
        this.wait.add(ticketId);
        return true;
    }

    serve() {
        // 队首下标追上数组末尾说明队列为空
        if (this.head >= this.q.length) return -1;
        const ticketId = this.q[this.head++];   // 叫走队首的号
        // 从等待集合里删掉，号才能再次被取
        this.wait.delete(ticketId);
        return ticketId;
    }

    waiting() {
        return this.wait.size;
    }
}
```

### C

```c
#include <stdbool.h>
#include <stdlib.h>

#define MAXCALL 10050   /* 调用上限，等待的号不会超过这个数 */
#define HASHSZ 32768    /* 哈希集合容量，为 2 的幂，便于用位与取模 */

typedef struct PickupDesk {
    int queue[MAXCALL];  /* 顺序队列：queue[head, tail) 是还在等待的号 */
    int head, tail;
    int key[HASHSZ];     /* 开放寻址哈希集合：-1 空位，-2 墓碑，其余是号 */
} PickupDesk;

/* 简单的整数哈希，把票号打散到表内 */
static unsigned hashOf(int x) {
    unsigned u = (unsigned)x;
    u ^= u >> 16;
    u *= 2654435761u;
    u ^= u >> 13;
    return u & (HASHSZ - 1);
}

/* 返回票号所在槽下标，不在集合中返回 -1 */
static int setFind(PickupDesk* obj, int ticketId) {
    int i = (int)hashOf(ticketId);
    while (obj->key[i] != -1) {          /* 遇到真正的空位才停，墓碑要继续找 */
        if (obj->key[i] == ticketId) return i;
        i = (i + 1) & (HASHSZ - 1);
    }
    return -1;
}

/* 插入票号：线性探测，空位或墓碑都能放 */
static void setInsert(PickupDesk* obj, int ticketId) {
    int i = (int)hashOf(ticketId);
    while (obj->key[i] >= 0) {
        i = (i + 1) & (HASHSZ - 1);
    }
    obj->key[i] = ticketId;
}

/* 删除票号：打墓碑，保证探测链后面的元素还能被查到 */
static void setErase(PickupDesk* obj, int ticketId) {
    int i = setFind(obj, ticketId);
    if (i >= 0) obj->key[i] = -2;
}

PickupDesk* pickupDeskCreate(void) {
    PickupDesk* obj = (PickupDesk*)malloc(sizeof(PickupDesk));
    obj->head = obj->tail = 0;
    for (int i = 0; i < HASHSZ; i++) obj->key[i] = -1;
    return obj;
}

void pickupDeskFree(PickupDesk* obj) {
    free(obj);
}

bool pickupDeskOrder(PickupDesk* obj, int ticketId) {
    if (setFind(obj, ticketId) >= 0) return false;  /* 已在等待队列里，判重失败 */
    obj->queue[obj->tail++] = ticketId;             /* 新号排到队尾 */
    setInsert(obj, ticketId);
    return true;
}

int pickupDeskServe(PickupDesk* obj) {
    if (obj->head == obj->tail) return -1;          /* 队列为空 */
    int ticketId = obj->queue[obj->head++];         /* 先来先叫 */
    setErase(obj, ticketId);                        /* 出队后删掉等待标记 */
    return ticketId;
}

int pickupDeskWaiting(PickupDesk* obj) {
    return obj->tail - obj->head;                   /* 队列中剩余元素个数 */
}
```
