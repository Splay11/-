## 解题思路

将所有黑棋坐标放入哈希集合。对四个方向 $(1,0),(0,1),(1,1),(1,-1)$：

- 对每个棋子，若其在该方向的**前一格**也有棋，则跳过（说明不是该方向连线的起点）；
- 否则沿该方向一路统计连续棋子数，达到 $5$ 即返回 `"YES"`。

$n<5$ 时不可能成五，自然会得到 `"NO"`。样例 3 中缺口会打断连续计数。

**常见假解**：只检查「恰好 5 子」而忽略更长连珠；把非相邻同线的子也算进长度；只检查一个方向的正半轴却从中段起算导致漏判。

## 复杂度分析

- 时间：$O(n)$，每子每方向至多沿连线扫描一次（起点约束使总扫描线性）。
- 空间：$O(n)$ 存集合。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def hasFiveInRow(self, blackChessPoses: List[List[int]]) -> str:
        s = {(x, y) for x, y in blackChessPoses}
        dirs = ((1, 0), (0, 1), (1, 1), (1, -1))
        for x, y in s:
            for dx, dy in dirs:
                # 只从该方向连线起点开始数
                if (x - dx, y - dy) in s:
                    continue
                cnt = 0
                cx, cy = x, y
                while (cx, cy) in s:
                    cnt += 1
                    if cnt >= 5:
                        return "YES"
                    cx += dx
                    cy += dy
        return "NO"
```

### Java

```java
import java.util.*;

public class Solution {
    public String hasFiveInRow(int[][] blackChessPoses) {
        Set<Long> s = new HashSet<>();
        for (int[] p : blackChessPoses) s.add(key(p[0], p[1]));
        int[][] dirs = {{1, 0}, {0, 1}, {1, 1}, {1, -1}};
        for (int[] p : blackChessPoses) {
            int x = p[0], y = p[1];
            for (int[] d : dirs) {
                int dx = d[0], dy = d[1];
                // 前一格有子则非起点
                if (s.contains(key(x - dx, y - dy))) continue;
                int cnt = 0, cx = x, cy = y;
                while (s.contains(key(cx, cy))) {
                    if (++cnt >= 5) return "YES";
                    cx += dx;
                    cy += dy;
                }
            }
        }
        return "NO";
    }

    private static long key(int x, int y) {
        return ((long) x << 6) | y; // y < 32 足够
    }
}
```

```cpp
#include <string>
#include <unordered_set>
#include <utility>
#include <vector>

using namespace std;

struct PairHash {
    size_t operator()(const pair<int, int>& p) const {
        return (size_t)p.first * 31u + (size_t)p.second;
    }
};

class Solution {
   public:
    string hasFiveInRow(vector<vector<int>>& blackChessPoses) {
        unordered_set<pair<int, int>, PairHash> s;
        for (auto& p : blackChessPoses) s.insert({p[0], p[1]});
        const int dirs[4][2] = {{1, 0}, {0, 1}, {1, 1}, {1, -1}};
        for (const auto& pt : s) {
            int x = pt.first, y = pt.second;
            for (auto& d : dirs) {
                int dx = d[0], dy = d[1];
                if (s.count({x - dx, y - dy})) continue;
                int cnt = 0, cx = x, cy = y;
                while (s.count({cx, cy})) {
                    if (++cnt >= 5) return "YES";
                    cx += dx;
                    cy += dy;
                }
            }
        }
        return "NO";
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

static char* dup_str(const char* s) {
    char* r = (char*)malloc(strlen(s) + 1);
    strcpy(r, s);
    return r;
}

char* hasFiveInRow(int** blackChessPoses, int blackChessPosesSize, int* blackChessPosesColSize) {
    (void)blackChessPosesColSize;
    int board[19][19];
    memset(board, 0, sizeof(board));
    for (int i = 0; i < blackChessPosesSize; i++) {
        int x = blackChessPoses[i][0], y = blackChessPoses[i][1];
        board[x][y] = 1;
    }
    const int dirs[4][2] = {{1, 0}, {0, 1}, {1, 1}, {1, -1}};
    for (int x = 0; x < 19; x++) {
        for (int y = 0; y < 19; y++) {
            if (!board[x][y]) continue;
            for (int d = 0; d < 4; d++) {
                int dx = dirs[d][0], dy = dirs[d][1];
                int px = x - dx, py = y - dy;
                // 只从线段端点开始数，避免重复
                if (px >= 0 && px < 19 && py >= 0 && py < 19 && board[px][py]) continue;
                int cnt = 0, cx = x, cy = y;
                while (cx >= 0 && cx < 19 && cy >= 0 && cy < 19 && board[cx][cy]) {
                    if (++cnt >= 5) return dup_str("YES");
                    cx += dx;
                    cy += dy;
                }
            }
        }
    }
    return dup_str("NO");
}
```
