## 解题思路

按题意模拟即可：

1. 将 `lettersStr` 按字符 ASCII 升序排序（稳定与否不影响，因只比码值）；
2. 对每个字符求 $pos=1..26$，再 $newPos=(pos^2\bmod 26)+1$，映射为大写字母；
3. 若原字符为大写，结果改小写，否则改大写。

**注意**：大小写取反相对的是**排序后**该位置的原字符，不是变换前未排序串中的对应字符。  
**常见假解**：先变换再排序；对 $26$ 取余后忘记 $+1$；大小写取反相对输入原串下标而非排序后字符。

## 复杂度分析

- 时间：$O(n\log n)$，主导为排序；$n\le 10^4$。
- 空间：$O(n)$ 存放结果（若原地排序则额外 $O(1)$ 辅助，不计输出）。

## 代码实现

### Python

```python
class Solution:
    def sortLetter(self, lettersStr: str) -> str:
        # ASCII 升序
        chars = sorted(lettersStr)
        out = []
        for c in chars:
            pos = ord(c.lower()) - ord("a") + 1
            # (pos^2 % 26) + 1 → 1..26
            new_pos = (pos * pos) % 26 + 1
            ch = chr(ord("A") + new_pos - 1)
            # 与排序后原字符大小写相反
            out.append(ch.lower() if c.isupper() else ch.upper())
        return "".join(out)
```

### Java

```java
import java.util.*;

public class Solution {
    public String sortLetter(String lettersStr) {
        char[] a = lettersStr.toCharArray();
        Arrays.sort(a); // ASCII 升序
        StringBuilder sb = new StringBuilder(a.length);
        for (char c : a) {
            int pos = Character.toLowerCase(c) - 'a' + 1;
            int np = (pos * pos) % 26 + 1;
            char ch = (char) ('A' + np - 1);
            // 大小写取反相对排序后的原字符
            sb.append(Character.isUpperCase(c) ? Character.toLowerCase(ch)
                                               : Character.toUpperCase(ch));
        }
        return sb.toString();
    }
}
```

### C++

```cpp
#include <algorithm>
#include <cctype>
#include <string>

using namespace std;

class Solution {
   public:
    string sortLetter(string lettersStr) {
        sort(lettersStr.begin(), lettersStr.end());
        string out;
        out.reserve(lettersStr.size());
        for (char c : lettersStr) {
            int pos = tolower((unsigned char)c) - 'a' + 1;
            int np = (pos * pos) % 26 + 1;
            char ch = char('A' + np - 1);
            out.push_back(isupper((unsigned char)c) ? (char)tolower(ch)
                                                    : (char)toupper(ch));
        }
        return out;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

static int cmp_char(const void* a, const void* b) {
    return *(const unsigned char*)a - *(const unsigned char*)b;
}

char* sortLetter(char* lettersStr) {
    int n = (int)strlen(lettersStr);
    char* sorted = (char*)malloc(n + 1);
    memcpy(sorted, lettersStr, n + 1);
    qsort(sorted, n, 1, cmp_char);
    char* out = (char*)malloc(n + 1);
    for (int i = 0; i < n; i++) {
        unsigned char c = (unsigned char)sorted[i];
        int pos = tolower(c) - 'a' + 1;
        int np = (pos * pos) % 26 + 1;
        char ch = (char)('A' + np - 1);
        // 原大写 -> 小写结果；原小写 -> 大写结果
        out[i] = isupper(c) ? (char)tolower((unsigned char)ch)
                            : (char)toupper((unsigned char)ch);
    }
    out[n] = '\0';
    free(sorted);
    return out;
}
```
