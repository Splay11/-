## 解题思路

本题是**树形（森林）结构上的设计题**：每个 span 至多有一个父节点；`inherit=false` 时开新根，形成多棵树并存。

维护：

1. `parent` / `children`：父子关系；
2. `order`：进入顺序列表（仍存活的 span）；
3. `latest`：当前最新 span（即 `order` 末尾）。

**enter**：若 `inherit` 且已有 span，则挂到 `latest` 下，否则无父；再把新 id 追加到 `order` 并更新 `latest`。

**log**：无 span 时返回 `msg`；否则从 `latest` 沿 `parent` 走到根，反转后用 `-` 拼接，再拼 `": " + msg`。

**leave**：从目标 id DFS/BFS 收集自身与全部子孙，从父的孩子列表断开，删掉这些节点，过滤 `order`，`latest` 取剩余 `order` 末尾（或置空）。

**常见假解**：

- `leave` 只删自身不删子孙；
- `leave` 后把 `latest` 错设为父节点，而不是「剩余中最后进入者」（例如删掉较新的旁支树后，应回到另一支上最晚进入的节点）；
- `log` 路径方向颠倒或漏写冒号后空格。

## 复杂度分析

- 单次操作在调用总数 $C\le 100$、span 个数同阶时，均可视为 $O(C)$；整次评测 $O(C^2)$ 可接受。
- 空间：$O(C)$。

## 代码实现

### Python

```python
from typing import Dict, List, Optional, Set


class LogSystem:
    def __init__(self):
        self.parent: Dict[int, Optional[int]] = {}
        self.children: Dict[int, List[int]] = {}
        self.order: List[int] = []
        self.latest: Optional[int] = None

    def enter(self, spanId: int, inherit: bool) -> None:
        p = None
        # inherit 且已有 span 时挂到当前最新
        if inherit and self.parent:
            p = self.latest
        self.parent[spanId] = p
        self.children[spanId] = []
        if p is not None:
            self.children[p].append(spanId)
        self.order.append(spanId)
        self.latest = spanId

    def log(self, msg: str) -> str:
        if self.latest is None:
            return msg
        path = []
        cur = self.latest
        # 自最新走到根，再反转为根→最新
        while cur is not None:
            path.append(cur)
            cur = self.parent[cur]
        path.reverse()
        return "-".join(str(x) for x in path) + ": " + msg

    def leave(self, spanId: int) -> None:
        doomed: Set[int] = set()

        def dfs(u: int) -> None:
            doomed.add(u)
            for v in self.children.get(u, []):
                dfs(v)

        dfs(spanId)
        p = self.parent.get(spanId)
        if p is not None and p in self.children:
            self.children[p] = [v for v in self.children[p] if v != spanId]
        for u in doomed:
            self.parent.pop(u, None)
            self.children.pop(u, None)
        # 剩余中最后进入者成为最新
        self.order = [x for x in self.order if x not in doomed]
        self.latest = self.order[-1] if self.order else None
```

### Java

```java
import java.util.*;

class LogSystem {
    private Map<Integer, Integer> parent = new HashMap<>();
    private Map<Integer, List<Integer>> children = new HashMap<>();
    private List<Integer> order = new ArrayList<>();
    private Integer latest = null;

    public LogSystem() {}

    public void enter(int spanId, boolean inherit) {
        Integer p = null;
        // inherit 且已有 span 时挂到当前最新
        if (inherit && !parent.isEmpty()) p = latest;
        parent.put(spanId, p);
        children.put(spanId, new ArrayList<>());
        if (p != null) children.get(p).add(spanId);
        order.add(spanId);
        latest = spanId;
    }

    public String log(String msg) {
        if (latest == null) return msg;
        ArrayList<Integer> path = new ArrayList<>();
        Integer cur = latest;
        while (cur != null) {
            path.add(cur);
            cur = parent.get(cur);
        }
        Collections.reverse(path);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < path.size(); i++) {
            if (i > 0) sb.append('-');
            sb.append(path.get(i));
        }
        sb.append(": ").append(msg);
        return sb.toString();
    }

    public void leave(int spanId) {
        HashSet<Integer> doomed = new HashSet<>();
        ArrayDeque<Integer> dq = new ArrayDeque<>();
        dq.add(spanId);
        while (!dq.isEmpty()) {
            int u = dq.poll();
            if (!doomed.add(u)) continue;
            for (int v : children.getOrDefault(u, Collections.emptyList())) dq.add(v);
        }
        Integer p = parent.get(spanId);
        if (p != null) children.get(p).remove(Integer.valueOf(spanId));
        for (int u : doomed) {
            parent.remove(u);
            children.remove(u);
        }
        ArrayList<Integer> nord = new ArrayList<>();
        for (int x : order) if (!doomed.contains(x)) nord.add(x);
        order = nord;
        latest = order.isEmpty() ? null : order.get(order.size() - 1);
    }
}

public class Solution {}
```

### C++

```cpp
#include <algorithm>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <vector>

using namespace std;

class LogSystem {
    unordered_map<int, int> parent;  // -1 = 无父
    unordered_map<int, vector<int>> children;
    vector<int> order;
    int latest;
    bool hasLatest;

   public:
    LogSystem() : latest(-1), hasLatest(false) {}

    void enter(int spanId, bool inherit) {
        int p = -1;
        if (inherit && !parent.empty()) p = latest;
        parent[spanId] = p;
        children[spanId] = {};
        if (p != -1) children[p].push_back(spanId);
        order.push_back(spanId);
        latest = spanId;
        hasLatest = true;
    }

    string log(string msg) {
        if (!hasLatest) return msg;
        vector<int> path;
        int cur = latest;
        while (cur != -1) {
            path.push_back(cur);
            cur = parent[cur];
        }
        reverse(path.begin(), path.end());
        string s;
        for (size_t i = 0; i < path.size(); i++) {
            if (i) s.push_back('-');
            s += to_string(path[i]);
        }
        s += ": ";
        s += msg;
        return s;
    }

    void leave(int spanId) {
        unordered_set<int> doomed;
        vector<int> st = {spanId};
        while (!st.empty()) {
            int u = st.back();
            st.pop_back();
            if (!doomed.insert(u).second) continue;
            for (int v : children[u]) st.push_back(v);
        }
        int p = parent[spanId];
        if (p != -1) {
            auto& ch = children[p];
            ch.erase(remove(ch.begin(), ch.end(), spanId), ch.end());
        }
        for (int u : doomed) {
            parent.erase(u);
            children.erase(u);
        }
        vector<int> nord;
        for (int x : order)
            if (!doomed.count(x)) nord.push_back(x);
        order.swap(nord);
        if (order.empty()) {
            hasLatest = false;
            latest = -1;
        } else {
            hasLatest = true;
            latest = order.back();
        }
    }
};
```

### C

```c
#include <stdbool.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct LogSystem LogSystem;

#define MAX_SPAN 1001

struct LogSystem {
    int parent[MAX_SPAN];
    int* children[MAX_SPAN];
    int childCnt[MAX_SPAN];
    int childCap[MAX_SPAN];
    char alive[MAX_SPAN];
    int* order;
    int orderCnt;
    int orderCap;
    int latest;
    int hasLatest;
};

LogSystem* logSystemCreate(void) {
    LogSystem* obj = (LogSystem*)calloc(1, sizeof(LogSystem));
    for (int i = 0; i < MAX_SPAN; i++) obj->parent[i] = -1;
    obj->latest = -1;
    obj->hasLatest = 0;
    return obj;
}

void logSystemFree(LogSystem* obj) {
    if (!obj) return;
    for (int i = 0; i < MAX_SPAN; i++) free(obj->children[i]);
    free(obj->order);
    free(obj);
}

static void pushChild(LogSystem* obj, int p, int c) {
    if (obj->childCnt[p] == obj->childCap[p]) {
        int nc = obj->childCap[p] ? obj->childCap[p] * 2 : 4;
        obj->children[p] = (int*)realloc(obj->children[p], sizeof(int) * (size_t)nc);
        obj->childCap[p] = nc;
    }
    obj->children[p][obj->childCnt[p]++] = c;
}

static void removeChild(LogSystem* obj, int p, int c) {
    int* ch = obj->children[p];
    int n = obj->childCnt[p], w = 0;
    for (int i = 0; i < n; i++)
        if (ch[i] != c) ch[w++] = ch[i];
    obj->childCnt[p] = w;
}

void logSystemEnter(LogSystem* obj, int spanId, bool inherit) {
    int p = -1;
    // 有任意活跃 span 时才可能 inherit；用 hasLatest 近似 parent 非空
    if (inherit && obj->hasLatest) p = obj->latest;
    obj->parent[spanId] = p;
    obj->alive[spanId] = 1;
    obj->childCnt[spanId] = 0;
    if (p != -1) pushChild(obj, p, spanId);
    if (obj->orderCnt == obj->orderCap) {
        int nc = obj->orderCap ? obj->orderCap * 2 : 8;
        obj->order = (int*)realloc(obj->order, sizeof(int) * (size_t)nc);
        obj->orderCap = nc;
    }
    obj->order[obj->orderCnt++] = spanId;
    obj->latest = spanId;
    obj->hasLatest = 1;
}

char* logSystemLog(LogSystem* obj, char* msg) {
    if (!obj->hasLatest) {
        char* r = (char*)malloc(strlen(msg) + 1);
        strcpy(r, msg);
        return r;
    }
    int path[MAX_SPAN], plen = 0;
    int cur = obj->latest;
    while (cur != -1) {
        path[plen++] = cur;
        cur = obj->parent[cur];
    }
    // reverse
    for (int i = 0, j = plen - 1; i < j; i++, j--) {
        int t = path[i];
        path[i] = path[j];
        path[j] = t;
    }
    char buf[512];
    int pos = 0;
    for (int i = 0; i < plen; i++) {
        if (i) buf[pos++] = '-';
        pos += sprintf(buf + pos, "%d", path[i]);
    }
    pos += sprintf(buf + pos, ": %s", msg);
    char* r = (char*)malloc((size_t)pos + 1);
    memcpy(r, buf, (size_t)pos + 1);
    return r;
}

void logSystemLeave(LogSystem* obj, int spanId) {
    char doomed[MAX_SPAN];
    memset(doomed, 0, sizeof(doomed));
    int st[MAX_SPAN], top = 0;
    st[top++] = spanId;
    while (top) {
        int u = st[--top];
        if (doomed[u]) continue;
        doomed[u] = 1;
        for (int i = 0; i < obj->childCnt[u]; i++) st[top++] = obj->children[u][i];
    }
    int p = obj->parent[spanId];
    if (p != -1) removeChild(obj, p, spanId);
    for (int u = 0; u < MAX_SPAN; u++) {
        if (!doomed[u]) continue;
        obj->alive[u] = 0;
        obj->parent[u] = -1;
        obj->childCnt[u] = 0;
    }
    int w = 0;
    for (int i = 0; i < obj->orderCnt; i++)
        if (!doomed[obj->order[i]]) obj->order[w++] = obj->order[i];
    obj->orderCnt = w;
    if (obj->orderCnt == 0) {
        obj->hasLatest = 0;
        obj->latest = -1;
    } else {
        obj->hasLatest = 1;
        obj->latest = obj->order[obj->orderCnt - 1];
    }
}
```
