## 解题思路

本题要求对数组做**按奇偶分组的局部排序**，而不是整体排序，也不是按下标奇偶划分。

**关键观察**：先扫描一遍，收集「当前值为奇数」的下标集合 $I_{\mathrm{odd}}$，以及「当前值为偶数」的下标集合 $I_{\mathrm{even}}$。然后：

1. 取出 $I_{\mathrm{odd}}$ 上的数值，**升序**排序后按原下标顺序写回；
2. 取出 $I_{\mathrm{even}}$ 上的数值，**降序**排序后按原下标顺序写回。

这是典型的「**分组 + 排序 + 回填**」：分组键是值的奇偶性，位置集合由分组结果决定。

**常见假解**：

- 误把「奇数位置」理解成**奇下标**（按下标奇偶分别排序）——应用「值奇偶」与「下标奇偶」不一致的数据即可卡掉；
- 奇数组降序 / 偶数组升序（方向搞反）；
- 对整个数组排序后试图再还原奇偶位置。

## 复杂度分析

- 时间：$O(n\log n)$，主导代价是对奇数组与偶数组分别排序，$n=\texttt{numbers.length}$；
- 空间：$O(n)$，存放下标列表与待排序的副本。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def sortOddEven(self, numbers: List[int]) -> List[int]:
        n = len(numbers)
        ans = numbers[:]
        # 按「当前值的奇偶」收集下标，而非按下标奇偶
        odd_idx = [i for i in range(n) if numbers[i] % 2 == 1]
        even_idx = [i for i in range(n) if numbers[i] % 2 == 0]
        # 奇数升序、偶数降序
        odds = sorted(numbers[i] for i in odd_idx)
        evens = sorted((numbers[i] for i in even_idx), reverse=True)
        for k, i in enumerate(odd_idx):
            ans[i] = odds[k]
        for k, i in enumerate(even_idx):
            ans[i] = evens[k]
        return ans
```

### Java

```java
import java.util.*;

public class Solution {
    public int[] sortOddEven(int[] numbers) {
        int n = numbers.length;
        int[] ans = Arrays.copyOf(numbers, n);
        ArrayList<Integer> oddIdx = new ArrayList<>();
        ArrayList<Integer> evenIdx = new ArrayList<>();
        // 按值的奇偶收集下标
        for (int i = 0; i < n; i++) {
            if (numbers[i] % 2 != 0) oddIdx.add(i);
            else evenIdx.add(i);
        }
        ArrayList<Integer> odds = new ArrayList<>();
        ArrayList<Integer> evens = new ArrayList<>();
        for (int i : oddIdx) odds.add(numbers[i]);
        for (int i : evenIdx) evens.add(numbers[i]);
        // 奇数升序、偶数降序后回填
        Collections.sort(odds);
        Collections.sort(evens, Collections.reverseOrder());
        for (int k = 0; k < oddIdx.size(); k++) ans[oddIdx.get(k)] = odds.get(k);
        for (int k = 0; k < evenIdx.size(); k++) ans[evenIdx.get(k)] = evens.get(k);
        return ans;
    }
}
```

```cpp
class Solution {

                odd_idx.push_back(i);
            else
                even_idx.push_back(i);

        vector<int> odds, evens;

        for (int i : even_idx) evens.push_back(numbers[i]);
        // 奇数升序、偶数降序后回填
```

### C

```c
#include <stdlib.h>

static int cmp_asc(const void* a, const void* b) {
    return *(const int*)a - *(const int*)b;
}
static int cmp_desc(const void* a, const void* b) {
    return *(const int*)b - *(const int*)a;
}

int* sortOddEven(int* numbers, int numbersSize, int* returnSize) {
    int n = numbersSize;
    int* ans = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) ans[i] = numbers[i];
    int* odd_idx = (int*)malloc(n * sizeof(int));
    int* even_idx = (int*)malloc(n * sizeof(int));
    int oc = 0, ec = 0;
    // 收集奇数位与偶数位下标
    for (int i = 0; i < n; i++) {
        if (numbers[i] % 2 != 0) odd_idx[oc++] = i;
        else even_idx[ec++] = i;
    }
    int* odds = (int*)malloc(oc * sizeof(int));
    int* evens = (int*)malloc(ec * sizeof(int));
    for (int k = 0; k < oc; k++) odds[k] = numbers[odd_idx[k]];
    for (int k = 0; k < ec; k++) evens[k] = numbers[even_idx[k]];
    // 奇数升序，偶数降序
    qsort(odds, oc, sizeof(int), cmp_asc);
    qsort(evens, ec, sizeof(int), cmp_desc);
    for (int k = 0; k < oc; k++) ans[odd_idx[k]] = odds[k];
    for (int k = 0; k < ec; k++) ans[even_idx[k]] = evens[k];
    free(odd_idx); free(even_idx); free(odds); free(evens);
    *returnSize = n;
    return ans;
}
```
