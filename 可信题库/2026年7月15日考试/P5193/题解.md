## 解题思路

合法单词只能来自：

1. **某一行**上从左到右的连续子串；或
2. **某一列**上从上到下的连续子串。

不允许从右往左、从下往上，也不允许折线 / 对角线。

由于 $n,m\le 10$，可直接枚举所有行子串与列子串，放入哈希集合；再对 `words` 中每个单词查询是否出现。不同单词之间格子可复用，因此不必考虑占用冲突。

**常见假解**：

- 允许反向（右→左 / 下→上）→ 样例中 `XC` 会误判为可行；
- 允许任意路径（类似 Word Search）→ `ACX` 等会误通过；
- 只检查行、漏掉列（或反之）。

## 复杂度分析

- 时间：$O(n^2 m + n m^2 + \sum |w_i|)$ 量级——枚举全部子串并哈希；查询单词与集合大小相关。在本题约束下可视为常数级。
- 空间：$O(n^2 m + n m^2)$，存储全部子串。

## 代码实现

### Python

```python
from typing import List


class Solution:
    def countMatchedWords(self, charMatrix: List[str], words: List[str]) -> int:
        n = len(charMatrix)
        m = len(charMatrix[0]) if n else 0
        seen = set()

        # 枚举每一行从左到右的连续子串
        for r in range(n):
            row = charMatrix[r]
            for i in range(m):
                for j in range(i, m):
                    seen.add(row[i : j + 1])

        # 枚举每一列从上到下的连续子串
        for c in range(m):
            col = "".join(charMatrix[r][c] for r in range(n))
            for i in range(n):
                for j in range(i, n):
                    seen.add(col[i : j + 1])

        return sum(1 for w in words if w in seen)
```
### c++

```cpp
#include <string>
#include <unordered_set>
#include <vector>

using namespace std;

class Solution {
   public:
    int countMatchedWords(vector<string>& charMatrix, vector<string>& words) {
        int n = (int)charMatrix.size();
        int m = n ? (int)charMatrix[0].size() : 0;
        unordered_set<string> seen;

        // 行：从左到右
        for (int r = 0; r < n; r++) {
            const string& row = charMatrix[r];
            for (int i = 0; i < m; i++)
                for (int j = i; j < m; j++)
                    seen.insert(row.substr(i, j - i + 1));
        }

        // 列：从上到下
        for (int c = 0; c < m; c++) {
            string col;
            for (int r = 0; r < n; r++) col.push_back(charMatrix[r][c]);
            for (int i = 0; i < n; i++)
                for (int j = i; j < n; j++)
                    seen.insert(col.substr(i, j - i + 1));
        }

        int ans = 0;
        for (const string& w : words)
            if (seen.count(w)) ans++;
        return ans;
    }
};
```

### Java

```java
import java.util.*;

public class Solution {
    public int countMatchedWords(String[] charMatrix, String[] words) {
        int n = charMatrix.length;
        int m = n == 0 ? 0 : charMatrix[0].length();
        HashSet<String> seen = new HashSet<>();

        // 行：从左到右
        for (int r = 0; r < n; r++) {
            String row = charMatrix[r];
            for (int i = 0; i < m; i++)
                for (int j = i; j < m; j++)
                    seen.add(row.substring(i, j + 1));
        }

        // 列：从上到下
        for (int c = 0; c < m; c++) {
            StringBuilder sb = new StringBuilder(n);
            for (int r = 0; r < n; r++) sb.append(charMatrix[r].charAt(c));
            String col = sb.toString();
            for (int i = 0; i < n; i++)
                for (int j = i; j < n; j++)
                    seen.add(col.substring(i, j + 1));
        }

        int ans = 0;
        for (String w : words)
            if (seen.contains(w)) ans++;
        return ans;
    }
}
```

### C

```c
#include <stdlib.h>
#include <string.h>

#define MAX_SUB 2048
#define MAX_LEN 16

int countMatchedWords(char** charMatrix, int charMatrixSize, char** words, int wordsSize) {
    int n = charMatrixSize;
    int m = n ? (int)strlen(charMatrix[0]) : 0;
    // 枚举全部合法子串并去重
    char seen[MAX_SUB][MAX_LEN];
    int seenCnt = 0;

    // 行：从左到右
    for (int r = 0; r < n; r++) {
        for (int i = 0; i < m; i++) {
            for (int j = i; j < m; j++) {
                int len = j - i + 1;
                char buf[MAX_LEN];
                memcpy(buf, charMatrix[r] + i, (size_t)len);
                buf[len] = '\0';
                int ok = 1;
                for (int t = 0; t < seenCnt; t++) {
                    if (strcmp(seen[t], buf) == 0) {
                        ok = 0;
                        break;
                    }
                }
                if (ok) strcpy(seen[seenCnt++], buf);
            }
        }
    }

    // 列：从上到下
    for (int c = 0; c < m; c++) {
        char col[MAX_LEN];
        for (int r = 0; r < n; r++) col[r] = charMatrix[r][c];
        col[n] = '\0';
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                int len = j - i + 1;
                char buf[MAX_LEN];
                memcpy(buf, col + i, (size_t)len);
                buf[len] = '\0';
                int ok = 1;
                for (int t = 0; t < seenCnt; t++) {
                    if (strcmp(seen[t], buf) == 0) {
                        ok = 0;
                        break;
                    }
                }
                if (ok) strcpy(seen[seenCnt++], buf);
            }
        }
    }

    int ans = 0;
    for (int i = 0; i < wordsSize; i++) {
        for (int t = 0; t < seenCnt; t++) {
            if (strcmp(seen[t], words[i]) == 0) {
                ans++;
                break;
            }
        }
    }
    return ans;
}
```
