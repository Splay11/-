## 解题思路

本题用队列模拟日志文件。每份文件记下模块号、序号和体积；另外用哈希表指向每个模块的当前文件，以及该模块下一份序号。

$putLog(mid, nbytes)$ 分两步：

1. **总额**：当 $totalSize + nbytes > totalCap$ 时，按生成顺序删掉最早的文件。若删掉的正是某模块当前文件，清掉该模块的当前指针。
2. **单文件限**：没有当前文件，或当前文件再写会越过 $fileCap$，就开一份新文件（序号加一）作为当前文件。然后把 $nbytes$ 加进当前文件，返回当前文件体积。

$listFiles$ 按队列顺序输出 $[mid, idx, sz]$。$totalSize$ 对仍在的文件求和。

不要在「因超单文件上限而关掉当前文件」时把它删掉，旧文件仍占额度。

## 复杂度分析

- 单次 $putLog$：删文件均摊 $O(1)$，开文件 $O(1)$。
- 单次 $listFiles$ / $totalSize$：$O(m)$，$m$ 为仍在的文件数。
- 累计调用 $q \le 100$，整体时间 $O(q^2)$ 可接受。
- 空间复杂度：$O(q)$。

## 代码实现

### Python

```python
from typing import Dict, List


class LogChunk:
    def __init__(self, mid: int, idx: int) -> None:
        self.mid = mid
        self.idx = idx
        self.sz = 0


class FileLogger:
    def __init__(self, fileCap: int, totalCap: int) -> None:
        self.cap = fileCap
        self.quota = totalCap
        self.files: List[LogChunk] = []
        self.cur: Dict[int, LogChunk] = {}
        self.seq: Dict[int, int] = {}

    def totalSize(self) -> int:
        return sum(f.sz for f in self.files)

    def _drop_oldest(self) -> None:
        # 按创建先后删掉最早的一份文件
        f = self.files.pop(0)
        if self.cur.get(f.mid) is f:
            del self.cur[f.mid]

    def putLog(self, mid: int, nbytes: int) -> int:
        while self.totalSize() + nbytes > self.quota:
            self._drop_oldest()
        cur = self.cur.get(mid)
        if cur is None or cur.sz + nbytes > self.cap:
            # 无当前文件或再写会超单文件上限，开新文件
            self.seq[mid] = self.seq.get(mid, 0) + 1
            cur = LogChunk(mid, self.seq[mid])
            self.files.append(cur)
            self.cur[mid] = cur
        cur.sz += nbytes
        return cur.sz

    def listFiles(self) -> List[List[int]]:
        return [[f.mid, f.idx, f.sz] for f in self.files]
```

### Java

```java
import java.util.*;

class FileLogger {
    static class LogChunk {
        int mid, idx, sz;
        LogChunk(int mid, int idx) {
            this.mid = mid;
            this.idx = idx;
        }
    }

    private int cap, quota;
    private List<LogChunk> files = new ArrayList<>();
    private Map<Integer, LogChunk> cur = new HashMap<>();
    private Map<Integer, Integer> seq = new HashMap<>();

    public FileLogger(int fileCap, int totalCap) {
        this.cap = fileCap;
        this.quota = totalCap;
    }

    public int totalSize() {
        int s = 0;
        for (LogChunk f : files) s += f.sz;
        return s;
    }

    private void dropOldest() {
        // 按创建先后删掉最早的一份文件
        LogChunk f = files.remove(0);
        if (cur.get(f.mid) == f) cur.remove(f.mid);
    }

    public int putLog(int mid, int nbytes) {
        while (totalSize() + nbytes > quota) dropOldest();
        LogChunk now = cur.get(mid);
        if (now == null || now.sz + nbytes > cap) {
            // 开新文件
            int nxt = seq.getOrDefault(mid, 0) + 1;
            seq.put(mid, nxt);
            now = new LogChunk(mid, nxt);
            files.add(now);
            cur.put(mid, now);
        }
        now.sz += nbytes;
        return now.sz;
    }

    public int[][] listFiles() {
        int[][] ans = new int[files.size()][3];
        for (int i = 0; i < files.size(); i++) {
            LogChunk f = files.get(i);
            ans[i][0] = f.mid;
            ans[i][1] = f.idx;
            ans[i][2] = f.sz;
        }
        return ans;
    }
}
```

### C++

```cpp
#include <unordered_map>
#include <vector>
using namespace std;

class FileLogger {
    struct LogChunk {
        int mid, idx, sz;
        LogChunk(int m, int i) : mid(m), idx(i), sz(0) {}
    };
    int cap, quota;
    vector<LogChunk*> files;
    unordered_map<int, LogChunk*> cur;
    unordered_map<int, int> seq;

    int usedSum() {
        int s = 0;
        for (auto* f : files) s += f->sz;
        return s;
    }

    void dropOldest() {
        // 按创建先后删掉最早的一份文件
        LogChunk* f = files.front();
        files.erase(files.begin());
        if (cur.count(f->mid) && cur[f->mid] == f) cur.erase(f->mid);
        delete f;
    }

public:
    FileLogger(int fileCap, int totalCap) : cap(fileCap), quota(totalCap) {}
    ~FileLogger() {
        for (auto* f : files) delete f;
    }

    int totalSize() { return usedSum(); }

    int putLog(int mid, int nbytes) {
        while (usedSum() + nbytes > quota) dropOldest();
        LogChunk* now = cur.count(mid) ? cur[mid] : nullptr;
        if (!now || now->sz + nbytes > cap) {
            // 开新文件
            int nxt = seq[mid] + 1;
            seq[mid] = nxt;
            now = new LogChunk(mid, nxt);
            files.push_back(now);
            cur[mid] = now;
        }
        now->sz += nbytes;
        return now->sz;
    }

    vector<vector<int>> listFiles() {
        vector<vector<int>> ans;
        for (auto* f : files) ans.push_back({f->mid, f->idx, f->sz});
        return ans;
    }
};
```

### JavaScript

```javascript
class FileLogger {
    constructor(fileCap, totalCap) {
        this.cap = fileCap;
        this.quota = totalCap;
        this.files = [];
        this.cur = new Map();
        this.seq = new Map();
    }

    totalSize() {
        return this.files.reduce((s, f) => s + f.sz, 0);
    }

    _dropOldest() {
        // 按创建先后删掉最早的一份文件
        const f = this.files.shift();
        if (this.cur.get(f.mid) === f) this.cur.delete(f.mid);
    }

    putLog(mid, nbytes) {
        while (this.totalSize() + nbytes > this.quota) this._dropOldest();
        let now = this.cur.get(mid);
        if (!now || now.sz + nbytes > this.cap) {
            // 开新文件
            const nxt = (this.seq.get(mid) || 0) + 1;
            this.seq.set(mid, nxt);
            now = { mid, idx: nxt, sz: 0 };
            this.files.push(now);
            this.cur.set(mid, now);
        }
        now.sz += nbytes;
        return now.sz;
    }

    listFiles() {
        return this.files.map((f) => [f.mid, f.idx, f.sz]);
    }
}
```

### C

```c
#include <stdlib.h>
#include <string.h>

typedef struct {
    int mid, idx, sz;
} logChunk;

typedef struct {
    int cap, quota, n;
    logChunk files[128];
    int seqG[128], seqV[128], sc;
    int curG[128], curI[128], cc; /* curI 为 files 下标，-1 表示没有当前文件 */
} fileLogger;

static int nrUsed(fileLogger* o) {
    int s = 0;
    for (int i = 0; i < o->n; i++) s += o->files[i].sz;
    return s;
}

static int nrGetSeq(fileLogger* o, int mid) {
    for (int i = 0; i < o->sc; i++) if (o->seqG[i] == mid) return i;
    o->seqG[o->sc] = mid;
    o->seqV[o->sc] = 0;
    return o->sc++;
}

static int nrGetCur(fileLogger* o, int mid) {
    for (int i = 0; i < o->cc; i++) if (o->curG[i] == mid) return i;
    o->curG[o->cc] = mid;
    o->curI[o->cc] = -1;
    return o->cc++;
}

static void nrDrop(fileLogger* o) {
    /* 丢掉下标 0，后面的下标都减一 */
    for (int i = 0; i < o->cc; i++) {
        if (o->curI[i] == 0) o->curI[i] = -1;
        else if (o->curI[i] > 0) o->curI[i]--;
    }
    memmove(o->files, o->files + 1, (o->n - 1) * sizeof(logChunk));
    o->n--;
}

fileLogger* fileLoggerCreate(int fileCap, int totalCap) {
    fileLogger* o = (fileLogger*)calloc(1, sizeof(fileLogger));
    o->cap = fileCap;
    o->quota = totalCap;
    return o;
}

void fileLoggerFree(fileLogger* o) { free(o); }

int fileLoggerTotalSize(fileLogger* o) { return nrUsed(o); }

int fileLoggerPutLog(fileLogger* o, int mid, int nbytes) {
    while (nrUsed(o) + nbytes > o->quota) nrDrop(o);
    int ci = nrGetCur(o, mid);
    int now = o->curI[ci];
    if (now < 0 || o->files[now].sz + nbytes > o->cap) {
        int si = nrGetSeq(o, mid);
        o->seqV[si]++;
        now = o->n++;
        o->files[now].mid = mid;
        o->files[now].idx = o->seqV[si];
        o->files[now].sz = 0;
        o->curI[ci] = now;
    }
    o->files[now].sz += nbytes;
    return o->files[now].sz;
}

logChunk* fileLoggerListFiles(fileLogger* o, int* returnSize) {
    *returnSize = o->n;
    return o->files;
}
```
