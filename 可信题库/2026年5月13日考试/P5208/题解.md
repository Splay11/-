## 解题思路

用两个列表维护新生代 / 老年代对象 id，并用哈希表存每个对象的 `cnt` 与是否可回收。

- **createObject**：放入新生代；若数量 $\ge$ `youngSize`，立即执行新生代 GC。
- **新生代 GC**：先删可回收 → 剩余 `cnt++` → `cnt>=2` 晋升老年代。
- **老年代 GC**：删可回收 → 剩余 `cnt++`（不晋升）。
- **getLiveObjects**：按 `(-cnt, -id)` 排序输出。

**常见假解**：

- 在插入前就按阈值触发 GC；
- 晋升条件写成经历两次 GC 才晋升但 `cnt` 计数错误；
- 排序写成 id 升序。

## 复杂度分析

- 单次操作与当前对象数线性相关；调用次数 $\le 200$，对象数同阶，整体可接受。
- 空间：$O(\text{对象数})$。

## 代码实现

### Python

```python
from typing import Dict, List


class GCSystem:
    def __init__(self, youngSize: int):
        self.young_size = youngSize
        self.objs: Dict[int, dict] = {}
        self.young: List[int] = []
        self.old: List[int] = []

    def createObject(self, objectId: int) -> None:
        self.objs[objectId] = {"cnt": 0, "reclaimable": False}
        self.young.append(objectId)
        # 加入后达到阈值则自动新生代 GC
        if len(self.young) >= self.young_size:
            self._gc_young()

    def markObjects(self, objectIds: List[int]) -> None:
        for oid in objectIds:
            self.objs[oid]["reclaimable"] = True

    def manualGC(self, generation: int) -> None:
        if generation == 0:
            self._gc_young()
        else:
            self._gc_old()

    def getLiveObjects(self, generation: int) -> List[int]:
        ids = list(self.young if generation == 0 else self.old)
        ids.sort(key=lambda x: (-self.objs[x]["cnt"], -x))
        return ids

    def _gc_young(self) -> None:
        remain = []
        for oid in self.young:
            if self.objs[oid]["reclaimable"]:
                del self.objs[oid]
            else:
                remain.append(oid)
        self.young = []
        for oid in remain:
            self.objs[oid]["cnt"] += 1
            if self.objs[oid]["cnt"] >= 2:
                self.old.append(oid)
            else:
                self.young.append(oid)

    def _gc_old(self) -> None:
        remain = []
        for oid in self.old:
            if self.objs[oid]["reclaimable"]:
                del self.objs[oid]
            else:
                remain.append(oid)
        self.old = []
        for oid in remain:
            self.objs[oid]["cnt"] += 1
            self.old.append(oid)
```

### Java

```java
import java.util.*;

class GCSystem {
    private static class Obj {
        int cnt;
        boolean reclaimable;
    }

    private final int youngSize;
    private final Map<Integer, Obj> objs = new HashMap<>();
    private final List<Integer> young = new ArrayList<>();
    private final List<Integer> old = new ArrayList<>();

    public GCSystem(int youngSize) { this.youngSize = youngSize; }

    public void createObject(int objectId) {
        objs.put(objectId, new Obj());
        young.add(objectId);
        if (young.size() >= youngSize) gcYoung();
    }

    public void markObjects(int[] objectIds) {
        for (int oid : objectIds) objs.get(oid).reclaimable = true;
    }

    public void manualGC(int generation) {
        if (generation == 0) gcYoung();
        else gcOld();
    }

    public int[] getLiveObjects(int generation) {
        List<Integer> ids = new ArrayList<>(generation == 0 ? young : old);
        ids.sort((a, b) -> {
            int ca = objs.get(a).cnt, cb = objs.get(b).cnt;
            if (ca != cb) return Integer.compare(cb, ca);
            return Integer.compare(b, a);
        });
        int[] ans = new int[ids.size()];
        for (int i = 0; i < ids.size(); i++) ans[i] = ids.get(i);
        return ans;
    }

    private void gcYoung() {
        List<Integer> remain = new ArrayList<>();
        for (int oid : young) {
            if (objs.get(oid).reclaimable) objs.remove(oid);
            else remain.add(oid);
        }
        young.clear();
        for (int oid : remain) {
            objs.get(oid).cnt += 1;
            if (objs.get(oid).cnt >= 2) old.add(oid);
            else young.add(oid);
        }
    }

    private void gcOld() {
        List<Integer> remain = new ArrayList<>();
        for (int oid : old) {
            if (objs.get(oid).reclaimable) objs.remove(oid);
            else remain.add(oid);
        }
        old.clear();
        for (int oid : remain) {
            objs.get(oid).cnt += 1;
            old.add(oid);
        }
    }
}

public class Solution {}
```

### C++

```cpp
#include <algorithm>
#include <unordered_map>
#include <vector>

using namespace std;

class GCSystem {
    struct Obj {
        int cnt;
        bool reclaimable;
    };
    int youngSize;
    unordered_map<int, Obj> objs;
    vector<int> young, old;

    void gcYoung() {
        vector<int> remain;
        for (int oid : young) {
            if (objs[oid].reclaimable) objs.erase(oid);
            else remain.push_back(oid);
        }
        young.clear();
        for (int oid : remain) {
            objs[oid].cnt += 1;
            if (objs[oid].cnt >= 2) old.push_back(oid);
            else young.push_back(oid);
        }
    }

    void gcOld() {
        vector<int> remain;
        for (int oid : old) {
            if (objs[oid].reclaimable) objs.erase(oid);
            else remain.push_back(oid);
        }
        old.clear();
        for (int oid : remain) {
            objs[oid].cnt += 1;
            old.push_back(oid);
        }
    }

   public:
    GCSystem(int youngSize) : youngSize(youngSize) {}

    void createObject(int objectId) {
        objs[objectId] = {0, false};
        young.push_back(objectId);
        if ((int)young.size() >= youngSize) gcYoung();
    }

    void markObjects(vector<int> objectIds) {
        for (int oid : objectIds) objs[oid].reclaimable = true;
    }

    void manualGC(int generation) {
        if (generation == 0) gcYoung();
        else gcOld();
    }

    vector<int> getLiveObjects(int generation) {
        vector<int> ids = generation == 0 ? young : old;
        sort(ids.begin(), ids.end(), [&](int a, int b) {
            if (objs[a].cnt != objs[b].cnt) return objs[a].cnt > objs[b].cnt;
            return a > b;
        });
        return ids;
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>

typedef struct GCSystem GCSystem;

typedef struct {
    int cnt;
    char reclaimable;
    char alive;
} Obj;

struct GCSystem {
    int youngSize;
    Obj* objs; // indexed by objectId; grow dynamically
    int objsCap;
    int* young;
    int youngCnt, youngCap;
    int* old;
    int oldCnt, oldCap;
};

static void ensureObj(GCSystem* obj, int id) {
    if (id >= obj->objsCap) {
        int nc = obj->objsCap ? obj->objsCap * 2 : 64;
        while (nc <= id) nc *= 2;
        obj->objs = (Obj*)realloc(obj->objs, sizeof(Obj) * (size_t)nc);
        memset(obj->objs + obj->objsCap, 0, sizeof(Obj) * (size_t)(nc - obj->objsCap));
        obj->objsCap = nc;
    }
}

static void pushInt(int** arr, int* cnt, int* cap, int v) {
    if (*cnt == *cap) {
        int nc = *cap ? *cap * 2 : 8;
        *arr = (int*)realloc(*arr, sizeof(int) * (size_t)nc);
        *cap = nc;
    }
    (*arr)[(*cnt)++] = v;
}

static void gcYoung(GCSystem* obj) {
    int* remain = (int*)malloc(sizeof(int) * (size_t)(obj->youngCnt + 1));
    int rc = 0;
    for (int i = 0; i < obj->youngCnt; i++) {
        int oid = obj->young[i];
        if (obj->objs[oid].reclaimable) {
            obj->objs[oid].alive = 0;
        } else {
            remain[rc++] = oid;
        }
    }
    obj->youngCnt = 0;
    for (int i = 0; i < rc; i++) {
        int oid = remain[i];
        obj->objs[oid].cnt += 1;
        if (obj->objs[oid].cnt >= 2)
            pushInt(&obj->old, &obj->oldCnt, &obj->oldCap, oid);
        else
            pushInt(&obj->young, &obj->youngCnt, &obj->youngCap, oid);
    }
    free(remain);
}

static void gcOld(GCSystem* obj) {
    int* remain = (int*)malloc(sizeof(int) * (size_t)(obj->oldCnt + 1));
    int rc = 0;
    for (int i = 0; i < obj->oldCnt; i++) {
        int oid = obj->old[i];
        if (obj->objs[oid].reclaimable) {
            obj->objs[oid].alive = 0;
        } else {
            remain[rc++] = oid;
        }
    }
    obj->oldCnt = 0;
    for (int i = 0; i < rc; i++) {
        int oid = remain[i];
        obj->objs[oid].cnt += 1;
        pushInt(&obj->old, &obj->oldCnt, &obj->oldCap, oid);
    }
    free(remain);
}

GCSystem* gCSystemCreate(int youngSize) {
    GCSystem* obj = (GCSystem*)calloc(1, sizeof(GCSystem));
    obj->youngSize = youngSize;
    return obj;
}

void gCSystemFree(GCSystem* obj) {
    if (!obj) return;
    free(obj->objs);
    free(obj->young);
    free(obj->old);
    free(obj);
}

void gCSystemCreateObject(GCSystem* obj, int objectId) {
    ensureObj(obj, objectId);
    obj->objs[objectId].cnt = 0;
    obj->objs[objectId].reclaimable = 0;
    obj->objs[objectId].alive = 1;
    pushInt(&obj->young, &obj->youngCnt, &obj->youngCap, objectId);
    if (obj->youngCnt >= obj->youngSize) gcYoung(obj);
}

void gCSystemMarkObjects(GCSystem* obj, int* objectIds, int objectIdsSize) {
    for (int i = 0; i < objectIdsSize; i++) {
        int oid = objectIds[i];
        ensureObj(obj, oid);
        obj->objs[oid].reclaimable = 1;
    }
}

void gCSystemManualGC(GCSystem* obj, int generation) {
    if (generation == 0) gcYoung(obj);
    else gcOld(obj);
}

typedef struct {
    int id, cnt;
} SortItem;

static int cmpLive(const void* a, const void* b) {
    const SortItem* x = (const SortItem*)a;
    const SortItem* y = (const SortItem*)b;
    if (x->cnt != y->cnt) return y->cnt - x->cnt; // cnt 降序
    return y->id - x->id;                         // id 降序
}

int* gCSystemGetLiveObjects(GCSystem* obj, int generation, int* returnSize) {
    int* src = generation == 0 ? obj->young : obj->old;
    int n = generation == 0 ? obj->youngCnt : obj->oldCnt;
    SortItem* items = (SortItem*)malloc(sizeof(SortItem) * (size_t)(n + 1));
    for (int i = 0; i < n; i++) {
        items[i].id = src[i];
        items[i].cnt = obj->objs[src[i]].cnt;
    }
    qsort(items, (size_t)n, sizeof(SortItem), cmpLive);
    int* out = (int*)malloc(sizeof(int) * (size_t)(n + 1));
    for (int i = 0; i < n; i++) out[i] = items[i].id;
    free(items);
    *returnSize = n;
    return out;
}
```
