## 解题思路

合法串形如：`字母 + 0* + 特征段 + 0*`，且去掉字母后的 8 位必须全为数字。

解析方式：

1. 检查长度、首字符为字母、其余为数字；
2. 在其余部分去掉前导 `0` 与尾随 `0`，中间即为特征段；若中间为空则非法；
3. 翻转特征段，再拼回原来的前导/尾随 `0` 个数。

**常见假解**：

- 未校验首字母后是否全数字；
- 把整段数字整体翻转（连同前后的 `0` 一起翻）；
- 全 `0` 后缀仍当合法。

## 复杂度分析

- 时间：$O(1)$（长度固定为 9）；
- 空间：$O(1)$。

## 代码实现

### Python

```python
class Solution:
    def flipWorkId(self, code: str) -> str:
        if len(code) != 9:
            return "ERROR"
        if not code[0].isalpha():
            return "ERROR"
        rest = code[1:]
        if not rest.isdigit():
            return "ERROR"
        # 剥离前导 0 / 尾随 0，中间为特征段
        i = 0
        while i < len(rest) and rest[i] == "0":
            i += 1
        if i == len(rest):
            return "ERROR"
        j = len(rest) - 1
        while j >= i and rest[j] == "0":
            j -= 1
        if j < i:
            return "ERROR"
        lead, trail = i, len(rest) - 1 - j
        feature = rest[i : j + 1]
        return "0" * lead + feature[::-1] + "0" * trail
```

### Java

```java
public class Solution {
    public String flipWorkId(String code) {
        if (code.length() != 9) return "ERROR";
        char c0 = code.charAt(0);
        if (!Character.isLetter(c0)) return "ERROR";
        String rest = code.substring(1);
        for (int k = 0; k < rest.length(); k++) {
            if (!Character.isDigit(rest.charAt(k))) return "ERROR";
        }
        int n = rest.length(), i = 0;
        while (i < n && rest.charAt(i) == '0') i++;
        if (i == n) return "ERROR";
        int j = n - 1;
        while (j >= i && rest.charAt(j) == '0') j--;
        if (j < i) return "ERROR";
        int lead = i, trail = n - 1 - j;
        StringBuilder feature = new StringBuilder(rest.substring(i, j + 1));
        feature.reverse();
        StringBuilder ans = new StringBuilder();
        for (int t = 0; t < lead; t++) ans.append('0');
        ans.append(feature);
        for (int t = 0; t < trail; t++) ans.append('0');
        return ans.toString();
    }
}
```

### C++

```cpp
#include <cctype>
#include <string>

using namespace std;

class Solution {
   public:
    string flipWorkId(string code) {
        if ((int)code.size() != 9) return "ERROR";
        if (!isalpha((unsigned char)code[0])) return "ERROR";
        string rest = code.substr(1);
        for (char c : rest)
            if (!isdigit((unsigned char)c)) return "ERROR";
        int n = (int)rest.size(), i = 0;
        while (i < n && rest[i] == '0') i++;
        if (i == n) return "ERROR";
        int j = n - 1;
        while (j >= i && rest[j] == '0') j--;
        if (j < i) return "ERROR";
        int lead = i, trail = n - 1 - j;
        string feature = rest.substr(i, j - i + 1);
        string rev(feature.rbegin(), feature.rend());
        return string(lead, '0') + rev + string(trail, '0');
    }
};
```

### C

```c
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

static char* dup_str(const char* s) {
    char* r = (char*)malloc(strlen(s) + 1);
    strcpy(r, s);
    return r;
}

char* flipWorkId(char* code) {
    if ((int)strlen(code) != 9) return dup_str("ERROR");
    if (!isalpha((unsigned char)code[0])) return dup_str("ERROR");
    char* rest = code + 1;
    for (int k = 0; rest[k]; k++)
        if (!isdigit((unsigned char)rest[k])) return dup_str("ERROR");
    int n = (int)strlen(rest), i = 0;
    while (i < n && rest[i] == '0') i++;
    if (i == n) return dup_str("ERROR");
    int j = n - 1;
    while (j >= i && rest[j] == '0') j--;
    if (j < i) return dup_str("ERROR");
    int lead = i, trail = n - 1 - j;
    int flen = j - i + 1;
    char* ans = (char*)malloc(lead + flen + trail + 1);
    int p = 0;
    for (int t = 0; t < lead; t++) ans[p++] = '0';
    // 翻转中间特征数字段
    for (int t = 0; t < flen; t++) ans[p++] = rest[j - t];
    for (int t = 0; t < trail; t++) ans[p++] = '0';
    ans[p] = '\0';
    return ans;
}
```
