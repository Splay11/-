## 解题思路

用哈希表 `key → (value, expireAt)` 模拟 **TTL 缓存**。

- **过期判定**：时刻 `now` 满足 `now >= expireAt` 即过期。`get` 遇过期返回 `-1`，但**不删除**条目（仍占容量）。
- **put**：写入/覆盖后若 `|map| > capacity`，按 `(expireAt, key)` 字典序最小淘汰一条（过期最早，其次键更小）。**不**顺带清理其它过期键。
- **purge(now)**：删除所有已过期键，返回删除个数。
- **size**：当前条目数（含已过期未 purge 的）。

调用量 $\le 4000$，容量 $\le 500$，每次淘汰线性扫描即可。

易错点：`get` 误删过期项；淘汰时按插入顺序/LRU；覆盖已有键却仍触发淘汰；`now == expireAt` 当成未过期。

## 复杂度分析

- `put` / `purge`：$O(\textit{capacity})$；`get` / `size`：$O(1)$。
- 空间：$O(\textit{capacity})$。

## 代码实现

### Python

```python
from typing import Dict, Tuple


class TTLCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.data: Dict[int, Tuple[int, int]] = {}  # key -> (value, expireAt)

    def put(self, key: int, value: int, expireAt: int) -> None:
        self.data[key] = (value, expireAt)
        if len(self.data) > self.capacity:
            # 淘汰过期最早；相同时键更小
            victim = min(self.data.items(), key=lambda kv: (kv[1][1], kv[0]))[0]
            del self.data[victim]

    def get(self, key: int, now: int) -> int:
        if key not in self.data:
            return -1
        value, expireAt = self.data[key]
        if now >= expireAt:  # 过期只返回 -1，不删
            return -1
        return value

    def purge(self, now: int) -> int:
        dead = [k for k, (_, exp) in self.data.items() if now >= exp]
        for k in dead:
            del self.data[k]
        return len(dead)

    def size(self) -> int:
        return len(self.data)
```

### Java

```java
import java.util.*;

class TTLCache {
    private int capacity;
    private Map<Integer, int[]> data; // key -> {value, expireAt}

    public TTLCache(int capacity) {
        this.capacity = capacity;
        this.data = new HashMap<>();
    }

    public void put(int key, int value, int expireAt) {
        data.put(key, new int[]{value, expireAt});
        if (data.size() > capacity) {
            // 找 (expireAt, key) 最小者淘汰
            int victim = 0, bestExp = Integer.MAX_VALUE, bestKey = Integer.MAX_VALUE;
            boolean first = true;
            for (Map.Entry<Integer, int[]> e : data.entrySet()) {
                int k = e.getKey(), exp = e.getValue()[1];
                if (first || exp < bestExp || (exp == bestExp && k < bestKey)) {
                    first = false;
                    victim = k;
                    bestExp = exp;
                    bestKey = k;
                }
            }
            data.remove(victim);
        }
    }

    public int get(int key, int now) {
        int[] node = data.get(key);
        if (node == null) return -1;
        if (now >= node[1]) return -1; // 过期占位仍保留
        return node[0];
    }

    public int purge(int now) {
        List<Integer> dead = new ArrayList<>();
        for (Map.Entry<Integer, int[]> e : data.entrySet()) {
            if (now >= e.getValue()[1]) dead.add(e.getKey());
        }
        for (int k : dead) data.remove(k);
        return dead.size();
    }

    public int size() {
        return data.size();
    }
}
```

### C++

```cpp
#include <unordered_map>
#include <vector>
#include <climits>
using namespace std;

class TTLCache {
    int capacity;
    unordered_map<int, pair<int, int>> data; // key -> {value, expireAt}
public:
    TTLCache(int capacity) : capacity(capacity) {}

    void put(int key, int value, int expireAt) {
        data[key] = {value, expireAt};
        if ((int)data.size() > capacity) {
            // 过期最早，其次键更小
            int victim = 0, bestExp = INT_MAX, bestKey = INT_MAX;
            bool first = true;
            for (auto& [k, node] : data) {
                int exp = node.second;
                if (first || exp < bestExp || (exp == bestExp && k < bestKey)) {
                    first = false;
                    victim = k;
                    bestExp = exp;
                    bestKey = k;
                }
            }
            data.erase(victim);
        }
    }

    int get(int key, int now) {
        auto it = data.find(key);
        if (it == data.end()) return -1;
        if (now >= it->second.second) return -1; // 不删除
        return it->second.first;
    }

    int purge(int now) {
        vector<int> dead;
        for (auto& [k, node] : data)
            if (now >= node.second) dead.push_back(k);
        for (int k : dead) data.erase(k);
        return (int)dead.size();
    }

    int size() { return (int)data.size(); }
};
```

### C

```c
#include <limits.h>
#include <stdlib.h>
#include <string.h>

typedef struct TTLCache TTLCache;

typedef struct {
    int key, value, expireAt;
    char used;
} Entry;

struct TTLCache {
    int capacity;
    Entry* data;
    int size;
    int capArr;
};

TTLCache* tTLCacheCreate(int capacity) {
    TTLCache* obj = (TTLCache*)calloc(1, sizeof(TTLCache));
    obj->capacity = capacity;
    obj->capArr = capacity + 8;
    obj->data = (Entry*)calloc((size_t)obj->capArr, sizeof(Entry));
    return obj;
}

void tTLCacheFree(TTLCache* obj) {
    if (!obj) return;
    free(obj->data);
    free(obj);
}

static int findKey(TTLCache* obj, int key) {
    for (int i = 0; i < obj->capArr; i++)
        if (obj->data[i].used && obj->data[i].key == key) return i;
    return -1;
}

void tTLCachePut(TTLCache* obj, int key, int value, int expireAt) {
    int idx = findKey(obj, key);
    if (idx >= 0) {
        obj->data[idx].value = value;
        obj->data[idx].expireAt = expireAt;
    } else {
        if (obj->size == obj->capArr) {
            int nc = obj->capArr * 2;
            obj->data = (Entry*)realloc(obj->data, sizeof(Entry) * (size_t)nc);
            memset(obj->data + obj->capArr, 0, sizeof(Entry) * (size_t)(nc - obj->capArr));
            obj->capArr = nc;
        }
        for (int i = 0; i < obj->capArr; i++) {
            if (!obj->data[i].used) {
                obj->data[i].used = 1;
                obj->data[i].key = key;
                obj->data[i].value = value;
                obj->data[i].expireAt = expireAt;
                obj->size++;
                break;
            }
        }
    }
    if (obj->size > obj->capacity) {
        // 淘汰 expireAt 最小；并列则 key 更小
        int victim = -1, bestExp = INT_MAX, bestKey = INT_MAX;
        for (int i = 0; i < obj->capArr; i++) {
            if (!obj->data[i].used) continue;
            int k = obj->data[i].key, exp = obj->data[i].expireAt;
            if (victim < 0 || exp < bestExp || (exp == bestExp && k < bestKey)) {
                victim = i;
                bestExp = exp;
                bestKey = k;
            }
        }
        if (victim >= 0) {
            obj->data[victim].used = 0;
            obj->size--;
        }
    }
}

int tTLCacheGet(TTLCache* obj, int key, int now) {
    int idx = findKey(obj, key);
    if (idx < 0) return -1;
    if (now >= obj->data[idx].expireAt) return -1; // 不删除
    return obj->data[idx].value;
}

int tTLCachePurge(TTLCache* obj, int now) {
    int dead = 0;
    for (int i = 0; i < obj->capArr; i++) {
        if (obj->data[i].used && now >= obj->data[i].expireAt) {
            obj->data[i].used = 0;
            obj->size--;
            dead++;
        }
    }
    return dead;
}

int tTLCacheSize(TTLCache* obj) {
    return obj->size;
}
```
