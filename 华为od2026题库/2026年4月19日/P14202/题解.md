## 解题思路

题目给定若干个用空格分隔的分辨率字符串，每个分辨率格式为 `宽x高`。
需要先判断每个分辨率所属的清晰度等级，再按题目要求排序后输出。

清晰度等级从低到高分别为：

- `720P`：`1280 × 720`
- `1080P`：`1920 × 1080`
- `2K`：`2560 × 1440`
- `4K`：`3840 × 2160`

分类规则如下：

1. 宽和高都必须同时大于等于某个等级的标准，才算满足该等级；
2. 需要优先匹配更高等级；
3. 若宽和高都低于 `720P`，则归为 `720P`；
4. 不允许交换宽高来匹配，例如 `2500x3200` 只能按原始宽高判断。

因此可以写一个函数，输入宽和高，返回该分辨率对应的等级编号：

- `4K` 返回 `3`
- `2K` 返回 `2`
- `1080P` 返回 `1`
- `720P` 返回 `0`

完成分类后，再对所有分辨率进行排序。排序优先级如下：

1. 先按清晰度等级从大到小排序；
2. 若清晰度等级相同，则按面积从大到小排序；
3. 若面积也相同，则按宽从大到小排序；
4. 若前三项都相同，则保持原字符串顺序也可以。

排序完成后，将结果按空格拼接输出即可。

## 复杂度分析

假设输入分辨率数量为 `n`，  
- 分类计算复杂度为 `O(n)`；  
- 排序复杂度为 `O(n log n)`；  
因此整体时间复杂度为 `O(n log n)`。  
空间复杂度为 `O(n)` 用于存储分辨率的属性信息。

```python
class Solution:
    def sort(self, sortResolutions: str) -> str:
        # 去掉首尾空白字符
        text = sortResolutions.strip()

        # 若输入为空，则直接返回空串
        if not text:
            return ""

        # 按空格切分出所有分辨率字符串
        items = text.split()

        arr = []
        for s in items:
            # 按 'x' 分割出宽和高
            w_str, h_str = s.split('x')
            w = int(w_str)
            h = int(h_str)

            # 判断清晰度等级，数值越大表示等级越高
            if w >= 3840 and h >= 2160:
                level = 3
            elif w >= 2560 and h >= 1440:
                level = 2
            elif w >= 1920 and h >= 1080:
                level = 1
            else:
                level = 0

            # 保存排序所需信息：原字符串、等级、面积、宽度
            arr.append((s, level, w * h, w))

        # 按题意排序：
        # 1. 清晰度等级从大到小
        # 2. 面积从大到小
        # 3. 宽从大到小
        arr.sort(key=lambda x: (-x[1], -x[2], -x[3]))

        # 拼接排序结果并返回
        return '"' + ' '.join(item[0] for item in arr) + '"'
```

```java
import java.util.*;

public class Solution {
    static class Node {
        String text;
        int level;
        long area;
        int width;

        Node(String text, int level, long area, int width) {
            this.text = text;
            this.level = level;
            this.area = area;
            this.width = width;
        }
    }

    public String sort(String sortResolutions) {
        // 去掉首尾空白字符
        String text = sortResolutions.trim();

        // 若输入为空，则直接返回空串
        if (text.isEmpty()) {
            return "";
        }

        // 按空格切分所有分辨率字符串
        String[] parts = text.split("\\s+");
        List<Node> list = new ArrayList<>();

        for (String s : parts) {
            // 按 'x' 分割出宽和高
            String[] wh = s.split("x");
            int w = Integer.parseInt(wh[0]);
            int h = Integer.parseInt(wh[1]);

            // 判断清晰度等级
            int level;
            if (w >= 3840 && h >= 2160) {
                level = 3;
            } else if (w >= 2560 && h >= 1440) {
                level = 2;
            } else if (w >= 1920 && h >= 1080) {
                level = 1;
            } else {
                level = 0;
            }

            // 保存排序所需信息
            list.add(new Node(s, level, 1L * w * h, w));
        }

        // 按题意排序
        list.sort((a, b) -> {
            if (a.level != b.level) return b.level - a.level;
            if (a.area != b.area) return Long.compare(b.area, a.area);
            if (a.width != b.width) return b.width - a.width;
            return 0;
        });

        // 拼接结果输出
        StringBuilder sb = new StringBuilder();
        sb.append('"');
        for (int i = 0; i < list.size(); i++) {
            if (i > 0) sb.append(' ');
            sb.append(list.get(i).text);
        }
        sb.append('"');
        return sb.toString();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    string sort(string sortResolutions) {
        // 去掉首尾空白字符
        while (!sortResolutions.empty() && isspace((unsigned char)sortResolutions.back())) {
            sortResolutions.pop_back();
        }
        int start = 0;
        while (start < (int)sortResolutions.size() && isspace((unsigned char)sortResolutions[start])) {
            start++;
        }
        sortResolutions = sortResolutions.substr(start);

        // 若输入为空则直接返回空串
        if (sortResolutions.empty()) {
            return "";
        }

        // 用输入流按空格切分分辨率字符串
        stringstream ss(sortResolutions);
        vector<tuple<string, int, long long, int>> arr;
        string s;

        while (ss >> s) {
            // 按 'x' 分割宽高
            int pos = s.find('x');
            int w = stoi(s.substr(0, pos));
            int h = stoi(s.substr(pos + 1));

            // 判断清晰度等级
            int level;
            if (w >= 3840 && h >= 2160) level = 3;
            else if (w >= 2560 && h >= 1440) level = 2;
            else if (w >= 1920 && h >= 1080) level = 1;
            else level = 0;

            // 保存排序信息
            arr.push_back({s, level, 1LL * w * h, w});
        }

        // 按题意排序
        sort(arr.begin(), arr.end(), [](auto& a, auto& b) {
            if (get<1>(a) != get<1>(b)) return get<1>(a) > get<1>(b);
            if (get<2>(a) != get<2>(b)) return get<2>(a) > get<2>(b);
            if (get<3>(a) != get<3>(b)) return get<3>(a) > get<3>(b);
            return false;
        });

        // 拼接结果字符串
        string ans;
        ans.push_back('"');
        for (int i = 0; i < arr.size(); i++) {
            if (i > 0) ans += ' ';
            ans += get<0>(arr[i]);
        }
        ans.push_back('"');
        return ans;
    }
};
```

```javascript
var sort = function(sortResolutions) {
    // 计算清晰度等级函数
    const getLevel = (w, h) => {
        if (w >= 3840 && h >= 2160) return 3;
        if (w >= 2560 && h >= 1440) return 2;
        if (w >= 1920 && h >= 1080) return 1;
        return 0;
    };

    // 去除首尾空白并分割分辨率字符串
    const items = sortResolutions.trim().split(/\s+/).filter(s => s.length > 0).map(s => {
        const [wStr, hStr] = s.split('x');
        const w = parseInt(wStr), h = parseInt(hStr);
        return { name: s, level: getLevel(w, h), area: w * h, width: w };
    });

    // 按题意排序
    items.sort((a, b) => {
        if (a.level !== b.level) return b.level - a.level;
        if (a.area !== b.area) return b.area - a.area;
        return b.width - a.width;
    });

    // 返回拼接结果
    return '"' + items.map(x => x.name).join(' ') + '"';
};
```

```c
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

typedef struct {
    char name[32];
    int level;
    long long area;
    int width;
} Res;

// 计算清晰度等级函数
static int _getLevel(int w, int h) {
    if (w >= 3840 && h >= 2160) return 3;
    if (w >= 2560 && h >= 1440) return 2;
    if (w >= 1920 && h >= 1080) return 1;
    return 0;
}

// 排序比较函数
static int _cmp(const void* a, const void* b) {
    const Res *ra = (const Res*)a, *rb = (const Res*)b;
    if (ra->level != rb->level) return rb->level - ra->level;
    if (ra->area != rb->area) return (rb->area > ra->area) ? 1 : -1;
    return rb->width - ra->width;
}

char* sort(char* sortResolutions) {
    int cap = 256, n = 0;
    Res* items = (Res*)malloc(cap * sizeof(Res));
    char* p = sortResolutions;

    // 按空格读取各分辨率字符串
    while (*p) {
        while (*p == ' ') p++;
        if (!*p) break;
        char* start = p;
        while (*p && *p != ' ') p++;
        int len = p - start;
        if (n >= cap) {
            cap *= 2;
            items = (Res*)realloc(items, cap * sizeof(Res));
        }

        char buf[32];
        if (len >= 32) len = 31;
        memcpy(buf, start, len);
        buf[len] = '\0';

        int w, h;
        sscanf(buf, "%dx%d", &w, &h);

        strcpy(items[n].name, buf);
        items[n].width = w;
        items[n].area = (long long)w * h;
        items[n].level = _getLevel(w, h);
        n++;
    }

    // 排序
    qsort(items, n, sizeof(Res), _cmp);

    // 拼接输出结果字符串
    int totalLen = 0;
    for (int i = 0; i < n; i++) {
        if (i > 0) totalLen++;
        totalLen += strlen(items[i].name);
    }

    char* result = (char*)malloc(totalLen + 3);
    int pos = 0;
    result[pos++] = '"';
    for (int i = 0; i < n; i++) {
        if (i > 0) result[pos++] = ' ';
        int slen = strlen(items[i].name);
        memcpy(result + pos, items[i].name, slen);
        pos += slen;
    }
    result[pos++] = '"';
    result[pos] = '\0';

    free(items);
    return result;
}
```
