## 解题思路

本题是**字符串分词 + 统计 + 二元共现**，可按以下步骤处理：

1. **分词**：对每条日志，用空格与英文标点 `, . ! ? ; :` 切分得到单词列表；转小写后参与比较。
2. **出现次数**：对每个关键词，遍历所有日志的所有单词，完整匹配则计数累加（同一日志内多次出现分别计数）。
3. **共现判定**：对每对关键词 $(i, j)$（$i < j$），统计有多少条日志**同时**包含两词各至少一次；若 $\ge 2$ 则输出下标对 $[i, j]$。
4. **输出顺序**：先输出全部关键词计数，再按 $(i, j)$ 字典序追加关联下标。

**常见假解**：

- 用子串搜索（`find`/`contains`）→ `errors` 会误计为 `error`；
- 共现按「出现次数乘积」或「同一日志多次出现算多次」→ 关联阈值判断错误；
- 只统计每条日志是否出现一次关键词总数 → 样例 3 的 $6$ 次计数错误；
- 关联对未按 $(i, j)$ 升序输出 → 样例 2 顺序错；
- 忽略大小写或标点边界 → 边界样例失败。

## 复杂度分析

设 $L = |logs| \le 1000$，$K = |keywords| \le 100$，每条日志长度 $\le 1000$，分词后单词数 $W = O(1000)$。

- 统计：$O(L \cdot W \cdot K)$；
- 共现：$O(K^2 \cdot L \cdot W)$；
- 在约束下约为 $10^8$ 量级以内，可接受；空间 $O(W)$ 临时分词。

```python
import re
from typing import List

# 与题面一致的单词切分规则
WORD_SPLIT = re.compile(r"[\s,.!?;:]+")


class Solution:
    def analyzeLogKeywords(self, logs: List[str], keywords: List[str]) -> List[int]:
        kws = [k.lower() for k in keywords]
        tokenized = [[w.lower() for w in WORD_SPLIT.split(log) if w] for log in logs]

        counts = [0] * len(keywords)
        for words in tokenized:
            for ki, kw in enumerate(kws):
                # 完整单词匹配，累加出现次数
                counts[ki] += sum(1 for w in words if w == kw)

        # 预处理每条日志出现了哪些关键词
        present = []
        for words in tokenized:
            word_set = set(words)
            present.append([kw in word_set for kw in kws])

        pairs: List[int] = []
        m = len(keywords)
        for i in range(m):
            for j in range(i + 1, m):
                co = sum(1 for row in present if row[i] and row[j])
                if co >= 2:
                    pairs.extend([i, j])

        return counts + pairs
```

```java
import java.util.ArrayList;

public class Solution {
    private static boolean isDelim(char c) {
        return Character.isWhitespace(c) || c == ',' || c == '.' || c == '!' ||
                c == '?' || c == ';' || c == ':';
    }

    private static ArrayList<String> tokenize(String log) {
        ArrayList<String> words = new ArrayList<>();
        StringBuilder cur = new StringBuilder();
        for (int i = 0; i < log.length(); i++) {
            char c = log.charAt(i);
            if (isDelim(c)) {
                if (cur.length() > 0) {
                    words.add(cur.toString().toLowerCase());
                    cur.setLength(0);
                }
            } else {
                cur.append(c);
            }
        }
        if (cur.length() > 0) words.add(cur.toString().toLowerCase());
        return words;
    }

    public int[] analyzeLogKeywords(String[] logs, String[] keywords) {
        int m = keywords.length;
        String[] kws = new String[m];
        for (int i = 0; i < m; i++) kws[i] = keywords[i].toLowerCase();

        int[] counts = new int[m];
        for (String log : logs) {
            ArrayList<String> words = tokenize(log);
            for (int ki = 0; ki < m; ki++) {
                for (String w : words) {
                    if (w.equals(kws[ki])) counts[ki]++;
                }
            }
        }

        ArrayList<Integer> pairs = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            for (int j = i + 1; j < m; j++) {
                int co = 0;
                for (String log : logs) {
                    ArrayList<String> words = tokenize(log);
                    boolean hasI = false, hasJ = false;
                    for (String w : words) {
                        if (w.equals(kws[i])) hasI = true;
                        if (w.equals(kws[j])) hasJ = true;
                    }
                    if (hasI && hasJ) co++;
                }
                // 至少 2 条不同日志共现才记录
                if (co >= 2) {
                    pairs.add(i);
                    pairs.add(j);
                }
            }
        }

        int[] ans = new int[m + pairs.size()];
        System.arraycopy(counts, 0, ans, 0, m);
        for (int i = 0; i < pairs.size(); i++) ans[m + i] = pairs.get(i);
        return ans;
    }
}
```

```cpp
#include <cctype>
#include <string>
#include <vector>

using namespace std;

static string toLower(const string& s) {
  string t = s;
  for (char& c : t) c = static_cast<char>(tolower(static_cast<unsigned char>(c)));
  return t;
}

static bool isDelim(char c) {
  return isspace(static_cast<unsigned char>(c)) || c == ',' || c == '.' ||
         c == '!' || c == '?' || c == ';' || c == ':';
}

static vector<string> tokenize(const string& log) {
  vector<string> words;
  string cur;
  for (char c : log) {
    if (isDelim(c)) {
      if (!cur.empty()) {
        words.push_back(toLower(cur));
        cur.clear();
      }
    } else {
      cur += c;
    }
  }
  if (!cur.empty()) words.push_back(toLower(cur));
  return words;
}

class Solution {
 public:
  vector<int> analyzeLogKeywords(vector<string>& logs, vector<string>& keywords) {
    int m = static_cast<int>(keywords.size());
    vector<string> kws;
    for (auto& k : keywords) kws.push_back(toLower(k));

    vector<int> counts(m, 0);
    for (auto& log : logs) {
      auto words = tokenize(log);
      for (int ki = 0; ki < m; ki++) {
        for (auto& w : words) {
          if (w == kws[ki]) counts[ki]++;
        }
      }
    }

    vector<int> pairs;
    for (int i = 0; i < m; i++) {
      for (int j = i + 1; j < m; j++) {
        int co = 0;
        for (auto& log : logs) {
          auto words = tokenize(log);
          bool hasI = false, hasJ = false;
          for (auto& w : words) {
            if (w == kws[i]) hasI = true;
            if (w == kws[j]) hasJ = true;
          }
          if (hasI && hasJ) co++;
        }
        if (co >= 2) {
          pairs.push_back(i);
          pairs.push_back(j);
        }
      }
    }

    counts.insert(counts.end(), pairs.begin(), pairs.end());
    return counts;
  }
};
```

```javascript
var analyzeLogKeywords = function(logs, keywords) {
    // 分词：按空格及英文标点切分，转小写
    function tokenize(log) {
        return log.split(/[\s,.!?;:]+/).filter(function(w) { return w.length > 0; }).map(function(w) { return w.toLowerCase(); });
    }

    // 关键词转小写
    var kws = keywords.map(function(k) { return k.toLowerCase(); });
    var m = kws.length;

    // 统计每个关键词的总出现次数
    var counts = new Array(m).fill(0);
    for (var li = 0; li < logs.length; li++) {
        var words = tokenize(logs[li]);
        for (var ki = 0; ki < m; ki++) {
            for (var wi = 0; wi < words.length; wi++) {
                if (words[wi] === kws[ki]) counts[ki]++;
            }
        }
    }

    // 统计关键词共现对（至少 2 条日志共现才输出）
    var pairs = [];
    for (var i = 0; i < m; i++) {
        for (var j = i + 1; j < m; j++) {
            var co = 0;
            for (var li = 0; li < logs.length; li++) {
                var words = tokenize(logs[li]);
                var hasI = false, hasJ = false;
                for (var wi = 0; wi < words.length; wi++) {
                    if (words[wi] === kws[i]) hasI = true;
                    if (words[wi] === kws[j]) hasJ = true;
                }
                if (hasI && hasJ) co++;
            }
            if (co >= 2) pairs.push(i, j);
        }
    }

    return counts.concat(pairs);
};
```

```c
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

// 判断是否是分词分隔符
static int isDelim(char c) {
    return isspace((unsigned char)c) || c == ',' || c == '.' ||
           c == '!' || c == '?' || c == ';' || c == ':';
}

// 字符串转小写
static void toLowerStr(char* s) {
    for (int i = 0; s[i]; i++) s[i] = (char)tolower((unsigned char)s[i]);
}

// 将日志分词为小写单词列表
static char** tokenize(const char* log, int* wordCount) {
    int cap = 1000;
    char** words = (char**)malloc(cap * sizeof(char*));
    *wordCount = 0;
    int len = (int)strlen(log);
    int curLen = 0;
    char cur[2000];

    for (int i = 0; i <= len; i++) {
        char c = log[i];
        if (c == '\0' || isDelim(c)) {
            if (curLen > 0) {
                cur[curLen] = '\0';
                toLowerStr(cur);
                words[*wordCount] = (char*)malloc(curLen + 1);
                strcpy(words[*wordCount], cur);
                (*wordCount)++;
                curLen = 0;
            }
        } else {
            cur[curLen++] = c;
        }
    }
    return words;
}

int* analyzeLogKeywords(char** logs, int logsSize, char** keywords, int keywordsSize, int* returnSize) {
    int m = keywordsSize;

    // 关键词转小写
    char** kws = (char**)malloc(m * sizeof(char*));
    for (int i = 0; i < m; i++) {
        kws[i] = (char*)malloc(strlen(keywords[i]) + 1);
        strcpy(kws[i], keywords[i]);
        toLowerStr(kws[i]);
    }

    // 统计每个关键词的总出现次数
    int* counts = (int*)calloc(m, sizeof(int));
    for (int li = 0; li < logsSize; li++) {
        int wc;
        char** words = tokenize(logs[li], &wc);
        for (int ki = 0; ki < m; ki++) {
            for (int wi = 0; wi < wc; wi++) {
                if (strcmp(words[wi], kws[ki]) == 0) counts[ki]++;
            }
        }
        for (int wi = 0; wi < wc; wi++) free(words[wi]);
        free(words);
    }

    // 统计关键词共现对（至少 2 条日志共现才输出）
    int pairCap = m * m * 2;
    int* pairs = (int*)malloc(pairCap * sizeof(int));
    int pairCount = 0;

    for (int i = 0; i < m; i++) {
        for (int j = i + 1; j < m; j++) {
            int co = 0;
            for (int li = 0; li < logsSize; li++) {
                int wc;
                char** words = tokenize(logs[li], &wc);
                int hasI = 0, hasJ = 0;
                for (int wi = 0; wi < wc; wi++) {
                    if (strcmp(words[wi], kws[i]) == 0) hasI = 1;
                    if (strcmp(words[wi], kws[j]) == 0) hasJ = 1;
                }
                if (hasI && hasJ) co++;
                for (int wi = 0; wi < wc; wi++) free(words[wi]);
                free(words);
            }
            if (co >= 2) {
                pairs[pairCount++] = i;
                pairs[pairCount++] = j;
            }
        }
    }

    // 组装结果：计数 + 关联下标对
    *returnSize = m + pairCount;
    int* result = (int*)malloc(*returnSize * sizeof(int));
    for (int i = 0; i < m; i++) result[i] = counts[i];
    for (int i = 0; i < pairCount; i++) result[m + i] = pairs[i];

    for (int i = 0; i < m; i++) free(kws[i]);
    free(kws);
    free(counts);
    free(pairs);
    return result;
}
```
