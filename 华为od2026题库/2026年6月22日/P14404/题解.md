## 解题思路

本题按子阵基准发电量做**一次**全局比例调整，再对结果**向上取整**。

1. 读取每个子阵的 `max_capacity`、`min_capacity`、`base_generation`，记基准数组 `base`，并计算 `total = sum(base)`、`min_total = sum(min)`。
2. **缩减**：若 `total > station_capacity`，总缩减量为 `total - station_capacity`。子阵 $i$ 的可缩减空间为 `base[i] - min[i]`，按空间占比分配缩减量；若可缩减空间之和为 $0$，说明全部已到下限仍超标，返回全 $0$。
3. **放大**：若 `total < min_total`，总增加量为 `min_total - total`。子阵 $i$ 的可放大空间为 `max[i] - base[i]`，按空间占比分配增加量；若可放大空间之和为 $0$，返回全 $0$。
4. 对调整后的实数值逐元素 `ceil` 得到整数答案。

核心观察：只需判断「超标」或「低于总下限」两种情形之一，按比例一次性分配即可；注意浮点误差与向上取整。

## 复杂度分析

- 时间复杂度：$O(n)$，其中 $n$ 为子阵个数；仅需两次线性扫描与一次分配。
- 空间复杂度：$O(n)$，存放基准、上下界与中间实数结果。

import math
from typing import List


class Solution:
    def predictGeneration(
        self, sub_arrays: List[List[int]], station_capacity: int
    ) -> List[int]:
        n = len(sub_arrays)
        if n == 0:
            return []

        base = [row[2] for row in sub_arrays]
        mins = [row[1] for row in sub_arrays]
        maxs = [row[0] for row in sub_arrays]

        total = sum(base)
        min_total = sum(mins)
        values = [float(x) for x in base]

        # 总基准超过电站容量：按可缩减空间比例下调
        if total > station_capacity:
            need = total - station_capacity
            spaces = [base[i] - mins[i] for i in range(n)]
            space_sum = sum(spaces)
            if space_sum == 0:
                return [0] * n
            for i in range(n):
                values[i] = base[i] - need * spaces[i] / space_sum
        # 总基准低于各子阵下限之和：按可放大空间比例上调
        elif total < min_total:
            need = min_total - total
            spaces = [maxs[i] - base[i] for i in range(n)]
            space_sum = sum(spaces)
            if space_sum == 0:
                return [0] * n
            for i in range(n):
                values[i] = base[i] + need * spaces[i] / space_sum

        # 向上取整；减微小量避免浮点边界误判
        return [math.ceil(v - 1e-12) for v in values]

public class Solution {
    public int[] predictGeneration(int[][] sub_arrays, int station_capacity) {

        if (n == 0) return new int[0];

        int[] base = new int[n];
        int[] mins = new int[n];
        int[] maxs = new int[n];
        for (int i = 0; i < n; i++) {
            maxs[i] = sub_arrays[i][0];
            mins[i] = sub_arrays[i][1];
            base[i] = sub_arrays[i][2];
        }

        long total = 0, minTotal = 0;
        for (int i = 0; i < n; i++) {
            total += base[i];
            minTotal += mins[i];
        }

        double[] values = new double[n];
        for (int i = 0; i < n; i++) values[i] = base[i];

        if (total > station_capacity) {
            long need = total - station_capacity;
            long spaceSum = 0;
            for (int i = 0; i < n; i++) spaceSum += base[i] - mins[i];
            // 全部已到下限仍超标，无法分配
            if (spaceSum == 0) return new int[n];
            for (int i = 0; i < n; i++) {
                double space = base[i] - mins[i];
                values[i] = base[i] - need * space / (double) spaceSum;
            }
        } else if (total < minTotal) {
            long need = minTotal - total;
            long spaceSum = 0;
            for (int i = 0; i < n; i++) spaceSum += maxs[i] - base[i];
            if (spaceSum == 0) return new int[n];
            for (int i = 0; i < n; i++) {
                double space = maxs[i] - base[i];
                values[i] = base[i] + need * space / (double) spaceSum;
            }
        }

        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            ans[i] = (int) Math.ceil(values[i] - 1e-12);
        }
        return ans;
    }

        total += base[i];
        min_total += mins[i];
