## 解题思路

本题是**图上的双动点相遇问题**，可拆成两部分：B 的位置是**完全确定的周期序列**，A 在图上**自由走最短路径**。核心结论：

- 玩家 B 不依赖 A，其位置只由回合 `t` 决定。把巡逻路径 `patrolPath`（一次单程）展开成循环序列：从起点走到末端后原路折返。例如 `[0,1,2]` 展开为 `[0,1,2,1]`（周期 4），`[5,6,7,5]` 展开为 `[5,6,7,5,7,6]`（周期 6）。一般地，`cycle = patrolPath + reverse(patrolPath[1..L-2])`，周期长度 `C = 2L-2`（`L=1` 时 B 原地不动，周期 1）。
- 玩家 A 每回合可移动到相邻房间或停留。因此 A 在第 `t` 回合**能到达**任意「从 `startA` 出发最短距离 `≤ t`」的房间（先走最短路，余下的回合原地等待即可）。
- 于是 A 与 B 在第 `t` 回合相遇，当且仅当存在房间 `u` 使 `B(t)==u` 且 `dist(startA,u) ≤ t`。我们只需从小到大枚举 `t`（从 1 开始），找到第一个满足条件的 `t` 即可。

**不可达（`-1`）判定**：若 B 巡逻经过的所有房间，与 A 所在的连通分量**完全没有交集**（例如 A 被孤立、或 B 只在另一连通块巡逻），则永远无法相遇，直接返回 `-1`。

**易错点（区分度设计）**：
- 朴素做法「求 A 到 B 某节点的最短路，直接返回该距离」是**错误**的。因为 B 在移动，A 到达该节点时 B 未必在那里（见样例 6 的 hack：距离为 2 但首次相遇是第 3 回合）。必须用「距离 `≤ t` 且 B 在第 `t` 回合正好在该节点」的时序条件。
- 必须区分「B 原地不动」（`L=1`）与「B 折返」两种情况，周期公式要对 `L=1` 正确退化。

算法主流程：
1. 用邻接表建无向图。
2. 从 `startA` 做 BFS，得到到每个房间的最少步数 `dist[]`。
3. 展开 B 的循环位置序列 `cycle`。
4. 若 `cycle` 中所有节点都与 A 不连通 → 返回 `-1`。
5. 枚举 `t = 1,2,3,…`：若 `dist[cycle[t % C]] ≤ t` 返回 `t`。

## 复杂度分析

- **时间复杂度**：建图与 BFS 为 `O(n + m)`（`n` 房间数、`m` 边数）。枚举回合最多到 `n + C`（因为 A 至多 `n-1` 步可达任意节点，而 B 每隔一个周期 `C` 重复访问同一节点），`C ≤ 2L-2 ≤ 2(n-1)`。整体约 `O(n + m + n + C) = O(n + m)`，对 `n < 100` 完全足够。
- **空间复杂度**：邻接表、距离数组、`cycle` 数组均为 `O(n + m)`，故 `O(n + m)`。

```python
from collections import deque

class Solution:
    def minMeetRounds(self, n, edges, startA, patrolPath):
        # 建无向图邻接表
        g = [[] for _ in range(n)]
        for u, v in edges:
            g[u].append(v)
            g[v].append(u)

        # 从 A 出发的 BFS 最短路：dist[u] 为到达房间 u 的最少步数
        INF = 10 ** 9
        dist = [INF] * n
        dist[startA] = 0
        q = deque([startA])
        while q:
            u = q.popleft()
            for v in g[u]:
                if dist[v] == INF:
                    dist[v] = dist[u] + 1
                    q.append(v)

        # 构造 B 的循环巡逻位置序列：单程走到末端后原路折返
        L = len(patrolPath)
        cycle = list(patrolPath)
        if L >= 2:
            cycle += patrolPath[-2:0:-1]  # 追加 patrolPath[1..L-2] 的反转
        C = len(cycle)

        # B 经过的房间若全部与 A 不连通，则永远追不上
        if not any(dist[u] < INF for u in cycle):
            return -1

        # 枚举回合 t：A 在第 t 回合可到达任意距 startA 不超过 t 的房间
        for t in range(1, 100000):
            u = cycle[t % C]
            if dist[u] <= t:
                return t
        return -1
```

```java
import java.util.*;

class Solution {
    public int minMeetRounds(int n, List<List<Integer>> edges, int startA, List<Integer> patrolPath) {
        // 建无向图邻接表
        List<Integer>[] g = new List[n];
        for (int i = 0; i < n; i++) g[i] = new ArrayList<>();
        for (List<Integer> e : edges) {
            int u = e.get(0), v = e.get(1);
            g[u].add(v); g[v].add(u);
        }

        // 从 A 出发的 BFS 最短路
        int INF = 1_000_000_000;
        int[] dist = new int[n];
        Arrays.fill(dist, INF);
        dist[startA] = 0;
        Queue<Integer> q = new LinkedList<>();
        q.offer(startA);
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : g[u]) {
                if (dist[v] == INF) {
                    dist[v] = dist[u] + 1;
                    q.offer(v);
                }
            }
        }

        // 构造 B 的循环巡逻位置序列
        int L = patrolPath.size();
        List<Integer> cycle = new ArrayList<>(patrolPath);
        if (L >= 2) {
            for (int i = L - 2; i >= 1; i--) cycle.add(patrolPath.get(i));
        }
        int C = cycle.size();

        // B 经过的房间若全部与 A 不连通，则永远追不上
        boolean reachable = false;
        for (int u : cycle) if (dist[u] < INF) reachable = true;
        if (!reachable) return -1;

        // 枚举回合 t：距离不超过 t 且 B 当时恰在该房间即相遇
        for (int t = 1; t < 100000; t++) {
            int u = cycle.get(t % C);
            if (dist[u] <= t) return t;
        }
        return -1;
    }
}
```

```cpp
class Solution {

    int minMeetRounds(int n, vector<vector<int>>& edges, int startA, vector<int>& patrolPath) {

            g[e[0]].push_back(e[1]);
            g[e[1]].push_back(e[0]);
        }

        // 从 A 出发的 BFS 最短路
        const int INF = 1e9;
        vector<int> dist(n, INF);
        dist[startA] = 0;
        queue<int> q;
        q.push(startA);
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : g[u]) {
                if (dist[v] == INF) {
                    dist[v] = dist[u] + 1;
                    q.push(v);
                }
            }
        }

        // 构造 B 的循环巡逻位置序列
        int L = (int)patrolPath.size();
        vector<int> cycle = patrolPath;
        if (L >= 2) {
            for (int i = L - 2; i >= 1; --i) cycle.push_back(patrolPath[i]);

            int u = cycle[t % C];
```

```javascript
    dist[startA] = 0;

    let head = 0;
    while (head < q.length) {
        const u = q[head++];
        for (const v of g[u]) {
            if (dist[v] === INF) {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }

    const L = patrolPath.length;
    const cycle = patrolPath.slice();
    if (L >= 2) {
        for (let i = L - 2; i >= 1; i--) cycle.push(patrolPath[i]);
```

```c
#include <stdlib.h>

int minMeetRounds(int n, int** edges, int edgesSize, int* edgesColSize,
                  int startA, int* patrolPath, int patrolPathSize) {
    // 建无向图邻接表（链表法）
    int* heads = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) heads[i] = -1;  // 链表头初始化为 -1
    int* to = (int*)malloc(2 * edgesSize * sizeof(int));
    int* nxt = (int*)malloc(2 * edgesSize * sizeof(int));
    int ecount = 0;
    for (int i = 0; i < edgesSize; i++) {
        int u = edges[i][0], v = edges[i][1];
        to[ecount] = v; nxt[ecount] = heads[u]; heads[u] = ecount++;
        to[ecount] = u; nxt[ecount] = heads[v]; heads[v] = ecount++;
    }

    // 从 A 出发的 BFS 最短路
    int INF = 1000000000;
    int* dist = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) dist[i] = INF;
    dist[startA] = 0;
    int* q = (int*)malloc(n * sizeof(int));
    int head = 0, tail = 0;
    q[tail++] = startA;
    while (head < tail) {
        int u = q[head++];
        for (int e = heads[u]; e != -1; e = nxt[e]) {
            int v = to[e];
            if (dist[v] == INF) {
                dist[v] = dist[u] + 1;
                q[tail++] = v;
            }
        }
    }

    // 构造 B 的循环巡逻位置序列
    int C = patrolPathSize;
    int* cycle = (int*)malloc(patrolPathSize * 2 * sizeof(int));
    for (int i = 0; i < patrolPathSize; i++) cycle[i] = patrolPath[i];
    if (patrolPathSize >= 2) {
        for (int i = patrolPathSize - 2; i >= 1; i--) cycle[C++] = patrolPath[i];
    }

    // B 经过的房间若全部与 A 不连通，则永远追不上
    int reachable = 0;
    for (int i = 0; i < C; i++) if (dist[cycle[i]] < INF) reachable = 1;
    int ans = -1;
    if (reachable) {
        for (int t = 1; t < 100000; t++) {
            int u = cycle[t % C];
            if (dist[u] <= t) { ans = t; break; }
        }
    }

    free(heads); free(to); free(nxt); free(dist); free(q); free(cycle);
    return ans;
}
```
