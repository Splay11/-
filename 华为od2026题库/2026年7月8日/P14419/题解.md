## 解题思路

本题核心是通过**并查集（Union-Find）**将输电线路连接的多个区域合并为连通分量（碳交易区），再对每个分量聚合统计，计算并比较减排量。

1. **边界预处理**：若 $N < 2$ 或输电线路数 $M = 0$，则不存在有效碳交易区（区域不足两个或没有连接），直接返回 $-1$。
2. **建立连通分量**：使用并查集将 `edges` 中每条边的两端区域合并到同一集合。
3. **聚合统计**：遍历所有区域，按所在连通分量的根节点汇总绿色减排总量、碳排放总量和区域数。
4. **计算减排量**：对每个区域数 $\geq 2$ 的分量，按公式 $\max(0,\ \text{绿色总量} - \text{碳总量}) \times \text{区域数}$ 计算，取最大值返回。

需要注意，区域数 $\geq 2$ 的连通分量就是有效碳交易区。即使该交易区内绿色减排总量不超过碳排放总量，其减排量也应为 $0$，而不是 $-1$。只有不存在任何有效碳交易区时才返回 $-1$。

相关算法：并查集（连通性），路径压缩优化。

## 复杂度分析

* 时间复杂度：$O(N + M \cdot \alpha(N))$，其中 $\alpha(N)$ 为反阿克曼函数（近乎常数）。并查集合并 $O(M \cdot \alpha(N))$，遍历聚合 $O(N)$，$N \leq 100$ 常数极小。
* 空间复杂度：$O(N)$，用于存储并查集 `parent` 数组及各分量统计数据。

from typing import List
from collections import defaultdict


class Solution:
    def maxCarbonReduction(self, green: List[int], carbon: List[int],
                           edges: List[List[int]]) -> int:
        n = len(green)
        # 无有效碳交易区：区域数不足 2 或无输电线路
        if n < 2 or len(edges) == 0:
            return -1

        # 并查集初始化
        parent = list(range(n))

        def find(x: int) -> int:
            while parent[x] != x:
                parent[x] = parent[parent[x]]  # 路径压缩
                x = parent[x]
            return x

        def union(x: int, y: int) -> None:
            px, py = find(x), find(y)
            if px != py:
                parent[px] = py

        # 根据输电线路合并区域
        for u, v in edges:
            union(u, v)

        # 按连通分量聚合绿色减排量和碳排放量
        comp_green = defaultdict(int)
        comp_carbon = defaultdict(int)
        comp_cnt = defaultdict(int)

        for i in range(n):
            root = find(i)
            comp_green[root] += green[i]
            comp_carbon[root] += carbon[i]
            comp_cnt[root] += 1

        # 计算每个碳交易区的减排量，取最大值
        ans = -1

        for root in comp_cnt:
            if comp_cnt[root] >= 2:
                diff = comp_green[root] - comp_carbon[root]
                reduction = max(0, diff) * comp_cnt[root]
                if reduction > ans:
                    ans = reduction

        return ans

import java.util.*;

class Solution {
    public int maxCarbonReduction(List<Integer> green, List<Integer> carbon,
                                  List<List<Integer>> edges) {
        int n = green.size();

        // 无有效碳交易区
        if (n < 2 || edges.isEmpty()) return -1;

        // 并查集初始化
        int[] parent = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }

        // 路径压缩查找，合并输电线路两端
        for (List<Integer> e : edges) {
            int px = find(parent, e.get(0));
            int py = find(parent, e.get(1));

            if (px != py) {
                parent[px] = py;
            }
        }

        // 聚合各分量数据
        int[] compGreen = new int[n];
        int[] compCarbon = new int[n];
        int[] compCnt = new int[n];

        for (int i = 0; i < n; i++) {
            int root = find(parent, i);
            compGreen[root] += green.get(i);
            compCarbon[root] += carbon.get(i);
            compCnt[root]++;
        }

        // 计算最大减排量
        int ans = -1;

        for (int i = 0; i < n; i++) {
            if (compCnt[i] >= 2) {
                int diff = compGreen[i] - compCarbon[i];
                int reduction = Math.max(0, diff) * compCnt[i];

                if (reduction > ans) {
                    ans = reduction;
                }
            }
        }

        return ans;
    }

    private int find(int[] parent, int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];  // 路径压缩
            x = parent[x];
        }

        return x;
    }
}

// 并查集查找（路径压缩）
static int find(int* parent, int x) {
    while (parent[x] != x) {
        parent[x] = parent[parent[x]];
        x = parent[x];
    }

    return x;
}

int maxCarbonReduction(int* green, int greenSize,
                       int* carbon, int carbonSize,
                       int** edges, int edgesRows,
                       int* edgesCols) {
    int n = greenSize;

    // 无有效碳交易区
    if (n < 2 || edgesRows == 0) return -1;

    // 并查集初始化
    int* parent = (int*)malloc(n * sizeof(int));

    for (int i = 0; i < n; i++) {
        parent[i] = i;
    }

    // 根据输电线路合并区域
    for (int i = 0; i < edgesRows; i++) {
        int px = find(parent, edges[i][0]);
        int py = find(parent, edges[i][1]);

        if (px != py) {
            parent[px] = py;
        }
    }

    // 聚合分量数据
    int* compGreen = (int*)calloc(n, sizeof(int));
    int* compCarbon = (int*)calloc(n, sizeof(int));
    int* compCnt = (int*)calloc(n, sizeof(int));

    for (int i = 0; i < n; i++) {
        int root = find(parent, i);
        compGreen[root] += green[i];
        compCarbon[root] += carbon[i];
        compCnt[root]++;
    }

    // 计算最大减排量
    int ans = -1;

    for (int i = 0; i < n; i++) {
        if (compCnt[i] >= 2) {
            int diff = compGreen[i] - compCarbon[i];
            int reduction = (diff > 0 ? diff : 0) * compCnt[i];

            if (reduction > ans) {
                ans = reduction;
            }
        }
    }

    // 释放动态分配的内存
    free(parent);
    free(compGreen);
    free(compCarbon);
    free(compCnt);

    return ans;
}
