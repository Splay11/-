## 解题思路

本题的核心考察全排列生成和字符串查重，分为三步完成：

1. **筛选纯净碎片**：遍历所有碎片字符串，检查每个字符串是否所有字母都唯一（无重复字母）。如果 `len(set(s)) == len(s)`，则该碎片为纯净碎片。
2. **按字母序排序**：将筛选出的纯净碎片按字母序排列，以保证最终输出的全排列方案符合字典序要求。
3. **生成全排列**：对排序后的纯净碎片列表生成所有全排列，每个排列用空格连接碎片形成魔法阵方案。

相关算法：哈希表（去重判断）、回溯（生成全排列）或内置全排列函数。

## 复杂度分析

- 时间复杂度：$O(N \cdot L + P! \cdot P \cdot L)$，其中 $N$ 为碎片数量（$N \leq 6$），$L$ 为字符串最大长度（$L \leq 26$），$P$ 为纯净碎片数量。滤除操作 $O(N \cdot L)$，排列生成 $O(P! \cdot P \cdot L)$，在数据范围内非常小。
- 空间复杂度：$O(P! \cdot P \cdot L)$，用于存储所有排列结果。

```python
from typing import List
from itertools import permutations


class Solution:
    def magicFragments(self, fragments: List[str]) -> List[str]:
        # 筛选纯净碎片：字符串中所有字母不重复
        pure = [s for s in fragments if len(set(s)) == len(s)]

        # 没有纯净碎片则返回空列表
        if not pure:
            return []

        # 生成所有排列，用空格连接碎片字符串
        result = []
        for perm in permutations(pure):
            result.append(" ".join(perm))

        # 按字母序排序后返回
        result.sort()
        return result
```

```java
import java.util.*;

class Solution {
    public List<String> magicFragments(List<String> fragments) {
        // 筛选纯净碎片：字符串中所有字母不重复
        List<String> pure = new ArrayList<>();
        for (String s : fragments) {
            Set<Character> set = new HashSet<>();
            for (char c : s.toCharArray()) {
                set.add(c);
            }
            // 集合大小等于原串长度，说明无重复字母
            if (set.size() == s.length()) {
                pure.add(s);
            }
        }

        if (pure.isEmpty()) return new ArrayList<>();

        // 按字母序排序，确保排列输出字典序
        Collections.sort(pure);

        // 回溯生成所有排列，用空格连接
        List<String> result = new ArrayList<>();
        boolean[] used = new boolean[pure.size()];
        backtrack(pure, used, new ArrayList<>(), result);
        return result;
    }

    private void backtrack(List<String> pure, boolean[] used,
                          List<String> current, List<String> result) {
        // 当前排列已包含所有纯净碎片，加入结果
        if (current.size() == pure.size()) {
            result.add(String.join(" ", current));
            return;
        }
        for (int i = 0; i < pure.size(); i++) {
            if (used[i]) continue;
            used[i] = true;
            current.add(pure.get(i));
            backtrack(pure, used, current, result);
            // 回溯：撤销选择
            current.remove(current.size() - 1);
            used[i] = false;
        }
    }
}
```

```cpp
#include <vector>
#include <string>
#include <algorithm>
#include <unordered_set>
using namespace std;

class Solution {
public:
    vector<string> magicFragments(vector<string>& fragments) {
        // 筛选纯净碎片：字符串中所有字母不重复
        vector<string> pure;
        for (auto& s : fragments) {
            unordered_set<char> st(s.begin(), s.end());
            if (st.size() == s.size()) {
                pure.push_back(s);
            }
        }

        if (pure.empty()) return {};

        // 按字母序排序，next_permutation 需要初始有序
        sort(pure.begin(), pure.end());

        // 使用 next_permutation 按字典序生成所有排列
        vector<string> result;
        do {
            string cur;
            for (int i = 0; i < (int)pure.size(); i++) {
                if (i > 0) cur += " ";
                cur += pure[i];
            }
            result.push_back(cur);
        } while (next_permutation(pure.begin(), pure.end()));

        return result;
    }
};
```

```javascript
/**
 * @param {string[]} fragments
 * @return {string[]}
 */
var magicFragments = function(fragments) {
    // 筛选纯净碎片：字符串中所有字母不重复
    const pure = fragments.filter(s => {
        const set = new Set(s);
        return set.size === s.length;
    });

    if (pure.length === 0) return [];

    // 按字母序排序
    pure.sort();

    // 回溯法生成所有排列，用空格连接
    const result = [];
    const used = new Array(pure.length).fill(false);
    const current = [];

    function backtrack() {
        if (current.length === pure.length) {
            result.push(current.join(' '));
            return;
        }
        for (let i = 0; i < pure.length; i++) {
            if (used[i]) continue;
            used[i] = true;
            current.push(pure[i]);
            backtrack();
            // 回溯：撤销选择
            current.pop();
            used[i] = false;
        }
    }

    backtrack();
    return result;
};
```

```c
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>

// 检查字符串是否所有字母不重复
static bool isPure(const char* s) {
    bool seen[26] = {false};
    for (int i = 0; s[i]; i++) {
        int idx = s[i] - 'a';
        if (seen[idx]) return false;
        seen[idx] = true;
    }
    return true;
}

// 字符串比较函数（用于 qsort 按字母序排序）
static int cmpStr(const void* a, const void* b) {
    return strcmp(*(const char**)a, *(const char**)b);
}

// 回溯生成排列
static void backtrack(char** pure, int n, bool* used, char** current,
                     int depth, char*** result, int* resultSize) {
    if (depth == n) {
        // 计算空格连接后的总长度并构造字符串
        int totalLen = 0;
        for (int i = 0; i < n; i++) {
            totalLen += (int)strlen(current[i]);
        }
        totalLen += n - 1;
        char* s = (char*)malloc(totalLen + 1);
        s[0] = '\0';
        for (int i = 0; i < n; i++) {
            if (i > 0) strcat(s, " ");
            strcat(s, current[i]);
        }
        result[*resultSize] = s;
        (*resultSize)++;
        return;
    }
    for (int i = 0; i < n; i++) {
        if (used[i]) continue;
        used[i] = true;
        current[depth] = pure[i];
        backtrack(pure, n, used, current, depth + 1, result, resultSize);
        // 回溯：撤销选择
        used[i] = false;
    }
}

char** magicFragments(char** fragments, int fragmentsSize, int* returnSize) {
    // 筛选纯净碎片
    char* pure[6];
    int pureCnt = 0;
    for (int i = 0; i < fragmentsSize; i++) {
        if (isPure(fragments[i])) {
            pure[pureCnt++] = fragments[i];
        }
    }

    if (pureCnt == 0) {
        *returnSize = 0;
        return NULL;
    }

    // 按字母序排序
    qsort(pure, pureCnt, sizeof(char*), cmpStr);

    // 计算排列数 = pureCnt!
    int total = 1;
    for (int i = 2; i <= pureCnt; i++) total *= i;

    char** result = (char**)malloc(total * sizeof(char*));
    int resultSize = 0;

    bool used[6] = {false};
    char* current[6];
    backtrack(pure, pureCnt, used, current, 0, &result, &resultSize);

    *returnSize = resultSize;
    return result;
}
```
