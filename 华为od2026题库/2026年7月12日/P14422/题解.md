# 查找满足条件的数据包

## 解题思路

题目要求：对每个数据包，在其**右侧**找到**第一个**同时满足「`weight` 相等且 `priority` 严格更大」的数据包，输出它的 1-based 编号，找不到则输出 `0`。

核心观察：条件 2 要求 `weight` 相等，因此「右侧第一个更优数据包」只可能在**同权重**的数据包之间产生。我们可以按 `weight` 把数据包分组，问题随即退化为：在每一组内，对每个元素求**右侧第一个 `priority` 更大的元素**（经典的「下一个更大元素」问题）。

算法选择：**单调栈（Monotonic Stack）**。
- 在每一组内，按原始顺序从**右向左**扫描，维护一个栈，栈中存放原始下标，且对应的 `priority` 保持**严格递增**（栈底到栈顶递增）。
- 处理当前元素时，先弹出栈顶所有 `priority <= 当前 priority` 的元素（它们不可能成为更左侧元素的答案，且当前元素比它们更近、优先级更高），此时栈顶即为右侧第一个 `priority` 更大的同权重数据包。
- 将当前下标入栈，继续向左。

由于只关心「同权重」，分组后用单调栈，整体每个元素入栈出栈各一次，时间复杂度 `O(n)`。

## 复杂度分析

- **时间复杂度**：`O(n)`。按 `weight` 分组为 `O(n)`；每组内从右向左单调栈扫描，每个元素最多入栈、出栈一次，合计 `O(n)`。
- **空间复杂度**：`O(n)`。分组存储所有下标与优先级，以及单调栈，最坏情况下（全部同权重）栈与分组结构共占 `O(n)`。

出题区分度与 hack 设计：
- 反例 1（朴素双重循环）：对全同权重且递增的 `n=10^5` 数据，朴素 `O(n^2)` 会超时，必须单调栈 `O(n)`。
- 反例 2（只比 `priority` 忽略 `weight`）：如样例 2，`[5,10]` 右侧 `[7,22]` 优先级更大但权重不同，正确输出 `0`，错解会误判为 `2`。
- 反例 3（用 `<` 而非 `<=` 弹出）：当存在相等 `priority` 时，相等不算「更大」，必须严格大于，否则会把相等优先级误判为答案。
- 边界：`n=1` 必输出 `[0]`；全同权重递减输出全 `0`；编号是 1-based 而非 0-based 下标。

```python
from typing import List
from collections import defaultdict


class Solution:
    def findPacket(self, packets: List[List[int]]) -> List[int]:
        n = len(packets)
        # 按 weight 分组，记录 (原始下标, priority)，保持原序
        groups = defaultdict(list)
        for i, (p, w) in enumerate(packets):
            groups[w].append((i, p))

        ans = [0] * n
        for w, lst in groups.items():
            # 同权重组内从右向左扫描，单调栈维护「严格递增的优先级」
            stack = []  # 存原始下标
            for idx, pr in reversed(lst):
                # 弹出优先级 <= 当前优先级的，保证栈顶是右侧第一个更大的
                while stack and packets[stack[-1]][0] <= pr:
                    stack.pop()
                if stack:
                    ans[idx] = stack[-1] + 1  # 1-based 数据包 ID
                stack.append(idx)
        return ans
```

```java
import java.util.*;

class Solution {
    public int[] findPacket(int[][] packets) {
        int n = packets.length;
        // 按 weight 分组，记录 (原始下标, priority)，保持原序
        Map<Integer, List<int[]>> groups = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int w = packets[i][1];
            groups.computeIfAbsent(w, k -> new ArrayList<>()).add(new int[]{i, packets[i][0]});
        }

        int[] ans = new int[n];
        for (List<int[]> lst : groups.values()) {
            // 同权重组内从右向左扫描，单调栈维护「严格递增的优先级」
            Deque<Integer> stack = new ArrayDeque<>();  // 存原始下标
            for (int k = lst.size() - 1; k >= 0; k--) {
                int idx = lst.get(k)[0];
                int pr = lst.get(k)[1];
                // 弹出优先级 <= 当前优先级的，保证栈顶是右侧第一个更大的
                while (!stack.isEmpty() && packets[stack.peek()][0] <= pr) {
                    stack.pop();
                }
                if (!stack.isEmpty()) {
                    ans[idx] = stack.peek() + 1;  // 1-based 数据包 ID
                }
                stack.push(idx);
            }
        }
        return ans;
    }
}
```

```cpp
#include <vector>
#include <unordered_map>
using namespace std;

class Solution {
public:
    vector<int> findPacket(vector<vector<int>>& packets) {
        int n = (int)packets.size();
        // 按 weight 分组，记录 (原始下标, priority)，保持原序
        unordered_map<int, vector<pair<int, int>>> groups;
        for (int i = 0; i < n; i++) {
            groups[packets[i][1]].push_back({i, packets[i][0]});
        }

        vector<int> ans(n, 0);
        for (auto& kv : groups) {
            vector<pair<int, int>>& lst = kv.second;
            // 同权重组内从右向左扫描，单调栈维护「严格递增的优先级」
            vector<int> stack;  // 存原始下标
            for (auto it = lst.rbegin(); it != lst.rend(); ++it) {
                int idx = it->first;
                int pr = it->second;
                // 弹出优先级 <= 当前优先级的，保证栈顶是右侧第一个更大的
                while (!stack.empty() && packets[stack.back()][0] <= pr) {
                    stack.pop_back();
                }
                if (!stack.empty()) {
                    ans[idx] = stack.back() + 1;  // 1-based 数据包 ID
                }
                stack.push_back(idx);
            }
        }
        return ans;
    }
};
```

```javascript
var findPacket = function(packets) {
    const n = packets.length;
    // 按 weight 分组，记录 (原始下标, priority)，保持原序
    const groups = new Map();
    for (let i = 0; i < n; i++) {
        const w = packets[i][1];
        if (!groups.has(w)) groups.set(w, []);
        groups.get(w).push({ idx: i, pr: packets[i][0] });
    }

    const ans = new Array(n).fill(0);
    for (const lst of groups.values()) {
        // 同权重组内从右向左扫描，单调栈维护「严格递增的优先级」
        const stack = [];  // 存原始下标
        for (let k = lst.length - 1; k >= 0; k--) {
            const idx = lst[k].idx;
            const pr = lst[k].pr;
            // 弹出优先级 <= 当前优先级的，保证栈顶是右侧第一个更大的
            while (stack.length && packets[stack[stack.length - 1]][0] <= pr) {
                stack.pop();
            }
            if (stack.length) {
                ans[idx] = stack[stack.length - 1] + 1;  // 1-based 数据包 ID
            }
            stack.push(idx);
        }
    }
    return ans;
};
```

```c
#include <stdlib.h>

/* 按 weight 分组的哈希表（拉链法） */
typedef struct Node {
    int weight;
    int* idx;     /* 原始下标数组（按序） */
    int* pr;      /* 优先级数组 */
    int cnt;
    int cap;
    struct Node* next;
} Node;

#define HASH_SIZE 100003

static Node* g_table[HASH_SIZE];

static unsigned int hash(int x) {
    unsigned int v = (unsigned int)(x < 0 ? -x : x);
    return v % HASH_SIZE;
}

static Node* findGroup(int weight) {
    unsigned int h = hash(weight);
    Node* cur = g_table[h];
    while (cur) {
        if (cur->weight == weight) return cur;
        cur = cur->next;
    }
    Node* node = (Node*)malloc(sizeof(Node));
    node->weight = weight;
    node->cap = 16;
    node->cnt = 0;
    node->idx = (int*)malloc(node->cap * sizeof(int));
    node->pr = (int*)malloc(node->cap * sizeof(int));
    node->next = g_table[h];
    g_table[h] = node;
    return node;
}

static void addToGroup(int weight, int idx, int pr) {
    Node* g = findGroup(weight);
    if (g->cnt >= g->cap) {
        g->cap *= 2;
        g->idx = (int*)realloc(g->idx, g->cap * sizeof(int));
        g->pr = (int*)realloc(g->pr, g->cap * sizeof(int));
    }
    g->idx[g->cnt] = idx;
    g->pr[g->cnt] = pr;
    g->cnt++;
}

int* findPacket(int** packets, int packetsSize, int* packetsColSize, int* returnSize) {
    /* 清空哈希表 */
    for (int i = 0; i < HASH_SIZE; i++) g_table[i] = NULL;

    /* 按 weight 分组 */
    for (int i = 0; i < packetsSize; i++) {
        addToGroup(packets[i][1], i, packets[i][0]);
    }

    int* ans = (int*)malloc((packetsSize > 0 ? packetsSize : 1) * sizeof(int));
    for (int i = 0; i < packetsSize; i++) ans[i] = 0;

    /* 遍历每个分组，同权重组内从右向左单调栈求「右侧第一个更大优先级」 */
    for (int h = 0; h < HASH_SIZE; h++) {
        Node* g = g_table[h];
        while (g) {
            int* stack = (int*)malloc((g->cnt > 0 ? g->cnt : 1) * sizeof(int));
            int sp = 0;
            for (int k = g->cnt - 1; k >= 0; k--) {
                int idx = g->idx[k];
                int pr = g->pr[k];
                while (sp > 0 && packets[stack[sp - 1]][0] <= pr) sp--;
                if (sp > 0) {
                    ans[idx] = stack[sp - 1] + 1;  /* 1-based 数据包 ID */
                }
                stack[sp++] = idx;
            }
            free(stack);
            g = g->next;
        }
    }

    /* 释放哈希表 */
    for (int h = 0; h < HASH_SIZE; h++) {
        Node* g = g_table[h];
        while (g) {
            Node* nx = g->next;
            free(g->idx);
            free(g->pr);
            free(g);
            g = nx;
        }
    }

    *returnSize = packetsSize;
    return ans;
}
```
