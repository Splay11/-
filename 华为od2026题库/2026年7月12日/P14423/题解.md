# 软件依赖树

## 解题思路

本题要求按**依赖顺序**输出项目最终生效的组件版本列表。核心是把「组件依赖关系」建模成一棵（可能含共享节点与环的）依赖树，再按题面规定的顺序遍历。

**关键规则拆解：**

1. **顺序规则**：从项目直属依赖出发，上层先输出；同层依赖按父组件出现顺序、同父内按依赖声明顺序。这恰好是**广度优先搜索（BFS，FIFO 队列）** 的展开顺序——先入队的先处理，子节点按声明顺序追加到队尾。
2. **去重规则**：某组件第一次被访问时记录并输出，之后再次出现直接跳过（既保证顺序唯一，也顺带处理了循环依赖）。
3. **版本规则**：以首次访问到的版本为准（去重天然保证）。
4. **排除规则**：依赖声明中的排除列表（如 `a:b:v1:c` 中的 `c`）表示「排除该依赖子树中的 `c`」。排除集合要**随依赖路径向下传递并叠加**：展开 `X` 时携带的排除集 = `X` 自身排除集 ∪ 各子规则携带的排除集；遇到被排除的组件则整棵子树跳过。

**算法主流程（BFS）：**

- 用哈希表 `rules` 记录每个组件的依赖规则列表（保持声明顺序），每条规则含 `(依赖名, 依赖版本, 子排除集)`。
- 队列元素为三元组 `(组件名, 版本, 当前排除集)`，初始放入所有直属依赖。
- 循环出队：若已访问则跳过；否则标记访问、输出 `name:version`，并把其所有未排除的子依赖（携带叠加后的排除集）入队。

## 复杂度分析

- **时间复杂度**：$O(N + M)$，其中 $N$ 为组件总数、$M$ 为依赖规则数。每个组件与每条规则均只被处理常数次；排除集合用哈希集合维护，成员判断与合并均摊 $O(1)$。
- **空间复杂度**：$O(N + M)$，用于存储规则表、访问集合、队列以及排除集合。

## 代码实现

### Python

```python
from collections import defaultdict, deque

class Solution:
    def getDependencyOrder(self, directDeps, depRules):

        # 规则表：组件名 -> 该组件的依赖列表（保持声明顺序）
        # 每条规则为 (依赖名, 依赖版本, 子排除集)
        rules = defaultdict(list)
        for rule in depRules:
            p = rule.split(':')                 # name, depname, depversion, exclusions?
            name, depname, depversion = p[0], p[1], p[2]
            excl_str = p[3] if len(p) > 3 else ""
            # 排除列表以逗号分隔，空串表示无排除
            child_excl = set(excl_str.split(',')) if excl_str else set()
            rules[name].append((depname, depversion, child_excl))

        seen = set()        # 已输出组件，用于去重
        result = []
        q = deque()         # FIFO 队列，保证上层先出、同层父优先

        # 初始化：放入所有项目直属依赖，并带上各自排除集
        for dep in directDeps:
            p = dep.split(':')                  # name, version, exclusions?
            name, version = p[0], p[1]
            excl_str = p[2] if len(p) > 2 else ""
            excl = set(excl_str.split(',')) if excl_str else set()
            q.append((name, version, excl))

        while q:
            name, version, excl = q.popleft()
            if name in seen:
                continue                        # 去重：重复组件只保留首次位置
            seen.add(name)
            result.append(name + ":" + version)

            # 展开当前组件的子依赖
            for depname, depversion, child_excl in rules.get(name, []):
                if depname in excl:
                    continue                    # 该依赖被排除（含其子树）
                # 排除集合随路径向下传递并叠加子规则排除项
                q.append((depname, depversion, excl | child_excl))
        return result
```

### Java

```java
import java.util.*;

class Solution {
    public List<String> getDependencyOrder(List<String> directDeps, List<String> depRules) {
        // 规则表：组件名 -> 依赖列表（依赖名, 依赖版本, 排除集合）
        Map<String, List<Object[]>> rules = new HashMap<>();
        for (String rule : depRules) {
            // 用 -1 保留末尾空字段：name, depname, depversion, exclusions?
            String[] p = rule.split(":", -1);
            String name = p[0], depname = p[1], depversion = p[2];
            Set<String> childExcl = new HashSet<>();
            if (p.length > 3 && !p[3].isEmpty())
                for (String e : p[3].split(",")) childExcl.add(e);
            rules.computeIfAbsent(name, k -> new ArrayList<>())
                 .add(new Object[]{depname, depversion, childExcl});
        }

        Set<String> seen = new HashSet<>();
        List<String> result = new ArrayList<>();
        // 队列元素：Object[]{组件名, 版本, 排除集合}
        Queue<Object[]> q = new LinkedList<>();
        for (String dep : directDeps) {
            String[] p = dep.split(":", -1);     // name, version, exclusions?
            String name = p[0], version = p[1];
            Set<String> excl = new HashSet<>();
            if (p.length > 2 && !p[2].isEmpty())
                for (String e : p[2].split(",")) excl.add(e);
            q.add(new Object[]{name, version, excl});
        }

        while (!q.isEmpty()) {
            Object[] cur = q.poll();
            String name = (String) cur[0];
            String version = (String) cur[1];
            @SuppressWarnings("unchecked")
            Set<String> excl = (Set<String>) cur[2];
            if (seen.contains(name)) continue;           // 去重
            seen.add(name);
            result.add(name + ":" + version);
            if (rules.containsKey(name)) {
                for (Object[] child : rules.get(name)) {
                    String depname = (String) child[0];
                    String depversion = (String) child[1];
                    @SuppressWarnings("unchecked")
                    Set<String> childExcl = (Set<String>) child[2];
                    if (excl.contains(depname)) continue;     // 被排除
                    Set<String> ne = new HashSet<>(excl);     // 排除集合向下传递并叠加
                    ne.addAll(childExcl);
                    q.add(new Object[]{depname, depversion, ne});
                }
            }
        }
        return result;
    }
}
```

### C++

```cpp
#include <vector>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <tuple>
using namespace std;

// 按 ':' 切分字符串（组件名/版本号不含 ':'，故切分无歧义）
static vector<string> splitColon(const string& s) {
    vector<string> res;
    string cur;
    for (char c : s) {
        if (c == ':') { res.push_back(cur); cur.clear(); }
        else cur += c;
    }
    res.push_back(cur);
    return res;
}

// 把 "a,b" 形式的排除串解析为集合
static unordered_set<string> parseExcl(const string& ex) {
    unordered_set<string> s;
    if (ex.empty()) return s;
    size_t pos = 0;
    while (pos < ex.size()) {
        size_t comma = ex.find(',', pos);
        if (comma == string::npos) { s.insert(ex.substr(pos)); break; }
        s.insert(ex.substr(pos, comma - pos));
        pos = comma + 1;
    }
    return s;
}

class Solution {
public:
    vector<string> getDependencyOrder(vector<string>& directDeps, vector<string>& depRules) {
        // 规则表：组件名 -> 依赖列表 (依赖名, 依赖版本, 排除集合)
        unordered_map<string, vector<pair<string, pair<string, unordered_set<string>>>>> rules;
        for (auto& rule : depRules) {
            vector<string> p = splitColon(rule);     // name, depname, depversion, exclusions?
            string name = p[0], depname = p[1], depversion = p[2];
            unordered_set<string> childExcl = (p.size() > 3) ? parseExcl(p[3]) : unordered_set<string>();
            rules[name].push_back({depname, {depversion, childExcl}});
        }

        unordered_set<string> seen;
        vector<string> result;
        using Elem = tuple<string, string, unordered_set<string>>;  // (组件名, 版本, 排除集)
        queue<Elem> q;
        for (auto& dep : directDeps) {
            vector<string> p = splitColon(dep);      // name, version, exclusions?
            string name = p[0], version = p[1];
            unordered_set<string> excl = (p.size() > 2) ? parseExcl(p[2]) : unordered_set<string>();
            q.push({name, version, excl});
        }

        while (!q.empty()) {
            Elem front = q.front(); q.pop();
            string name = get<0>(front);
            string version = get<1>(front);
            unordered_set<string> excl = get<2>(front);
            if (seen.count(name)) continue;          // 去重
            seen.insert(name);
            result.push_back(name + ":" + version);
            if (rules.count(name)) {
                for (auto& child : rules[name]) {
                    string depname = child.first;
                    string depversion = child.second.first;
                    if (excl.count(depname)) continue;   // 被排除
                    unordered_set<string> ne = excl;     // 排除集合向下传递并叠加
                    for (auto& e : child.second.second) ne.insert(e);
                    q.push({depname, depversion, ne});
                }
            }
        }
        return result;
    }
};
```

### JavaScript

```javascript
var getDependencyOrder = function(directDeps, depRules) {
    // 规则表：组件名 -> 依赖列表 [依赖名, 依赖版本, 排除集合]
    const rules = {};
    for (const rule of depRules) {
        const p = rule.split(':');               // name, depname, depversion, exclusions?
        const name = p[0], depname = p[1], depversion = p[2];
        const childExcl = new Set();
        if (p.length > 3 && p[3]) p[3].split(',').forEach(e => childExcl.add(e));
        if (!rules[name]) rules[name] = [];
        rules[name].push([depname, depversion, childExcl]);
    }

    const seen = new Set();
    const result = [];
    const q = [];                                 // 队列元素：[组件名, 版本, 排除集合]
    for (const dep of directDeps) {
        const p = dep.split(':');                 // name, version, exclusions?
        const name = p[0], version = p[1];
        const excl = new Set();
        if (p.length > 2 && p[2]) p[2].split(',').forEach(e => excl.add(e));
        q.push([name, version, excl]);
    }

    while (q.length) {
        const cur = q.shift();                   // FIFO：上层先出、同层父优先、同父按声明序
        const name = cur[0], version = cur[1], excl = cur[2];
        if (seen.has(name)) continue;            // 去重
        seen.add(name);
        result.push(name + ':' + version);
        if (rules[name]) {
            for (const child of rules[name]) {
                const depname = child[0], depversion = child[1], childExcl = child[2];
                if (excl.has(depname)) continue;     // 被排除
                const ne = new Set(excl);            // 排除集合向下传递并叠加
                childExcl.forEach(e => ne.add(e));
                q.push([depname, depversion, ne]);
            }
        }
    }
    return result;
};
```

### C

```c
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

#define MAX_RULES 8000
#define MAX_Q 8000
#define MAX_SEEN 8000

// 跨平台 strdup（避免 _POSIX_C_SOURCE 依赖）
static char* my_strdup(const char* s) {
    char* p = (char*)malloc(strlen(s) + 1);
    strcpy(p, s);
    return p;
}

// 判断 name 是否出现在逗号分隔的排除串 excl 中
static int in_excl(const char* excl, const char* name) {
    if (!excl || !excl[0]) return 0;
    char* buf = my_strdup(excl);
    char* tok = strtok(buf, ",");
    int found = 0;
    while (tok) {
        if (strcmp(tok, name) == 0) { found = 1; break; }
        tok = strtok(NULL, ",");
    }
    free(buf);
    return found;
}

char** getDependencyOrder(char** directDeps, int directDepsSize,
                          char** depRules, int depRulesSize, int* returnSize) {
    // 解析依赖规则：name -> (depname, depversion, child_excl)
    char* r_name[MAX_RULES];
    char* r_dep[MAX_RULES];
    char* r_ver[MAX_RULES];
    char* r_excl[MAX_RULES];
    int ruleCount = 0;

    for (int i = 0; i < depRulesSize && ruleCount < MAX_RULES; i++) {
        char* s = my_strdup(depRules[i]);
        char* parts[4];
        int cnt = 0;
        char* tok = strtok(s, ":");
        while (tok && cnt < 4) { parts[cnt++] = tok; tok = strtok(NULL, ":"); }
        if (cnt < 3) { free(s); continue; }
        r_name[ruleCount] = my_strdup(parts[0]);
        r_dep[ruleCount] = my_strdup(parts[1]);
        r_ver[ruleCount] = my_strdup(parts[2]);
        r_excl[ruleCount] = (cnt > 3 && parts[3][0]) ? my_strdup(parts[3]) : my_strdup("");
        ruleCount++;
        free(s);
    }

    char* seen[MAX_SEEN];
    int seenCount = 0;

    char* q_name[MAX_Q];
    char* q_ver[MAX_Q];
    char* q_excl[MAX_Q];
    int qh = 0, qt = 0;

    // 初始化：放入所有项目直属依赖，并带上各自排除集
    for (int i = 0; i < directDepsSize; i++) {
        char* s = my_strdup(directDeps[i]);
        char* parts[3];
        int cnt = 0;
        char* tok = strtok(s, ":");
        while (tok && cnt < 3) { parts[cnt++] = tok; tok = strtok(NULL, ":"); }
        if (cnt < 2) { free(s); continue; }
        char* name = my_strdup(parts[0]);
        char* ver = my_strdup(parts[1]);
        char* excl = (cnt > 2 && parts[2][0]) ? my_strdup(parts[2]) : my_strdup("");
        free(s);
        q_name[qt] = name; q_ver[qt] = ver; q_excl[qt] = excl; qt++;
    }

    char** result = (char**)malloc(MAX_SEEN * sizeof(char*));
    int resCount = 0;

    while (qh < qt) {
        char* name = q_name[qh];
        char* ver = q_ver[qh];
        char* excl = q_excl[qh];
        qh++;

        // 去重：重复组件只保留首次出现
        int dup = 0;
        for (int i = 0; i < seenCount; i++)
            if (strcmp(seen[i], name) == 0) { dup = 1; break; }
        if (dup) { free(name); free(ver); free(excl); continue; }
        seen[seenCount++] = my_strdup(name);

        // 记录结果 "name:version"
        char* item = (char*)malloc(strlen(name) + strlen(ver) + 2);
        sprintf(item, "%s:%s", name, ver);
        result[resCount++] = item;

        // 展开子依赖
        for (int r = 0; r < ruleCount; r++) {
            if (strcmp(r_name[r], name) != 0) continue;
            if (in_excl(excl, r_dep[r])) continue;       // 该依赖被排除（含其子树）
            // 新排除集合 = excl ∪ r_excl[r]
            char* ne;
            if (!excl[0]) ne = my_strdup(r_excl[r]);
            else if (!r_excl[r][0]) ne = my_strdup(excl);
            else {
                ne = (char*)malloc(strlen(excl) + strlen(r_excl[r]) + 2);
                sprintf(ne, "%s,%s", excl, r_excl[r]);
            }
            q_name[qt] = my_strdup(r_dep[r]);
            q_ver[qt] = my_strdup(r_ver[r]);
            q_excl[qt] = ne;
            qt++;
        }
        free(name); free(ver); free(excl);
    }

    *returnSize = resCount;
    return result;
}
```
