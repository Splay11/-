## 解题思路

本题是一个带衰减约束的调度优化问题。有 $n$ 个菌子（$n \le 15$），每个菌子加工需 $5$ 小时，总时间不超过 $total$ 小时。每个菌子 $i$ 有初始价值 $v_i$ 和衰减速度 $d_i$，若在第 $j$ 个被加工（$j$ 从 $0$ 开始），实际价值为 $v_i - d_i \times j \times 5$。若实际价值 $\le 0$，则跳过不加工。求最大总价值。

**核心观察：最优加工顺序是确定的**——在所选菌子集合固定的情况下，应按照衰减速度 $d_i$ **降序**加工（衰减快的先加工，减少等待带来的损失）。这可以通过交换相邻菌子的贪心论证：若 $d_a > d_b$，则先加工 $a$ 后加工 $b$ 总是优于反过来。

基于此，算法为：
1. 将所有菌子按 $d_i$ 降序排序；
2. 进行 $0/1$ 背包式 DP：$dp[j]$ 表示已加工 $j$ 个菌子时的最大总价值；
3. 遍历每个菌子，从后向前更新：若加工该菌子作为第 $j$ 个（等待 $j \times 5$ 小时），实际价值 $> 0$ 则转移 $dp[j+1] = \max(dp[j+1], dp[j] + v_i - d_i \times j \times 5)$；
4. 答案为 $\max_{0 \le j \le \min(n, total/5)} dp[j]$。

时间复杂度 $O(n \cdot \max\_cnt)$，其中 $\max\_cnt = \min(n, total/5) \le 15$。

## 复杂度分析

- **时间复杂度**：$O(n \log n + n \cdot \max\_cnt)$。排序 $O(n \log n)$，DP $O(n \cdot \min(n, total/5))$。由于 $n \le 15$，实际运行极快。
- **空间复杂度**：$O(n + \max\_cnt)$。需要存储菌子信息数组和 DP 数组。

```python
from typing import List


class Solution:
    def maxMushroomValue(self, count: int, total: int,
                         values: List[int], decays: List[int]) -> int:
        max_cnt = min(count, total // 5)

        # 按衰减速度降序排列（贪心：衰减快的先加工）
        items = list(zip(decays, values))
        items.sort(key=lambda x: -x[0])

        # dp[j] = 加工 j 个菌子的最大总价值
        dp = [0] * (max_cnt + 1)

        for d, v in items:
            for j in range(max_cnt - 1, -1, -1):
                wait = j * 5
                cur = v - d * wait
                if cur > 0 and dp[j] + cur > dp[j + 1]:
                    dp[j + 1] = dp[j] + cur

        return max(dp)
```

```java
import java.util.*;

class Solution {
    public int maxMushroomValue(int count, int total,
                                List<Integer> values, List<Integer> decays) {
        int maxCnt = Math.min(count, total / 5);

        // 按衰减速度降序排列
        int[][] items = new int[count][2];
        for (int i = 0; i < count; i++) {
            items[i][0] = decays.get(i);
            items[i][1] = values.get(i);
        }
        Arrays.sort(items, (a, b) -> b[0] - a[0]);

        int[] dp = new int[maxCnt + 1];

        for (int[] item : items) {
            int d = item[0];
            int v = item[1];
            for (int j = maxCnt - 1; j >= 0; j--) {
                int wait = j * 5;
                int cur = v - d * wait;
                if (cur > 0) {
                    dp[j + 1] = Math.max(dp[j + 1], dp[j] + cur);
                }
            }
        }

        int ans = 0;
        for (int val : dp) ans = Math.max(ans, val);
        return ans;
    }
}
```

```cpp
class Solution {
public:
    int maxMushroomValue(int count, int total,
                         vector<int>& values, vector<int>& decays) {
        int maxCnt = min(count, total / 5);

        // 将菌子按衰减速度降序排列
        vector<pair<int, int>> items;
        for (int i = 0; i < count; i++) {
            items.push_back({decays[i], values[i]});
        }
        sort(items.begin(), items.end(),
             [](const pair<int, int>& a, const pair<int, int>& b) {
                 return a.first > b.first;
             });

        vector<int> dp(maxCnt + 1, 0);

        for (auto& item : items) {
            int d = item.first, v = item.second;
            for (int j = maxCnt - 1; j >= 0; j--) {
                int wait = j * 5;
                int cur = v - d * wait;
                if (cur > 0) {
                    dp[j + 1] = max(dp[j + 1], dp[j] + cur);
                }
            }
        }

        return *max_element(dp.begin(), dp.end());
    }
};
```

```c
#include <stdlib.h>
#include <string.h>

int cmp_desc(const void* a, const void* b) {
    const int* ia = *(const int* const*)a;
    const int* ib = *(const int* const*)b;
    if (ia[0] > ib[0]) return -1;
    if (ia[0] < ib[0]) return 1;
    return 0;
}

int maxMushroomValue(int count, int total,
                     int* values, int valuesSize,
                     int* decays, int decaysSize) {
    int maxCnt = count < total / 5 ? count : total / 5;

    int** items = (int**)malloc(count * sizeof(int*));
    for (int i = 0; i < count; i++) {
        items[i] = (int*)malloc(2 * sizeof(int));
        items[i][0] = decays[i];
        items[i][1] = values[i];
    }
    qsort(items, count, sizeof(int*), cmp_desc);

    int* dp = (int*)calloc(maxCnt + 1, sizeof(int));

    for (int i = 0; i < count; i++) {
        int d = items[i][0], v = items[i][1];
        for (int j = maxCnt - 1; j >= 0; j--) {
            int wait = j * 5;
            int cur = v - d * wait;
            if (cur > 0) {
                int newVal = dp[j] + cur;
                if (newVal > dp[j + 1]) dp[j + 1] = newVal;
            }
        }
    }

    int ans = 0;
    for (int j = 0; j <= maxCnt; j++)
        if (dp[j] > ans) ans = dp[j];

    for (int i = 0; i < count; i++) free(items[i]);
    free(items);
    free(dp);

    return ans;
}
```
