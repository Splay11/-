## 解题思路

本题的核心是将灯带颜色变换建模为**确定性有限状态自动机**。

**关键观察：**
- 灯带固定 16 盏灯，每盏灯只有红（R）或绿（G）两种颜色，总共 `2^16 = 65536` 种可能状态。
- 变换规则是确定的：下一状态完全由当前状态决定。
- 由于状态空间有限（65536），从任意初始状态出发，最多经过 65536 步必然进入循环。

**算法设计（状态压缩 + 循环节检测）：**

1. **状态压缩**：用整数 `state` 的二进制位表示灯带状态，第 i 位为 0 表示红色（R），为 1 表示绿色（G）。这样 16 盏灯只需要一个 int。

2. **单步变换**：对于中间位置 i（1 ≤ i ≤ 14），检查左右邻居 `state[i-1]` 和 `state[i+1]` 是否一致。一致则设为绿色（置 1），不一致则设为红色（置 0）。边界位置 0 和 15 只有单一邻居，始终变为红色。

3. **循环节检测**：由于 `t` 最大可达 `10^7`，远超状态数，逐秒模拟会超时。使用哈希表记录每个状态首次出现的时刻：
   - 若当前状态在哈希表中出现过（设首次出现时刻为 `prev`），说明进入循环，循环长度为 `step - prev`。
   - 剩余步数 `remain = (t - step) % cycleLen`，直接从历史记录中取出目标状态，结束模拟。

**时间复杂度：** O(S)，其中 S 为状态空间大小（≤ 65536）。空间复杂度：O(S)。

## 复杂度分析

- **时间复杂度**：O(2^16) = O(65536)。虽然 t 可达 10^7，但通过循环节检测，实际模拟步数不会超过状态总数 65536。每步需要 O(16) 检查 14 个中间位置，总体在可接受范围内。

- **空间复杂度**：O(2^16) = O(65536)。需要哈希表记录每个状态首次出现的时间，以及存储状态历史序列。

class Solution:
    def lightStripTransform(self, lights: str, t: int) -> str:
        # 将字符串转为整数位表示，R=0, G=1
        state = 0
        for i, ch in enumerate(lights):
            if ch == 'G':
                state |= (1 << i)

        # 记录每个状态首次出现的时间
        seen = {state: 0}
        history = [state]

        for step in range(1, t + 1):
            # 计算下一状态
            # 边界位置（0和15）只有单一邻居，走规则2→红色(R=0)
            # 中间位置：左右邻居一致→绿色(G=1)
            nxt = 0
            for i in range(1, 15):
                left = (state >> (i - 1)) & 1
                right = (state >> (i + 1)) & 1
                if left == right:
                    nxt |= (1 << i)
            state = nxt

            # 检测循环：当前状态之前出现过，说明进入循环
            if state in seen:
                prev = seen[state]
                cycle_len = step - prev       # 循环节长度
                remain = (t - step) % cycle_len  # 剩余步数取模
                state = history[prev + remain]   # 直接定位最终状态
                break

            seen[state] = step
            history.append(state)

        # 将最终状态转为字符串
        result = []
        for i in range(16):
            result.append('G' if (state >> i) & 1 else 'R')
        return ''.join(result)

            // 边界位置（0和15）只有单一邻居，走规则2→红色(R=0)
            // 中间位置：左右邻居一致→绿色(G=1)

                    nxt |= (1 << i);
                }

            // 检测循环：当前状态之前出现过，说明进入循环
            if (seen.containsKey(state)) {
                int prev = seen.get(state);

                state = history.get(prev + remain);  // 直接定位最终状态
                break;
            }

            seen.put(state, step);
            history.add(state);
        }

        // 将最终状态转为字符串
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 16; i++) {
            sb.append(((state >> i) & 1) == 1 ? 'G' : 'R');
        }
        return sb.toString();
    }

#include <string.h>
#include <stdlib.h>

/**
 * @param lights 初始灯带状态字符串（不含引号）
 * @param t 秒数
 * @return t 秒后的灯带状态字符串（不含引号，需调用方 free）
 */
char* lightStripTransform(char* lights, int t) {
    // 将字符串转为整数位表示，R=0, G=1
    int state = 0;
    for (int i = 0; i < 16; i++) {
        if (lights[i] == 'G') {
            state |= (1 << i);
        }
    }

    // 记录每个状态首次出现的时间，状态数最多 65536
    int seen[65536];
    for (int i = 0; i < 65536; i++) seen[i] = -1;
    seen[state] = 0;
    int history[65536];
    history[0] = state;
    int historySize = 1;

    for (int step = 1; step <= t; step++) {
        // 计算下一状态
        // 边界位置（0和15）只有单一邻居，走规则2→红色(R=0)
        // 中间位置：左右邻居一致→绿色(G=1)
        int nxt = 0;
        for (int i = 1; i < 15; i++) {
            int left = (state >> (i - 1)) & 1;
            int right = (state >> (i + 1)) & 1;
            if (left == right) {
                nxt |= (1 << i);
            }
        }
        state = nxt;

        // 检测循环：当前状态之前出现过，说明进入循环
        if (seen[state] != -1) {
            int prev = seen[state];
            int cycleLen = step - prev;          // 循环节长度
            int remain = (t - step) % cycleLen;  // 剩余步数取模
            state = history[prev + remain];       // 直接定位最终状态
            break;
        }

        seen[state] = step;
        history[historySize++] = state;
    }

    // 将最终状态转为字符串（不含引号）
    char* result = (char*)malloc(17);
    for (int i = 0; i < 16; i++) {
        result[i] = ((state >> i) & 1) ? 'G' : 'R';
    }
    result[16] = '\0';
    return result;
}
