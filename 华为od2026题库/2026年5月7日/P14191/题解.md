## 解题思路

本题是**字符串模拟**结合**哈希集合**与**排序**。

1. **规范化**：题意规定命令行首尾无意义的空格需忽略，因此对每条命令先做 `strip()`（C++/Java 为跳过首尾空格子串）。
2. **前缀匹配**：若规范化后的命令**以给定前缀开头**（`startswith` / `startsWith` / `compare(0, len, prefix)==0`），则该命令参与统计。
3. **取“前缀后的第一个关键字”**：从命令中去掉已匹配的前缀，得到后缀；对后缀再次去掉首尾空格；从第一个非空格字符起，连续读到下一个空格（或行尾），即为一个关键字。若后缀为空（例如整行恰好等于前缀），则该命令不产生关键字。
4. **去重与输出顺序**：题意要求相同关键字只保留一个，且最终按**字典序**输出，故用 `set` / `TreeSet` 收集，再转为有序列表（C++ 中 `set` 已有序）。

核心代码接口为 LeetCode 核心模式：`Solution.FindNextKeywords(commands, prefix)`，平台侧只需解析输入、调用该函数并格式化输出即可。

## 复杂度分析

- **时间复杂度**：$O(N \cdot L)$，其中 \(N\) 为命令条数，\(L\) 为单条命令最大长度（题面 \(L \le 200\)）。每条命令仅做常数次扫描与哈希插入；排序的对象是去重后的关键字个数，不超过 \(N\)，故排序部分为 \(O(M \log M)\)，\(M \le N\)，整体由线性扫描主导。
- **空间复杂度**：\(O(M)\)，存储去重后的关键字集合；与命令总长度相比仍为题目范围内的常数级额外空间。

```python
from typing import List

class Solution:
    def FindNextKeywords(self, commands: List[str], prefix: str) -> List[str]:
        # 用集合去重，最后排序输出
        seen = set()
        for raw in commands:
            # 忽略命令首尾无意义空格
            s = raw.strip()
            # 以前缀开头才处理
            if not s.startswith(prefix):
                continue
            # 前缀之后的部分，再去掉首尾空格
            rest = s[len(prefix):].strip()
            if not rest:
                continue
            # 第一个关键字：到下一个空格为止
            sp = rest.find(" ")
            word = rest if sp == -1 else rest[:sp]
            if word:
                seen.add(word)
        # 按字典序排序输出
        return sorted(seen)
```

```java
import java.util.*;

public class Solution {
    public String[] FindNextKeywords(String[] commands, String prefix) {
        // TreeSet 同时完成去重与字典序排序
        TreeSet<String> seen = new TreeSet<>();
        for (String raw : commands) {
            String s = raw.trim(); // 去除首尾空格
            if (!s.startsWith(prefix)) {
                continue; // 不符合前缀跳过
            }
            String rest = s.substring(prefix.length()).trim(); // 去除前缀和首尾空格
            if (rest.isEmpty()) {
                continue;
            }
            int sp = rest.indexOf(' '); // 查找第一个空格
            String word = sp < 0 ? rest : rest.substring(0, sp); // 提取关键字
            if (!word.isEmpty()) {
                seen.add(word);
            }
        }
        // 返回有序结果
        return seen.toArray(new String[0]);
    }
}
```

```cpp
#include <algorithm>
#include <set>
#include <string>
#include <vector>
using namespace std;

class Solution {
public:
    vector<string> FindNextKeywords(vector<string> &commands, string prefix) {
        set<string> seen;  // 有序集合，天然字典序
        for (const string &raw : commands) {
            // 去掉首尾空格
            size_t b = raw.find_first_not_of(' ');
            if (b == string::npos) continue;
            size_t e = raw.find_last_not_of(' ');
            string s = raw.substr(b, e - b + 1);
            // 检查前缀
            if (s.size() < prefix.size() || s.compare(0, prefix.size(), prefix) != 0) continue;
            // 去除前缀并再次去掉首尾空格
            string rest = s.substr(prefix.size());
            b = rest.find_first_not_of(' ');
            if (b == string::npos) continue;
            e = rest.find_last_not_of(' ');
            rest = rest.substr(b, e - b + 1);
            // 提取第一个关键字
            size_t sp = rest.find(' ');
            string word = (sp == string::npos) ? rest : rest.substr(0, sp);
            if (!word.empty()) seen.insert(word);
        }
        // 转为向量返回
        return vector<string>(seen.begin(), seen.end());
    }
};
```

```javascript
var FindNextKeywords = function(commands, prefix) {
    const seen = new Set();
    for (const raw of commands) {
        // 忽略命令首尾无意义空格
        const s = raw.trim();
        // 以前缀开头才处理
        if (!s.startsWith(prefix)) continue;
        // 前缀之后的部分，再去掉首尾空格
        const rest = s.substring(prefix.length).trim();
        if (rest.length === 0) continue;
        // 第一个关键字：到下一个空格为止
        const sp = rest.indexOf(' ');
        const word = sp === -1 ? rest : rest.substring(0, sp);
        if (word.length > 0) seen.add(word);
    }
    // 转为数组并按字典序排序
    return Array.from(seen).sort();
};
```

```c
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>

// 字符串比较函数，用于 qsort 排序
static int _cmpstr(const void* a, const void* b) {
    return strcmp(*(const char**)a, *(const char**)b);
}

char** FindNextKeywords(char** commands, int commandsSize, char* prefix, int* returnSize) {
    char** result = (char**)malloc(commandsSize * sizeof(char*));
    int resultCount = 0;
    int prefixLen = strlen(prefix);

    for (int i = 0; i < commandsSize; i++) {
        // 跳过命令行首部空格
        const char* s = commands[i];
        while (*s == ' ') s++;
        // 检查是否以前缀开头
        if (strncmp(s, prefix, prefixLen) != 0) continue;
        // 前缀之后的剩余部分，去掉首部空格
        const char* rest = s + prefixLen;
        while (*rest == ' ') rest++;
        if (*rest == '\0') continue;
        // 提取第一个关键字（到下一个空格或字符串末尾）
        const char* p = rest;
        while (*p != '\0' && *p != ' ') p++;
        int wordLen = p - rest;
        // 去重检查
        bool duplicate = false;
        for (int j = 0; j < resultCount; j++) {
            if (strlen(result[j]) == (size_t)wordLen &&
                strncmp(result[j], rest, wordLen) == 0) {
                duplicate = true;
                break;
            }
        }
        if (duplicate) continue;
        // 添加新关键字
        result[resultCount] = (char*)malloc(wordLen + 1);
        strncpy(result[resultCount], rest, wordLen);
        result[resultCount][wordLen] = '\0';
        resultCount++;
    }
    // 按字典序排序
    qsort(result, resultCount, sizeof(char*), _cmpstr);
    *returnSize = resultCount;
    return result;
}
```
