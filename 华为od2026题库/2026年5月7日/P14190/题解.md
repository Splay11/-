## 解题思路

一次优化操作选择的是当前所有正数货物中某一个连续段，并让这一段全部减少 1。可以把数组看成柱状图：在某个高度层 `h` 上，所有 `goodProceeTime[i] >= h` 的位置会形成若干个连续段，每个连续段长度就是在这一层执行一次优化能减少的总处理时间。

因此，问题转化为：统计所有高度层上的正连续段长度，从中选择最多 `optimize` 个最大的长度，使总减少量最大，最终答案为原始总和减去最大减少量。

为了高效统计这些长度，按处理时间从高到低激活位置，并用并查集维护当前高度层的连续段。每个连通段在若干个连续高度层中保持不变，记录它的起始高度；当它因为更低高度的新位置加入而合并时，先结算旧连续段贡献了多少次对应长度。最后得到 `countByLen[len]`，表示长度为 `len` 的优化收益出现了多少次。再从大到小取前 `optimize` 个收益即可。

## 复杂度分析

设数组长度为 `n`，最大处理时间为 `V`。

每个位置只会被激活一次，每次最多与左右相邻位置合并，所有并查集操作总复杂度近似为 `O(n)`；按高度遍历需要 `O(V)`，最后按长度统计需要 `O(n)`。

时间复杂度：`O(n + V)`。

空间复杂度：`O(n + V)`。

在本题约束 `n <= 10^5`、`goodProceeTime[i] <= 10^5` 下可以稳定通过。

```python
from typing import List

class Solution:
    def minProcessTime(self, goodProceeTime: List[int], optimize: int) -> int:
        n = len(goodProceeTime)
        total = sum(goodProceeTime)
        if optimize <= 0 or total == 0:
            return total

        # 按高度分组，后续从高到低激活货物位置
        max_value = max(goodProceeTime, default=0)
        by_height = [[] for _ in range(max_value + 1)]
        for i, value in enumerate(goodProceeTime):
            if value > 0:
                by_height[value].append(i)

        # 并查集初始化
        parent = list(range(n))
        size = [1] * n
        start_height = [0] * n
        active = [False] * n
        count_by_len = [0] * (n + 1)

        # 查找并查集的根
        def find(x: int) -> int:
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        # 结算连续段在高度变化间隔内的收益贡献
        def close_component(root: int, h: int) -> None:
            times = start_height[root] - h
            if times > 0:
                count_by_len[size[root]] += times

        # 合并两个连续段
        def merge(a: int, b: int, h: int) -> int:
            ra, rb = find(a), find(b)
            if ra == rb:
                return ra
            # 合并前结算旧段贡献
            close_component(ra, h)
            close_component(rb, h)
            if size[ra] < size[rb]:
                ra, rb = rb, ra
            parent[rb] = ra
            size[ra] += size[rb]
            start_height[ra] = h
            return ra

        # 从高到低逐层激活
        for h in range(max_value, 0, -1):
            for idx in by_height[h]:
                active[idx] = True
                parent[idx] = idx
                size[idx] = 1
                start_height[idx] = h
                root = idx
                if idx > 0 and active[idx - 1]:
                    root = merge(root, idx - 1, h)
                if idx + 1 < n and active[idx + 1]:
                    root = merge(root, idx + 1, h)

        # 收尾：处理未结算的段
        for i in range(n):
            if active[i] and find(i) == i:
                close_component(i, 0)

        # 从大到小选取最优操作
        saved = 0
        left = optimize
        for length in range(n, 0, -1):
            if left == 0:
                break
            take = min(left, count_by_len[length])
            saved += take * length
            left -= take
        return total - saved
```

```java
import java.util.*;

class Solution {
    private int[] parent;
    private int[] size;
    private int[] startHeight;
    private long[] countByLen;

    public long minProcessTime(int[] goodProceeTime, int optimize) {
        int n = goodProceeTime.length;
        long total = 0;
        int maxValue = 0;
        for (int value : goodProceeTime) {
            total += value;
            maxValue = Math.max(maxValue, value);
        }
        if (optimize <= 0 || total == 0) {
            return total;
        }

        // 按高度分组
        List<Integer>[] byHeight = new ArrayList[maxValue + 1];
        for (int i = 0; i <= maxValue; i++) {
            byHeight[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            if (goodProceeTime[i] > 0) {
                byHeight[goodProceeTime[i]].add(i);
            }
        }

        // 并查集初始化
        parent = new int[n];
        size = new int[n];
        startHeight = new int[n];
        countByLen = new long[n + 1];
        boolean[] active = new boolean[n];
        Arrays.fill(size, 1);

        // 从高到低遍历
        for (int h = maxValue; h >= 1; h--) {
            for (int idx : byHeight[h]) {
                active[idx] = true;
                parent[idx] = idx;
                size[idx] = 1;
                startHeight[idx] = h;
                int root = idx;
                if (idx > 0 && active[idx - 1]) {
                    root = merge(root, idx - 1, h);
                }
                if (idx + 1 < n && active[idx + 1]) {
                    root = merge(root, idx + 1, h);
                }
            }
        }

        // 收尾结算
        for (int i = 0; i < n; i++) {
            if (active[i] && find(i) == i) {
                closeComponent(i, 0);
            }
        }

        // 选取最大收益的段
        long saved = 0;
        long left = optimize;
        for (int len = n; len >= 1 && left > 0; len--) {
            long take = Math.min(left, countByLen[len]);
            saved += take * len;
            left -= take;
        }
        return total - saved;
    }

    private int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    // 结算连续段在高度变化间隔内的收益贡献
    private void closeComponent(int root, int h) {
        long times = (long) startHeight[root] - h;
        if (times > 0) {
            countByLen[size[root]] += times;
        }
    }

    // 合并两个连续段
    private int merge(int a, int b, int h) {
        int ra = find(a);
        int rb = find(b);
        if (ra == rb) {
            return ra;
        }
        closeComponent(ra, h);
        closeComponent(rb, h);
        if (size[ra] < size[rb]) {
            int tmp = ra;
            ra = rb;
            rb = tmp;
        }
        parent[rb] = ra;
        size[ra] += size[rb];
        startHeight[ra] = h;
        return ra;
    }
}
```

```cpp
#include <vector>
#include <numeric>
#include <algorithm>
using namespace std;

class Solution {
public:
    long long minProcessTime(vector<int>& goodProceeTime, int optimize) {
        int n = (int)goodProceeTime.size();
        long long total = 0;
        int maxValue = 0;
        for (int x : goodProceeTime) {
            total += x;
            maxValue = max(maxValue, x);
        }
        if (optimize <= 0 || total == 0) return total;

        // 按高度分组
        vector<vector<int>> byHeight(maxValue + 1);
        for (int i = 0; i < n; ++i) {
            if (goodProceeTime[i] > 0) byHeight[goodProceeTime[i]].push_back(i);
        }

        // 并查集初始化
        vector<int> parent(n), sz(n, 1), startHeight(n, 0);
        vector<char> active(n, 0);
        vector<long long> countByLen(n + 1, 0);

        auto findRoot = [&](int x) {
            int r = x;
            while (parent[r] != r) r = parent[r];
            while (parent[x] != x) {
                int p = parent[x];
                parent[x] = r;
                x = p;
            }
            return r;
        };

        // 结算连续段在高度变化间隔内的收益贡献
        auto closeComponent = [&](int root, int h) {
            long long times = (long long)startHeight[root] - h;
            if (times > 0) countByLen[sz[root]] += times;
        };

        // 合并两个连续段
        auto mergeComponent = [&](int a, int b, int h) {
            int ra = findRoot(a), rb = findRoot(b);
            if (ra == rb) return ra;
            closeComponent(ra, h);
            closeComponent(rb, h);
            if (sz[ra] < sz[rb]) swap(ra, rb);
            parent[rb] = ra;
            sz[ra] += sz[rb];
            startHeight[ra] = h;
            return ra;
        };

        // 从高到低逐层激活
        for (int h = maxValue; h >= 1; --h) {
            for (int idx : byHeight[h]) {
                active[idx] = 1;
                parent[idx] = idx;
                sz[idx] = 1;
                startHeight[idx] = h;
                int root = idx;
                if (idx > 0 && active[idx - 1]) root = mergeComponent(root, idx - 1, h);
                if (idx + 1 < n && active[idx + 1]) root = mergeComponent(root, idx + 1, h);
            }
        }

        // 收尾
        for (int i = 0; i < n; ++i) {
            if (active[i] && findRoot(i) == i) closeComponent(i, 0);
        }

        // 选取最大收益的段
        long long saved = 0;
        long long left = optimize;
        for (int len = n; len >= 1 && left > 0; --len) {
            long long take = min(left, countByLen[len]);
            saved += take * len;
            left -= take;
        }
        return total - saved;
    }
};
```

```javascript
var minProcessTime = function(goodProceeTime, optimize) {
    const n = goodProceeTime.length;
    let total = goodProceeTime.reduce((a, b) => a + b, 0);
    let maxVal = Math.max(...goodProceeTime);
    if (optimize <= 0 || total === 0) return total;

    // 按高度分组
    const byHeight = new Array(maxVal + 1);
    for (let i = 0; i <= maxVal; i++) byHeight[i] = [];
    for (let i = 0; i < n; i++) {
        if (goodProceeTime[i] > 0) byHeight[goodProceeTime[i]].push(i);
    }

    // 并查集初始化
    const parent = new Array(n);
    const size = new Array(n);
    const startHeight = new Array(n);
    const active = new Array(n).fill(false);
    const countByLen = new Array(n + 1).fill(0);

    function find(x) {
        while (parent[x] !== x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }

    // 结算连续段在高度变化间隔内的收益贡献
    function closeComponent(root, h) {
        const times = startHeight[root] - h;
        if (times > 0) {
            countByLen[size[root]] += times;
        }
    }

    // 合并两个连续段
    function merge(a, b, h) {
        let ra = find(a), rb = find(b);
        if (ra === rb) return ra;
        closeComponent(ra, h);
        closeComponent(rb, h);
        if (size[ra] < size[rb]) [ra, rb] = [rb, ra];
        parent[rb] = ra;
        size[ra] += size[rb];
        startHeight[ra] = h;
        return ra;
    }

    // 从高到低遍历
    for (let h = maxVal; h >= 1; h--) {
        for (const idx of byHeight[h]) {
            active[idx] = true;
            parent[idx] = idx;
            size[idx] = 1;
            startHeight[idx] = h;
            let root = idx;
            if (idx > 0 && active[idx - 1]) root = merge(root, idx - 1, h);
            if (idx + 1 < n && active[idx + 1]) root = merge(root, idx + 1, h);
        }
    }

    // 收尾
    for (let i = 0; i < n; i++) {
        if (active[i] && find(i) === i) closeComponent(i, 0);
    }

    // 选取最大收益的段
    let saved = 0;
    let left = optimize;
    for (let len = n; len >= 1 && left > 0; len--) {
        const take = Math.min(left, countByLen[len]);
        saved += take * len;
        left -= take;
    }

    return total - saved;
};
```

```c
#include <stdlib.h>

static int _parent[100000];
static int _size[100000];
static int _startHeight[100000];
static char _active[100000];
static long long _countByLen[100001];

static int _find(int x) {
    while (_parent[x] != x) {
        _parent[x] = _parent[_parent[x]];
        x = _parent[x];
    }
    return x;
}

// 结算连续段在高度变化间隔内的收益贡献
static void _closeComponent(int root, int h) {
    long long times = (long long)_startHeight[root] - h;
    if (times > 0) {
        _countByLen[_size[root]] += times;
    }
}

// 合并两个连续段
static int _merge(int a, int b, int h) {
    int ra = _find(a), rb = _find(b);
    if (ra == rb) return ra;
    _closeComponent(ra, h);
    _closeComponent(rb, h);
    if (_size[ra] < _size[rb]) {
        int tmp = ra; ra = rb; rb = tmp;
    }
    _parent[rb] = ra;
    _size[ra] += _size[rb];
    _startHeight[ra] = h;
    return ra;
}

long long minProcessTime(int* goodProceeTime, int goodProceeTimeSize, int optimize) {
    int n = goodProceeTimeSize;
    long long total = 0;
    int maxVal = 0;
    for (int i = 0; i < n; i++) {
        total += goodProceeTime[i];
        if (goodProceeTime[i] > maxVal) maxVal = goodProceeTime[i];
    }
    if (optimize <= 0 || total == 0) return total;

    // 按高度分组
    int** byHeight = (int**)calloc(maxVal + 1, sizeof(int*));
    int* byHeightCnt = (int*)calloc(maxVal + 1, sizeof(int));
    for (int i = 0; i < n; i++) {
        if (goodProceeTime[i] > 0) byHeightCnt[goodProceeTime[i]]++;
    }
    for (int h = 1; h <= maxVal; h++) {
        if (byHeightCnt[h] > 0) {
            byHeight[h] = (int*)malloc(byHeightCnt[h] * sizeof(int));
            byHeightCnt[h] = 0;
        }
    }
    for (int i = 0; i < n; i++) {
        int v = goodProceeTime[i];
        if (v > 0) byHeight[v][byHeightCnt[v]++] = i;
    }

    // 并查集初始化
    for (int i = 0; i < n; i++) {
        _active[i] = 0;
    }
    for (int i = 0; i <= n; i++) {
        _countByLen[i] = 0;
    }

    // 从高到低逐层激活
    for (int h = maxVal; h >= 1; h--) {
        for (int j = 0; j < byHeightCnt[h]; j++) {
            int idx = byHeight[h][j];
            _active[idx] = 1;
            _parent[idx] = idx;
            _size[idx] = 1;
            _startHeight[idx] = h;
            int root = idx;
            if (idx > 0 && _active[idx - 1]) root = _merge(root, idx - 1, h);
            if (idx + 1 < n && _active[idx + 1]) root = _merge(root, idx + 1, h);
        }
    }

    // 收尾
    for (int i = 0; i < n; i++) {
        if (_active[i] && _find(i) == i) _closeComponent(i, 0);
    }

    // 选取最大收益的段
    long long saved = 0;
    long long left = optimize;
    for (int len = n; len >= 1 && left > 0; len--) {
        long long take = left < _countByLen[len] ? left : _countByLen[len];
        saved += take * len;
        left -= take;
    }

    // 清理
    for (int h = 1; h <= maxVal; h++) {
        free(byHeight[h]);
    }
    free(byHeight);
    free(byHeightCnt);

    return total - saved;
}
```
