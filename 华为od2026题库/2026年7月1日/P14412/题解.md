## 解题思路

本题需要统计物品件数并按规则排序，核心是**哈希表 + 自定义排序**。

**算法步骤**：

1. 遍历入库序列 `items`，用哈希表统计每种编号的出现次数 `cnt`，同时记录每种编号首次出现的下标 `first`。
2. 将所有不同的编号提取为列表。
3. 对列表按自定义规则排序：
   - 第一关键字：件数**降序**（多的在前）；
   - 第二关键字：首次出现位置**升序**（件数相同时，先入库的在前）。
4. 返回排序后的编号列表。

**示例分析**（以 `[1,2,2,3,3,3]` 为例）：
- 编号 1 出现 1 次，首次在位置 0
- 编号 2 出现 2 次，首次在位置 1
- 编号 3 出现 3 次，首次在位置 3
- 排序后：`[3, 2, 1]`

## 复杂度分析

- **时间复杂度**：$O(n + k \log k)$，其中 $n$ 为入库总件数，$k$ 为物品种类数。统计频率需 $O(n)$，排序需 $O(k \log k)$。由于 $n \le 100$ 且 $k \le 10$，实际运行极快。
- **空间复杂度**：$O(k)$，哈希表存储每种编号的频率和首次位置。

```python
class Solution:
    def warehouseInventory(self, items: list[int]) -> list[int]:
        # 统计每种编号的出现次数
        cnt = {}
        # 记录每种编号首次出现的位置
        first = {}
        for i, x in enumerate(items):
            cnt[x] = cnt.get(x, 0) + 1
            if x not in first:
                first[x] = i
        # 获取所有不同编号
        ids = list(cnt.keys())
        # 排序：件数降序，件数相同按首次出现位置升序
        ids.sort(key=lambda x: (-cnt[x], first[x]))
        return ids
```

```java
import java.util.*;

class Solution {
    public List<Integer> warehouseInventory(List<Integer> items) {
        // 统计每种编号的出现次数
        Map<Integer, Integer> cnt = new HashMap<>();
        // 记录每种编号首次出现的位置
        Map<Integer, Integer> first = new HashMap<>();
        int n = items.size();
        for (int i = 0; i < n; i++) {
            int x = items.get(i);
            cnt.put(x, cnt.getOrDefault(x, 0) + 1);
            if (!first.containsKey(x)) {
                first.put(x, i);
            }
        }
        // 获取所有不同编号
        List<Integer> ids = new ArrayList<>(cnt.keySet());
        // 排序：件数降序，件数相同按首次出现位置升序
        ids.sort((a, b) -> {
            int cmp = cnt.get(b) - cnt.get(a);
            if (cmp != 0) return cmp;
            return first.get(a) - first.get(b);
        });
        return ids;
    }
}
```

```cpp
#include <vector>
#include <unordered_map>
#include <algorithm>
using namespace std;

class Solution {
public:
    vector<int> warehouseInventory(vector<int>& items) {
        // 统计每种编号的出现次数
        unordered_map<int, int> cnt;
        // 记录每种编号首次出现的位置
        unordered_map<int, int> first;
        int n = items.size();
        for (int i = 0; i < n; ++i) {
            int x = items[i];
            cnt[x]++;
            if (first.find(x) == first.end()) {
                first[x] = i;
            }
        }
        // 获取所有不同编号
        vector<int> ids;
        for (auto& p : cnt) {
            ids.push_back(p.first);
        }
        // 排序：件数降序，件数相同按首次出现位置升序
        sort(ids.begin(), ids.end(), [&](int a, int b) {
            if (cnt[a] != cnt[b]) return cnt[a] > cnt[b];
            return first[a] < first[b];
        });
        return ids;
    }
};
```

```javascript
var warehouseInventory = function(items) {
    // 统计每种编号的出现次数和首次位置
    var cnt = {};
    var first = {};
    for (var i = 0; i < items.length; i++) {
        var x = items[i];
        cnt[x] = (cnt[x] || 0) + 1;
        if (!(x in first)) first[x] = i;
    }
    // 获取所有不同编号
    var ids = Object.keys(cnt).map(Number);
    // 排序：件数降序，件数相同按首次出现位置升序
    ids.sort(function(a, b) {
        if (cnt[b] !== cnt[a]) return cnt[b] - cnt[a];
        return first[a] - first[b];
    });
    return ids;
};
```

```c
#include <stdlib.h>

typedef struct {
    int id;
    int cnt;
    int first;
} Item;

static int cmp(const void* a, const void* b) {
    Item* x = (Item*)a;
    Item* y = (Item*)b;
    if (x->cnt != y->cnt) return y->cnt - x->cnt; // 件数降序
    return x->first - y->first;                     // 首次出现升序
}

int* warehouseInventory(int* items, int itemsSize, int* returnSize) {
    // 统计每种编号：暴力查找（n ≤ 100）
    int unique[100], cnt[100], firstPos[100];
    int uniCount = 0;

    for (int i = 0; i < itemsSize; i++) {
        int x = items[i];
        int found = -1;
        for (int j = 0; j < uniCount; j++) {
            if (unique[j] == x) { found = j; break; }
        }
        if (found == -1) {
            unique[uniCount] = x;
            cnt[uniCount] = 1;
            firstPos[uniCount] = i;
            uniCount++;
        } else {
            cnt[found]++;
        }
    }

    Item* arr = (Item*)malloc(uniCount * sizeof(Item));
    for (int i = 0; i < uniCount; i++) {
        arr[i].id = unique[i];
        arr[i].cnt = cnt[i];
        arr[i].first = firstPos[i];
    }

    qsort(arr, uniCount, sizeof(Item), cmp);

    int* result = (int*)malloc(uniCount * sizeof(int));
    for (int i = 0; i < uniCount; i++) result[i] = arr[i].id;
    *returnSize = uniCount;
    free(arr);
    return result;
}
```
