## 解题思路

本题要求在连续子序列中，满足至少包含 $K$ 个正数，且最多可丢弃 $M$ 个负数的条件下，最大化灵力值总和。

### 核心算法：枚举 + 小顶堆维护

**关键观察**：
1. 题目需要选择一段**连续**的子序列，而丢弃操作是在选定子序列后进行的
2. $M \le 5$ 非常小，这意味着我们可以在枚举区间时，用一个小顶堆（大小为 $M$）维护区间内绝对值最大的最多 $M$ 个负数
3. 丢弃负数的收益 = 丢弃的那些负数的绝对值之和（因为丢弃后相当于把这些负数从总和中移除，总和增加其绝对值）

**算法步骤**：
- 枚举左端点 $l$（$0$ 到 $N-1$），向右扩展右端点 $r$
- 扩展过程中维护：
  - 当前区间原始总和 `curSum`
  - 当前区间正数个数 `posCount`
  - 一个小顶堆 `minHeap` 存区间内绝对值最大的最多 $M$ 个负数，以及堆中元素绝对值之和 `negSum`
- 当 `posCount >= K` 时，当前区间有效，答案更新为 `max(ans, curSum + negSum)`
- 对于每个负数：如果堆未满则直接加入；否则若当前负数绝对值大于堆顶最小值，替换之

**时间复杂度**：
- 枚举 $l$ 和 $r$ 共 $O(N^2)$ 对区间
- 每个负数入堆/出堆操作为 $O(\log M)$，$M \le 5$ 时近似常数
- 总复杂度 $O(N^2)$，$N \le 10^4$ 时约为 $5 \times 10^7$ 次操作，在合理时限内可完成

**空间复杂度**：$O(N)$（前缀和数组 + 堆）

## 复杂度分析

- **时间复杂度**：$O(N^2)$，其中 $N \le 10^4$。最坏情况下约 $5 \times 10^7$ 次区间扩展，每次常数级操作（堆大小为 $M \le 5$），在 C++/Java/Python 均可接受
- **空间复杂度**：$O(N)$，用于前缀和和正数计数数组；运行时额外 $O(M)$ 堆空间

```python
import heapq

class Solution:
    def maxSpiritPower(self, N, K, M, A):
        # 前缀和与正数计数，方便快速判断
        pref = [0] * (N + 1)
        pos_cnt = [0] * (N + 1)
        for i in range(1, N + 1):
            pref[i] = pref[i - 1] + A[i - 1]
            pos_cnt[i] = pos_cnt[i - 1] + (1 if A[i - 1] > 0 else 0)

        # 全数组正数不足 K，无解
        if pos_cnt[N] < K:
            return -1

        ans = -10**18  # 足够小的初始值

        # 枚举左端点，向右扩展
        for l in range(N):
            cur_sum = 0       # 当前区间原始和
            pos_count = 0     # 当前区间正数个数
            min_heap = []     # 小顶堆，存负数绝对值（用于取最大M个）
            neg_sum = 0       # 堆中负数绝对值之和

            for r in range(l, N):
                val = A[r]
                cur_sum += val
                if val > 0:
                    pos_count += 1
                elif val < 0:
                    abs_val = -val
                    # 维护大小为 M 的小顶堆，存最大的 M 个负数绝对值
                    if len(min_heap) < M:
                        heapq.heappush(min_heap, abs_val)
                        neg_sum += abs_val
                    else:
                        # 如果当前负数绝对值比堆顶（最小的）大，替换
                        if min_heap and abs_val > min_heap[0]:
                            neg_sum -= heapq.heappop(min_heap)
                            heapq.heappush(min_heap, abs_val)
                            neg_sum += abs_val

                # 正数够 K 个时更新答案
                if pos_count >= K:
                    ans = max(ans, cur_sum + neg_sum)

        return ans if ans > -10**17 else -1
```

```java
import java.util.*;

class Solution {
    public long maxSpiritPower(int N, int K, int M, List<Integer> A) {
        // 前缀和与正数计数
        long[] pref = new long[N + 1];
        int[] posCnt = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            pref[i] = pref[i - 1] + A.get(i - 1);
            posCnt[i] = posCnt[i - 1] + (A.get(i - 1) > 0 ? 1 : 0);
        }

        // 全数组正数不足 K，无解
        if (posCnt[N] < K) {
            return -1;
        }

        long ans = Long.MIN_VALUE;

        // 枚举左端点，向右扩展
        for (int l = 0; l < N; l++) {
            long curSum = 0;        // 当前区间原始和
            int posCount = 0;       // 当前区间正数个数
            // 小顶堆维护区间内绝对值最大的最多 M 个负数
            PriorityQueue<Long> minHeap = new PriorityQueue<>();
            long negSum = 0;        // 堆中负数绝对值之和

            for (int r = l; r < N; r++) {
                int val = A.get(r);
                curSum += val;
                if (val > 0) {
                    posCount++;
                } else if (val < 0) {
                    long absVal = -(long) val;
                    // 维护大小为 M 的小顶堆
                    if (minHeap.size() < M) {
                        minHeap.offer(absVal);
                        negSum += absVal;
                    } else {
                        // 如果当前负数绝对值比堆顶大，替换
                        if (!minHeap.isEmpty() && absVal > minHeap.peek()) {
                            negSum -= minHeap.poll();
                            minHeap.offer(absVal);
                            negSum += absVal;
                        }
                    }
                }

                // 正数够 K 个时更新答案
                if (posCount >= K) {
                    ans = Math.max(ans, curSum + negSum);
                }
            }
        }

        return ans == Long.MIN_VALUE ? -1 : ans;
    }
}
```

```javascript
var maxSpiritPower = function(N, K, M, A) {
    // 全数组正数不足 K，无解
    var totalPos = 0;
    for (var i = 0; i < N; i++) {
        if (A[i] > 0) totalPos++;
    }
    if (totalPos < K) return -1;

    var ans = -Infinity;

    // 枚举左端点，向右扩展
    for (var l = 0; l < N; l++) {
        var curSum = 0;
        var posCount = 0;
        // 用数组维护绝对值最大的最多 M 个负数（小顶堆性质）
        var heap = [];  // 升序数组，heap[0] 最小
        var negSum = 0;

        for (var r = l; r < N; r++) {
            var val = A[r];
            curSum += val;
            if (val > 0) {
                posCount++;
            } else if (val < 0 && M > 0) {
                var absVal = -val;
                if (heap.length < M) {
                    // 插入并维护升序（小顶堆）
                    heap.push(absVal);
                    var j = heap.length - 1;
                    while (j > 0 && heap[j] < heap[(j - 1) >> 1]) {
                        var tmp = heap[j];
                        heap[j] = heap[(j - 1) >> 1];
                        heap[(j - 1) >> 1] = tmp;
                        j = (j - 1) >> 1;
                    }
                    negSum += absVal;
                } else if (absVal > heap[0]) {
                    // 替换堆顶并下沉
                    negSum = negSum - heap[0] + absVal;
                    heap[0] = absVal;
                    var i = 0, sz = heap.length;
                    while (1) {
                        var left = 2 * i + 1, right = 2 * i + 2, smallest = i;
                        if (left < sz && heap[left] < heap[smallest]) smallest = left;
                        if (right < sz && heap[right] < heap[smallest]) smallest = right;
                        if (smallest === i) break;
                        var tmp = heap[i];
                        heap[i] = heap[smallest];
                        heap[smallest] = tmp;
                        i = smallest;
                    }
                }
            }

            // 正数够 K 个时更新答案
            if (posCount >= K) {
                var cand = curSum + negSum;
                if (cand > ans) ans = cand;
            }
        }
    }

    return ans === -Infinity ? -1 : ans;
};
```

```c
#include <stdlib.h>
#include <limits.h>

// 小顶堆：维护区间内绝对值最大的最多 M 个负数
typedef struct {
    long long data[6];  // M <= 5
    int size;
    long long sum;
} MinHeap;

// 插入一个值到小顶堆，M 为最大容量
static void heapPush(MinHeap* h, int M, long long val) {
    if (h->size < M) {
        // 堆未满，直接插入并上浮
        int i = h->size++;
        while (i > 0) {
            int p = (i - 1) / 2;
            if (h->data[p] <= val) break;
            h->data[i] = h->data[p];
            i = p;
        }
        h->data[i] = val;
        h->sum += val;
    } else if (val > h->data[0]) {
        // 堆已满，若当前值大于堆顶（最小值），则替换
        h->sum -= h->data[0];
        h->sum += val;
        h->data[0] = val;
        // 下沉
        int i = 0, sz = h->size;
        while (1) {
            int left = 2 * i + 1, right = 2 * i + 2, smallest = i;
            if (left < sz && h->data[left] < h->data[smallest]) smallest = left;
            if (right < sz && h->data[right] < h->data[smallest]) smallest = right;
            if (smallest == i) break;
            long long tmp = h->data[i];
            h->data[i] = h->data[smallest];
            h->data[smallest] = tmp;
            i = smallest;
        }
    }
}

long long maxSpiritPower(int N, int K, int M, int* A) {
    // 全数组正数不足 K，无解
    int totalPos = 0;
    for (int i = 0; i < N; i++) {
        if (A[i] > 0) totalPos++;
    }
    if (totalPos < K) return -1;

    long long ans = LLONG_MIN;

    // 枚举左端点，向右扩展
    for (int l = 0; l < N; l++) {
        long long curSum = 0;
        int posCount = 0;
        MinHeap heap;
        heap.size = 0;
        heap.sum = 0;

        for (int r = l; r < N; r++) {
            int val = A[r];
            curSum += val;
            if (val > 0) {
                posCount++;
            } else if (val < 0 && M > 0) {
                long long absVal = -(long long)val;
                heapPush(&heap, M, absVal);
            }

            // 正数够 K 个时更新答案
            if (posCount >= K) {
                long long cand = curSum + heap.sum;
                if (cand > ans) ans = cand;
            }
        }
    }

    return ans == LLONG_MIN ? -1 : ans;
}
```
