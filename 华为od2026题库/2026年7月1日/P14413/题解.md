## 解题思路

本题是经典「三数之和」的变体，额外增加了**奇偶性约束**：三元组中至少有两个元素为奇数。

**算法核：排序 + 枚举 + 双指针**

1. 首先将数组升序排序，便于跳过重复元素和控制双指针方向。
2. 枚举第一个元素 `nums[i]`（`i` 从 `0` 到 `n-3`），若与前一元素相同则跳过以去重。
3. 对固定的 `i`，使用双指针 `j = i+1`、`k = n-1` 在剩余部分中搜索：
   - 若 `nums[i] + nums[j] + nums[k] == target`，统计三个数的奇数个数；若 ≥ 2 则加入答案，然后同时移动 `j`、`k` 并跳过重复值。
   - 若和小于 `target`，`j` 右移。
   - 若和大于 `target`，`k` 左移。
4. 由于已排序，答案中的每个三元组自然升序，组间字典序也自然满足。

**去重关键**：固定第一个元素和双指针移动时，均需跳过连续相同值，避免产生重复三元组。

## 复杂度分析

- **时间复杂度**：$O(n^2)$。排序 $O(n \log n)$，枚举 `i` 和双指针总计 $O(n^2)$。
- **空间复杂度**：$O(1)$（不计输出数组）。排序使用原地排序，双指针不需要额外空间。

class Solution:
    def threeSumWithParity(self, nums: list[int], target: int) -> list[list[int]]:
        # 排序便于双指针和去重
        nums.sort()
        n = len(nums)
        ans = []
        for i in range(n - 2):
            # 跳过重复的第一个元素，避免重复三元组
            if i > 0 and nums[i] == nums[i - 1]:
                continue
            j, k = i + 1, n - 1
            while j < k:
                s = nums[i] + nums[j] + nums[k]
                if s == target:
                    # 统计三个数中的奇数个数
                    odd = (nums[i] & 1) + (nums[j] & 1) + (nums[k] & 1)
                    if odd >= 2:
                        ans.append([nums[i], nums[j], nums[k]])
                    # 跳过重复的 j 和 k
                    while j < k and nums[j] == nums[j + 1]:
                        j += 1
                    while j < k and nums[k] == nums[k - 1]:
                        k -= 1
                    j += 1
                    k -= 1
                elif s < target:
                    j += 1
                else:
                    k -= 1
        return ans

import java.util.*;

class Solution {
    public List<List<Integer>> threeSumWithParity(List<Integer> nums, int target) {
        // 排序便于双指针和去重
        Collections.sort(nums);
        int n = nums.size();
        List<List<Integer>> ans = new ArrayList<>();
        for (int i = 0; i < n - 2; i++) {
            // 跳过重复的第一个元素，避免重复三元组
            if (i > 0 && nums.get(i).equals(nums.get(i - 1))) continue;
            int j = i + 1, k = n - 1;
            while (j < k) {
                long s = (long)nums.get(i) + nums.get(j) + nums.get(k);
                if (s == target) {
                    // 统计三个数中的奇数个数
                    int odd = (nums.get(i) & 1) + (nums.get(j) & 1) + (nums.get(k) & 1);
                    if (odd >= 2) {
                        ans.add(Arrays.asList(nums.get(i), nums.get(j), nums.get(k)));
                    }
                    // 跳过重复的 j 和 k
                    while (j < k && nums.get(j).equals(nums.get(j + 1))) j++;
                    while (j < k && nums.get(k).equals(nums.get(k - 1))) k--;
                    j++; k--;
                } else if (s < target) {
                    j++;
                } else {
                    k--;
                }
            }
        }
        return ans;
    }
}

#include <vector>
#include <algorithm>
using namespace std;

class Solution {
public:
    vector<vector<int>> threeSumWithParity(vector<int>& nums, int target) {
        // 排序便于双指针和去重
        sort(nums.begin(), nums.end());
        int n = nums.size();
        vector<vector<int>> ans;
        for (int i = 0; i < n - 2; ++i) {
            // 跳过重复的第一个元素，避免重复三元组
            if (i > 0 && nums[i] == nums[i - 1]) continue;
            int j = i + 1, k = n - 1;
            while (j < k) {
                // 使用 long long 防止 int 溢出
                long long s = (long long)nums[i] + nums[j] + nums[k];
                if (s == target) {
                    // 统计三个数中的奇数个数
                    int odd = (nums[i] & 1) + (nums[j] & 1) + (nums[k] & 1);
                    if (odd >= 2) {
                        ans.push_back({nums[i], nums[j], nums[k]});
                    }
                    // 跳过重复的 j 和 k
                    while (j < k && nums[j] == nums[j + 1]) ++j;
                    while (j < k && nums[k] == nums[k - 1]) --k;
                    ++j; --k;
                } else if (s < target) {
                    ++j;
                } else {
                    --k;
                }
            }
        }
        return ans;
    }
};

#include <stdlib.h>

// 升序排序比较函数
static int cmp(const void* a, const void* b) {
    int x = *(int*)a, y = *(int*)b;
    if (x < y) return -1;
    if (x > y) return 1;
    return 0;
}

int** threeSumWithParity(int* nums, int numsSize, int target, int* returnSize, int** returnColumnSizes) {
    // 排序，便于双指针和去重
    qsort(nums, numsSize, sizeof(int), cmp);

    int cap = 4096;
    int** ans = (int**)malloc(cap * sizeof(int*));
    *returnColumnSizes = (int*)malloc(cap * sizeof(int));
    int cnt = 0;

    for (int i = 0; i < numsSize - 2; i++) {
        // 跳过重复的第一个元素
        if (i > 0 && nums[i] == nums[i - 1]) continue;
        int j = i + 1, k = numsSize - 1;
        while (j < k) {
            long long s = (long long)nums[i] + nums[j] + nums[k];
            if (s == target) {
                // 统计奇数个数
                int odd = (nums[i] & 1) + (nums[j] & 1) + (nums[k] & 1);
                if (odd >= 2) {
                    if (cnt >= cap) {
                        cap *= 2;
                        ans = (int**)realloc(ans, cap * sizeof(int*));
                        *returnColumnSizes = (int*)realloc(*returnColumnSizes, cap * sizeof(int));
                    }
                    ans[cnt] = (int*)malloc(3 * sizeof(int));
                    ans[cnt][0] = nums[i];
                    ans[cnt][1] = nums[j];
                    ans[cnt][2] = nums[k];
                    (*returnColumnSizes)[cnt] = 3;
                    cnt++;
                }
                // 跳过重复的 j 和 k
                while (j < k && nums[j] == nums[j + 1]) j++;
                while (j < k && nums[k] == nums[k - 1]) k--;
                j++;
                k--;
            } else if (s < target) {
                j++;
            } else {
                k--;
            }
        }
    }

    *returnSize = cnt;
    return ans;
}
