## 解题思路

本题是**字符串解析 + 模拟分组**，可按以下步骤处理：

1. **合法性检查**：遍历 $SN$，若出现非字母、非数字且非 `-` 的字符，直接返回 `""`。
2. **提取有效字符**：按从左到右顺序收集所有字母与数字，忽略 `-`；若收集结果为空（如全为破折号），返回 `""`。
3. **统一大小写**：将小写字母转为大写。
4. **按 $m$ 分组**：
   - 设有效字符总数为 $n$，余数 $r = n \bmod m$；
   - 若 $r \neq 0$，第一段长度为 $r$，其余每段长度为 $m$；
   - 若 $r = 0$，每段长度均为 $m$。
5. 用 `-` 连接各段输出。

**常见假解**：

- 保留原破折号位置再插入新破折号 → 应先去掉所有 `-$ 再重排；
- 忘记转大写 → `2n` 应输出 `2N`；
- $n \bmod m \neq 0$ 时仍从第一段取 $m$ 个字符 → 首段长度应为余数 $r$；
- $n \bmod m = 0$ 时错误地让首段更短 → 此时应全部均分，无特例首段；
- 把 `-` 当成有效字符计入分组 → 只统计字母数字；
- 遇到 `@`、空格等仍继续处理 → 异常场景应直接返回 `""`。

## 复杂度分析

设 $n$ 为 $SN$ 长度，$|sn| < 10000$，$m \le 10$。

- 时间：$O(n)$，单次遍历提取与分组；
- 空间：$O(n)$ 存储提取后的字符序列。

```python
class Solution:
    def rearrangeSN(self, sn: str, m: int) -> str:
        # 仅允许字母、数字与破折号；其它字符视为异常
        for c in sn:
            if not (c.isalnum() or c == "-"):
                return ""

        chars = []
        # 按原顺序提取字母数字并转大写
        for c in sn:
            if c.isalnum():
                chars.append(c.upper())

        if not chars:
            return ""

        n = len(chars)
        rem = n % m
        groups = []
        idx = 0
        # 有余数时首段长度不足 m
        if rem != 0:
            groups.append("".join(chars[idx : idx + rem]))
            idx += rem
        while idx < n:
            groups.append("".join(chars[idx : idx + m]))
            idx += m
        return "-".join(groups)
```

```java
public class Solution {
    public String rearrangeSN(String sn, int m) {
        // 异常字符：非字母数字且非破折号
        for (int i = 0; i < sn.length(); i++) {
            char c = sn.charAt(i);
            if (!Character.isLetterOrDigit(c) && c != '-') return "";
        }

        StringBuilder chars = new StringBuilder();
        for (int i = 0; i < sn.length(); i++) {
            char c = sn.charAt(i);
            if (Character.isLetterOrDigit(c)) {
                chars.append(Character.toUpperCase(c));
            }
        }
        if (chars.length() == 0) return "";

        int n = chars.length();
        int rem = n % m;
        StringBuilder ans = new StringBuilder();
        int idx = 0;
        // 首段承接余数长度
        if (rem != 0) {
            if (ans.length() > 0) ans.append('-');
            ans.append(chars, idx, idx + rem);
            idx += rem;
        }
        while (idx < n) {
            if (ans.length() > 0) ans.append('-');
            ans.append(chars, idx, idx + m);
            idx += m;
        }
        return ans.toString();
    }
}
```

```cpp
#include <cctype>
#include <string>
#include <vector>

using namespace std;

class Solution {
 public:
  string rearrangeSN(string sn, int m) {
    // 校验字符集：仅字母、数字、破折号合法
    for (char c : sn) {
      if (!isalnum((unsigned char)c) && c != '-') return "";
    }

    vector<char> chars;
    for (char c : sn) {
      if (isalnum((unsigned char)c)) {
        chars.push_back((char)toupper((unsigned char)c));
      }
    }
    if (chars.empty()) return "";

    int n = (int)chars.size();
    int rem = n % m;
    vector<string> groups;
    int idx = 0;
    // 不能整除 m 时，余数段放在第一段
    if (rem != 0) {
      groups.emplace_back(chars.begin(), chars.begin() + rem);
      idx += rem;
    }
    while (idx < n) {
      groups.emplace_back(chars.begin() + idx, chars.begin() + idx + m);
      idx += m;
    }

    string ans;
    for (size_t i = 0; i < groups.size(); i++) {
      if (i) ans.push_back('-');
      ans += groups[i];
    }
    return ans;
  }
};
```

```javascript
/**
 * @param {string} sn - 一个有效的 SN 地址字符串
 * @param {number} m - 每个分组恰好包含的字符数
 * @return {string} - 重新排列后的 SN 地址字符串（含外层双引号）
 */
var rearrangeSN = function(sn, m) {
    // 合法性检查：仅允许字母、数字、破折号
    for (var k = 0; k < sn.length; k++) {
        var ch = sn[k];
        if (!((ch >= '0' && ch <= '9') ||
              (ch >= 'A' && ch <= 'Z') ||
              (ch >= 'a' && ch <= 'z') ||
               ch === '-')) {
            return '""';
        }
    }

    // 提取字母数字并转大写
    var chars = [];
    for (var k = 0; k < sn.length; k++) {
        var ch = sn[k];
        if ((ch >= '0' && ch <= '9') ||
            (ch >= 'A' && ch <= 'Z') ||
            (ch >= 'a' && ch <= 'z')) {
            chars.push(ch.toUpperCase());
        }
    }

    var n = chars.length;
    // 无有效字符
    if (n === 0) return '""';

    var r = n % m;
    var groups = [];
    var idx = 0;
    // 首段（若有余数）
    if (r !== 0) {
        groups.push(chars.slice(idx, idx + r).join(''));
        idx += r;
    }
    // 其后每段长 m
    while (idx < n) {
        groups.push(chars.slice(idx, idx + m).join(''));
        idx += m;
    }
    return '"' + groups.join('-') + '"';
};
```

```c
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char* rearrangeSN(char* sn, int m) {
    // 合法性检查：只允许字母、数字、破折号
    int len = (int)strlen(sn);

    for (int i = 0; i < len; i++) {
        char c = sn[i];
        if (!isalnum((unsigned char)c) && c != '-') {
            // 异常场景，返回空字符串
            char* result = (char*)malloc(3);
            result[0] = '"';
            result[1] = '"';
            result[2] = '\0';
            return result;
        }
    }

    // 提取字母数字并转大写
    char* chars = (char*)malloc(len + 1);
    int cnt = 0;
    for (int i = 0; i < len; i++) {
        char c = sn[i];
        if (isalnum((unsigned char)c)) {
            chars[cnt++] = (char)toupper((unsigned char)c);
        }
    }
    chars[cnt] = '\0';

    // 无有效字符
    if (cnt == 0) {
        free(chars);
        char* result = (char*)malloc(3);
        result[0] = '"';
        result[1] = '"';
        result[2] = '\0';
        return result;
    }

    // 分组：余数 r，首段长 r（若 r>0），其余每段长 m
    int n = cnt;
    int r = n % m;
    int groups = (n + m - 1) / m;
    int resultLen = 2 + n + (groups - 1) + 1;
    char* result = (char*)malloc(resultLen);

    int pos = 0;
    result[pos++] = '"';
    int idx = 0;
    // 首段（若有余数）
    if (r != 0) {
        for (int j = 0; j < r; j++) result[pos++] = chars[idx++];
        if (idx < n) result[pos++] = '-';
    }
    // 其后每段长 m
    while (idx < n) {
        for (int j = 0; j < m; j++) result[pos++] = chars[idx++];
        if (idx < n) result[pos++] = '-';
    }
    result[pos++] = '"';
    result[pos] = '\0';

    free(chars);
    return result;
}
```
