## 解题思路

本题是**固定规则**下的离散事件模拟，不是求最优调度。

1. 按时间维护事件队列：**车辆到达**（优先级低）与**充电桩释放**（优先级高）；同一时刻先处理释放再处理到达。
2. 用**小根堆**维护 $N$ 个桩的最早空闲时刻；用**队列**维护按到达先后（FCFS）等待的车辆。
3. 在时刻 $t$ 尝试分配：若队首车辆满足 $t-AT \le WT$，则占用最早可用的桩，开始充电并在 $t+CT$ 时刻产生释放事件；否则队首失败出队。
4. 统计失败出队次数即为答案。

**区分度**：

- 忽略「同刻先释放后到达」会错样例 2 类边界。
- 等待队列不按到达顺序会错多车同到场景。
- 把 $WT$ 当成充电时长或把失败车辆仍占桩都会错。

## 复杂度分析

- 时间：$O((M + K)\log M)$，$K$ 为成功充电次数（每辆车至多一次成功分配），事件堆与桩堆均为对数级。
- 空间：$O(M + N)$。

```python
import heapq
from typing import List


class Solution:
    def countFailedCharging(self, n: int, cars: List[List[int]]) -> int:
        events = []
        for i, (at, _ct, _wt) in enumerate(cars):
            heapq.heappush(events, (at, 1, "A", i))  # 到达事件，同刻优先级 1

        piles = [0] * n
        heapq.heapify(piles)  # 桩空闲时刻小根堆
        waiting = []
        failed = 0

        while events:
            t = events[0][0]
            batch = []
            while events and events[0][0] == t:
                batch.append(heapq.heappop(events))
            batch.sort(key=lambda x: x[1])  # 先处理释放(0)再到达(1)

            for _time, _pri, kind, data in batch:
                if kind == "F":
                    heapq.heappush(piles, t)  # 桩在 t 时刻可用
                else:
                    waiting.append(data)

            while piles and piles[0] <= t and waiting:
                car = waiting[0]
                at, ct, wt = cars[car]
                if t - at > wt:  # 等待超时，失败
                    waiting.pop(0)
                    failed += 1
                    continue
                heapq.heappop(piles)
                finish = t + ct
                heapq.heappush(events, (finish, 0, "F", 0))  # 充电结束释放
                waiting.pop(0)

        return failed
```

```java
import java.util.ArrayList;
import java.util.List;
import java.util.PriorityQueue;

public class Solution {
    private static class Event implements Comparable<Event> {
        int time, pri, type, id;

        Event(int time, int pri, int type, int id) {
            this.time = time;
            this.pri = pri;
            this.type = type;
            this.id = id;
        }

        public int compareTo(Event o) {
            if (time != o.time) return Integer.compare(time, o.time);
            return Integer.compare(pri, o.pri);
        }
    }

    public int countFailedCharging(int n, int[][] cars) {
        PriorityQueue<Event> events = new PriorityQueue<>();
        for (int i = 0; i < cars.length; i++) {
            events.offer(new Event(cars[i][0], 1, 1, i));
        }

        PriorityQueue<Integer> piles = new PriorityQueue<>();
        for (int i = 0; i < n; i++) piles.offer(0);

        List<Integer> waiting = new ArrayList<>();
        int failed = 0;

        while (!events.isEmpty()) {
            int t = events.peek().time;
            List<Event> batch = new ArrayList<>();
            while (!events.isEmpty() && events.peek().time == t) {
                batch.add(events.poll());
            }
            batch.sort((a, b) -> Integer.compare(a.pri, b.pri));

            for (Event e : batch) {
                if (e.type == 0) {
                    piles.offer(t);
                } else {
                    waiting.add(e.id);
                }
            }

            while (!piles.isEmpty() && piles.peek() <= t && !waiting.isEmpty()) {
                int car = waiting.get(0);
                int at = cars[car][0], ct = cars[car][1], wt = cars[car][2];
                if (t - at > wt) {
                    waiting.remove(0);
                    failed++;
                    continue;
                }
                piles.poll();
                events.offer(new Event(t + ct, 0, 0, 0));
                waiting.remove(0);
            }
        }
        return failed;
    }
}
```

```cpp
#include <algorithm>
#include <queue>
#include <tuple>
#include <vector>

using namespace std;

class Solution {
 public:
  int countFailedCharging(int n, vector<vector<int>>& cars) {
    using Event = tuple<int, int, char, int>;
    priority_queue<Event, vector<Event>, greater<Event>> events;
    for (int i = 0; i < (int)cars.size(); i++) {
      events.push({cars[i][0], 1, 'A', i});
    }

    priority_queue<int, vector<int>, greater<int>> piles;
    for (int i = 0; i < n; i++) piles.push(0);

    vector<int> waiting;
    int failed = 0;

    while (!events.empty()) {
      int t = get<0>(events.top());
      vector<Event> batch;
      while (!events.empty() && get<0>(events.top()) == t) {
        batch.push_back(events.top());
        events.pop();
      }
      sort(batch.begin(), batch.end(),
           [](const Event& a, const Event& b) { return get<1>(a) < get<1>(b); });

      for (auto& e : batch) {
        if (get<2>(e) == 'F')
          piles.push(t);
        else
          waiting.push_back(get<3>(e));
      }

      while (!piles.empty() && piles.top() <= t && !waiting.empty()) {
        int car = waiting[0];
        int at = cars[car][0], ct = cars[car][1], wt = cars[car][2];
        if (t - at > wt) {
          waiting.erase(waiting.begin());
          failed++;
          continue;
        }
        piles.pop();
        events.push({t + ct, 0, 'F', 0});
        waiting.erase(waiting.begin());
      }
    }
    return failed;
  }
};
```

```javascript
/**
 * @param {number} n 充电桩数量
 * @param {number[][]} cars 每辆车的 [到达时间, 充电时间, 最大等待时长]
 * @return {number} 充电失败的车辆数量
 */
var countFailedCharging = function(n, cars) {
    // 事件小根堆：按 (time, priority) 排序，0=释放 1=到达
    var events = [];
    for (var i = 0; i < cars.length; i++) {
        events.push({ time: cars[i][0], pri: 1, type: 1, idx: i });
    }
    events.sort(function(a, b) { return a.time - b.time || a.pri - b.pri; });

    // 充电桩最早可用时间的小根堆
    var piles = [];
    for (var i = 0; i < n; i++) piles.push(0);
    piles.sort(function(a, b) { return a - b; });

    var waiting = [];
    var failed = 0;

    while (events.length > 0) {
        var t = events[0].time;
        var batch = [];
        while (events.length > 0 && events[0].time === t) {
            batch.push(events.shift());
        }
        // 同时间先处理释放(pri=0)再处理到达(pri=1)
        batch.sort(function(a, b) { return a.pri - b.pri; });

        for (var i = 0; i < batch.length; i++) {
            var e = batch[i];
            if (e.type === 0) {
                piles.push(t);
                piles.sort(function(a, b) { return a - b; });
            } else {
                waiting.push(e.idx);
            }
        }

        // 把可用桩分配给等待队列中的车(FCFS)
        while (piles.length > 0 && piles[0] <= t && waiting.length > 0) {
            var carIdx = waiting[0];
            var at = cars[carIdx][0], ct = cars[carIdx][1], wt = cars[carIdx][2];
            if (t - at > wt) {
                waiting.shift();
                failed++;
                continue;
            }
            piles.shift();
            events.push({ time: t + ct, pri: 0, type: 0, idx: 0 });
            events.sort(function(a, b) { return a.time - b.time || a.pri - b.pri; });
            waiting.shift();
        }
    }

    return failed;
};
```

```c
#include <stdlib.h>

/* --- 事件最小堆 (按 time 排序，time 相同时按 priority 排序) --- */
typedef struct { int time, pri, type, idx; } Evt;
typedef struct { Evt* a; int sz, cap; } EHeap;

static void epush(EHeap* h, int t, int p, int ty, int id) {
    if (h->sz == h->cap) {
        h->cap = h->cap ? h->cap * 2 : 64;
        h->a = (Evt*)realloc(h->a, h->cap * sizeof(Evt));
    }
    int i = h->sz++;
    while (i > 0) {
        int pi = (i - 1) / 2;
        if (h->a[pi].time < t) break;
        if (h->a[pi].time == t && h->a[pi].pri <= p) break;
        h->a[i] = h->a[pi]; i = pi;
    }
    h->a[i].time = t; h->a[i].pri = p; h->a[i].type = ty; h->a[i].idx = id;
}

static int eempty(EHeap* h) { return h->sz == 0; }
static int etop_t(EHeap* h) { return h->a[0].time; }

static void epop(EHeap* h, int* t, int* p, int* ty, int* id) {
    *t = h->a[0].time; *p = h->a[0].pri; *ty = h->a[0].type; *id = h->a[0].idx;
    h->a[0] = h->a[--h->sz];
    int i = 0;
    while (1) {
        int l = 2 * i + 1, r = 2 * i + 2, best = i;
        if (l < h->sz && (h->a[l].time < h->a[best].time ||
            (h->a[l].time == h->a[best].time && h->a[l].pri < h->a[best].pri))) best = l;
        if (r < h->sz && (h->a[r].time < h->a[best].time ||
            (h->a[r].time == h->a[best].time && h->a[r].pri < h->a[best].pri))) best = r;
        if (best == i) break;
        Evt tmp = h->a[i]; h->a[i] = h->a[best]; h->a[best] = tmp; i = best;
    }
}

/* --- 整数最小堆 (充电桩释放时间) --- */
typedef struct { int* a; int sz, cap; } IHeap;

static void ipush(IHeap* h, int v) {
    if (h->sz == h->cap) {
        h->cap = h->cap ? h->cap * 2 : 64;
        h->a = (int*)realloc(h->a, h->cap * sizeof(int));
    }
    int i = h->sz++;
    while (i > 0 && h->a[(i - 1) / 2] > v) { h->a[i] = h->a[(i - 1) / 2]; i = (i - 1) / 2; }
    h->a[i] = v;
}

static int itop(IHeap* h) { return h->sz > 0 ? h->a[0] : -1; }
static int iempty(IHeap* h) { return h->sz == 0; }

static void ipop(IHeap* h) {
    h->a[0] = h->a[--h->sz];
    int i = 0;
    while (1) {
        int l = 2 * i + 1, r = 2 * i + 2, best = i;
        if (l < h->sz && h->a[l] < h->a[best]) best = l;
        if (r < h->sz && h->a[r] < h->a[best]) best = r;
        if (best == i) break;
        int tmp = h->a[i]; h->a[i] = h->a[best]; h->a[best] = tmp; i = best;
    }
}

/* --- 主算法：事件驱动充电桩调度 --- */
int countFailedCharging(int n, int** cars, int carsSize) {
    EHeap ev = {NULL, 0, 0};
    for (int i = 0; i < carsSize; i++) epush(&ev, cars[i][0], 1, 1, i);

    IHeap pl = {NULL, 0, 0};
    for (int i = 0; i < n; i++) ipush(&pl, 0);

    int* wait = (int*)malloc(carsSize * sizeof(int));
    int wh = 0, wc = 0, failed = 0;

    while (!eempty(&ev)) {
        int ct = etop_t(&ev);

        Evt* batch = (Evt*)malloc(ev.sz * sizeof(Evt));
        int bsz = 0;
        while (!eempty(&ev) && etop_t(&ev) == ct) {
            int t, p, ty, id;
            epop(&ev, &t, &p, &ty, &id);
            batch[bsz].time = t; batch[bsz].pri = p; batch[bsz].type = ty; batch[bsz].idx = id;
            bsz++;
        }

        // 同时间先处理释放(pri=0)再处理到达(pri=1)
        for (int i = 0; i < bsz - 1; i++)
            for (int j = 0; j < bsz - 1 - i; j++)
                if (batch[j].pri > batch[j + 1].pri) {
                    Evt tmp = batch[j]; batch[j] = batch[j + 1]; batch[j + 1] = tmp;
                }

        for (int i = 0; i < bsz; i++) {
            if (batch[i].type == 0) ipush(&pl, ct);
            else wait[wc++] = batch[i].idx;
        }
        free(batch);

        // 分配可用充电桩给等待队列中的车辆(FCFS)
        while (!iempty(&pl) && itop(&pl) <= ct && wh < wc) {
            int car = wait[wh];
            int at = cars[car][0], carct = cars[car][1], wt = cars[car][2];
            if (ct - at > wt) { wh++; failed++; continue; }
            ipop(&pl);
            epush(&ev, ct + carct, 0, 0, 0);
            wh++;
        }
    }

    free(ev.a); free(pl.a); free(wait);
    return failed;
}
```
