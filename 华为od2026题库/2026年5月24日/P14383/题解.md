## 解题思路

数据范围 $n \le 15$，可直接 **枚举子集**（位掩码 $0 \ldots 2^n-1$）：

1. 将 `timestamps` **排序**，便于按索引取元素；
2. 对每个 `mask`，取出被选中的时间戳列表 `picked`；
3. 双重循环检查任意两元素之差是否 $\ge$ `minInterval`；
4. 合法则答案 $+1$（空集 `mask=0` 自动合法）。

也可用排序后的 DP：按最后一个选中的下标计数，但 $2^{15}$ 暴力足够且不易写错。

**区分度**：若忘记排序，乱序下标与子集元素对应会错；若只检查相邻元素会漏掉非相邻冲突对。

## 复杂度分析

- 时间：$O(2^n \cdot n^2)$，$n \le 15$ 时约 $3 \times 10^4$ 次检查，可忽略。
- 空间：$O(n)$ 存选中元素。

```python
from typing import List

class Solution:
    def countValidPlans(self, timestamps: List[int], minInterval: int) -> int:
        ts = sorted(timestamps)  # 排序后按 mask 位取元素
        n = len(ts)
        ans = 0
        for mask in range(1 << n):  # 枚举所有子集，含空集
            picked = [ts[i] for i in range(n) if mask & (1 << i)]
            ok = True
            for i in range(len(picked)):
                for j in range(i + 1, len(picked)):
                    if picked[j] - picked[i] < minInterval:  # 任意一对都要满足间隔
                        ok = False
                        break
                if not ok:
                    break
            if ok:
                ans += 1
        return ans
```

```java
public class Solution {
    public int countValidPlans(int[] timestamps, int minInterval) {
        int[] ts = timestamps.clone();
        java.util.Arrays.sort(ts);
        int n = ts.length, ans = 0;

            for (int i = 0; i < n; i++)

            boolean ok = true;
            for (int i = 0; i < picked.size() && ok; i++)
                for (int j = i + 1; j < picked.size(); j++)
                    if (picked.get(j) - picked.get(i) < minInterval) ok = false;
            if (ok) ans++;
        }
        return ans;
    }
}
```

```c
int countValidPlans(int* timestamps, int timestampsSize, int minInterval) {
    int n = timestampsSize;
    int cmp(const void* a, const void* b) {
        return (*(int*)a - *(int*)b);
    }
    qsort(timestamps, n, sizeof(int), cmp);
    int total = 1 << n;
    int ans = 0;
    for (int mask = 0; mask < total; mask++) {
        int picked[16];
        int cnt = 0;
        for (int i = 0; i < n; i++)
            if (mask & (1 << i)) picked[cnt++] = timestamps[i];
        int valid = 1;
        for (int i = 0; i < cnt && valid; i++)
            for (int j = i + 1; j < cnt; j++)
                if (picked[j] - picked[i] < minInterval) valid = 0;
        if (valid) ans++;
    }
    return ans;
}
```
