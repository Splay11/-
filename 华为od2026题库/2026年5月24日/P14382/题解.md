## 解题思路

本题是**自定义词法分析 + 从左到右求值 + 结果钳制 + 按字节取反**的综合实现题。

1. **合法性**：先检查字符集（仅字母数字与 `+/-`）、长度 $\le 10^4$，再用有限状态机式地扫描「数 + 运算符 + 数 + …」结构；`-` 仅可作为二元运算符，不能引出负数字面。
2. **读数**：
   - `0x/0X` 后读十六进制，数值 $\le 999$；
   - `0o/0O` 后读八进制（仅 `0-7`），数值 $\le 999$；
   - 否则按十进制读连续数字（允许前导零），数值 $\le 999$。
3. **运算**：按出现顺序做加减，得到有符号中间结果。
4. **钳制**：结果压到 $[-255,255]$。
5. **输出**：`out = (~result) & 0xFF`，格式化为带双引号的 `"0x"` + 两位大写十六进制（如 `"0xBB"`）；任一步失败返回 `"NA"`。

**易错点**：`023` 是十进制 $23$ 而非八进制；`0o89` 非法；含空格或 `=` 直接 `NA`；取反是对**钳制后的 int** 做 Python/Java 语义下的按位非再截低 8 位。

## 复杂度分析

- 时间：$O(|s|)$，单次扫描。
- 空间：$O(1)$。

```python
class Solution:
    def processExpression(self, inputStr: str) -> str:
        s = inputStr
        if len(s) > 10000:
            return '"NA"'
        allowed = set("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789+-")
        for ch in s:
            if ch not in allowed:
                return '"NA"'
        if not s:
            return '"NA"'
        pos = 0

        def parse_number():
            nonlocal pos
            n = len(s)
            if pos >= n:
                return False, 0
            if s[pos] == "0" and pos + 1 < n and s[pos + 1] in "xX":
                pos += 2
                val, ok = 0, False
                while pos < n:
                    c = s[pos]
                    if c.isdigit():
                        d = ord(c) - 48
                    elif "a" <= c <= "f":
                        d = ord(c) - 87
                    elif "A" <= c <= "F":
                        d = ord(c) - 55
                    else:
                        break
                    ok, val = True, val * 16 + d
                    pos += 1
                return (ok and val <= 999, val) if ok else (False, 0)
            if s[pos] == "0" and pos + 1 < n and s[pos + 1] in "oO":
                pos += 2
                val, ok = 0, False
                while pos < n and "0" <= s[pos] <= "7":
                    ok, val = True, val * 8 + (ord(s[pos]) - 48)
                    pos += 1
                return (ok and val <= 999, val) if ok else (False, 0)
            if not s[pos].isdigit():
                return False, 0
            val = 0
            while pos < n and s[pos].isdigit():
                val = val * 10 + (ord(s[pos]) - 48)
                pos += 1
            return (val <= 999, val) if val <= 999 else (False, 0)

        ok, result = parse_number()
        if not ok:
            return '"NA"'
        while pos < len(s):
            op = s[pos]
            if op not in "+-":
                return '"NA"'
            pos += 1
            ok, val = parse_number()
            if not ok:
                return '"NA"'
            result = result + val if op == "+" else result - val
        if pos != len(s):
            return '"NA"'
        if result > 255:
            result = 255
        elif result < -255:
            result = -255
        return '"0x%02X"' % ((~result) & 0xFF)
```

```java
public class Solution {
    private String s;
    private int pos;

    private boolean isAllowed(char ch) {
        if (ch >= 'a' && ch <= 'z') return true;
        if (ch >= 'A' && ch <= 'Z') return true;
        if (ch >= '0' && ch <= '9') return true;
        return ch == '+' || ch == '-';
    }

    private boolean parseNumber(int[] out) {
        int n = s.length();
        if (pos >= n) return false;
        // 0x/0X：十六进制字面量
        if (s.charAt(pos) == '0' && pos + 1 < n) {
            char next = s.charAt(pos + 1);
            if (next == 'x' || next == 'X') {
                pos += 2;
                int val = 0;
                boolean ok = false;
                while (pos < n) {
                    char c = s.charAt(pos);
                    int d;
                    if (c >= '0' && c <= '9') d = c - '0';
                    else if (c >= 'a' && c <= 'f') d = c - 'a' + 10;
                    else if (c >= 'A' && c <= 'F') d = c - 'A' + 10;
                    else break;
                    ok = true;
                    val = val * 16 + d;
                    pos++;
                }
                if (!ok || val > 999) return false;
                out[0] = val;
                return true;
            }
            // 0o/0O：八进制字面量（仅 0-7）
            if (next == 'o' || next == 'O') {
                pos += 2;
                int val = 0;
                boolean ok = false;
                while (pos < n && s.charAt(pos) >= '0' && s.charAt(pos) <= '7') {
                    ok = true;
                    val = val * 8 + (s.charAt(pos) - '0');
                    pos++;
                }
                if (!ok || val > 999) return false;
                out[0] = val;
                return true;
            }
        }
        // 其余按十进制读连续数字（允许前导零）
        if (!Character.isDigit(s.charAt(pos))) return false;
        int val = 0;
        while (pos < n && Character.isDigit(s.charAt(pos))) {
            val = val * 10 + (s.charAt(pos) - '0');
            pos++;
        }
        if (val > 999) return false;
        out[0] = val;
        return true;
    }

    public String processExpression(String inputStr) {
        s = inputStr;
        pos = 0;
        if (s.length() > 10000) return "\"NA\"";
        // 仅允许字母、数字与 +/-
        for (int i = 0; i < s.length(); i++) {
            if (!isAllowed(s.charAt(i))) return "\"NA\"";
        }
        if (s.isEmpty()) return "\"NA\"";

        int[] num = new int[1];
        if (!parseNumber(num)) return "\"NA\"";
        int result = num[0];

        // 从左到右交替读运算符与操作数
        while (pos < s.length()) {
            char op = s.charAt(pos++);
            if (op != '+' && op != '-') return "\"NA\"";
            if (!parseNumber(num)) return "\"NA\"";
            result = (op == '+') ? result + num[0] : result - num[0];
        }
        if (pos != s.length()) return "\"NA\"";

        // 钳制到 [-255, 255]
        if (result > 255) result = 255;
        else if (result < -255) result = -255;

        // 按位取反后保留低 8 位，输出带双引号
        int out = (~result) & 0xFF;
        return String.format("\"0x%02X\"", out);
    }
}
```

```cpp
class Solution {
    string s;
    size_t pos;

    bool parseNumber(int &val) {
        size_t n = s.size();
        if (pos >= n) {
            return false;
        }
        // 0x/0X：十六进制字面量
        if (s[pos] == '0' && pos + 1 < n) {
            char next = s[pos + 1];
            if (next == 'x' || next == 'X') {
                pos += 2;
                int v = 0;
                bool ok = false;
                while (pos < n) {
                    char c = s[pos];
                    int d;
                    if (c >= '0' && c <= '9')
                        d = c - '0';
                    else if (c >= 'a' && c <= 'f')
                        d = c - 'a' + 10;
                    else if (c >= 'A' && c <= 'F')
                        d = c - 'A' + 10;
                    else
                        break;
                    ok = true;
                    v = v * 16 + d;
                    pos++;
                }
                if (!ok || v > 999) return false;
                val = v;
                return true;
            }
            // 0o/0O：八进制字面量（仅 0-7）
            if (next == 'o' || next == 'O') {
                pos += 2;
                int v = 0;
                bool ok = false;
                while (pos < n && s[pos] >= '0' && s[pos] <= '7') {
                    ok = true;
                    v = v * 8 + (s[pos] - '0');
                    pos++;
                }
                if (!ok || v > 999) return false;
                val = v;
                return true;
            }
        }
        // 其余按十进制读连续数字（允许前导零）
        if (!isdigit((unsigned char)s[pos])) return false;
        int v = 0;
        while (pos < n && isdigit((unsigned char)s[pos])) {
            v = v * 10 + (s[pos] - '0');
            pos++;
        }
        if (v > 999) return false;
        val = v;
        return true;
    }

    string processExpression(string inputStr) {
        s = inputStr;
        pos = 0;
        if (s.size() > 10000) return "\"NA\"";
        // 仅允许字母、数字与 +/-
        for (char ch : s) {
            if (!((ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') ||
                  (ch >= '0' && ch <= '9') || ch == '+' || ch == '-')) {
                return "\"NA\"";
            }
        }
        if (s.empty()) return "\"NA\"";

        int val = 0;
        if (!parseNumber(val)) return "\"NA\"";
        int result = val;

        // 从左到右交替读运算符与操作数
        while (pos < s.size()) {
            char op = s[pos++];
            if (op != '+' && op != '-') return "\"NA\"";
            if (!parseNumber(val)) return "\"NA\"";
            result = (op == '+') ? result + val : result - val;
        }
        if (pos != s.size()) return "\"NA\"";

        // 钳制到 [-255, 255]
        if (result > 255)
            result = 255;
        else if (result < -255)
            result = -255;

        // 对钳制后的值按位取反，再保留低 8 位，输出带双引号
        int out = (~result) & 0xFF;
        char buf[16];
        snprintf(buf, sizeof(buf), "\"0x%02X\"", out);
        return string(buf);
    }
```

```javascript
/**
 * P14382 表达式处理 - JavaScript 解答
 * @param {string} inputStr 输入的表达式字符串（不含外层双引号）
 * @return {string} 十六进制结果或 "NA"
 */
var processExpression = function(inputStr) {
    var s = inputStr;
    var n = s.length;

    // 长度检查
    if (n > 10000) return '"NA"';

    // 字符合法性检查：仅允许字母、数字与 +/-
    for (var i = 0; i < n; i++) {
        var c = s.charCodeAt(i);
        var valid = (c >= 48 && c <= 57)
            || (c >= 65 && c <= 90)
            || (c >= 97 && c <= 122)
            || c === 43 || c === 45;
        if (!valid) return '"NA"';
    }

    if (n === 0) return '"NA"';

    var pos = 0;

    // 解析整数：十六进制 / 八进制 / 十进制
    function parseNumber() {
        if (pos >= n) return { ok: false };

        // 0x / 0X 十六进制
        if (s[pos] === '0' && pos + 1 < n && (s[pos + 1] === 'x' || s[pos + 1] === 'X')) {
            pos += 2;
            var v = 0, ok = false;
            while (pos < n) {
                var c = s.charCodeAt(pos), d;
                if (c >= 48 && c <= 57) d = c - 48;
                else if (c >= 97 && c <= 102) d = c - 87;
                else if (c >= 65 && c <= 70) d = c - 55;
                else break;
                ok = true; v = v * 16 + d; pos++;
            }
            if (!ok || v > 999) return { ok: false };
            return { ok: true, val: v };
        }

        // 0o / 0O 八进制（仅 0-7）
        if (s[pos] === '0' && pos + 1 < n && (s[pos + 1] === 'o' || s[pos + 1] === 'O')) {
            pos += 2;
            var v = 0, ok = false;
            while (pos < n && s[pos] >= '0' && s[pos] <= '7') {
                ok = true; v = v * 8 + (s.charCodeAt(pos) - 48); pos++;
            }
            if (!ok || v > 999) return { ok: false };
            return { ok: true, val: v };
        }

        // 十进制
        if (s.charCodeAt(pos) < 48 || s.charCodeAt(pos) > 57) return { ok: false };
        var v = 0;
        while (pos < n && s.charCodeAt(pos) >= 48 && s.charCodeAt(pos) <= 57) {
            v = v * 10 + (s.charCodeAt(pos) - 48); pos++;
        }
        if (v > 999) return { ok: false };
        return { ok: true, val: v };
    }

    var first = parseNumber();
    if (!first.ok) return '"NA"';
    var result = first.val;

    // 从左到右交替读取运算符和操作数
    while (pos < n) {
        var op = s[pos++];
        if (op !== '+' && op !== '-') return '"NA"';
        var num = parseNumber();
        if (!num.ok) return '"NA"';
        result = (op === '+') ? result + num.val : result - num.val;
    }

    if (pos !== n) return '"NA"';

    // 钳制到 [-255, 255]
    if (result > 255) result = 255;
    else if (result < -255) result = -255;

    // 按位取反后保留低 8 位
    var out = (~result) & 0xFF;
    var hex = out.toString(16).toUpperCase();
    if (hex.length < 2) hex = '0' + hex;
    return '"0x' + hex + '"';
};
```

```c
/**
 * P14382 表达式处理 - C 语言解答
 * @param s 输入的表达式字符串（不含外层双引号）
 * @return 十六进制结果或 "NA"（带双引号），使用静态缓冲区
 */
char* processExpression(char* s) {
    static char buf[32];
    int n = strlen(s);

    // 长度与字符合法性检查
    if (n > 10000) { strcpy(buf, "\"NA\""); return buf; }
    for (int i = 0; i < n; i++) {
        char c = s[i];
        if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') ||
              (c >= '0' && c <= '9') || c == '+' || c == '-')) {
            strcpy(buf, "\"NA\""); return buf;
        }
    }
    if (n == 0) { strcpy(buf, "\"NA\""); return buf; }

    int pos = 0;

    // 解析整数：0x/0X 十六进制、0o/0O 八进制、十进制
    bool parseNumber(int* val) {
        if (pos >= n) return false;
        if (s[pos] == '0' && pos + 1 < n) {
            if (s[pos + 1] == 'x' || s[pos + 1] == 'X') {
                pos += 2; int v = 0; bool ok = false;
                while (pos < n) { int d; char c = s[pos];
                    if (c >= '0' && c <= '9') d = c - '0';
                    else if (c >= 'a' && c <= 'f') d = c - 'a' + 10;
                    else if (c >= 'A' && c <= 'F') d = c - 'A' + 10;
                    else break;
                    ok = true; v = v * 16 + d; pos++;
                }
                if (!ok || v > 999) return false;
                *val = v; return true;
            }
            if (s[pos + 1] == 'o' || s[pos + 1] == 'O') {
                pos += 2; int v = 0; bool ok = false;
                while (pos < n && s[pos] >= '0' && s[pos] <= '7')
                    { ok = true; v = v * 8 + (s[pos] - '0'); pos++; }
                if (!ok || v > 999) return false;
                *val = v; return true;
            }
        }
        if (!isdigit((unsigned char)s[pos])) return false;
        int v = 0;
        while (pos < n && isdigit((unsigned char)s[pos]))
            { v = v * 10 + (s[pos] - '0'); pos++; }
        if (v > 999) return false;
        *val = v; return true;
    }

    int result;
    if (!parseNumber(&result)) { strcpy(buf, "\"NA\""); return buf; }

    // 从左到右求值
    while (pos < n) {
        char op = s[pos++];
        if (op != '+' && op != '-') { strcpy(buf, "\"NA\""); return buf; }
        int val;
        if (!parseNumber(&val)) { strcpy(buf, "\"NA\""); return buf; }
        result = (op == '+') ? result + val : result - val;
    }
    if (pos != n) { strcpy(buf, "\"NA\""); return buf; }

    // 钳制与按位取反
    if (result > 255) result = 255;
    else if (result < -255) result = -255;
    int out = (~result) & 0xFF;
    sprintf(buf, "\"0x%02X\"", out);
    return buf;
}
```
