## 解题思路

本题要求实现语义化版本号（SemVer）的比较器，找出列表中最大的版本号，同时需要处理格式合法性检查。

### 核心思路

1. **格式合法性检查**：遍历所有版本号，逐一校验格式：
   - 禁止字符：`_`、`~`
   - 边界限制：不能以 `-`、`+`、`.` 开头或结尾
   - `+` 最多出现一次
   - `Major.Minor.Patch` 必须是三段纯数字，无前导零（`"0"` 本身除外），值在 `[0, 2^31-1]` 内
   - 先行版/构建信息段内部不允许连续的点（`..`）、不能含 `_`/`~`
   - 若存在任何非法版本号，直接返回 `""`

2. **版本号解析**：将合法版本号拆分为三部分：
   - `Major.Minor.Patch`：按 `.` 切分后转为整数
   - 先行版（pre-release）：第一个 `-` 到第一个 `+`（或字符串末尾）之间的部分
   - 构建信息（build）：第一个 `+` 之后的部分

3. **版本号比较**：按优先级从高到低比较：
   - 先比较 `Major` → `Minor` → `Patch`（数值比较，大的优先）
   - 再比较先行版：有信息 > 无信息；按 `.` 切分后逐段比较，每段纯数字按数值比较，非纯数字按 ASCII 字典序比较（`1 < A < a`），前缀相同时内容多的优先
   - 最后比较构建信息：规则同先行版

4. **找最大值**：遍历列表，记录当前最大版本号的索引，相等时保留第一个。

### 关键细节

- 纯数字标识符按**数值**比较（如 `10 > 2`），非纯数字按 **ASCII 字典序**比较（如 `2 > 10` 因为 `'2' > '1'`）
- 比较先行版/构建信息时，若一方为空、另一方非空，非空方更大
- 前缀相同（如 `a-b` vs `a-b-c`），内容多的优先

## 复杂度分析

- **时间复杂度**：$O(N \cdot L)$，其中 $N$ 是版本号数量（$\le 100$），$L$ 是每个版本号的最大长度（$\le 1024$）。格式检查需要遍历每个字符，比较操作也需要遍历标识符段，总体在线性范围内。
- **空间复杂度**：$O(L)$，解析过程中需要临时存储切分后的字段，为常数级额外空间。

class Solution:
    def findLatestVersion(self, versions):
        # 先检查所有版本号的格式合法性
        for v in versions:
            if not self._isValid(v):
                return ""

        # 按规则比较，找出最大的版本号
        best_idx = 0
        for i in range(1, len(versions)):
            if self._compare(versions[i], versions[best_idx]) > 0:
                best_idx = i
        return versions[best_idx]

    def _isValid(self, s):
        """检查版本号格式是否合法"""
        if not s:
            return False
        # 禁止字符：_ ~
        if '_' in s or '~' in s:
            return False
        # 不能以 - + . 开头或结尾
        if s[0] in '-+.' or s[-1] in '-+.':
            return False
        # + 最多出现一次
        if s.count('+') > 1:
            return False

        plus_pos = s.find('+')
        minus_pos = s.find('-')

        # 提取 Major.Minor.Patch 部分
        if minus_pos != -1:
            if plus_pos != -1 and minus_pos > plus_pos:
                return False  # - 必须在 + 之前
            core = s[:minus_pos]
        elif plus_pos != -1:
            core = s[:plus_pos]
        else:
            core = s

        # 检查 Major.Minor.Patch 必须是三段纯数字无前导零
        parts = core.split('.')
        if len(parts) != 3:
            return False
        for p in parts:
            if not p.isdigit():
                return False
            if len(p) > 1 and p[0] == '0':
                return False
            if int(p) > (1 << 31) - 1:
                return False

        # 检查先行版和构建信息段合法性
        pre_release = ""
        build = ""
        if minus_pos != -1:
            if plus_pos != -1 and plus_pos > minus_pos:
                pre_release = s[minus_pos + 1:plus_pos]
                build = s[plus_pos + 1:]
            elif plus_pos != -1:
                pre_release = s[minus_pos + 1:plus_pos]
                build = s[plus_pos + 1:]
            else:
                pre_release = s[minus_pos + 1:]
        elif plus_pos != -1:
            build = s[plus_pos + 1:]

        if not self._checkIdentifiers(pre_release):
            return False
        if not self._checkIdentifiers(build):
            return False

        return True

    def _checkIdentifiers(self, s):
        """检查标识符段合法性：不含 _ ~，不含空段，不以 . 开头或结尾"""
        if not s:
            return True
        if '_' in s or '~' in s:
            return False
        if s.startswith('.') or s.endswith('.'):
            return False
        if '..' in s:
            return False
        return True

    def _compare(self, a, b):
        """比较两个版本号：>0 表示 a > b"""
        a_core, a_pre, a_build = self._parse(a)
        b_core, b_pre, b_build = self._parse(b)

        # 1. 比较 Major.Minor.Patch（数值）
        for i in range(3):
            if a_core[i] != b_core[i]:
                return a_core[i] - b_core[i]

        # 2. 比较先行版
        pre_cmp = self._compareIdentifiers(a_pre, b_pre)
        if pre_cmp != 0:
            return pre_cmp

        # 3. 比较构建信息
        return self._compareIdentifiers(a_build, b_build)

    def _parse(self, s):
        """解析版本号为 (core_list, pre_string, build_string)"""
        plus_pos = s.find('+')
        minus_pos = s.find('-')

        core = s
        pre_release = ""
        build = ""

        if minus_pos != -1:
            core = s[:minus_pos]
            if plus_pos != -1 and plus_pos > minus_pos:
                pre_release = s[minus_pos + 1:plus_pos]
                build = s[plus_pos + 1:]
            elif plus_pos != -1:
                build = s[plus_pos + 1:]
                pre_release = s[minus_pos + 1:plus_pos]
            else:
                pre_release = s[minus_pos + 1:]
        elif plus_pos != -1:
            core = s[:plus_pos]
            build = s[plus_pos + 1:]

        core_parts = [int(x) for x in core.split('.')]
        return core_parts, pre_release, build

    def _compareIdentifiers(self, a, b):
        """比较先行版或构建信息：有信息 > 无信息"""
        if not a and not b:
            return 0
        if not a:
            return -1  # 无信息 < 有信息
        if not b:
            return 1

        a_parts = a.split('.')
        b_parts = b.split('.')
        for i in range(min(len(a_parts), len(b_parts))):
            cmp = self._comparePart(a_parts[i], b_parts[i])
            if cmp != 0:
                return cmp

        # 前缀相同，内容多者优先
        return len(a_parts) - len(b_parts)

    def _comparePart(self, a, b):
        """比较单个标识符段"""
        a_is_num = a.isdigit()
        b_is_num = b.isdigit()

        if a_is_num and b_is_num:
            # 纯数字按数值比较
            return int(a) - int(b)
        if a_is_num:
            return -1  # 数字 < 非数字（ASCII 序）
        if b_is_num:
            return 1
        # 非纯数字按 ASCII 字典序比较
        if a < b:
            return -1
        elif a > b:
            return 1
        return 0

import java.util.List;

class Solution {
    public String findLatestVersion(List<String> versions) {
        // 先检查所有版本号的格式合法性
        for (String v : versions) {
            if (!isValid(v)) return "";
        }

        // 按规则比较，找出最大的版本号
        int best = 0;
        for (int i = 1; i < versions.size(); i++) {
            if (compare(versions.get(i), versions.get(best)) > 0) {
                best = i;
            }
        }
        return versions.get(best);
    }

    // 判断是否为纯数字
    private boolean isPureNumber(String s) {
        if (s.isEmpty()) return false;
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) < '0' || s.charAt(i) > '9') return false;
        }
        return true;
    }

    // 检查标识符段合法性
    private boolean checkIdentifiers(String s) {
        if (s.isEmpty()) return true;
        if (s.charAt(0) == '.' || s.charAt(s.length() - 1) == '.') return false;
        if (s.contains("..")) return false;
        if (s.contains("_") || s.contains("~")) return false;
        return true;
    }

    // 格式合法性检查
    private boolean isValid(String s) {
        if (s.isEmpty()) return false;
        // 禁止字符
        if (s.contains("_") || s.contains("~")) return false;
        char first = s.charAt(0), last = s.charAt(s.length() - 1);
        if (first == '-' || first == '+' || first == '.') return false;
        if (last == '-' || last == '+' || last == '.') return false;

        int plusPos = s.indexOf('+');
        if (plusPos != -1 && s.indexOf('+', plusPos + 1) != -1) return false;

        int minusPos = s.indexOf('-');
        if (minusPos != -1 && plusPos != -1 && minusPos > plusPos) return false;

        // 提取 Major.Minor.Patch
        String core;
        if (minusPos != -1) core = s.substring(0, minusPos);
        else if (plusPos != -1) core = s.substring(0, plusPos);
        else core = s;

        String[] parts = core.split("\\.");
        if (parts.length != 3) return false;
        for (String p : parts) {
            if (!isPureNumber(p)) return false;
            if (p.length() > 1 && p.charAt(0) == '0') return false;
            if (Long.parseLong(p) > (1L << 31) - 1) return false;
        }

        // 提取并检查先行版和构建信息
        String preRelease = "", build = "";
        if (minusPos != -1) {
            if (plusPos != -1 && plusPos > minusPos) {
                preRelease = s.substring(minusPos + 1, plusPos);
                build = s.substring(plusPos + 1);
            } else if (plusPos != -1) {
                build = s.substring(plusPos + 1);
                preRelease = s.substring(minusPos + 1, plusPos);
            } else {
                preRelease = s.substring(minusPos + 1);
            }
        } else if (plusPos != -1) {
            build = s.substring(plusPos + 1);
        }

        return checkIdentifiers(preRelease) && checkIdentifiers(build);
    }

    // 比较两个版本号
    private int compare(String a, String b) {
        // 解析 core 的三段数字
        int aPlus = a.indexOf('+'), aMinus = a.indexOf('-');
        int bPlus = b.indexOf('+'), bMinus = b.indexOf('-');

        String aCore, bCore;
        if (aMinus != -1) aCore = a.substring(0, aMinus);
        else if (aPlus != -1) aCore = a.substring(0, aPlus);
        else aCore = a;
        if (bMinus != -1) bCore = b.substring(0, bMinus);
        else if (bPlus != -1) bCore = b.substring(0, bPlus);
        else bCore = b;

        String[] aParts = aCore.split("\\.");
        String[] bParts = bCore.split("\\.");
        // 比较 Major.Minor.Patch（数值）
        for (int i = 0; i < 3; i++) {
            int va = Integer.parseInt(aParts[i]);
            int vb = Integer.parseInt(bParts[i]);
            if (va != vb) return va < vb ? -1 : 1;
        }

        // 提取先行版
        String aPre = "", bPre = "", aBuild = "", bBuild = "";
        if (aMinus != -1) {
            if (aPlus != -1 && aPlus > aMinus) {
                aPre = a.substring(aMinus + 1, aPlus);
                aBuild = a.substring(aPlus + 1);
            } else if (aPlus != -1) {
                aBuild = a.substring(aPlus + 1);
                aPre = a.substring(aMinus + 1, aPlus);
            } else {
                aPre = a.substring(aMinus + 1);
            }
        } else if (aPlus != -1) aBuild = a.substring(aPlus + 1);

        if (bMinus != -1) {
            if (bPlus != -1 && bPlus > bMinus) {
                bPre = b.substring(bMinus + 1, bPlus);
                bBuild = b.substring(bPlus + 1);
            } else if (bPlus != -1) {
                bBuild = b.substring(bPlus + 1);
                bPre = b.substring(bMinus + 1, bPlus);
            } else {
                bPre = b.substring(bMinus + 1);
            }
        } else if (bPlus != -1) bBuild = b.substring(bPlus + 1);

        // 比较先行版
        int preCmp = compareIdentifiers(aPre, bPre);
        if (preCmp != 0) return preCmp;

        // 比较构建信息
        return compareIdentifiers(aBuild, bBuild);
    }

    // 比较标识符列表
    private int compareIdentifiers(String a, String b) {
        if (a.isEmpty() && b.isEmpty()) return 0;
        if (a.isEmpty()) return -1;
        if (b.isEmpty()) return 1;

        String[] aParts = a.split("\\.");
        String[] bParts = b.split("\\.");
        int minLen = Math.min(aParts.length, bParts.length);
        for (int i = 0; i < minLen; i++) {
            int cmp = comparePart(aParts[i], bParts[i]);
            if (cmp != 0) return cmp;
        }
        return Integer.compare(aParts.length, bParts.length);
    }

    // 比较单个标识符段
    private int comparePart(String a, String b) {
        boolean aNum = isPureNumber(a), bNum = isPureNumber(b);
        if (aNum && bNum) {
            long va = Long.parseLong(a), vb = Long.parseLong(b);
            return Long.compare(va, vb);
        }
        if (aNum) return -1;  // 数字 < 非数字
        if (bNum) return 1;
        return a.compareTo(b);
    }
}

class Solution {
public:
    string findLatestVersion(vector<string>& versions) {
        // 先检查所有版本号的格式合法性
        for (const string& v : versions) {
            if (!isValid(v)) return "";
        }

        // 按规则比较，找出最大的版本号
        int best = 0;
        for (int i = 1; i < (int)versions.size(); i++) {
            if (compare(versions[i], versions[best]) > 0) best = i;
        }
        return versions[best];
    }

private:
    bool isPureNumber(const string& s) const {
        if (s.empty()) return false;
        for (char c : s) if (c < '0' || c > '9') return false;
        return true;
    }

    bool checkIdentifiers(const string& s) const {
        if (s.empty()) return true;
        if (s.front() == '.' || s.back() == '.') return false;
        if (s.find("..") != string::npos) return false;
        if (s.find('_') != string::npos || s.find('~') != string::npos) return false;
        return true;
    }

    bool isValid(const string& s) const {
        if (s.empty()) return false;
        // 禁止字符
        if (s.find('_') != string::npos || s.find('~') != string::npos) return false;
        if (s.front() == '-' || s.front() == '+' || s.front() == '.') return false;
        if (s.back() == '-' || s.back() == '+' || s.back() == '.') return false;

        size_t plusPos = s.find('+');
        if (plusPos != string::npos && s.find('+', plusPos + 1) != string::npos) return false;

        size_t minusPos = s.find('-');
        if (minusPos != string::npos && plusPos != string::npos && minusPos > plusPos) return false;

        // 提取 Major.Minor.Patch
        string core;
        if (minusPos != string::npos) core = s.substr(0, minusPos);
        else if (plusPos != string::npos) core = s.substr(0, plusPos);
        else core = s;

        // 按 . 切分并检查三段纯数字
        vector<string> parts;
        size_t start = 0;
        while (start <= core.size()) {
            size_t pos = core.find('.', start);
            if (pos == string::npos) {
                parts.push_back(core.substr(start));
                break;
            }
            parts.push_back(core.substr(start, pos - start));
            start = pos + 1;
        }
        if (parts.size() != 3) return false;
        for (const string& p : parts) {
            if (!isPureNumber(p)) return false;
            if (p.size() > 1 && p[0] == '0') return false;
            if (stoll(p) > (1LL << 31) - 1) return false;
        }

        // 检查先行版和构建信息
        string preRelease, build;
        if (minusPos != string::npos) {
            if (plusPos != string::npos && plusPos > minusPos) {
                preRelease = s.substr(minusPos + 1, plusPos - minusPos - 1);
                build = s.substr(plusPos + 1);
            } else if (plusPos != string::npos) {
                build = s.substr(plusPos + 1);
                preRelease = s.substr(minusPos + 1, plusPos - minusPos - 1);
            } else {
                preRelease = s.substr(minusPos + 1);
            }
        } else if (plusPos != string::npos) {
            build = s.substr(plusPos + 1);
        }

        return checkIdentifiers(preRelease) && checkIdentifiers(build);
    }

    // 按 . 切分字符串
    vector<string> split(const string& s) const {
        vector<string> res;
        size_t start = 0;
        while (start <= s.size()) {
            size_t pos = s.find('.', start);
            if (pos == string::npos) { res.push_back(s.substr(start)); break; }
            res.push_back(s.substr(start, pos - start));
            start = pos + 1;
        }
        return res;
    }

    int comparePart(const string& a, const string& b) const {
        bool aNum = isPureNumber(a), bNum = isPureNumber(b);
        if (aNum && bNum) {
            long long va = stoll(a), vb = stoll(b);
            if (va != vb) return va < vb ? -1 : 1;
            return 0;
        }
        if (aNum) return -1;  // 数字 < 非数字
        if (bNum) return 1;
        return a.compare(b);  // ASCII 字典序
    }

    int compareIdentifiers(const string& a, const string& b) const {
        if (a.empty() && b.empty()) return 0;
        if (a.empty()) return -1;  // 无信息 < 有信息
        if (b.empty()) return 1;

        auto aParts = split(a);
        auto bParts = split(b);
        size_t minLen = min(aParts.size(), bParts.size());
        for (size_t i = 0; i < minLen; i++) {
            int cmp = comparePart(aParts[i], bParts[i]);
            if (cmp != 0) return cmp;
        }
        // 前缀相同，内容多者优先
        if (aParts.size() != bParts.size())
            return aParts.size() < bParts.size() ? -1 : 1;
        return 0;
    }

    int compare(const string& a, const string& b) const {
        size_t aPlus = a.find('+'), aMinus = a.find('-');
        size_t bPlus = b.find('+'), bMinus = b.find('-');

        // 提取 core 并比较 Major.Minor.Patch
        string aCore, bCore;
        if (aMinus != string::npos) aCore = a.substr(0, aMinus);
        else if (aPlus != string::npos) aCore = a.substr(0, aPlus);
        else aCore = a;
        if (bMinus != string::npos) bCore = b.substr(0, bMinus);
        else if (bPlus != string::npos) bCore = b.substr(0, bPlus);
        else bCore = b;

        auto aParts = split(aCore), bParts = split(bCore);
        for (int i = 0; i < 3; i++) {
            int va = stoi(aParts[i]), vb = stoi(bParts[i]);
            if (va != vb) return va < vb ? -1 : 1;
        }

        // 提取先行版和构建信息
        string aPre, bPre, aBuild, bBuild;
        if (aMinus != string::npos) {
            if (aPlus != string::npos && aPlus > aMinus) {
                aPre = a.substr(aMinus + 1, aPlus - aMinus - 1);
                aBuild = a.substr(aPlus + 1);
            } else if (aPlus != string::npos) {
                aBuild = a.substr(aPlus + 1);
                aPre = a.substr(aMinus + 1, aPlus - aMinus - 1);
            } else aPre = a.substr(aMinus + 1);
        } else if (aPlus != string::npos) aBuild = a.substr(aPlus + 1);

        if (bMinus != string::npos) {
            if (bPlus != string::npos && bPlus > bMinus) {
                bPre = b.substr(bMinus + 1, bPlus - bMinus - 1);

                bBuild = b.substr(bPlus + 1);
                bPre = b.substr(bMinus + 1, bPlus - bMinus - 1);
            } else bPre = b.substr(bMinus + 1);
        } else if (bPlus != string::npos) bBuild = b.substr(bPlus + 1);

        if (preCmp != 0) return preCmp;
        return compareIdentifiers(aBuild, bBuild);
    }

    for (const p of parts) {
        if (!isPureNumber(p)) return false;
        if (p.length > 1 && p[0] === '0') return false;
        if (parseInt(p, 10) > 2147483647) return false;
    }

    // 提取并检查先行版和构建信息
    let preRelease = "", build = "";
    if (minusPos !== -1) {
        if (plusPos !== -1 && plusPos > minusPos) {
            preRelease = s.substring(minusPos + 1, plusPos);
            build = s.substring(plusPos + 1);
        } else if (plusPos !== -1) {
            build = s.substring(plusPos + 1);
            preRelease = s.substring(minusPos + 1, plusPos);
        } else {
            preRelease = s.substring(minusPos + 1);
        }
    } else if (plusPos !== -1) {
        build = s.substring(plusPos + 1);
    }

    return checkIdentifiers(preRelease) && checkIdentifiers(build);
}

function compare(a, b) {
    const aPlus = a.indexOf('+'), aMinus = a.indexOf('-');
    const bPlus = b.indexOf('+'), bMinus = b.indexOf('-');

    // 提取 core 并比较 Major.Minor.Patch
    let aCore, bCore;
    if (aMinus !== -1) aCore = a.substring(0, aMinus);
    else if (aPlus !== -1) aCore = a.substring(0, aPlus);
    else aCore = a;
    if (bMinus !== -1) bCore = b.substring(0, bMinus);
    else if (bPlus !== -1) bCore = b.substring(0, bPlus);
    else bCore = b;

    const aCoreParts = aCore.split('.'), bCoreParts = bCore.split('.');
    for (let i = 0; i < 3; i++) {
        const va = parseInt(aCoreParts[i], 10), vb = parseInt(bCoreParts[i], 10);
        if (va !== vb) return va < vb ? -1 : 1;
    }

    // 提取先行版和构建信息
    let aPre = "", bPre = "", aBuild = "", bBuild = "";
    if (aMinus !== -1) {
        if (aPlus !== -1 && aPlus > aMinus) {
            aPre = a.substring(aMinus + 1, aPlus);
            aBuild = a.substring(aPlus + 1);
        } else if (aPlus !== -1) {
            aBuild = a.substring(aPlus + 1);
            aPre = a.substring(aMinus + 1, aPlus);
        } else aPre = a.substring(aMinus + 1);
    } else if (aPlus !== -1) aBuild = a.substring(aPlus + 1);

    if (bMinus !== -1) {
        if (bPlus !== -1 && bPlus > bMinus) {
            bPre = b.substring(bMinus + 1, bPlus);
            bBuild = b.substring(bPlus + 1);
        } else if (bPlus !== -1) {
            bBuild = b.substring(bPlus + 1);
            bPre = b.substring(bMinus + 1, bPlus);
        } else bPre = b.substring(bMinus + 1);
    } else if (bPlus !== -1) bBuild = b.substring(bPlus + 1);

    // 比较先行版
    const preCmp = compareIdentifiers(aPre, bPre);
    if (preCmp !== 0) return preCmp;
    // 比较构建信息
    return compareIdentifiers(aBuild, bBuild);
}

// 比较标识符列表
function compareIdentifiers(a, b) {
    if (a.length === 0 && b.length === 0) return 0;
    if (a.length === 0) return -1;  // 无信息 < 有信息
    if (b.length === 0) return 1;

    const aParts = a.split('.'), bParts = b.split('.');
    const minLen = Math.min(aParts.length, bParts.length);
    for (let i = 0; i < minLen; i++) {
        const cmp = comparePart(aParts[i], bParts[i]);
        if (cmp !== 0) return cmp;
    }
    // 前缀相同，内容多者优先
    return aParts.length - bParts.length;
}

// 比较单个标识符段
function comparePart(a, b) {
    const aNum = isPureNumber(a), bNum = isPureNumber(b);
    if (aNum && bNum) {
        const va = parseInt(a, 10), vb = parseInt(b, 10);
        return va - vb;
    }
    if (aNum) return -1;  // 数字 < 非数字
    if (bNum) return 1;
    // ASCII 字典序
    if (a < b) return -1;
    if (a > b) return 1;
    return 0;
}

        if (s[i] < '0' || s[i] > '9') return false;

static bool checkIdentifiers(const char* s) {
    if (!s || !*s) return true;
    int len = (int)strlen(s);
    if (s[0] == '.' || s[len - 1] == '.') return false;
    if (strstr(s, "..")) return false;
    if (strchr(s, '_') || strchr(s, '~')) return false;
    return true;

// 格式合法性检查
static bool isValid(const char* s) {
    if (!s || !*s) return false;
    int len = (int)strlen(s);
    // 禁止字符
    if (strchr(s, '_') || strchr(s, '~')) return false;
    if (s[0] == '-' || s[0] == '+' || s[0] == '.') return false;
    if (s[len - 1] == '-' || s[len - 1] == '+' || s[len - 1] == '.') return false;

    const char* plusPos = strchr(s, '+');
    if (plusPos && strchr(plusPos + 1, '+')) return false;

    const char* minusPos = strchr(s, '-');
    if (minusPos && plusPos && minusPos > plusPos) return false;

    // 提取 Major.Minor.Patch 并检查三段纯数字
    int coreLen;
    if (minusPos) coreLen = (int)(minusPos - s);
    else if (plusPos) coreLen = (int)(plusPos - s);
    else coreLen = len;

    char* core = (char*)malloc(coreLen + 1);
    strncpy(core, s, coreLen); core[coreLen] = '\0';

    // 统计点号并逐段检查
    int dotCount = 0;
    for (int i = 0; core[i]; i++)
        if (core[i] == '.') dotCount++;
    if (dotCount != 2) { free(core); return false; }

    char* saveptr;
    char* tok = strtok_r(core, ".", &saveptr);
    int partCnt = 0;
    while (tok) {
        partCnt++;
        if (!isPureNumber(tok)) { free(core); return false; }
        int tokLen = (int)strlen(tok);
        if (tokLen > 1 && tok[0] == '0') { free(core); return false; }
        if (atoll(tok) > (1LL << 31) - 1) { free(core); return false; }
        tok = strtok_r(NULL, ".", &saveptr);
    }
    free(core);
    if (partCnt != 3) return false;

    // 检查先行版和构建信息
    if (minusPos) {
        if (plusPos && plusPos > minusPos) {
            int preLen = (int)(plusPos - minusPos - 1);
            char* pre = (char*)malloc(preLen + 1);
            strncpy(pre, minusPos + 1, preLen); pre[preLen] = '\0';
            bool ok = checkIdentifiers(pre) && checkIdentifiers(plusPos + 1);
            free(pre);
            if (!ok) return false;
        } else if (plusPos) {
            int preLen = (int)(plusPos - minusPos - 1);
            char* pre = (char*)malloc(preLen + 1);
            strncpy(pre, minusPos + 1, preLen); pre[preLen] = '\0';
            bool ok = checkIdentifiers(pre) && checkIdentifiers(plusPos + 1);
            free(pre);

        }
    } else if (plusPos) {
        if (!checkIdentifiers(plusPos + 1)) return false;
    }
    return true;
}

char* findLatestVersion(char** versions, int versionsSize, int* resultSize) {

            char* empty = (char*)malloc(1);
            empty[0] = '\0';
            return empty;
        }
    }

    // 按规则比较，找出最大的版本号
    int best = 0;
    for (int i = 1; i < versionsSize; i++) {
        if (compare(versions[i], versions[best]) > 0)
            best = i;
    }
    return strdup(versions[best]);
}
