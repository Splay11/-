## 解题思路

本题是**单机顺序调度 + 截止时间约束下的收益最大化**。任务必须全部执行且不可并行，执行顺序会影响每个任务的完成时刻，进而决定能否拿到收益。

**关键观察**：$n \le 20$，可用**状压 DP** 枚举「已完成任务集合」，并记录达成当前最大收益所需的**最小完成时刻**。

状态设计：

- `dp[mask]`：已完成任务集合为 `mask` 时，`(最大总收益, 达成该收益的最小完成时刻)`。
- 转移：固定 `mask` 中**最后一个执行**的任务 `j`，则前序集合为 `prev = mask ^ (1<<j)`。
  - 完成时刻 `finish = dp[prev].time + duration[j]`；
  - 若 `finish <= deadline[j]` 则贡献 `profit[j]`，否则贡献 $0$；
  - 在同等收益下保留更小的完成时刻，便于后续任务更容易按时完成。

**常见假解**：

- **按截止时间升序贪心**：先做长任务可能挤掉「短时长、紧截止、高收益」任务（如先做 $dur=3$ 再做 $dur=1,deadline=2$ 会错过收益 $100$）；
- **按收益降序贪心**：先做长高收益任务可能使多个中等收益任务集体超时；
- **全排列暴力**（$20!$）：规模爆炸；
- **忽略「同收益取更小完成时刻」**：可能漏掉更优后续组合；
- **把问题当成固定区间加权调度**（类似 LeetCode 1235 的大 $n$ 做法）：本题 $n$ 小但顺序影响完成时刻，需状压而非仅按结束时间排序。

## 复杂度分析

设 $n \le 20$。

- 时间：$O(n \cdot 2^n)$，共 $2^n$ 个状态，每个枚举最后一个任务 $O(n)$；
- 空间：$O(2^n)$ 存 DP 数组。

from typing import List, Tuple


class Solution:
    def maximumProfit(
        self, duration: List[int], deadline: List[int], profit: List[int]
    ) -> int:
        n = len(duration)
        # dp[mask] = (最大收益, 达成该收益的最小完成时刻)
        dp: List[Tuple[int, int]] = [(0, 0)] * (1 << n)
        for mask in range(1, 1 << n):
            best_profit, best_time = 0, 0
            for j in range(n):
                if (mask >> j) & 1 == 0:
                    continue
                # 假设任务 j 为当前子集中最后一个执行
                prev_mask = mask ^ (1 << j)
                prev_profit, prev_time = dp[prev_mask]
                finish = prev_time + duration[j]
                gain = profit[j] if finish <= deadline[j] else 0
                cand_profit = prev_profit + gain
                cand_time = finish
                # 收益更大优先；收益相同时保留更小完成时刻
                if cand_profit > best_profit or (
                    cand_profit == best_profit and cand_time < best_time
                ):
                    best_profit, best_time = cand_profit, cand_time
            dp[mask] = (best_profit, best_time)
        return dp[(1 << n) - 1][0]

public class Solution {
    public int maximumProfit(int[] duration, int[] deadline, int[] profit) {
        int n = duration.length;
        int[] dpProfit = new int[1 << n];
        int[] dpTime = new int[1 << n];
        for (int mask = 1; mask < (1 << n); mask++) {
            int bestProfit = 0, bestTime = 0;
            for (int j = 0; j < n; j++) {
                if (((mask >> j) & 1) == 0) continue;
                int prevMask = mask ^ (1 << j);
                int finish = dpTime[prevMask] + duration[j];
                // 超时则该任务收益为 0
                int gain = finish <= deadline[j] ? profit[j] : 0;
                int candProfit = dpProfit[prevMask] + gain;
                int candTime = finish;
                if (candProfit > bestProfit
                        || (candProfit == bestProfit && candTime < bestTime)) {
                    bestProfit = candProfit;
                    bestTime = candTime;
                }
            }
            dpProfit[mask] = bestProfit;
            dpTime[mask] = bestTime;
        }
        return dpProfit[(1 << n) - 1];
    }
}

class Solution {
 public:
  int maximumProfit(vector<int>& duration, vector<int>& deadline,
                    vector<int>& profit) {
    int n = duration.size();
    vector<pair<int, int>> dp(1 << n, {0, 0});
    for (int mask = 1; mask < (1 << n); mask++) {
      int bestProfit = 0, bestTime = 0;
      for (int j = 0; j < n; j++) {
        if (((mask >> j) & 1) == 0) continue;
        int prevMask = mask ^ (1 << j);
        auto [prevProfit, prevTime] = dp[prevMask];
        int finish = prevTime + duration[j];
        int gain = (finish <= deadline[j]) ? profit[j] : 0;
        int candProfit = prevProfit + gain;
        int candTime = finish;
        // 同收益时保留更小完成时刻，利于后续任务
        if (candProfit > bestProfit ||
            (candProfit == bestProfit && candTime < bestTime)) {
          bestProfit = candProfit;
          bestTime = candTime;
        }
      }
      dp[mask] = {bestProfit, bestTime};
    }
    return dp[(1 << n) - 1].first;
  }
};

#include <stdio.h>

#include <string.h>

int maximumProfit(int* duration, int* deadline, int* profit, int n) {
    int total = 1 << n;
    int* dpProfit = (int*)malloc(total * sizeof(int));
    int* dpTime = (int*)malloc(total * sizeof(int));
    memset(dpProfit, 0, total * sizeof(int));
    memset(dpTime, 0, total * sizeof(int));

    for (int mask = 1; mask < total; mask++) {
        int bestProfit = 0, bestTime = 0;
        for (int j = 0; j < n; j++) {
            if (!((mask >> j) & 1)) continue;
            int prevMask = mask ^ (1 << j);
            int finish = dpTime[prevMask] + duration[j];
            // 超过截止时间则该任务收益为 0
            int gain = (finish <= deadline[j]) ? profit[j] : 0;
            int candProfit = dpProfit[prevMask] + gain;
            int candTime = finish;
            // 收益更大优先；收益相同时保留更小完成时刻
            if (candProfit > bestProfit ||
                (candProfit == bestProfit && candTime < bestTime)) {
                bestProfit = candProfit;
                bestTime = candTime;
            }
        }
        dpProfit[mask] = bestProfit;
        dpTime[mask] = bestTime;
    }

    int ans = dpProfit[total - 1];
    free(dpProfit);
    free(dpTime);
    return ans;
}
