## 解题思路

把每个水站看成图中的一个点，把每条管道看成图中的边。

对于管道 $[u, v, type]$：

- 当 $type = 0$ 时，表示水只能从 $u$ 流向 $v$，因此只加入一条有向边 $u \to v$。
- 当 $type = 1$ 时，表示水可以双向流动，因此加入两条有向边 $u \to v$ 和 $v \to u$。

所有源头站一定有水，因此从所有源头站同时出发，沿着水流方向进行搜索。凡是能够被搜索到的水站，都可以获得供水；没有被搜索到的水站，就是孤立站。

最后按编号从小到大扫描 $0$ 到 $n-1$，把未访问过的水站加入答案。因为扫描顺序本身就是升序，所以结果天然有序。

## 复杂度分析

设水站数量为 $n$，管道数量为 $m$。

建图需要遍历所有管道，时间复杂度为 $O(m)$。

搜索过程中，每个水站最多被访问一次，每条有向边最多被遍历一次。双向管道会被拆成两条有向边，因此总边数仍然是 $O(m)$。搜索时间复杂度为 $O(n + m)$。

最终扫描所有水站得到答案，时间复杂度为 $O(n)$。

所以总时间复杂度为 $O(n + m)$。

邻接表、访问数组和搜索队列都需要额外空间，因此空间复杂度为 $O(n + m)$。

```python
class Solution:
    def findIsolatedStations(self, n, sources, pipes):
        # 建立邻接表，graph[x] 表示水可以从 x 直接流向的水站列表
        graph = [[] for _ in range(n)]

        # 根据管道类型建图
        for u, v, typ in pipes:
            if typ == 0:
                # 单向管道：水只能从 u 流向 v
                graph[u].append(v)
            else:
                # 双向管道：水可以在 u 和 v 之间双向流动
                graph[u].append(v)
                graph[v].append(u)

        # visited[i] 表示 i 号水站是否可以从某个源头站获得供水
        visited = [False] * n

        # 使用队列进行广度优先搜索，所有源头站同时入队
        queue = []
        head = 0

        for s in sources:
            # 源头站一定有水，若尚未访问则加入搜索队列
            if 0 <= s < n and not visited[s]:
                visited[s] = True
                queue.append(s)

        # 从所有源头站出发，沿着水流方向扩展
        while head < len(queue):
            cur = queue[head]
            head += 1

            # 当前水站有水，则它能流到的下一批水站也可以获得供水
            for nxt in graph[cur]:
                if not visited[nxt]:
                    visited[nxt] = True
                    queue.append(nxt)

        # 按编号从小到大收集没有供水的水站
        ans = []
        for i in range(n):
            if not visited[i]:
                ans.append(i)

        return ans
```

```java
class Solution {
    public int[] findIsolatedStations(int n, int[] sources, int[][] pipes) {
        // degree[i] 表示 i 号水站向外可以直接供水的管道数量
        int[] degree = new int[n];

        // 第一次遍历管道，统计每个点的出边数量，便于建立紧凑的邻接表
        for (int i = 0; i < pipes.length; i++) {
            int u = pipes[i][0];
            int v = pipes[i][1];
            int typ = pipes[i][2];

            if (typ == 0) {
                // 单向管道只产生一条从 u 到 v 的边
                degree[u]++;
            } else {
                // 双向管道等价于两条方向相反的有向边
                degree[u]++;
                degree[v]++;
            }
        }

        // graph[u] 中保存所有可以从 u 直接到达的水站
        int[][] graph = new int[n][];
        for (int i = 0; i < n; i++) {
            graph[i] = new int[degree[i]];
        }

        // pos[i] 表示当前已经向 graph[i] 中填入了多少个邻接点
        int[] pos = new int[n];

        // 第二次遍历管道，真正填充邻接表
        for (int i = 0; i < pipes.length; i++) {
            int u = pipes[i][0];
            int v = pipes[i][1];
            int typ = pipes[i][2];

            if (typ == 0) {
                // 单向管道：只允许水从 u 流向 v
                graph[u][pos[u]++] = v;
            } else {
                // 双向管道：两个方向都可以流动
                graph[u][pos[u]++] = v;
                graph[v][pos[v]++] = u;
            }
        }

        // visited[i] 表示 i 号水站是否可以获得供水
        boolean[] visited = new boolean[n];

        // 用数组模拟队列，最多只会有 n 个水站入队
        int[] queue = new int[n];
        int head = 0;
        int tail = 0;

        // 所有源头站同时作为搜索起点
        for (int i = 0; i < sources.length; i++) {
            int s = sources[i];
            if (0 <= s && s < n && !visited[s]) {
                visited[s] = true;
                queue[tail++] = s;
            }
        }

        // 广度优先搜索，沿着水流方向寻找所有可供水水站
        while (head < tail) {
            int cur = queue[head++];

            for (int i = 0; i < graph[cur].length; i++) {
                int nxt = graph[cur][i];

                if (!visited[nxt]) {
                    visited[nxt] = true;
                    queue[tail++] = nxt;
                }
            }
        }

        // 先统计孤立站数量，用于创建长度准确的结果数组
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                count++;
            }
        }

        // 按编号从小到大写入答案，因此结果天然有序
        int[] ans = new int[count];
        int idx = 0;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                ans[idx++] = i;
            }
        }

        return ans;
    }
}
```

```cpp
class Solution {
public:
    vector<int> findIsolatedStations(int n, vector<int>& sources, vector<vector<int>>& pipes) {
        // 建立邻接表，graph[u] 保存所有可以由 u 直接供水到达的水站
        vector<vector<int>> graph(n);

        // 根据管道类型加入有向边
        for (vector<int>& pipe : pipes) {
            int u = pipe[0];
            int v = pipe[1];
            int typ = pipe[2];

            if (typ == 0) {
                // 单向管道：水只能从 u 流向 v
                graph[u].push_back(v);
            } else {
                // 双向管道：水可以从 u 到 v，也可以从 v 到 u
                graph[u].push_back(v);
                graph[v].push_back(u);
            }
        }

        // visited[i] 表示 i 号水站是否已经确认可以获得供水
        vector<int> visited(n, 0);

        // 使用数组模拟队列，避免递归导致栈空间风险
        vector<int> queue(n);
        int head = 0;
        int tail = 0;

        // 所有源头站都是搜索起点
        for (int s : sources) {
            if (0 <= s && s < n && !visited[s]) {
                visited[s] = 1;
                queue[tail++] = s;
            }
        }

        // 广度优先搜索：从有水的水站出发，沿着管道方向继续供水
        while (head < tail) {
            int cur = queue[head++];

            for (int nxt : graph[cur]) {
                if (!visited[nxt]) {
                    visited[nxt] = 1;
                    queue[tail++] = nxt;
                }
            }
        }

        // 按编号从小到大收集没有被访问到的水站
        vector<int> ans;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                ans.push_back(i);
            }
        }

        return ans;
    }
};
```

```javascript
var findIsolatedStations = function(n, sources, pipes) {
    // 构建邻接表，graph[i] 表示 i 能直接供水到的水站列表
    const graph = Array.from({ length: n }, () => []);

    for (const [u, v, typ] of pipes) {
        graph[u].push(v);
        if (typ === 1) {
            // 双向管道需要增加反向边
            graph[v].push(u);
        }
    }

    // visited[i] 表示 i 是否已获得供水
    const visited = new Array(n).fill(false);
    const queue = [];
    let head = 0;

    // 先将所有源头站入队
    for (const s of sources) {
        if (s >= 0 && s < n && !visited[s]) {
            visited[s] = true;
            queue.push(s);
        }
    }

    // BFS 搜索所有可达水站
    while (head < queue.length) {
        const cur = queue[head++];
        for (const nxt of graph[cur]) {
            if (!visited[nxt]) {
                visited[nxt] = true;
                queue.push(nxt);
            }
        }
    }

    // 收集没有水的孤立水站
    const ans = [];
    for (let i = 0; i < n; i++) {
        if (!visited[i]) {
            ans.push(i);
        }
    }

    return ans;
};
```

```c
int* findIsolatedStations(int n, int* sources, int sourcesSize, int** pipes, int pipesSize, int* returnSize) {
    // 统计每个水站的出度，用于构建邻接表
    int* degree = (int*)calloc(n, sizeof(int));
    for (int i = 0; i < pipesSize; i++) {
        int u = pipes[i][0], v = pipes[i][1], typ = pipes[i][2];
        degree[u]++;
        if (typ == 1) degree[v]++;
    }

    // 分配邻接表内存
    int** graph = (int**)malloc(n * sizeof(int*));
    for (int i = 0; i < n; i++) {
        graph[i] = (int*)malloc(degree[i] * sizeof(int));
    }

    // 填充邻接表
    int* pos = (int*)calloc(n, sizeof(int));
    for (int i = 0; i < pipesSize; i++) {
        int u = pipes[i][0], v = pipes[i][1], typ = pipes[i][2];
        graph[u][pos[u]++] = v;
        if (typ == 1) {
            graph[v][pos[v]++] = u;
        }
    }
    free(pos);

    // visited[i] 表示是否获得供水
    int* visited = (int*)calloc(n, sizeof(int));
    // 队列模拟 BFS
    int* queue = (int*)malloc(n * sizeof(int));
    int head = 0, tail = 0;

    // 源头站入队
    for (int i = 0; i < sourcesSize; i++) {
        int s = sources[i];
        if (s >= 0 && s < n && !visited[s]) {
            visited[s] = 1;
            queue[tail++] = s;
        }
    }

    // BFS 扩展供水范围
    while (head < tail) {
        int cur = queue[head++];
        for (int i = 0; i < degree[cur]; i++) {
            int nxt = graph[cur][i];
            if (!visited[nxt]) {
                visited[nxt] = 1;
                queue[tail++] = nxt;
            }
        }
    }

    // 统计孤立站数量
    int count = 0;
    for (int i = 0; i < n; i++) {
        if (!visited[i]) count++;
    }

    // 生成答案数组
    int* ans = (int*)malloc(count * sizeof(int));
    *returnSize = count;
    int idx = 0;
    for (int i = 0; i < n; i++) {
        if (!visited[i]) {
            ans[idx++] = i;
        }
    }

    // 释放内存
    for (int i = 0; i < n; i++) free(graph[i]);
    free(graph);
    free(degree);
    free(visited);
    free(queue);

    return ans;
}
```
