## 解题思路

本题在**文件路径前缀树（逻辑上按 `/` 分段）**上做统计：对每个文件，把它的大小累加到「目标目录 `target` 的一级子项」上。

- **目标是否存在**：若存在某个文件路径 `p` 满足 `p == target` 或 `p` 以 `target + '/'` 为前缀，则认为 `target` 在文件系统中出现；否则返回空列表。
- **一级子项划分**：对满足 `p` 以 `target + '/'` 开头的文件，令相对路径 `rel = p[len(target)+1:]`。若 `rel` 中不含 `/`，则该文件本身就是一级子项，累加到 `p`；否则一级子项为 `target + '/' + rel` 的第一段路径前缀。
- **目录大小**：题意规定目录占用为其内部所有文件大小之和；本题输入只给出文件，因此「一级子目录」的大小等于所有路径以该子目录为前缀的文件大小之和，与上面的累加方式一致。
- **答案**：在所有一级子项的聚合大小中取最大值，将所有达到最大值的子项路径按**字典序升序**输出。若目标存在但没有任何子路径文件，则返回空列表。

涉及算法：**哈希表（`unordered_map` / `dict` / `HashMap`）聚合**与**排序**；本质是单次线性扫描的路径解析。

## 复杂度分析

- **时间复杂度**：$O(n \cdot L)$，其中 $n$ 为 `files` 长度，$L$ 为路径平均长度；每次只需在相对路径中查找第一个 `/` 并做常数次哈希更新。题面保证路径长度与深度较小，也可视为 $O(n)$。
- **空间复杂度**：$O(n)$，存储每个一级子项的累计值；答案列表规模不超过一级子项个数，亦为 $O(n)$。

```python
from typing import List


class Solution:
    def findMaxOccupiedPaths(self, target: str, files: List[str], sizes: List[int]) -> List[str]:
        # 核心：判断 target 是否出现在「路径或真子路径前缀」意义下；必须用 target+'/' 判断子树，避免 "/dir" 匹配 "/dir1"
        exists = any(
            p == target or (len(p) > len(target) and p.startswith(target + "/"))
            for p in files
        )
        if not exists:
            return []
        # 核心：一级子项 -> 该子项下所有文件大小之和（目录无单独 size，只聚合其下文件）
        agg = {}
        for p, sz in zip(files, sizes):
            if p == target:
                # 若 target 本身是文件路径，不产生「子项」桶，跳过
                continue
            if not (len(p) > len(target) and p.startswith(target + "/")):
                continue
            # 关键：相对路径第一段即「直接挂在 target 下」的那一层，决定哈希键
            rel = p[len(target) + 1 :]
            slash = rel.find("/")
            if slash < 0:
                child = p  # 文件直接在 target 下，一级子项就是该文件全路径
            else:
                child = target + "/" + rel[:slash]  # 否则归入名为第一段的一级子目录
            agg[child] = agg.get(child, 0) + int(sz)
        if not agg:
            return []
        # 关键：全局最大占用，再筛出所有等于最大的键
        mx = max(agg.values())
        out = [k for k, v in agg.items() if v == mx]
        out.sort()  # 题意：并列时按路径字典序升序
        return out
```

```java
import java.util.*;

class Solution {
    public List<String> findMaxOccupiedPaths(String target, List<String> files, List<Integer> sizes) {
        // 核心：同 Python，用「相等或 target+'/' 真前缀」判断 target 是否在文件系统中出现
        boolean exists = false;
        for (String p : files) {
            if (p.equals(target)
                    || (p.length() > target.length()
                        && p.startsWith(target)
                        && p.charAt(target.length()) == '/')) {
                exists = true;
                break;
            }
        }
        if (!exists) {
            return new ArrayList<>();
        }
        // 核心：每个一级子项路径 -> 其下所有文件 size 之和（Long 防累加溢出）
        Map<String, Long> agg = new HashMap<>();
        for (int i = 0; i < files.size(); i++) {
            String p = files.get(i);
            if (p.equals(target)) {
                continue;
            }
            // 关键：必须同时满足 startsWith 与下一字符为 '/'，避免 "/dir" 与 "/dir1" 串前缀误判
            if (p.length() <= target.length()
                    || !p.startsWith(target)
                    || p.charAt(target.length()) != '/') {
                continue;
            }
            String rel = p.substring(target.length() + 1);
            int slash = rel.indexOf('/');
            String child;
            if (slash < 0) {
                child = p;
            } else {
                child = target + "/" + rel.substring(0, slash);
            }
            long sz = sizes.get(i).longValue();
            agg.merge(child, sz, Long::sum);
        }
        if (agg.isEmpty()) {
            return new ArrayList<>();
        }
        // 关键：先求最大占用，再收集所有等于最大的路径
        long mx = 0;
        for (long v : agg.values()) {
            mx = Math.max(mx, v);
        }
        List<String> ans = new ArrayList<>();
        for (Map.Entry<String, Long> e : agg.entrySet()) {
            if (e.getValue() == mx) {
                ans.add(e.getKey());
            }
        }
        Collections.sort(ans);  // 题意：并列最大时字典序
        return ans;
    }
}
```

```cpp
#include <algorithm>
#include <cstdint>
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;

class Solution {
public:
    vector<string> findMaxOccupiedPaths(string target, vector<string>& files, vector<int>& sizes) {
        // 核心：判断 target 是否作为某条文件路径本身或「target/」严格前缀出现
        bool exists = false;
        for (const string& p : files) {
            if (p == target
                || (p.size() > target.size() && p.compare(0, target.size(), target) == 0
                    && p[target.size()] == '/')) {
                exists = true;
                break;
            }
        }
        if (!exists) {
            return {};
        }
        // 核心：一级子项聚合；int64_t 存和，避免多文件大 size 相加溢出 int
        unordered_map<string, int64_t> agg;
        for (size_t i = 0; i < files.size(); ++i) {
            const string& p = files[i];
            if (p == target) {
                continue;
            }
            // 关键：compare 前缀后必须检查 target 长度处是否为 '/'，区分 /dir 与 /dir1
            if (p.size() <= target.size() || p.compare(0, target.size(), target) != 0
                || p[target.size()] != '/') {
                continue;
            }
            string rel = p.substr(target.size() + 1);
            size_t slash = rel.find('/');
            string child;
            if (slash == string::npos) {
                child = p;
            } else {
                child = target + "/" + rel.substr(0, slash);
            }
            agg[child] += static_cast<int64_t>(sizes[i]);
        }
        if (agg.empty()) {
            return {};
        }
        // 关键：最大占用及所有取到最大的路径
        int64_t mx = 0;
        for (const auto& kv : agg) {
            mx = max(mx, kv.second);
        }
        vector<string> ans;
        for (const auto& kv : agg) {
            if (kv.second == mx) {
                ans.push_back(kv.first);
            }
        }
        sort(ans.begin(), ans.end());  // 题意：并列时字典序升序
        return ans;
    }
};
```

```javascript
var findMaxOccupiedPaths = function(target, files, sizes) {
    // 核心：判断 target 是否在文件系统中出现
    var exists = false;
    var tlen = target.length;
    for (var i = 0; i < files.length; i++) {
        var p = files[i];
        if (p === target ||
            (p.length > tlen && p.charCodeAt(tlen) === 47 /* '/' */ &&
             p.lastIndexOf(target, 0) === 0)) {
            exists = true;
            break;
        }
    }
    if (!exists) return [];

    // 核心：一级子项 -> 该子项下所有文件大小之和
    var agg = Object.create(null);
    for (var i = 0; i < files.length; i++) {
        var p = files[i];
        if (p === target) continue;
        if (p.length <= tlen || p.charCodeAt(tlen) !== 47 /* '/' */ ||
            p.lastIndexOf(target, 0) !== 0) continue;

        var rel = p.substring(tlen + 1);
        var slash = rel.indexOf('/');
        var child;
        if (slash < 0) {
            child = p;
        } else {
            child = p.substring(0, tlen + 1 + slash);
        }
        agg[child] = (agg[child] || 0) + sizes[i];
    }

    var keys = Object.keys(agg);
    if (keys.length === 0) return [];

    // 关键：全局最大占用，再筛出所有等于最大的键
    var mx = 0;
    for (var j = 0; j < keys.length; j++) {
        if (agg[keys[j]] > mx) mx = agg[keys[j]];
    }
    var ans = [];
    for (var k = 0; k < keys.length; k++) {
        if (agg[keys[k]] === mx) ans.push(keys[k]);
    }
    ans.sort();  // 题意：并列时按路径字典序升序
    return ans;
};
```

```c
#include <string.h>
#include <stdlib.h>

// 哈希表条目
typedef struct {
    char* key;
    long long val;
} Entry;

// 简单开地址哈希表
typedef struct {
    Entry* entries;
    int size;
    int cap;
} HashMap;

static unsigned long hash_str(const char* s) {
    unsigned long h = 5381;
    while (*s) h = ((h << 5) + h) + (unsigned char)(*s++);
    return h;
}

static void hm_init(HashMap* hm, int cap) {
    hm->cap = cap;
    hm->size = 0;
    hm->entries = (Entry*)calloc((size_t)cap, sizeof(Entry));
}

static void hm_ensure(HashMap* hm) {
    if (hm->size * 2 < hm->cap) return;
    int newCap = hm->cap * 2;
    Entry* old = hm->entries;
    int oldCap = hm->cap;
    hm->cap = newCap;
    hm->entries = (Entry*)calloc((size_t)newCap, sizeof(Entry));
    hm->size = 0;
    for (int i = 0; i < oldCap; i++) {
        if (old[i].key) {
            unsigned long h = hash_str(old[i].key) % (unsigned long)newCap;
            while (hm->entries[h].key)
                h = (h + 1) % (unsigned long)newCap;
            hm->entries[h] = old[i];
            hm->size++;
        }
    }
    free(old);
}

static void hm_put(HashMap* hm, const char* key, long long val) {
    hm_ensure(hm);
    unsigned long h = hash_str(key) % (unsigned long)hm->cap;
    while (hm->entries[h].key && strcmp(hm->entries[h].key, key) != 0)
        h = (h + 1) % (unsigned long)hm->cap;
    if (hm->entries[h].key) {
        hm->entries[h].val += val;
    } else {
        size_t klen = strlen(key);
        hm->entries[h].key = (char*)malloc(klen + 1);
        if (hm->entries[h].key) {
            memcpy(hm->entries[h].key, key, klen + 1);
        }
        hm->entries[h].val = val;
        hm->size++;
    }
}

static int cmp_str(const void* a, const void* b) {
    return strcmp(*(const char**)a, *(const char**)b);
}

char** findMaxOccupiedPaths(const char* target,
                            char** files, int filesSize,
                            int* sizes, int sizesSize,
                            int* resultCount) {
    // 判断 target 是否在文件系统中存在
    int exists = 0;
    size_t tlen = strlen(target);
    for (int i = 0; i < filesSize && !exists; i++) {
        size_t plen = strlen(files[i]);
        if (strcmp(files[i], target) == 0) {
            exists = 1;
        } else if (plen > tlen &&
                   strncmp(files[i], target, tlen) == 0 &&
                   files[i][tlen] == '/') {
            exists = 1;
        }
    }
    if (!exists) {
        *resultCount = 0;
        return NULL;
    }

    HashMap agg;
    hm_init(&agg, 64);

    char child_buf[2048];

    for (int i = 0; i < filesSize; i++) {
        const char* p = files[i];
        size_t plen = strlen(p);

        // 跳过 target 本身
        if (strcmp(p, target) == 0) continue;

        // 必须以 target+'/' 为前缀
        if (plen <= tlen || strncmp(p, target, tlen) != 0 || p[tlen] != '/')
            continue;

        const char* rel = p + tlen + 1;
        const char* slash = strchr(rel, '/');

        if (!slash) {
            // 直接挂在 target 下的一级文件
            hm_put(&agg, p, sizes[i]);
        } else {
            // 一级子目录：target + "/" + 第一段路径名
            size_t seg = (size_t)(slash - rel);
            int pos = 0;
            memcpy(child_buf + pos, target, tlen);
            pos += (int)tlen;
            child_buf[pos++] = '/';
            memcpy(child_buf + pos, rel, seg);
            pos += (int)seg;
            child_buf[pos] = '\0';
            hm_put(&agg, child_buf, sizes[i]);
        }
    }

    if (agg.size == 0) {
        free(agg.entries);
        *resultCount = 0;
        return NULL;
    }

    // 找最大占用
    long long mx = 0;
    for (int i = 0; i < agg.cap; i++) {
        if (agg.entries[i].key && agg.entries[i].val > mx)
            mx = agg.entries[i].val;
    }

    // 收集所有达到最大的路径
    int ans_cap = 8, cnt = 0;
    char** ans = (char**)malloc((size_t)ans_cap * sizeof(char*));
    for (int i = 0; i < agg.cap; i++) {
        if (agg.entries[i].key && agg.entries[i].val == mx) {
            if (cnt >= ans_cap) {
                ans_cap *= 2;
                ans = (char**)realloc(ans, (size_t)ans_cap * sizeof(char*));
            }
            ans[cnt++] = agg.entries[i].key;
        } else if (agg.entries[i].key) {
            free(agg.entries[i].key);
        }
    }
    free(agg.entries);

    qsort(ans, (size_t)cnt, sizeof(char*), cmp_str);
    *resultCount = cnt;
    return ans;
}
```
