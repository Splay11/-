## 解题思路

本题可以使用哈希表进行统计。

由于班级中可能存在重名学生，需要先按照题目约定生成合法候选人姓名：

* 第一次出现的姓名保持原名；
* 第二次出现的同名学生编号为 `姓名1`；
* 第三次出现为 `姓名2`；
* 以此类推。

例如：

```text
["Zhangsan", "Lisi", "Wangwu", "Zhangsan"]
```

对应合法姓名为：

```text
Zhangsan, Lisi, Wangwu, Zhangsan1
```

然后遍历投票数据：

* 如果投票人数大于全班人数，直接返回 `"Invalid election."`；
* 如果选票中的姓名不在合法候选人集合中，则视为废票；
* 否则用哈希表统计该候选人的票数。

最后遍历所有合法候选人，找到票数最多的人：

* 票数更高则更新答案；
* 票数相同，则选择字典序更小的姓名；
* 如果没有任何有效票，则选举失败。

本题核心算法是哈希表计数。

## 复杂度分析

设学生人数为 `n`，投票数为 `m`。

时间复杂度：`O(n + m)`

* 构造合法候选人姓名需要遍历学生数组，复杂度为 `O(n)`；
* 统计投票需要遍历投票数组，复杂度为 `O(m)`；
* 最后遍历候选人找答案，复杂度为 `O(n)`。

空间复杂度：`O(n)`

* 需要哈希表保存姓名出现次数、合法候选人集合和票数统计。

```python
class Solution:
    def electMonitor(self, students: list[str], votes: list[str]) -> str:
        n = len(students)

        # 如果票数多于全班人数，选举无效
        if len(votes) > n:
            return "Invalid election."

        name_count = {}
        candidates = []

        # 生成符合重名规则的合法候选人姓名
        for name in students:
            cnt = name_count.get(name, 0)
            if cnt == 0:
                real_name = name
            else:
                real_name = name + str(cnt)
            name_count[name] = cnt + 1
            candidates.append(real_name)

        candidate_set = set(candidates)
        vote_count = {}

        # 统计有效选票
        for vote in votes:
            if vote in candidate_set:
                vote_count[vote] = vote_count.get(vote, 0) + 1

        # 没有有效票，选举失败
        if not vote_count:
            return "Invalid election."

        ans = ""
        max_votes = -1

        # 找出票数最多的人，票数相同取字典序更小
        for name in candidates:
            cnt = vote_count.get(name, 0)
            if cnt > max_votes or (cnt == max_votes and name < ans):
                max_votes = cnt
                ans = name

        return ans
```

```java
import java.util.*;

class Solution {
    public String electMonitor(String[] students, String[] votes) {
        int n = students.length;

        // 如果票数多于全班人数，选举无效
        if (votes.length > n) {
            return "Invalid election.";
        }

        Map<String, Integer> nameCount = new HashMap<>();
        List<String> candidates = new ArrayList<>();

        // 按重名规则生成合法候选人姓名
        for (String name : students) {
            int cnt = nameCount.getOrDefault(name, 0);
            String realName;

            if (cnt == 0) {
                realName = name;
            } else {
                realName = name + cnt;
            }

            nameCount.put(name, cnt + 1);
            candidates.add(realName);
        }

        Set<String> candidateSet = new HashSet<>(candidates);
        Map<String, Integer> voteCount = new HashMap<>();

        // 统计有效选票
        for (String vote : votes) {
            if (candidateSet.contains(vote)) {
                voteCount.put(vote, voteCount.getOrDefault(vote, 0) + 1);
            }
        }

        // 没有任何有效票，选举失败
        if (voteCount.isEmpty()) {
            return "Invalid election.";
        }

        String ans = "";
        int maxVotes = -1;

        // 找出票数最多的人，票数相同取字典序更小
        for (String name : candidates) {
            int cnt = voteCount.getOrDefault(name, 0);

            if (cnt > maxVotes || (cnt == maxVotes && (ans.equals("") || name.compareTo(ans) < 0))) {
                maxVotes = cnt;
                ans = name;
            }
        }

        return ans;
    }
}
```

```cpp
using namespace std;

class Solution {
public:
    string electMonitor(vector<string>& students, vector<string>& votes) {
        int n = students.size();

        // 如果票数多于全班人数，选举无效
        if (votes.size() > n) {
            return "Invalid election.";
        }

        unordered_map<string, int> nameCount;
        vector<string> candidates;

        // 生成符合重名规则的合法候选人姓名
        for (string name : students) {
            int cnt = nameCount[name];
            string realName;

            if (cnt == 0) {
                realName = name;
            } else {
                realName = name + to_string(cnt);
            }

            nameCount[name]++;
            candidates.push_back(realName);
        }

        unordered_set<string> candidateSet(candidates.begin(), candidates.end());
        unordered_map<string, int> voteCount;

        // 统计有效选票
        for (string vote : votes) {
            if (candidateSet.count(vote)) {
                voteCount[vote]++;
            }
        }

        // 没有有效票，选举失败
        if (voteCount.empty()) {
            return "Invalid election.";
        }

        string ans = "";
        int maxVotes = -1;

        // 找出票数最多的人，票数相同取字典序更小
        for (string name : candidates) {
            int cnt = voteCount[name];

            if (cnt > maxVotes || (cnt == maxVotes && name < ans)) {
                maxVotes = cnt;
                ans = name;
            }
        }

        return ans;
    }
};
```

```javascript
/**
 * @param {string[]} students
 * @param {string[]} votes
 * @return {string}
 */
var electMonitor = function(students, votes) {
    var n = students.length;

    // 票数多于全班人数，选举无效
    if (votes.length > n) return "Invalid election.";

    // 生成符合重名规则的合法候选人姓名
    var nameCount = {};
    var candidates = [];
    for (var i = 0; i < n; i++) {
        var name = students[i];
        var cnt = nameCount[name] || 0;
        var realName;
        if (cnt === 0) {
            realName = name;
        } else {
            realName = name + cnt;
        }
        nameCount[name] = cnt + 1;
        candidates.push(realName);
    }

    // 构建候选人集合
    var candidateSet = {};
    for (var i = 0; i < candidates.length; i++) {
        candidateSet[candidates[i]] = true;
    }

    // 统计有效选票
    var voteCount = {};
    for (var i = 0; i < votes.length; i++) {
        var vote = votes[i];
        if (candidateSet[vote]) {
            voteCount[vote] = (voteCount[vote] || 0) + 1;
        }
    }

    // 没有有效票，选举失败
    if (Object.keys(voteCount).length === 0) return "Invalid election.";

    // 找出票数最多的人，票数相同取字典序更小
    var ans = "";
    var maxVotes = -1;
    for (var i = 0; i < candidates.length; i++) {
        var cnt = voteCount[candidates[i]] || 0;
        if (cnt > maxVotes || (cnt === maxVotes && candidates[i] < ans)) {
            maxVotes = cnt;
            ans = candidates[i];
        }
    }

    return ans;
};
```

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct {
    const char* name;
    int cnt;
} NC;

typedef struct {
    char* name;
    int cnt;
} VC;

char* electMonitor(char** students, int studentsSize, char** votes, int votesSize) {
    // 票数多于全班人数，选举无效
    if (votesSize > studentsSize) return "Invalid election.";

    // 动态分配候选人姓名统计表
    int ncCap = 256, ncSize = 0;
    NC* nameCount = (NC*)malloc(ncCap * sizeof(NC));
    int candCap = 256, candSize = 0;
    char** cands = (char**)malloc(candCap * sizeof(char*));

    // 生成符合重名规则的合法候选人姓名
    for (int i = 0; i < studentsSize; i++) {
        char* name = students[i];
        int idx = -1;
        for (int j = 0; j < ncSize; j++) {
            if (strcmp(nameCount[j].name, name) == 0) { idx = j; break; }
        }
        char tmp[256];
        if (idx == -1) {
            strcpy(tmp, name);
            if (ncSize >= ncCap) { ncCap *= 2; nameCount = (NC*)realloc(nameCount, ncCap * sizeof(NC)); }
            nameCount[ncSize].name = name;
            nameCount[ncSize].cnt = 1;
            ncSize++;
        } else {
            sprintf(tmp, "%s%d", name, nameCount[idx].cnt);
            nameCount[idx].cnt++;
        }
        if (candSize >= candCap) { candCap *= 2; cands = (char**)realloc(cands, candCap * sizeof(char*)); }
        cands[candSize] = (char*)malloc(strlen(tmp) + 1);
        strcpy(cands[candSize], tmp);
        candSize++;
    }

    // 统计有效选票
    int vcCap = 256, vcSize = 0;
    VC* voteCount = (VC*)malloc(vcCap * sizeof(VC));
    for (int i = 0; i < votesSize; i++) {
        char* vote = votes[i];
        int valid = 0;
        for (int j = 0; j < candSize; j++) {
            if (strcmp(cands[j], vote) == 0) { valid = 1; break; }
        }
        if (!valid) continue;
        int idx = -1;
        for (int j = 0; j < vcSize; j++) {
            if (strcmp(voteCount[j].name, vote) == 0) { idx = j; break; }
        }
        if (idx == -1) {
            if (vcSize >= vcCap) { vcCap *= 2; voteCount = (VC*)realloc(voteCount, vcCap * sizeof(VC)); }
            voteCount[vcSize].name = vote;
            voteCount[vcSize].cnt = 1;
            vcSize++;
        } else {
            voteCount[idx].cnt++;
        }
    }

    if (vcSize == 0) {
        for (int i = 0; i < candSize; i++) free(cands[i]);
        free(cands); free(nameCount); free(voteCount);
        return "Invalid election.";
    }

    // 找出票数最多的人，票数相同取字典序更小
    char* ans = "";
    int maxVotes = -1;
    for (int i = 0; i < candSize; i++) {
        int cnt = 0;
        for (int j = 0; j < vcSize; j++) {
            if (strcmp(voteCount[j].name, cands[i]) == 0) { cnt = voteCount[j].cnt; break; }
        }
        if (cnt > maxVotes || (cnt == maxVotes && (ans[0] == '\0' || strcmp(cands[i], ans) < 0))) {
            maxVotes = cnt;
            ans = cands[i];
        }
    }

    char* result = (char*)malloc(strlen(ans) + 1);
    strcpy(result, ans);
    for (int i = 0; i < candSize; i++) free(cands[i]);
    free(cands); free(nameCount); free(voteCount);
    return result;
}
```
