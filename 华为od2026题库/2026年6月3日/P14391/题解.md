## 解题思路

本题要求每次 `query target` 时，统计**以当前最新到达的流水为右端点**，且区间和等于 `target` 的连续区间个数。

使用 **前缀和 + 哈希表**。

设当前所有已添加数据的前缀和为 `sum`。

如果某个区间 `[l, r]` 的和为 `target`，且 `r` 是当前最新元素，那么：

```text
prefix[r] - prefix[l - 1] = target
```

也就是：

```text
prefix[l - 1] = prefix[r] - target
```

所以每次查询时，只需要统计之前出现过多少个前缀和等于：

```text
sum - target
```

关键点：

当执行 `add x` 时，新的元素会成为最新右端点。因此，在更新当前前缀和之前，要先把“当前前缀和”加入哈希表，因为它可以作为新区间左端点前一位的前缀和。

流程如下：

```text
初始 sum = 0，哈希表 cnt 为空

add x:
    cnt[sum] += 1
    sum += x

query target:
    返回 cnt[sum - target]
```

这样可以避免把当前完整前缀和也算进去，从而避免错误统计空区间。

## 复杂度分析

设 `n = ops.length`。

时间复杂度：`O(n)`

每个操作只进行一次哈希表查询或更新，平均时间复杂度为 `O(1)`。

空间复杂度：`O(n)`

最坏情况下，每次 `add` 都产生不同的前缀和，需要存入哈希表。

```python
from typing import List
from collections import defaultdict

class Solution:
    def monitor(self, ops: List[str], vals: List[int]) -> List[int]:
        cnt = defaultdict(int)  # 记录可作为区间左端点前一位的前缀和出现次数
        prefix_sum = 0          # 当前所有 add 数据的前缀和
        ans = []

        for op, val in zip(ops, vals):
            if op == "add":
                # 当前 prefix_sum 可以作为新区间左端点前一位的前缀和
                cnt[prefix_sum] += 1
                # 加入新的流水数据
                prefix_sum += val
            else:
                target = val
                # 需要找到 prefix[l - 1] = prefix_sum - target
                ans.append(cnt[prefix_sum - target])

        return ans
```

```java
class Solution {
    public List<Integer> monitor(String[] ops, int[] vals) {
        Map<Long, Integer> cnt = new HashMap<>(); // 记录前缀和出现次数
        long prefixSum = 0;                       // 当前前缀和

                ans.add(cnt.getOrDefault(prefixSum - target, 0));
            }
        }

        return ans;
    }
}
```

```c
/**
 * 实时流处理监控
 * 算法：前缀和 + 哈希表
 * 时间复杂度 O(n)，空间复杂度 O(n)
 */

#define HASH_SIZE 200003

typedef struct Node {
    long long key;
    int val;
    struct Node* next;
} Node;

static Node* hash_table[HASH_SIZE];

static void hash_inc(long long key) {
    unsigned int h = (unsigned int)((key % HASH_SIZE + HASH_SIZE) % HASH_SIZE);
    Node* p = hash_table[h];
    while (p) {
        if (p->key == key) { p->val++; return; }
        p = p->next;
    }
    Node* n = (Node*)malloc(sizeof(Node));
    n->key = key; n->val = 1; n->next = hash_table[h]; hash_table[h] = n;
}

static int hash_get(long long key) {
    unsigned int h = (unsigned int)((key % HASH_SIZE + HASH_SIZE) % HASH_SIZE);
    Node* p = hash_table[h];
    while (p) { if (p->key == key) return p->val; p = p->next; }
    return 0;
}

int* monitor(char** ops, int opsSize, int* vals, int valsSize, int* returnSize) {
    // 初始化哈希表
    memset(hash_table, 0, sizeof(hash_table));

    // 统计查询次数
    int qcnt = 0;
    for (int i = 0; i < opsSize; i++)
        if (strcmp(ops[i], "query") == 0) qcnt++;
    *returnSize = qcnt;
    if (qcnt == 0) return NULL;
    int* ans = (int*)malloc(qcnt * sizeof(int));
    int ans_idx = 0;

    long long prefixSum = 0;

    for (int i = 0; i < opsSize; i++) {
        if (strcmp(ops[i], "add") == 0) {
            // 当前 prefixSum 可作为新区间左端点前一位的前缀和
            hash_inc(prefixSum);
            // 加入新的流水数据
            prefixSum += vals[i];
        } else {
            long long target = vals[i];
            // 需要找 prefix[l-1] = prefixSum - target
            ans[ans_idx++] = hash_get(prefixSum - target);
        }
    }

    // 释放哈希表内存
    for (int i = 0; i < HASH_SIZE; i++) {
        Node* p = hash_table[i];
        while (p) { Node* t = p; p = p->next; free(t); }
    }

    return ans;
}
```
