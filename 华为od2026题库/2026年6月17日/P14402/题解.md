## 解题思路

本题是**自定义排序 + 模拟数组操作**，可按以下步骤处理：

1. **popcount 排序键**：对 $32$ 位有符号整数，统计二进制中 `1` 的个数（含符号位），记为 `popcount32(x)`；排序键为 `(popcount32(x), x)` 升序。Python 可用 `bin(x & 0xFFFFFFFF).count('1')`，Java 用 `Integer.bitCount`，C++ 对 `unsigned` 做 `__builtin_popcount`。
2. **初始排序**：进入操作循环前先对 `data` 排序。
3. **单次操作**：读取排序后数组下标 `i, j`：
   - 计算 `merged = arr[i] | arr[j]`（`i == j` 时即 `arr[i] | arr[i]`）；
   - 删除：`i == j` 时 `pop(i)` 一次；否则先删较大下标再删较小下标，避免下标错位；
   - `append(merged)` 后重新按步骤 $1$ 排序。
4. **空操作**：`operations` 为空时，输出排序后的 `data` 即可。

**常见假解**：

- 用无符号或绝对值 popcount，负数排序错 → 样例 3 失败；
- `i == j` 时删两个元素 → 数组长度异常；
- `i != j` 时先删小下标 → 大下标移位导致删错元素；
- 每次操作后忘记重排 → 样例 2 逐步状态全错；
- popcount 相同时未按数值升序 → 合并前取错元素。

## 复杂度分析

设 $n = |data| \le 100$，$m = |operations| \le 10$。

- 单次排序 $O(n \log n)$，每次操作后数组长度至多 $n + m$，总长度 $O(n + m)$；
- 时间：$O(m \cdot n \log n)$，在约束下足够；
- 空间：$O(n + m)$ 存数组副本。

```python
from typing import List


def popcount32(x: int) -> int:
    # 按 32 位有符号整数统计 1 的个数
    return bin(x & 0xFFFFFFFF).count("1")


class Solution:
    def processDataArray(self, data: List[int], operations: List[List[int]]) -> List[int]:
        arr = list(data)

        def sort_arr() -> None:
            arr.sort(key=lambda v: (popcount32(v), v))

        sort_arr()
        for op in operations:
            i, j = op[0], op[1]
            a = arr[i]
            b = arr[j] if i != j else a
            merged = a | b
            if i == j:
                # 同下标只删除一个元素
                arr.pop(i)
            else:
                lo, hi = min(i, j), max(i, j)
                arr.pop(hi)
                arr.pop(lo)
            arr.append(merged)
            sort_arr()
        return arr
```

```java
import java.util.Arrays;

public class Solution {
    private static int popcount32(int x) {
        return Integer.bitCount(x);
    }

    public int[] processDataArray(int[] data, int[][] operations) {
        int[] arr = Arrays.copyOf(data, data.length);
        sortArr(arr);
        for (int[] op : operations) {
            int i = op[0], j = op[1];
            int a = arr[i];
            int b = (i == j) ? a : arr[j];
            int merged = a | b;
            if (i == j) {
                arr = removeAt(arr, i);
            } else {
                int lo = Math.min(i, j), hi = Math.max(i, j);
                arr = removeAt(arr, hi);
                arr = removeAt(arr, lo);
            }
            arr = append(arr, merged);
            sortArr(arr);
        }
        return arr;
    }

    private static void sortArr(int[] arr) {
        Integer[] boxed = new Integer[arr.length];
        for (int k = 0; k < arr.length; k++) boxed[k] = arr[k];
        Arrays.sort(boxed, (u, v) -> {
            int cu = popcount32(u), cv = popcount32(v);
            if (cu != cv) return cu - cv;
            return Integer.compare(u, v);
        });
        for (int k = 0; k < arr.length; k++) arr[k] = boxed[k];
    }

    private static int[] removeAt(int[] arr, int idx) {
        int[] res = new int[arr.length - 1];
        for (int k = 0, p = 0; k < arr.length; k++) {
            if (k != idx) res[p++] = arr[k];
        }
        return res;
    }

    private static int[] append(int[] arr, int val) {
        int[] res = Arrays.copyOf(arr, arr.length + 1);
        res[arr.length] = val;
        return res;
    }
}
```

```cpp
#include <algorithm>
#include <vector>

using namespace std;

static int popcount32(int x) {
  return __builtin_popcount(static_cast<unsigned>(x));
}

class Solution {
 public:
  vector<int> processDataArray(vector<int>& data, vector<vector<int>>& operations) {
    vector<int> arr = data;
    auto sortArr = [&]() {
      sort(arr.begin(), arr.end(), [](int a, int b) {
        int ca = popcount32(a), cb = popcount32(b);
        if (ca != cb) return ca < cb;
        return a < b;
      });
    };
    sortArr();
    for (auto& op : operations) {
      int i = op[0], j = op[1];
      int a = arr[i];
      int b = (i == j) ? a : arr[j];
      int merged = a | b;
      if (i == j) {
        arr.erase(arr.begin() + i);
      } else {
        int lo = min(i, j), hi = max(i, j);
        arr.erase(arr.begin() + hi);
        arr.erase(arr.begin() + lo);
      }
      arr.push_back(merged);
      sortArr();
    }
    return arr;
  }
};
```

```javascript
var processDataArray = function(data, operations) {
    // 32位popcount
    function popcount32(x) {
        var u = x >>> 0; // 转为无符号32位
        var cnt = 0;
        while (u) { cnt++; u &= u - 1; }
        return cnt;
    }

    // 按(popcount, 数值)升序排序
    function sortArr(arr) {
        arr.sort(function(u, v) {
            var cu = popcount32(u), cv = popcount32(v);
            if (cu !== cv) return cu - cv;
            return u - v;
        });
    }

    var arr = data.slice();
    sortArr(arr);

    for (var k = 0; k < operations.length; k++) {
        var i = operations[k][0], j = operations[k][1];
        var a = arr[i];
        var b = (i === j) ? a : arr[j];
        var merged = a | b;

        if (i === j) {
            arr.splice(i, 1);
        } else {
            var lo = Math.min(i, j), hi = Math.max(i, j);
            arr.splice(hi, 1);
            arr.splice(lo, 1);
        }
        arr.push(merged);
        sortArr(arr);
    }

    return arr;
};
```

```c
#include <stdlib.h>

// 32位popcount：统计二进制中1的个数
static int popcount32(int x) {
    unsigned u = (unsigned)x;
    int cnt = 0;
    while (u) { cnt++; u &= u - 1; }
    return cnt;
}

// 比较函数：先按popcount升序，再按数值升序（用比较避免int减法溢出）
static int cmp(const void* a, const void* b) {
    int x = *(int*)a, y = *(int*)b;
    int cx = popcount32(x), cy = popcount32(y);
    if (cx != cy) return cx - cy;
    if (x < y) return -1;
    if (x > y) return 1;
    return 0;
}

// 从数组中删除指定下标
static void removeAt(int* arr, int* size, int idx) {
    for (int i = idx; i < *size - 1; i++) arr[i] = arr[i + 1];
    (*size)--;
}

int* processDataArray(int* data, int dataSize, int** operations, int operationsSize, int* operationsColSize, int* returnSize) {
    int cap = dataSize + operationsSize + 1;
    int* arr = (int*)malloc(cap * sizeof(int));
    int size = dataSize;
    for (int i = 0; i < size; i++) arr[i] = data[i];

    // 初始排序
    qsort(arr, size, sizeof(int), cmp);

    for (int op = 0; op < operationsSize; op++) {
        int i = operations[op][0];
        int j = operations[op][1];
        int a = arr[i];
        int b = (i == j) ? a : arr[j];
        int merged = a | b;

        if (i == j) {
            removeAt(arr, &size, i);
        } else {
            int lo = i < j ? i : j;
            int hi = i > j ? i : j;
            removeAt(arr, &size, hi);
            removeAt(arr, &size, lo);
        }
        arr[size++] = merged;
        qsort(arr, size, sizeof(int), cmp);
    }

    *returnSize = size;
    int* result = (int*)malloc(size * sizeof(int));
    for (int i = 0; i < size; i++) result[i] = arr[i];
    free(arr);
    return result;
}
```
