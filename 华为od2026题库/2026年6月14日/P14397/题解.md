## 解题思路

本题是**字符分类提取 + 模拟拼接**，可按以下步骤处理：

1. **扫描分类**：从左到右遍历 $S$，将数字按出现顺序放入数组 $D$，字母放入数组 $L$。
2. **特判**：若 $D$ 或 $L$ 为空（只含数字或只含字母），直接返回原串 $S$。
3. **交替展开**：对 $i=0,1,\ldots,\min(|D|,|L|)-1$：
   - 先追加 $D[i]$；
   - 令 $n$ 为 $D[i]$ 的数值；若 $n>0$，将 $L[i]$ 重复 $n$ 次追加（$n=0$ 时不输出字母）。
4. **拼接余量**：若 $|D|>|L|$，将 $D$ 从下标 $|L|$ 起的后缀直接追加；若 $|L|>|D|$，同理追加 $L$ 的余下部分。

**常见假解**：

- 按原串位置交替而非「提取后的 $D$/$L$」交替 → 样例 2 会错；
- 数字为 $0$ 仍输出一次字母 → `A0` 应得 `0` 而非 `0A`；
- 忘记拼接较长一侧的余量 → `ABC0123` 会丢末尾 `3`；
- 字母重复次数用 $n+1$ 或固定重复 $1$ 次。

## 复杂度分析

设 $n=|S|\le 100$，输出长度在最坏情况下约为 $O(n^2)$（大量数字 `9` 与高重复），但 $n$ 很小。

- 时间：$O(n + |output|)$，一次扫描加拼接；
- 空间：$O(n + |output|)$ 存分类结果与答案。

```python
class Solution:
    def processString(self, s: str) -> str:
        digits = [c for c in s if c.isdigit()]
        letters = [c for c in s if c.isalpha()]
        # 只含字母或只含数字：原样返回
        if not digits or not letters:
            return s
        parts = []
        n = min(len(digits), len(letters))
        for i in range(n):
            d = digits[i]
            parts.append(d)
            cnt = ord(d) - ord("0")
            # 数字为 0 时不输出对应字母
            if cnt > 0:
                parts.append(letters[i] * cnt)
        # 较长一侧的余量直接拼接
        if len(digits) > len(letters):
            parts.extend(digits[n:])
        elif len(letters) > len(digits):
            parts.extend(letters[n:])
        return "".join(parts)
```

```java
public class Solution {
    public String processString(String s) {
        StringBuilder digits = new StringBuilder();
        StringBuilder letters = new StringBuilder();
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c >= '0' && c <= '9') digits.append(c);
            else if (c >= 'A' && c <= 'Z') letters.append(c);
        }
        // 只含一类字符时直接返回原串
        if (digits.length() == 0 || letters.length() == 0) return s;
        StringBuilder res = new StringBuilder();
        int n = Math.min(digits.length(), letters.length());
        for (int i = 0; i < n; i++) {
            char d = digits.charAt(i);
            res.append(d);
            int cnt = d - '0';
            // 数字为 0 时不输出对应字母
            if (cnt > 0) {
                for (int k = 0; k < cnt; k++) res.append(letters.charAt(i));
            }
        }
        // 拼接较长一侧的余量
        if (digits.length() > n) res.append(digits.substring(n));
        else if (letters.length() > n) res.append(letters.substring(n));
        return res.toString();
    }
}
```

```cpp
#include <algorithm>
#include <string>

using namespace std;

class Solution {
 public:
  string processString(string s) {
    string digits, letters;
    for (char c : s) {
      if (c >= '0' && c <= '9') digits += c;
      else if (c >= 'A' && c <= 'Z') letters += c;
    }
    // 只含一类字符时直接返回原串
    if (digits.empty() || letters.empty()) return s;
    string res;
    int n = min((int)digits.size(), (int)letters.size());
    for (int i = 0; i < n; i++) {
      res += digits[i];
      int cnt = digits[i] - '0';
      // 数字为 0 表示该位置字母不输出
      if (cnt > 0) res += string(cnt, letters[i]);
    }
    // 拼接较长一侧的余量
    if ((int)digits.size() > n) res += digits.substr(n);
    else if ((int)letters.size() > n) res += letters.substr(n);
    return res;
  }
};
```

```javascript
var processString = function(s) {
    // 分离数字和字母
    var digits = [];
    var letters = [];
    for (var i = 0; i < s.length; i++) {
        var c = s[i];
        if (c >= '0' && c <= '9') {
            digits.push(c);
        } else if (c >= 'A' && c <= 'Z') {
            letters.push(c);
        }
    }
    // 只含一类字符时直接返回原串
    if (digits.length === 0 || letters.length === 0) {
        return s;
    }
    var parts = [];
    var n = Math.min(digits.length, letters.length);
    // 交替输出
    for (var i = 0; i < n; i++) {
        parts.push(digits[i]); // 先输出数字
        var cnt = parseInt(digits[i]);
        // 数字为 0 时不输出对应字母
        if (cnt > 0) {
            for (var k = 0; k < cnt; k++) {
                parts.push(letters[i]);
            }
        }
    }
    // 拼接较长一侧的余量
    if (digits.length > n) {
        for (var i = n; i < digits.length; i++) {
            parts.push(digits[i]);
        }
    } else if (letters.length > n) {
        for (var i = n; i < letters.length; i++) {
            parts.push(letters[i]);
        }
    }
    return parts.join('');
};
```

```c
char* processString(char* s) {
    int len = strlen(s);
    // 分离数字和字母
    char* digits = (char*)malloc(len + 1);
    char* letters = (char*)malloc(len + 1);
    int d_cnt = 0, l_cnt = 0;
    for (int i = 0; i < len; i++) {
        if (s[i] >= '0' && s[i] <= '9') {
            digits[d_cnt++] = s[i];
        } else if (s[i] >= 'A' && s[i] <= 'Z') {
            letters[l_cnt++] = s[i];
        }
    }
    digits[d_cnt] = '\0';
    letters[l_cnt] = '\0';

    // 只含一类字符时直接返回原串
    if (d_cnt == 0 || l_cnt == 0) {
        free(digits);
        free(letters);
        char* result = (char*)malloc(len + 1);
        strcpy(result, s);
        return result;
    }

    // 计算输出需要的空间
    int max_out = 0;
    int n = d_cnt < l_cnt ? d_cnt : l_cnt;
    for (int i = 0; i < n; i++) {
        max_out++; // 数字本身
        int cnt = digits[i] - '0';
        if (cnt > 0) max_out += cnt; // 重复字母
    }
    // 较长一侧的余量
    if (d_cnt > l_cnt) max_out += d_cnt - n;
    else if (l_cnt > d_cnt) max_out += l_cnt - n;

    char* result = (char*)malloc(max_out + 1);
    int pos = 0;

    // 交替输出：先数字，再按数字次数重复字母
    for (int i = 0; i < n; i++) {
        result[pos++] = digits[i];
        int cnt = digits[i] - '0';
        // 数字为 0 时不输出对应字母
        if (cnt > 0) {
            for (int k = 0; k < cnt; k++) {
                result[pos++] = letters[i];
            }
        }
    }
    // 拼接较长一侧的余量
    if (d_cnt > n) {
        strcpy(result + pos, digits + n);
        pos += d_cnt - n;
    } else if (l_cnt > n) {
        strcpy(result + pos, letters + n);
        pos += l_cnt - n;
    }
    result[pos] = '\0';

    free(digits);
    free(letters);
    return result;
}
```
