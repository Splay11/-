## 解题思路

本题是**进制转换 + 自定义排序**，可按以下步骤处理：

1. **进制转换**：对每个 $nums[i]$，若其为 $0$ 则直接得到 `"0"`；否则不断对 $base$ 取余得到数位，映射为 `0-9a-f` 后逆序拼接。
2. **排序键**：转换后的字符串在 $base$ 进制下的十进制值，等于原整数 $nums[i]$ 本身，因此可直接按 $nums[i]$ **降序**排序下标。
3. **输出**：按排序后的下标，依次输出对应的 $base$ 进制字符串。

**常见假解**：

- 按字符串**字典序**降序而非按数值 → 如 base $16$ 时 `"64"`（$100$）字典序大于 `"ff"`（$255$），会排错；
- 按**升序**排列 → 与题意相反；
- $0$ 未特判或输出空串 → 应输出 `"0"`；
- 字母用大写 `A-F` → 题面要求小写；
- 把字符串再解析成十进制比较时用了错误的 $base$。

## 复杂度分析

设 $n = |nums| \le 100$，单次数值 $\le 10^6$，进制 $\le 16$，转换长度 $L = O(\log_{base} nums[i]) \le 20$。

- 时间：$O(n \cdot L + n \log n)$，转换加排序；
- 空间：$O(n \cdot L)$ 存转换结果与答案。

```python
class Solution:
    def sortConvertedNums(self, nums: List[int], base: int) -> List[str]:
        def to_base(n: int) -> str:
            if n == 0:
                return "0"
            digits = "0123456789abcdef"
            parts = []
            while n:
                parts.append(digits[n % base])
                n //= base
            return "".join(reversed(parts))

        pairs = [(x, to_base(x)) for x in nums]
        # 按原整数降序，等价于按 base 进制字符串的十进制值降序
        pairs.sort(key=lambda p: p[0], reverse=True)
        return [s for _, s in pairs]
```

```java
public class Solution {
    public String[] sortConvertedNums(int[] nums, int base) {
        int n = nums.length;
        String[] converted = new String[n];
        for (int i = 0; i < n; i++) {
            converted[i] = toBase(nums[i], base);
        }
        int[] order = new int[n];
        for (int i = 0; i < n; i++) order[i] = i;
        // 选择排序下标：按原整数降序，等价于按进制串十进制值降序
        for (int i = 0; i < n; i++) {
            int best = i;
            for (int j = i + 1; j < n; j++) {
                if (nums[order[j]] > nums[order[best]]) best = j;
            }
            int tmp = order[i];
            order[i] = order[best];
            order[best] = tmp;
        }
        String[] ans = new String[n];
        for (int i = 0; i < n; i++) ans[i] = converted[order[i]];
        return ans;
    }

    private String toBase(int n, int base) {
        if (n == 0) return "0";
        String digits = "0123456789abcdef";
        StringBuilder sb = new StringBuilder();
        while (n > 0) {
            sb.append(digits.charAt(n % base));
            n /= base;
        }
        return sb.reverse().toString();
    }
}
```

```cpp
#include <algorithm>
#include <string>
#include <vector>

using namespace std;

class Solution {
 public:
  vector<string> sortConvertedNums(vector<int>& nums, int base) {
    int n = (int)nums.size();
    vector<string> converted(n);
    for (int i = 0; i < n; i++) converted[i] = toBase(nums[i], base);
    vector<int> order(n);
    for (int i = 0; i < n; i++) order[i] = i;
    // 按下标排序：nums 降序等价于进制串十进制值降序
    sort(order.begin(), order.end(), [&](int a, int b) {
      return nums[a] > nums[b];
    });
    vector<string> ans(n);
    for (int i = 0; i < n; i++) ans[i] = converted[order[i]];
    return ans;
  }

 private:
  string toBase(int n, int base) {
    if (n == 0) return "0";
    const string digits = "0123456789abcdef";
    string s;
    while (n > 0) {
      s.push_back(digits[n % base]);
      n /= base;
    }
    reverse(s.begin(), s.end());
    return s;
  }
};
```

```javascript
var sortConvertedNums = function(nums, base) {
    // 进制字符映射
    var digits = "0123456789abcdef";
    // 将数字转换为 base 进制字符串
    function toBase(n) {
        if (n === 0) return "0";
        var parts = [];
        while (n > 0) {
            parts.push(digits[n % base]);
            n = Math.floor(n / base);
        }
        return parts.reverse().join('');
    }
    // 转换每个数字
    var converted = nums.map(function(x) { return toBase(x); });
    // 按原整数降序排列下标
    var order = [];
    for (var i = 0; i < nums.length; i++) order.push(i);
    order.sort(function(a, b) { return nums[b] - nums[a]; });
    // 按排序后的下标收集结果
    var result = [];
    for (var i = 0; i < order.length; i++) {
        result.push(converted[order[i]]);
    }
    return result;
};
```

```c
char** sortConvertedNums(int* nums, int numsSize, int base, int* returnSize) {
    *returnSize = numsSize;
    // 字符映射表
    const char* digits = "0123456789abcdef";

    // 转换为 base 进制字符串
    char** converted = (char**)malloc(numsSize * sizeof(char*));
    for (int i = 0; i < numsSize; i++) {
        int n = nums[i];
        if (n == 0) {
            converted[i] = (char*)malloc(2);
            converted[i][0] = '0';
            converted[i][1] = '\0';
            continue;
        }
        char buf[32];
        int pos = 0;
        while (n > 0) {
            buf[pos++] = digits[n % base];
            n /= base;
        }
        // 逆序拼回
        converted[i] = (char*)malloc(pos + 1);
        for (int j = 0; j < pos; j++) {
            converted[i][j] = buf[pos - 1 - j];
        }
        converted[i][pos] = '\0';
    }

    // 按原整数降序排列下标
    int* order = (int*)malloc(numsSize * sizeof(int));
    for (int i = 0; i < numsSize; i++) order[i] = i;
    // 选择排序
    for (int i = 0; i < numsSize; i++) {
        int best = i;
        for (int j = i + 1; j < numsSize; j++) {
            if (nums[order[j]] > nums[order[best]]) best = j;
        }
        int tmp = order[i];
        order[i] = order[best];
        order[best] = tmp;
    }

    // 按排序后的下标收集结果
    char** result = (char**)malloc(numsSize * sizeof(char*));
    for (int i = 0; i < numsSize; i++) {
        result[i] = converted[order[i]];
    }
    free(converted);
    free(order);
    return result;
}
```
