## 解题思路

本题要求将数字字符串从一个字符集（源进制）表示转换为另一个字符集（目标进制）表示。

**核心算法：反复除法（Repeated Division）**

设源进制基数为 $S = \text{len}(sourceDigits)$，目标进制基数为 $T = \text{len}(targetDigits)$。

方法一（Python / Java BigInteger / JavaScript BigInt）：先将源进制字符串转为十进制大整数，再通过反复对 $T$ 取余数转换为目标进制。

方法二（C / C++ 手写大数）：直接在原进制下模拟"除以 $T$"的过程，收集余数。由于数值可能非常大（长度可达 100 位，在二进制下约 $2^{100}$），C/C++ 需要用数组/字符串模拟大数除法。

**反复除法的步骤：**

1. 将输入数字的每个字符映射为数值 $0 \dots S-1$，存入数组 `digits`。
2. 去掉前导零（但若全为零则保留一个 $0$）。
3. 循环：
   - 从左到右遍历 `digits`，用进位变量 `carry` 记录"当前余数"；
   - 对每个数位 $d$：`cur = carry * S + d`，商位 `q = cur / T`，更新 `carry = cur % T`；
   - 跳过商的前导零，将剩余商位作为下一轮的被除数；
   - 本轮最终 `carry` 即为目标进制的一个数位（对应 `targetDigits[carry]`），插入结果头部。
4. 循环直到 `digits` 为空，返回结果字符串。

**特殊情况：** 输入为 $0$ 时应返回 `targetDigits[0]`（即结果不能含前导零，除非数值本身就是零）。

## 复杂度分析

- **时间复杂度**：$O(n \cdot \log_S V)$，其中 $n \le 100$ 为输入长度，$V$ 为输入数值的大小。每次除法的代价与当前被除数长度成正比，被除数长度随余数收集逐步缩短。实际运行中 $n$ 较小，完全可行。
- **空间复杂度**：$O(n)$，需要存储被除数的数值数组和结果字符串。

```python
class Solution:
    def convertNumber(self, num: str, sourceDigits: str, targetDigits: str) -> str:
        src_base = len(sourceDigits)     # 源进制基数
        tgt_base = len(targetDigits)     # 目标进制基数

        # 构建字符 → 数值的映射表
        char_to_val = {c: i for i, c in enumerate(sourceDigits)}

        # 将源进制字符串转为十进制大整数（Python 自动支持大整数）
        value = 0
        for c in num:
            value = value * src_base + char_to_val[c]

        # 处理零的特殊情况
        if value == 0:
            return targetDigits[0]

        # 十进制 → 目标进制：反复除以 T，收集余数
        result = []
        while value > 0:
            result.append(targetDigits[value % tgt_base])
            value //= tgt_base

        # 余数逆序即为目标进制表示
        return ''.join(reversed(result))
```

```java
class Solution {
    public String convertNumber(String num, String sourceDigits, String targetDigits) {
        int srcBase = sourceDigits.length();   // 源进制基数
        int tgtBase = targetDigits.length();   // 目标进制基数

        // 构建字符 → 数值的映射
        int[] valMap = new int[128];
        for (int i = 0; i < srcBase; i++)
            valMap[sourceDigits.charAt(i)] = i;

        // 将输入串每位映射为数值，存入链表（方便删除前导零）
        java.util.LinkedList<Integer> digits = new java.util.LinkedList<>();
        for (int i = 0; i < num.length(); i++)
            digits.add(valMap[num.charAt(i)]);

        // 去掉前导零（若全为零则保留一位 0）
        while (digits.size() > 1 && digits.getFirst() == 0)
            digits.removeFirst();

        // 反复除以目标进制，收集余数
        StringBuilder result = new StringBuilder();
        while (!digits.isEmpty()) {
            int carry = 0;                          // 进位/余数
            java.util.LinkedList<Integer> next = new java.util.LinkedList<>();
            for (int d : digits) {
                carry = carry * srcBase + d;
                int q = carry / tgtBase;            // 当前商位
                if (!next.isEmpty() || q != 0)      // 跳过商的前导零
                    next.add(q);
                carry %= tgtBase;                   // 更新余数
            }
            result.insert(0, targetDigits.charAt(carry)); // 余数插入结果头部
            digits = next;
        }
        return result.toString();
    }
}
```

```cpp
#include <string>
#include <vector>
using namespace std;

class Solution {
public:
    string convertNumber(string num, string sourceDigits, string targetDigits) {
        int srcBase = (int)sourceDigits.size();   // 源进制基数
        int tgtBase = (int)targetDigits.size();   // 目标进制基数

        // 构建字符 → 数值的映射
        int valMap[256] = {};
        for (int i = 0; i < srcBase; i++)
            valMap[(unsigned char)sourceDigits[i]] = i;

        // 将输入串每位映射为数值
        vector<int> digits;
        for (char c : num)
            digits.push_back(valMap[(unsigned char)c]);

        // 去掉前导零（若全为零则保留一位 0）
        int start = 0;
        while (start + 1 < (int)digits.size() && digits[start] == 0)
            start++;
        if (start > 0)
            digits.erase(digits.begin(), digits.begin() + start);

        // 反复除以目标进制，收集余数
        string result;
        while (!digits.empty()) {
            int carry = 0;                       // 进位/余数
            vector<int> next;
            for (int d : digits) {
                carry = carry * srcBase + d;
                int q = carry / tgtBase;         // 当前商位
                if (!next.empty() || q != 0)     // 跳过商的前导零
                    next.push_back(q);
                carry %= tgtBase;                // 更新余数
            }
            result = targetDigits[carry] + result; // 余数插入结果头部
            digits = move(next);
        }
        return result;
    }
};
```

```javascript
var convertNumber = function(num, sourceDigits, targetDigits) {
    const srcBase = sourceDigits.length;   // 源进制基数
    const tgtBase = targetDigits.length;   // 目标进制基数

    // 构建字符 → 数值的映射
    const valMap = {};
    for (let i = 0; i < srcBase; i++)
        valMap[sourceDigits[i]] = i;

    // 将输入串每位映射为数值
    const digits = [];
    for (const c of num)
        digits.push(valMap[c]);

    // 去掉前导零（若全为零则保留一位 0）
    while (digits.length > 1 && digits[0] === 0)
        digits.shift();

    // 反复除以目标进制，收集余数（大数除法）
    let result = '';
    while (digits.length > 0) {
        let carry = 0;                         // 进位/余数
        const next = [];
        for (const d of digits) {
            carry = carry * srcBase + d;
            const q = Math.floor(carry / tgtBase); // 当前商位
            if (next.length > 0 || q !== 0)        // 跳过商的前导零
                next.push(q);
            carry %= tgtBase;                      // 更新余数
        }
        result = targetDigits[carry] + result;     // 余数插入结果头部
        digits = next;
    }
    return result;
};
```

```c
#include <stdlib.h>
#include <string.h>

char* convertNumber(char* num, char* sourceDigits, char* targetDigits) {
    int srcBase = (int)strlen(sourceDigits);   // 源进制基数
    int tgtBase = (int)strlen(targetDigits);   // 目标进制基数

    // 构建字符 → 数值的映射
    int valMap[256] = {0};
    for (int i = 0; i < srcBase; i++)
        valMap[(unsigned char)sourceDigits[i]] = i;

    // 将输入串每位映射为数值数组
    int n = (int)strlen(num);
    int* digits = (int*)malloc(n * sizeof(int));
    int dLen = n;
    for (int i = 0; i < n; i++)
        digits[i] = valMap[(unsigned char)num[i]];

    // 去掉前导零（若全为零则保留一位 0）
    int start = 0;
    while (start < dLen - 1 && digits[start] == 0)
        start++;
    if (start > 0) {
        memmove(digits, digits + start, (dLen - start) * sizeof(int));
        dLen -= start;
    }

    // 结果缓冲区（保守估计长度）
    int maxResult = dLen * 8 + 2;
    char* result = (char*)malloc(maxResult);
    int rLen = 0;

    // 反复除以目标进制，收集余数
    while (dLen > 0) {
        int carry = 0;                              // 进位/余数
        int newLen = 0;
        for (int i = 0; i < dLen; i++) {
            carry = carry * srcBase + digits[i];
            int q = carry / tgtBase;                // 当前商位
            if (newLen > 0 || q != 0)               // 跳过商的前导零
                digits[newLen++] = q;
            carry %= tgtBase;                       // 更新余数
        }
        // 余数插入结果头部
        memmove(result + 1, result, rLen);
        result[0] = targetDigits[carry];
        rLen++;
        dLen = newLen;
    }
    result[rLen] = '\0';

    free(digits);
    return result;  // 调用者需自行释放 result
}
```
