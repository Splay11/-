## 解题思路

本题要求将数组重构，使偶数下标存放升序偶数、奇数下标存放升序奇数。由于题目保证了数组中偶数和奇数各占一半，且输出数组唯一确定，可以直接采用「分离 + 排序 + 交替填充」的策略。

**算法步骤：**

1. 遍历原数组，将偶数元素放入 `evens` 列表，奇数元素放入 `odds` 列表。
2. 分别对 `evens` 和 `odds` 进行升序排序。
3. 构建结果数组，下标 $i$ 为偶数时填入 `evens[i/2]`，下标 $i$ 为奇数时填入 `odds[i/2]`。

**正确性说明：** 由于题目保证恰好各一半，排序后按奇偶下标交替填入自然满足所有要求。唯一性由「排序后顺序固定」保证。

**常见错误：**
- 试图在原数组上原地交换而不排序——奇数/偶数内部也需要有序。
- 使用固定大小的计数排序时没注意值域（$0 \sim 1000$，可以使用，但不是必须的优化）。

## 复杂度分析

- **时间复杂度：** $O(n \log n)$，其中 $n$ 是数组长度。分离操作 $O(n)$，排序 $O(n \log n)$，构建结果 $O(n)$。
- **空间复杂度：** $O(n)$，需要两个额外数组分别存放偶数和奇数，以及结果数组。

由于值域仅为 $0 \sim 1000$，也可以使用计数排序将时间优化到 $O(n + M)$（$M$ 为值域大小），但 $O(n \log n)$ 已满足 $n \le 20000$ 的约束。

```python
from typing import List

class Solution:
    def sortArrayByParity(self, nums: List[int]) -> List[int]:
        # 分离偶数和奇数
        evens = [x for x in nums if x % 2 == 0]
        odds = [x for x in nums if x % 2 == 1]
        # 分别升序排序
        evens.sort()
        odds.sort()
        # 构建结果：偶数下标放升序偶数，奇数下标放升序奇数
        n = len(nums)
        res = [0] * n
        for i in range(n):
            if i % 2 == 0:
                res[i] = evens[i // 2]  # 偶数下标，取第 i/2 个偶数
            else:
                res[i] = odds[i // 2]   # 奇数下标，取第 i/2 个奇数
        return res
```

```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

class Solution {
    public int[] sortArrayByParity(int[] nums) {
        int n = nums.length;
        // 分离偶数和奇数
        List<Integer> evens = new ArrayList<>();
        List<Integer> odds = new ArrayList<>();
        for (int x : nums) {
            if (x % 2 == 0) evens.add(x);
            else odds.add(x);
        }
        // 分别升序排序
        Collections.sort(evens);
        Collections.sort(odds);
        // 构建结果：偶数下标放升序偶数，奇数下标放升序奇数
        int[] res = new int[n];
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                res[i] = evens.get(i / 2);  // 偶数下标，取第 i/2 个偶数
            } else {
                res[i] = odds.get(i / 2);   // 奇数下标，取第 i/2 个奇数
            }
        }
        return res;
    }
}
```

```c
#include <stdlib.h>

// qsort 比较函数，实现升序排序
static int cmpAsc(const void* a, const void* b) {
    int va = *(const int*)a;
    int vb = *(const int*)b;
    // 三向比较，避免整数溢出
    if (va < vb) return -1;
    if (va > vb) return 1;
    return 0;
}

/**
 * @param nums 输入数组
 * @param numsSize 数组长度
 * @param returnSize 返回数组的长度（由函数设置）
 * @return 重构后的数组（调用者需 free）
 */
int* sortArrayByParity(int* nums, int numsSize, int* returnSize) {
    int n = numsSize;
    *returnSize = n;

    // 收集偶数和奇数
    int* evens = (int*)malloc(n * sizeof(int));
    int* odds = (int*)malloc(n * sizeof(int));
    int eCnt = 0, oCnt = 0;
    for (int i = 0; i < n; i++) {
        if (nums[i] % 2 == 0) {
            evens[eCnt++] = nums[i];  // 收集偶数
        } else {
            odds[oCnt++] = nums[i];   // 收集奇数
        }
    }

    // 分别升序排序
    qsort(evens, eCnt, sizeof(int), cmpAsc);
    qsort(odds, oCnt, sizeof(int), cmpAsc);

    // 构建结果：偶数下标放升序偶数，奇数下标放升序奇数
    int* res = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        if (i % 2 == 0) {
            res[i] = evens[i / 2];  // 偶数下标，取第 i/2 个偶数
        } else {
            res[i] = odds[i / 2];   // 奇数下标，取第 i/2 个奇数
        }
    }

    // 释放临时数组
    free(evens);
    free(odds);
    return res;
}
```
