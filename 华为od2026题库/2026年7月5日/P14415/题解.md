## 解题思路

本题要求根据单循环足球联赛的比赛结果，计算积分和净胜球排名，输出前三名。核心考察**哈希表统计 + 排序 + 合法性校验**。

### 算法框架

1. **合法性校验**：
   - `teamNum` 不在 $[3, 15]$ 范围内 → 返回空列表
   - 从 `matches` 收集实际参赛队伍，去重后数量不等于 `teamNum` → 返回空列表
   - 比赛场次超过 $\frac{teamNum \times (teamNum - 1)}{2}$ → 返回空列表
   - 存在两队之间没有比赛记录 → 返回空列表

2. **积分与净胜球统计**：
   - 遍历所有比赛，对每场比赛 `[队伍A, 进球A, 进球B, 队伍B]`：
     - 胜方得 3 分，平局双方各得 1 分
     - 更新进球数和失球数，用于计算净胜球
   - 判断是否存在两支队伍的**积分**和**净胜球**完全相同 → 返回空列表

3. **排序输出**：
   - 按积分降序、净胜球降序排序
   - 返回前 3 名队伍名称

### 关键点

- 由于 $teamNum \le 15$，比赛数 $M \le 105$，直接 $O(N^2)$ 校验对阵关系和并列即可通过
- 并列判定必须**同时满足**积分相同和净胜球相同，仅积分相同而净胜球不同是合法的

## 复杂度分析

- **时间复杂度**：$O(M + N^2)$，其中 $M$ 为比赛场数，$N$ 为队伍数（$N \le 15$）。遍历 `matches` 统计积分和净胜球需 $O(M)$，检查每对队伍的对阵关系和并列需 $O(N^2)$。
- **空间复杂度**：$O(N)$，哈希表存储每支队伍的积分、进球数和失球数。

```python
from typing import List

class Solution:
    def getTopThree(self, teamNum: int, matches: List[List[str]]) -> List[str]:
        # 1. 检查 teamNum 合法性
        if teamNum < 3 or teamNum > 15:
            return []

        # 2. 收集所有队伍并去重，数量必须等于 teamNum
        teams_set = set()
        for m in matches:
            teams_set.add(m[0])
            teams_set.add(m[3])
        teams = list(teams_set)
        if len(teams) != teamNum:
            return []

        # 3. 初始化统计数据
        points = {t: 0 for t in teams}
        goals_for = {t: 0 for t in teams}
        goals_against = {t: 0 for t in teams}
        played = set()  # 记录已比赛的对阵

        for m in matches:
            a, ga, gb, b = m[0], int(m[1]), int(m[2]), m[3]
            # 记录已比赛的对阵（排序后存入集合避免重复）
            played.add(tuple(sorted([a, b])))

            # 累加进球与失球
            goals_for[a] += ga
            goals_against[a] += gb
            goals_for[b] += gb
            goals_against[b] += ga

            # 计分规则：胜 3 分，平 1 分，负 0 分
            if ga > gb:
                points[a] += 3
            elif ga < gb:
                points[b] += 3
            else:
                points[a] += 1
                points[b] += 1

        # 4. 检查比赛场次是否超限
        max_matches = teamNum * (teamNum - 1) // 2
        if len(matches) > max_matches:
            return []

        # 5. 检查每对队伍是否都有比赛记录
        n = len(teams)
        for i in range(n):
            for j in range(i + 1, n):
                if tuple(sorted([teams[i], teams[j]])) not in played:
                    return []  # 有队伍对之间缺少比赛

        # 6. 检查是否存在积分和净胜球完全相同的队伍
        for i in range(n):
            for j in range(i + 1, n):
                gd_i = goals_for[teams[i]] - goals_against[teams[i]]
                gd_j = goals_for[teams[j]] - goals_against[teams[j]]
                if points[teams[i]] == points[teams[j]] and gd_i == gd_j:
                    return []  # 并列且不可区分

        # 7. 排序：积分降序 → 净胜球降序，取前 3 名
        teams.sort(key=lambda t: (-points[t], -(goals_for[t] - goals_against[t])))
        return teams[:3]
```

```java
import java.util.*;

class Solution {
    public List<String> getTopThree(int teamNum, List<List<String>> matches) {
        // 1. 检查 teamNum 合法性
        if (teamNum < 3 || teamNum > 15) {
            return new ArrayList<>();
        }

        // 2. 收集所有队伍并去重
        Set<String> teamSet = new HashSet<>();
        for (List<String> m : matches) {
            teamSet.add(m.get(0));
            teamSet.add(m.get(3));
        }
        List<String> teams = new ArrayList<>(teamSet);
        if (teams.size() != teamNum) {
            return new ArrayList<>();
        }

        // 3. 初始化统计数据
        Map<String, Integer> points = new HashMap<>();
        Map<String, Integer> goalsFor = new HashMap<>();
        Map<String, Integer> goalsAgainst = new HashMap<>();
        for (String t : teams) {
            points.put(t, 0);
            goalsFor.put(t, 0);
            goalsAgainst.put(t, 0);
        }
        Set<String> played = new HashSet<>();

        for (List<String> m : matches) {
            String a = m.get(0), b = m.get(3);
            int ga = Integer.parseInt(m.get(1));
            int gb = Integer.parseInt(m.get(2));

            // 记录已比赛的对阵
            String key = a.compareTo(b) < 0 ? a + "|" + b : b + "|" + a;
            played.add(key);

            // 累加进球与失球
            goalsFor.put(a, goalsFor.get(a) + ga);
            goalsAgainst.put(a, goalsAgainst.get(a) + gb);
            goalsFor.put(b, goalsFor.get(b) + gb);
            goalsAgainst.put(b, goalsAgainst.get(b) + ga);

            // 计分规则
            if (ga > gb) {
                points.put(a, points.get(a) + 3);
            } else if (ga < gb) {
                points.put(b, points.get(b) + 3);
            } else {
                points.put(a, points.get(a) + 1);
                points.put(b, points.get(b) + 1);
            }
        }

        // 4. 检查比赛场次
        int maxMatches = teamNum * (teamNum - 1) / 2;
        if (matches.size() > maxMatches) {
            return new ArrayList<>();
        }

        // 5. 检查每对队伍是否都有比赛
        int n = teams.size();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                String key = teams.get(i).compareTo(teams.get(j)) < 0
                    ? teams.get(i) + "|" + teams.get(j)
                    : teams.get(j) + "|" + teams.get(i);
                if (!played.contains(key)) {
                    return new ArrayList<>();
                }
            }
        }

        // 6. 检查积分和净胜球并列
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int gdI = goalsFor.get(teams.get(i)) - goalsAgainst.get(teams.get(i));
                int gdJ = goalsFor.get(teams.get(j)) - goalsAgainst.get(teams.get(j));
                if (points.get(teams.get(i)).equals(points.get(teams.get(j))) && gdI == gdJ) {
                    return new ArrayList<>();
                }
            }
        }

        // 7. 排序取前三
        teams.sort((a, b) -> {
            int cmp = points.get(b).compareTo(points.get(a));
            if (cmp != 0) return cmp;
            int gdA = goalsFor.get(a) - goalsAgainst.get(a);
            int gdB = goalsFor.get(b) - goalsAgainst.get(b);
            return Integer.compare(gdB, gdA);
        });

        return new ArrayList<>(teams.subList(0, Math.min(3, teams.size())));
    }
}
```

```cpp
#include <vector>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <algorithm>
using namespace std;

class Solution {
public:
    vector<string> getTopThree(int teamNum, vector<vector<string>>& matches) {
        // 1. 检查 teamNum 合法性
        if (teamNum < 3 || teamNum > 15) return {};

        // 2. 收集队伍并验证数量
        unordered_set<string> teamSet;
        for (auto& m : matches) {
            teamSet.insert(m[0]);
            teamSet.insert(m[3]);
        }
        vector<string> teams(teamSet.begin(), teamSet.end());
        if ((int)teams.size() != teamNum) return {};

        // 3. 初始化统计
        unordered_map<string, int> points, goalsFor, goalsAgainst;
        for (auto& t : teams) {
            points[t] = 0;
            goalsFor[t] = 0;
            goalsAgainst[t] = 0;
        }
        unordered_set<string> played;

        for (auto& m : matches) {
            string a = m[0], b = m[3];
            int ga = stoi(m[1]), gb = stoi(m[2]);

            // 记录对阵（字典序小的在前）
            string key = a < b ? a + "|" + b : b + "|" + a;
            played.insert(key);

            // 累加进球失球
            goalsFor[a] += ga;
            goalsAgainst[a] += gb;
            goalsFor[b] += gb;
            goalsAgainst[b] += ga;

            // 计分
            if (ga > gb) points[a] += 3;
            else if (ga < gb) points[b] += 3;
            else { points[a]++; points[b]++; }
        }

        // 4. 检查比赛场次
        int maxMatches = teamNum * (teamNum - 1) / 2;
        if ((int)matches.size() > maxMatches) return {};

        // 5. 检查每对队伍是否都有比赛
        int n = (int)teams.size();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                string key = teams[i] < teams[j]
                    ? teams[i] + "|" + teams[j]
                    : teams[j] + "|" + teams[i];
                if (played.find(key) == played.end()) return {};
            }
        }

        // 6. 检查积分+净胜球并列
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int gdI = goalsFor[teams[i]] - goalsAgainst[teams[i]];
                int gdJ = goalsFor[teams[j]] - goalsAgainst[teams[j]];
                if (points[teams[i]] == points[teams[j]] && gdI == gdJ) return {};
            }
        }

        // 7. 排序取前三
        sort(teams.begin(), teams.end(), [&](const string& a, const string& b) {
            if (points[a] != points[b]) return points[a] > points[b];
            int gdA = goalsFor[a] - goalsAgainst[a];
            int gdB = goalsFor[b] - goalsAgainst[b];
            return gdA > gdB;
        });
        return {teams.begin(), teams.begin() + min(3, (int)teams.size())};
    }
};
```

```javascript
/**
 * @param {number} teamNum
 * @param {string[][]} matches
 * @return {string[]}

var getTopThree = function(teamNum, matches) {
    // 1. 检查 teamNum 合法性
    if (teamNum < 3 || teamNum > 15) return [];

    // 2. 收集所有队伍并去重
    const teamSet = new Set();
    for (const m of matches) {
        teamSet.add(m[0]);
        teamSet.add(m[3]);
    }
    const teams = [...teamSet];
    if (teams.length !== teamNum) return [];

    // 3. 初始化统计
    const points = {};
    const goalsFor = {};
    const goalsAgainst = {};
    for (const t of teams) {
        points[t] = 0;
        goalsFor[t] = 0;
        goalsAgainst[t] = 0;
    }
    const played = new Set();

    for (const m of matches) {
        const a = m[0], b = m[3];
        const ga = parseInt(m[1]), gb = parseInt(m[2]);

        // 记录已比赛的对阵
        const key = a < b ? a + '|' + b : b + '|' + a;
        played.add(key);

        // 累加进球与失球
        goalsFor[a] += ga;
        goalsAgainst[a] += gb;
        goalsFor[b] += gb;
        goalsAgainst[b] += ga;

        // 计分
        if (ga > gb) points[a] += 3;
        else if (ga < gb) points[b] += 3;
        else { points[a]++; points[b]++; }
    }

    // 4. 检查比赛场次
    if (matches.length > teamNum * (teamNum - 1) / 2) return [];

    // 5. 检查每对队伍是否都有比赛
    const n = teams.length;
    for (let i = 0; i < n; i++) {
        for (let j = i + 1; j < n; j++) {
            const key = teams[i] < teams[j]
                ? teams[i] + '|' + teams[j]
                : teams[j] + '|' + teams[i];
            if (!played.has(key)) return [];
        }
    }

    // 6. 检查积分+净胜球并列
    for (let i = 0; i < n; i++) {
        for (let j = i + 1; j < n; j++) {
            const gdI = goalsFor[teams[i]] - goalsAgainst[teams[i]];
            const gdJ = goalsFor[teams[j]] - goalsAgainst[teams[j]];
            if (points[teams[i]] === points[teams[j]] && gdI === gdJ) return [];
        }
    }

    // 7. 排序取前三
    teams.sort((a, b) => {
        if (points[b] !== points[a]) return points[b] - points[a];
        const gdA = goalsFor[a] - goalsAgainst[a];
        const gdB = goalsFor[b] - goalsAgainst[b];
        return gdB - gdA;
    });
    return teams.slice(0, Math.min(3, teams.length));
};
```

```c
#include <string.h>
#include <stdlib.h>

char** getTopThree(int teamNum, char*** matches, int matchesSize, int* returnSize) {
    // 返回空结果的处理函数
    void returnEmpty() {
        *returnSize = 0;
        // 返回空结果：分配一个以 NULL 结尾的空指针
    }

    // 1. 检查 teamNum 合法性
    if (teamNum < 3 || teamNum > 15) {
        *returnSize = 0;
        return NULL;
    }

    // 2. 收集队伍（去重）
    char* teams[16];
    int teamCount = 0;
    for (int i = 0; i < matchesSize; i++) {
        char** m = matches[i];
        for (int k = 0; k < 2; k++) {
            char* name = k == 0 ? m[0] : m[3];
            int found = 0;
            for (int j = 0; j < teamCount; j++) {
                if (strcmp(teams[j], name) == 0) { found = 1; break; }
            }
            if (!found) teams[teamCount++] = name;
        }
    }
    if (teamCount != teamNum) {
        *returnSize = 0;
        return NULL;
    }

    // 3. 统计积分、进球、失球
    int points[16] = {0};
    int goalsFor[16] = {0};
    int goalsAgainst[16] = {0};
    int played[256] = {0};  // 位图标记已比赛的对阵

    for (int k = 0; k < matchesSize; k++) {
        char** m = matches[k];
        char *a = m[0], *b = m[3];
        int ga = atoi(m[1]), gb = atoi(m[2]);

        // 找到队伍下标
        int ia = -1, ib = -1;
        for (int i = 0; i < teamCount; i++) {
            if (strcmp(teams[i], a) == 0) ia = i;
            if (strcmp(teams[i], b) == 0) ib = i;
        }

        // 标记对阵
        int pi = ia < ib ? ia : ib;
        int pj = ia < ib ? ib : ia;
        played[pi * 16 + pj] = 1;

        // 累加
        goalsFor[ia] += ga; goalsAgainst[ia] += gb;
        goalsFor[ib] += gb; goalsAgainst[ib] += ga;

        // 计分
        if (ga > gb) points[ia] += 3;
        else if (ga < gb) points[ib] += 3;
        else { points[ia]++; points[ib]++; }
    }

    // 4. 检查比赛场次
    int maxMatches = teamNum * (teamNum - 1) / 2;
    if (matchesSize > maxMatches) {
        *returnSize = 0;
        return NULL;
    }

    // 5. 检查每对队伍是否都有比赛
    for (int i = 0; i < teamCount; i++) {
        for (int j = i + 1; j < teamCount; j++) {
            if (!played[i * 16 + j]) {
                *returnSize = 0;
                return NULL;
            }
        }
    }

    // 6. 检查积分+净胜球并列
    for (int i = 0; i < teamCount; i++) {
        for (int j = i + 1; j < teamCount; j++) {
            int gdI = goalsFor[i] - goalsAgainst[i];
            int gdJ = goalsFor[j] - goalsAgainst[j];
            if (points[i] == points[j] && gdI == gdJ) {
                *returnSize = 0;
                return NULL;
            }
        }
    }

    // 7. 冒泡排序（按积分降序、净胜球降序）
    int order[16];
    for (int i = 0; i < teamCount; i++) order[i] = i;
    for (int i = 0; i < teamCount; i++) {
        for (int j = i + 1; j < teamCount; j++) {
            int pi = points[order[i]], pj = points[order[j]];
            int gdI = goalsFor[order[i]] - goalsAgainst[order[i]];
            int gdJ = goalsFor[order[j]] - goalsAgainst[order[j]];
            if (pi < pj || (pi == pj && gdI < gdJ)) {
                int tmp = order[i];
                order[i] = order[j];
                order[j] = tmp;
            }
        }
    }

    // 8. 构造返回数组（前三名）
    *returnSize = 3;
    char** result = (char**)malloc(4 * sizeof(char*));
    for (int i = 0; i < 3; i++) {
        result[i] = teams[order[i]];
    }
    result[3] = NULL;  // NULL 结尾
    return result;
}
```
