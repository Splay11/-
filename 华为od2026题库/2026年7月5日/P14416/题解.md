## 解题思路

本题要求在事务标签的约束下，将 SQL 语句行贪心分配到文件中并返回文件数。核心考察**并查集 + 约束传递 + 贪心分配**。

### 算法框架

1. **合法性校验**：`splitLine` 不在 $[1, 10000]$ 或 `sqlText` 为空 → 返回 0。

2. **解析事务标签**：遍历每一行，按 `;` 拆分语句，提取 `[Tn]` 格式的事务标签，记录每个标签涉及的行号集合。

3. **约束传递（并查集）**：
   - 对每个事务标签，将其涉及的所有行号合并（同一标签的语句必须在同一文件）。
   - 由于同行中可能出现多个不同标签（如一行同时有 `[T1]` 和 `[T2]`），并查集自然传递约束。

4. **分组排序**：将行号按连通分量分组，每组记录（首次出现行号、行数），按首次出现行号排序。

5. **贪心分配文件**：
   - 按首次出现顺序处理每组；
   - 当前文件能放下（组行数 ≤ 剩余容量）则放入；
   - 否则新建文件（即使组行数超过单文件限制也必须完整放入）。

### 关键点

- 约束传递：若 A 与 B 同标签、B 所在行还含 C 的标签，则 A、B、C 都在同一组
- 组行数可能超过 `splitLine`，但不可拆分——这是题目显式规定的
- 无事务标签的行作为独立单行组存在
- 输出只需文件个数，不需要具体分配方案

## 复杂度分析

- **时间复杂度**：$O(N + T + N \cdot \alpha(N))$，其中 $N$ 为 `sqlText` 行数，$T$ 为总语句数（含 `;` 拆分）。并查集操作接近 $O(1)$，排序 $O(N \log N)$，但 $N$ 受限于总语句 100000，可在时限内完成。
- **空间复杂度**：$O(N)$，存储标签到行号的映射及并查集。

```python
from typing import List

class Solution:
    def splitSQLToFiles(self, splitLine: int, sqlText: List[str]) -> int:
        # 1. 合法性校验
        if splitLine < 1 or splitLine > 10000:
            return 0
        n = len(sqlText)
        if n == 0:
            return 0

        # 2. 解析每行的事务标签，建 tag → {行号} 映射
        tag_to_rows = {}
        for idx, row in enumerate(sqlText):
            statements = row.split(';')
            # 去掉末尾空串（行以 ; 结尾时产生）
            while statements and statements[-1] == '':
                statements.pop()
            for stmt in statements:
                stmt = stmt.strip()
                if not stmt:
                    continue
                # 提取 [Tn] 标签
                if stmt.startswith('['):
                    end = stmt.find(']')
                    if end != -1:
                        tag = stmt[1:end]
                        if tag.startswith('T'):
                            tag_to_rows.setdefault(tag, set()).add(idx)

        # 3. 并查集：按事务标签合并行号
        parent = list(range(n))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(x, y):
            px, py = find(x), find(y)
            if px != py:
                parent[px] = py

        for rows in tag_to_rows.values():
            rows_list = list(rows)
            for i in range(1, len(rows_list)):
                union(rows_list[0], rows_list[i])

        # 4. 按连通分量分组 → (首次出现行号, 行数)
        groups = {}
        for i in range(n):
            root = find(i)
            groups.setdefault(root, set()).add(i)

        group_list = []
        for root, rows in groups.items():
            group_list.append((min(rows), len(rows)))
        group_list.sort()  # 按首次出现行号排序

        # 5. 贪心分配文件
        files = 0
        cur_size = 0
        for _, gsize in group_list:
            if cur_size == 0 or cur_size + gsize > splitLine:
                files += 1                           # 新建文件（即使超限也独占）
                cur_size = gsize
            else:
                cur_size += gsize                    # 放入当前文件

        return files
```

```java
import java.util.*;

class Solution {
    public int splitSQLToFiles(int splitLine, List<String> sqlText) {
        // 1. 合法性校验
        if (splitLine < 1 || splitLine > 10000) return 0;
        int n = sqlText.size();
        if (n == 0) return 0;

        // 2. 解析事务标签
        Map<String, Set<Integer>> tagToRows = new HashMap<>();
        for (int idx = 0; idx < n; idx++) {
            String[] statements = sqlText.get(idx).split(";");
            // 去掉末尾空串
            int end = statements.length;
            while (end > 0 && statements[end - 1].isEmpty()) end--;

            for (int s = 0; s < end; s++) {
                String stmt = statements[s].trim();
                if (stmt.isEmpty() || !stmt.startsWith("[")) continue;
                int br = stmt.indexOf(']');
                if (br == -1) continue;
                String tag = stmt.substring(1, br);
                if (tag.startsWith("T")) {
                    tagToRows.computeIfAbsent(tag, k -> new HashSet<>()).add(idx);
                }
            }
        }

        // 3. 并查集
        int[] parent = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;

        int[] find = new int[1];
        // 递归 find（用循环展开）
        for (Set<Integer> rows : tagToRows.values()) {
            if (rows.size() <= 1) continue;
            Integer[] arr = rows.toArray(new Integer[0]);
            // 找 root
            int root = arr[0];
            while (parent[root] != root) root = parent[root];
            for (int i = 1; i < arr.length; i++) {
                int r = arr[i];
                while (parent[r] != r) r = parent[r];
                if (r != root) parent[r] = root;
            }
        }
        // 路径压缩
        for (int i = 0; i < n; i++) {
            int r = i;
            while (parent[r] != r) r = parent[r];
            parent[i] = r;
        }

        // 4. 分组
        Map<Integer, int[]> groups = new HashMap<>(); // root → [minIdx, count]
        for (int i = 0; i < n; i++) {
            int root = parent[i];
            int[] g = groups.get(root);
            if (g == null) {
                groups.put(root, new int[]{i, 1});
            } else {
                if (i < g[0]) g[0] = i;
                g[1]++;
            }
        }

        List<int[]> glist = new ArrayList<>(groups.values());
        glist.sort((a, b) -> Integer.compare(a[0], b[0]));

        // 5. 贪心分配
        int files = 0, curSize = 0;
        for (int[] g : glist) {
            if (curSize == 0 || curSize + g[1] > splitLine) {
                files++;              // 新建文件（即使超限也独占）
                curSize = g[1];
            } else {
                curSize += g[1];      // 放入当前文件
            }
        }

        return files;
    }
}
```

```cpp
#include <vector>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <algorithm>
using namespace std;

class Solution {
public:
    int splitSQLToFiles(int splitLine, vector<string>& sqlText) {
        // 1. 合法性校验
        if (splitLine < 1 || splitLine > 10000) return 0;
        int n = (int)sqlText.size();
        if (n == 0) return 0;

        // 2. 解析标签 → 行号集合
        unordered_map<string, unordered_set<int>> tagToRows;
        for (int idx = 0; idx < n; idx++) {
            const string& row = sqlText[idx];
            size_t start = 0;
            while (start <= row.size()) {
                size_t semi = row.find(';', start);
                string stmt = (semi == string::npos)
                    ? row.substr(start) : row.substr(start, semi - start);
                // 跳过空串（末尾无分号时可能为空）
                if (!stmt.empty()) {
                    // 去除前后空格
                    size_t s = stmt.find_first_not_of(' ');
                    if (s != string::npos) {
                        size_t e = stmt.find_last_not_of(' ');
                        stmt = stmt.substr(s, e - s + 1);
                    }
                    if (!stmt.empty() && stmt[0] == '[') {
                        size_t br = stmt.find(']');
                        if (br != string::npos) {
                            string tag = stmt.substr(1, br - 1);
                            if (!tag.empty() && tag[0] == 'T')
                                tagToRows[tag].insert(idx);
                        }
                    }
                }
                if (semi == string::npos) break;
                start = semi + 1;
            }
        }

        // 3. 并查集
        vector<int> parent(n);
        for (int i = 0; i < n; i++) parent[i] = i;
        for (auto& kv : tagToRows) {
            const auto& rows = kv.second;
            if (rows.size() <= 1) continue;
            int first = *rows.begin();
            for (auto it = next(rows.begin()); it != rows.end(); ++it) {
                // 合并到 first
                int px = first, py = *it;
                while (parent[px] != px) px = parent[px];
                while (parent[py] != py) py = parent[py];
                if (px != py) parent[py] = px;
            }
        }
        // 路径压缩
        for (int i = 0; i < n; i++) {
            int r = i;
            while (parent[r] != r) r = parent[r];
            parent[i] = r;
        }

        // 4. 分组 → (首次出现行号, 行数)
        unordered_map<int, pair<int, int>> groups;
        for (int i = 0; i < n; i++) {
            int root = parent[i];
            auto it = groups.find(root);
            if (it == groups.end())
                groups[root] = {i, 1};
            else {
                it->second.first = min(it->second.first, i);
                it->second.second++;
            }
        }
        vector<pair<int, int>> glist;
        for (auto& kv : groups) glist.push_back(kv.second);
        sort(glist.begin(), glist.end());

        // 5. 贪心分配
        int files = 0, curSize = 0;
        for (auto& g : glist) {
            if (curSize == 0 || curSize + g.second > splitLine) {
                files++;              // 新建文件（即使超限也独占）
                curSize = g.second;
            } else {
                curSize += g.second;  // 放入当前文件
            }
        }
        return files;
    }
};
```

```javascript
/**
 * @param {number} splitLine
 * @param {string[]} sqlText
 * @return {number}

var splitSQLToFiles = function(splitLine, sqlText) {
    // 1. 合法性校验
    if (splitLine < 1 || splitLine > 10000) return 0;
    const n = sqlText.length;
    if (n === 0) return 0;

    // 2. 解析标签 → 行号集合
    const tagToRows = {};
    for (let idx = 0; idx < n; idx++) {
        const statements = sqlText[idx].split(';');
        const end = statements.length > 0 && statements[statements.length - 1] === ''
            ? statements.length - 1 : statements.length;
        for (let s = 0; s < end; s++) {
            const stmt = statements[s].trim();
            if (!stmt || !stmt.startsWith('[')) continue;
            const br = stmt.indexOf(']');
            if (br === -1) continue;
            const tag = stmt.substring(1, br);
            if (tag.startsWith('T')) {
                if (!tagToRows[tag]) tagToRows[tag] = new Set();
                tagToRows[tag].add(idx);
            }
        }
    }

    // 3. 并查集
    const parent = Array.from({ length: n }, (_, i) => i);
    const find = (x) => {
        while (parent[x] !== x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    };
    const union = (x, y) => {
        const px = find(x), py = find(y);
        if (px !== py) parent[px] = py;
    };

    for (const rows of Object.values(tagToRows)) {
        const arr = [...rows];
        for (let i = 1; i < arr.length; i++)
            union(arr[0], arr[i]);
    }

    // 4. 分组
    const groups = {};
    for (let i = 0; i < n; i++) {
        const root = find(i);
        if (!groups[root]) groups[root] = [i, 0];
        if (i < groups[root][0]) groups[root][0] = i;
        groups[root][1]++;
    }
    const glist = Object.values(groups).sort((a, b) => a[0] - b[0]);

    // 5. 贪心分配
    let files = 0, curSize = 0;
    for (const [, gsize] of glist) {
        if (curSize === 0 || curSize + gsize > splitLine) {
            files++;              // 新建文件（即使超限也独占）
            curSize = gsize;
        } else {
            curSize += gsize;     // 放入当前文件
        }
    }

    return files;
};
```

```c
#include <string.h>
#include <stdlib.h>

int splitSQLToFiles(int splitLine, char** sqlText, int sqlTextSize) {
    // 1. 合法性校验
    if (splitLine < 1 || splitLine > 10000) return 0;
    int n = sqlTextSize;
    if (n == 0) return 0;

    // 2. 解析标签
    // 用邻接表记录每个标签涉及的行（最多 1000 标签 × 若干行）
    int** tagRows = (int**)calloc(1001, sizeof(int*));
    int* tagSizes = (int*)calloc(1001, sizeof(int));
    int* tagCaps = (int*)calloc(1001, sizeof(int));

    for (int idx = 0; idx < n; idx++) {
        const char* row = sqlText[idx];
        int rowLen = (int)strlen(row);
        int pos = 0;
        while (pos < rowLen) {
            // 找 ; 分隔
            int semi = pos;
            while (semi < rowLen && row[semi] != ';') semi++;
            int stmtLen = semi - pos;

            if (stmtLen > 0) {
                const char* stmt = row + pos;
                // 跳过前导空格
                while (stmtLen > 0 && stmt[0] == ' ') { stmt++; stmtLen--; }
                // 检查 [T
                if (stmtLen > 2 && stmt[0] == '[' && stmt[1] == 'T') {
                    int br = 2;
                    while (br < stmtLen && stmt[br] != ']') br++;
                    if (br < stmtLen) {
                        // 提取标签号
                        int tagNum = 0;
                        for (int k = 2; k < br; k++) {
                            if (stmt[k] >= '0' && stmt[k] <= '9')
                                tagNum = tagNum * 10 + (stmt[k] - '0');
                        }
                        if (tagNum >= 1 && tagNum <= 1000) {
                            // 加入 tagRows[tagNum]
                            if (tagSizes[tagNum] >= tagCaps[tagNum]) {
                                tagCaps[tagNum] = tagCaps[tagNum] == 0 ? 4 : tagCaps[tagNum] * 2;
                                tagRows[tagNum] = (int*)realloc(tagRows[tagNum],
                                    tagCaps[tagNum] * sizeof(int));
                            }
                            // 去重：只加首次出现的行
                            int dup = 0;
                            for (int k = 0; k < tagSizes[tagNum]; k++) {
                                if (tagRows[tagNum][k] == idx) { dup = 1; break; }
                            }
                            if (!dup)
                                tagRows[tagNum][tagSizes[tagNum]++] = idx;
                        }
                    }
                }
            }

            pos = semi + 1;
            if (semi == rowLen) break;
        }
    }

    // 3. 并查集
    int* parent = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) parent[i] = i;

    // find 内联
    for (int t = 1; t <= 1000; t++) {
        if (tagSizes[t] <= 1) continue;
        int first = tagRows[t][0];
        // 压缩 first
        int fRoot = first;
        while (parent[fRoot] != fRoot) fRoot = parent[fRoot];
        for (int k = 1; k < tagSizes[t]; k++) {
            int x = tagRows[t][k];
            int xRoot = x;
            while (parent[xRoot] != xRoot) xRoot = parent[xRoot];
            if (xRoot != fRoot) parent[xRoot] = fRoot;
        }
    }
    // 路径压缩
    for (int i = 0; i < n; i++) {
        int r = i;
        while (parent[r] != r) r = parent[r];
        parent[i] = r;
    }

    // 4. 分组
    int* groupMin = (int*)malloc(n * sizeof(int));
    int* groupCnt = (int*)calloc(n, sizeof(int));
    for (int i = 0; i < n; i++) {
        int r = parent[i];
        if (groupCnt[r] == 0)
            groupMin[r] = i;
        else if (i < groupMin[r])
            groupMin[r] = i;
        groupCnt[r]++;
    }

    // 收集非空组并冒泡排序
    int* order = (int*)malloc(n * sizeof(int));
    int orderSize = 0;
    for (int i = 0; i < n; i++) {
        if (groupCnt[i] > 0)
            order[orderSize++] = i;
    }
    for (int i = 0; i < orderSize; i++) {
        for (int j = i + 1; j < orderSize; j++) {
            if (groupMin[order[i]] > groupMin[order[j]]) {
                int tmp = order[i];
                order[i] = order[j];
                order[j] = tmp;
            }
        }
    }

    // 5. 贪心分配
    int files = 0, curSize = 0;
    for (int i = 0; i < orderSize; i++) {
        int gsize = groupCnt[order[i]];
        if (curSize == 0 || curSize + gsize > splitLine) {
            files++;              // 新建文件（即使超限也独占）
            curSize = gsize;
        } else {
            curSize += gsize;     // 放入当前文件
        }
    }

    // 清理
    for (int t = 1; t <= 1000; t++) free(tagRows[t]);
    free(tagRows); free(tagSizes); free(tagCaps);
    free(parent); free(groupMin); free(groupCnt); free(order);

    return files;
}
```
