## 算法框架

层序序列中的 `null` 只表示当前节点的某个孩子不存在。`null` 本身不会进入队列，也不会继续拥有孩子。

因此不能直接通过数组下标计算父节点，需要使用一个队列还原真实的父子关系：

`1`. 根节点入队。
`2`. 每次取出一个非空节点。
`3`. 从序列中依次读取两个位置，分别作为它的左右孩子。
`4`. 非空孩子记录父节点和深度，并加入队列。
`5`. `null` 只占用一个孩子位置，但不加入队列。

完成父节点关系构建后，继续使用原来的 `LCA` 上溯方法即可。

## 复杂度分析

设序列长度为 `n`，树高为 `h`：

* 构建父节点关系：`O(n)`
* 查找两个节点：`O(n)`
* LCA 上溯：`O(h)`
* 总时间复杂度：`O(n)`
* 空间复杂度：`O(n)`

注意二叉树可能退化为链，因此树高最坏为 `O(n)`，不一定是 `O(log n)`。

```python
class Solution:
    def minJumps(self, treeLevelOrder: str, frm: str, to: str) -> int:
        nodes = [node.strip() for node in treeLevelOrder.split(',')]
        n = len(nodes)

        if n == 0 or nodes[0] == "null":
            return -1

        # parent[i] 表示序列下标 i 对应节点的真实父节点下标
        parent = [-1] * n
        depth = [-1] * n

        # 队列中只保存非空节点
        queue = [0]
        depth[0] = 0

        head = 0
        index = 1

        # 每个非空节点依次消费两个孩子位置
        while head < len(queue) and index < n:
            current = queue[head]
            head += 1

            for _ in range(2):
                if index >= n:
                    break

                if nodes[index] != "null":
                    parent[index] = current
                    depth[index] = depth[current] + 1
                    queue.append(index)

                index += 1

        # 只查找真实可达的非空节点
        pos_frm = -1
        pos_to = -1

        for i, node in enumerate(nodes):
            if depth[i] == -1:
                continue

            if node == frm:
                pos_frm = i
            if node == to:
                pos_to = i

        if pos_frm == -1 or pos_to == -1:
            return -1

        d1 = depth[pos_frm]
        d2 = depth[pos_to]
        orig_d1 = d1
        orig_d2 = d2

        # 将两个节点提升到相同深度
        while d1 > d2:
            pos_frm = parent[pos_frm]
            d1 -= 1

        while d2 > d1:
            pos_to = parent[pos_to]
            d2 -= 1

        # 同时上溯寻找最近公共祖先
        while pos_frm != pos_to:
            pos_frm = parent[pos_frm]
            pos_to = parent[pos_to]

        return orig_d1 + orig_d2 - 2 * depth[pos_frm]
```

```java
import java.util.Arrays;

class Solution {
    public int minJumps(String treeLevelOrder, String frm, String to) {
        String[] nodes = treeLevelOrder.split(",", -1);

        for (int i = 0; i < nodes.length; i++) {
            nodes[i] = nodes[i].trim();
        }

        int n = nodes.length;

        if (n == 0 || nodes[0].equals("null")) {
            return -1;
        }

        int[] parent = new int[n];
        int[] depth = new int[n];

        Arrays.fill(parent, -1);
        Arrays.fill(depth, -1);

        // 使用 int 数组模拟队列，避免大量 Integer 装箱
        int[] queue = new int[n];
        int head = 0;
        int tail = 0;

        queue[tail++] = 0;
        depth[0] = 0;

        int index = 1;

        while (head < tail && index < n) {
            int current = queue[head++];

            for (int child = 0; child < 2 && index < n;
                 child++, index++) {

                if (!nodes[index].equals("null")) {
                    parent[index] = current;
                    depth[index] = depth[current] + 1;
                    queue[tail++] = index;
                }
            }
        }

        int posFrm = -1;
        int posTo = -1;

        for (int i = 0; i < n; i++) {
            if (depth[i] == -1) {
                continue;
            }

            if (nodes[i].equals(frm)) {
                posFrm = i;
            }

            if (nodes[i].equals(to)) {
                posTo = i;
            }
        }

        if (posFrm == -1 || posTo == -1) {
            return -1;
        }

        int d1 = depth[posFrm];
        int d2 = depth[posTo];
        int originalD1 = d1;
        int originalD2 = d2;

            posFrm = parent[posFrm];
            d1--;

            posTo = parent[posTo];
            d2--;

            posFrm = parent[posFrm];

    }
}
```

```cpp
#include <string>
#include <vector>
using namespace std;

class Solution {
public:
    int minJumps(string treeLevelOrder, string frm, string to) {
        // 按逗号拆分并去空格
        vector<string> nodes;

        size_t start = 0;

        while (start <= treeLevelOrder.size()) {
            size_t comma = treeLevelOrder.find(',', start);

            string node = comma == string::npos
                ? treeLevelOrder.substr(start)
                : treeLevelOrder.substr(start, comma - start);

            size_t left = node.find_first_not_of(" \t\r\n");

            if (left != string::npos) {
                size_t right = node.find_last_not_of(" \t\r\n");
                node = node.substr(left, right - left + 1);
            } else {
                node.clear();
            }

            nodes.push_back(node);

            if (comma == string::npos) {
                break;
            }

            start = comma + 1;
        }

        int n = static_cast<int>(nodes.size());

        if (n == 0 || nodes[0] == "null") {
            return -1;
        }

        vector<int> parent(n, -1);
        vector<int> depth(n, -1);

        // 队列中只保存非空节点的序列下标
        vector<int> queue;
        queue.reserve(n);

        queue.push_back(0);
        depth[0] = 0;

        int head = 0;
        int index = 1;

        while (head < static_cast<int>(queue.size()) && index < n) {
            int current = queue[head++];

            for (int child = 0; child < 2 && index < n;
                 child++, index++) {

                if (nodes[index] != "null") {
                    parent[index] = current;
                    depth[index] = depth[current] + 1;
                    queue.push_back(index);
```

```c
#include <ctype.h>
#include <stdlib.h>
#include <string.h>

static char* trim(char* str) {
    while (*str && isspace((unsigned char)*str)) {
        str++;
    }

    char* end = str + strlen(str);

    while (end > str && isspace((unsigned char)*(end - 1))) {
        end--;
    }

    *end = '\0';
    return str;
}

int minJumps(char* treeLevelOrder, char* frm, char* to) {
    if (treeLevelOrder == NULL || frm == NULL || to == NULL) {
        return -1;
    }

    // 统计序列元素数量，用于分配数组
    int nodeCount = 1;

    for (char* p = treeLevelOrder; *p; p++) {
        if (*p == ',') {
            nodeCount++;
        }
    }

    int* parent = (int*)malloc(sizeof(int) * nodeCount);
    int* depth = (int*)malloc(sizeof(int) * nodeCount);
    int* queue = (int*)malloc(sizeof(int) * nodeCount);

    if (parent == NULL || depth == NULL || queue == NULL) {
        free(parent);
        free(depth);
        free(queue);
        return -1;
    }

    for (int i = 0; i < nodeCount; i++) {
        parent[i] = -1;
        depth[i] = -1;
    }

    char* token = strtok(treeLevelOrder, ",");

    if (token == NULL) {
        free(parent);
        free(depth);
        free(queue);
        return -1;
    }

    token = trim(token);

    if (*token == '\0' || strcmp(token, "null") == 0) {
        free(parent);
        free(depth);
        free(queue);
        return -1;
    }

    int head = 0;
    int tail = 0;

    queue[tail++] = 0;
    depth[0] = 0;

    int posFrm = strcmp(token, frm) == 0 ? 0 : -1;
    int posTo = strcmp(token, to) == 0 ? 0 : -1;

    int index = 0;
    int childSlot = 0;

    while ((token = strtok(NULL, ",")) != NULL) {
        index++;
        token = trim(token);

        // 队列为空说明后续节点不属于这棵树
        if (head >= tail) {
            continue;
        }

        int currentParent = queue[head];

        if (*token != '\0' && strcmp(token, "null") != 0) {
            parent[index] = currentParent;
            depth[index] = depth[currentParent] + 1;
            queue[tail++] = index;

            if (strcmp(token, frm) == 0) {
                posFrm = index;
            }

            if (strcmp(token, to) == 0) {
                posTo = index;
            }
        }

        childSlot++;

        // 当前父节点的左右孩子都已经处理完
        if (childSlot == 2) {
            childSlot = 0;
            head++;
        }
    }

    if (posFrm == -1 || posTo == -1) {
        free(parent);
        free(depth);
        free(queue);
        return -1;
    }

    int d1 = depth[posFrm];
    int d2 = depth[posTo];
    int originalD1 = d1;
    int originalD2 = d2;

    while (d1 > d2) {
        posFrm = parent[posFrm];
        d1--;
    }

    while (d2 > d1) {
        posTo = parent[posTo];
        d2--;
    }

    while (posFrm != posTo) {
        posFrm = parent[posFrm];
        posTo = parent[posTo];
    }

    int result =
        originalD1 + originalD2 - 2 * depth[posFrm];

    free(parent);
    free(depth);
    free(queue);

    return result;
}
```
