## 解题思路

这题本质上是 **字符串解析 + 哈希表模拟**。

我们需要按顺序解析批量命令，每条命令独立执行；即使某条失败，也不能影响后续命令继续处理。
因此最适合用：

* **字符串分割**：把整体字符串拆成一条条命令
* **哈希表**：维护当前系统中的 `rule_id -> rule_index`

### 核心规则整理

对于每条命令：

1. **命令关键字必须合法**

   * 只能是 `add_rule`、`mod_rule`、`del_rule`

2. **属性必须合法**

   * `rule_id`、`rule_index` 的值都必须是数字，且范围在 `1 ~ 9999`
   * 缺少必要属性也算失败

3. **各操作的业务规则**

   * `add_rule`：

     * 必须有 `rule_id` 和 `rule_index`
     * 当前 `rule_id` 不存在才能添加成功
   * `mod_rule`：

     * 必须有 `rule_id` 和 `rule_index`
     * 当前 `rule_id` 必须存在
     * 新旧 `rule_index` 必须不同
   * `del_rule`：

     * 必须有 `rule_id`
     * 当前 `rule_id` 必须存在


### 实现方法

用一个哈希表记录当前规则：

* 键：`rule_id`
* 值：`rule_index`

然后遍历每条命令：

1. 先拆出操作关键字和各属性
2. 校验关键字、属性格式、数值范围
3. 按操作类型模拟执行
4. 若失败则失败次数加一

### 复杂度分析

设总字符串长度为 `n`，命令条数为 `m`。

* **时间复杂度：`O(n)`**

  * 每个字符最多被扫描、分割、解析一次
  * 哈希表增删改查均摊 `O(1)`
* **空间复杂度：`O(m)`**

  * 最多存下所有成功添加的规则

这个复杂度对于题目中“不超过 1000 条命令”的数据范围完全足够。

```python
class Solution:
    def getErrorCount(self, rules: str) -> int:
        # 判断字符串是否是 1~9999 的合法数字
        def valid_num(s: str) -> bool:
            if not s or not s.isdigit():
                return False
            x = int(s)
            return 1 <= x <= 9999

        # 空串直接返回 0
        if not rules:
            return 0

        # 去掉整体首尾可能存在的双引号
        if len(rules) >= 2 and rules[0] == '"' and rules[-1] == '"':
            rules = rules[1:-1]

        # 按每条命令切分
        cmds = rules.split("][")

        # 哈希表：rule_id -> rule_index
        mp = {}
        ans = 0

        for cmd in cmds:
            # 去掉每条命令首尾可能残留的方括号
            cmd = cmd.strip("[]")
            parts = cmd.split()

            # 至少要有操作关键字
            if not parts:
                ans += 1
                continue

            op = parts[0]
            attrs = {}

            ok = True

            # 解析属性
            for i in range(1, len(parts)):
                if '=' not in parts[i]:
                    ok = False
                    break
                key, val = parts[i].split('=', 1)
                if key not in ("rule_id", "rule_index"):
                    ok = False
                    break
                if not valid_num(val):
                    ok = False
                    break
                attrs[key] = int(val)

            if not ok:
                ans += 1
                continue

            # 分类处理
            if op == "add_rule":
                # add 必须同时有 rule_id 和 rule_index
                if "rule_id" not in attrs or "rule_index" not in attrs:
                    ans += 1
                    continue
                rule_id = attrs["rule_id"]
                rule_index = attrs["rule_index"]
                if rule_id in mp:
                    ans += 1
                else:
                    mp[rule_id] = rule_index

            elif op == "mod_rule":
                # mod 必须同时有 rule_id 和 rule_index
                if "rule_id" not in attrs or "rule_index" not in attrs:
                    ans += 1
                    continue
                rule_id = attrs["rule_id"]
                rule_index = attrs["rule_index"]
                if rule_id not in mp or mp[rule_id] == rule_index:
                    ans += 1
                else:
                    mp[rule_id] = rule_index

            elif op == "del_rule":
                # del 只要求有 rule_id
                if "rule_id" not in attrs:
                    ans += 1
                    continue
                rule_id = attrs["rule_id"]
                if rule_id not in mp:
                    ans += 1
                else:
                    del mp[rule_id]

            else:
                # 非法操作关键字
                ans += 1

        return ans
```

```java
import java.util.*;

class Solution {
    public int getErrorCount(String rules) {
        // 空串直接返回 0
        if (rules == null || rules.length() == 0) {
            return 0;
        }

        // 去掉整体首尾可能存在的双引号
        if (rules.length() >= 2 && rules.charAt(0) == '"' && rules.charAt(rules.length() - 1) == '"') {
            rules = rules.substring(1, rules.length() - 1);
        }

        // 按每条命令切分
        String[] cmds = rules.split("\\]\\[");

        // 哈希表：rule_id -> rule_index
        Map<Integer, Integer> map = new HashMap<>();
        int ans = 0;

        for (String cmd : cmds) {
            // 去掉每条命令首尾可能残留的方括号
            cmd = trimBrackets(cmd);
            if (cmd.length() == 0) {
                ans++;
                continue;
            }

            String[] parts = cmd.split(" ");
            if (parts.length == 0) {
                ans++;
                continue;
            }

            String op = parts[0];
            Map<String, Integer> attrs = new HashMap<>();
            boolean ok = true;

            // 解析属性
            for (int i = 1; i < parts.length; i++) {
                int idx = parts[i].indexOf('=');
                if (idx == -1) {
                    ok = false;
                    break;
                }

                String key = parts[i].substring(0, idx);
                String val = parts[i].substring(idx + 1);

                if (!key.equals("rule_id") && !key.equals("rule_index")) {
                    ok = false;
                    break;
                }

                if (!validNum(val)) {
                    ok = false;
                    break;
                }

                attrs.put(key, Integer.parseInt(val));
            }

            if (!ok) {
                ans++;
                continue;
            }

            // 分类处理
            if (op.equals("add_rule")) {
                // add 必须同时有 rule_id 和 rule_index
                if (!attrs.containsKey("rule_id") || !attrs.containsKey("rule_index")) {
                    ans++;
                    continue;
                }

                int ruleId = attrs.get("rule_id");
                int ruleIndex = attrs.get("rule_index");

                if (map.containsKey(ruleId)) {
                    ans++;
                } else {
                    map.put(ruleId, ruleIndex);
                }

            } else if (op.equals("mod_rule")) {
                // mod 必须同时有 rule_id 和 rule_index
                if (!attrs.containsKey("rule_id") || !attrs.containsKey("rule_index")) {
                    ans++;
                    continue;
                }

                int ruleId = attrs.get("rule_id");
                int ruleIndex = attrs.get("rule_index");

                if (!map.containsKey(ruleId) || map.get(ruleId) == ruleIndex) {
                    ans++;
                } else {
                    map.put(ruleId, ruleIndex);
                }

            } else if (op.equals("del_rule")) {
                // del 只要求有 rule_id
                if (!attrs.containsKey("rule_id")) {
                    ans++;
                    continue;
                }

                int ruleId = attrs.get("rule_id");

                if (!map.containsKey(ruleId)) {
                    ans++;
                } else {
                    map.remove(ruleId);
                }

            } else {
                // 非法操作关键字
                ans++;
            }
        }

        return ans;
    }

    // 判断字符串是否是 1~9999 的合法数字
    private boolean validNum(String s) {
        if (s == null || s.length() == 0) {
            return false;
        }

        for (int i = 0; i < s.length(); i++) {
            if (!Character.isDigit(s.charAt(i))) {
                return false;
            }
        }

        int x = Integer.parseInt(s);
        return x >= 1 && x <= 9999;
    }

    // 去掉字符串首尾的 '[' 和 ']'
    private String trimBrackets(String s) {
        int l = 0, r = s.length() - 1;
        while (l <= r && s.charAt(l) == '[') {
            l++;
        }
        while (l <= r && s.charAt(r) == ']') {
            r--;
        }
        return s.substring(l, r + 1);
    }
}
```

```cpp
#include <string>
#include <vector>
#include <unordered_map>
using namespace std;

class Solution {
public:
    int getErrorCount(string rules) {
        // 空串直接返回 0
        if (rules.empty()) {
            return 0;
        }

        // 去掉整体首尾可能存在的双引号
        if (rules.size() >= 2 && rules.front() == '"' && rules.back() == '"') {
            rules = rules.substr(1, rules.size() - 2);
        }

        // 按每条命令切分
        vector<string> cmds = splitCommands(rules);

        // 哈希表：rule_id -> rule_index
        unordered_map<int, int> mp;
        int ans = 0;

        for (string cmd : cmds) {
            // 去掉每条命令首尾可能残留的方括号
            cmd = trimBrackets(cmd);
            if (cmd.empty()) {
                ans++;
                continue;
            }

            vector<string> parts = splitBySpace(cmd);
            if (parts.empty()) {
                ans++;
                continue;
            }

            string op = parts[0];
            unordered_map<string, int> attrs;
            bool ok = true;

            // 解析属性
            for (int i = 1; i < (int)parts.size(); i++) {
                int pos = parts[i].find('=');
                if (pos == string::npos) {
                    ok = false;
                    break;
                }

                string key = parts[i].substr(0, pos);
                string val = parts[i].substr(pos + 1);

                if (key != "rule_id" && key != "rule_index") {
                    ok = false;
                    break;
                }

                if (!validNum(val)) {
                    ok = false;
                    break;
                }

                attrs[key] = stoi(val);
            }

            if (!ok) {
                ans++;
                continue;
            }

            // 分类处理
            if (op == "add_rule") {
                // add 必须同时有 rule_id 和 rule_index
                if (!attrs.count("rule_id") || !attrs.count("rule_index")) {
                    ans++;
                    continue;
                }

                int rule_id = attrs["rule_id"];
                int rule_index = attrs["rule_index"];

                if (mp.count(rule_id)) {
                    ans++;
                } else {
                    mp[rule_id] = rule_index;
                }

            } else if (op == "mod_rule") {
                // mod 必须同时有 rule_id 和 rule_index
                if (!attrs.count("rule_id") || !attrs.count("rule_index")) {
                    ans++;
                    continue;
                }

                int rule_id = attrs["rule_id"];
                int rule_index = attrs["rule_index"];

                if (!mp.count(rule_id) || mp[rule_id] == rule_index) {
                    ans++;
                } else {
                    mp[rule_id] = rule_index;
                }

            } else if (op == "del_rule") {
                // del 只要求有 rule_id
                if (!attrs.count("rule_id")) {
                    ans++;
                    continue;
                }

                int rule_id = attrs["rule_id"];

                if (!mp.count(rule_id)) {
                    ans++;
                } else {
                    mp.erase(rule_id);
                }

            } else {
                // 非法操作关键字
                ans++;
            }
        }

        return ans;
    }

private:
    // 判断字符串是否是 1~9999 的合法数字
    bool validNum(const string& s) {
        if (s.empty()) {
            return false;
        }
        for (char c : s) {
            if (c < '0' || c > '9') {
                return false;
            }
        }
        int x = stoi(s);
        return x >= 1 && x <= 9999;
    }

    // 去掉字符串首尾的 '[' 和 ']'
    string trimBrackets(const string& s) {
        int l = 0, r = (int)s.size() - 1;
        while (l <= r && s[l] == '[') l++;
        while (l <= r && s[r] == ']') r--;
        if (l > r) return "";
        return s.substr(l, r - l + 1);
    }

    // 按 "][" 切分命令
    vector<string> splitCommands(const string& s) {
        vector<string> res;
        int start = 0;
        for (int i = 0; i + 1 < (int)s.size(); i++) {
            if (s[i] == ']' && s[i + 1] == '[') {
                res.push_back(s.substr(start, i - start + 1));
                start = i + 1;
            }
        }
        res.push_back(s.substr(start));
        return res;
    }

    // 按空格切分
    vector<string> splitBySpace(const string& s) {
        vector<string> res;
        string cur;
        for (char c : s) {
            if (c == ' ') {
                if (!cur.empty()) {
                    res.push_back(cur);
                    cur.clear();
                }
            } else {
                cur.push_back(c);
            }
        }
        if (!cur.empty()) {
            res.push_back(cur);
        }
        return res;
    }
};
```

```javascript
/**
 * @param {string} rules - 批量操作的字符串，格式为[cmd][cmd][cmd]
 * @return {number} - 配置失败的次数
 */
var getErrorCount = function(rules) {
    // 判断字符串是否为 1~9999 的合法数字
    function validNum(s) {
        if (!s || s.length === 0) return false;
        for (let i = 0; i < s.length; i++) {
            if (s[i] < '0' || s[i] > '9') return false;
        }
        const x = parseInt(s, 10);
        return x >= 1 && x <= 9999;
    }

    // 空串直接返回 0
    if (!rules || rules.length === 0) return 0;

    // 去掉整体首尾可能存在的双引号
    if (rules.length >= 2 && rules[0] === '"' && rules[rules.length - 1] === '"') {
        rules = rules.slice(1, -1);
    }
    if (rules.length === 0) return 0;

    // 按 "][" 切分命令
    const cmds = rules.split('][');

    // 哈希表：rule_id -> rule_index
    const mp = new Map();
    let ans = 0;

    for (let rawCmd of cmds) {
        // 去掉首尾可能残留的方括号
        let cmd = rawCmd.trim();
        if (cmd.length > 0 && cmd[0] === '[') cmd = cmd.slice(1);
        if (cmd.length > 0 && cmd[cmd.length - 1] === ']') cmd = cmd.slice(0, -1);

        const parts = cmd.split(' ');
        if (parts.length === 0 || parts[0] === '') {
            ans++;
            continue;
        }

        const op = parts[0];
        const attrs = {};
        let ok = true;

        // 解析属性
        for (let i = 1; i < parts.length; i++) {
            const eqIdx = parts[i].indexOf('=');
            if (eqIdx === -1) { ok = false; break; }
            const key = parts[i].substring(0, eqIdx);
            const val = parts[i].substring(eqIdx + 1);

            if (key !== 'rule_id' && key !== 'rule_index') { ok = false; break; }
            if (!validNum(val)) { ok = false; break; }
            attrs[key] = parseInt(val, 10);
        }

        if (!ok) { ans++; continue; }

        // 分类处理
        if (op === 'add_rule') {
            if (!('rule_id' in attrs) || !('rule_index' in attrs)) { ans++; continue; }
            const ruleId = attrs['rule_id'];
            const ruleIndex = attrs['rule_index'];
            if (mp.has(ruleId)) { ans++; continue; }
            mp.set(ruleId, ruleIndex);
        } else if (op === 'mod_rule') {
            if (!('rule_id' in attrs) || !('rule_index' in attrs)) { ans++; continue; }
            const ruleId = attrs['rule_id'];
            const ruleIndex = attrs['rule_index'];
            if (!mp.has(ruleId) || mp.get(ruleId) === ruleIndex) { ans++; continue; }
            mp.set(ruleId, ruleIndex);
        } else if (op === 'del_rule') {
            if (!('rule_id' in attrs)) { ans++; continue; }
            const ruleId = attrs['rule_id'];
            if (!mp.has(ruleId)) { ans++; continue; }
            mp.delete(ruleId);
        } else {
            // 非法操作关键字
            ans++;
        }
    }

    return ans;
};
```

```c
#include <string.h>
#include <stdlib.h>
#include <ctype.h>

/**
 * @param rules 批量操作字符串，格式 "[cmd][cmd]..."
 * @return 配置失败的次数
 */
int getErrorCount(char* rules) {
    int len = strlen(rules);
    if (len == 0) return 0;

    // 去掉首尾可能存在的双引号
    int start = 0, end = len - 1;
    if (len >= 2 && rules[0] == '"' && rules[end] == '"') {
        start = 1;
        end = len - 2;
    }
    if (start > end) return 0;

    // 用静态数组存储 rule_id -> rule_index 映射（id 范围 1~9999）
    static int hasEntry[10000];
    static int mp[10000];

    // 重置全局状态（支持多次调用）
    memset(hasEntry, 0, sizeof(hasEntry));

    int errors = 0;
    int i = start;
    while (i <= end) {
        // 找到命令开始 '['
        while (i <= end && rules[i] != '[') i++;
        if (i > end) break;
        i++; // 跳过 '['

        // 找到命令结束 ']'
        int cmdStart = i;
        int depth = 0;
        while (i <= end) {
            if (rules[i] == '[') depth++;
            else if (rules[i] == ']') {
                if (depth == 0) break;
                depth--;
            }
            i++;
        }
        int cmdEnd = i - 1;
        if (i > end) break;
        i++; // 跳过 ']'

        // 解析命令内容：按空格分割 token
        int pos = cmdStart;
        int has_id = 0, has_index = 0;
        int rule_id = 0, rule_index = 0;
        int attrOk = 1;
        char op[20] = "";
        int opSet = 0;

        while (pos <= cmdEnd) {
            // 跳过空格
            while (pos <= cmdEnd && rules[pos] == ' ') pos++;
            if (pos > cmdEnd) break;

            // 读取一个 token
            int tokStart = pos;
            while (pos <= cmdEnd && rules[pos] != ' ' && rules[pos] != '=') pos++;

            if (!opSet) {
                // 第一个 token 是操作关键字
                int tokLen = pos - tokStart;
                if (tokLen >= 20) { attrOk = 0; break; }
                memcpy(op, rules + tokStart, tokLen);
                op[tokLen] = '\0';
                opSet = 1;
                if (pos > cmdEnd || rules[pos] != ' ') { attrOk = 0; break; }
                continue;
            }

            // 后续 token 是 key=value
            int keyLen = pos - tokStart;
            if (pos > cmdEnd || rules[pos] != '=') { attrOk = 0; break; }
            pos++; // 跳过 '='

            // 读取 value
            int valStart = pos;
            while (pos <= cmdEnd && rules[pos] != ' ') pos++;
            int valLen = pos - valStart;
            if (valLen == 0) { attrOk = 0; break; }

            // 校验 value 为 1~9999 的数字
            int v = 0;
            for (int k = 0; k < valLen; k++) {
                if (rules[valStart + k] < '0' || rules[valStart + k] > '9') {
                    attrOk = 0;
                    break;
                }
                v = v * 10 + (rules[valStart + k] - '0');
            }
            if (!attrOk || v < 1 || v > 9999) { attrOk = 0; break; }

            // 校验 key
            if (keyLen == 7 && memcmp(rules + tokStart, "rule_id", 7) == 0) {
                has_id = 1; rule_id = v;
            } else if (keyLen == 10 && memcmp(rules + tokStart, "rule_index", 10) == 0) {
                has_index = 1; rule_index = v;
            } else {
                attrOk = 0; break;
            }
        }

        if (!attrOk || !opSet) { errors++; continue; }

        // 分类处理
        if (strcmp(op, "add_rule") == 0) {
            if (!has_id || !has_index) { errors++; continue; }
            if (hasEntry[rule_id]) { errors++; continue; }
            hasEntry[rule_id] = 1;
            mp[rule_id] = rule_index;
        } else if (strcmp(op, "mod_rule") == 0) {
            if (!has_id || !has_index) { errors++; continue; }
            if (!hasEntry[rule_id] || mp[rule_id] == rule_index) { errors++; continue; }
            mp[rule_id] = rule_index;
        } else if (strcmp(op, "del_rule") == 0) {
            if (!has_id) { errors++; continue; }
            if (!hasEntry[rule_id]) { errors++; continue; }
            hasEntry[rule_id] = 0;
        } else {
            errors++;
        }
    }

    return errors;
}
```
