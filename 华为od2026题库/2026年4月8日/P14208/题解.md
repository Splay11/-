### 解题思路

这道题的核心是 **哈希表**。

我们需要统计某个月份过生日的员工人数，但有一个关键条件：

* **同一员工如果被重复录入，只以最后一次录入为准**
* **同一员工不能重复统计**

所以最直接的方法是：

1. 用哈希表记录每个员工“最后一次录入”的生日月份

   * 遍历员工列表和生日列表
   * 从生日字符串中解析出月份
   * 用 `员工姓名 -> 月份` 存入哈希表
   * 如果员工重复出现，新的月份会覆盖旧的月份，正好符合“以最后一次录入结果为准”

2. 再遍历哈希表，统计月份等于目标月份的员工数量即可

这属于典型的 **哈希去重 + 覆盖更新** 的问题。

### 复杂度分析

设员工总数为 `n`。

* **时间复杂度：`O(n)`**

  * 第一次遍历列表，建立“员工 -> 最后生日月份”的映射，耗时 `O(n)`
  * 第二次遍历哈希表统计答案，最多也是 `O(n)`
  * 总体为 `O(n)`

* **空间复杂度：`O(n)`**

  * 哈希表最多存储每个员工一次，因此额外空间为 `O(n)`

由于题目中员工人数不超过 100，这个复杂度完全合适。

```python
class Solution:
    def countBirthdayGifts(self, month, employees, birthdays):
        # 哈希表：记录每个员工最后一次录入的生日月份
        last_month = {}

        # 遍历所有录入信息，后出现的同名员工会覆盖前面的记录
        for i in range(len(employees)):
            name = employees[i]
            birthday = birthdays[i]

            # 按 '/' 分割，取中间部分作为月份
            m = int(birthday.split('/')[1])
            last_month[name] = m

        # 统计最后生日月份等于目标月份的员工个数
        ans = 0
        for m in last_month.values():
            if m == month:
                ans += 1

        return ans
```

```java
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Solution {
    public int countBirthdayGifts(int month, List<String> employees, List<String> birthdays) {
        // 哈希表：记录每个员工最后一次录入的生日月份
        Map<String, Integer> lastMonth = new HashMap<>();

        // 遍历所有录入信息，后面的同名员工会覆盖前面的记录
        for (int i = 0; i < employees.size(); i++) {
            String name = employees.get(i);
            String birthday = birthdays.get(i);

            // 按 "/" 分割，取月份
            String[] parts = birthday.split("/");
            int m = Integer.parseInt(parts[1]);

            lastMonth.put(name, m);
        }

        // 统计目标月份的人数
        int ans = 0;
        for (int m : lastMonth.values()) {
            if (m == month) {
                ans++;
            }
        }

        return ans;
    }
}
```

```cpp
#include <vector>
#include <string>
#include <unordered_map>
using namespace std;

class Solution {
public:
    int countBirthdayGifts(int month, vector<string>& employees, vector<string>& birthdays) {
        // 哈希表：记录每个员工最后一次录入的生日月份
        unordered_map<string, int> lastMonth;

        // 遍历所有录入信息，后面的同名员工会覆盖前面的记录
        for (int i = 0; i < employees.size(); i++) {
            string name = employees[i];
            string birthday = birthdays[i];

            // 解析月份：格式为 Year/Month/Day
            int firstSlash = birthday.find('/');
            int secondSlash = birthday.find('/', firstSlash + 1);
            int m = stoi(birthday.substr(firstSlash + 1, secondSlash - firstSlash - 1));

            lastMonth[name] = m;
        }

        // 统计目标月份的人数
        int ans = 0;
        for (auto& entry : lastMonth) {
            if (entry.second == month) {
                ans++;
            }
        }

        return ans;
    }
};
```

```javascript
/**
 * @param {number} month 要发放礼物的月份
 * @param {string[]} employees 员工姓名列表
 * @param {string[]} birthdays 对应的员工生日日期列表
 * @return {number} 该月份要准备的礼品个数
 */
var countBirthdayGifts = function(month, employees, birthdays) {
    // 用 object 记录每个员工最后一次录入的生日月份
    const lastMonth = {};
    for (let i = 0; i < employees.length; i++) {
        const m = parseInt(birthdays[i].split('/')[1], 10);
        lastMonth[employees[i]] = m;
    }
    // 统计生日月份等于目标月份的员工数
    let cnt = 0;
    for (const m of Object.values(lastMonth)) {
        if (m === month) cnt++;
    }
    return cnt;
};
```

```c
#include <string.h>
#include <stdlib.h>

/**
 * @param month 要统计的月份（1~12）
 * @param employees 员工姓名数组
 * @param birthdays 员工生日日期数组
 * @param size 数组大小
 * @return 当月应准备的生日礼物数量
 */
int countBirthdayGifts(int month, char** employees, char** birthdays, int size) {
    // 员工数 ≤ 100，用简单数组模拟 name -> 最后录入月份的映射
    char* names[100];
    int months[100];
    int nameCount = 0;

    for (int i = 0; i < size; i++) {
        // 从生日字符串 "YYYY/M/D" 或 "YYYY/MM/DD" 中解析月份
        const char* b = birthdays[i];
        const char* p = b;
        while (*p && *p != '/') p++;   // 跳过年份
        if (!*p) continue;
        p++;                           // 跳过第一个 '/'
        int m = 0;
        while (*p && *p != '/') {
            m = m * 10 + (*p - '0');   // 手动解析月份数字
            p++;
        }

        // 查找该员工是否已有记录
        int found = -1;
        for (int j = 0; j < nameCount; j++) {
            if (strcmp(names[j], employees[i]) == 0) {
                found = j;
                break;
            }
        }
        if (found >= 0) {
            months[found] = m;         // 覆盖为最后一次录入的月份
        } else {
            names[nameCount] = employees[i];
            months[nameCount] = m;
            nameCount++;
        }
    }

    // 统计生日月份等于目标月份的员工数
    int cnt = 0;
    for (int i = 0; i < nameCount; i++) {
        if (months[i] == month) cnt++;
    }
    return cnt;
}
```
