## 解题思路

本题在 $n \times n$ 网格上，从固定起点 $(0, n/2)$ 走到固定终点 $(n-1, n/2)$，四向移动。每个哨兵 $(gx, gy)$ 会使切比雪夫距离 $\max(|x-gx|,|y-gy|)\le 1$ 的格子全部不可走（即以其为中心的 $3\times 3$ 九宫格并到禁区集合里）。

算法上这是**无权图最短路**问题：把每个可走格子看作顶点，四邻边权为 $1$。用 **BFS** 求从起点到终点的最短边数；若不可达返回 `[0,0]`。

在统计**最短路径条数**时，在 BFS 松弛过程中维护 `ways[x][y]`：从起点到 $(x,y)$ 的最短路条数。当从 $(x,y)$ 扩展到邻居 $(nx,ny)$ 时：

- 若 $(nx,ny)$ 首次被访问，则 `dist[nx][ny]=dist[x][y]+1`，`ways[nx][ny]=ways[x][y]`；
- 若已被访问且 `dist[nx][ny]==dist[x][y]+1`（说明是同层最短路上的另一条前驱），则 `ways[nx][ny]+=ways[x][y]`。

题面要求的「路径长度」为**路径上的格子数**，即最短路边数 $+1$，故第二维答案为 `dist[终点]+1`。

易错点：九宫格是**切比雪夫半径 1**，不是曼哈顿距离 $1$；多哨兵的禁区取**并集**；起点或终点落在任一哨兵九宫格内时应视为不可行。

## 复杂度分析

- 时间复杂度：$O(n^2)$。每个格子最多入队一次，每条边常数次检查；$n<30$，网格规模不超过 $900$。
- 空间复杂度：$O(n^2)$，存储 `dist`、`ways` 与禁区标记。

```python
from collections import deque
from typing import List


class Solution:
    def countShortestPaths(self, n: int, guards: List[List[int]]) -> List[int]:
        # 起点在底行正中，终点在顶行正中（题面坐标：行 x，列 y，左下为 (0,0)）
        sx, sy = 0, n // 2
        ex, ey = n - 1, n // 2
        # ban[x][y] 为 True 表示该格落在某个哨兵的九宫格内，不可站立
        ban = [[False] * n for _ in range(n)]
        for g in guards:
            gx, gy = int(g[0]), int(g[1])
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    nx, ny = gx + dx, gy + dy
                    if 0 <= nx < n and 0 <= ny < n:
                        ban[nx][ny] = True
        # 起终点若已在禁区内，直接无解
        if ban[sx][sy] or ban[ex][ey]:
            return [0, 0]
        dist = [[-1] * n for _ in range(n)]
        ways = [[0] * n for _ in range(n)]
        dist[sx][sy] = 0
        ways[sx][sy] = 1
        q = deque([(sx, sy)])
        dirs = ((1, 0), (-1, 0), (0, 1), (0, -1))
        while q:
            x, y = q.popleft()
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if nx < 0 or nx >= n or ny < 0 or ny >= n or ban[nx][ny]:
                    continue
                nd = dist[x][y] + 1
                if dist[nx][ny] == -1:
                    # 第一次到达 (nx,ny)，最短路长度确定
                    dist[nx][ny] = nd
                    ways[nx][ny] = ways[x][y]
                    q.append((nx, ny))
                elif dist[nx][ny] == nd:
                    # 另一条同样长度的最短路汇入
                    ways[nx][ny] += ways[x][y]
        if dist[ex][ey] == -1:
            return [0, 0]
        # 第二维：顶点数 = 边数 + 1
        return [ways[ex][ey], dist[ex][ey] + 1]
```

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;

class Solution {
    public List<Integer> countShortestPaths(int n, List<List<Integer>> guards) {
        int sy = n / 2;
        int sx = 0, ex = n - 1, ey = sy;
        boolean[][] ban = new boolean[n][n];
        // 将每个哨兵的 3x3 禁区并入 ban
        for (List<Integer> g : guards) {
            int gx = g.get(0), gy = g.get(1);
            for (int dx = -1; dx <= 1; dx++) {
                for (int dy = -1; dy <= 1; dy++) {
                    int nx = gx + dx, ny = gy + dy;
                    if (nx >= 0 && nx < n && ny >= 0 && ny < n) {
                        ban[nx][ny] = true;
                    }
                }
            }
        }
        if (ban[sx][sy] || ban[ex][ey]) {
            return make(0, 0);
        }
        int[][] dist = new int[n][n];
        long[][] ways = new long[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dist[i][j] = -1;
            }
        }
        int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        dist[sx][sy] = 0;
        ways[sx][sy] = 1;
        Deque<int[]> q = new ArrayDeque<>();
        q.add(new int[] {sx, sy});
        while (!q.isEmpty()) {
            int[] cur = q.pollFirst();
            int x = cur[0], y = cur[1];
            for (int[] d : dirs) {
                int nx = x + d[0], ny = y + d[1];
                if (nx < 0 || nx >= n || ny < 0 || ny >= n || ban[nx][ny]) {
                    continue;
                }
                int nd = dist[x][y] + 1;
                if (dist[nx][ny] == -1) {
                    dist[nx][ny] = nd;
                    ways[nx][ny] = ways[x][y];
                    q.addLast(new int[] {nx, ny});
                } else if (dist[nx][ny] == nd) {
                    // 同一最短路层上的多条前驱，累加方案数
                    ways[nx][ny] += ways[x][y];
                }
            }
        }
        if (dist[ex][ey] == -1) {
            return make(0, 0);
        }
        return make((int) ways[ex][ey], dist[ex][ey] + 1);
    }

    private static List<Integer> make(int a, int b) {
        List<Integer> r = new ArrayList<>();
        r.add(a);
        r.add(b);
        return r;
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    vector<int> countShortestPaths(int n, vector<vector<int>>& guards) {
        // 题面坐标：左下为 (0,0)，入口底行正中，司令部顶行正中
        int sy = n / 2;
        int sx = 0, ex = n - 1, ey = sy;
        // ban[x][y]=1 表示该格落在某个哨兵的九宫格（切比雪夫距离<=1）内，不可走
        vector<vector<char>> ban(n, vector<char>(n, 0));
        for (auto& g : guards) {
            int gx = g[0], gy = g[1];
            for (int dx = -1; dx <= 1; dx++) {
                for (int dy = -1; dy <= 1; dy++) {
                    int nx = gx + dx, ny = gy + dy;
                    if (nx >= 0 && nx < n && ny >= 0 && ny < n) {
                        ban[nx][ny] = 1;
                    }
                }
            }
        }
        // 起点或终点已在禁区内，无法出发或无法到达
        if (ban[sx][sy] || ban[ex][ey]) {
            return {0, 0};
        }
        const int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};  // 上、下、左、右四邻
        vector<vector<int>> dist(n, vector<int>(n, -1));             // -1 表示未访问
        vector<vector<long long>> ways(n, vector<long long>(n, 0));   // 最短路条数，可能较大用 long long
        deque<pair<int, int>> q;
        dist[sx][sy] = 0;
        ways[sx][sy] = 1;  // 起点方案数为 1
        q.push_back({sx, sy});
        while (!q.empty()) {
            int x = q.front().first, y = q.front().second;
            q.pop_front();
            for (int k = 0; k < 4; k++) {
                int nx = x + dirs[k][0], ny = y + dirs[k][1];
                if (nx < 0 || nx >= n || ny < 0 || ny >= n || ban[nx][ny]) {
                    continue;
                }
                int nd = dist[x][y] + 1;  // 走到邻居的边数 = 当前最短路 +1
                if (dist[nx][ny] == -1) {
                    // 首次到达：最短路长度确定，方案数继承自前驱
                    dist[nx][ny] = nd;
                    ways[nx][ny] = ways[x][y];
                    q.push_back({nx, ny});
                } else if (dist[nx][ny] == nd) {
                    // 同一最短路层上另有前驱指向 (nx,ny)，累加方案数
                    ways[nx][ny] += ways[x][y];
                }
            }
        }
        if (dist[ex][ey] == -1) {
            return {0, 0};
        }
        // 返回值：条数；路径「长度」为题面规定的顶点数 = 边数 + 1
        return {(int)ways[ex][ey], dist[ex][ey] + 1};
    }
};
```

```javascript
/**
 * @param {number} n - 敌军阵营的边长
 * @param {number[][]} guards - 哨兵位置列表 [x, y]
 * @return {number[]} - 返回 [最短路径条数, 最短路径长度]
 */
var countShortestPaths = function(n, guards) {
    // 起点：底行正中 (0, n/2)；终点：顶行正中 (n-1, n/2)
    const sy = n >> 1;
    const sx = 0, ex = n - 1, ey = sy;

    // 标记每个哨兵的 3×3 九宫格（切比雪夫距离 ≤ 1）为禁区
    const ban = Array.from({ length: n }, () => new Array(n).fill(false));
    for (const g of guards) {
        const gx = g[0], gy = g[1];
        for (let dx = -1; dx <= 1; dx++) {
            for (let dy = -1; dy <= 1; dy++) {
                const nx = gx + dx, ny = gy + dy;
                if (nx >= 0 && nx < n && ny >= 0 && ny < n) {
                    ban[nx][ny] = true;
                }
            }
        }
    }

    // 起点或终点在禁区内则不可达
    if (ban[sx][sy] || ban[ex][ey]) return [0, 0];

    // BFS：dist 存最短边数（-1 表示未访问），ways 存方案数
    const dist = Array.from({ length: n }, () => new Array(n).fill(-1));
    const ways = Array.from({ length: n }, () => new Array(n).fill(0));
    const dirs = [[1, 0], [-1, 0], [0, 1], [0, -1]];

    dist[sx][sy] = 0;
    ways[sx][sy] = 1;
    const q = [[sx, sy]];

    for (let qi = 0; qi < q.length; qi++) {
        const [x, y] = q[qi];
        for (const [dx, dy] of dirs) {
            const nx = x + dx, ny = y + dy;
            if (nx < 0 || nx >= n || ny < 0 || ny >= n || ban[nx][ny]) continue;
            const nd = dist[x][y] + 1;
            if (dist[nx][ny] === -1) {
                // 首次到达：最短路长度确定，方案数继承自前驱
                dist[nx][ny] = nd;
                ways[nx][ny] = ways[x][y];
                q.push([nx, ny]);
            } else if (dist[nx][ny] === nd) {
                // 同一最短路层上另有前驱，累加方案数
                ways[nx][ny] += ways[x][y];
            }
        }
    }

    // 可达时返回 [条数, 顶点数（边数 + 1）]
    if (dist[ex][ey] === -1) return [0, 0];
    return [ways[ex][ey], dist[ex][ey] + 1];
};
```

```c
#include <stdlib.h>
#include <string.h>

/**
 * @param n 敌军阵营的边长
 * @param guards 哨兵位置数组
 * @param guardsSize 哨兵数量
 * @param guardsColSize 每个哨兵坐标子数组的长度（恒为2）
 * @param returnSize 返回数组长度（恒为2）
 * @return [最短路径条数, 最短路径长度]
 */
int* countShortestPaths(int n, int** guards, int guardsSize, int* guardsColSize, int* returnSize) {
    *returnSize = 2;
    int* result = (int*)malloc(2 * sizeof(int));
    result[0] = 0;
    result[1] = 0;

    // 起点：底行正中 (0, n/2)；终点：顶行正中 (n-1, n/2)
    int sy = n / 2;
    int sx = 0, ex = n - 1, ey = sy;

    // 标记每个哨兵的 3×3 九宫格（切比雪夫距离 ≤ 1）为禁区
    char** ban = (char**)malloc(n * sizeof(char*));
    for (int i = 0; i < n; i++) {
        ban[i] = (char*)calloc(n, sizeof(char));
    }
    for (int i = 0; i < guardsSize; i++) {
        int gx = guards[i][0], gy = guards[i][1];
        for (int dx = -1; dx <= 1; dx++) {
            for (int dy = -1; dy <= 1; dy++) {
                int nx = gx + dx, ny = gy + dy;
                if (nx >= 0 && nx < n && ny >= 0 && ny < n) {
                    ban[nx][ny] = 1;
                }
            }
        }
    }

    // 起点或终点在禁区内则不可达
    if (ban[sx][sy] || ban[ex][ey]) {
        for (int i = 0; i < n; i++) free(ban[i]);
        free(ban);
        return result;
    }

    // BFS：dist 存最短边数（-1 表示未访问），ways 存方案数
    int** dist = (int**)malloc(n * sizeof(int*));
    long long** ways = (long long**)malloc(n * sizeof(long long*));
    for (int i = 0; i < n; i++) {
        dist[i] = (int*)malloc(n * sizeof(int));
        ways[i] = (long long*)malloc(n * sizeof(long long));
        for (int j = 0; j < n; j++) {
            dist[i][j] = -1;
            ways[i][j] = 0;
        }
    }

    int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};  // 上下左右
    dist[sx][sy] = 0;
    ways[sx][sy] = 1;

    // 数组模拟队列
    int** q = (int**)malloc(n * n * sizeof(int*));
    int head = 0, tail = 0;
    q[tail] = (int*)malloc(2 * sizeof(int));
    q[tail][0] = sx; q[tail][1] = sy;
    tail++;

    while (head < tail) {
        int x = q[head][0], y = q[head][1];
        free(q[head]);
        head++;
        for (int k = 0; k < 4; k++) {
            int nx = x + dirs[k][0], ny = y + dirs[k][1];
            if (nx < 0 || nx >= n || ny < 0 || ny >= n || ban[nx][ny]) continue;
            int nd = dist[x][y] + 1;
            if (dist[nx][ny] == -1) {
                // 首次到达：最短路长度确定，方案数继承自前驱
                dist[nx][ny] = nd;
                ways[nx][ny] = ways[x][y];
                q[tail] = (int*)malloc(2 * sizeof(int));
                q[tail][0] = nx; q[tail][1] = ny;
                tail++;
            } else if (dist[nx][ny] == nd) {
                // 同一最短路层上另有前驱，累加方案数
                ways[nx][ny] += ways[x][y];
            }
        }
    }

    // 可达时填写结果：条数 + 顶点数（边数 + 1）
    if (dist[ex][ey] != -1) {
        result[0] = (int)ways[ex][ey];
        result[1] = dist[ex][ey] + 1;
    }

    // 释放内存
    for (int i = 0; i < n; i++) {
        free(ban[i]);
        free(dist[i]);
        free(ways[i]);
    }
    free(ban);
    free(dist);
    free(ways);
    free(q);

    return result;
}
```
