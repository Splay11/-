## 解题思路

本题的关键是：**每个参与者只按照初始队列顺序处理一次**，并且移动到队首或队尾的人，其相对顺序要和初始队列保持一致。

因此不需要真的在链表中反复移动节点，可以使用**分组模拟 + 顺序拼接**的方法。

按照规则优先级处理每个节点：

1. 如果号码不在 `[0, 10^9]` 范围内，直接返回 `"-1"`。
2. 如果号码是 `3` 的倍数，淘汰，不加入结果。
3. 否则，如果十进制表示中包含数字 `3`，加入 `back` 分组，表示最终移动到队尾。
4. 否则，如果十进制表示中包含数字 `2`，加入 `front` 分组，表示最终移动到队首。
5. 否则加入 `middle` 分组，表示位置相对保持不变。

由于题目要求移动到队首、队尾的元素相对顺序都保持初始顺序，所以最终答案为：

```text
front + middle + back
```

如果最终没有剩余数字，返回 `"-1"`。

本题主要使用的数据结构是**数组/列表**，本质是一次线性扫描的**模拟算法**。

## 2. 复杂度分析

设链表长度为 `m`。

每个节点只遍历一次。判断数字是否包含 `2` 或 `3` 时，最多检查十进制的 10 位数字，因为 `n <= 10^9`。

因此：

```text
时间复杂度：O(m)
```

虽然每个数字需要检查数位，但位数是常数级，所以整体仍为线性复杂度。

需要额外使用三个列表保存最终分组：

```text
空间复杂度：O(m)
```

```python
# LeetCode 通常会预定义 ListNode
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next

class Solution:
    def gameResult(self, head):
        front = []   # 包含数字 2，最终移动到队首
        middle = []  # 既不淘汰，也不移动
        back = []    # 包含数字 3，最终移动到队尾

        cur = head
        while cur:
            x = cur.val

            # 非法数字，直接返回 -1
            if x < 0 or x > 1000000000:
                return "-1"

            # 规则 1：3 的倍数直接淘汰
            if x % 3 == 0:
                cur = cur.next
                continue

            s = str(x)

            # 规则 2 优先于规则 3
            if '3' in s:
                back.append(str(x))
            elif '2' in s:
                front.append(str(x))
            else:
                middle.append(str(x))

            cur = cur.next

        ans = front + middle + back

        # 没有剩余参与者时返回 -1
        if not ans:
            return "-1"

        return " ".join(ans)
```

```java
import java.util.*;

class Solution {
    public String gameResult(ListNode head) {
        List<String> front = new ArrayList<>();   // 包含数字 2，最终移动到队首
        List<String> middle = new ArrayList<>();  // 既不淘汰，也不移动
        List<String> back = new ArrayList<>();    // 包含数字 3，最终移动到队尾

        ListNode cur = head;

        while (cur != null) {
            long x = cur.val; // ListNode.val 是 long，不能用 int 接收

            // 非法数字，直接返回 -1
            if (x < 0 || x > 1000000000L) {
                return "-1";
            }

            // 规则 1：3 的倍数直接淘汰
            if (x % 3 == 0) {
                cur = cur.next;
                continue;
            }

            // 规则 2 优先于规则 3
            if (containsDigit(x, 3)) {
                back.add(String.valueOf(x));
            } else if (containsDigit(x, 2)) {
                front.add(String.valueOf(x));
            } else {
                middle.add(String.valueOf(x));
            }

            cur = cur.next;
        }

        List<String> ans = new ArrayList<>();

        // 最终顺序：移到队首组 + 保持不变组 + 移到队尾组
        ans.addAll(front);
        ans.addAll(middle);
        ans.addAll(back);

        // 如果全部被淘汰，返回 -1
        if (ans.isEmpty()) {
            return "-1";
        }

        return String.join(" ", ans);
    }

    // 判断数字 x 的十进制表示中是否包含 digit
    private boolean containsDigit(long x, int digit) {
        if (x == 0) {
            return digit == 0;
        }

        while (x > 0) {
            if (x % 10 == digit) {
                return true;
            }
            x /= 10;
        }

        return false;
    }
}
```

```cpp
class Solution {
public:
    string gameResult(ListNode* head) {
        vector<string> front;   // 包含数字 2，最终移动到队首
        vector<string> middle;  // 既不淘汰，也不移动
        vector<string> back;    // 包含数字 3，最终移动到队尾

        while (cur != nullptr) {
            long long x = cur->val;

            // 非法数字，直接返回 -1
            if (x < 0 || x > 1000000000LL) {
                return "-1";
            }

            // 规则 1：3 的倍数直接淘汰

                continue;
            }

            // 规则 2 优先于规则 3
            if (containsDigit(x, 3)) {
                back.push_back(to_string(x));
            } else if (containsDigit(x, 2)) {
                front.push_back(to_string(x));
            } else {
                middle.push_back(to_string(x));
            }

            cur = cur->next;
        }

        vector<string> ans;

        // 按照最终顺序拼接：队首移动组 + 不变组 + 队尾移动组
        for (string s : front) ans.push_back(s);
        for (string s : middle) ans.push_back(s);
        for (string s : back) ans.push_back(s);

        // 没有剩余参与者时返回 -1
        if (ans.empty()) {
            return "-1";
        }

        string res = ans[0];
        for (int i = 1; i < ans.size(); i++) {
            res += " " + ans[i];
        }

        return res;
    }

private:
    // 判断数字 x 的十进制表示中是否包含 digit

            return digit == 0;
        }

        while (x > 0) {
            if (x % 10 == digit) {
                return true;
            }
            x /= 10;
        }

        return false;
    }
```

```javascript
var gameResult = function(head) {
    var front = [];   // 包含数字 2，最终移动到队首

            cur = cur.next;
            continue;

        var s = String(x);
        if (s.indexOf('3') !== -1) {
            back.push(s);
        } else if (s.indexOf('2') !== -1) {
            front.push(s);
        } else {
            middle.push(s);

        cur = cur.next;

    return ans.join(' ');
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

// 判断十进制表示中是否包含 digit
int containsDigit(int x, int d) {
    if (x == 0) return d == 0;
    while (x > 0) {
        if (x % 10 == d) return 1;
        x /= 10;
    }
    return 0;
}

char* gameResult(struct ListNode* head) {
    // 第一遍：统计非淘汰节点数量（同时检查非法值）
    int total = 0;
    struct ListNode* cur = head;
    while (cur) {
        int x = cur->val;
        if (x < 0 || x > 1000000000) {
            char* r = (char*)malloc(3);
            strcpy(r, "-1");
            return r;
        }
        if (x % 3 != 0) total++;
        cur = cur->next;
    }

    // 无剩余
    if (total == 0) {
        char* r = (char*)malloc(3);
        strcpy(r, "-1");
        return r;
    }

    // 动态分配分组数组
    int* front = (int*)malloc(total * sizeof(int));
    int* middle = (int*)malloc(total * sizeof(int));
    int* back = (int*)malloc(total * sizeof(int));
    int fCnt = 0, mCnt = 0, bCnt = 0;

    // 第二遍：分组
    cur = head;
    while (cur) {
        int x = cur->val;
        if (x % 3 == 0) {
            cur = cur->next;
            continue;
        }
        if (containsDigit(x, 3)) {
            back[bCnt++] = x;
        } else if (containsDigit(x, 2)) {
            front[fCnt++] = x;
        } else {
            middle[mCnt++] = x;
        }
        cur = cur->next;
    }

    // 拼接结果字符串
    int cap = total * 12;
    char* result = (char*)malloc(cap + 1);
    result[0] = '\0';

    int first = 1;
    for (int i = 0; i < fCnt; i++) {
        if (!first) strcat(result, " ");
        char buf[16];
        sprintf(buf, "%d", front[i]);
        strcat(result, buf);
        first = 0;
    }
    for (int i = 0; i < mCnt; i++) {
        if (!first) strcat(result, " ");
        char buf[16];
        sprintf(buf, "%d", middle[i]);
        strcat(result, buf);
        first = 0;
    }
    for (int i = 0; i < bCnt; i++) {
        if (!first) strcat(result, " ");
        char buf[16];
        sprintf(buf, "%d", back[i]);
        strcat(result, buf);
        first = 0;
    }

    free(front);
    free(middle);
    free(back);
    return result;
}
```
