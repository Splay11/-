## 解题思路

本题要求在**环形数组**中选择一段连续子数组，长度至少为 `1`，最多为 `n`，并且子数组和能被 `k` 整除，同时子数组和尽可能大。

环形数组问题可以转化为普通数组问题：

将数组复制一遍，得到长度为 `2n` 的新数组。
环形上的任意一段连续魔法石，都可以看成这个新数组中的一段连续子数组，但长度不能超过 `n`。

问题变成：

> 在长度为 `2n` 的数组中，找一个长度在 `[1, n]` 之间的连续子数组，使其和能被 `k` 整除，并且和最大。

使用算法：

**前缀和 + 同余分组 + 单调队列**

设前缀和为 `prefix`。

如果一段子数组 `[l, r]` 的和能被 `k` 整除，则有：

```text
prefix[r] - prefix[l - 1] 能被 k 整除
```

也就是：

```text
prefix[r] % k == prefix[l - 1] % k
```

所以对于当前右端点 `r`，我们只需要在之前的前缀和中，找到一个与当前前缀和余数相同的最小前缀和。

因为：

```text
子数组和 = 当前前缀和 - 之前前缀和
```

想让子数组和最大，就要让之前那个前缀和尽可能小。

同时，子数组长度不能超过 `n`，所以之前的前缀下标必须在滑动窗口范围内。

因此，对每个余数维护一个单调递增队列：

* 队列中存储前缀和下标和前缀和值
* 队首始终是当前窗口内该余数对应的最小前缀和
* 每次移动右端点时，删除超过长度限制的旧前缀和
* 用当前前缀和减去队首前缀和更新答案
* 再把当前前缀和加入对应余数的单调队列

如果最终没有找到任何合法子数组，则返回 `0`。

需要注意：

* 数组中有负数，不能用普通双指针。
* 前缀和可能很大，Java 和 C++ 要使用 `long` / `long long`。
* 取模时要处理负数前缀和。

## 复杂度分析

时间复杂度：`O(n)`

虽然遍历的是复制后的数组，长度为 `2n`，但每个前缀和最多入队一次、出队一次，所以总时间复杂度仍然是 `O(n)`。

空间复杂度：`O(n + k)`

最多维护 `k` 个单调队列，所有队列中元素总数不会超过窗口大小级别，因此空间复杂度为 `O(n)`。

```python
from typing import List
from collections import deque

class Solution:
    def maxEnergyDivisibleByK(self, nums: List[int], n: int, k: int) -> int:
        # 每个余数维护一个单调队列，存储二元组：(前缀下标, 前缀和)
        queues = [deque() for _ in range(k)]

        # 初始前缀和为 0，下标为 0，余数为 0
        queues[0].append((0, 0))

        prefix = 0
        ans = -10**30
        found = False

        # 遍历复制后的数组，下标 r 表示当前前缀和位置
        for r in range(1, 2 * n + 1):
            prefix += nums[(r - 1) % n]
            # 处理负数取模
            mod = prefix % k

            q = queues[mod]

            # 删除长度超过 n 的旧前缀和
            while q and q[0][0] < r - n:
                q.popleft()

            # 队首是当前窗口内同余数的最小前缀和
            if q:
                ans = max(ans, prefix - q[0][1])
                found = True

            # 将当前前缀和加入对应余数的单调队列
            # 保持队列中的前缀和值单调递增
            while q and q[-1][1] >= prefix:
                q.pop()

            q.append((r, prefix))

        return ans if found else 0
```

```java
import java.util.*;

class Solution {
    public long maxEnergyDivisibleByK(int[] nums, int n, int k) {
        // 每个余数维护一个单调队列，队列元素为 long[]{前缀下标, 前缀和}
        ArrayDeque<long[]>[] queues = new ArrayDeque[k];

        for (int i = 0; i < k; i++) {
            queues[i] = new ArrayDeque<>();
        }

        // 初始前缀和为 0，下标为 0，余数为 0
        queues[0].offerLast(new long[]{0, 0});

        long prefix = 0;
        long ans = Long.MIN_VALUE;
        boolean found = false;

        // 遍历复制后的数组
        for (int r = 1; r <= 2 * n; r++) {
            prefix += nums[(r - 1) % n];

            int mod = getMod(prefix, k);
            ArrayDeque<long[]> q = queues[mod];

            // 删除长度超过 n 的旧前缀和
            while (!q.isEmpty() && q.peekFirst()[0] < r - n) {
                q.pollFirst();
            }

            // 队首是当前窗口内同余数的最小前缀和
            if (!q.isEmpty()) {
                ans = Math.max(ans, prefix - q.peekFirst()[1]);
                found = true;
            }

            // 将当前前缀和加入对应余数的单调队列
            // 保持队列中的前缀和值单调递增
            while (!q.isEmpty() && q.peekLast()[1] >= prefix) {
                q.pollLast();
            }

            q.offerLast(new long[]{r, prefix});
        }

        return found ? ans : 0;
    }

    // 处理负数取模
    private int getMod(long x, int k) {
        int res = (int)(x % k);
        if (res < 0) {
            res += k;
        }
        return res;
    }
}
```

```cpp
#include <vector>
#include <deque>
#include <climits>
using namespace std;

class Solution {
public:
    long long maxEnergyDivisibleByK(vector<int>& nums, int n, int k) {
        // 每个余数维护一个单调队列，存储 pair<前缀下标, 前缀和>
        vector<deque<pair<int, long long>>> queues(k);

        // 初始前缀和为 0，下标为 0，余数为 0
        queues[0].push_back({0, 0});

        long long prefix = 0;
        long long ans = LLONG_MIN;
        bool found = false;

        // 处理负数取模
        auto getMod = [&](long long x) {
            int res = x % k;
            if (res < 0) {
                res += k;
            }
            return res;
        };

        // 遍历复制后的数组
        for (int r = 1; r <= 2 * n; r++) {
            prefix += nums[(r - 1) % n];

            int mod = getMod(prefix);
            auto& q = queues[mod];

            // 删除长度超过 n 的旧前缀和
            while (!q.empty() && q.front().first < r - n) {
                q.pop_front();
            }

            // 队首是当前窗口内同余数的最小前缀和
            if (!q.empty()) {
                ans = max(ans, prefix - q.front().second);
                found = true;
            }

            // 将当前前缀和加入对应余数的单调队列
            // 保持队列中的前缀和值单调递增
            while (!q.empty() && q.back().second >= prefix) {
                q.pop_back();
            }

            q.push_back({r, prefix});
        }

        return found ? ans : 0;
    }
};
```

```javascript
/**
 * 环形数组最大能被 k 整除的子数组和
 * 算法：前缀和 + 同余分组 + 单调队列
 * 时间复杂度 O(n)，空间复杂度 O(n + k)
 *
 * @param {number[]} nums - 魔法石的能量值数组
 * @param {number} n - 魔法石数量
 * @param {number} k - 除数
 * @return {number} - 能被 k 整除的最大总能量值，若不存在则返回 0
 */
var maxEnergyDivisibleByK = function(nums, n, k) {
    // 每个余数维护一个单调队列，队列元素为 [前缀下标, 前缀和]
    const queues = Array.from({ length: k }, () => []);

    // 初始：前缀下标 0，前缀和 0，余数 0
    queues[0].push([0, 0]);

    let prefix = 0;
    let ans = -Infinity;
    let found = false;

    // 遍历复制后的数组（2n 长度），r 表示前缀和位置
    for (let r = 1; r <= 2 * n; r++) {
        prefix += nums[(r - 1) % n];
        // 处理负数取模
        const mod = ((prefix % k) + k) % k;

        const q = queues[mod];

        // 删除长度超过 n 的旧前缀和
        while (q.length > 0 && q[0][0] < r - n) {
            q.shift();
        }

        // 队首是当前窗口内同余数的最小前缀和
        if (q.length > 0) {
            ans = Math.max(ans, prefix - q[0][1]);
            found = true;
        }

        // 将当前前缀和加入单调队列，保持前缀和值单调递增
        while (q.length > 0 && q[q.length - 1][1] >= prefix) {
            q.pop();
        }
        q.push([r, prefix]);
    }

    return found ? ans : 0;
};
```

```c
/**
 * 环形数组最大能被 k 整除的子数组和
 * 算法：前缀和 + 同余分组 + 单调队列
 * 时间复杂度 O(n)，空间复杂度 O(n + k)
 *
 * @param nums 魔法石能量值数组
 * @param n 魔法石数量
 * @param k 除数
 * @return 能被 k 整除的最大总能量值；若不存在则返回 0
 */
long long maxEnergyDivisibleByK(int* nums, int n, int k) {
    // 为每个余数维护一个单调队列，用数组模拟队列
    int* q_head = (int*)calloc(k, sizeof(int));
    int* q_tail = (int*)calloc(k, sizeof(int));
    int** q_idx = (int**)malloc(k * sizeof(int*));
    long long** q_val = (long long**)malloc(k * sizeof(long long*));
    for (int i = 0; i < k; i++) {
        q_idx[i] = (int*)malloc((2 * n + 5) * sizeof(int));
        q_val[i] = (long long*)malloc((2 * n + 5) * sizeof(long long));
    }

    // 初始：前缀下标 0，前缀和 0，余数 0
    q_val[0][q_tail[0]] = 0;
    q_idx[0][q_tail[0]] = 0;
    q_tail[0]++;

    long long prefix = 0;
    long long ans = -1000000000000000000LL;
    int found = 0;

    // 遍历复制后的数组（2n 长度），r 表示前缀和位置
    for (int r = 1; r <= 2 * n; r++) {
        prefix += nums[(r - 1) % n];
        // 处理负数取模
        int mod = (int)(prefix % k);
        if (mod < 0) mod += k;

        // 删除长度超过 n 的旧前缀和
        while (q_head[mod] < q_tail[mod] && q_idx[mod][q_head[mod]] < r - n) {
            q_head[mod]++;
        }

        // 队首是当前窗口内同余数的最小前缀和
        if (q_head[mod] < q_tail[mod]) {
            long long cur = prefix - q_val[mod][q_head[mod]];
            if (cur > ans) ans = cur;
            found = 1;
        }

        // 将当前前缀和加入单调队列，保持前缀和值单调递增
        while (q_head[mod] < q_tail[mod] && q_val[mod][q_tail[mod] - 1] >= prefix) {
            q_tail[mod]--;
        }
        q_val[mod][q_tail[mod]] = prefix;
        q_idx[mod][q_tail[mod]] = r;
        q_tail[mod]++;
    }

    // 释放动态分配的内存
    for (int i = 0; i < k; i++) {
        free(q_idx[i]);
        free(q_val[i]);
    }
    free(q_idx);
    free(q_val);
    free(q_head);
    free(q_tail);

    return found ? ans : 0;
}
```
