## 解题思路

题目要求在字符串中提取所有连续数字串对应的整数，并找出其中能被给定正整数整除的最大值。

需要注意以下规则：

1. 字符串只允许包含英文大小写字母和数字，出现其他字符直接返回 $-1$。
2. 字符串长度必须在 $1$ 到 $10000$ 之间。
3. 除数必须在 $1$ 到 $99$ 之间。
4. 连续数字串不可拆分，例如 `2222` 只能作为整数 `2222` 判断，不能拆成 `222` 和 `2`。
5. 数字串允许有前导零，例如 `064` 解析为 `64`，`000003` 解析为 `3`，`0000` 解析为 `0`。
6. 每个解析出的整数数值必须在 $0$ 到 $999$ 之间，超过范围说明输入不合法，返回 $-1$。
7. 如果没有找到任何能被除数整除的整数，返回 $-1$。

做法是从左到右扫描字符串：

- 遇到字母，继续扫描。
- 遇到非法字符，直接返回 $-1$。
- 遇到数字，读取完整的连续数字串。
- 对数字串忽略前导零后计算数值。
- 如果有效数字部分超过三位，或者最终值大于 $999$，说明该整数超出范围，返回 $-1$。
- 若该数能被 `inputDivisor` 整除，则更新答案最大值。

## 复杂度分析

设字符串长度为 $n$。

时间复杂度：$O(n)$。

每个字符最多被扫描常数次。

空间复杂度：$O(1)$。

除输入字符串外，只使用了有限个变量。

```python
class Solution:
    def getMaxDivisibleNumber(self, inputStr, inputDivisor):
        # 校验参数类型，防止模板解析异常后传入非法类型
        if not isinstance(inputStr, str):
            return -1
        if not isinstance(inputDivisor, int) or isinstance(inputDivisor, bool):
            return -1

        n = len(inputStr)

        # 校验字符串长度和除数范围
        if n < 1 or n > 10000:
            return -1
        if inputDivisor < 1 or inputDivisor > 99:
            return -1

        answer = -1
        i = 0

        while i < n:
            ch = inputStr[i]

            # 判断当前字符是否合法：只能是英文大小写字母或数字
            is_digit = '0' <= ch <= '9'
            is_letter = ('a' <= ch <= 'z') or ('A' <= ch <= 'Z')

            if not is_digit and not is_letter:
                return -1

            # 如果当前字符是字母，直接继续
            if not is_digit:
                i += 1
                continue

            # 当前字符是数字，读取完整的连续数字串
            value = 0
            significant_count = 0

            while i < n and '0' <= inputStr[i] <= '9':
                digit = ord(inputStr[i]) - ord('0')

                # 忽略前导零；一旦出现非零数字，后续零也属于有效位
                if digit != 0 or significant_count > 0:
                    significant_count += 1
                    value = value * 10 + digit

                    # 数值范围要求为 0 到 999，超过三位有效数字一定非法
                    if significant_count > 3 or value > 999:
                        return -1

                i += 1

            # 全是前导零的数字串，例如 0、00、000，解析值为 0
            if value % inputDivisor == 0:
                answer = max(answer, value)

        return answer
```

```java
class Solution {
    public int getMaxDivisibleNumber(String inputStr, int inputDivisor) {
        // 校验空字符串对象
        if (inputStr == null) {
            return -1;
        }

        int n = inputStr.length();

        // 校验字符串长度和除数范围
        if (n < 1 || n > 10000) {
            return -1;
        }
        if (inputDivisor < 1 || inputDivisor > 99) {
            return -1;
        }

        int answer = -1;
        int i = 0;

        while (i < n) {
            char ch = inputStr.charAt(i);

            // 判断当前字符是否合法：只能是英文大小写字母或数字
            if (!isDigit(ch) && !isLetter(ch)) {
                return -1;
            }

            // 如果当前字符是字母，直接继续
            if (!isDigit(ch)) {
                i++;
                continue;
            }

            // 当前字符是数字，读取完整的连续数字串
            int value = 0;
            int significantCount = 0;

            while (i < n && isDigit(inputStr.charAt(i))) {
                int digit = inputStr.charAt(i) - '0';

                // 忽略前导零；一旦出现非零数字，后续零也属于有效位
                if (digit != 0 || significantCount > 0) {
                    significantCount++;
                    value = value * 10 + digit;

                    // 数值范围要求为 0 到 999，超过三位有效数字一定非法
                    if (significantCount > 3 || value > 999) {
                        return -1;
                    }
                }

                i++;
            }

            // 全是前导零的数字串，例如 0、00、000，解析值为 0
            if (value % inputDivisor == 0) {
                answer = Math.max(answer, value);
            }
        }

        return answer;
    }

    private boolean isDigit(char ch) {
        return ch >= '0' && ch <= '9';
    }

    private boolean isLetter(char ch) {
        return (ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z');
    }
}
```

```javascript
/**
 * @param {string} inputStr
 * @param {number} inputDivisor
 * @return {number}
 */
var getMaxDivisibleNumber = function(inputStr, inputDivisor) {
    // 校验输入合法性
    if (typeof inputStr !== 'string' || inputStr.length < 1 || inputStr.length > 10000) {
        return -1;
    }
    if (!Number.isInteger(inputDivisor) || inputDivisor < 1 || inputDivisor > 99) {
        return -1;
    }

    // 检查是否有非法字符，只允许 A-Z、a-z、0-9
    if (!/^[A-Za-z0-9]+$/.test(inputStr)) {
        return -1;
    }

    // 提取所有连续数字串
    const matches = inputStr.match(/\d+/g);
    if (!matches) return -1;

    let maxDivisible = -1;

    for (const numStr of matches) {
        // 解析为整数
        const val = parseInt(numStr, 10);

        // 检查数值范围是否合法
        if (val < 0 || val > 999) {
            return -1;
        }

        // 判断能否被除数整除，更新最大值
        if (val % inputDivisor === 0) {
            if (val > maxDivisible) {
                maxDivisible = val;
            }
        }
    }

    return maxDivisible;
};
```

```c
#include <string.h>
#include <stdbool.h>

int getMaxDivisibleNumber(char* inputStr, int inputDivisor) {
    int n = (int)strlen(inputStr);
    // 校验字符串长度和除数范围
    if (n < 1 || n > 10000) {
        return -1;
    }
    if (inputDivisor < 1 || inputDivisor > 99) {
        return -1;
    }

    int answer = -1;
    int i = 0;

    while (i < n) {
        char ch = inputStr[i];

        // 合法字符校验：仅限字母或数字
        if (!((ch >= '0' && ch <= '9') ||
              (ch >= 'a' && ch <= 'z') ||
              (ch >= 'A' && ch <= 'Z'))) {
            return -1; // 出现非法字符
        }

        // 字母直接跳过
        if (ch < '0' || ch > '9') {
            i++;
            continue;
        }

        // 数字串开始，读取完整连续数字
        int value = 0;
        int significantCount = 0; // 有效位数（不含前导零）
        bool hasNonZero = false;

        while (i < n && inputStr[i] >= '0' && inputStr[i] <= '9') {
            int digit = inputStr[i] - '0';
            if (digit != 0) {
                hasNonZero = true;
                significantCount++;
                value = value * 10 + digit;
                // 超过范围视为非法
                if (significantCount > 3 || value > 999) {
                    return -1;
                }
            } else {
                // 非前导零才累计有效位数
                if (hasNonZero) {
                    significantCount++;
                    value = value * 10;
                    if (significantCount > 3 || value > 999) {
                        return -1;
                    }
                }
            }
            i++;
        }

        // 全是前导零，值为 0
        if (!hasNonZero) {
            value = 0;
        }

        // 能被整除则更新答案
        if (value % inputDivisor == 0) {
            if (value > answer) {
                answer = value;
            }
        }
    }

    return answer;
}
```
