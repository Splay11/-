## 解题思路

把用户和关注关系看成一张有向图。关系 `[a,b]` 表示从用户 `a` 可以沿一条有向边到达用户 `b`。

需要查询从 `myId` 出发，最短距离不超过 `maxHop` 的用户，并筛选出与 `myId` 至少有一个共同兴趣的用户。

处理步骤如下：

1. 先用 `nodes` 建立用户到兴趣集合的映射。
2. 根据 `relations` 建立有向邻接表。
   - 如果一条关系中的任意一端用户不存在，直接忽略。
3. 从 `myId` 开始做广度优先搜索，得到每个可达用户的最短距离。
   - `maxHop` 如果大于 `100`，按 `100` 处理。
   - BFS 天然保证第一次访问某个点时得到的是最短距离。
4. 遍历所有可达用户：
   - 排除 `myId` 自己。
   - 计算它和 `myId` 的兴趣交集。
   - 如果交集非空，则加入答案。
   - 交集内部按 ASCII 码序从小到大排序。
5. 最后将答案按：
   - 最短距离从小到大；
   - 距离相同按用户 ID 的整数值从小到大；
   
   进行排序。

## 复杂度分析

设用户数为 $n$，关注关系数为 $m$，每个用户兴趣数量上限为 $c$，其中题目给出 $c \le 10$。

建图需要 $O(n + m)$。

BFS 最多访问所有用户和所有有效边，复杂度为 $O(n + m)$。

计算兴趣交集时，每个用户最多处理 $c$ 个兴趣，因此复杂度为 $O(nc)$。

最终对满足条件的用户排序，若结果数量为 $r$，复杂度为 $O(r \log r)$。

总时间复杂度为：

$O(n + m + nc + r \log r)$

由于 $c \le 10$，也可以近似看作：

$O(n + m + r \log r)$

空间复杂度为：

$O(n + m)$

```python
from collections import deque

class Solution:
    def queryFriends(self, nodes, relations, myId, maxHop):
        # 最大跳数按题意限制到 100
        hop_limit = min(int(maxHop), 100)

        # 建立用户 ID 到兴趣集合的映射
        interests = {}
        for row in nodes:
            if not row:
                continue
            uid = row[0]
            interests[uid] = set(row[1:])

        # 起始用户一定存在，直接取出它的兴趣集合
        my_interests = interests[myId]

        # 初始化邻接表，只给存在的用户建表
        graph = {uid: [] for uid in interests}

        # 建立有向图，非法关系直接忽略
        for edge in relations:
            if len(edge) < 2:
                continue
            a, b = edge[0], edge[1]
            if a in interests and b in interests:
                graph[a].append(b)

        # BFS 计算从 myId 到其他用户的最短距离
        dist = {myId: 0}
        q = deque([myId])

        while q:
            cur = q.popleft()
            cur_dist = dist[cur]

            # 已经达到最大跳数时，不再继续向外扩展
            if cur_dist == hop_limit:
                continue

            for nxt in graph.get(cur, []):
                if nxt not in dist:
                    dist[nxt] = cur_dist + 1
                    q.append(nxt)

        items = []

        # 遍历所有可达用户，筛选有共同兴趣的用户
        for uid, d in dist.items():
            if uid == myId:
                continue

            # 计算共同兴趣，并按 ASCII 码序排序
            common = sorted(my_interests & interests[uid])
            if not common:
                continue

            # 保存排序键和输出行
            items.append((d, int(uid), [uid] + common))

        # 先按最短距离排序，再按 ID 整数值排序
        items.sort(key=lambda x: (x[0], x[1]))

        # 只返回题目要求的二维数组
        return [row for _, _, row in items]
```

```java
class Solution {
    public java.util.List<java.util.List<String>> queryFriends(
            java.util.List<java.util.List<String>> nodes,
            java.util.List<java.util.List<String>> relations,
            String myId,
            int maxHop) {

        // 最大跳数按题意限制到 100
        int hopLimit = Math.min(maxHop, 100);

        // 建立用户 ID 到兴趣集合的映射
        java.util.Map<String, java.util.Set<String>> interests = new java.util.HashMap<>();
        for (java.util.List<String> row : nodes) {
            if (row.isEmpty()) {
                continue;
            }

            String uid = row.get(0);
            java.util.Set<String> set = new java.util.HashSet<>();

            for (int i = 1; i < row.size(); i++) {
                set.add(row.get(i));
            }

            interests.put(uid, set);
        }

        // 起始用户一定存在，直接取出它的兴趣集合
        java.util.Set<String> myInterests = interests.get(myId);

        // 初始化邻接表，只给存在的用户建表
        java.util.Map<String, java.util.List<String>> graph = new java.util.HashMap<>();
        for (String uid : interests.keySet()) {
            graph.put(uid, new java.util.ArrayList<>());
        }

        // 建立有向图，非法关系直接忽略
        for (java.util.List<String> edge : relations) {
            if (edge.size() < 2) {
                continue;
            }

            String a = edge.get(0);
            String b = edge.get(1);

            if (interests.containsKey(a) && interests.containsKey(b)) {
                graph.get(a).add(b);
            }
        }

        // BFS 计算从 myId 到其他用户的最短距离
        java.util.Map<String, Integer> dist = new java.util.HashMap<>();
        java.util.Queue<String> queue = new java.util.ArrayDeque<>();

        dist.put(myId, 0);
        queue.offer(myId);

        while (!queue.isEmpty()) {
            String cur = queue.poll();
            int curDist = dist.get(cur);

            // 已经达到最大跳数时，不再继续向外扩展
            if (curDist == hopLimit) {
                continue;
            }

            for (String nxt : graph.get(cur)) {
                if (!dist.containsKey(nxt)) {
                    dist.put(nxt, curDist + 1);
                    queue.offer(nxt);
                }
            }
        }

        // 用内部类保存排序键和输出行
        class Item {
            int distance;
            int id;
            java.util.List<String> row;

            Item(int distance, int id, java.util.List<String> row) {
                this.distance = distance;
                this.id = id;
                this.row = row;
            }
        }

        java.util.List<Item> items = new java.util.ArrayList<>();

        // 遍历所有可达用户，筛选有共同兴趣的用户
        for (java.util.Map.Entry<String, Integer> entry : dist.entrySet()) {
            String uid = entry.getKey();

            if (uid.equals(myId)) {
                continue;
            }

            // 计算共同兴趣
            java.util.List<String> common = new java.util.ArrayList<>();
            for (String tag : myInterests) {
                if (interests.get(uid).contains(tag)) {
                    common.add(tag);
                }
            }

            if (common.isEmpty()) {
                continue;
            }

            // 共同兴趣按 ASCII 码序排序
            java.util.Collections.sort(common);

            java.util.List<String> row = new java.util.ArrayList<>();
            row.add(uid);
            row.addAll(common);

            items.add(new Item(entry.getValue(), Integer.parseInt(uid), row));
        }

        // 先按最短距离排序，再按 ID 整数值排序
        java.util.Collections.sort(items, (a, b) -> {
            if (a.distance != b.distance) {
                return Integer.compare(a.distance, b.distance);
            }
            return Integer.compare(a.id, b.id);
        });

        // 只返回题目要求的二维数组
        java.util.List<java.util.List<String>> answer = new java.util.ArrayList<>();
        for (Item item : items) {
            answer.add(item.row);
        }

        return answer;
    }
}
```

```cpp
class Solution {
public:
    vector<vector<string>> queryFriends(vector<vector<string>>& nodes,
                                        vector<vector<string>>& relations,
                                        string myId,
                                        int maxHop) {
        // 最大跳数按题意限制到 100
        int hopLimit = min(maxHop, 100);

        // 建立用户 ID 到兴趣集合的映射
        unordered_map<string, unordered_set<string>> interests;
        for (auto& row : nodes) {
            if (row.empty()) {
                continue;
            }

            string uid = row[0];
            unordered_set<string> set;

            for (int i = 1; i < (int)row.size(); i++) {
                set.insert(row[i]);
            }

            interests[uid] = set;
        }

        // 起始用户一定存在，直接取出它的兴趣集合
        unordered_set<string>& myInterests = interests[myId];

        // 初始化邻接表，只给存在的用户建表
        unordered_map<string, vector<string>> graph;
        for (auto& kv : interests) {
            graph[kv.first] = vector<string>();
        }

        // 建立有向图，非法关系直接忽略
        for (auto& edge : relations) {
            if (edge.size() < 2) {
                continue;
            }

            string a = edge[0];
            string b = edge[1];

            if (interests.count(a) && interests.count(b)) {
                graph[a].push_back(b);
            }
        }

        // BFS 计算从 myId 到其他用户的最短距离
        unordered_map<string, int> dist;
        queue<string> q;

        dist[myId] = 0;
        q.push(myId);

        while (!q.empty()) {
            string cur = q.front();
            q.pop();

            int curDist = dist[cur];

            // 已经达到最大跳数时，不再继续向外扩展
            if (curDist == hopLimit) {
                continue;
            }

            for (const string& nxt : graph[cur]) {
                if (!dist.count(nxt)) {
                    dist[nxt] = curDist + 1;
                    q.push(nxt);
                }
            }
        }

        // 保存排序键和输出行：最短距离、整数 ID、结果行
        vector<tuple<int, int, vector<string>>> items;

        // 遍历所有可达用户，筛选有共同兴趣的用户
        for (auto& kv : dist) {
            string uid = kv.first;

            if (uid == myId) {
                continue;
            }

            // 计算共同兴趣
            vector<string> common;
            for (const string& tag : myInterests) {
                if (interests[uid].count(tag)) {
                    common.push_back(tag);
                }
            }

            if (common.empty()) {
                continue;
            }

            // 共同兴趣按 ASCII 码序排序
            sort(common.begin(), common.end());

            vector<string> row;
            row.push_back(uid);
            for (const string& tag : common) {
                row.push_back(tag);
            }

            items.push_back(make_tuple(kv.second, stoi(uid), row));
        }

        // 先按最短距离排序，再按 ID 整数值排序
        sort(items.begin(), items.end(), [](const auto& a, const auto& b) {
            if (get<0>(a) != get<0>(b)) {
                return get<0>(a) < get<0>(b);
            }
            return get<1>(a) < get<1>(b);
        });

        // 只返回题目要求的二维数组
        vector<vector<string>> answer;
        for (auto& item : items) {
            answer.push_back(get<2>(item));
        }

        return answer;
    }
};
```

```javascript
/**
 * @param {string[][]} nodes - 用户节点数据
 * @param {string[][]} relations - 关注关系列表
 * @param {string} myId - 起始用户ID
 * @param {number} maxHop - 最大跳数
 * @return {string[][]} - 满足条件的用户ID及共同兴趣列表
 */
var queryFriends = function(nodes, relations, myId, maxHop) {
    // 最大跳数限制到 100
    let hopLimit = Math.min(maxHop, 100);

    // 建立用户到兴趣集合的映射
    let interests = new Map();
    for (let row of nodes) {
        let uid = row[0];
        interests.set(uid, new Set(row.slice(1)));
    }

    // 起始用户兴趣集合
    let myInterests = interests.get(myId);

    // 建立有向邻接表
    let graph = new Map();
    for (let uid of interests.keys()) graph.set(uid, []);
    for (let edge of relations) {
        let a = edge[0], b = edge[1];
        if (interests.has(a) && interests.has(b)) {
            graph.get(a).push(b);
        }
    }

    // BFS 计算最短距离
    let dist = new Map();
    let queue = [myId];
    dist.set(myId, 0);
    let head = 0;
    while (head < queue.length) {
        let cur = queue[head++];
        let curDist = dist.get(cur);
        if (curDist === hopLimit) continue;
        for (let nxt of graph.get(cur)) {
            if (!dist.has(nxt)) {
                dist.set(nxt, curDist + 1);
                queue.push(nxt);
            }
        }
    }

    // 收集满足条件的用户
    let items = [];
    for (let [uid, d] of dist) {
        if (uid === myId) continue;

        // 计算共同兴趣
        let common = [];
        for (let tag of myInterests) {
            if (interests.get(uid).has(tag)) {
                common.push(tag);
            }
        }
        if (common.length === 0) continue;

        // 共同兴趣按 ASCII 码序排序
        common.sort();

        items.push({
            dist: d,
            id: parseInt(uid),
            row: [uid, ...common]
        });
    }

    // 按距离排序，距离相同按 ID 整数值排序
    items.sort((a, b) => a.dist !== b.dist ? a.dist - b.dist : a.id - b.id);

    return items.map(item => item.row);
};
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

// 简单整数队列
typedef struct {
    int* data;
    int front, rear, capacity;
} Queue;

static Queue* qcreate(int cap) {
    Queue* q = (Queue*)malloc(sizeof(Queue));
    q->data = (int*)malloc(cap * sizeof(int));
    q->front = q->rear = 0;
    q->capacity = cap;
    return q;
}
static void qdestroy(Queue* q) { free(q->data); free(q); }
static void qpush(Queue* q, int v) { q->data[q->rear++] = v; }
static int qpop(Queue* q) { return q->data[q->front++]; }
static bool qempty(Queue* q) { return q->front == q->rear; }

// 结果项：距离、整数ID、行起止
typedef struct { int dist, id, idx, rowStart, rowLen; } Item;
static int itemCmp(const void* a, const void* b) {
    Item* x = (Item*)a; Item* y = (Item*)b;
    if (x->dist != y->dist) return x->dist - y->dist;
    return x->id - y->id;
}

// 开放寻址哈希表：ID字符串 → 数组索引
#define HASH_CAP 20011
#define HASH_EMPTY -1
typedef struct { char* key; int value; } HashEntry;
static int hashFunc(const char* s) {
    unsigned int h = 5381;
    while (*s) h = ((h << 5) + h) + (unsigned char)(*s++);
    return h % HASH_CAP;
}
static void hashInit(HashEntry* ht) {
    for (int i = 0; i < HASH_CAP; i++) { ht[i].key = NULL; ht[i].value = HASH_EMPTY; }
}
static void hashPut(HashEntry* ht, const char* key, int value) {
    int idx = hashFunc(key);
    while (ht[idx].key != NULL && ht[idx].value != HASH_EMPTY) idx = (idx + 1) % HASH_CAP;
    ht[idx].key = (char*)key; ht[idx].value = value;
}
static int hashGet(HashEntry* ht, const char* key) {
    int idx = hashFunc(key);
    while (ht[idx].key != NULL && ht[idx].value != HASH_EMPTY) {
        if (strcmp(ht[idx].key, key) == 0) return ht[idx].value;
        idx = (idx + 1) % HASH_CAP;
    }
    return HASH_EMPTY;
}
static int strCmp(const void* a, const void* b) { return strcmp(*(const char**)a, *(const char**)b); }

char*** queryFriends(char*** nodes, int nodesSize, char*** relations, int relationsSize,
                     char* myId, int maxHop, int* returnSize) {
    int hopLimit = maxHop > 100 ? 100 : maxHop;

    // 1. 构建 ID → 索引映射
    HashEntry* idToIdx = (HashEntry*)malloc(HASH_CAP * sizeof(HashEntry));
    hashInit(idToIdx);
    int* nodeIdInt = (int*)malloc(nodesSize * sizeof(int));
    for (int i = 0; i < nodesSize; i++) { hashPut(idToIdx, nodes[i][0], i); nodeIdInt[i] = atoi(nodes[i][0]); }

    // 2. 查找起始用户索引
    int myIdx = hashGet(idToIdx, myId);
    if (myIdx == HASH_EMPTY) { *returnSize = 0; free(idToIdx); free(nodeIdInt); return NULL; }

    // 3. 统计每个节点的出度
    int* outDeg = (int*)calloc(nodesSize, sizeof(int));
    for (int i = 0; i < relationsSize; i++) {
        int aIdx = hashGet(idToIdx, relations[i][0]), bIdx = hashGet(idToIdx, relations[i][1]);
        if (aIdx != HASH_EMPTY && bIdx != HASH_EMPTY) outDeg[aIdx]++;
    }

    // 4. 按出度分配邻接表
    int** adj = (int**)malloc(nodesSize * sizeof(int*));
    int* adjSz = (int*)calloc(nodesSize, sizeof(int));
    for (int i = 0; i < nodesSize; i++) adj[i] = outDeg[i] > 0 ? (int*)malloc(outDeg[i] * sizeof(int)) : NULL;
    for (int i = 0; i < relationsSize; i++) {
        int aIdx = hashGet(idToIdx, relations[i][0]), bIdx = hashGet(idToIdx, relations[i][1]);
        if (aIdx != HASH_EMPTY && bIdx != HASH_EMPTY) adj[aIdx][adjSz[aIdx]++] = bIdx;
    }
    free(outDeg);

    // 5. BFS 计算最短距离
    int* dist = (int*)malloc(nodesSize * sizeof(int));
    bool* vis = (bool*)calloc(nodesSize, sizeof(bool));
    for (int i = 0; i < nodesSize; i++) dist[i] = -1;
    Queue* q = qcreate(nodesSize + 5);
    dist[myIdx] = 0; vis[myIdx] = true; qpush(q, myIdx);
    while (!qempty(q)) {
        int cur = qpop(q), curDist = dist[cur];
        if (curDist == hopLimit) continue;
        for (int i = 0; i < adjSz[cur]; i++) {
            int nxt = adj[cur][i];
            if (!vis[nxt]) { vis[nxt] = true; dist[nxt] = curDist + 1; qpush(q, nxt); }
        }
    }

    // 6. 统计兴趣数量
    int* intCount = (int*)malloc(nodesSize * sizeof(int));
    for (int i = 0; i < nodesSize; i++) { intCount[i] = 0; while (nodes[i][intCount[i]]) intCount[i]++; intCount[i]--; }

    // 7. 收集结果：遍历可达用户，寻找共同兴趣
    int maxRows = nodesSize * 12;
    char** stringPool = (char**)malloc(maxRows * sizeof(char*));
    int poolSize = 0;
    Item* items = (Item*)malloc(nodesSize * sizeof(Item));
    int itemCnt = 0;
    char* commonBuf[11];  // 最多10个共同兴趣

    for (int uid = 0; uid < nodesSize; uid++) {
        if (uid == myIdx || dist[uid] == -1) continue;

        // 计算共同兴趣
        int cc = 0;
        for (int i = 1; i <= intCount[myIdx] && cc < 11; i++)
            for (int j = 1; j <= intCount[uid]; j++)
                if (strcmp(nodes[myIdx][i], nodes[uid][j]) == 0) { commonBuf[cc++] = nodes[myIdx][i]; break; }
        if (cc == 0) continue;

        if (cc > 1) qsort(commonBuf, cc, sizeof(char*), strCmp);

        if (poolSize + 1 + cc > maxRows) {
            maxRows += nodesSize * 2;
            stringPool = (char**)realloc(stringPool, maxRows * sizeof(char*));
        }

        items[itemCnt].dist = dist[uid]; items[itemCnt].id = nodeIdInt[uid]; items[itemCnt].idx = uid;
        items[itemCnt].rowStart = poolSize;

        // 复制用户ID字符串
        char* idStr = nodes[uid][0];
        stringPool[poolSize] = (char*)malloc(strlen(idStr) + 1);
        strcpy(stringPool[poolSize++], idStr);

        // 复制共同兴趣
        for (int i = 0; i < cc; i++) {
            stringPool[poolSize] = (char*)malloc(strlen(commonBuf[i]) + 1);
            strcpy(stringPool[poolSize++], commonBuf[i]);
        }
        items[itemCnt].rowLen = 1 + cc;
        itemCnt++;
    }

    // 8. 排序：先按距离，再按ID整数值
    qsort(items, itemCnt, sizeof(Item), itemCmp);

    // 9. 构造返回结果
    *returnSize = itemCnt;
    char*** result = itemCnt > 0 ? (char***)malloc(itemCnt * sizeof(char**)) : NULL;
    for (int i = 0; i < itemCnt; i++) {
        int len = items[i].rowLen;
        char** row = (char**)malloc((len + 1) * sizeof(char*));
        int start = items[i].rowStart;
        for (int j = 0; j < len; j++) row[j] = stringPool[start + j];
        row[len] = NULL;
        result[i] = row;
    }

    // 10. 释放临时内存
    free(intCount);
    for (int i = 0; i < nodesSize; i++) if (adj[i]) free(adj[i]);
    free(adj); free(adjSz); free(dist); free(vis);
    qdestroy(q); free(items); free(stringPool); free(idToIdx); free(nodeIdInt);
    return result;
}
```
