## 解题思路

先把每个数据包拆成两个数组：

- `ids[i]` 表示第 `i` 个数据包的 `id`
- `priorities[i]` 表示第 `i` 个数据包的 `priority`

题目要求：对于每个长度为 $k$ 的滑动窗口，找到窗口内每个数据包右边第一个 `priority` 更高的数据包，并输出那个数据包的 `id`。如果一个窗口里没有任何位置满足条件，则跳过该窗口。

直接在每个窗口内暴力向右找，单个窗口需要 $O(k^2)$，总复杂度会偏高。这里可以先对整个数组做一次预处理：

1. 使用单调递减栈，求出每个位置右边第一个更高优先级的位置 `nextGreater[i]`
2. 再枚举每个窗口 $[l, r]$
3. 对窗口中的每个位置 `i`：
   - 如果 `nextGreater[i] != -1`
   - 并且 `nextGreater[i] <= r`
   - 说明这个“右边第一个更高优先级数据包”就在当前窗口里
   - 把对应的 `id` 记录下来
4. 当前窗口如果没有记录任何 `id`，就跳过

为什么这样判断是正确的？

因为 `nextGreater[i]` 表示在整个数组中，位置 `i` 右边第一个优先级更高的位置。  
如果这个位置已经落在窗口右边之外，那么在 `i` 和窗口右端之间一定不存在更早出现的、更高优先级的数据包，否则那个更早的位置才应该是 `nextGreater[i]`。

## 复杂度分析

设有效数据包数量为 $m$。

- 单调栈预处理 `nextGreater`：$O(m)$
- 枚举所有窗口并检查窗口内元素：$O((m-k+1)\cdot k)$
- 总时间复杂度：$O(m + (m-k+1)\cdot k)$
- 空间复杂度：$O(m)$

由于题目中 $k \le 100$，窗口内逐个检查是可行的。

```python
class Solution:
    def processPackets(self, n, k, packets):
        # 为防止 n 与实际长度不一致，取较小值
        m = min(n, len(packets))
        if k <= 0 or k > m:
            return []

        # 拆出 ids 与 priorities
        ids = [packet[0] for packet in packets[:m]]
        priorities = [packet[1] for packet in packets[:m]]

        # next_greater[i] 表示右侧第一个更高 priority 的位置下标
        next_greater = [-1] * m
        stack = []

        # 单调递减栈求 next_greater
        for i in range(m):
            while stack and priorities[stack[-1]] < priorities[i]:
                prev = stack.pop()
                next_greater[prev] = i
            stack.append(i)

        ans = []
        for start in range(m - k + 1):
            end = start + k - 1
            window = []
            for i in range(start, end + 1):
                j = next_greater[i]
                # 在窗口内才有效
                if j != -1 and j <= end:
                    window.append(ids[j])
            # 非空窗口结果才加入
            if window:
                ans.append(window)
        return ans
```

```java
class Solution {
    public java.util.List<java.util.List<Long>> processPackets(int n, int k, java.util.List<java.util.List<Long>> packets) {
        int m = Math.min(n, packets.size());
        java.util.List<java.util.List<Long>> ans = new java.util.ArrayList<>();
        if (k <= 0 || k > m) return ans;

        long[] ids = new long[m];
        long[] priorities = new long[m];
        for (int i = 0; i < m; i++) {
            ids[i] = packets.get(i).get(0);
            priorities[i] = packets.get(i).get(1);
        }

        int[] nextGreater = new int[m];
        java.util.Arrays.fill(nextGreater, -1);
        int[] stack = new int[m];
        int top = -1;

        for (int i = 0; i < m; i++) {
            while (top >= 0 && priorities[stack[top]] < priorities[i]) {
                nextGreater[stack[top--]] = i;
            }
            stack[++top] = i;
        }

        for (int start = 0; start + k <= m; start++) {
            int end = start + k - 1;
            java.util.List<Long> window = new java.util.ArrayList<>();
            for (int i = start; i <= end; i++) {
                int j = nextGreater[i];
                if (j != -1 && j <= end) {
                    window.add(ids[j]);
                }
            }
            if (!window.isEmpty()) ans.add(window);
        }
        return ans;
    }
}
```

```cpp
class Solution {
public:
    std::vector<std::vector<long long>> processPackets(int n, int k, const std::vector<std::vector<long long>>& packets) {
        int m = std::min<int>(n, packets.size());
        std::vector<std::vector<long long>> ans;
        if (k <= 0 || k > m) return ans;

        std::vector<long long> ids(m), priorities(m);
        for (int i = 0; i < m; ++i) {
            ids[i] = packets[i][0];
            priorities[i] = packets[i][1];
        }

        std::vector<int> nextGreater(m, -1);
        std::vector<int> stack;

        for (int i = 0; i < m; ++i) {
            while (!stack.empty() && priorities[stack.back()] < priorities[i]) {
                nextGreater[stack.back()] = i;
                stack.pop_back();
            }
            stack.push_back(i);
        }

        for (int start = 0; start + k <= m; ++start) {
            int end = start + k - 1;
            std::vector<long long> window;
            for (int i = start; i <= end; ++i) {
                int j = nextGreater[i];
                if (j != -1 && j <= end) window.push_back(ids[j]);
            }
            if (!window.empty()) ans.push_back(window);
        }
        return ans;
    }
```

```javascript
            nextGreater[stack.pop()] = i;
        }
        stack.push(i);

    for (let start = 0; start + k <= m; start++) {

        const window = [];
```

```c
int** processPackets(int n, int k, char** packets, int packetsSize, int* returnSize) {
    int m = (n < packetsSize) ? n : packetsSize;
    *returnSize = 0;
    if (k <= 0 || k > m) return NULL;

    int* ids = malloc(sizeof(int) * m);
    long long* priorities = malloc(sizeof(long long) * m);
    for (int i = 0; i < m; i++) {
        char* colon = strchr(packets[i], ':');
        ids[i] = atoi(packets[i]);
        priorities[i] = colon ? atoll(colon + 1) : 0;
    }

    int* nextGreater = malloc(sizeof(int) * m);
    for (int i = 0; i < m; i++) nextGreater[i] = -1;

    int* stack = malloc(sizeof(int) * m);
    int top = -1;
    for (int i = 0; i < m; i++) {
        while (top >= 0 && priorities[stack[top]] < priorities[i]) {
            nextGreater[stack[top--]] = i;
        }
        stack[++top] = i;
    }

    int maxWindows = m - k + 1;
    int** ans = malloc(sizeof(int*) * maxWindows);

    for (int start = 0; start + k <= m; start++) {
        int end = start + k - 1;
        int* window = calloc(k, sizeof(int));
        int wc = 0;
        for (int i = start; i <= end; i++) {
            int j = nextGreater[i];
            if (j != -1 && j <= end) window[wc++] = ids[j];
        }
        if (wc > 0) ans[(*returnSize)++] = window;
        else free(window);
    }

    free(stack);
    free(nextGreater);
    free(priorities);
    free(ids);
    return ans;
}
```
