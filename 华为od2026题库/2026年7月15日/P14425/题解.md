## 解题思路

本题核心是**字符串解析 + 日期合法性校验 + 去重 + 排序**，属于模拟/字符串处理类题目，不涉及复杂算法，关键在于把规则严谨地落地，避免边界错误。

整体流程：

1. **逐字符串解析**：对每个输入字符串，按优先级判定分隔符——含 `/` 按斜杠 `yyyy/mm/dd` 解析，否则含 `-` 按横杠 `yyyy-mm-dd` 解析；两者皆无（或混合）则视为非法，直接忽略。
2. **拆段校验**：用分隔符把字符串拆成 3 段。若段数不为 3（例如混合分隔符 `2021/02-20` 拆不出 3 段），或某段含非数字字符，均判非法。
3. **日期合法性**：校验年份 `1000~2100`、月份 `1~12`、日不超过当月天数（平年 2 月 28 天，闰年 2 月 29 天）。闰年规则：能被 4 整除且不能被 100 整除，或能被 400 整除。
4. **统一格式 + 去重**：把合法日期格式化为 `yyyy-mm-dd`（月/日补前导零），用集合去重（同一日期只保留一次）。
5. **排序输出**：由于格式定宽（`yyyy-mm-dd` 每位固定），**字典序即日期递增序**，直接对字符串排序即可。若去重后为空，返回仅含 `"NULL"` 的数组。

> 易错点：混合分隔符（`2021/02-20`）必须判非法；`2021/02/29` 在平年非法；年份边界 `0999`/`2101` 非法；全部非法要返回 `["NULL"]` 而非空数组。

## 复杂度分析

设输入数组长度为 $n$，单个字符串长度上界为常数（题目给定 $\le 50$）。

- **时间复杂度**：$O(n \log n)$。解析与校验每个字符串为 $O(1)$（长度有常数上界），共 $O(n)$；去重后最多 $n$ 个日期，排序为 $O(n \log n)$。整体由排序主导。
- **空间复杂度**：$O(n)$。需要存储去重集合与结果列表，最坏情况下每个输入都是互不相同的合法日期。

该复杂度完全满足题目数据范围（$n \le 200$），即使取最大规模也毫无压力。

from typing import List


class Solution:
    def normalizeDates(self, dates: List[str]) -> List[str]:
        # 闰年判定：能被 4 整除且不能被 100 整除，或能被 400 整除
        def is_leap(y: int) -> bool:
            return (y % 4 == 0 and y % 100 != 0) or (y % 400 == 0)

        # 返回某年某月的天数，2 月按闰年调整
        def days_in_month(y: int, m: int) -> int:
            days = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
            if m == 2 and is_leap(y):
                return 29
            return days[m - 1]

        # 解析单个字符串，成功返回 (y, m, d)，失败返回 None
        def parse(s: str):
            # 优先级：含 '/' 按斜杠分隔，否则含 '-' 按横杠分隔，否则非法
            if '/' in s:
                parts = s.split('/')
            elif '-' in s:
                parts = s.split('-')
            else:
                return None
            # 必须恰好拆出 3 段（混合分隔符会拆不出 3 段，自然判非法）
            if len(parts) != 3:
                return None
            try:
                y, m, d = int(parts[0]), int(parts[1]), int(parts[2])
            except ValueError:
                return None
            # 年份范围、月份范围、日范围逐一校验
            if not (1000 <= y <= 2100):
                return None
            if not (1 <= m <= 12):
                return None
            if not (1 <= d <= days_in_month(y, m)):
                return None
            return (y, m, d)

        seen = set()        # 用于去重
        result = []
        for s in dates:
            p = parse(s)
            if p is None:
                continue         # 非法格式直接忽略
            # 统一格式化为 yyyy-mm-dd
            key = "%04d-%02d-%02d" % (p[0], p[1], p[2])
            if key in seen:
                continue         # 同一日期只保留一次
            seen.add(key)
            result.append(key)

        if not result:
            return ["NULL"]    # 全部非法时返回 ["NULL"]
        result.sort()          # 定宽格式，字典序即日期递增序
        return result

import java.util.*;

class Solution {
    // 闰年判定
    private static boolean isLeap(int y) {
        return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
    }
    // 某年某月的天数
    private static int daysInMonth(int y, int m) {
        int[] d = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        if (m == 2 && isLeap(y)) return 29;
        return d[m - 1];
    }
    // 解析单个字符串，成功返回 true 并写入 ymd
    private static boolean parse(String s, int[] ymd) {
        char delim;
        if (s.indexOf('/') >= 0) delim = '/';        // 优先级：斜杠
        else if (s.indexOf('-') >= 0) delim = '-';    // 其次：横杠
        else return false;                             // 无分隔符：非法

        String[] parts = s.split(Character.toString(delim));
        if (parts.length != 3) return false;           // 必须恰好 3 段

        // 每段必须是纯数字（含非数字即非法，混合分隔符也会在此暴露）
        for (String p : parts) {
            if (p.isEmpty()) return false;
            for (int i = 0; i < p.length(); i++)
                if (!Character.isDigit(p.charAt(i))) return false;
        }
        int y = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        int d = Integer.parseInt(parts[2]);
        if (y < 1000 || y > 2100) return false;
        if (m < 1 || m > 12) return false;
        if (d < 1 || d > daysInMonth(y, m)) return false;
        ymd[0] = y; ymd[1] = m; ymd[2] = d;
        return true;
    }

    public List<String> normalizeDates(List<String> dates) {
        Set<String> seen = new HashSet<>();    // 去重
        List<String> result = new ArrayList<>();
        for (String s : dates) {
            int[] ymd = new int[3];
            if (!parse(s, ymd)) continue;       // 非法忽略
            // 统一格式化为 yyyy-mm-dd
            String key = String.format("%04d-%02d-%02d", ymd[0], ymd[1], ymd[2]);
            if (seen.contains(key)) continue;    // 同日期只保留一次
            seen.add(key);
            result.add(key);
        }
        if (result.isEmpty()) {
            List<String> r = new ArrayList<>();
            r.add("NULL");     // 全部非法返回 ["NULL"]
            return r;
        }
        Collections.sort(result);    // 定宽格式，字典序即日期序
        return result;
    }
}

#include <vector>
#include <string>
#include <set>
#include <algorithm>
#include <cstdio>
#include <cctype>
using namespace std;

class Solution {
public:
    // 闰年判定
    static bool isLeap(int y) {
        return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
    }
    // 某年某月的天数
    static int daysInMonth(int y, int m) {
        static const int d[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        if (m == 2 && isLeap(y)) return 29;
        return d[m - 1];
    }
    // 解析单个字符串，成功返回 true 并写出 y,m,d
    static bool parse(const string& s, int& y, int& m, int& d) {
        char delim;
        if (s.find('/') != string::npos) delim = '/';      // 优先级：斜杠
        else if (s.find('-') != string::npos) delim = '-';  // 其次：横杠
        else return false;                                   // 无分隔符：非法

        // 按分隔符拆成至多 3 段
        vector<string> parts;
        string cur;
        for (char c : s) {
            if (c == delim) { parts.push_back(cur); cur.clear(); }
            else cur += c;
        }
        parts.push_back(cur);
        if (parts.size() != 3) return false;    // 必须恰好 3 段

        // 每段必须是纯数字（含非数字即非法，混合分隔符也会在此暴露）
        for (auto& p : parts) {
            if (p.empty()) return false;
            for (char c : p) if (!isdigit((unsigned char)c)) return false;
        }
        y = stoi(parts[0]); m = stoi(parts[1]); d = stoi(parts[2]);
        if (y < 1000 || y > 2100) return false;
        if (m < 1 || m > 12) return false;
        if (d < 1 || d > daysInMonth(y, m)) return false;
        return true;
    }

    vector<string> normalizeDates(vector<string>& dates) {
        set<string> seen;        // 去重
        vector<string> result;
        for (auto& s : dates) {
            int y, m, d;
            if (!parse(s, y, m, d)) continue;    // 非法忽略
            char buf[16];
            snprintf(buf, sizeof(buf), "%04d-%02d-%02d", y, m, d);
            string key(buf);
            if (seen.count(key)) continue;        // 同日期只保留一次
            seen.insert(key);
            result.push_back(key);
        }
        if (result.empty()) {
            vector<string> r;
            r.push_back("NULL");    // 全部非法返回 ["NULL"]
            return r;
        }
        sort(result.begin(), result.end());  // 定宽格式，字典序即日期序
        return result;
    }
};

/**
 * @param {string[]} dates
 * @return {string[]}
 */
function normalizeDates(dates) {
    // 闰年判定
    function isLeap(y) {
        return (y % 4 === 0 && y % 100 !== 0) || (y % 400 === 0);
    }
    // 某年某月的天数
    function daysInMonth(y, m) {
        const d = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
        if (m === 2 && isLeap(y)) return 29;
        return d[m - 1];
    }
    // 解析单个字符串，成功返回 [y,m,d]，失败返回 null
    function parse(s) {
        let delim;
        if (s.indexOf('/') >= 0) delim = '/';         // 优先级：斜杠
        else if (s.indexOf('-') >= 0) delim = '-';     // 其次：横杠
        else return null;                              // 无分隔符：非法

        const parts = s.split(delim);
        if (parts.length !== 3) return null;           // 必须恰好 3 段

        // 每段必须是纯数字
        for (const p of parts) {
            if (p.length === 0) return null;
            if (!/^\d+$/.test(p)) return null;
        }
        const y = parseInt(parts[0], 10);
        const m = parseInt(parts[1], 10);
        const d = parseInt(parts[2], 10);
        if (y < 1000 || y > 2100) return null;
        if (m < 1 || m > 12) return null;
        if (d < 1 || d > daysInMonth(y, m)) return null;
        return [y, m, d];
    }

    const seen = new Set();    // 去重
    const result = [];
    for (const s of dates) {
        const p = parse(s);
        if (p === null) continue;    // 非法忽略
        // 统一格式化为 yyyy-mm-dd
        const key = String(p[0]).padStart(4, '0') + '-' +
                    String(p[1]).padStart(2, '0') + '-' +
                    String(p[2]).padStart(2, '0');
        if (seen.has(key)) continue;   // 同日期只保留一次
        seen.add(key);
        result.push(key);
    }
    if (result.length === 0) return ["NULL"];   // 全部非法返回 ["NULL"]
    result.sort();    // 定宽格式，字典序即日期序
    return result;
}

#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdio.h>

// 闰年判定
static int isLeap(int y) {
    return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
}
// 某年某月的天数
static int daysInMonth(int y, int m) {
    static const int d[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    if (m == 2 && isLeap(y)) return 29;
    return d[m - 1];
}
// 字符串比较，供 qsort 使用
static int cmpStr(const void* a, const void* b) {
    return strcmp(*(const char**)a, *(const char**)b);
}
// dates：输入字符串数组；datesSize：数组长度
// returnSize：输出数组长度（通过指针返回）
// 返回值为动态分配的字符串数组（以 NULL 结尾），调用方负责释放
char** normalizeDates(char** dates, int datesSize, int* returnSize) {
// 解析单个字符串，成功返回 1 并写出 y,m,d
static int parseDate(const char* s, int* y, int* m, int* d) {
        char delim = 0;
        if (strchr(s, '/')) delim = '/';         // 优先级：斜杠
        else if (strchr(s, '-')) delim = '-';    // 其次：横杠
        else return 0;                           // 无分隔符：非法

        char part0[16], part1[16], part2[16];
        const char* p = s;
        char* parts[3] = {part0, part1, part2};
        int idx = 0, cur = 0;
        for (; *p; p++) {
            if (*p == delim) {
                parts[idx][cur] = '\0';
                idx++; cur = 0;
                if (idx > 2) return 0;    // 超过 3 段：非法
            } else {
                parts[idx][cur++] = *p;
            }
        }
        parts[idx][cur] = '\0';
        if (idx != 2) return 0;           // 必须恰好 3 段

        // 每段必须是纯数字
        for (int i = 0; i < 3; i++) {
            if (parts[i][0] == '\0') return 0;
            for (char* q = parts[i]; *q; q++)
                if (!isdigit((unsigned char)*q)) return 0;
        }
        *y = atoi(parts[0]); *m = atoi(parts[1]); *d = atoi(parts[2]);
        if (*y < 1000 || *y > 2100) return 0;
        if (*m < 1 || *m > 12) return 0;
        if (*d < 1 || *d > daysInMonth(*y, *m)) return 0;
        return 1;
    }

    // 先用临时数组收集去重后的日期
    char** tmp = (char**)malloc(sizeof(char*) * (datesSize + 1));
    int cnt = 0;
    for (int i = 0; i < datesSize; i++) {
        int y, m, d;
        if (!parseDate(dates[i], &y, &m, &d)) continue;    // 非法忽略
        char key[16];
        sprintf(key, "%04d-%02d-%02d", y, m, d);
        // 去重：与已收集的比较
        int dup = 0;
        for (int j = 0; j < cnt; j++)
            if (strcmp(tmp[j], key) == 0) { dup = 1; break; }
        if (dup) continue;
        tmp[cnt] = (char*)malloc(strlen(key) + 1);
        strcpy(tmp[cnt], key);
        cnt++;
    }
    if (cnt == 0) {
        // 全部非法：返回仅含 "NULL" 的数组
        char** res = (char**)malloc(sizeof(char*) * 2);
        res[0] = (char*)malloc(5);
        strcpy(res[0], "NULL");
        res[1] = NULL;
        *returnSize = 1;
        free(tmp);
        return res;
    }
    // 定宽格式，字典序即日期序
    qsort(tmp, cnt, sizeof(char*), cmpStr);
    char** res = (char**)malloc(sizeof(char*) * (cnt + 1));
    for (int i = 0; i < cnt; i++) {
        res[i] = (char*)malloc(strlen(tmp[i]) + 1);
        strcpy(res[i], tmp[i]);
    }
    res[cnt] = NULL;
    *returnSize = cnt;
    for (int i = 0; i < cnt; i++) free(tmp[i]);
    free(tmp);
    return res;
}
