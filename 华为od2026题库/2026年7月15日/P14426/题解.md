## 解题思路

本题求从城市 $1$ 到城市 $n$ 在预算 $w$ 内的**最小花费路线**。由于每条路线的费用 $cost \ge 1$（均为正数），这是一个**单源最短路径**问题，使用 **Dijkstra 算法**即可。

核心观察：在所有从 $1$ 到 $n$ 的路线中，「花费最少」的那条就是最短路径。因此：

- 先用 Dijkstra 求出 $1$ 到 $n$ 的最短距离 $d$；
- 若 $d \le w$，说明该最优路线在预算内，直接返回 $d$；
- 否则所有路线都超过预算（最短的都超了，更长的更超），返回 $-1$。

边权均为正，Dijkstra 的正确性有保证；城市数 $n \le 100$、路线数 $m \le 1000$，规模很小，堆优化 Dijkstra 或 $O(n^2)$ 的朴素 Dijkstra 都能轻松通过。

## 复杂度分析

- **时间复杂度**：$O(m \log n)$（用小根堆维护当前距离最小的未访问城市，每次松弛最多 $m$ 条边）。由于 $n \le 100, m \le 1000$，即使朴素 $O(n^2)$ 实现也仅约 $10^4$ 次操作，远在时限内。
- **空间复杂度**：$O(n + m)$，用于存储邻接表、距离数组与优先队列。

```python
from typing import List
import heapq


class Solution:
    def minCost(self, n: int, m: int, w: int, roads: List[List[int]]) -> int:
        # 建图：g[u] 存 (v, cost) 的邻接表
        g = [[] for _ in range(n + 1)]
        for u, v, cost in roads:
            g[u].append((v, cost))

        INF = 10 ** 9
        dist = [INF] * (n + 1)
        dist[1] = 0  # 起点到自身花费为 0

        # 小根堆，按累计花费取出最小城市
        pq = [(0, 1)]  # (累计花费, 城市编号)
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue  # 过期记录，跳过
            for v, c in g[u]:
                # 松弛：经过 u 到 v 更便宜则更新
                if dist[u] + c < dist[v]:
                    dist[v] = dist[u] + c
                    heapq.heappush(pq, (dist[v], v))

        # 最短花费不超过预算则返回，否则无解
        return dist[n] if dist[n] <= w else -1
```

```java
import java.util.*;

class Solution {
    public int minCost(int n, int m, int w, int[][] roads) {
        // 建图：g[u] 存 (v, cost) 的邻接表
        List<int[]>[] g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
        for (int[] e : roads) {
            g[e[0]].add(new int[]{e[1], e[2]});
        }

        final int INF = 1_000_000_000;
        int[] dist = new int[n + 1];
        Arrays.fill(dist, INF);
        dist[1] = 0;  // 起点花费为 0

        // 小根堆，按累计花费取最小城市
        PriorityQueue<int[]> pq = new PriorityQueue<>(
            (a, b) -> Integer.compare(a[0], b[0]));
        pq.offer(new int[]{0, 1});  // (累计花费, 城市编号)
        while (!pq.isEmpty()) {
            int[] cur = pq.poll();
            int d = cur[0], u = cur[1];
            if (d > dist[u]) continue;  // 过期记录，跳过
            for (int[] edge : g[u]) {
                int v = edge[0], c = edge[1];
                // 松弛：经过 u 到 v 更便宜则更新
                if (dist[u] + c < dist[v]) {
                    dist[v] = dist[u] + c;
                    pq.offer(new int[]{dist[v], v});
                }
            }
        }

        // 最短花费不超过预算则返回，否则无解
        return dist[n] <= w ? dist[n] : -1;
    }
}
```

```cpp
#include <vector>

using namespace std;

class Solution {
public:
    int minCost(int n, int m, int w, vector<vector<int>>& roads) {
        // 建图：g[u] 存 (v, cost) 的邻接表
        vector<vector<pair<int, int>>> g(n + 1);
        for (size_t i = 0; i < roads.size(); i++) {
            int u = roads[i][0], v = roads[i][1], c = roads[i][2];
            g[u].push_back(make_pair(v, c));
        }

        const int INF = 1000000000;
        vector<int> dist(n + 1, INF);
        dist[1] = 0;  // 起点花费为 0

        // 小根堆，按累计花费取最小城市
        priority_queue<pair<int, int>, vector<pair<int, int>>,
                       greater<pair<int, int>>> pq;
        pq.push(make_pair(0, 1));  // (累计花费, 城市编号)
        while (!pq.empty()) {
            int d = pq.top().first;
            int u = pq.top().second;
            pq.pop();
            if (d > dist[u]) continue;  // 过期记录，跳过
            for (size_t i = 0; i < g[u].size(); i++) {
                int v = g[u][i].first;
                int c = g[u][i].second;
                // 松弛：经过 u 到 v 更便宜则更新
                if (dist[u] + c < dist[v]) {
                    dist[v] = dist[u] + c;
                    pq.push(make_pair(dist[v], v));
                }
            }
        }

        // 最短花费不超过预算则返回，否则无解
        if (dist[n] <= w) return dist[n];
        return -1;
    }
};
```

```javascript
function minCost(n, m, w, roads) {
    // 建图：g[u] 存 [v, cost] 的邻接表
    const g = Array.from({ length: n + 1 }, () => []);
    for (const [u, v, cost] of roads) {
        g[u].push([v, cost]);
    }

    const INF = 1e9;
    const dist = new Array(n + 1).fill(INF);
    const vis = new Array(n + 1).fill(false);
    dist[1] = 0;  // 起点花费为 0

    // 城市数 n ≤ 100，用 O(n^2) 的 Dijkstra（线性选最小未访问点）
    for (let it = 0; it < n; it++) {
        // 选未访问中距离最小的城市
        let u = -1, best = INF;
        for (let i = 1; i <= n; i++) {
            if (!vis[i] && dist[i] < best) {
                best = dist[i];
                u = i;
            }
        }
        if (u === -1) break;  // 剩余城市均不可达
        vis[u] = true;
        // 松弛所有出边
        for (const [v, c] of g[u]) {
            if (dist[u] + c < dist[v]) dist[v] = dist[u] + c;
        }
    }

    // 最短花费不超过预算则返回，否则无解
    return dist[n] <= w ? dist[n] : -1;
}
```

```c
#include <stdlib.h>

// 邻接表边节点
typedef struct Edge {
    int v;
    int cost;
    struct Edge* next;
} Edge;

int minCost(int n, int m, int w, int** roads, int roadsSize, int* roadsColSize) {
    // 建图：邻接表（头插法）
    Edge** g = (Edge**)calloc(n + 1, sizeof(Edge*));
    for (int i = 0; i < roadsSize; i++) {
        int u = roads[i][0], v = roads[i][1], c = roads[i][2];
        Edge* e = (Edge*)malloc(sizeof(Edge));
        e->v = v;
        e->cost = c;
        e->next = g[u];
        g[u] = e;
    }

    const int INF = 1000000000;
    int* dist = (int*)malloc((n + 1) * sizeof(int));
    int* vis = (int*)calloc(n + 1, sizeof(int));
    for (int i = 1; i <= n; i++) dist[i] = INF;
    dist[1] = 0;  // 起点花费为 0

    // 城市数 n ≤ 100，用 O(n^2) 的 Dijkstra（线性选最小未访问点）
    for (int it = 0; it < n; it++) {
        int u = -1, best = INF;
        for (int i = 1; i <= n; i++) {
            if (!vis[i] && dist[i] < best) {
                best = dist[i];
                u = i;
            }
        }
        if (u == -1) break;  // 剩余城市均不可达
        vis[u] = 1;
        // 松弛所有出边
        for (Edge* e = g[u]; e; e = e->next) {
            int v = e->v, c = e->cost;
            if (dist[u] + c < dist[v]) dist[v] = dist[u] + c;
        }
    }

    int ans = (dist[n] <= w) ? dist[n] : -1;

    // 释放内存
    for (int i = 1; i <= n; i++) {
        Edge* e = g[i];
        while (e) {
            Edge* nx = e->next;
            free(e);
            e = nx;
        }
    }
    free(g);
    free(dist);
    free(vis);
    return ans;
}
```
