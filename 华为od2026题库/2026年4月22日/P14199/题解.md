## 解题思路

这题本质上是一个**按时间轴模拟 + 分类统计**的问题。

一天只有 $1440$ 分钟，而三挡电价的规则是按“**当天从 $0:00$ 开始**”来划分的，因此最直接且最稳妥的做法就是：

1. 先把输入的开关机时间转成分钟。
2. 把一天中的每一分钟都看成一个单位，按规则预处理它属于哪一档。
3. 再统计设备开机区间 $[start, end)$ 内，每一分钟分别落在哪一档。

### 三挡规则如何落实

#### 第一档

固定时段：

* $12:00$ - $13:30$
* $17:30$ - $18:00$

这两个区间内的分钟，直接归为第一档。

#### 第二档

题目说的是：

* 从每天 $0:00$ 开始
* **排除第一档时段后**
* 累积前 $10$ 小时，也就是 $600$ 分钟
* 这些分钟归为第二档

这其实就是一个**顺序扫描 + 贪心标记**：

* 从第 $0$ 分钟扫描到第 $1439$ 分钟
* 遇到**不属于第一档**的分钟，就优先分给第二档
* 直到第二档累计满 $600$ 分钟为止

#### 第三档

剩余未被分到第一档、第二档的分钟，全部归第三档。

### 为什么这种方法最合适

这题最容易出错的地方在于：

* 第二档不是“开机后的前 $10$ 小时”，而是“**当天从 $0:00$ 开始累计的前 $10$ 小时（排除第一档）**”
* 第一档和第二档存在优先级关系
* 时间区间要按分钟精确处理，且要注意开关机区间通常按 **左闭右开** $[start, end)$ 处理

因此，用“逐分钟模拟”最清晰，也最不容易写错。

## 复杂度分析

一天固定只有 $1440$ 分钟。

* **时间复杂度**：$O(1440)$，也就是 $O(1)$  
  预处理一天的档位需要扫描一次，统计开机区间再扫描一次，都是常数级。
* **空间复杂度**：$O(1440)$，也就是 $O(1)$  
  用一个长度为 $1440$ 的数组记录每一分钟所属档位。

之所以可以认为是 $O(1)$，是因为一天的分钟数是固定常数，不随输入规模增长。

```python
from typing import List

class Solution:
    def timeClassification(self, timeRange: str) -> List[int]:
        # 去掉多余的空格与引号
        timeRange = timeRange.strip()
        if len(timeRange) >= 2 and timeRange[0] == '"' and timeRange[-1] == '"':
            timeRange = timeRange[1:-1]

        # 拆分起止时间
        start_str, end_str = timeRange.split()
        h1, m1 = map(int, start_str.split(":"))
        h2, m2 = map(int, end_str.split(":"))

        start = h1 * 60 + m1
        end = h2 * 60 + m2

        # 三个档位分钟统计
        ans = [0, 0, 0]
        second_cnt = 0  # 第二档累计分钟数

        for i in range(start, end):
            # 第一档固定优先判断
            if (12 * 60 <= i < 13 * 60 + 30) or (17 * 60 + 30 <= i < 18 * 60):
                ans[0] += 1
            elif second_cnt < 600:
                # 非第一档的前 600 分钟属于第二档
                ans[1] += 1
                second_cnt += 1
            else:
                # 其余分钟归第三档
                ans[2] += 1

        return ans
```

```java
import java.util.*;

public class Solution {
    public int[] timeClassification(String timeRange) {
        // 去除引号与空格
        timeRange = timeRange.trim();
        if (timeRange.length() >= 2 && timeRange.charAt(0) == '"' && timeRange.charAt(timeRange.length() - 1) == '"') {
            timeRange = timeRange.substring(1, timeRange.length() - 1);
        }

        // 拆分时间段
        String[] parts = timeRange.split(" ");
        String[] startArr = parts[0].split(":");
        String[] endArr = parts[1].split(":");

        int start = Integer.parseInt(startArr[0]) * 60 + Integer.parseInt(startArr[1]);
        int end = Integer.parseInt(endArr[0]) * 60 + Integer.parseInt(endArr[1]);

        int[] ans = new int[3];
        int secondCnt = 0;

        for (int i = start; i < end; i++) {
            // 第一档固定优先
            if ((i >= 12 * 60 && i < 13 * 60 + 30) ||
                (i >= 17 * 60 + 30 && i < 18 * 60)) {
                ans[0]++;
            } else if (secondCnt < 600) {
                // 非第一档累计前 600 分钟属于第二档
                ans[1]++;
                secondCnt++;
            } else {
                // 超过后进入第三档
                ans[2]++;
            }
        }

        return ans;
    }
}
```

```cpp
#include <vector>
#include <string>
using namespace std;

class Solution {
public:
    vector<int> timeClassification(string timeRange) {
        // 去除首尾引号
        if (!timeRange.empty() && timeRange[0] == '"') {
            timeRange = timeRange.substr(1);
        }
        if (!timeRange.empty() && timeRange.back() == '"') {
            timeRange.pop_back();
        }

        // 分割字符串
        int blank = timeRange.find(' ');
        string a = timeRange.substr(0, blank);
        string b = timeRange.substr(blank + 1);

        int p1 = a.find(':');
        int p2 = b.find(':');

        int start = stoi(a.substr(0, p1)) * 60 + stoi(a.substr(p1 + 1));
        int end = stoi(b.substr(0, p2)) * 60 + stoi(b.substr(p2 + 1));

        vector<int> ans(3, 0);
        int secondCnt = 0;

        for (int i = start; i < end; i++) {
            // 第一档优先
            if ((i >= 12 * 60 && i < 13 * 60 + 30) ||
                (i >= 17 * 60 + 30 && i < 18 * 60)) {
                ans[0]++;
            } else if (secondCnt < 600) {
                // 非第一档前 600 分钟属第二档
                ans[1]++;
                secondCnt++;
            } else {
                // 其他归第三档
                ans[2]++;
            }
        }

        return ans;
    }
};
```

```javascript
var timeClassification = function(timeRange) {

    const parts = timeRange.match(/\d+/g).map(Number);
    const start = parts[0] * 60 + parts[1];
    const end = parts[2] * 60 + parts[3];

        else if (cumulNonTier1 < 600) {
            result[1]++;
            cumulNonTier1++;
        } else result[2]++;
    }
    return result;
};
```

```c
// 判断是否为第一档时间
static int _isTier1(int minute) {
    return (minute >= 720 && minute < 810) || (minute >= 1050 && minute < 1080);
}

int* timeClassification(char* timeRange, int* returnSize) {
    // 去除首尾可能的引号
    if (timeRange[0] == '"') timeRange++;
    int len = (int)strlen(timeRange);
    if (len > 0 && timeRange[len - 1] == '"') timeRange[len - 1] = '\0';

    // 解析输入时间
    int h1, m1, h2, m2;
    sscanf(timeRange, "%d:%d %d:%d", &h1, &m1, &h2, &m2);
    int start = h1 * 60 + m1;
    int end = h2 * 60 + m2;

    *returnSize = 3;
    int* result = (int*)calloc(3, sizeof(int));
    int cumulNonTier1 = 0;

    for (int m = start; m < end; m++) {
        if (_isTier1(m)) result[0]++;
        else if (cumulNonTier1 < 600) {
            result[1]++;
            cumulNonTier1++;
        } else result[2]++;
    }
    return result;
}
```
