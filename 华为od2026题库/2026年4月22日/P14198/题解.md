## 解题思路

这题本质是 **字符串匹配 + 哈希表计数**。

核心有两步：

### 第一步：预处理所有“合法写法”

每个学生的完整姓名都可能对应若干种可接受的选票写法：

* 如果名字里 **没有 `-`**，那只能写 **全称**
* 如果名字里 **有 `-`**，则可以写其中 **若干个连续完整段**

  * 例如 `maimaiti-aierken-batuer`
  * 合法写法包括：

    * `maimaiti`
    * `aierken`
    * `batuer`
    * `maimaiti-aierken`
    * `aierken-batuer`
    * `maimaiti-aierken-batuer`

注意：

* 不能写半段，比如 `mai`、`aier`
* 选票上的名字必须 **唯一匹配** 某个学生，否则这张票无效

所以我们可以先枚举每个名字的所有连续段写法，建立映射：

* `写法 -> 唯一对应的完整姓名`
* 如果某种写法对应了多个学生，则标记为 **歧义**

这一步使用的算法思想是：

* **字符串拆分**
* **枚举连续区间**
* **哈希表预处理**


### 第二步：统计有效票数

遍历 `ballotTickets`：

* 如果该选票能在预处理映射中找到，且是 **唯一匹配**，则给对应学生加一票
* 否则忽略这张票

然后根据题意检查选举是否合法：

1. 原始输入总票数 `ballotTickets.size()` 超过 `3 * N`，直接无效
2. 某位同学得票数超过 `N`，无效
3. 没有任何同学被选中（即没有一张有效票），无效

最后从有效得票中选出：

* 票数最多者当选
* 如果票数相同，按字符串字典序最小者当选


### 实现细节

为了方便判断“是否唯一匹配”，可以用一个哈希表：

* 第一次出现某种写法时，记录它对应的完整姓名
* 如果后面发现同一种写法对应了不同姓名，则把它标记成特殊值，表示 **歧义**

之后唱票时：

* 若映射值不是歧义标记，则该票有效
* 否则无效

这就是典型的 **哈希预处理 + 计数** 做法，逻辑清晰且实现简洁。


## 复杂度分析

设：

* 学生人数为 $N$
* 第 $i$ 个名字按 `-` 分段后有 $k_i$ 段
* 总选票数为 $M$

### 时间复杂度

预处理每个名字时，需要枚举所有连续段，数量为：

$ \sum_{i=1}^{N} O(k_i^2) $

统计选票时，每张票查询哈希表为 $O(1)$，共 $O(M)$

因此总时间复杂度为：

$ O\left(\sum_{i=1}^{N} k_i^2 + M\right) $

这个复杂度是合适的，因为每种合法写法只需要预处理一次，后续查询非常快。

### 空间复杂度

主要是哈希表存储所有可能写法，以及计票表：

$ O\left(\sum_{i=1}^{N} k_i^2 + N\right) $

```python
from typing import List
from collections import defaultdict

class Solution:
    def getClassMonitor(self, names: List[str], ballotTickets: List[str]) -> str:
        n = len(names)
        
        # 原始总票数超过 3N，直接无效
        if len(ballotTickets) > 3 * n:
            return "Invalid election."
        
        # special 用来表示该写法存在歧义，不能唯一匹配
        special = "#INVALID#"
        mapping = {}
        
        # 预处理每个名字的所有合法连续段写法
        for name in names:
            parts = name.split('-')
            m = len(parts)
            for i in range(m):
                cur = []
                for j in range(i, m):
                    cur.append(parts[j])
                    key = "-".join(cur)
                    # 记录写法映射关系
                    if key not in mapping:
                        mapping[key] = name
                    elif mapping[key] != name:
                        mapping[key] = special
        
        # 统计有效票
        cnt = defaultdict(int)
        valid_votes = 0
        
        for ticket in ballotTickets:
            # 若票对应到唯一学生，计票 +1
            if ticket in mapping and mapping[ticket] != special:
                person = mapping[ticket]
                cnt[person] += 1
                valid_votes += 1
        
        # 没有任何有效票，则选举无效
        if valid_votes == 0:
            return "Invalid election."
        
        # 某位同学得票数超过班级总人数，则无效
        for v in cnt.values():
            if v > n:
                return "Invalid election."
        
        # 选出票数最多；若并列，选字典序更小者
        ans = ""
        max_vote = -1
        for name in cnt:
            if cnt[name] > max_vote or (cnt[name] == max_vote and name < ans):
                max_vote = cnt[name]
                ans = name
        
        return ans
```

```java
import java.util.*;

public class Solution {
    /**
     * 班长选举：根据名字与选票映射关系确定当选人
     */
    public String getClassMonitor (ArrayList<String> names, ArrayList<String> ballotTickets) {
        int n = names.size();
        
        // 原始总票数超过 3N，直接无效
        if (ballotTickets.size() > 3 * n) {
            return "Invalid election.";
        }
        
        // special 标记：该写法有歧义
        String special = "#INVALID#";
        Map<String, String> mapping = new HashMap<>();
        
        // 预处理每个名字的所有合法连续段写法
        for (String name : names) {
            String[] parts = name.split("-");
            int m = parts.length;
            
            for (int i = 0; i < m; i++) {
                StringBuilder sb = new StringBuilder();
                for (int j = i; j < m; j++) {
                    if (j > i) sb.append("-");
                    sb.append(parts[j]);
                    String key = sb.toString();
                    
                    if (!mapping.containsKey(key)) {
                        mapping.put(key, name);
                    } else if (!mapping.get(key).equals(name)) {
                        mapping.put(key, special);
                    }
                }
            }
        }
        
        // 统计有效票数
        Map<String, Integer> cnt = new HashMap<>();
        int validVotes = 0;
        
        for (String ticket : ballotTickets) {
            if (mapping.containsKey(ticket) && !mapping.get(ticket).equals(special)) {
                String person = mapping.get(ticket);
                cnt.put(person, cnt.getOrDefault(person, 0) + 1);
                validVotes++;
            }
        }
        
        // 没有任何有效票，则选举无效
        if (validVotes == 0) return "Invalid election.";
        
        // 某位同学得票数超过班级总人数，则无效
        for (int v : cnt.values()) {
            if (v > n) return "Invalid election.";
        }
        
        // 选出票数最多者；若票数相同，取字典序更小者
        String ans = "";
        int maxVote = -1;
        
        for (String name : cnt.keySet()) {
            int vote = cnt.get(name);
            if (vote > maxVote || (vote == maxVote && name.compareTo(ans) < 0)) {
                maxVote = vote;
                ans = name;
            }
        }
        
        return ans;
    }
}
```

```cpp
#include <vector>
#include <string>
#include <unordered_map>
using namespace std;

class Solution {
public:
    string getClassMonitor(vector<string>& names, vector<string>& ballotTickets) {
        int n = (int)names.size();
        
        // 原始总票数超过 3N，直接无效
        if ((int)ballotTickets.size() > 3 * n) {
            return "Invalid election.";
        }
        
        // 特殊标记：表示该写法有歧义
        const string special = "#INVALID#";
        unordered_map<string, string> mapping;
        
        // 预处理每个名字的所有合法连续段写法
        for (string& name : names) {
            vector<string> parts = split(name, '-');
            int m = (int)parts.size();
            
            for (int i = 0; i < m; i++) {
                string cur = "";
                for (int j = i; j < m; j++) {
                    if (j > i) cur += "-";
                    cur += parts[j];
                    
                    if (!mapping.count(cur)) mapping[cur] = name;
                    else if (mapping[cur] != name) mapping[cur] = special;
                }
            }
        }
        
        // 统计有效票
        unordered_map<string, int> cnt;
        int validVotes = 0;
        
        for (string& ticket : ballotTickets) {
            if (mapping.count(ticket) && mapping[ticket] != special) {
                string person = mapping[ticket];
                cnt[person]++;
                validVotes++;
            }
        }
        
        if (validVotes == 0) return "Invalid election.";
        
        for (auto& p : cnt)
            if (p.second > n) return "Invalid election.";
        
        // 找出票数最多且字典序最小者
        string ans = "";
        int maxVote = -1;
        for (auto& p : cnt) {
            if (p.second > maxVote || (p.second == maxVote && p.first < ans)) {
                maxVote = p.second;
                ans = p.first;
            }
        }
        return ans;
    }
    
private:
    // 按分隔符拆分字符串
    vector<string> split(const string& s, char delim) {
        vector<string> res;
        string cur;
        for (char c : s) {
            if (c == delim) {
                res.push_back(cur);
                cur.clear();
            } else cur += c;
        }
        res.push_back(cur);
        return res;
    }
};
```

```javascript
var getClassMonitor = function(names, ballotTickets) {
    const N = names.length;
    if (ballotTickets.length > 3 * N) return "Invalid election.";

    const mapping = new Map();
    // 预处理所有合法连续段写法
    for (const name of names) {
        const parts = name.split('-');
        const m = parts.length;
        for (let i = 0; i < m; i++) {
            for (let j = i; j < m; j++) {
                const key = parts.slice(i, j + 1).join('-');
                if (!mapping.has(key)) mapping.set(key, name);
                else if (mapping.get(key) !== name) mapping.set(key, '#INVALID#');
            }
        }
    }

    const votes = new Map();
    let validVotes = 0;
    for (const ticket of ballotTickets) {
        const match = mapping.get(ticket);
        if (match && match !== '#INVALID#') {
            votes.set(match, (votes.get(match) || 0) + 1);
            validVotes++;
        }
    }

    if (validVotes === 0) return "Invalid election.";
    for (const v of votes.values()) if (v > N) return "Invalid election.";

    let best = null, maxVotes = -1;
    for (const [name, cnt] of votes) {
        if (cnt > maxVotes || (cnt === maxVotes && name < best)) {
            maxVotes = cnt;
            best = name;
        }
    }
    return best;
};
```

```c
#include <stdlib.h>
#include <string.h>

// 判断票据是否为该学生名字的连续段
static int _isMatch(const char* name, const char* ticket) {
    int nLen = (int)strlen(name), tLen = (int)strlen(ticket);
    if (tLen > nLen) return 0;
    // 添加分隔符辅助匹配
    char* nw = (char*)malloc(nLen + 3);
    char* tw = (char*)malloc(tLen + 3);
    nw[0] = '-'; memcpy(nw + 1, name, nLen); nw[nLen + 1] = '-'; nw[nLen + 2] = '\0';
    tw[0] = '-'; memcpy(tw + 1, ticket, tLen); tw[tLen + 1] = '-'; tw[tLen + 2] = '\0';
    int ret = strstr(nw, tw) != NULL;
    free(nw); free(tw);
    return ret;
}

char* getClassMonitor(char** names, int namesSize, char** ballotTickets, int ballotTicketsSize) {
    int N = namesSize, M = ballotTicketsSize;
    if (M > 3 * N) return "Invalid election.";

    int* votes = (int*)calloc(N, sizeof(int));
    for (int b = 0; b < M; b++) {
        int matchIdx = -1;
        for (int i = 0; i < N; i++) {
            if (_isMatch(names[i], ballotTickets[b])) {
                if (matchIdx == -1) matchIdx = i;
                else { matchIdx = -2; break; } // 多人匹配则无效
            }
        }
        if (matchIdx >= 0) votes[matchIdx]++;
    }

    int totalValid = 0;
    char* best = NULL;
    int bestVotes = 0;
    for (int i = 0; i < N; i++) {
        totalValid += votes[i];
        if (votes[i] > N) { free(votes); return "Invalid election."; }
        if (votes[i] > bestVotes) { bestVotes = votes[i]; best = names[i]; }
        else if (votes[i] == bestVotes && bestVotes > 0 && strcmp(names[i], best) < 0) best = names[i];
    }
    free(votes);
    if (totalValid == 0 || bestVotes == 0) return "Invalid election.";
    return best;
}
```
