## 解题思路

给定多个仅由小写英文字母组成的片段，需要提取这些片段的公共特征。

核心观察如下：

1. 某个字母想要出现在答案中，必须在每个片段里都至少出现一次。
2. 如果某个字母在所有片段中都出现，那么它在答案中出现的次数，应该等于它在所有片段中出现次数的最小值。
3. 最终答案要求按字母从小到大排列，因此只需要按 `a` 到 `z` 的顺序处理即可。

具体做法：

1. 准备一个长度为 26 的数组 `minCnt`，初始时设为一个很大的值。
2. 枚举每个片段，统计当前片段中每个字母的出现次数，记为 `cnt`。
3. 用 `minCnt[i] = min(minCnt[i], cnt[i])` 更新每个字母在所有片段中的最小出现次数。
4. 所有片段处理完成后，按 `a` 到 `z` 的顺序，把每个字母重复 `minCnt[i]` 次加入答案。
5. 如果没有任何字母满足条件，最终答案自然就是空字符串。

这就是典型的“多个字符串的字符多重集交集”问题。

## 复杂度分析

设所有片段的总长度为 `L`，片段数量为 `n`。

- 时间复杂度：`O(L + 26n)`，由于 26 是常数，本质上可以看作 `O(L)`
- 空间复杂度：`O(26)`，只使用了若干个定长计数数组

```python
class Solution:
    def featureExtraction(self, docs):
        # 记录每个字母在所有片段中的最小出现次数
        min_cnt = [10 ** 9] * 26

        # 逐个片段统计字母频次
        for s in docs:
            cnt = [0] * 26
            for ch in s:
                cnt[ord(ch) - ord('a')] += 1

            # 更新全局最小值
            for i in range(26):
                if cnt[i] < min_cnt[i]:
                    min_cnt[i] = cnt[i]

        # 按字典序构造答案
        ans = []
        for i in range(26):
            if min_cnt[i] > 0:
                ans.append(chr(ord('a') + i) * min_cnt[i])

        # 输出格式要求加双引号
        return '"' + ''.join(ans) + '"'
```

```java
public class Solution {
    public String featureExtraction(String[] docs) {
        // 记录每个字母在所有片段中的最小出现次数
        int[] minCnt = new int[26];
        java.util.Arrays.fill(minCnt, Integer.MAX_VALUE);

        // 逐个片段统计字母频次
        for (String s : docs) {
            int[] cnt = new int[26];
            for (int i = 0; i < s.length(); i++) {
                cnt[s.charAt(i) - 'a']++;
            }

            // 更新全局最小值
            for (int i = 0; i < 26; i++) {
                minCnt[i] = Math.min(minCnt[i], cnt[i]);
            }
        }

        // 按字典序构造答案
        StringBuilder ans = new StringBuilder();
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < minCnt[i]; j++) {
                ans.append((char) ('a' + i));
            }
        }

        // 输出格式要求加双引号
        return "\"" + ans.toString() + "\"";
    }
}
```

```javascript
var featureExtraction = function(docs) {
    // 用 Infinity 初始化最小计数
    const minCnt = new Array(26).fill(Infinity);
    // 记录某字母是否在所有片段中都出现
    const present = new Array(26).fill(true);

    // 遍历每个片段
    for (const doc of docs) {
        const cnt = new Array(26).fill(0);
        for (const ch of doc) cnt[ch.charCodeAt(0) - 97]++;
        for (let c = 0; c < 26; c++) {
            if (cnt[c] === 0) present[c] = false;
            else minCnt[c] = Math.min(minCnt[c], cnt[c]);
        }
    }

    // 构造答案字符串
    let result = '';
    for (let c = 0; c < 26; c++) {
        if (present[c]) result += String.fromCharCode(97 + c).repeat(minCnt[c]);
    }

    // 输出格式要求加双引号
    return '"' + result + '"';
};
```

```c
#include <stdlib.h>
#include <string.h>
#include <limits.h>

char* featureExtraction(char** docs, int docsSize) {
    int minCnt[26], present[26];

    // 初始化最小计数为最大值，标记为全体出现
    for (int c = 0; c < 26; c++) {
        minCnt[c] = INT_MAX;
        present[c] = 1;
    }

    // 统计每个片段
    for (int i = 0; i < docsSize; i++) {
        int cnt[26] = {0};
        for (int j = 0; docs[i][j]; j++)
            cnt[docs[i][j] - 'a']++;

        // 更新最小计数与存在标记
        for (int c = 0; c < 26; c++) {
            if (cnt[c] == 0)
                present[c] = 0;
            else if (cnt[c] < minCnt[c])
                minCnt[c] = cnt[c];
        }
    }

    // 计算结果长度
    int total = 0;
    for (int c = 0; c < 26; c++)
        if (present[c]) total += minCnt[c];

    // 构建结果字符串
    char* result = (char*)malloc(total + 3); // 多分配2个位置用于双引号和终止符
    int pos = 0;
    result[pos++] = '"'; // 添加左双引号

    for (int c = 0; c < 26; c++) {
        if (present[c]) {
            for (int k = 0; k < minCnt[c]; k++)
                result[pos++] = 'a' + c;
        }
    }

    result[pos++] = '"'; // 添加右双引号
    result[pos] = '\0';  // 字符串结束
    return result;
}
```
