## 解题思路

本题要求统计字符串 `L` 的所有不同重排中，满足“相同字母不能相邻”的排列数量。

因为 `L` 长度不超过 `12`，可以使用 **DFS + 记忆化搜索 + 计数数组** 来解决。

核心思路：

1. 统计每个大写字母出现次数。
2. 使用 DFS 枚举当前位置可以放置的字母。
3. 每次选择一个剩余数量大于 `0`，且不等于上一个放置字母的字母。
4. 放置后减少该字母数量，继续递归。
5. 当所有字母都放完时，说明找到一种合法排列，返回 `1`。
6. 为了避免重复计算，使用记忆化搜索，状态由：

   * 当前剩余字母数量
   * 上一个放置的字母
   
   共同决定。

由于直接使用计数数组而不是交换字符，所以天然避免了重复排列的问题。


## 复杂度分析

设字符串长度为 `n`，不同字母数量最多为 `26`。

时间复杂度为：

$$
O(26 \times 状态数)
$$

每个状态最多尝试 `26` 个字母，状态由所有字母剩余数量和上一个字母决定。因为 `n <= 12`，状态数量是可以接受的。

空间复杂度为：

$$
O(状态数)
$$

主要来自记忆化搜索的哈希表，以及递归调用栈。递归深度最多为 `n`。

```python
from typing import List
from functools import lru_cache

class Solution:
    def countValidPatterns(self, L: str) -> int:
        # 统计每个大写字母出现次数
        cnt = [0] * 26
        for ch in L:
            cnt[ord(ch) - ord('A')] += 1

        @lru_cache(None)
        def dfs(state, pre):
            # 如果所有字母都用完，说明得到一种合法排列
            if sum(state) == 0:
                return 1

            ans = 0
            state = list(state)

            # 枚举当前可以放置的字母
            for i in range(26):
                # 剩余数量大于 0，且不能和上一个字母相同
                if state[i] > 0 and i != pre:
                    state[i] -= 1
                    ans += dfs(tuple(state), i)
                    state[i] += 1

            return ans

        # 从初始状态开始DFS搜索
        return dfs(tuple(cnt), -1)
```

```java
import java.util.*;

public class Solution {
    private Map<String, Integer> memo = new HashMap<>();

    public int countValidPatterns(String L) {
        // 统计每个大写字母出现次数
        int[] cnt = new int[26];
        for (int i = 0; i < L.length(); i++) {
            cnt[L.charAt(i) - 'A']++;
        }

        // 开始深度优先搜索
        return dfs(cnt, -1);
    }

    private int dfs(int[] cnt, int pre) {
        // 构造记忆化搜索的状态 key
        StringBuilder sb = new StringBuilder();
        sb.append(pre).append('#');
        int left = 0;

        for (int x : cnt) {
            sb.append(x).append(',');
            left += x;
        }

        // 所有字母都放完，找到一种合法排列
        if (left == 0) {
            return 1;
        }

        String key = sb.toString();
        if (memo.containsKey(key)) {
            return memo.get(key);
        }

        int ans = 0;

        // 枚举当前可以放置的字母
        for (int i = 0; i < 26; i++) {
            // 剩余数量大于 0，且不能和上一个字母相同
            if (cnt[i] > 0 && i != pre) {
                cnt[i]--;
                ans += dfs(cnt, i);
                cnt[i]++;
            }
        }

        // 记忆当前状态结果
        memo.put(key, ans);
        return ans;
    }
}
```

```cpp
class Solution {
private:
    unordered_map<string, int> memo;

    int dfs(vector<int>& cnt, int pre) {
        // 构造记忆化搜索的状态 key
        string key = to_string(pre) + "#";
        int left = 0;

        for (int x : cnt) {
            key += to_string(x) + ",";
            left += x;
        }

        // 所有字母都放完，找到一种合法排列
        if (left == 0) {
            return 1;
        }

        // 检查缓存
        if (memo.count(key)) {
            return memo[key];
        }

        int ans = 0;

        // 枚举当前可以放置的字母
        for (int i = 0; i < 26; i++) {
            // 剩余数量大于 0，且不能和上一个字母相同
            if (cnt[i] > 0 && i != pre) {
                cnt[i]--;
                ans += dfs(cnt, i);
                cnt[i]++;
            }
        }

        // 保存记忆化结果
        memo[key] = ans;
        return ans;
```

```javascript
var countValidPatterns = function(L) {
    // 统计每个大写字母出现次数
    const cnt = new Array(26).fill(0);
    for (let i = 0; i < L.length; i++) {
        cnt[L.charCodeAt(i) - 65]++;

    function dfs(cnt, prev) {
        // 构建状态键用于记忆化
        const key = prev + '#' + cnt.join(',');
        if (memo.has(key)) return memo.get(key);

        // 统计剩余总字母数
        let total = 0;
        for (let i = 0; i < 26; i++) total += cnt[i];
        if (total === 0) return 1;

        let ans = 0;
        // 枚举当前可以放置的字母
        for (let i = 0; i < 26; i++) {
            if (cnt[i] > 0 && i !== prev) {
                cnt[i]--;
                ans += dfs(cnt, i);
                cnt[i]++;
            }
        }

        memo.set(key, ans);
        return ans;
    }

    // 从初始状态开始搜索
    return dfs(cnt, -1);
};
```

```c
#include <string.h>
#include <stdio.h>

#define MAX_MEMO 50000

static char _memo_key[MAX_MEMO][256];
static int _memo_val[MAX_MEMO];
static int _memo_cnt = 0;

// 构建唯一的状态键
static void _build_key(int* cnt, int prev, char* key) {
    int pos = 0;
    pos += sprintf(key + pos, "%d", prev);
    for (int i = 0; i < 26; i++) {
        pos += sprintf(key + pos, ",%d", cnt[i]);
    }
}

// 检查是否已存在记忆化结果
static int _check_memo(int* cnt, int prev, int* result) {
    char key[256];
    _build_key(cnt, prev, key);
    for (int i = 0; i < _memo_cnt; i++) {
        if (strcmp(_memo_key[i], key) == 0) {
            *result = _memo_val[i];
            return 1;
        }
    }
    return 0;
}

// 记忆新状态结果
static void _add_memo(int* cnt, int prev, int val) {
    if (_memo_cnt >= MAX_MEMO) return;
    _build_key(cnt, prev, _memo_key[_memo_cnt]);
    _memo_val[_memo_cnt] = val;
    _memo_cnt++;
}

// 深度优先搜索主函数
static int _dfs(int* cnt, int prev) {
    int memo_result;
    if (_check_memo(cnt, prev, &memo_result)) return memo_result;

    int total = 0;
    for (int i = 0; i < 26; i++) total += cnt[i];
    if (total == 0) return 1;

    int ans = 0;
    for (int i = 0; i < 26; i++) {
        if (cnt[i] > 0 && i != prev) {
            cnt[i]--;
            ans += _dfs(cnt, i);
            cnt[i]++;
        }
    }

    _add_memo(cnt, prev, ans);
    return ans;
}

// 入口函数
int countValidPatterns(char* L) {
    int cnt[26] = {0};
    for (int i = 0; L[i]; i++) {
        cnt[L[i] - 'A']++;
    }
    _memo_cnt = 0;
    return _dfs(cnt, -1);
}
```
