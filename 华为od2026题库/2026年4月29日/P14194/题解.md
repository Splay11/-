## 解题思路

本题需要找出所有出现次数大于等于 $3$ 次的日志标识，并按照以下规则排序：

1. 出现次数越多，排序越靠前  
2. 出现次数相同，首次出现位置越早，排序越靠前  

可以使用哈希表完成统计：

1. 遍历数组 $logs$  
2. 用哈希表 $cnt$ 记录每个日志标识出现次数  
3. 用哈希表 $first$ 记录每个日志标识首次出现的位置  
4. 遍历 $cnt$，筛选出出现次数大于等于 $3$ 的日志标识  
5. 对结果数组排序，排序规则为出现次数降序，首次出现位置升序  

使用的算法是哈希统计和自定义排序。

## 复杂度分析

设日志总量为 $n$，不同日志标识数量为 $m$。  

时间复杂度为 $O(n + m \log m)$，其中统计次数需要 $O(n)$，排序最多需要 $O(m \log m)$。  

空间复杂度为 $O(m)$，需要存储每个日志标识的出现次数和首次出现位置。

```python
        ans.sort(key=lambda x: (-cnt[x], first[x]))
```

```java
import java.util.*;

public class Solution {
    /**
     * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
     *
     * 找出所有出现次数大于等于 3 次的日志标识
     * @param logs string字符串一维数组 日志标识数组
     * @return string字符串一维数组
     */
    public String[] findAnomalyLogs(String[] logs) {
        Map<String, Integer> cnt = new HashMap<>();
        Map<String, Integer> first = new HashMap<>();

        // 统计每个日志标识出现次数，并记录首次出现位置
        for (int i = 0; i < logs.length; i++) {
            String s = logs[i];
            cnt.put(s, cnt.getOrDefault(s, 0) + 1);
            if (!first.containsKey(s)) {
                first.put(s, i);
            }
        }

        List<String> ans = new ArrayList<>();

        // 筛选出现次数大于等于 3 次的日志标识
        for (String s : cnt.keySet()) {
            if (cnt.get(s) >= 3) {
                ans.add(s);
            }
        }

        // 按出现次数降序，首次出现位置升序排序
        Collections.sort(ans, new Comparator<String>() {
            public int compare(String a, String b) {
                if (!cnt.get(a).equals(cnt.get(b))) {
                    return cnt.get(b) - cnt.get(a);
                }
                return first.get(a) - first.get(b);
            }
        });

        // 将列表转为数组返回
        String[] res = new String[ans.size()];
        for (int i = 0; i < ans.size(); i++) {
            res[i] = ans.get(i);
        }

        return res;
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    /**
     * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
     *
     * 找出所有出现次数大于等于 3 次的日志标识
     * @param logs string字符串vector 日志标识数组
     * @return string字符串vector
     */
    vector<string> findAnomalyLogs(vector<string>& logs) {
        unordered_map<string, int> cnt;
        unordered_map<string, int> first;

        // 统计每个日志标识出现次数，并记录首次出现位置
        for (int i = 0; i < (int)logs.size(); i++) {
            string s = logs[i];
            cnt[s]++;
            if (first.find(s) == first.end()) {
                first[s] = i;
            }
        }

        vector<string> ans;

        // 筛选出现次数大于等于 3 次的日志标识
        for (auto &p : cnt) {
            if (p.second >= 3) {
                ans.push_back(p.first);
            }
        }

        // 按出现次数降序，首次出现位置升序排序
        sort(ans.begin(), ans.end(), [&](const string& a, const string& b) {
            if (cnt[a] != cnt[b]) {
                return cnt[a] > cnt[b];
            }
            return first[a] < first[b];
        });

        return ans;
    }
};
```

```javascript
var findAnomalyLogs = function(logs) {
    const cnt = new Map();
    const first = new Map();

    // 统计每个日志标识出现次数，并记录首次出现位置
    for (let i = 0; i < logs.length; i++) {
        const log = logs[i];
        cnt.set(log, (cnt.get(log) || 0) + 1);
        if (!first.has(log)) first.set(log, i);
    }

    const result = [];

    // 筛选出现次数大于等于 3 次的日志标识
    for (const [log, count] of cnt) {
        if (count >= 3) result.push(log);
    }

    // 排序：出现次数降序，首次出现位置升序
    result.sort((a, b) => {
        const diff = cnt.get(b) - cnt.get(a);
        if (diff !== 0) return diff;
        return first.get(a) - first.get(b);
    });

    return result;
};
```

```c
#include <string.h>
#include <stdlib.h>

typedef struct {
    char* name;
    int count;
    int firstIdx;
} LogEntry;

// 比较函数：出现次数降序，首次出现位置升序
static int _cmp(const void* a, const void* b) {
    const LogEntry* x = (const LogEntry*)a;
    const LogEntry* y = (const LogEntry*)b;
    if (x->count != y->count) return y->count - x->count;
    return x->firstIdx - y->firstIdx;
}

char** findAnomalyLogs(char** logs, int logsSize, int* returnSize) {
    // 建立日志条目数组以统计次数
    LogEntry* entries = (LogEntry*)malloc(logsSize * sizeof(LogEntry));
    int entryCount = 0;

    for (int i = 0; i < logsSize; i++) {
        int found = -1;
        for (int j = 0; j < entryCount; j++) {
            if (strcmp(entries[j].name, logs[i]) == 0) {
                found = j;
                break;
            }
        }
        if (found == -1) {
            entries[entryCount].name = logs[i];
            entries[entryCount].count = 1;
            entries[entryCount].firstIdx = i;
            entryCount++;
        } else {
            entries[found].count++;
        }
    }

    // 筛选出现次数 >= 3 次的日志标识
    LogEntry* candidates = (LogEntry*)malloc(entryCount * sizeof(LogEntry));
    int candCount = 0;
    for (int i = 0; i < entryCount; i++) {
        if (entries[i].count >= 3) {
            candidates[candCount++] = entries[i];
        }
    }

    // 根据排序规则进行排序
    qsort(candidates, candCount, sizeof(LogEntry), _cmp);

    // 构建返回数组
    char** result = (char**)malloc(candCount * sizeof(char*));
    for (int i = 0; i < candCount; i++) {
        result[i] = (char*)malloc(strlen(candidates[i].name) + 1);
        strcpy(result[i], candidates[i].name);
    }

    *returnSize = candCount;
    free(entries);
    free(candidates);
    return result;
}
```
