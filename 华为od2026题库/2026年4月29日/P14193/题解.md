## 解题思路

本题要求实现一个**操作历史管理器**，支持三种命令：

* `execute 操作描述`：执行一个新操作，并清空当前状态之后的所有历史记录  
* `undo`：回退到上一个操作状态  
* `redo`：前进到下一个操作状态  

核心数据结构使用**双向链表**。

我们可以建立一个虚拟头节点 `head`，表示“从未执行过任何操作”的初始状态，其操作描述为空字符串 `""`。

再用一个指针 `cur` 表示当前所在的操作状态：

* 执行 `execute desc`：
  * 删除 `cur` 后面的所有 redo 历史  
  * 新建节点保存 `desc`  
  * 接到 `cur` 后面  
  * `cur` 移动到新节点  
* 执行 `undo`：
  * 如果 `cur == head`，说明已经在初始状态，返回 `"undo failed"`  
  * 否则 `cur = cur.prev`  
* 执行 `redo`：
  * 如果 `cur.next == null`，说明没有可重做的操作，返回 `"redo failed"`  
  * 否则 `cur = cur.next`  

所有命令执行完后，返回当前操作描述：

* 如果当前在初始状态，返回空字符串 `""`  
* 如果中途失败，立即返回失败字符串，不继续执行后续命令  


## 复杂度分析

设命令数量为 `n`。

时间复杂度：`O(n)`  
每条命令只会进行常数次指针移动或节点连接，因此总时间复杂度为 `O(n)`。

空间复杂度：`O(n)`  
最坏情况下，所有命令都是 `execute`，双向链表中需要保存所有历史操作，因此空间复杂度为 `O(n)`。

```python
from typing import List

class Solution:
    def execute_command(self, command: List[List[str]]) -> str:
        # 定义双向链表节点
        class Node:
            def __init__(self, val: str):
                self.val = val
                self.prev = None
                self.next = None

        # 创建虚拟头节点，表示初始状态
        head = Node("")
        cur = head

        for cmd in command:
            # 兼容 [[]] 这种空命令样例
            if not cmd:
                continue

            op = cmd[0]

            if op == "execute":
                # 执行新命令，清除当前之后的 redo 历史
                desc = cmd[1]
                cur.next = None

                # 新建节点保存操作描述
                node = Node(desc)
                node.prev = cur
                cur.next = node
                cur = node

            elif op == "undo":
                # 回退至上一个状态
                if cur == head:
                    return "undo failed"
                cur = cur.prev

            else:
                # redo 操作
                if cur.next is None:
                    return "redo failed"
                cur = cur.next

        # 返回当前操作描述
        return cur.val
```

```java
import java.util.*;

public class Solution {
    public String execute_command(String[][] command) {
        List<String> history = new ArrayList<>();
        int cur = -1; // 当前索引位置，-1 表示初始状态

        for (String[] cmd : command) {
            // 兼容空命令
            if (cmd.length == 0) {
                continue;
            }

            String op = cmd[0];

            if (op.equals("execute")) {
                // execute 命令才有第二个元素
                if (cmd.length < 2) {
                    continue;
                }

                // 清空当前状态之后的 redo 历史
                while (history.size() > cur + 1) {
                    history.remove(history.size() - 1);
                }

                // 新建操作节点并存入历史
                history.add(cmd[1]);
                cur++;
            } else if (op.equals("undo")) {
                // 回退操作
                if (cur == -1) {
                    return "undo failed";
                }
                cur--;
            } else if (op.equals("redo")) {
                // 重做操作
                if (cur + 1 >= history.size()) {
                    return "redo failed";
                }
                cur++;
            }
        }

        // 返回当前操作状态描述
        if (cur == -1) {
            return "";
        }
        return history.get(cur);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    string execute_command(vector<vector<string>>& command) {
        vector<string> history;
        int cur = -1; // 当前索引位置，-1 表示初始状态

        for (auto& cmd : command) {
            // 兼容空命令
            if (cmd.empty()) continue;

            string op = cmd[0];

            if (op == "execute") {
                // execute 命令才有第二个元素
                if (cmd.size() < 2) continue;

                // 清空当前状态之后的 redo 历史
                while ((int)history.size() > cur + 1) {
                    history.pop_back();
                }

                // 新建操作节点并存入历史
                history.push_back(cmd[1]);
                cur++;
            } else if (op == "undo") {
                // 回退操作
                if (cur == -1) {
                    return "undo failed";
                }
                cur--;
            } else if (op == "redo") {
                // 重做操作
                if (cur + 1 >= (int)history.size()) {
                    return "redo failed";
                }
                cur++;
            }
        }

        // 返回当前操作状态描述
        if (cur == -1) return "";
        return history[cur];
    }
};
```

```c
#include <string.h>
#include <stdlib.h>

typedef struct Node {
    char* desc;
    struct Node* prev;
    struct Node* next;
} Node;

// 释放链表内存
static void _freeList(Node* head) {
    Node* p = head;
    while (p) {
        Node* tmp = p;
        p = p->next;
        free(tmp->desc);
        free(tmp);
    }
}

char* execute_command(char*** command, int commandSize, int* commandColSize) {
    // 创建虚拟头节点，表示初始状态
    Node* head = (Node*)malloc(sizeof(Node));
    head->desc = (char*)malloc(1);
    head->desc[0] = '\0';
    head->prev = NULL;
    head->next = NULL;

    Node* cur = head;

    for (int i = 0; i < commandSize; i++) {
        char* op = command[i][0];
        if (strcmp(op, "execute") == 0) {
            // 清除当前之后的所有 redo 历史
            Node* p = cur->next;
            while (p) {
                Node* tmp = p;
                p = p->next;
                free(tmp->desc);
                free(tmp);
            }
            cur->next = NULL;

            // 创建新节点保存操作描述
            Node* newNode = (Node*)malloc(sizeof(Node));
            newNode->desc = (char*)malloc(strlen(command[i][1]) + 1);
            strcpy(newNode->desc, command[i][1]);
            newNode->prev = cur;
            newNode->next = NULL;
            cur->next = newNode;
            cur = newNode;
        } else if (strcmp(op, "undo") == 0) {
            // 回退操作
            if (cur == head) {
                char* msg = (char*)malloc(13);
                strcpy(msg, "undo failed");
                _freeList(head);
                return msg;
            }
            cur = cur->prev;
        } else if (strcmp(op, "redo") == 0) {
            // 重做操作
            if (cur->next == NULL) {
                char* msg = (char*)malloc(13);
                strcpy(msg, "redo failed");
                _freeList(head);
                return msg;
            }
            cur = cur->next;
        }
    }

    // 返回当前操作描述
    char* result = (char*)malloc(strlen(cur->desc) + 1);
    strcpy(result, cur->desc);
    _freeList(head);
    return result;
}
```
