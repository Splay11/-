## 解题思路

题目要求实现函数 `classifyIPv4`，输入一个字符串形式的 IPv4 地址，返回它所属的地址类型。

先判断 IP 地址是否合法：

1. 地址必须能被 `.` 分成恰好 4 段。
2. 每一段不能为空。
3. 每一段只能包含数字字符。
4. 除了单独的 `"0"` 以外，多位数字不能有前导 `0`。
5. 每一段转换成整数后，必须在 $0$ 到 $255$ 之间。

如果合法，再根据第一段数字判断类型：

- 第一段为 $0$，返回 `"R"`。
- 第一段为 $1$ 到 $126$，返回 `"A"`。
- 第一段为 $127$，返回 `"L"`。
- 第一段为 $128$ 到 $191$，返回 `"B"`。
- 第一段为 $192$ 到 $223$，返回 `"C"`。
- 第一段为 $224$ 到 $239$ 时，虽然属于 D 类地址，但样例 4 明确要求输出 `"F"`，因此本题中返回 `"F"`。
- 第一段为 $240$ 到 $255$，返回 `"E"`。

## 复杂度分析

设输入字符串长度为 $n$。

时间复杂度为 $O(n)$，因为只需要扫描输入字符串中的每个字符常数次。

空间复杂度为 $O(1)$，IPv4 地址固定只有 4 段，额外存储空间不随输入规模增长。

```python
class Solution:
    def classifyIPv4(self, ip: str) -> str:
        # 使用点号切分字符串，IPv4 地址必须恰好有 4 段
        parts = ip.split(".")
        if len(parts) != 4:
            return "F"

        nums = []

        # 逐段检查是否合法
        for part in parts:
            # 每一段不能为空，例如 "1..1.1" 非法
            if part == "":
                return "F"

            # 每一段必须全部由数字字符组成
            for ch in part:
                if ch < "0" or ch > "9":
                    return "F"

            # 除了数字 0 本身，多位数字不允许出现前导 0
            if len(part) > 1 and part[0] == "0":
                return "F"

            # 转换为整数后，必须在 0 到 255 之间
            value = int(part)
            if value < 0 or value > 255:
                return "F"

            nums.append(value)

        # IP 地址类型只由第一段决定
        first = nums[0]

        if first == 0:
            return "R"
        if 1 <= first <= 126:
            return "A"
        if first == 127:
            return "L"
        if 128 <= first <= 191:
            return "B"
        if 192 <= first <= 223:
            return "C"

        # 按照样例要求，D 类地址输出 F
        if 224 <= first <= 239:
            return "F"

        if 240 <= first <= 255:
            return "E"

        return "F"
```

```java
class Solution {
    public String classifyIPv4(String ip) {
        if (ip == null) {
            return "F";
        }

        // 使用 -1 保留空段，例如 "1.2.3." 会切出末尾空段
        String[] parts = ip.split("\\.", -1);
        if (parts.length != 4) {
            return "F";
        }

        int[] nums = new int[4];

        // 逐段检查是否合法
        for (int i = 0; i < 4; i++) {
            String part = parts[i];

            // 每一段不能为空
            if (part.length() == 0) {
                return "F";
            }

            // 除了数字 0 本身，多位数字不允许出现前导 0
            if (part.length() > 1 && part.charAt(0) == '0') {
                return "F";
            }

            int value = 0;

            // 手动解析整数，避免非法字符导致异常
            for (int j = 0; j < part.length(); j++) {
                char ch = part.charAt(j);

                // 每一段必须全部由数字字符组成
                if (ch < '0' || ch > '9') {
                    return "F";
                }

                value = value * 10 + (ch - '0');

                // IPv4 每段最大值为 255，超过即可直接判非法
                if (value > 255) {
                    return "F";
                }
            }

            nums[i] = value;
        }

        // IP 地址类型只由第一段决定
        int first = nums[0];

        if (first == 0) {
            return "R";
        }
        if (first >= 1 && first <= 126) {
            return "A";
        }
        if (first == 127) {
            return "L";
        }
        if (first >= 128 && first <= 191) {
            return "B";
        }
        if (first >= 192 && first <= 223) {
            return "C";
        }

        // 按照样例要求，D 类地址输出 F
        if (first >= 224 && first <= 239) {
            return "F";
        }

        if (first >= 240 && first <= 255) {
            return "E";
        }

        return "F";
    }
}
```

```javascript
/**
 * classifyIPv4 - 判断 IPv4 地址的分类
 * @param {string} ip IPv4 地址字符串
 * @return {string} 地址类型："A" / "B" / "C" / "E" / "R" / "L" / "F"（不合法）
 */
var classifyIPv4 = function(ip) {
    // 按点号切分，必须恰好 4 段
    var parts = ip.split('.');
    if (parts.length !== 4) return 'F';

    var nums = [];

    // 逐段合法性检查
    for (var i = 0; i < parts.length; i++) {
        var part = parts[i];

        // 空段非法
        if (part.length === 0) return 'F';

        // 非数字字符检查
        for (var j = 0; j < part.length; j++) {
            if (part[j] < '0' || part[j] > '9') return 'F';
        }

        // 前导零检查（单独 "0" 合法，"01" 等非法）
        if (part.length > 1 && part[0] === '0') return 'F';

        // 转为整数并检查范围
        var val = parseInt(part, 10);
        if (val < 0 || val > 255) return 'F';

        nums.push(val);
    }

    // 按第一段分类
    var first = nums[0];

    if (first === 0) return 'R';
    if (first >= 1 && first <= 126) return 'A';
    if (first === 127) return 'L';
    if (first >= 128 && first <= 191) return 'B';
    if (first >= 192 && first <= 223) return 'C';
    // D 类按测试数据要求返回 "F"
    if (first >= 224 && first <= 239) return 'F';
    if (first >= 240 && first <= 255) return 'E';

    return 'F';
};
```

```c
/**
 * classifyIPv4 - 判断 IPv4 地址的分类
 * @param ip 输入的IPv4地址字符串，如"192.168.0.1"
 * @return 返回对应的类型："A"、"B"、"C"、"E"、"R"、"L"或"F"（不合法）
 */
char* classifyIPv4(char* ip) {
    int segCount = 0;       // 已完成的段数
    int num = 0;            // 当前段数值
    int numStarted = 0;     // 当前段是否已经开始解析数字
    int firstSeg = -1;      // 第一段的值

    for (int i = 0; ; i++) {
        char ch = ip[i];

        // 遇到分隔符 '.' 或字符串结束符 '\0'
        if (ch == '.' || ch == '\0') {
            // 空段非法（如 "1..1.1" 或 "1."）
            if (!numStarted) return "F";

            segCount++;
            // 记录第一段的值，用于后续分类
            if (firstSeg == -1) firstSeg = num;

            // 每段数值必须在 0~255 范围内
            if (num > 255) return "F";

            // 字符串结束，跳出循环
            if (ch == '\0') break;

            // 重置当前段状态
            num = 0;
            numStarted = 0;
        }
        // 数字字符
        else if (ch >= '0' && ch <= '9') {
            // 段首字符为 '0' 时，后一个字符不能也是数字（防止前导零）
            if (!numStarted && ch == '0' && ip[i + 1] >= '0' && ip[i + 1] <= '9')
                return "F";

            numStarted = 1;
            num = num * 10 + (ch - '0');
        }
        // 非数字字符，直接判非法
        else {
            return "F";
        }
    }

    // 必须恰好 4 段
    if (segCount != 4) return "F";

    // 按第一段范围分类
    if (firstSeg == 0) return "R";
    if (firstSeg >= 1 && firstSeg <= 126) return "A";
    if (firstSeg == 127) return "L";
    if (firstSeg >= 128 && firstSeg <= 191) return "B";
    if (firstSeg >= 192 && firstSeg <= 223) return "C";
    // D 类（224-239）按测试数据要求返回 "F"
    if (firstSeg >= 224 && firstSeg <= 239) return "F";
    if (firstSeg >= 240 && firstSeg <= 255) return "E";

    return "F";
}
```
