## 解题思路

题目要求根据前序遍历字符串和中序遍历字符串还原二叉树，再按规则删除指定节点，最后输出后序遍历结果。

首先需要校验输入是否合法：

1. 两个遍历字符串长度都必须在 $2$ 到 $26$ 之间。
2. 两个字符串只能包含大写字母。
3. 两个字符串内部都不能有重复节点。
4. 两个字符串长度必须一致。
5. 两个字符串包含的节点集合必须完全相同。
6. 根据前序和中序无法递归构造出二叉树时，返回空串。

校验通过后，利用前序遍历的第一个字符确定当前根节点，再在中序遍历区间中找到根节点位置。根节点左侧为左子树，右侧为右子树，递归构造即可。

删除节点时，需要同时记录目标节点和它的父节点：

1. 如果待删除字符不是大写字母，不删除。
2. 如果找不到待删除节点，不删除。
3. 如果待删除节点是根节点，不删除。
4. 如果待删除节点是叶子节点，直接断开父节点指针。
5. 如果待删除节点是内部节点：
   - 若它是父节点的左子节点，则令父节点的左指针指向该节点的左子树，同时丢弃该节点及其右子树。
   - 若它是父节点的右子节点，则令父节点的右指针指向该节点的右子树，同时丢弃该节点及其左子树。

最后对处理后的二叉树做后序遍历，得到答案字符串。

## 复杂度分析

设节点数为 $n$。

构造二叉树时，每个节点只会被创建一次，配合中序位置表可以在 $O(1)$ 时间找到根节点位置，因此时间复杂度为 $O(n)$。

查找待删除节点需要遍历一次二叉树，后序遍历也需要遍历一次二叉树，因此总时间复杂度为 $O(n)$。

递归构造和后序遍历的栈空间在最坏情况下为 $O(n)$，二叉树本身也需要 $O(n)$ 空间，因此空间复杂度为 $O(n)$。

class Solution:
    class TreeNode:
        def __init__(self, val: str):
            # 保存当前节点的字符值
            self.val = val
            # 保存左子节点
            self.left = None
            # 保存右子节点
            self.right = None

    def buildAfterDelete(self, preorderStr: str, inorderStr: str, beDeletedNode: str) -> str:
        # 先检查两个字符串的基础合法性
        if not self._is_valid_string(preorderStr) or not self._is_valid_string(inorderStr):
            return ""

        # 长度必须相同
        if len(preorderStr) != len(inorderStr):
            return ""

        # 节点值必须唯一
        if len(set(preorderStr)) != len(preorderStr) or len(set(inorderStr)) != len(inorderStr):
            return ""

        # 两个遍历字符串的节点集合必须完全相同
        if set(preorderStr) != set(inorderStr):
            return ""

        # 记录每个字符在中序遍历中的位置，方便递归时快速定位根节点
        inorder_pos = {}
        for i, ch in enumerate(inorderStr):
            inorder_pos[ch] = i

        # 使用列表包装标记，方便在递归过程中修改
        ok = [True]

        # 构造二叉树
        root = self._build_tree(
            preorderStr,
            0,
            len(preorderStr) - 1,
            inorderStr,
            0,
            len(inorderStr) - 1,
            inorder_pos,
            ok
        )

        # 如果无法构造出合法二叉树，返回空串
        if not ok[0] or root is None:
            return ""

        # 删除指定节点
        self._delete_node(root, beDeletedNode)

        # 生成后序遍历结果
        ans = []
        self._postorder(root, ans)
        return "".join(ans)

    def _is_valid_string(self, s: str) -> bool:
        # 长度必须在 2 到 26 之间
        if s is None or len(s) < 2 or len(s) > 26:
            return False

        # 所有字符都必须是大写字母
        for ch in s:
            if ch < "A" or ch > "Z":
                return False
        return True

    def _build_tree(self, preorder, pre_l, pre_r, inorder, in_l, in_r, inorder_pos, ok):
        # 空区间表示空子树
        if pre_l > pre_r:
            return None

        # 前序和中序区间长度必须一致
        if pre_r - pre_l != in_r - in_l:
            ok[0] = False
            return None

        # 前序区间第一个字符是当前子树根节点
        root_val = preorder[pre_l]

        # 根节点必须出现在当前中序区间内
        if root_val not in inorder_pos:
            ok[0] = False
            return None

        mid = inorder_pos[root_val]
        if mid < in_l or mid > in_r:
            ok[0] = False
            return None

        # 根据中序位置计算左子树大小
        left_size = mid - in_l

        # 左子树大小不能超过当前前序区间剩余长度
        if pre_l + left_size > pre_r:
            ok[0] = False
            return None

        root = self.TreeNode(root_val)

        # 递归构造左子树
        root.left = self._build_tree(
            preorder,
            pre_l + 1,
            pre_l + left_size,
            inorder,
            in_l,
            mid - 1,
            inorder_pos,
            ok
        )

        # 递归构造右子树
        root.right = self._build_tree(
            preorder,
            pre_l + left_size + 1,
            pre_r,
            inorder,
            mid + 1,
            in_r,
            inorder_pos,
            ok
        )

        return root

    def _delete_node(self, root, target):
        # 待删除字符不是单个大写字母时，不进行删除
        if target is None or len(target) != 1 or target < "A" or target > "Z":
            return

        # 根节点不能删除
        if root.val == target:
            return

        # 查找目标节点及其父节点
        parent, node = self._find_parent(root, target)

        # 找不到目标节点时，不进行删除
        if node is None:
            return

        # 判断目标节点是父节点的左孩子还是右孩子
        is_left_child = parent.left is node

        # 叶子节点直接删除
        if node.left is None and node.right is None:
            if is_left_child:
                parent.left = None
            else:
                parent.right = None
            return

        # 内部节点如果是左子节点，就把它的左子树挂到父节点左侧
        if is_left_child:
            parent.left = node.left
        # 内部节点如果是右子节点，就把它的右子树挂到父节点右侧
        else:
            parent.right = node.right

    def _find_parent(self, root, target):
        # 深度优先搜索目标节点，同时记录父节点
        stack = [(None, root)]

        while stack:
            parent, node = stack.pop()
            if node is None:
                continue

            if node.val == target:
                return parent, node

            stack.append((node, node.right))
            stack.append((node, node.left))

        return None, None

    def _postorder(self, node, ans):
        # 空节点不产生输出
        if node is None:
            return

        # 后序遍历顺序：左子树、右子树、根节点
        self._postorder(node.left, ans)
        self._postorder(node.right, ans)
        ans.append(node.val)

class Solution {
    static class TreeNode {
        // 当前节点的字符值
        char val;
        // 左子节点
        TreeNode left;
        // 右子节点
        TreeNode right;

        TreeNode(char val) {
            this.val = val;
        }
    }

    private boolean buildOk;

    public String buildAfterDelete(String preorderStr, String inorderStr, char beDeletedNode) {
        // 先检查两个字符串的基础合法性
        if (!isValidString(preorderStr) || !isValidString(inorderStr)) {
            return "";
        }

        // 长度必须相同
        if (preorderStr.length() != inorderStr.length()) {
            return "";
        }

        // 节点值必须唯一
        if (!isUnique(preorderStr) || !isUnique(inorderStr)) {
            return "";
        }

        // 两个遍历字符串的节点集合必须完全相同
        if (!sameSet(preorderStr, inorderStr)) {
            return "";
        }

        // 记录每个字符在中序遍历中的位置，方便递归时快速定位根节点
        int[] inorderPos = new int[256];
        for (int i = 0; i < inorderPos.length; i++) {
            inorderPos[i] = -1;
        }
        for (int i = 0; i < inorderStr.length(); i++) {
            inorderPos[inorderStr.charAt(i)] = i;
        }

        buildOk = true;

        // 构造二叉树
        TreeNode root = buildTree(
            preorderStr,
            0,
            preorderStr.length() - 1,
            inorderStr,
            0,
            inorderStr.length() - 1,
            inorderPos
        );

        // 如果无法构造出合法二叉树，返回空串
        if (!buildOk || root == null) {
            return "";
        }

        // 删除指定节点
        deleteNode(root, beDeletedNode);

        // 生成后序遍历结果
        StringBuilder ans = new StringBuilder();
        postorder(root, ans);
        return ans.toString();
    }

    private boolean isValidString(String s) {
        // 长度必须在 2 到 26 之间
        if (s == null || s.length() < 2 || s.length() > 26) {
            return false;
        }

        // 所有字符都必须是大写字母
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            if (ch < 'A' || ch > 'Z') {
                return false;
            }
        }
        return true;
    }

    private boolean isUnique(String s) {
        // 使用布尔数组判断是否存在重复字符
        boolean[] seen = new boolean[256];
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            if (seen[ch]) {
                return false;
            }
            seen[ch] = true;
        }
        return true;
    }

    private boolean sameSet(String a, String b) {
        // 由于长度已相同，只要每个字符出现集合相同即可
        boolean[] seenA = new boolean[256];
        boolean[] seenB = new boolean[256];

        for (int i = 0; i < a.length(); i++) {
            seenA[a.charAt(i)] = true;
            seenB[b.charAt(i)] = true;
        }

        for (int i = 0; i < 256; i++) {
            if (seenA[i] != seenB[i]) {
                return false;
            }
        }
        return true;
    }

    private TreeNode buildTree(
        String preorder,
        int preL,
        int preR,
        String inorder,
        int inL,
        int inR,
        int[] inorderPos
    ) {
        // 空区间表示空子树
        if (preL > preR) {
            return null;
        }

        // 前序和中序区间长度必须一致
        if (preR - preL != inR - inL) {
            buildOk = false;
            return null;
        }

        // 前序区间第一个字符是当前子树根节点
        char rootVal = preorder.charAt(preL);

        // 根节点必须出现在当前中序区间内
        int mid = inorderPos[rootVal];
        if (mid < inL || mid > inR) {
            buildOk = false;
            return null;
        }

        // 根据中序位置计算左子树大小
        int leftSize = mid - inL;

        // 左子树大小不能超过当前前序区间剩余长度
        if (preL + leftSize > preR) {
            buildOk = false;
            return null;
        }

        TreeNode root = new TreeNode(rootVal);

        // 递归构造左子树
        root.left = buildTree(
            preorder,
            preL + 1,
            preL + leftSize,
            inorder,
            inL,
            mid - 1,
            inorderPos
        );

        // 递归构造右子树
        root.right = buildTree(
            preorder,
            preL + leftSize + 1,
            preR,
            inorder,
            mid + 1,
            inR,
            inorderPos
        );

        return root;
    }

    private void deleteNode(TreeNode root, char target) {
        // 待删除字符不是大写字母时，不进行删除
        if (target < 'A' || target > 'Z') {
            return;
        }

        // 根节点不能删除
        if (root.val == target) {
            return;
        }

        // 查找目标节点及其父节点
        TreeNode[] result = findParent(root, target);
        TreeNode parent = result[0];
        TreeNode node = result[1];

        // 找不到目标节点时，不进行删除
        if (node == null) {
            return;
        }

        // 判断目标节点是父节点的左孩子还是右孩子
        boolean isLeftChild = parent.left == node;

        // 叶子节点直接删除
        if (node.left == null && node.right == null) {
            if (isLeftChild) {
                parent.left = null;
            } else {
                parent.right = null;
            }
            return;
        }

        // 内部节点如果是左子节点，就把它的左子树挂到父节点左侧
        if (isLeftChild) {
            parent.left = node.left;
        }
        // 内部节点如果是右子节点，就把它的右子树挂到父节点右侧
        else {
            parent.right = node.right;
        }
    }

    private TreeNode[] findParent(TreeNode root, char target) {
        // 深度优先搜索目标节点，同时记录父节点
        Deque<TreeNode[]> stack = new ArrayDeque<>();
        stack.push(new TreeNode[]{null, root});

        while (!stack.isEmpty()) {
            TreeNode[] current = stack.pop();
            TreeNode parent = current[0];
            TreeNode node = current[1];

            if (node == null) {
                continue;
            }

            if (node.val == target) {
                return new TreeNode[]{parent, node};
            }

            stack.push(new TreeNode[]{node, node.right});
            stack.push(new TreeNode[]{node, node.left});
        }

        return new TreeNode[]{null, null};
    }

    private void postorder(TreeNode node, StringBuilder ans) {
        // 空节点不产生输出
        if (node == null) {
            return;
        }

        // 后序遍历顺序：左子树、右子树、根节点
        postorder(node.left, ans);
        postorder(node.right, ans);
        ans.append(node.val);
    }
}

class Solution {
private:
    struct TreeNode {
        // 当前节点的字符值
        char val;
        // 左子节点
        TreeNode* left;
        // 右子节点
        TreeNode* right;

        TreeNode(char v) {
            val = v;
            left = nullptr;
            right = nullptr;
        }
    };

    bool buildOk;

public:
    string buildAfterDelete(string preorderStr, string inorderStr, char beDeletedNode) {
        // 先检查两个字符串的基础合法性
        if (!isValidString(preorderStr) || !isValidString(inorderStr)) {
            return "";
        }

        // 长度必须相同
        if (preorderStr.size() != inorderStr.size()) {
            return "";
        }

        // 节点值必须唯一
        if (!isUnique(preorderStr) || !isUnique(inorderStr)) {
            return "";
        }

        // 两个遍历字符串的节点集合必须完全相同
        if (!sameSet(preorderStr, inorderStr)) {
            return "";
        }

        // 记录每个字符在中序遍历中的位置，方便递归时快速定位根节点
        vector<int> inorderPos(256, -1);
        for (int i = 0; i < (int)inorderStr.size(); i++) {
            inorderPos[(unsigned char)inorderStr[i]] = i;
        }

        buildOk = true;

        // 构造二叉树
        TreeNode* root = buildTree(
            preorderStr,
            0,
            (int)preorderStr.size() - 1,
            inorderStr,
            0,
            (int)inorderStr.size() - 1,
            inorderPos
        );

        // 如果无法构造出合法二叉树，返回空串
        if (!buildOk || root == nullptr) {
            clearTree(root);
            return "";
        }

        // 删除指定节点
        deleteNode(root, beDeletedNode);

        // 生成后序遍历结果
        string ans;
        postorder(root, ans);

        // 释放剩余二叉树节点
        clearTree(root);

        return ans;
    }

private:
    bool isValidString(const string& s) {
        // 长度必须在 2 到 26 之间
        if ((int)s.size() < 2 || (int)s.size() > 26) {
            return false;
        }

        // 所有字符都必须是大写字母
        for (char ch : s) {
            if (ch < 'A' || ch > 'Z') {
                return false;
            }
        }
        return true;
    }

    bool isUnique(const string& s) {
        // 使用布尔数组判断是否存在重复字符
        vector<bool> seen(256, false);

        for (char ch : s) {
            unsigned char idx = (unsigned char)ch;
            if (seen[idx]) {
                return false;
            }
            seen[idx] = true;
        }
        return true;
    }

    bool sameSet(const string& a, const string& b) {
        // 由于长度已相同，只要每个字符出现集合相同即可
        vector<bool> seenA(256, false);
        vector<bool> seenB(256, false);

        for (char ch : a) {
            seenA[(unsigned char)ch] = true;
        }
        for (char ch : b) {
            seenB[(unsigned char)ch] = true;
        }

        for (int i = 0; i < 256; i++) {
            if (seenA[i] != seenB[i]) {
                return false;
            }
        }
        return true;
    }

    TreeNode* buildTree(
        const string& preorder,
        int preL,
        int preR,
        const string& inorder,
        int inL,
        int inR,
        const vector<int>& inorderPos
    ) {
        // 空区间表示空子树
        if (preL > preR) {
            return nullptr;
        }

        // 前序和中序区间长度必须一致
        if (preR - preL != inR - inL) {
            buildOk = false;
            return nullptr;
        }

        // 前序区间第一个字符是当前子树根节点
        char rootVal = preorder[preL];

        // 根节点必须出现在当前中序区间内
        int mid = inorderPos[(unsigned char)rootVal];
        if (mid < inL || mid > inR) {
            buildOk = false;
            return nullptr;
        }

        // 根据中序位置计算左子树大小
        int leftSize = mid - inL;

        // 左子树大小不能超过当前前序区间剩余长度
        if (preL + leftSize > preR) {
            buildOk = false;
            return nullptr;
        }

        TreeNode* root = new TreeNode(rootVal);

        // 递归构造左子树
        root->left = buildTree(
            preorder,
            preL + 1,
            preL + leftSize,
            inorder,
            inL,
            mid - 1,
            inorderPos
        );

        // 递归构造右子树
        root->right = buildTree(
            preorder,
            preL + leftSize + 1,
            preR,
            inorder,
            mid + 1,
            inR,
            inorderPos
        );

        return root;
    }

    void deleteNode(TreeNode* root, char target) {
        // 待删除字符不是大写字母时，不进行删除
        if (target < 'A' || target > 'Z') {
            return;
        }

        // 根节点不能删除
        if (root->val == target) {
            return;
        }

        // 查找目标节点及其父节点
        pair<TreeNode*, TreeNode*> result = findParent(root, target);
        TreeNode* parent = result.first;
        TreeNode* node = result.second;

        // 找不到目标节点时，不进行删除
        if (node == nullptr) {
            return;
        }

        // 判断目标节点是父节点的左孩子还是右孩子
        bool isLeftChild = parent->left == node;

        // 叶子节点直接删除
        if (node->left == nullptr && node->right == nullptr) {
            if (isLeftChild) {
                parent->left = nullptr;
            } else {
                parent->right = nullptr;
            }
            clearTree(node);
            return;
        }

        // 内部节点如果是左子节点，就保留它的左子树，丢弃它和右子树
        if (isLeftChild) {
            TreeNode* keep = node->left;
            node->left = nullptr;
            clearTree(node);
            parent->left = keep;
        }
        // 内部节点如果是右子节点，就保留它的右子树，丢弃它和左子树
        else {
            TreeNode* keep = node->right;
            node->right = nullptr;
            clearTree(node);
            parent->right = keep;
        }
    }

    pair<TreeNode*, TreeNode*> findParent(TreeNode* root, char target) {
        // 深度优先搜索目标节点，同时记录父节点
        stack<pair<TreeNode*, TreeNode*>> st;
        st.push({nullptr, root});

        while (!st.empty()) {
            auto current = st.top();
            st.pop();

            TreeNode* parent = current.first;
            TreeNode* node = current.second;

            if (node == nullptr) {
                continue;
            }

            if (node->val == target) {
                return {parent, node};
            }

            st.push({node, node->right});
            st.push({node, node->left});
        }

        return {nullptr, nullptr};
    }

    void postorder(TreeNode* node, string& ans) {
        // 空节点不产生输出
        if (node == nullptr) {
            return;
        }

        // 后序遍历顺序：左子树、右子树、根节点
        postorder(node->left, ans);
        postorder(node->right, ans);
        ans.push_back(node->val);
    }

    void clearTree(TreeNode* node) {
        // 递归释放二叉树节点
        if (node == nullptr) {
            return;
        }

        clearTree(node->left);
        clearTree(node->right);
        delete node;
    }
};

/**
 * TreeNode - 二叉树节点
 */
function TreeNode(val) {
    this.val = val;
    this.left = null;
    this.right = null;
}

/**
 * buildAfterDelete - 构造二叉树并删除指定节点，返回后序遍历结果
 * @param {string} preorderStr 前序遍历字符串
 * @param {string} inorderStr 中序遍历字符串
 * @param {string} beDeletedNode 待删除节点字符
 * @return {string} 后序遍历结果
 */
var buildAfterDelete = function(preorderStr, inorderStr, beDeletedNode) {
    // 字符串合法性检查：长度必须在 2 到 26 之间，且只能包含大写字母
    function isValid(s) {
        if (!s || s.length < 2 || s.length > 26) return false;
        for (var i = 0; i < s.length; i++) {
            if (s[i] < 'A' || s[i] > 'Z') return false;
        }
        return true;
    }
    if (!isValid(preorderStr) || !isValid(inorderStr)) return "";

    // 长度必须相同
    if (preorderStr.length !== inorderStr.length) return "";

    // 节点值必须唯一
    function isUnique(s) {
        var seen = {};
        for (var i = 0; i < s.length; i++) {
            if (seen[s[i]]) return false;
            seen[s[i]] = true;
        }
        return true;
    }
    if (!isUnique(preorderStr) || !isUnique(inorderStr)) return "";

    // 两个字符串的节点集合必须完全相同
    var setA = {}, setB = {};
    for (var i = 0; i < preorderStr.length; i++) setA[preorderStr[i]] = true;
    for (var i = 0; i < inorderStr.length; i++) setB[inorderStr[i]] = true;
    for (var k in setA) { if (!setB[k]) return ""; setB[k] = false; }
    for (var k in setB) { if (setB[k]) return ""; }

    // 建立中序遍历位置映射
    var inorderPos = {};
    for (var i = 0; i < inorderStr.length; i++) {
        inorderPos[inorderStr[i]] = i;
    }

    var ok = true;

    // 递归构造二叉树
    function buildTree(pre, preL, preR, inOrder, inL, inR) {
        if (preL > preR) return null;
        if (preR - preL !== inR - inL) { ok = false; return null; }

        var rootVal = pre[preL];
        var mid = inorderPos[rootVal];
        if (mid === undefined || mid < inL || mid > inR) { ok = false; return null; }

        var leftSize = mid - inL;
        var root = new TreeNode(rootVal);
        root.left = buildTree(pre, preL + 1, preL + leftSize, inOrder, inL, mid - 1);
        root.right = buildTree(pre, preL + leftSize + 1, preR, inOrder, mid + 1, inR);
        return root;
    }

    var root = buildTree(preorderStr, 0, preorderStr.length - 1, inorderStr, 0, inorderStr.length - 1);
    if (!ok || !root) return "";

    // 删除指定节点
    var target = beDeletedNode;
    if (target && target.length === 1 && target >= 'A' && target <= 'Z' && root.val !== target) {
        // 查找目标节点及其父节点
        function findParent(root, target) {
            var stack = [[null, root]];
            while (stack.length > 0) {
                var pair = stack.pop();
                var parent = pair[0], node = pair[1];
                if (!node) continue;
                if (node.val === target) return { parent: parent, node: node };
                stack.push([node, node.right]);
                stack.push([node, node.left]);
            }
            return { parent: null, node: null };
        }

        var res = findParent(root, target);
        var parent = res.parent, node = res.node;

        if (node) {
            var isLeftChild = parent.left === node;
            if (!node.left && !node.right) {
                // 叶子节点：直接删除
                if (isLeftChild) parent.left = null;
                else parent.right = null;
            } else {
                // 内部节点：保留对应子树
                if (isLeftChild) parent.left = node.left;
                else parent.right = node.right;
            }
        }
    }

    // 后序遍历
    function postorder(node, arr) {
        if (!node) return;
        postorder(node.left, arr);
        postorder(node.right, arr);
        arr.push(node.val);
    }

    var result = [];
    postorder(root, result);
    return result.join('');
};

#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

// 二叉树节点结构
typedef struct TreeNode {
    char val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 创建新节点
TreeNode* createNode(char val) {
    TreeNode* node = (TreeNode*)malloc(sizeof(TreeNode));
    node->val = val;
    node->left = NULL;
    node->right = NULL;
    return node;
}

// 递归释放二叉树
void freeTree(TreeNode* node) {
    if (node == NULL) return;
    freeTree(node->left);
    freeTree(node->right);
    free(node);
}

// 查找目标节点及其父节点（栈式迭代 DFS）
void findParent(TreeNode* root, char target, TreeNode** parent, TreeNode** node) {
    TreeNode* stack[100];
    TreeNode* pstack[100];
    int top = 0;
    stack[top] = root;
    pstack[top] = NULL;
    top++;

    while (top > 0) {
        top--;
        TreeNode* cur = stack[top];
        TreeNode* par = pstack[top];
        if (cur == NULL) continue;
        if (cur->val == target) { *parent = par; *node = cur; return; }
        stack[top] = cur->right; pstack[top] = cur; top++;
        stack[top] = cur->left; pstack[top] = cur; top++;
    }
    *parent = NULL; *node = NULL;
}

// 删除指定节点
void deleteNode(TreeNode* root, const char* beDeletedNode) {
    if (beDeletedNode == NULL || strlen(beDeletedNode) != 1) return;
    char target = beDeletedNode[0];
    if (target < 'A' || target > 'Z') return;
    if (root->val == target) return;

    TreeNode *parent = NULL, *node = NULL;
    findParent(root, target, &parent, &node);
    if (node == NULL) return;

    bool isLeftChild = (parent->left == node);

    if (node->left == NULL && node->right == NULL) {
        // 叶子节点：直接删除
        if (isLeftChild) parent->left = NULL;
        else parent->right = NULL;
        free(node);
        return;
    }

    // 内部节点：保留对应子树，丢弃该节点
    if (isLeftChild) {
        parent->left = node->left;
        node->left = NULL;
    } else {
        parent->right = node->right;
        node->right = NULL;
    }
    freeTree(node);
}

// 后序遍历
void postorder(TreeNode* node, char* buf, int* idx) {
    if (node == NULL) return;
    postorder(node->left, buf, idx);
    postorder(node->right, buf, idx);
    buf[(*idx)++] = node->val;
}

// 合法性检查
bool isValidString(const char* s) {
    if (s == NULL) return false;
    int len = strlen(s);
    if (len < 2 || len > 26) return false;
    for (int i = 0; i < len; i++)
        if (s[i] < 'A' || s[i] > 'Z') return false;
    return true;
}

bool isUniqueStr(const char* s) {
    bool seen[128] = {false};
    for (int i = 0; s[i]; i++) {
        unsigned char c = (unsigned char)s[i];
        if (seen[c]) return false;
        seen[c] = true;
    }
    return true;
}

bool sameSet(const char* a, const char* b) {
    bool setA[128] = {false}, setB[128] = {false};
    for (int i = 0; a[i]; i++) setA[(unsigned char)a[i]] = true;
    for (int i = 0; b[i]; i++) setB[(unsigned char)b[i]] = true;
    for (int i = 0; i < 128; i++)
        if (setA[i] != setB[i]) return false;
    return true;
}

static bool buildOk;

// 递归构造二叉树
TreeNode* buildTree(const char* preorder, int preL, int preR,
                    const char* inorder, int inL, int inR, int* inorderPos) {
    if (preL > preR) return NULL;
    if (preR - preL != inR - inL) { buildOk = false; return NULL; }

    char rootVal = preorder[preL];
    int mid = inorderPos[(unsigned char)rootVal];
    if (mid < inL || mid > inR || mid < 0) { buildOk = false; return NULL; }

    int leftSize = mid - inL;
    TreeNode* root = createNode(rootVal);
    root->left = buildTree(preorder, preL + 1, preL + leftSize, inorder, inL, mid - 1, inorderPos);
    root->right = buildTree(preorder, preL + leftSize + 1, preR, inorder, mid + 1, inR, inorderPos);
    return root;
}

/**
 * buildAfterDelete - 构造二叉树并删除指定节点，返回后序遍历结果
 */
char* buildAfterDelete(char* preorderStr, char* inorderStr, char* beDeletedNode) {
    if (!isValidString(preorderStr) || !isValidString(inorderStr))
        return strdup("");
    if (strlen(preorderStr) != strlen(inorderStr)) return strdup("");
    if (!isUniqueStr(preorderStr) || !isUniqueStr(inorderStr)) return strdup("");
    if (!sameSet(preorderStr, inorderStr)) return strdup("");

    int inorderPos[256], n = strlen(inorderStr);
    for (int i = 0; i < 256; i++) inorderPos[i] = -1;
    for (int i = 0; i < n; i++) inorderPos[(unsigned char)inorderStr[i]] = i;

    buildOk = true;
    TreeNode* root = buildTree(preorderStr, 0, n - 1, inorderStr, 0, n - 1, inorderPos);
    if (!buildOk || root == NULL) return strdup("");

    deleteNode(root, beDeletedNode);

    char* ans = (char*)malloc(n + 1);
    int idx = 0;
    postorder(root, ans, &idx);
    ans[idx] = '\0';
    freeTree(root);
    return ans;
}
