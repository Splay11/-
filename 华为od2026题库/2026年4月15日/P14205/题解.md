### 解题思路

这题本质上是一次**线性扫描 + 贪心判断**。

因为题目已经明确规定：

* 连续两个 `u` 一定表示按了 1 次 `j`
* 连续两个 `t` 一定表示按了 1 次 `b`
* 不会出现 `u` 后紧跟 `j`、`t` 后紧跟 `b` 这种会造成歧义的情况

所以我们从左到右扫描字符串时：

* 如果当前位置和下一位组成 `"uu"`，就统计一次按键 `j`
* 如果当前位置和下一位组成 `"tt"`，就统计一次按键 `b`
* 否则，当前位置字符本身就是正常按下的按键，直接统计该字符

这就是一个典型的**双指针/模拟**思路：

* 指针 `i` 表示当前处理到的位置
* 若匹配到 `"uu"` 或 `"tt"`，则 `i += 2`
* 否则 `i += 1`

统计出每个按键的次数后，再按照题目要求排序：

1. 按键次数降序
2. 次数相同，按键字符升序（这里失灵按键按原字符 `b`、`j` 参与排序）

最后做一次转义输出：

* `'0' ~ '9'` → `0 ~ 9`
* `'a' ~ 'z'` → `10 ~ 35`

因此：

* 数字字符 `c` 的映射值是 `c - '0'`
* 字母字符 `c` 的映射值是 `c - 'a' + 10`

### 复杂度分析

设字符串长度为 `n`。

#### 时间复杂度

* 扫描字符串一次：`O(n)`
* 统计的按键种类最多只有 36 个（`a-z` + `0-9`），排序复杂度是 `O(36 log 36)`，可视为常数

所以总时间复杂度为：

**O(n)**

#### 空间复杂度

* 统计按键次数最多只会存 36 种字符

所以空间复杂度为：

**O(1)**

因为键盘范围固定，额外空间是常数级。

```python
from typing import List
from collections import Counter

class Solution:
    def countKeys(self, s: str) -> List[List[int]]:
        # 统计每个按键被按下的次数
        cnt = Counter()
        i = 0
        n = len(s)

        while i < n:
            # 优先匹配失灵按键对应的转义串
            if i + 1 < n and s[i] == 'u' and s[i + 1] == 'u':
                cnt['j'] += 1
                i += 2
            elif i + 1 < n and s[i] == 't' and s[i + 1] == 't':
                cnt['b'] += 1
                i += 2
            else:
                # 否则就是正常按键
                cnt[s[i]] += 1
                i += 1

        # 字符转义函数
        def encode(c: str) -> int:
            if '0' <= c <= '9':
                return ord(c) - ord('0')
            return ord(c) - ord('a') + 10

        # 按次数降序，字符升序排序
        keys = sorted(cnt.items(), key=lambda x: (-x[1], x[0]))

        # 生成结果
        ans = []
        for c, num in keys:
            ans.append([encode(c), num])

        return ans
```

```java
import java.util.*;

class Solution {
    public List<List<Integer>> countKeys(String s) {
        // 统计每个按键被按下的次数
        Map<Character, Integer> cnt = new HashMap<>();
        int i = 0, n = s.length();

        while (i < n) {
            // 优先匹配失灵按键对应的转义串
            if (i + 1 < n && s.charAt(i) == 'u' && s.charAt(i + 1) == 'u') {
                cnt.put('j', cnt.getOrDefault('j', 0) + 1);
                i += 2;
            } else if (i + 1 < n && s.charAt(i) == 't' && s.charAt(i + 1) == 't') {
                cnt.put('b', cnt.getOrDefault('b', 0) + 1);
                i += 2;
            } else {
                // 否则就是正常按键
                char c = s.charAt(i);
                cnt.put(c, cnt.getOrDefault(c, 0) + 1);
                i++;
            }
        }

        // 转成列表后排序
        List<Character> list = new ArrayList<>(cnt.keySet());
        Collections.sort(list, (a, b) -> {
            int ca = cnt.get(a), cb = cnt.get(b);
            if (ca != cb) {
                return cb - ca; // 次数降序
            }
            return a - b; // 字符升序
        });

        // 构造答案
        List<List<Integer>> ans = new ArrayList<>();
        for (char c : list) {
            List<Integer> pair = new ArrayList<>();
            pair.add(encode(c));
            pair.add(cnt.get(c));
            ans.add(pair);
        }

        return ans;
    }

    // 字符转义
    private int encode(char c) {
        if (c >= '0' && c <= '9') {
            return c - '0';
        }
        return c - 'a' + 10;
    }
}
```

```cpp
#include <vector>
#include <string>
#include <unordered_map>
#include <algorithm>
using namespace std;

class Solution {
public:
    vector<vector<int>> countKeys(string s) {
        // 统计每个按键被按下的次数
        unordered_map<char, int> cnt;
        int i = 0, n = s.size();

        while (i < n) {
            // 优先匹配失灵按键对应的转义串
            if (i + 1 < n && s[i] == 'u' && s[i + 1] == 'u') {
                cnt['j']++;
                i += 2;
            } else if (i + 1 < n && s[i] == 't' && s[i + 1] == 't') {
                cnt['b']++;
                i += 2;
            } else {
                // 否则就是正常按键
                cnt[s[i]]++;
                i++;
            }
        }

        // 提取所有出现过的按键
        vector<pair<char, int>> arr;
        for (auto &p : cnt) {
            arr.push_back(p);
        }

        // 按次数降序，字符升序排序
        sort(arr.begin(), arr.end(), [](const pair<char, int>& a, const pair<char, int>& b) {
            if (a.second != b.second) {
                return a.second > b.second;
            }
            return a.first < b.first;
        });

        // 构造答案
        vector<vector<int>> ans;
        for (auto &p : arr) {
            ans.push_back({encode(p.first), p.second});
        }

        return ans;
    }

private:
    // 字符转义
    int encode(char c) {
        if (c >= '0' && c <= '9') {
            return c - '0';
        }
        return c - 'a' + 10;
    }
};
```

```javascript
var countKeys = function(s) {
    // cnt[encoded] = count
    var cnt = new Array(36).fill(0);
    var n = s.length;
    var i = 0;

    // 解析按键
    while (i < n) {
        if (i + 1 < n && s[i] === 'u' && s[i + 1] === 'u') {
            // uu → j (encoded as 19)
            cnt[19]++;
            i += 2;
        } else if (i + 1 < n && s[i] === 't' && s[i + 1] === 't') {
            // tt → b (encoded as 11)
            cnt[11]++;
            i += 2;
        } else {
            // 普通按键
            var c = s[i];
            var enc;
            if (c >= '0' && c <= '9') {
                enc = c.charCodeAt(0) - 48; // '0' = 48
            } else {
                enc = c.charCodeAt(0) - 97 + 10; // 'a' = 97
            }
            cnt[enc]++;
            i++;
        }
    }

    // 收集按键，按次数降序排序
    var result = [];
    for (var k = 0; k < 36; k++) {
        if (cnt[k] > 0) {
            result.push([k, cnt[k]]);
        }
    }

    // 按次数降序，同次数按键值升序
    result.sort(function(a, b) {
        if (b[1] !== a[1]) return b[1] - a[1];
        return a[0] - b[0];
    });

    return result;
};
```

```c
#include <stdlib.h>
#include <string.h>

int** countKeys(char* s, int* returnSize) {
    int cnt[36] = {0};
    int n = strlen(s);
    int i = 0;

    // 解析按键
    while (i < n) {
        if (i + 1 < n && s[i] == 'u' && s[i + 1] == 'u') {
            // uu → j (encoded as 19)
            cnt[19]++;
            i += 2;
        } else if (i + 1 < n && s[i] == 't' && s[i + 1] == 't') {
            // tt → b (encoded as 11)
            cnt[11]++;
            i += 2;
        } else {
            // 普通按键
            char c = s[i];
            int enc;
            if (c >= '0' && c <= '9') {
                enc = c - '0';
            } else {
                enc = c - 'a' + 10;
            }
            cnt[enc]++;
            i++;
        }
    }

    // 统计有效按键数
    int num = 0;
    for (int k = 0; k < 36; k++) {
        if (cnt[k] > 0) num++;
    }

    // 分配结果数组
    int** res = (int**)malloc(num * sizeof(int*));
    int idx = 0;
    // 按次数降序，同次数按键值升序
    for (int freq = 500; freq >= 1; freq--) {
        for (int k = 0; k < 36; k++) {
            if (cnt[k] == freq) {
                res[idx] = (int*)malloc(2 * sizeof(int));
                res[idx][0] = k;
                res[idx][1] = freq;
                idx++;
            }
        }
    }

    *returnSize = num;
    return res;
}
```
