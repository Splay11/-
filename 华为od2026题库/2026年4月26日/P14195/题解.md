## 解题思路

题意给出模块集合与依赖列表，每条依赖 **`[x, y]` 表示 $x$ 依赖 $y$**，即合法构建顺序中 **$y$ 必须出现在 $x$ 的左侧**。在图论上，将每条依赖建成有向边 **$y \rightarrow x$**（先构建 $y$，再构建 $x$），问题转化为求 **所有拓扑排序**。

- **判环**：若图存在有向环，则不存在能包含全部顶点的拓扑序，应返回空数组。可用 **Kahn 算法**（不断删除入度为 $0$ 的顶点并删边）统计能否删完 $N$ 个顶点；若不能，说明有环。
- **枚举所有拓扑序**：在确认无环后，用 **DFS + 回溯** 维护当前路径 `path` 与动态入度表 `indeg`：
  - 每一步在「尚未放入路径且当前入度为 $0$」的模块中，按 **模块名字典序** 依次尝试（保证搜索顺序稳定，但最终仍需对生成的整行字符串列表排序，与题意「多条结果按字典序」一致）。
  - 选中模块 $m$ 后，将其加入路径，并对所有 $m \rightarrow v$ 的边执行 `indeg[v]--`；回溯时恢复。
  - 当路径长度达到 $N$ 时，将 `" ".join(path)` 加入答案。
- **输出格式**：将所有完整顺序字符串收集后 **`sorted`** 一次，保证与题面「按字典序排序后输出」一致。

**区分度 / 易错点**：依赖方向不要反（应建 $y \rightarrow x$）；有环时不能只靠 DFS「走不满 $N$」就当作无输出——必须先做全局可拓扑性判定；多解时必须对**结果字符串**排序，而不是只对每一步候选排序。

**测试设计简述**：样例与官方一致；另含链、菱形、无依赖全排列、并行汇合、单点/双边等小数据；在题面保证「合法拓扑序数量 $<10!$」前提下控制规模，避免指数爆炸超出评测资源。

## 复杂度分析

设 $N$ 为模块数，$M$ 为依赖条数，$K$ 为合法拓扑序条数（题面保证 $K < 10!$）。

- **建图与判环（Kahn）**：时间 $O(N+M)$，空间 $O(N+M)$。
- **枚举所有拓扑序**：搜索树规模与 $K$ 及 DAG 结构相关；在题面保证下可接受。实现上每步需扫描入度为 $0$ 的候选，粗略上界可写为 $O(K \cdot N^2)$ 的保守估计；数据保证 $K$ 不大，实际可过。
- **空间**：除邻接表外，递归栈深度 $O(N)$，路径数组 $O(N)$。

```python
from collections import defaultdict, deque
from typing import Dict, List

class Solution:
    def allBuildOrders(self, modules: List[str], dependencies: List[List[str]]) -> List[str]:
        if not modules:
            return []
        # indeg[x]：当前图中还有多少条依赖指向 x
        indeg: Dict[str, int] = {m: 0 for m in modules}
        # g[u]：所有从 u 出发的边（即 u 必须先于其中的节点）
        g: Dict[str, List[str]] = defaultdict(list)
        for a, b in dependencies:
            if a not in indeg or b not in indeg:
                continue
            g[b].append(a)
            indeg[a] += 1

        # 使用 Kahn 算法判环
        q = deque([m for m in modules if indeg[m] == 0])
        tmp = indeg.copy()
        seen = 0
        while q:
            u = q.popleft()
            seen += 1
            for v in g[u]:
                tmp[v] -= 1
                if tmp[v] == 0:
                    q.append(v)
        if seen != len(modules):
            # 存在环，无合法构建顺序
            return []

        indeg_mut = indeg.copy()
        path: List[str] = []  # 当前构建路径
        out: List[str] = []   # 所有合法顺序结果

        def dfs() -> None:
            if len(path) == len(modules):
                # 完成一种构建顺序，加入结果
                out.append(" ".join(path))
                return
            # 找出当前所有可选（入度为0且未使用）的模块
            cand = sorted([m for m in modules if m not in path and indeg_mut[m] == 0])
            for m in cand:
                path.append(m)
                for nei in g[m]:
                    indeg_mut[nei] -= 1
                dfs()
                for nei in g[m]:
                    indeg_mut[nei] += 1
                path.pop()

        dfs()
        return sorted(out)
```

```java
import java.util.*;

class Solution {
    public List<String> allBuildOrders(List<String> modules, List<List<String>> dependencies) {
        if (modules.isEmpty()) {
            return new ArrayList<>();
        }
        Map<String, Integer> indeg = new HashMap<>();
        for (String m : modules) {
            indeg.put(m, 0);
        }
        Map<String, List<String>> g = new HashMap<>();
        for (List<String> e : dependencies) {
            String a = e.get(0), b = e.get(1);
            if (!indeg.containsKey(a) || !indeg.containsKey(b)) {
                continue;
            }
            g.computeIfAbsent(b, k -> new ArrayList<>()).add(a);
            indeg.put(a, indeg.get(a) + 1);
        }

        // Kahn 判环
        Map<String, Integer> tmp = new HashMap<>(indeg);
        Deque<String> q = new ArrayDeque<>();
        for (String m : modules) {
            if (tmp.get(m) == 0) q.addLast(m);
        }
        int seen = 0;
        while (!q.isEmpty()) {
            String u = q.removeFirst();
            seen++;
            for (String v : g.getOrDefault(u, new ArrayList<>())) {
                tmp.put(v, tmp.get(v) - 1);
                if (tmp.get(v) == 0) q.addLast(v);
            }
        }
        if (seen != modules.size()) return new ArrayList<>();

        Map<String, Integer> indegMut = new HashMap<>(indeg);
        List<String> path = new ArrayList<>();
        List<String> out = new ArrayList<>();

        dfsAll(modules, g, indegMut, path, out);
        Collections.sort(out);
        return out;
    }

    private void dfsAll(List<String> modules, Map<String, List<String>> g,
                        Map<String, Integer> indegMut, List<String> path, List<String> out) {
        if (path.size() == modules.size()) {
            out.add(String.join(" ", path));
            return;
        }
        Set<String> inPath = new HashSet<>(path);
        List<String> cand = new ArrayList<>();
        for (String m : modules) {
            if (!inPath.contains(m) && indegMut.get(m) == 0) {
                cand.add(m);
            }
        }
        Collections.sort(cand);
        for (String m : cand) {
            path.add(m);
            for (String nei : g.getOrDefault(m, new ArrayList<>())) {
                indegMut.put(nei, indegMut.get(nei) - 1);
            }
            dfsAll(modules, g, indegMut, path, out);
            for (String nei : g.getOrDefault(m, new ArrayList<>())) {
                indegMut.put(nei, indegMut.get(nei) + 1);
            }
            path.remove(path.size() - 1);
        }
    }
}
```

```cpp
#include <algorithm>
#include <queue>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <vector>
using namespace std;

class Solution {
public:
    vector<string> allBuildOrders(vector<string>& modules, vector<vector<string>>& dependencies) {
        if (modules.empty()) return {};
        unordered_map<string, int> indeg;
        unordered_map<string, vector<string>> g;
        for (auto& m : modules) indeg[m] = 0;
        for (auto& e : dependencies) {
            const string& a = e[0];
            const string& b = e[1];
            if (!indeg.count(a) || !indeg.count(b)) continue;
            g[b].push_back(a);
            indeg[a]++;
        }

        // Kahn 判环
        auto tmp = indeg;
        queue<string> q;
        for (auto& m : modules)
            if (tmp[m] == 0) q.push(m);
        int seen = 0;
        while (!q.empty()) {
            string u = q.front(); q.pop();
            seen++;
            for (auto& v : g[u])
                if (--tmp[v] == 0) q.push(v);
        }
        if (seen != (int)modules.size()) return {};

        unordered_map<string, int> indegMut = indeg;
        vector<string> path, out;
        dfsAll(modules, g, indegMut, path, out);
        sort(out.begin(), out.end());
        return out;
    }

private:
    void dfsAll(vector<string>& modules, unordered_map<string, vector<string>>& g,
                unordered_map<string, int>& indegMut, vector<string>& path, vector<string>& out) {
        if ((int)path.size() == (int)modules.size()) {
            string line;
            for (size_t i = 0; i < path.size(); i++) {
                if (i) line += ' ';
                line += path[i];
            }
            out.push_back(line);
            return;
        }
        unordered_set<string> inPath(path.begin(), path.end());
        vector<string> cand;
        for (auto& m : modules)
            if (!inPath.count(m) && indegMut[m] == 0) cand.push_back(m);
        sort(cand.begin(), cand.end());
        for (auto& m : cand) {
            path.push_back(m);
            for (auto& nei : g[m]) indegMut[nei]--;
            dfsAll(modules, g, indegMut, path, out);
            for (auto& nei : g[m]) indegMut[nei]++;
            path.pop_back();
        }
    }
};
```

```javascript
/**
 * @param {string[]} modules 模块名称数组
 * @param {string[][]} dependencies 依赖关系 [x, y] 表示 x 依赖 y
 * @return {string[]} 所有合法构建顺序（以空格分隔），按字典序排序
 */
var allBuildOrders = function(modules, dependencies) {
    const n = modules.length;
    const nameToIdx = new Map();
    modules.forEach((m, i) => nameToIdx.set(m, i));

    // 建图
    const adj = Array.from({ length: n }, () => []);
    const indeg = new Array(n).fill(0);
    for (const [a, b] of dependencies) {
        const bi = nameToIdx.get(b);
        const ai = nameToIdx.get(a);
        if (bi != null && ai != null) {
            adj[bi].push(ai);
            indeg[ai]++;
        }
    }

    // 判环
    const indegCopy = [...indeg];
    const q = [];
    for (let i = 0; i < n; i++) if (indegCopy[i] === 0) q.push(i);
    let visited = 0;
    while (q.length) {
        const u = q.shift();
        visited++;
        for (const v of adj[u]) if (--indegCopy[v] === 0) q.push(v);
    }
    if (visited < n) return [];

    // DFS 枚举所有拓扑序
    const results = [];
    const path = [];
    const curIndeg = [...indeg];

    function dfs() {
        if (path.length === n) {
            results.push(path.map(i => modules[i]).join(' '));
            return;
        }
        for (let i = 0; i < n; i++) {
            if (curIndeg[i] !== 0) continue;
            curIndeg[i] = -1;
            path.push(i);
            for (const v of adj[i]) curIndeg[v]--;
            dfs();
            for (const v of adj[i]) curIndeg[v]++;
            path.pop();
            curIndeg[i] = 0;
        }
    }

    dfs();
    return results.sort();
};
```

```c
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>

#define MAX_N 100
#define MAX_RESULT 50000

static int _n;
static int _adj[MAX_N][MAX_N];
static int _adjCnt[MAX_N];
static int _indeg[MAX_N];
static char* _names[MAX_N];

static char* _results[MAX_RESULT];
static int _resultCnt;
static int _path[MAX_N];
static int _pathLen;

static int _cmp_result(const void* a, const void* b) {
    return strcmp(*(const char**)a, *(const char**)b);
}

// 深度优先枚举所有拓扑排序
static void _dfs(int* indeg) {
    if (_pathLen == _n) {
        int totalLen = 0;
        for (int i = 0; i < _n; i++)
            totalLen += strlen(_names[_path[i]]) + 1;
        _results[_resultCnt] = (char*)malloc(totalLen);
        int pos = 0;
        for (int i = 0; i < _n; i++) {
            int len = strlen(_names[_path[i]]);
            memcpy(_results[_resultCnt] + pos, _names[_path[i]], len);
            pos += len;
            if (i < _n - 1) _results[_resultCnt][pos++] = ' ';
        }
        _results[_resultCnt][pos] = '\0';
        _resultCnt++;
        return;
    }
    for (int cand = 0; _resultCnt < MAX_RESULT && cand < _n; cand++) {
        if (indeg[cand] != 0) continue;
        indeg[cand] = -1;
        _path[_pathLen++] = cand;
        for (int k = 0; k < _adjCnt[cand]; k++) indeg[_adj[cand][k]]--;
        _dfs(indeg);
        for (int k = 0; k < _adjCnt[cand]; k++) indeg[_adj[cand][k]]++;
        _pathLen--;
        indeg[cand] = 0;
    }
}

char** allBuildOrders(char** modules, int modulesSize,
                      char*** dependencies, int dependenciesSize, int* returnSize) {
    _n = modulesSize;
    for (int i = 0; i < _n; i++) {
        _names[i] = modules[i];
        _adjCnt[i] = 0;
        _indeg[i] = 0;
    }

    // 建图：a 依赖 b => b -> a
    for (int d = 0; d < dependenciesSize; d++) {
        char* a = dependencies[d][0];
        char* b = dependencies[d][1];
        int ai = -1, bi = -1;
        for (int i = 0; i < _n; i++) {
            if (strcmp(_names[i], a) == 0) ai = i;
            if (strcmp(_names[i], b) == 0) bi = i;
        }
        if (ai >= 0 && bi >= 0) {
            _adj[bi][_adjCnt[bi]++] = ai;
            _indeg[ai]++;
        }
    }

    // Kahn 判环
    int indeg_copy[MAX_N];
    memcpy(indeg_copy, _indeg, sizeof(_indeg));
    int q[MAX_N], h = 0, t = 0;
    for (int i = 0; i < _n; i++)
        if (indeg_copy[i] == 0) q[t++] = i;
    int visited = 0;
    while (h < t) {
        int u = q[h++];
        visited++;
        for (int k = 0; k < _adjCnt[u]; k++) {
            int v = _adj[u][k];
            if (--indeg_copy[v] == 0) q[t++] = v;
        }
    }
    if (visited < _n) {
        *returnSize = 0;
        return NULL;
    }

    _resultCnt = 0;
    _pathLen = 0;
    memcpy(indeg_copy, _indeg, sizeof(_indeg));
    _dfs(indeg_copy);

    qsort(_results, _resultCnt, sizeof(char*), _cmp_result);
    char** result = (char**)malloc(_resultCnt * sizeof(char*));
    for (int i = 0; i < _resultCnt; i++) result[i] = _results[i];
    *returnSize = _resultCnt;
    return result;
}
```
