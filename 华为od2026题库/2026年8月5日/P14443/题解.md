## 解题思路

本题要求将一组大写字母去重后排序，然后将连续字母区间（长度 ≥ 3）合并为"起始-结束"格式，其余字母单独列出。

**核心思路：扫描合并**

1. **去重排序**：由于字母只有 A-Z（最多 26 个），可以用一个长度为 26 的布尔数组 `seen[26]` 标记出现过的字母，或使用 `set` 去重后排序。这样天然保证了按字母序处理。

2. **扫描连续区间**：遍历有序字母列表，对于每个起点 `i`，向右扩展找到最长的连续字母终点 `j`（即满足 `letters[k+1] - letters[k] == 1`）。

3. **按规则输出**：
   - 若区间长度 `j - i + 1 >= 3`：输出 `X-Y` 格式
   - 否则（长度 = 1 或 2）：每个字母单独输出
   - 各项之间用逗号分隔

**关键边界**：
- 输入字母可能有重复，需先去重
- 字母不连续时（如 A、D 之间跳过了 B、C），区间自然断开
- 最多 26 个字母，算法在常数时间内完成

## 复杂度分析

- **时间复杂度**：$O(|\Sigma|)$，其中 $\Sigma = 26$ 为字母表大小。去重、排序、扫描合并均只涉及最多 26 个字母，实际为常数时间。
- **空间复杂度**：$O(|\Sigma|)$，用于存储去重后的字母列表和结果字符串。

```python
class Solution:
    def mergeBroadcastChannels(self, n: int, channels: str) -> str:
        # 去重并排序：过滤空格，用 set 去重后排序
        letters = sorted(set(channels.replace(" ", "")))
        if not letters:
            return ""

        result = []
        i = 0
        while i < len(letters):
            j = i
            # 向右扩展，找到连续字母区间的末尾
            while j + 1 < len(letters) and ord(letters[j + 1]) - ord(letters[j]) == 1:
                j += 1
            length = j - i + 1
            if length >= 3:
                # 连续 ≥3 个字母，合并为 X-Y 格式
                result.append(f"{letters[i]}-{letters[j]}")
            else:
                # 不足 3 个，各自单独列出
                for k in range(i, j + 1):
                    result.append(letters[k])
            i = j + 1

        return ",".join(result)
```

```java
import java.util.*;

class Solution {
    public String mergeBroadcastChannels(int n, String channels) {
        // 去重并排序：用 TreeSet 自动去重和排序
        Set<Character> set = new TreeSet<>();
        for (int i = 0; i < channels.length(); i++) {
            char c = channels.charAt(i);
            if (c >= 'A' && c <= 'Z') {
                set.add(c);
            }
        }

        List<Character> letters = new ArrayList<>(set);

        StringBuilder result = new StringBuilder();
        int i = 0;
        int m = letters.size();
        while (i < m) {
            int j = i;
            // 向右扩展，找到连续字母区间的末尾
            while (j + 1 < m && letters.get(j + 1) - letters.get(j) == 1) {
                j++;
            }
            int length = j - i + 1;
            if (length >= 3) {
                // 连续 ≥3 个字母，合并为 X-Y 格式
                if (result.length() > 0) result.append(",");
                result.append(letters.get(i));
                result.append("-");
                result.append(letters.get(j));
            } else {
                // 不足 3 个，各自单独列出
                for (int k = i; k <= j; k++) {
                    if (result.length() > 0) result.append(",");
                    result.append(letters.get(k));
                }
            }
            i = j + 1;
        }
        return result.toString();
    }
}
```

```cpp
#include <string>
#include <vector>
#include <set>
using namespace std;

class Solution {
public:
    string mergeBroadcastChannels(int n, string channels) {
        // 去重：用 set 自动去重并排序
        set<char> s;
        for (char c : channels) {
            if (c >= 'A' && c <= 'Z') {
                s.insert(c);
            }
        }

        // 转为有序列表
        vector<char> letters(s.begin(), s.end());

        string result;
        int i = 0;
        int m = (int)letters.size();
        while (i < m) {
            int j = i;
            // 向右扩展，找到连续字母区间的末尾
            while (j + 1 < m && letters[j + 1] - letters[j] == 1) {
                j++;
            }
            int length = j - i + 1;
            if (length >= 3) {
                // 连续 ≥3 个字母，合并为 X-Y 格式
                if (!result.empty()) result += ",";
                result += letters[i];
                result += "-";
                result += letters[j];
            } else {
                // 不足 3 个，各自单独列出
                for (int k = i; k <= j; k++) {
                    if (!result.empty()) result += ",";
                    result += letters[k];
                }
            }
            i = j + 1;
        }
        return result;
    }
};
```

```javascript
var mergeBroadcastChannels = function(n, channels) {
    // 去重并排序：过滤空格，用 Set 去重后排序
    const letters = [...new Set(channels.replace(/ /g, '').split(''))].sort();
    if (letters.length === 0) return "";

    const result = [];
    let i = 0;
    while (i < letters.length) {
        let j = i;
        // 向右扩展，找到连续字母区间的末尾
        while (j + 1 < letters.length &&
               letters[j + 1].charCodeAt(0) - letters[j].charCodeAt(0) === 1) {
            j++;
        }
        const length = j - i + 1;
        if (length >= 3) {
            // 连续 ≥3 个字母，合并为 X-Y 格式
            result.push(letters[i] + "-" + letters[j]);
        } else {
            // 不足 3 个，各自单独列出
            for (let k = i; k <= j; k++) {
                result.push(letters[k]);
            }
        }
        i = j + 1;
    }
    return result.join(",");
};
```

```c
#include <string.h>
#include <stdlib.h>

char* mergeBroadcastChannels(int n, const char* channels) {
    // 去重：用布尔数组标记出现的字母
    int seen[26] = {0};
    for (int i = 0; channels[i]; i++) {
        if (channels[i] >= 'A' && channels[i] <= 'Z') {
            seen[channels[i] - 'A'] = 1;
        }
    }

    // 结果缓冲区（最多 26 个单独字母 + 25 个逗号 + 间隔符）
    char* result = (char*)malloc(100);
    result[0] = '\0';

    int i = 0;
    while (i < 26) {
        // 跳过未出现的字母
        if (!seen[i]) { i++; continue; }
        int j = i;
        // 向右扩展，找到连续字母区间的末尾
        while (j + 1 < 26 && seen[j + 1]) {
            j++;
        }
        int length = j - i + 1;
        if (length >= 3) {
            // 连续 ≥3 个字母，合并为 X-Y 格式
            if (result[0] != '\0') strcat(result, ",");
            char seg[4];
            seg[0] = 'A' + i;
            seg[1] = '-';
            seg[2] = 'A' + j;
            seg[3] = '\0';
            strcat(result, seg);
        } else {
            // 不足 3 个，各自单独列出
            for (int k = i; k <= j; k++) {
                if (result[0] != '\0') strcat(result, ",");
                char ch[2];
                ch[0] = 'A' + k;
                ch[1] = '\0';
                strcat(result, ch);
            }
        }
        i = j + 1;
    }
    return result;
}
```
