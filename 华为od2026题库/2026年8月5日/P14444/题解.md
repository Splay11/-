## 解题思路

本题的核心是**等长子网划分**：给定一个主网段 CIDR，需要划分出若干掩码长度相同的子网，且每个子网的可用主机数能容纳对应部门的需求。

算法分为以下几步：

### 1. 解析 CIDR 主网段

将 CIDR 格式字符串（如 `"192.168.1.0/24"`）拆分为 IP 地址和掩码位数两部分。IP 地址按 `.` 分割为 4 个字节，合并为一个 32 位无符号整数，方便后续做偏移计算。

### 2. 确定子网掩码

找出所有部门中的**最大主机需求数** `maxReq`。子网的可用主机数为 $2^h - 2$（减去网络地址和广播地址），因此需要找到最小的 $h$ 使得 $2^h - 2 \geq maxReq$。

新子网掩码位数 = $32 - h$。若新掩码位数小于原掩码位数（即子网需要的地址空间比主网段还大），则无法分配，返回空数组。

### 3. 检查子网数量

子网编号使用 $newMask - maskLen$ 个二进制位，可划分的子网总数为 $2^{newMask - maskLen}$。若这个数量小于部门数 $N$，则无法满足，返回空数组。

### 4. 连续分配子网

每个子网的大小为 $2^h$。第 $i$ 个子网的起始地址为：

$$baseIP + i \times 2^h$$

将计算结果转换为点分十进制 IP 格式，拼接上新掩码位数，即为该子网的 CIDR 表示。

### 时间复杂度

- 解析 CIDR：$O(1)$
- 找最大需求：$O(N)$
- 计算 $h$：最多 32 次循环，$O(1)$
- 生成结果：$O(N)$

总体时间复杂度 $O(N)$，空间复杂度 $O(N)$（存储输出数组）。

## 复杂度分析

- **时间复杂度**：$O(N)$，其中 $N$ 为部门数量。主要开销在于遍历需求数组找最大值，以及生成 $N$ 个结果字符串。
- **空间复杂度**：$O(N)$，需要存储 $N$ 个子网 CIDR 字符串。

```python
from typing import List


class Solution:
    def allocateSubnets(self, cidr: str, n: int, requirements: List[int]) -> List[str]:
        # 解析 CIDR: "192.168.1.0/24"
        ip_str, mask_str = cidr.split('/')
        mask_len = int(mask_str)

        # 掩码位数合法性检查：至少保留 2 位主机位
        if mask_len < 8 or mask_len > 30:
            return []

        # 将 IP 转换为 32 位整数，方便做偏移计算
        base_ip = self._ip_to_int(ip_str)

        # 找到最大需求，以确定子网掩码
        max_req = max(requirements)

        # 计算所需主机位数 h: 2^h - 2 >= max_req
        h = 0
        while (1 << h) - 2 < max_req:
            h += 1
        if h < 2:
            h = 2  # 至少保留 2 位主机位

        new_mask = 32 - h
        # 新掩码不能小于原掩码（不能缩小网络范围）
        if new_mask < mask_len:
            return []

        # 可划分子网数量 = 2^(new_mask - mask_len)
        subnet_bits = new_mask - mask_len
        subnet_count = 1 << subnet_bits
        if subnet_count < n:
            return []

        # 每个子网大小 = 2^h，连续分配
        subnet_size = 1 << h
        result = []
        for i in range(n):
            subnet_ip = base_ip + i * subnet_size
            result.append(self._int_to_ip(subnet_ip) + "/" + str(new_mask))
        return result

    def _ip_to_int(self, ip: str) -> int:
        """IP 地址字符串 "a.b.c.d" -> 32 位整数"""
        a, b, c, d = map(int, ip.split('.'))
        return (a << 24) | (b << 16) | (c << 8) | d

    def _int_to_ip(self, ip: int) -> str:
        """32 位整数 -> IP 地址字符串 "a.b.c.d" """
        return ".".join(str((ip >> s) & 0xFF) for s in (24, 16, 8, 0))
```

```java
import java.util.ArrayList;
import java.util.List;

class Solution {
    public List<String> allocateSubnets(String cidr, int n, List<Integer> requirements) {
        // 解析 CIDR: "192.168.1.0/24"
        int slash = cidr.indexOf('/');
        String ipStr = cidr.substring(0, slash);
        int maskLen = Integer.parseInt(cidr.substring(slash + 1));

        // 掩码位数合法性检查
        if (maskLen < 8 || maskLen > 30) return new ArrayList<>();

        // 将 IP 转换为 32 位无符号整数
        long baseIP = ipToLong(ipStr);

        // 找到最大需求
        int maxReq = 0;
        for (int r : requirements) maxReq = Math.max(maxReq, r);

        // 计算所需主机位数 h: 2^h - 2 >= maxReq
        int h = 0;
        while ((1L << h) - 2 < maxReq) h++;
        if (h < 2) h = 2;  // 至少保留 2 位主机位

        int newMask = 32 - h;
        // 新掩码不能小于原掩码（不能缩小网络范围）
        if (newMask < maskLen) return new ArrayList<>();

        // 可划分子网数量 = 2^(newMask - maskLen)
        int subnetBits = newMask - maskLen;
        long subnetCount = 1L << subnetBits;
        if (subnetCount < n) return new ArrayList<>();

        // 每个子网大小 = 2^h，连续分配
        long subnetSize = 1L << h;
        List<String> result = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            long subnetIP = baseIP + i * subnetSize;
            result.add(longToIp(subnetIP) + "/" + newMask);
        }
        return result;
    }

    private long ipToLong(String ip) {
        String[] parts = ip.split("\\.");
        // 将四个字节合并为一个长整型，避免符号位问题
        long a = Long.parseLong(parts[0]);
        long b = Long.parseLong(parts[1]);
        long c = Long.parseLong(parts[2]);
        long d = Long.parseLong(parts[3]);
        return (a << 24) | (b << 16) | (c << 8) | d;
    }

    private String longToIp(long ip) {
        // 分别取每 8 位还原为点分十进制
        return (ip >> 24 & 0xFF) + "." + (ip >> 16 & 0xFF) + "." +
               (ip >> 8 & 0xFF) + "." + (ip & 0xFF);
    }
}
```

```cpp
#include <string>
#include <vector>
#include <algorithm>
#include <sstream>
using namespace std;

class Solution {
public:
    vector<string> allocateSubnets(string cidr, int n, vector<int>& requirements) {
        // 解析 CIDR: "192.168.1.0/24"
        size_t slash = cidr.find('/');
        string ipStr = cidr.substr(0, slash);
        int maskLen = stoi(cidr.substr(slash + 1));

        // 掩码位数合法性检查
        if (maskLen < 8 || maskLen > 30) return {};

        // 将 IP 转换为 32 位无符号整数
        unsigned int baseIP = ipToUint(ipStr);

        // 找到最大需求
        int maxReq = 0;
        for (int r : requirements) maxReq = max(maxReq, r);

        // 计算所需主机位数 h: 2^h - 2 >= maxReq
        int h = 0;
        while ((1LL << h) - 2 < maxReq) h++;
        if (h < 2) h = 2;  // 至少保留 2 位主机位

        int newMask = 32 - h;
        // 新掩码不能小于原掩码（不能缩小网络范围）
        if (newMask < maskLen) return {};

        // 可划分子网数量 = 2^(newMask - maskLen)
        int subnetBits = newMask - maskLen;
        long long subnetCount = 1LL << subnetBits;
        if (subnetCount < n) return {};

        // 每个子网大小 = 2^h，连续分配
        unsigned int subnetSize = 1u << h;
        vector<string> result;
        for (int i = 0; i < n; i++) {
            unsigned int subnetIP = baseIP + i * subnetSize;
            result.push_back(uintToIp(subnetIP) + "/" + to_string(newMask));
        }
        return result;
    }

private:
    // IP 地址字符串 "a.b.c.d" -> 32 位无符号整数
    unsigned int ipToUint(const string& ip) {
        unsigned int a, b, c, d;
        char dot;
        istringstream iss(ip);
        iss >> a >> dot >> b >> dot >> c >> dot >> d;
        return (a << 24) | (b << 16) | (c << 8) | d;
    }

    // 32 位无符号整数 -> IP 地址字符串 "a.b.c.d"
    string uintToIp(unsigned int ip) {
        return to_string((ip >> 24) & 0xFF) + "." +
               to_string((ip >> 16) & 0xFF) + "." +
               to_string((ip >> 8) & 0xFF) + "." +
               to_string(ip & 0xFF);
    }
};
```

```javascript
/**
 * @param {string} cidr - 主网段 CIDR
 * @param {number} n - 部门数量
 * @param {number[]} requirements - 各部门最小主机需求数
 * @return {string[]} - 分配的子网 CIDR 字符串数组
 */
var allocateSubnets = function(cidr, n, requirements) {
    // 解析 CIDR: "192.168.1.0/24"
    const slash = cidr.indexOf('/');
    const ipStr = cidr.substring(0, slash);
    const maskLen = parseInt(cidr.substring(slash + 1));

    // 掩码位数合法性检查
    if (maskLen < 8 || maskLen > 30) return [];

    // 将 IP 转换为 32 位无符号整数（>>> 0 保持无符号语义）
    const baseIP = ipToUint(ipStr);

    // 找到最大需求
    let maxReq = 0;
    for (const r of requirements) maxReq = Math.max(maxReq, r);

    // 计算所需主机位数 h: 2^h - 2 >= maxReq
    // 用 Math.pow 避免 JS 位运算 32 位溢出
    let h = 0;
    while (Math.pow(2, h) - 2 < maxReq) h++;
    if (h < 2) h = 2;

    const newMask = 32 - h;
    if (newMask < maskLen) return [];  // 不能缩小网络范围

    // 可划分子网数量
    const subnetCount = Math.pow(2, newMask - maskLen);
    if (subnetCount < n) return [];

    // 每个子网大小 = 2^h，连续分配
    const subnetSize = Math.pow(2, h);
    const result = [];
    for (let i = 0; i < n; i++) {
        const subnetIP = baseIP + i * subnetSize;
        result.push(uintToIp(subnetIP) + "/" + newMask);
    }
    return result;
};

// 辅助函数：IP 地址 <-> 整数互转
function ipToUint(ip) {
    const [a, b, c, d] = ip.split('.').map(Number);
    return ((a << 24) | (b << 16) | (c << 8) | d) >>> 0;
}

function uintToIp(ip) {
    return (ip >>> 24 & 0xFF) + "." + (ip >>> 16 & 0xFF) + "." +
           (ip >>> 8 & 0xFF) + "." + (ip & 0xFF);
}
```

```c
#include <string.h>
#include <stdlib.h>
#include <stdio.h>

// IP 地址 "a.b.c.d" -> 32 位无符号整数
static unsigned int ipToUint(const char* ip) {
    unsigned int a, b, c, d;
    sscanf(ip, "%u.%u.%u.%u", &a, &b, &c, &d);
    return (a << 24) | (b << 16) | (c << 8) | d;
}

// 32 位整数 -> IP 地址 "a.b.c.d"
static void uintToIp(unsigned int ip, char* out) {
    sprintf(out, "%u.%u.%u.%u",
            (ip >> 24) & 0xFF, (ip >> 16) & 0xFF,
            (ip >> 8) & 0xFF, ip & 0xFF);
}

char** allocateSubnets(const char* cidr, int n, int* requirements,
                       int requirementsSize, int* returnSize) {
    *returnSize = 0;

    // 解析 CIDR: "192.168.1.0/24"
    char ipStr[20];
    int maskLen;
    sscanf(cidr, "%[^/]/%d", ipStr, &maskLen);

    // 掩码合法性检查
    if (maskLen < 8 || maskLen > 30) return NULL;

    // IP 转整数
    unsigned int baseIP = ipToUint(ipStr);

    // 找最大需求
    int maxReq = 0;
    for (int i = 0; i < requirementsSize; i++)
        if (requirements[i] > maxReq) maxReq = requirements[i];

    // 计算主机位数 h: 2^h - 2 >= maxReq
    int h = 0;
    while ((1LL << h) - 2 < maxReq) h++;
    if (h < 2) h = 2;

    int newMask = 32 - h;
    if (newMask < maskLen) return NULL;  // 不能缩小网络

    // 检查子网数量
    int subnetBits = newMask - maskLen;
    long long subnetCount = 1LL << subnetBits;
    if (subnetCount < n) return NULL;

    // 连续分配子网
    unsigned int subnetSize = 1u << h;
    char** result = (char**)malloc(n * sizeof(char*));
    for (int i = 0; i < n; i++) {
        unsigned int subnetIP = baseIP + i * subnetSize;
        char ipBuf[20];
        uintToIp(subnetIP, ipBuf);
        result[i] = (char*)malloc(24);
        sprintf(result[i], "%s/%d", ipBuf, newMask);
    }
    *returnSize = n;
    return result;
}
```
