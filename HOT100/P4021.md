先想清楚本题所有可能情况:共$n!$ 种，高中学过的排列：考虑总共$n$个位置，第一个位置填$n$个，第二个位置就只能填$n - 1$ 个 ， 以此类推，结合分步相乘得到:

$$n \times (n - 1) \times (n - 2) \times ... \times 1 = n!$$

**我们回溯的过程其实和这个计算过程如出一辙** ： 考虑三个位置可以填，从左往右逐一填写，每次考虑这个位置可以填哪些数。

例如$1\ 2\ 3$ ， 我们用一个演示视频来展示搜索回溯的过程👇

代码层面：

1.设计一个递归参数`index`  记录当前填写到第几个位置了，

2.设计一个全局`curr` 代表当前生成的排列。

3.我们从上面的视频可以知道，在填写的过程中每次都是从`1`开始尝试填写数字，但是要保证排列的不重复性质，也就是需要知道要填的数字是否在`curr`中出现过，所以我们又需要设计一个全局变量`used`来判断这个数当下是否被填写

```python
class Solution:
    def permute(self, nums: List[int]) -> List[List[int]]:
        n = len(nums)
        # ans 用来存储所有排列结果
        ans = []
        # curr 表示当前正在生成的排列
        curr = []
        # used 标记每个数字当前是否已经被放入 curr
        used = [False] * n

        def dfs(index):
            # 当 index == n，说明 n 个位置都已经填满，得到一个完整排列
            if index == n:
                ans.append(curr[:])
                return

            # 当前这个位置可以尝试填写任意一个还没有使用过的数字
            for i in range(n):
                if not used[i]:
                    # 选择 nums[i]
                    used[i] = True
                    curr.append(nums[i])

                    # 继续填写下一个位置
                    dfs(index + 1)

                    # 回溯，撤销选择
                    curr.pop()
                    used[i] = False

        dfs(0)
        return ans
```

```cpp
class Solution {
public:
    vector<vector<int>> ans;
    vector<int> curr;
    vector<bool> used;

    void dfs(vector<int>& nums, int index) {
        int n = nums.size();

        // 当 index == n，说明 n 个位置都已经填满，得到一个完整排列
        if (index == n) {
            ans.push_back(curr);
            return;
        }

        // 当前这个位置可以尝试填写任意一个还没有使用过的数字
        for (int i = 0; i < n; i++) {
            if (!used[i]) {
                // 选择 nums[i]
                used[i] = true;
                curr.push_back(nums[i]);

                // 继续填写下一个位置
                dfs(nums, index + 1);

                // 回溯，撤销选择
                curr.pop_back();
                used[i] = false;
            }
        }
    }

    vector<vector<int>> permute(vector<int>& nums) {
        int n = nums.size();

        // 初始化结果、当前排列和使用标记
        ans.clear();
        curr.clear();
        used = vector<bool>(n, false);

        // 从第 0 个位置开始填写
        dfs(nums, 0);

        return ans;
    }
};
```

```java
class Solution {
    // ans 用来存储所有排列结果
    private List<List<Integer>> ans = new ArrayList<>();
    // curr 表示当前正在生成的排列
    private List<Integer> curr = new ArrayList<>();
    // used 标记每个数字当前是否已经被放入 curr
    private boolean[] used;

    public List<List<Integer>> permute(int[] nums) {
        int n = nums.length;

        ans.clear();
        curr.clear();
        used = new boolean[n];

        // 从第 0 个位置开始填写
        dfs(nums, 0);

        return ans;
    }

    private void dfs(int[] nums, int index) {
        int n = nums.length;

        // 当 index == n，说明 n 个位置都已经填满，得到一个完整排列
        if (index == n) {
            ans.add(new ArrayList<>(curr));
            return;
        }

        // 当前这个位置可以尝试填写任意一个还没有使用过的数字
        for (int i = 0; i < n; i++) {
            if (!used[i]) {
                // 选择 nums[i]
                used[i] = true;
                curr.add(nums[i]);

                // 继续填写下一个位置
                dfs(nums, index + 1);

                // 回溯，撤销选择
                curr.remove(curr.size() - 1);
                used[i] = false;
            }
        }
    }
}
```

```go
func permute(nums []int) [][]int {
	n := len(nums)

	// ans 用来存储所有排列结果
	ans := [][]int{}
	// curr 表示当前正在生成的排列
	curr := []int{}
	// used 标记每个数字当前是否已经被放入 curr
	used := make([]bool, n)

	var dfs func(int)
	dfs = func(index int) {
		// 当 index == n，说明 n 个位置都已经填满，得到一个完整排列
		if index == n {
			tmp := make([]int, len(curr))
			copy(tmp, curr)
			ans = append(ans, tmp)
			return
		}

		// 当前这个位置可以尝试填写任意一个还没有使用过的数字
		for i := 0; i < n; i++ {
			if !used[i] {
				// 选择 nums[i]
				used[i] = true
				curr = append(curr, nums[i])

				// 继续填写下一个位置
				dfs(index + 1)

				// 回溯，撤销选择
				curr = curr[:len(curr)-1]
				used[i] = false
			}
		}
	}

	// 从第 0 个位置开始填写
	dfs(0)

	return ans
}
```

```javascript
function permute(nums) {
    const n = nums.length;

    // ans 用来存储所有排列结果
    const ans = [];
    // curr 表示当前正在生成的排列
    const curr = [];
    // used 标记每个数字当前是否已经被放入 curr
    const used = new Array(n).fill(false);

    function dfs(index) {
        // 当 index === n，说明 n 个位置都已经填满，得到一个完整排列
        if (index === n) {
            ans.push([...curr]);
            return;

        for (let i = 0; i < n; i++) {

                curr.push(nums[i]);

                dfs(index + 1);

                curr.pop();
                used[i] = false;
            }
        }
    }

    // 从第 0 个位置开始填写
    dfs(0);

    return ans;
}
```

```c
void dfs(int* nums, int numsSize, int index, int* used, int* curr, int currSize,
         int** ans, int* returnSize, int** returnColumnSizes) {
    // 当 index == numsSize，说明所有位置都已经填满，得到一个完整排列
    if (index == numsSize) {
        ans[*returnSize] = (int*)malloc(sizeof(int) * numsSize);
        for (int i = 0; i < numsSize; i++) {
            ans[*returnSize][i] = curr[i];
        }
        (*returnColumnSizes)[*returnSize] = numsSize;
        (*returnSize)++;
        return;
    }

    // 当前这个位置可以尝试填写任意一个还没有使用过的数字
    for (int i = 0; i < numsSize; i++) {
        if (!used[i]) {
            // 选择 nums[i]
            used[i] = 1;
            curr[currSize] = nums[i];

            // 继续填写下一个位置
            dfs(nums, numsSize, index + 1, used, curr, currSize + 1,
                ans, returnSize, returnColumnSizes);

            // 回溯，撤销选择
            used[i] = 0;
        }
    }
}

int** permute(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    // 最多有 numsSize! 个排列，numsSize 最大通常较小，这里先计算结果数量
    int total = 1;
    for (int i = 1; i <= numsSize; i++) {
        total *= i;
    }

    // ans 用来存储所有排列结果
    int** ans = (int**)malloc(sizeof(int*) * total);
    *returnColumnSizes = (int*)malloc(sizeof(int) * total);
    *returnSize = 0;

    // curr 表示当前正在生成的排列
    int* curr = (int*)malloc(sizeof(int) * numsSize);
    // used 标记每个数字当前是否已经被放入 curr
    int* used = (int*)malloc(sizeof(int) * numsSize);

    for (int i = 0; i < numsSize; i++) {
        used[i] = 0;
    }

    // 从第 0 个位置开始填写
    dfs(nums, numsSize, 0, used, curr, 0, ans, returnSize, returnColumnSizes);

    return ans;
}
```

### 面试问答

**1. 本题的核心思路是什么？**

这道题的核心思路是**回溯**。按位置从左到右依次填数，每次在所有还没使用过的数字里选一个放到当前位置，然后递归去填下一个位置。递归过程中，用 `used` 记录哪些数已经用过，用 `curr` 记录当前排列；当所有位置都填完时，就得到了一种合法排列。整个过程结束就把所有排列给搜出来了。

时间复杂度是 $O(n * n!)$ , 因为总共有$n!$种可能，每次存储`curr`到`ans`的操作为$O(n)$的。
