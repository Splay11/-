**思路**

本题和上一题极其类似。我们只需要先用哈希统计元素出现频次。然后对(频次,值)二元组用固定大小为$k$的小根堆维护，最后输出小根堆的前$k$小的值即可

```python
import heapq
from typing import List

class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        # 统计每个元素的频率
        freq = {}
        for num in nums:
            freq[num] = freq.get(num, 0) + 1

        # 构建一个大小为k的小根堆
        heap = []
        for num, freq in freq.items():
            heapq.heappush(heap, [freq, num])
            if len(heap) > k:
                heapq.heappop(heap)

        # 堆中剩下的就是前k个高频元素，堆顶就是第k个高频元素
        return [item[1] for item in heap]
```

```cpp
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        // 统计每个元素的频率
        unordered_map<int, int> freq;
        for (int num : nums) {
            freq[num]++;
        }

        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> heap;

        for (auto& item : freq) {
            int num = item.first;
            int count = item.second;

            heap.push({count, num});
            if ((int)heap.size() > k) {
                heap.pop();
            }
        }

        // 堆中剩下的就是前k个高频元素，堆顶就是第k个高频元素
        vector<int> ans;
        while (!heap.empty()) {
            ans.push_back(heap.top().second);
            heap.pop();
        }

        return ans;
    }
```

```java
class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        // 统计每个元素的频率
        Map<Integer, Integer> freq = new HashMap<>();

        for (int num : nums) {
            freq.put(num, freq.getOrDefault(num, 0) + 1);
        }

        // 构建一个大小为k的小根堆
        PriorityQueue<int[]> heap = new PriorityQueue<>((a, b) -> {
            if (a[0] != b[0]) {
                return a[0] - b[0];
            }
            return a[1] - b[1];
        });

        for (Map.Entry<Integer, Integer> item : freq.entrySet()) {
            int num = item.getKey();
            int count = item.getValue();

            heap.offer(new int[]{count, num});
            if (heap.size() > k) {
                heap.poll();
            }
        }

        // 堆中剩下的就是前k个高频元素，堆顶就是第k个高频元素
        int[] ans = new int[k];
        int index = 0;

        while (!heap.isEmpty()) {
            ans[index++] = heap.poll()[1];
        }

        return ans;
    }
}
```

```go
type Pair struct {
	count int
	num   int
}

type MinHeap []Pair

func (h MinHeap) Len() int {
	return len(h)
}

func (h MinHeap) Less(i, j int) bool {
	if h[i].count != h[j].count {
		return h[i].count < h[j].count
	}
	return h[i].num < h[j].num
}

func (h MinHeap) Swap(i, j int) {
	h[i], h[j] = h[j], h[i]
}

func (h *MinHeap) Push(x interface{}) {
	*h = append(*h, x.(Pair))
}

func (h *MinHeap) Pop() interface{} {
	old := *h
	n := len(old)
	item := old[n-1]
	*h = old[:n-1]
	return item
}

func topKFrequent(nums []int, k int) []int {
	// 统计每个元素的频率
	freq := make(map[int]int)
	for _, num := range nums {
		freq[num]++
	}

	// 构建一个大小为k的小根堆
	heap := &MinHeap{}
	for num, count := range freq {
		heap.Push(heap, Pair{count, num})
		if heap.Len() > k {
			heap.Pop(heap)
		}
	}

	// 堆中剩下的就是前k个高频元素，堆顶就是第k个高频元素
	ans := []int{}
	for heap.Len() > 0 {
		item := heap.Pop(heap).(Pair)
		ans = append(ans, item.num)
	}

	return ans
}
```

```javascript
var topKFrequent = function(nums, k) {
    // 统计每个元素的频率
    const freq = new Map();

    for (const num of nums) {
        freq.set(num, (freq.get(num) || 0) + 1);
    }

    // 构建一个大小为k的小根堆
    const heap = [];

    const less = function(a, b) {
        if (a[0] !== b[0]) {
            return a[0] < b[0];
        }
        return a[1] < b[1];
    };

    const heapifyUp = function(index) {
        while (index > 0) {
            const parent = Math.floor((index - 1) / 2);

            if (!less(heap[index], heap[parent])) {
                break;
            }

            const temp = heap[index];
            heap[index] = heap[parent];
            heap[parent] = temp;

            index = parent;
        }
    };

    const heapifyDown = function(index) {
        while (true) {
            const left = index * 2 + 1;
            const right = index * 2 + 2;
            let smallest = index;

            if (left < heap.length && less(heap[left], heap[smallest])) {
                smallest = left;
            }
            if (right < heap.length && less(heap[right], heap[smallest])) {
                smallest = right;
            }

            if (smallest === index) {
                break;
            }

            const temp = heap[index];
            heap[index] = heap[smallest];
            heap[smallest] = temp;

            index = smallest;
        }
    };

    for (const [num, count] of freq) {
        heap.push([count, num]);
        heapifyUp(heap.length - 1);

        if (heap.length > k) {
            heap[0] = heap[heap.length - 1];
            heap.pop();
            heapifyDown(0);
        }
    }

    // 堆中剩下的就是前k个高频元素，堆顶就是第k个高频元素
    const ans = [];
    for (const item of heap) {
        ans.push(item[1]);
    }

    return ans;
};
```

```c
typedef struct {
    int num;
    int count;
} Node;

int less(Node a, Node b) {
    if (a.count != b.count) {
        return a.count < b.count;
    }
    return a.num < b.num;
}

void heapifyUp(Node* heap, int index) {
    while (index > 0) {
        int parent = (index - 1) / 2;

        if (!less(heap[index], heap[parent])) {
            break;
        }

        Node temp = heap[index];
        heap[index] = heap[parent];
        heap[parent] = temp;

        index = parent;
    }
}

void heapifyDown(Node* heap, int size, int index) {
    while (1) {
        int left = index * 2 + 1;
        int right = index * 2 + 2;
        int smallest = index;

        if (left < size && less(heap[left], heap[smallest])) {
            smallest = left;
        }
        if (right < size && less(heap[right], heap[smallest])) {
            smallest = right;
        }

        if (smallest == index) {
            break;
        }

        Node temp = heap[index];
        heap[index] = heap[smallest];
        heap[smallest] = temp;

        index = smallest;
    }
}

void heapPush(Node* heap, int* size, Node node) {
    heap[*size] = node;
    (*size)++;
    heapifyUp(heap, *size - 1);
}

void heapPop(Node* heap, int* size) {
    heap[0] = heap[*size - 1];
    (*size)--;

    if (*size > 0) {
        heapifyDown(heap, *size, 0);
    }
}

unsigned int getHash(int x) {
    unsigned int h = (unsigned int)x;
    h ^= h >> 16;
    h *= 0x7feb352d;
    h ^= h >> 15;
    h *= 0x846ca68b;
    h ^= h >> 16;
    return h;
}

/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* topKFrequent(int* nums, int numsSize, int k, int* returnSize) {
    // 统计每个元素的频率
    int capacity = numsSize * 4 + 1;
    int* keys = (int*)malloc(sizeof(int) * capacity);
    int* counts = (int*)malloc(sizeof(int) * capacity);
    int* used = (int*)malloc(sizeof(int) * capacity);

    for (int i = 0; i < capacity; i++) {
        used[i] = 0;
        counts[i] = 0;
    }

    for (int i = 0; i < numsSize; i++) {
        int num = nums[i];
        int index = getHash(num) % capacity;

        while (used[index] && keys[index] != num) {
            index++;
            if (index == capacity) {
                index = 0;
            }
        }

        if (!used[index]) {
            used[index] = 1;
            keys[index] = num;
            counts[index] = 1;
        } else {
            counts[index]++;
        }
    }

    // 构建一个大小为k的小根堆
    Node* heap = (Node*)malloc(sizeof(Node) * (k + 1));
    int heapSize = 0;

    for (int i = 0; i < capacity; i++) {
        if (!used[i]) {
            continue;
        }

        Node node;
        node.num = keys[i];
        node.count = counts[i];

        heapPush(heap, &heapSize, node);
        if (heapSize > k) {
            heapPop(heap, &heapSize);
        }
    }

    // 堆中剩下的就是前k个高频元素，堆顶就是第k个高频元素
    int* ans = (int*)malloc(sizeof(int) * k);
    *returnSize = k;

    for (int i = 0; i < k; i++) {
        ans[i] = heap[i].num;
    }

    free(keys);
    free(counts);
    free(used);
    free(heap);

    return ans;
}
```
