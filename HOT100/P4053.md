### 思路

假设`s=abbdac` , `t=aab`  , 最短覆盖子串是`s[1:5] = abbda` 。 

为了更好的实现本题，我们可以将 一个子串`x`是否覆盖`t`  这一问题 转化为 `x` 的所有字符的出现次数 **都大于等于**  `t`中对应的字符出现的次数👇

![](/file/2/6CDZv5pxEgGXOEmmCzU3p.png =80%x)





上一题维护的性质是：窗口内不能有重复字符

而这题维护的性质变成了：**窗口必须包含 `t` 的所有字符，且每个字符的出现次数都不少于 `t`** 

所以这题的基本套路仍然是：

- 右指针不断向右扩展，尝试让当前窗口变成“合法窗口”
- 一旦当前窗口已经合法，就不断右移左指针，尽量缩小窗口
- 在缩小的过程中更新最短答案

一句话:**滑动窗口一旦合法，就立刻缩；缩的过程中更新最优答案。**

```python
class Solution:
    def minWindow(self, s: str, t: str) -> str:
        # 统计 t 中每个字符需要出现的次数
        need = {}
        for ch in t:
            need[ch] = need.get(ch, 0) + 1
        
        # 统计当前窗口中每个字符出现的次数
        window = {}
        
        # 左指针
        i = 0
        
        # 当前满足要求的字符种类数
        valid = 0
        
        # 最优答案的起点
        start = 0
        
        # 记录最短覆盖子串的长度，初始设为不可能达到的值
        min_len = len(s) + 1
        
        # 右指针不断向右扩展
        for j in range(len(s)):
            ch = s[j]
            
            # 右指针加入一个字符，更新窗口信息
            if ch in need:
                window[ch] = window.get(ch, 0) + 1
                
                # 如果当前字符的出现次数恰好达到要求，说明又有一种字符满足条件
                if window[ch] == need[ch]:
                    valid += 1
            
            # 当窗口已经包含 t 的所有字符时，尝试缩小左边界
            while valid == len(need):
                # 先用当前合法窗口更新答案
                if j - i + 1 < min_len:
                    min_len = j - i + 1
                    start = i
                
                # 准备移出左端点字符
                left_ch = s[i]
                if left_ch in need:
                    # 如果移出前这个字符刚好满足要求，那么移出后窗口将不再合法
                    if window[left_ch] == need[left_ch]:
                        valid -= 1
                    
                    # 更新窗口中该字符的出现次数
                    window[left_ch] -= 1
                
                # 左指针右移，继续尝试缩小窗口
                i += 1
        
        # 如果 min_len 没有被更新，说明不存在覆盖子串
        if min_len == len(s) + 1:
            return ""
        
        # 返回最短覆盖子串
        return s[start:start + min_len]
```

```java
class Solution {
    public String minWindow(String s, String t) {
        // 统计 t 中每个字符需要出现的次数
        Map<Character, Integer> need = new HashMap<>();
        for (char ch : t.toCharArray()) {
            need.put(ch, need.getOrDefault(ch, 0) + 1);
        }
        
        // 统计当前窗口中每个字符出现的次数
        Map<Character, Integer> window = new HashMap<>();
        
        // 左指针
        int i = 0;
        
        // 当前满足要求的字符种类数
        int valid = 0;
        
        // 最优答案的起点
        int start = 0;
        
        // 记录最短覆盖子串的长度，初始设为不可能达到的值
        int minLen = s.length() + 1;
        
        // 右指针不断向右扩展
        for (int j = 0; j < s.length(); j++) {
            char ch = s.charAt(j);
            
            // 右指针加入一个字符，更新窗口信息
            if (need.containsKey(ch)) {
                window.put(ch, window.getOrDefault(ch, 0) + 1);
                
                // 如果当前字符的出现次数恰好达到要求，说明又有一种字符满足条件
                if (window.get(ch).intValue() == need.get(ch).intValue()) {
                    valid++;
                }
            }
            
            // 当窗口已经包含 t 的所有字符时，尝试缩小左边界
            while (valid == need.size()) {
                // 先用当前合法窗口更新答案
                if (j - i + 1 < minLen) {
                    minLen = j - i + 1;
                    start = i;
                }
                
                // 准备移出左端点字符
                char leftCh = s.charAt(i);
                if (need.containsKey(leftCh)) {
                    // 如果移出前这个字符刚好满足要求，那么移出后窗口将不再合法
                    if (window.get(leftCh).intValue() == need.get(leftCh).intValue()) {
                        valid--;
                    }
                    
                    // 更新窗口中该字符的出现次数
                    window.put(leftCh, window.get(leftCh) - 1);
                }
                
                // 左指针右移，继续尝试缩小窗口
                i++;
            }
        }
        
        // 如果 minLen 没有被更新，说明不存在覆盖子串
        if (minLen == s.length() + 1) {
            return "";
        }
        
        // 返回最短覆盖子串
        return s.substring(start, start + minLen);
    }
}
```

```javascript
function minWindow(s, t) {
    // 统计 t 中每个字符需要出现的次数
    const need = new Map();
    for (const ch of t) {
        need.set(ch, (need.get(ch) || 0) + 1);
    }

    // 统计当前窗口中每个字符出现的次数
    const window = new Map();

    // 左指针
    let i = 0;

    // 当前满足要求的字符种类数
    let valid = 0;

    // 最优答案的起点
    let start = 0;

    // 记录最短覆盖子串的长度，初始设为不可能达到的值
    let minLen = s.length + 1;

    // 右指针不断向右扩展
    for (let j = 0; j < s.length; j++) {
        const ch = s[j];

        // 右指针加入一个字符，更新窗口信息
        if (need.has(ch)) {
            window.set(ch, (window.get(ch) || 0) + 1);

            // 如果当前字符的出现次数恰好达到要求，说明又有一种字符满足条件
            if (window.get(ch) === need.get(ch)) {
                valid++;
            }
        }

        // 当窗口已经包含 t 的所有字符时，尝试缩小左边界
        while (valid === need.size) {
            // 先用当前合法窗口更新答案
            if (j - i + 1 < minLen) {
                minLen = j - i + 1;
                start = i;
            }

            // 准备移出左端点字符
            const leftCh = s[i];

            if (need.has(leftCh)) {
                // 如果移出前这个字符刚好满足要求，那么移出后窗口将不再合法
                if (window.get(leftCh) === need.get(leftCh)) {
                    valid--;
                }

                // 更新窗口中该字符的出现次数
                window.set(leftCh, window.get(leftCh) - 1);
            }

            // 左指针右移，继续尝试缩小窗口
            i++;
        }
    }

    // 如果 minLen 没有被更新，说明不存在覆盖子串
    if (minLen === s.length + 1) {
        return "";
    }

    // 返回最短覆盖子串
    return s.substring(start, start + minLen);
}
```

```go
func minWindow(s string, t string) string {
	// 统计 t 中每个字符需要出现的次数
	need := make(map[byte]int)
	for i := 0; i < len(t); i++ {
		need[t[i]]++
	}

	// 统计当前窗口中每个字符出现的次数
	window := make(map[byte]int)

	// 左指针
	i := 0

	// 当前满足要求的字符种类数
	valid := 0

	// 最优答案的起点
	start := 0

	// 记录最短覆盖子串的长度，初始设为不可能达到的值
	minLen := len(s) + 1

	// 右指针不断向右扩展
	for j := 0; j < len(s); j++ {
		ch := s[j]

		// 右指针加入一个字符，更新窗口信息
		if _, ok := need[ch]; ok {
			window[ch]++

			// 如果当前字符的出现次数恰好达到要求，说明又有一种字符满足条件
			if window[ch] == need[ch] {
				valid++
			}
		}

		// 当窗口已经包含 t 的所有字符时，尝试缩小左边界
		for valid == len(need) {
			// 先用当前合法窗口更新答案
			if j-i+1 < minLen {
				minLen = j - i + 1
				start = i
			}

			// 准备移出左端点字符
			leftCh := s[i]

			if _, ok := need[leftCh]; ok {
				// 如果移出前这个字符刚好满足要求，那么移出后窗口将不再合法
				if window[leftCh] == need[leftCh] {
					valid--
				}

				// 更新窗口中该字符的出现次数
				window[leftCh]--
			}

			// 左指针右移，继续尝试缩小窗口
			i++
		}
	}

	// 如果 minLen 没有被更新，说明不存在覆盖子串
	if minLen == len(s)+1 {
		return ""
	}

	// 返回最短覆盖子串
	return s[start : start+minLen]
}
```

```c
char* minWindow(char *s, char *t) {
    // 统计 t 中每个字符需要出现的次数
    int need[256] = {0};

    // 统计当前窗口中每个字符出现的次数
    int window[256] = {0};

    // 记录 t 中一共有多少种不同字符
    int needSize = 0;

    for (int i = 0; t[i] != '\0'; i++) {
        unsigned char ch = t[i];

        // 第一次出现这个字符，字符种类数加一
        if (need[ch] == 0) {
            needSize++;
        }

        need[ch]++;
    }

    // 左指针
    int i = 0;

    // 当前满足要求的字符种类数
    int valid = 0;

    // 最优答案的起点
    int start = 0;

    // 获取字符串 s 的长度
    int n = strlen(s);

    // 记录最短覆盖子串的长度，初始设为不可能达到的值
    int minLen = n + 1;

    // 右指针不断向右扩展
    for (int j = 0; j < n; j++) {
        unsigned char ch = s[j];

        // 右指针加入一个字符，更新窗口信息
        if (need[ch] > 0) {
            window[ch]++;

            // 如果当前字符的出现次数恰好达到要求，说明又有一种字符满足条件
            if (window[ch] == need[ch]) {
                valid++;
            }
        }

        // 当窗口已经包含 t 的所有字符时，尝试缩小左边界
        while (valid == needSize) {
            // 先用当前合法窗口更新答案
            if (j - i + 1 < minLen) {
                minLen = j - i + 1;
                start = i;
            }

            // 准备移出左端点字符
            unsigned char leftCh = s[i];

            if (need[leftCh] > 0) {
                // 如果移出前这个字符刚好满足要求，那么移出后窗口将不再合法
                if (window[leftCh] == need[leftCh]) {
                    valid--;
                }

                // 更新窗口中该字符的出现次数
                window[leftCh]--;
            }

            // 左指针右移，继续尝试缩小窗口
            i++;
        }
    }

    // 用静态数组保存答案，避免返回局部变量地址
    static char ans[1000005];

    // 如果 minLen 没有被更新，说明不存在覆盖子串
    if (minLen == n + 1) {
        ans[0] = '\0';
        return ans;
    }

    // 截取最短覆盖子串
    strncpy(ans, s + start, minLen);
    ans[minLen] = '\0';

    return ans;
}
```

#### 为什么可以这样做？

这题同样依赖单调性。设 $f(i)$ 表示：固定左端点 $i$ 时，第一个能够覆盖 `t` 的最小右端点位置。不难证明：$f(i)$是**单调非降**的。而右端点本质上就是在找这个$f(i)$，所以右端点不用回退，单方向移动即可。这里塔子哥不再展开叙述，写了第一题的同学自行思考。

#### 这题的更新答案为什么要写在while里？

因为 `while valid == len(need)` 说明：**当前窗口已经是一个合法覆盖子串**

但我们要求的是**最短**覆盖子串，所以只要窗口合法，就应该立刻尝试更新答案，然后继续收缩左边界。

### 面试问答

#### 1. 这个算法的整体思路是什么？

使用的是同向双指针。

右指针负责不断向右扩展区间，使窗口包含 `t` 的所有字符。
当窗口已经合法后，左指针继续右移，在保证窗口仍然合法的前提下尽量缩小区间，并在这个过程中更新最短答案。

------

#### 3. 这个算法的复杂度是多少？

时间复杂度是：$O(n)$

因为左右指针都只会单调地向右移动，不会回退。
