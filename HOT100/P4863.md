```python
class MinHeap:
    def __init__(self):
        # 初始化堆数组和当前堆大小
        self.heap = []
        self.size = 0

    def push(self, x: int) -> None:
        # 将元素插入到堆的末尾
        self.heap.append(x)
        self.size += 1

        # 将新插入的元素上浮到合适位置
        self.heapify_up(self.size - 1)

    def pop(self):
        if self.size == 0:
            return None

        # 记录堆顶最小值
        ans = self.heap[0]

        # 用最后一个元素覆盖堆顶
        self.heap[0] = self.heap[self.size - 1]
        self.heap.pop()
        self.size -= 1

        # 从堆顶开始下沉，维护小根堆
        if self.size > 0:
            self.heapify_down(0)

        return ans

    def heapify_up(self, index: int) -> None:
        # 不断和父节点比较，如果当前节点更小就向上交换
        while index > 0:
            parent = (index - 1) // 2

            if self.heap[parent] <= self.heap[index]:
                break

            self.heap[parent], self.heap[index] = self.heap[index], self.heap[parent]
            index = parent

    def heapify_down(self, index: int) -> None:
        # 不断和左右孩子中较小的节点比较，如果当前节点更大就向下交换
        while True:
            left = index * 2 + 1
            right = index * 2 + 2
            smallest = index

            if left < self.size and self.heap[left] < self.heap[smallest]:
                smallest = left

            if right < self.size and self.heap[right] < self.heap[smallest]:
                smallest = right

            if smallest == index:
                break

            self.heap[index], self.heap[smallest] = self.heap[smallest], self.heap[index]
            index = smallest


def heap_sort(arr: List[int]) -> List[int]:
    min_heap = MinHeap()

    # 所有元素依次入堆
    for x in arr:
        min_heap.push(x)

    ans = []

    # 每次取出当前堆顶最小值，得到升序结果
    while min_heap.size > 0:
        ans.append(min_heap.pop())

    return ans
```

```cpp
class MinHeap {
public:
    vector<int> heap;
    int size;

    MinHeap() {
        // 初始化堆大小
        size = 0;
    }

    void push(int x) {
        // 将元素插入到堆的末尾
        heap.push_back(x);
        size++;

        // 将新插入的元素上浮到合适位置
        heapifyUp(size - 1);
    }

    int pop() {
        // 记录堆顶最小值
        int ans = heap[0];

        // 用最后一个元素覆盖堆顶
        heap[0] = heap[size - 1];
        heap.pop_back();
        size--;

        // 从堆顶开始下沉，维护小根堆
        if (size > 0) {
            heapifyDown(0);
        }

        return ans;
    }

    void heapifyUp(int index) {
        // 不断和父节点比较，如果当前节点更小就向上交换
        while (index > 0) {
            int parent = (index - 1) / 2;

            if (heap[parent] <= heap[index]) {
                break;
            }

            swap(heap[parent], heap[index]);
            index = parent;
        }
    }

    void heapifyDown(int index) {
        // 不断和左右孩子中较小的节点比较，如果当前节点更大就向下交换
        while (true) {
            int left = index * 2 + 1;
            int right = index * 2 + 2;
            int smallest = index;

            if (left < size && heap[left] < heap[smallest]) {
                smallest = left;
            }

            if (right < size && heap[right] < heap[smallest]) {
                smallest = right;
            }

            if (smallest == index) {
                break;
            }

            swap(heap[index], heap[smallest]);
            index = smallest;
        }
    }
};

class Solution {
public:
    vector<int> heapSort(vector<int>& arr) {
        MinHeap minHeap;

        // 所有元素依次入堆

            minHeap.push(x);
        }

        vector<int> ans;

        // 每次取出当前堆顶最小值，得到升序结果
        while (minHeap.size > 0) {
            ans.push_back(minHeap.pop());
        }

        return ans;
    }
};
```

```java
class MinHeap {
    ArrayList<Integer> heap;
    int size;

    public MinHeap() {
        // 初始化堆数组和当前堆大小
        heap = new ArrayList<Integer>();
        size = 0;
    }

    public void push(int x) {
        // 将元素插入到堆的末尾
        heap.add(x);
        size++;

        // 将新插入的元素上浮到合适位置
        heapifyUp(size - 1);
    }

    public int pop() {
        // 记录堆顶最小值
        int ans = heap.get(0);

        // 用最后一个元素覆盖堆顶
        heap.set(0, heap.get(size - 1));
        heap.remove(size - 1);
        size--;

        // 从堆顶开始下沉，维护小根堆
        if (size > 0) {
            heapifyDown(0);
        }

        return ans;
    }

    public void heapifyUp(int index) {
        // 不断和父节点比较，如果当前节点更小就向上交换
        while (index > 0) {
            int parent = (index - 1) / 2;

            if (heap.get(parent) <= heap.get(index)) {
                break;
            }

            int temp = heap.get(parent);
            heap.set(parent, heap.get(index));
            heap.set(index, temp);

            index = parent;
        }
    }

    public void heapifyDown(int index) {
        // 不断和左右孩子中较小的节点比较，如果当前节点更大就向下交换
        while (true) {
            int left = index * 2 + 1;
            int right = index * 2 + 2;
            int smallest = index;

            if (left < size && heap.get(left) < heap.get(smallest)) {
                smallest = left;
            }

            if (right < size && heap.get(right) < heap.get(smallest)) {
                smallest = right;
            }

            if (smallest == index) {
                break;
            }

            int temp = heap.get(index);
            heap.set(index, heap.get(smallest));
            heap.set(smallest, temp);

            index = smallest;
        }
    }
}

class Solution {
    public int[] heapSort(int[] arr) {
        MinHeap minHeap = new MinHeap();

        // 所有元素依次入堆
        for (int x : arr) {
            minHeap.push(x);
        }

        int[] ans = new int[arr.length];

        // 每次取出当前堆顶最小值，得到升序结果
        for (int i = 0; i < arr.length; i++) {
            ans[i] = minHeap.pop();
        }

        return ans;
    }
}
```

```go
	size int
}

func (h *MinHeap) push(x int) {
	// 将元素插入到堆的末尾
	h.heap = append(h.heap, x)
	h.size++

	// 将新插入的元素上浮到合适位置
	h.heapifyUp(h.size - 1)
}

func (h *MinHeap) pop() int {
	// 记录堆顶最小值
	ans := h.heap[0]

	// 用最后一个元素覆盖堆顶
	h.heap[0] = h.heap[h.size-1]
	h.heap = h.heap[:h.size-1]
	h.size--

	// 从堆顶开始下沉，维护小根堆
	if h.size > 0 {
		h.heapifyDown(0)
	}

	return ans
}

func (h *MinHeap) heapifyUp(index int) {
	// 不断和父节点比较，如果当前节点更小就向上交换
	for index > 0 {
		parent := (index - 1) / 2

		if h.heap[parent] <= h.heap[index] {
			break
		}

		h.heap[parent], h.heap[index] = h.heap[index], h.heap[parent]
		index = parent
	}
}

func (h *MinHeap) heapifyDown(index int) {

	for {
		left := index*2 + 1
		right := index*2 + 2
		smallest := index

		if left < h.size && h.heap[left] < h.heap[smallest] {
			smallest = left
		}
```

```c
typedef struct {
    int* heap;
    int size;
    int capacity;
} MinHeap;

void heapifyUp(MinHeap* obj, int index) {
    // 不断和父节点比较，如果当前节点更小就向上交换
    while (index > 0) {
        int parent = (index - 1) / 2;

        if (obj->heap[parent] <= obj->heap[index]) {
            break;
        }

        int temp = obj->heap[parent];
        obj->heap[parent] = obj->heap[index];
        obj->heap[index] = temp;

        index = parent;
    }
}

void heapifyDown(MinHeap* obj, int index) {
    // 不断和左右孩子中较小的节点比较，如果当前节点更大就向下交换
    while (1) {
        int left = index * 2 + 1;
        int right = index * 2 + 2;
        int smallest = index;

        if (left < obj->size && obj->heap[left] < obj->heap[smallest]) {
            smallest = left;
        }

        if (right < obj->size && obj->heap[right] < obj->heap[smallest]) {
            smallest = right;
        }

        if (smallest == index) {
            break;
        }

        int temp = obj->heap[index];
        obj->heap[index] = obj->heap[smallest];
        obj->heap[smallest] = temp;

        index = smallest;
    }
}

void push(MinHeap* obj, int x) {
    // 将元素插入到堆的末尾
    obj->heap[obj->size] = x;
    obj->size++;

    // 将新插入的元素上浮到合适位置
    heapifyUp(obj, obj->size - 1);
}

int pop(MinHeap* obj) {
    // 记录堆顶最小值
    int ans = obj->heap[0];

    // 用最后一个元素覆盖堆顶
    obj->heap[0] = obj->heap[obj->size - 1];
    obj->size--;

    // 从堆顶开始下沉，维护小根堆
    if (obj->size > 0) {
        heapifyDown(obj, 0);
    }

    return ans;
}

int* heapSort(int* arr, int arrSize, int* returnSize) {
    MinHeap minHeap;
    minHeap.heap = (int*)malloc(sizeof(int) * arrSize);
    minHeap.size = 0;
    minHeap.capacity = arrSize;

    // 所有元素依次入堆
    for (int i = 0; i < arrSize; i++) {
        push(&minHeap, arr[i]);
    }

    int* ans = (int*)malloc(sizeof(int) * arrSize);
    *returnSize = arrSize;

    // 每次取出当前堆顶最小值，得到升序结果
    for (int i = 0; i < arrSize; i++) {
        ans[i] = pop(&minHeap);
    }

    free(minHeap.heap);
    return ans;
}
```
