**思路**

使用两个堆，大根堆维护左半边的元素（也就是小于中位数的那些数），小根堆维护右半边的元素（也就是大于等于中位数的那些数）

查询的时候，如果两个堆的大小相等，此时数组长度为偶数，则取两者的堆顶的平均值，否则取大小更大的那个堆的堆顶即可👇

从上图我们也可以理解为什么左半边需要大根堆，右半边需要小根堆。这是因为我们需要取左半边里的最大值以及右半边的最小值才能得到中位数。

#warning-box

**插入新元素时如何调整**？两步走

**1.先判断新元素在哪半边，插入**

**2.调整堆使得平衡**

举个例子：假设在$[5,3,1,4,2]$ 的基础上再来一个$4$，那么考虑$4$大于右边的最小值，所以$4$一定是属于右半边的👇

此时发现插入到右边堆会导致**左右堆个数差别大于等于2**，所以需要将右边堆顶放到左边堆

import heapq

class MedianFinder:
    def __init__(self):
        self.left = [] # 大根堆
        self.right = [] # 小根堆

    def addNum(self, num):
        # 先插入到正确的堆
        if self.right and num >= self.right[0]:
            heapq.heappush(self.right, num)
        else:
            heapq.heappush(self.left, -num)

        # 平衡两个堆
        if len(self.left) > len(self.right) + 1:
            heapq.heappush(self.right, -heapq.heappop(self.left))
        if len(self.right) > len(self.left) + 1:
            heapq.heappush(self.left, -heapq.heappop(self.right))

    def findMedian(self):
        # 如果堆不平衡，则输出个数多的堆的堆顶
        if len(self.left) > len(self.right):
            return -self.left[0]
        elif len(self.left) < len(self.right):
            return self.right[0]

        # 如果堆平衡，此时元素总数为偶数，输出两个堆顶的平均值
        else:
            return (-self.left[0] + self.right[0]) / 2

class MedianFinder {
public:
    priority_queue<int> left; 
    priority_queue<int, vector<int>, greater<int> > right;

    MedianFinder() {
    }

    void addNum(int num) {
        // 先插入到正确的堆
        if (!right.empty() && num >= right.top()) {
            right.push(num);
        } else {
            left.push(num);
        }

        // 平衡两个堆
        if ((int)left.size() > (int)right.size() + 1) {
            right.push(left.top());
            left.pop();
        }
        if ((int)right.size() > (int)left.size() + 1) {
            left.push(right.top());
            right.pop();
        }
    }

    double findMedian() {
        // 如果堆不平衡，则输出个数多的堆的堆顶
        if (left.size() > right.size()) {
            return left.top();
        } else if (left.size() < right.size()) {
            return right.top();
        }

        // 如果堆平衡，此时元素总数为偶数，输出两个堆顶的平均值
        else {
            return (left.top() + right.top()) / 2.0;
        }
    }
};

class MedianFinder {
    PriorityQueue<Integer> left;
    PriorityQueue<Integer> right;

    public MedianFinder() {
        left = new PriorityQueue<>(Collections.reverseOrder()); // 大根堆
        right = new PriorityQueue<>(); // 小根堆
    }

    public void addNum(int num) {
        // 先插入到正确的堆
        if (!right.isEmpty() && num >= right.peek()) {
            right.offer(num);
        } else {
            left.offer(num);
        }

        // 平衡两个堆
        if (left.size() > right.size() + 1) {
            right.offer(left.poll());
        }
        if (right.size() > left.size() + 1) {
            left.offer(right.poll());
        }
    }

    public double findMedian() {
        // 如果堆不平衡，则输出个数多的堆的堆顶

            return left.peek();
        } else if (left.size() < right.size()) {
            return right.peek();

            return (left.peek() + right.peek()) / 2.0;

import "container/heap"

type MaxHeap []int

func (h MaxHeap) Len() int {
	return len(h)
}

func (h MaxHeap) Less(i, j int) bool {
	return h[i] > h[j]
}

func (h MaxHeap) Swap(i, j int) {
	h[i], h[j] = h[j], h[i]
}

func (h *MaxHeap) Push(x interface{}) {
	*h = append(*h, x.(int))
}

func (h *MaxHeap) Pop() interface{} {
	old := *h
	n := len(old)
	x := old[n-1]
	*h = old[:n-1]
	return x
}

type MinHeap []int

func (h MinHeap) Len() int {
	return len(h)
}

func (h MinHeap) Less(i, j int) bool {
	return h[i] < h[j]
}

func (h MinHeap) Swap(i, j int) {
	h[i], h[j] = h[j], h[i]
}

func (h *MinHeap) Push(x interface{}) {
	*h = append(*h, x.(int))
}

func (h *MinHeap) Pop() interface{} {
	old := *h
	n := len(old)
	x := old[n-1]
	*h = old[:n-1]
	return x
}

type MedianFinder struct {
	left  MaxHeap // 大根堆
	right MinHeap // 小根堆
}

func Constructor() MedianFinder {
	return MedianFinder{
		left:  MaxHeap{},
		right: MinHeap{},
	}
}

func (mf *MedianFinder) AddNum(num int) {
	// 先插入到正确的堆
	if len(mf.right) > 0 && num >= mf.right[0] {
		heap.Push(&mf.right, num)
	} else {
		heap.Push(&mf.left, num)
	}

	// 平衡两个堆
	if len(mf.left) > len(mf.right)+1 {
		heap.Push(&mf.right, heap.Pop(&mf.left).(int))
	}
	if len(mf.right) > len(mf.left)+1 {
		heap.Push(&mf.left, heap.Pop(&mf.right).(int))
	}
}

func (mf *MedianFinder) FindMedian() float64 {
	// 如果堆不平衡，则输出个数多的堆的堆顶
	if len(mf.left) > len(mf.right) {
		return float64(mf.left[0])
	} else if len(mf.left) < len(mf.right) {
		return float64(mf.right[0])
	}

	// 如果堆平衡，此时元素总数为偶数，输出两个堆顶的平均值
	return float64(mf.left[0]+mf.right[0]) / 2.0
}

var MedianFinder = function () {
    this.left = []; // 大根堆
    this.right = []; // 小根堆
};

MedianFinder.prototype.pushLeft = function (num) {
    this.left.push(num);

    let index = this.left.length - 1;
    while (index > 0) {
        let parent = Math.floor((index - 1) / 2);

        if (this.left[parent] >= this.left[index]) {
            break;
        }

        let temp = this.left[parent];
        this.left[parent] = this.left[index];
        this.left[index] = temp;

        index = parent;
    }
};

MedianFinder.prototype.popLeft = function () {
    let ans = this.left[0];
    this.left[0] = this.left[this.left.length - 1];
    this.left.pop();

    let index = 0;
    while (true) {
        let left = index * 2 + 1;
        let right = index * 2 + 2;
        let largest = index;

        if (left < this.left.length && this.left[left] > this.left[largest]) {
            largest = left;
        }
        if (right < this.left.length && this.left[right] > this.left[largest]) {
            largest = right;
        }

        if (largest === index) {
            break;
        }

        let temp = this.left[index];
        this.left[index] = this.left[largest];
        this.left[largest] = temp;

        index = largest;
    }

    return ans;
};

typedef struct {
    int* left; // 大根堆
    int leftSize;
    int leftCapacity;

    int* right; // 小根堆
    int rightSize;
    int rightCapacity;
} MedianFinder;

void pushLeft(MedianFinder* obj, int num) {
    if (obj->leftSize == obj->leftCapacity) {
        obj->leftCapacity *= 2;
        obj->left = (int*)realloc(obj->left, sizeof(int) * obj->leftCapacity);
    }

    obj->left[obj->leftSize] = num;
    obj->leftSize++;

    int index = obj->leftSize - 1;
    while (index > 0) {
        int parent = (index - 1) / 2;

        if (obj->left[parent] >= obj->left[index]) {
            break;
        }

        int temp = obj->left[parent];
        obj->left[parent] = obj->left[index];
        obj->left[index] = temp;

        index = parent;
    }
}

int popLeft(MedianFinder* obj) {
    int ans = obj->left[0];
    obj->left[0] = obj->left[obj->leftSize - 1];
    obj->leftSize--;

    int index = 0;
    while (1) {
        int left = index * 2 + 1;
        int right = index * 2 + 2;
        int largest = index;

        if (left < obj->leftSize && obj->left[left] > obj->left[largest]) {
            largest = left;
        }
        if (right < obj->leftSize && obj->left[right] > obj->left[largest]) {
            largest = right;
        }

        if (largest == index) {
            break;
        }

        obj->left[index] = obj->left[largest];
        obj->left[largest] = temp;

        index = largest;
    }

    return ans;
}

void pushRight(MedianFinder* obj, int num) {

        index = parent;

}

MedianFinder* medianFinderCreate() {
    MedianFinder* obj = (MedianFinder*)malloc(sizeof(MedianFinder));

void medianFinderAddNum(MedianFinder* obj, int num) {
    // 先插入到正确的堆
    if (obj->rightSize > 0 && num >= obj->right[0]) {
        pushRight(obj, num);

**1. 本题的核心思路是什么？**

本题核心思路：使用两个堆，一个堆存储小于等于中位数的元素，一个堆存储大于中位数的元素。

1.当插入一个元素时，先插入到正确的堆，然后平衡两个堆，保证两个堆的元素个数差不超过1。

2.当查找中位数时，如果堆不平衡，则输出个数多的堆的堆顶。

3.如果堆平衡，此时元素总数为偶数，输出两个堆顶的平均值。
