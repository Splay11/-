### 思路：哈希 + 贪心

一个朴素的思路是：用数组构建哈希，枚举所有点$x$作为起点，循环地哈希查询$x+1,x+2,...$来寻找它的最长连续序列，一直到不存在$x + k$ , 此时以$x$作为起点的长度就是$k$

来看一下这段代码：

```python
class Solution:
    def longestConsecutive(self, nums):
        if not nums:
            return 0

        num_set = set(nums)  # 将数组存入哈希集合
        longest_streak = 0

        for num in num_set:
            # 枚举当前位置作为起点
            current_num = num
            current_streak = 1

            # 向右拓展序列长度
            while current_num + 1 in num_set:  # 计算序列长度
                current_num += 1
                current_streak += 1

            longest_streak = max(longest_streak, current_streak)  # 更新最大长度

        return longest_streak
```

```c
    int key;
    int used;
} HashNode;

static unsigned int hashInt(int x) {
    return (unsigned int)x * 2654435761u;
}

static int contains(HashNode* table, int size, int key) {
    unsigned int idx = hashInt(key) % size;

    while (table[idx].used) {
        if (table[idx].key == key) {
            return 1;
        }
        idx = (idx + 1) % size;
    }

    return 0;
}

static void insert(HashNode* table, int size, int key) {
    unsigned int idx = hashInt(key) % size;

    while (table[idx].used) {
        if (table[idx].key == key) {
            return;
        }
        idx = (idx + 1) % size;
    }

    table[idx].used = 1;
    table[idx].key = key;
}

int longestConsecutive(int* nums, int numsSize) {
    if (numsSize == 0) {
        return 0;
    }

    int tableSize = numsSize * 3 + 1;
    HashNode* num_set = (HashNode*)calloc(tableSize, sizeof(HashNode)); // 将数组存入哈希集合

    for (int i = 0; i < numsSize; i++) {
        insert(num_set, tableSize, nums[i]);
    }

    int longest_streak = 0;

    for (int i = 0; i < numsSize; i++) {

            current_num += 1;
            current_streak += 1;
        }

        if (current_streak > longest_streak) { // 更新最大长度
            longest_streak = current_streak;
        }
    }

    free(num_set);
```

#code-switcher

这样的复杂度是$O(n^2)$ 的，因为可能存在如下序列使得每个点都需要往后寻找很多数:

$[1,2,3,...,n]$

一个比较巧妙的点在于：**考虑最优解的性质：**

$$[x_1,x_2,x_3,...,x_k]$$

中的开头元素：$x_1$ 一定满足 : 数组中不存在$y$ 满足$y = x_1 - 1$

原因很简单，因为如果有这样的$y = x_1 - 1$ , 那么显然最优解可以更大。


换句话说，我们上述算法可以优化的点是：没必要枚举所有点作为起点，而是枚举"不存在$x - 1$" 的$x$ 作为起点。

因为那些存在$x - 1$ 的 $x$ 作为 起点，一定不会比现在的答案更优。

所以上述代码只需要改一行即可：

```python
from typing import List


class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        # 如果数组为空，直接返回 0
        if not nums:
            return 0

                    current_streak += 1

        # 返回最长连续序列长度
```

```cpp
                    current_streak += 1;

                longest_streak = max(longest_streak, current_streak);
```

```java
    public int longestConsecutive(int[] nums) {
        // 如果数组为空，直接返回 0
        if (nums == null || nums.length == 0) {
            return 0;
        }

        // 将数组存入哈希集合
        Set<Integer> numSet = new HashSet<>();
        for (int num : nums) {
            numSet.add(num);
        }

        int longestStreak = 0;

            if (!numSet.contains(num - 1)) {

                    currentStreak += 1;

                // 更新最大长度
                longestStreak = Math.max(longestStreak, currentStreak);

        }

        // 返回最长连续序列长度
        return longestStreak;
    }
}
```

```c
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int key;
    bool used;
} HashEntry;

static unsigned int hashValue(int key, int capacity) {
    unsigned int x = (unsigned int)key;
    x ^= x >> 16;
    return x % capacity;
}

static bool contains(HashEntry* table, int capacity, int key) {
    unsigned int pos = hashValue(key, capacity);

    while (table[pos].used) {
        if (table[pos].key == key) {
            return true;
        }
        pos = (pos + 1) % capacity;
    }

    return false;
}

static void insert(HashEntry* table, int capacity, int key) {
    unsigned int pos = hashValue(key, capacity);

    while (table[pos].used) {
        if (table[pos].key == key) {
            return;
        }
        pos = (pos + 1) % capacity;
    }

    table[pos].key = key;
    table[pos].used = true;
}

int longestConsecutive(int* nums, int numsSize) {
    // 如果数组为空，直接返回 0
    if (numsSize == 0) {
        return 0;
    }

    // 创建哈希集合
    int capacity = numsSize * 3 + 1;
    HashEntry* table = (HashEntry*)calloc(capacity, sizeof(HashEntry));

    // 将数组元素加入哈希集合
    for (int i = 0; i < numsSize; i++) {
        insert(table, capacity, nums[i]);
    }

    // 记录最长连续序列长度
    int longestStreak = 0;

    // 遍历原数组
    for (int i = 0; i < numsSize; i++) {
        int num = nums[i];

        // 重点：只有当 num - 1 不存在时，num 才可能是连续序列的起点
        if (!contains(table, capacity, num - 1)) {
            int currentNum = num;
            int currentStreak = 1;

            // 向右扩展，计算当前连续序列长度
            while (contains(table, capacity, currentNum + 1)) {
                currentNum++;
                currentStreak++;
            }

            // 更新最大长度
            if (currentStreak > longestStreak) {
                longestStreak = currentStreak;
            }
        }
    }

    free(table);

    // 返回最长连续序列长度
    return longestStreak;
}
```

这样的复杂度是$O(n)$ 的，因为每个数只会被遍历一遍。

证明：`for`枚举所有数需要$n$次。`while`循环枚举所有`非起点`元素，而`非起点`元素最多也只有$n$个，所以复杂度为$O(n)$ 

### 面试问答

**1.这道题的整体思路是什么？**

用数组构建哈希集合，枚举所有不存在$num - 1$的值$num$ 作为起点，通过不断的在哈希集合中查询当前数字加1的方式来统计连续的数字有多少个，更新最长序列的长度。

时间复杂度为:$O(n)$ , 因为每个元素恰好被访问一次。
