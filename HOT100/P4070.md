**思路**

本题还是一样的回溯模型：递归的从左往右考虑每个位置的数字，循环来确定每个位置所代表的数字，然后继续递归确定下一个位置。

为了方便实现，在回溯之前，先列出映射表，代表每个数字所对应的字母列表。

```java
    public List<String> letterCombinations(String digits) {
        // 数字到字母的映射表
        Map<Character, List<Character>> phoneMap = new HashMap<>();
        phoneMap.put('2', Arrays.asList('a', 'b', 'c'));
        phoneMap.put('3', Arrays.asList('d', 'e', 'f'));
        phoneMap.put('4', Arrays.asList('g', 'h', 'i'));
        phoneMap.put('5', Arrays.asList('j', 'k', 'l'));
        phoneMap.put('6', Arrays.asList('m', 'n', 'o'));
        phoneMap.put('7', Arrays.asList('p', 'q', 'r', 's'));
        phoneMap.put('8', Arrays.asList('t', 'u', 'v'));
        phoneMap.put('9', Arrays.asList('w', 'x', 'y', 'z'));

        List<String> ans = new ArrayList<>();
        // curr 表示当前正在构造的字母组合
        StringBuilder curr = new StringBuilder();

        dfs(0, digits, phoneMap, curr, ans);
        return ans;
    }

    private void dfs(int index, String digits, Map<Character, List<Character>> phoneMap, StringBuilder curr, List<String> ans) {
        // 所有位置都已经填完，得到一个完整组合
        if (index == digits.length()) {
            ans.add(curr.toString());
            return;
        }

        // 枚举当前数字可以对应的每一个字母
        for (char ch : phoneMap.get(digits.charAt(index))) {
            // 选择当前字母
            curr.append(ch);
            // 继续递归处理下一个位置
            dfs(index + 1, digits, phoneMap, curr, ans);
            // 回溯，撤销选择
            curr.deleteCharAt(curr.length() - 1);
        }
    }
}
```

```go
	phoneMap := map[byte]string{
		'2': "abc",
		'3': "def",
		'4': "ghi",
		'5': "jkl",
		'6': "mno",
		'7': "pqrs",
		'8': "tuv",
		'9': "wxyz",
	}

	n := len(digits)
	ans := []string{}
	// curr 表示当前正在构造的字母组合
	curr := []byte{}

	var dfs func(int)
	dfs = func(index int) {
		// 所有位置都已经填完，得到一个完整组合
		if index == n {
			ans = append(ans, string(curr))
			return
		}

		// 枚举当前数字可以对应的每一个字母
		letters := phoneMap[digits[index]]
		for i := 0; i < len(letters); i++ {
			// 选择当前字母
			curr = append(curr, letters[i])
			// 继续递归处理下一个位置
			dfs(index + 1)
			// 回溯，撤销选择
			curr = curr[:len(curr)-1]
		}
	}

	dfs(0)
	return ans
}
```

```c
void dfs(char* digits, int n, int index, char* curr, char** ans, int* returnSize, char* phoneMap[]) {
    // 所有位置都已经填完，得到一个完整组合
    if (index == n) {
        ans[*returnSize] = (char*)malloc(sizeof(char) * (n + 1));
        for (int i = 0; i < n; i++) {
            ans[*returnSize][i] = curr[i];
        }
        ans[*returnSize][n] = '\0';
        (*returnSize)++;
        return;
    }

    // 枚举当前数字可以对应的每一个字母
    char* letters = phoneMap[digits[index] - '0'];
    for (int i = 0; letters[i] != '\0'; i++) {
        // 选择当前字母
        curr[index] = letters[i];
        // 继续递归处理下一个位置
        dfs(digits, n, index + 1, curr, ans, returnSize, phoneMap);
    }
}

char** letterCombinations(char* digits, int* returnSize) {
    // 数字到字母的映射表
    char* phoneMap[10];
    phoneMap[2] = "abc";
    phoneMap[3] = "def";
    phoneMap[4] = "ghi";
    phoneMap[5] = "jkl";
    phoneMap[6] = "mno";
    phoneMap[7] = "pqrs";
    phoneMap[8] = "tuv";
    phoneMap[9] = "wxyz";

    int n = 0;
    while (digits[n] != '\0') {
        n++;
    }

    int total = 1;
    for (int i = 0; i < n; i++) {
        if (digits[i] == '7' || digits[i] == '9') {
            total *= 4;
        } else {
            total *= 3;
        }
    }

    char** ans = (char**)malloc(sizeof(char*) * total);
    char* curr = (char*)malloc(sizeof(char) * (n + 1));
    *returnSize = 0;

    dfs(digits, n, 0, curr, ans, returnSize, phoneMap);

    return ans;
}
```

### 面试问答

**1. 本题的核心思路是什么？**

这道题的核心思路是**回溯**。我们按顺序从左到右处理每一位数字，每次根据映射表找到当前数字对应的所有字母，然后依次选择一个字母放到当前位置，继续递归处理下一位数字。当所有位置都填完时，就得到一种合法答案。

时间复杂度是 $O(3^m × 4^n)$，其中 `m` 是对应 3 个字母的数字个数，`n` 是对应 4 个字母的数字个数；空间复杂度是 $O(|digits|)$ ,

ps:$|x|$操作代表取$x$的长度。
