```python
import sys
import heapq


# 处理所有操作，返回所有 PM 操作的答案
def solve_operations(heap, operations):
    ans = []

    # 将初始数组建成最小堆
    heapq.heapify(heap)

    for op in operations:
        # 插入操作
        if op[0] == "I":
            x = int(op[1])
            heapq.heappush(heap, x)

        # 查询当前最小值
        elif op[0] == "PM":
            ans.append(str(heap[0]))

        # 删除当前最小值
        else:
            heapq.heappop(heap)

    return ans


def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1

    heap = []
    for _ in range(n):
        heap.append(int(data[idx]))
        idx += 1

    operations = []
    for _ in range(m):
        op = data[idx]
        idx += 1

        if op == "I":
            x = data[idx]
            idx += 1
            operations.append((op, x))
        else:
            operations.append((op,))

    ans = solve_operations(heap, operations)
    print("\n".join(ans))


if __name__ == "__main__":
    main()
```

```java
import java.util.PriorityQueue;
import java.util.Scanner;
import java.util.ArrayList;

public class Main {

    // 处理所有操作，返回所有 PM 操作的答案
    public static ArrayList<Integer> solveOperations(PriorityQueue<Integer> heap, int m, Scanner sc) {
        ArrayList<Integer> ans = new ArrayList<>();

        for (int i = 0; i < m; i++) {
            String op = sc.next();

            // 插入操作
            if (op.equals("I")) {
                int x = sc.nextInt();
                heap.offer(x);
            }

            // 查询当前最小值
            else if (op.equals("PM")) {
                ans.add(heap.peek());
            }

            // 删除当前最小值
            else {
                heap.poll();
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        PriorityQueue<Integer> heap = new PriorityQueue<>();

        // 读入初始数组并加入最小堆
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            heap.offer(x);
        }

        ArrayList<Integer> ans = solveOperations(heap, m, sc);

        StringBuilder sb = new StringBuilder();

        // 输出所有查询结果
        for (int x : ans) {
            sb.append(x).append('\n');
        }

        System.out.print(sb.toString());
        sc.close();
    }
}
```

```Go
package main

import (
	"bufio"
	"container/heap"
	"fmt"
	"os"
)

// 定义一个整数最小堆
type MinHeap []int

func (h MinHeap) Len() int {
	return len(h)
}

// 小根堆：值小的优先级更高
func (h MinHeap) Less(i, j int) bool {
	return h[i] < h[j]
}

func (h MinHeap) Swap(i, j int) {
	h[i], h[j] = h[j], h[i]
}

// 向堆中插入元素
func (h *MinHeap) Push(x interface{}) {
	*h = append(*h, x.(int))
}

// 删除堆顶元素
func (h *MinHeap) Pop() interface{} {
	old := *h
	n := len(old)
	x := old[n-1]
	*h = old[:n-1]
	return x
}

func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()

	var n, m int
	fmt.Fscan(in, &n, &m)

	// 读入初始数组
	h := &MinHeap{}
	for i := 0; i < n; i++ {
		var x int
		fmt.Fscan(in, &x)
		*h = append(*h, x)
	}

	// 将初始数组建成最小堆
	heap.Init(h)

	for i := 0; i < m; i++ {
		var op string
		fmt.Fscan(in, &op)

		// 插入操作
		if op == "I" {
			var x int
			fmt.Fscan(in, &x)
			heap.Push(h, x)
		} else if op == "PM" {
			// 查询当前最小值
			fmt.Fprintln(out, (*h)[0])
		} else {
			// 删除当前最小值
			heap.Pop(h)
		}
	}
}
```

```javascript
const fs = require("fs");

const input = fs.readFileSync(0, "utf8").trim().split(/\s+/);
let idx = 0;

const n = Number(input[idx++]);
const m = Number(input[idx++]);

// 手写最小堆
class MinHeap {
    constructor() {
        this.heap = [];
    }

    // 交换两个位置的元素
    swap(i, j) {
        const temp = this.heap[i];
        this.heap[i] = this.heap[j];
        this.heap[j] = temp;
    }

    // 插入元素
    push(x) {
        this.heap.push(x);
        this.up(this.heap.length - 1);
    }

    // 向上调整
    up(i) {
        while (i > 0) {
            const parent = Math.floor((i - 1) / 2);
            if (this.heap[parent] <= this.heap[i]) break;

            this.swap(parent, i);
            i = parent;
        }
    }

    // 删除并返回堆顶元素
    pop() {
        const top = this.heap[0];
        const last = this.heap.pop();

        if (this.heap.length > 0) {
            this.heap[0] = last;
            this.down(0);
        }

        return top;
    }

    // 向下调整
    down(i) {
        const n = this.heap.length;

        while (true) {
            let smallest = i;
            const left = i * 2 + 1;
            const right = i * 2 + 2;

            if (left < n && this.heap[left] < this.heap[smallest]) {
                smallest = left;
            }

            if (right < n && this.heap[right] < this.heap[smallest]) {
                smallest = right;
            }

            if (smallest === i) break;

            this.swap(i, smallest);
            i = smallest;
        }
    }

    // 查询堆顶最小值
    top() {
        return this.heap[0];
    }
}

const heap = new MinHeap();

// 读入初始数组并插入最小堆
for (let i = 0; i < n; i++) {
    heap.push(Number(input[idx++]));
}

    const op = input[idx++];

        const x = Number(input[idx++]);
        heap.push(x);
    }

    // 查询当前最小值

        ans.push(String(heap.top()));
```

```c
#include <stdio.h>

#define MAXN 200005

int heap[MAXN];
int sz = 0;

// 交换两个元素
void swap(int *a, int *b) {
    int t = *a;
    *a = *b;
    *b = t;
}

// 向上调整
void up(int i) {
    while (i > 1) {
        int parent = i / 2;

        // 如果父节点已经不大于当前节点，则满足最小堆性质
        if (heap[parent] <= heap[i]) {
            break;
        }

        swap(&heap[parent], &heap[i]);
        i = parent;
    }
}

// 向下调整
void down(int i) {
    while (1) {
        int smallest = i;
        int left = i * 2;
        int right = i * 2 + 1;

        if (left <= sz && heap[left] < heap[smallest]) {
            smallest = left;
        }

        if (right <= sz && heap[right] < heap[smallest]) {
            smallest = right;
        }

        if (smallest == i) {
            break;
        }

        swap(&heap[i], &heap[smallest]);
        i = smallest;
    }
}

// 插入元素
void push(int x) {
    heap[++sz] = x;
    up(sz);
}

// 删除堆顶最小值
void pop() {
    heap[1] = heap[sz--];
    down(1);
}

// 查询堆顶最小值
int top() {
    return heap[1];
}

int main() {
    int n, m;
    scanf("%d %d", &n, &m);

    // 读入初始数组并插入最小堆
    for (int i = 0; i < n; i++) {
        int x;
        scanf("%d", &x);
        push(x);
    }

    for (int i = 0; i < m; i++) {
        char op[5];
        scanf("%s", op);

        // 插入操作
        if (op[0] == 'I') {
            int x;
            scanf("%d", &x);
            push(x);
        }

        // 查询当前最小值
        else if (op[0] == 'P') {
            printf("%d\n", top());
        }

        // 删除当前最小值
        else {
            pop();
        }
    }

    return 0;
}
```
