**思路**

右视图 本质是 从上到下输出 层序遍历中的每一层最右侧的那个节点。

所以在 [LeetCode 102. 二叉树的层序遍历](https://codefun2000.com/codenote/hot100/P0041)的方法一的基础上，只需要每次将本层最后一个节点放入`res` , 而不是整行放入 即可。

**代码只需要改两行**。

```python id="g2pnjy"
class Solution:
    def rightSideView(self, root):
        # 空树直接返回空数组
        if not root:
            return []

        # q 存放当前层的所有节点
        q = [root]

        # res 存放每一层最右侧节点的值
        res = [root.val]

        # 每一轮处理一整层节点
        while q:
            # q_next_level 存放下一层的所有节点
            q_next_level = []

            # val_arr 存放下一层所有节点的值
            val_arr = []

            # 从左到右遍历当前层节点，并拓展下一层
            for node in q:
                # 先加入左孩子
                if node.left:
                    q_next_level.append(node.left)
                    val_arr.append(node.left.val)

                # 再加入右孩子
                if node.right:
                    q_next_level.append(node.right)
                    val_arr.append(node.right.val)

            # 进入下一层
            q = q_next_level

            # 如果下一层不为空，就把下一层最右侧节点加入答案
            if val_arr:
                res.append(val_arr[-1])

        return res
```

```go id="usog66"
func rightSideView(root *TreeNode) []int {
	// 空树直接返回空数组
	if root == nil {
		return []int{}
	}

	// q 存放当前层的所有节点
	q := []*TreeNode{root}

	// res 存放每一层最右侧节点的值
	res := []int{root.Val}

	// 每一轮处理一整层节点
	for len(q) > 0 {
		// qNextLevel 存放下一层的所有节点
		qNextLevel := make([]*TreeNode, 0)

		// valArr 存放下一层所有节点的值
		valArr := make([]int, 0)

		// 从左到右遍历当前层节点，并拓展下一层
		for _, node := range q {
			// 先加入左孩子
			if node.Left != nil {
				qNextLevel = append(qNextLevel, node.Left)
				valArr = append(valArr, node.Left.Val)
			}

			// 再加入右孩子
			if node.Right != nil {
				qNextLevel = append(qNextLevel, node.Right)
				valArr = append(valArr, node.Right.Val)
			}
		}

		// 进入下一层
		q = qNextLevel

		// 如果下一层不为空，就把下一层最右侧节点加入答案
		if len(valArr) > 0 {
			res = append(res, valArr[len(valArr)-1])
		}
	}

	return res
}
```

```c id="4zfhse"
int* rightSideView(struct TreeNode* root, int* returnSize) {
    // 空树直接返回空数组
    if (root == NULL) {
        *returnSize = 0;
        return (int*)malloc(sizeof(int) * 0);
    }

    // res 存放每一层最右侧节点的值
    int* res = (int*)malloc(sizeof(int) * 100005);
    *returnSize = 0;
    res[(*returnSize)++] = root->val;

    // q 存放当前层的所有节点
    struct TreeNode** q = (struct TreeNode**)malloc(sizeof(struct TreeNode*) * 100005);
    int qSize = 0;
    q[qSize++] = root;

    // 每一轮处理一整层节点
    while (qSize > 0) {
        // qNextLevel 存放下一层的所有节点
        struct TreeNode** qNextLevel = (struct TreeNode**)malloc(sizeof(struct TreeNode*) * 100005);
        int nextSize = 0;

        // valArr 存放下一层所有节点的值
        int* valArr = (int*)malloc(sizeof(int) * 100005);
        int valSize = 0;

        // 从左到右遍历当前层节点，并拓展下一层
        for (int i = 0; i < qSize; i++) {
            struct TreeNode* node = q[i];

            // 先加入左孩子
            if (node->left != NULL) {
                qNextLevel[nextSize++] = node->left;
                valArr[valSize++] = node->left->val;
            }

            // 再加入右孩子
            if (node->right != NULL) {
                qNextLevel[nextSize++] = node->right;
                valArr[valSize++] = node->right->val;
            }
        }

        // 如果下一层不为空，就把下一层最右侧节点加入答案
        if (valSize > 0) {
            res[(*returnSize)++] = valArr[valSize - 1];
        }

        // 进入下一层
        free(q);
        free(valArr);
        q = qNextLevel;
        qSize = nextSize;
    }

    free(q);
    return res;
}
```
