# LCA的两种形态

一棵树上的最近公共祖先 `LCA` 有两种可能的形态：

1. `p` 和 `q` 是祖先-后代关系  
   比如 `p` 在 `q` 的祖先链上（或反过来），那么 `LCA` 就是更上面的那个节点

2. `p` 和 `q` 分别位于某个节点的左右子树  
   这时它们第一次“汇合”的那个节点，就是最近公共祖先。

# 方法一：统计节点法（自底向上的DFS）

一个直观的想法(**性质1**)：如果当前子树同时存在`p,q`节点，那么`LCA`一定也存在于该子树，而且根节点可能就是`LCA`👇

![img](https://codefun2000.com/p/4259/file/%E6%80%A7%E8%B4%A81-1.svg?type=additional_file)

当然满足**性质1**的子树的根也不一定就是`LCA`👇

![img](https://codefun2000.com/p/4259/file/%E6%80%A7%E8%B4%A81-2.svg?type=additional_file)

现在的问题是：一棵树可能有很多个子树都满足**性质1**，如何找到**最下方的那一个子树的根**呢：**自底向上的DFS!**

在这个问题里，我们去统计每个节点作为根的子树中有多少个目标点，**第一个满足`目标点数量 = 2` 的节点就是答案**

#success-box

1.定义一个全局变量`LCA = None` 

2.定义`dfs(node)` 返回 以`node`为根的子树中目标点的个数`cnt`。 

3.自底向上的DFS,统计左右子树的目标点，以及检查`node`是否属于`p , q` 以得到`cnt`。

4.每次检查`node`是否满足**性质1(cnt = 2)** , `LCA`就是DFS过程中最早的满足**性质1**的节点。

```python
from collections import deque
from typing import Optional


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def lowestCommonAncestor(
        self,
        root: Optional[TreeNode],
        p: TreeNode,
        q: TreeNode,
    ) -> Optional[TreeNode]:
        lca = None
        # dfs 统计以node为根的子树中目标点的个数
        def dfs(node: Optional[TreeNode]) -> Optional[TreeNode]:
            nonlocal lca
            if node is None:
                return 0
            cnt = 0
            if node is p or node is q:
                cnt += 1
            cnt += dfs(node.left)
            cnt += dfs(node.right)
            if cnt == 2:
                # 如果当前子树中同时存在p和q，那么最先满足条件的节点就是LCA
                # 已经找到过一次LCA，那么后续的一定不会比这个更近
                if lca is None:
                    lca = node
            # 返回以node为根的子树中目标点的个数
            return cnt 
        dfs(root)
        return lca


def build_tree(values: list[str]) -> tuple[Optional[TreeNode], dict[int, TreeNode]]:
    if not values or values[0] == "null":
        return None, {}

    root = TreeNode(int(values[0]))
    mp = {root.val: root}
    q = deque([root])
    idx = 1

    while q and idx < len(values):
        cur = q.popleft()

        if idx < len(values) and values[idx] != "null":
            left = TreeNode(int(values[idx]))
            cur.left = left
            mp[left.val] = left
            q.append(left)
        idx += 1

        if idx < len(values) and values[idx] != "null":
            right = TreeNode(int(values[idx]))
            cur.right = right
            mp[right.val] = right
            q.append(right)
        idx += 1

    return root, mp


def main():
    # ACM 输入：
    # 第 1 行：层序序列（用 null 表示空）
    # 第 2 行：p q（节点值）
    tree_tokens = input().split()
    p_val, q_val = map(int, input().split())

    root, mp = build_tree(tree_tokens)
    p = mp[p_val]
    q = mp[q_val]

    ans = Solution().lowestCommonAncestor(root, p, q)
    print(ans.val if ans is not None else "")


if __name__ == "__main__":
    main()

#include <bits/stdc++.h>
using namespace std;


class TreeNode {
public:
    int val;
    TreeNode* left;
    TreeNode* right;

    TreeNode(int val = 0, TreeNode* left = nullptr, TreeNode* right = nullptr) {
        this->val = val;
        this->left = left;
        this->right = right;
    }
};


class Solution {
public:
    TreeNode* lowestCommonAncestor(
        TreeNode* root,
        TreeNode* p,
        TreeNode* q
    ) {
        TreeNode* lca = nullptr;

        // dfs 统计以node为根的子树中目标点的个数
        function<int(TreeNode*)> dfs = [&](TreeNode* node) -> int {
            if (node == nullptr) {
                return 0;
            }

            int cnt = 0;

            if (node == p || node == q) {
                cnt += 1;
            }

            cnt += dfs(node->left);
            cnt += dfs(node->right);

            if (cnt == 2) {
                // 如果当前子树中同时存在p和q，那么最先满足条件的节点就是LCA
                // 已经找到过一次LCA，那么后续的一定不会比这个更近
                if (lca == nullptr) {
                    lca = node;
                }
            }

            // 返回以node为根的子树中目标点的个数
            return cnt;
        };

        dfs(root);
        return lca;
    }
};


pair<TreeNode*, unordered_map<int, TreeNode*>> build_tree(vector<string>& values) {
    if (values.empty() || values[0] == "null") {
        return {nullptr, {}};
    }

    TreeNode* root = new TreeNode(stoi(values[0]));
    unordered_map<int, TreeNode*> mp;
    mp[root->val] = root;

    queue<TreeNode*> q;
    q.push(root);

    int idx = 1;

    while (!q.empty() && idx < values.size()) {
        TreeNode* cur = q.front();
        q.pop();

        if (idx < values.size() && values[idx] != "null") {
            TreeNode* left = new TreeNode(stoi(values[idx]));
            cur->left = left;
            mp[left->val] = left;
            q.push(left);
        }
        idx += 1;

        if (idx < values.size() && values[idx] != "null") {
            TreeNode* right = new TreeNode(stoi(values[idx]));
            cur->right = right;
            mp[right->val] = right;
            q.push(right);
        }
        idx += 1;
    }

    return {root, mp};
}


int main() {

    // 第 1 行：层序序列（用 null 表示空）
    // 第 2 行：p q（节点值）

    stringstream ss(line);
    vector<string> tree_tokens;
    string token;

    while (ss >> token) {
        tree_tokens.push_back(token);
    }

    int p_val, q_val;
    cin >> p_val >> q_val;

    auto [root, mp] = build_tree(tree_tokens);

    TreeNode* p = mp[p_val];
    TreeNode* q = mp[q_val];

    TreeNode* ans = Solution().lowestCommonAncestor(root, p, q);

    if (ans != nullptr) {
        cout << ans->val << endl;
    } else {
        cout << "" << endl;
    }

    return 0;
}

class BuildResult {
    TreeNode root;
    Map<Integer, TreeNode> mp;

    BuildResult(TreeNode root, Map<Integer, TreeNode> mp) {
        this.root = root;
        this.mp = mp;
    }
}


public class Main {
    public static BuildResult build_tree(String[] values) {
        if (values.length == 0 || values[0].equals("null")) {
            return new BuildResult(null, new HashMap<>());
        }

        TreeNode root = new TreeNode(Integer.parseInt(values[0]));
        Map<Integer, TreeNode> mp = new HashMap<>();
        mp.put(root.val, root);

        Queue<TreeNode> q = new ArrayDeque<>();
        q.offer(root);

        int idx = 1;

        while (!q.isEmpty() && idx < values.length) {
            TreeNode cur = q.poll();

            if (idx < values.length && !values[idx].equals("null")) {
                TreeNode left = new TreeNode(Integer.parseInt(values[idx]));
                cur.left = left;
                mp.put(left.val, left);
                q.offer(left);
            }
            idx += 1;

            if (idx < values.length && !values[idx].equals("null")) {
                TreeNode right = new TreeNode(Integer.parseInt(values[idx]));
                cur.right = right;
                mp.put(right.val, right);
                q.offer(right);
            }
            idx += 1;
        }

        return new BuildResult(root, mp);
    }

        BuildResult result = build_tree(tree_tokens);
        TreeNode root = result.root;
        Map<Integer, TreeNode> mp = result.mp;

        TreeNode p = mp.get(p_val);
        TreeNode q = mp.get(q_val);

        TreeNode ans = new Solution().lowestCommonAncestor(root, p, q);

        System.out.println(ans != null ? ans.val : "");
    }
}

## 面试问答

**1. 本题的思路是什么？**

**自底向上的DFS（后序遍历）**：定义 `dfs(root)` 返回“当前子树里目标点个数`cnt`”。当遍历到某个点`node` 第一次满足`cnt = 2` 时 就是`LCA`。

这个过程每个节点只访问一次，时间复杂度 `O(n)`，空间复杂度主要是递归栈，最坏 `O(n)`。

# 方法二：收集节点法(更加简洁，但同时更难理解)

和方法一的唯一区别是：我们不用全局变量`lca` 来记录答案，而是将其融入到`dfs`的过程里去往上返回。

所以这个方法的`dfs` 返回的值是`p/q/LCA` 其中的一个，根据不同情况进行讨论

我们先来直接看做法，先对整体算法有个把握，然后再深入讲每个点的原理👇

#warning-box

**递归函数 `dfs(root)` 的含义**：

1.返回值是“在 `root` 这棵子树里，找到的目标点，它可能是 `p/q/LCA` 节点中的某一个”

2.如果当前子树里都没有 `p/q`，返回 `None`

**递归规则**：

1.递归出口 : `root = None`，返回 `None`

2.如果 `root` 就是 `p` 或 `q`，直接返回 `root`

3.递归查左子树得 `left`，查右子树得 `right`

4.若 `left` 和 `right` 都非空，说明 `p、q` 分别在两边，当前 `root` 就是 LCA

5.否则返回 `left or right`（把命中的那个往上传）

#warning-box

接下来详细讲一下这个算法的原理：假设当前遍历到 `root`👇

#warning-box

**情况1:`root` 恰好是 `p` 或 `q`  , 此时意味着我们搜到了一个节点，直接返回`root` 即可**

#warning-box

**说明**：这种情况涵盖了两种不同的case:

case1：子树内还有另外一个目标点，恰好对应 **LCA的形态1** ， 此时我们等于是找到了更高的那一个点，也就是`LCA` ， 返回这个点即可。

case2：子树内没有目标点 ，此时直接返回该点。


![img](https://codefun2000.com/p/4259/file/lca_return_hit_cases.svg?type=additional_file)

#success-box

tips:从上图可以发现，只要我们找到了一个`root`属于`p/q` , 无论它下面还有没有另外一个点，我们都可以直接返回，只不过这两种情况返回的时候，返回值本身的意义不同。case1返回的是`lca` , case2返回的是`p/q`中的一个。但这个方法妙就妙在，它并不关心你到底返回的是哪一种，只要你返回的是`p/q/lca`其中一个就行。

#success-box


情况2:`root` 不是 `p` 或 `q` , 那继续递归左右子树，搜寻目标值👇
 
#warning-box
**2.1 如果递归后左子树和右子树分别都收集上来一个目标点，此时`root` 就是 `LCA` , 返回命中的`LCA`**

#warning-box

**说明**：此时就等于是找到了一个最下方的，子树内有两个目标点的 根节点。根据**方法一**的分析，这个点就是`LCA`

![img](https://codefun2000.com/p/4259/file/lca_double_hit_case.svg?type=additional_file)


#warning-box

**2.2 如果递归后只收集上来一个目标点，直接返回即可**。

#warning-box

**说明**：这种情况对应两种不同的case：

case1：这个点已经是`LCA`了，这就说明我们的递归函数已经帮我们找到了`LCA`，无需其他多余操作，应当直接返回。

case2：这个点是`p/q` 中的其中一个，但是另外一边子树没有目标点。此时`LCA`位于更上方，那么自然需要返回这个点给更上方的节点做决策。

![img](https://codefun2000.com/p/4259/file/lca_single_hit_return_cases.svg?type=additional_file)

**情况3:`root = None`  ， 代表我们到达了叶子节点的下一层，即递归出口，直接返回**。

![img](https://codefun2000.com/p/4259/file/lca_none_base_case.svg?type=additional_file)


这四种情况合起来，就是 DFS 的完整决策。

总结：这个方法之所以被称作"收集节点法"，因为我们在遍历的过程中，本质上是搜索`p,q`的实际位置，一旦遇到就往上返回。且找到第一个相遇的位置时得到LCA ， 接下来就改成将LCA继续往上返回。**整个过程就像是`p/q`两个节点会不断的往上层节点跳，到两者汇合时，再继续让LCA往上跳**。

class Solution:
    def lowestCommonAncestor(self, root, p, q):
        # dfs(node): 在以 node 为根的子树中收集 p / q / LCA
        def dfs(node):
            # 情况3：空节点，说明当前子树没有目标点
            if node is None:
                return None

            # 情况1：当前节点就是 p 或 q，直接把当前节点向上返回
            if node is p or node is q:
                return node

            # 递归去左子树收集目标点
            left = dfs(node.left)

            # 递归去右子树收集目标点
            right = dfs(node.right)

            # 情况2.1：左右子树都收集到了节点
            # 说明 p 和 q 分别在当前节点两侧，当前节点就是 LCA
            if left is not None and right is not None:
                return node

            # 情况2.2：只有一边收集到了节点
            # 可能是 p / q，也可能是已经找到的 LCA，继续向上返回
            if left is not None:
                return left
            return right

        return dfs(root)

var lowestCommonAncestor = function(root, p, q) {
    // dfs(node): 在以 node 为根的子树中收集 p / q / LCA
    const dfs = function(node) {
        // 情况3：空节点，说明当前子树没有目标点
        if (node === null) {
            return null;
        }

        // 情况1：当前节点就是 p 或 q，直接把当前节点向上返回
        if (node === p || node === q) {
            return node;
        }

        // 递归去左子树收集目标点
        let left = dfs(node.left);

        // 递归去右子树收集目标点
        let right = dfs(node.right);

        // 情况2.1：左右子树都收集到了节点
        // 说明 p 和 q 分别在当前节点两侧，当前节点就是 LCA
        if (left !== null && right !== null) {
            return node;
        }

        // 情况2.2：只有一边收集到了节点
        // 可能是 p / q，也可能是已经找到的 LCA，继续向上返回
        if (left !== null) {
            return left;
        }
        return right;
    };

    return dfs(root);
};
```

```c
// dfs(node): 在以 node 为根的子树中收集 p / q / LCA
struct TreeNode* dfs(struct TreeNode* node, struct TreeNode* p, struct TreeNode* q) {
    // 情况3：空节点，说明当前子树没有目标点
    if (node == NULL) {
        return NULL;
    }

    // 情况1：当前节点就是 p 或 q，直接把当前节点向上返回
    if (node == p || node == q) {
        return node;
    }

    // 递归去左子树收集目标点
    struct TreeNode* left = dfs(node->left, p, q);

    // 递归去右子树收集目标点
    struct TreeNode* right = dfs(node->right, p, q);

    // 情况2.1：左右子树都收集到了节点
    // 说明 p 和 q 分别在当前节点两侧，当前节点就是 LCA
    if (left != NULL && right != NULL) {
        return node;
    }

    // 情况2.2：只有一边收集到了节点
    // 可能是 p / q，也可能是已经找到的 LCA，继续向上返回
    if (left != NULL) {
        return left;
    }
    return right;
}

struct TreeNode* lowestCommonAncestor(struct TreeNode* root, struct TreeNode* p, struct TreeNode* q) {
    return dfs(root, p, q);
}
```

## 复杂度分析

- 时间复杂度：`O(n)`  
  每个节点最多访问一次。

- 空间复杂度：`O(n)`  
  递归栈最坏 `O(n)`（树退化成链），建树队列和映射表也是 `O(n)`。

## 面试问答

**1. 本题的思路是什么？**

后序遍历这颗树：定义递归函数返回“当前子树里命中的节点”，可能是 `p`、`q` 或它们的最近公共祖先。递归时先看当前节点，如果本身是 `p` 或 `q`，直接返回当前节点；再递归左右子树，如果左右都非空，说明 `p` 和 `q` 分别在两边，根据后序遍历的特性，当前节点就是 LCA；如果只有一边非空，就把那一边继续向上返回；。这样每个节点只访问一次，时间复杂度 `O(n)`，空间复杂度主要是递归栈，最坏 `O(n)`。
