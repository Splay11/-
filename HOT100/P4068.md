本题本质上就是给定一张**有向图**，判断是否存在一个合法的拓扑序列。

在基础知识部分说过了，**图中如果存在环，则不存在合法的拓扑序列。**

但我们在代码实现的过程中无需特别判环，只需要统计已经处理的课程个数，如果其大小不等于$numCourses$ , 则代表图中有环无法被消解，则输出$false$ , 否则输出$true$

**特别注意**，在算法讲解的时候，我们的依赖关系是正的，每次根据后驱寻找前驱。**但是代码实现过程中，通过邻接表找前驱并不好实现，真正好实现的是通过前驱找后驱，所以将关系反着来建边**，这样才能更方便的通过邻接表寻找那些依赖的关系。

```python
class Solution:
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        # edges[x] 存放以 x 为先修课的所有后续课程
        edges = [[] for _ in range(numCourses)]

        # in_degree[i] 表示学习课程 i 之前，还需要完成多少门先修课
        in_degree = [0] * numCourses

        # 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
        for a, b in prerequisites:
            edges[b].append(a)
            in_degree[a] += 1

        # 所有入度为 0 的课程，都可以最开始学习
        q = []
        for i in range(numCourses):
            if in_degree[i] == 0:
                q.append(i)

        # count 记录已经可以完成学习的课程数量
        count = 0

        # 拓扑排序，不断消除入度为 0 的课程
        while q:
            u = q.pop()
            count += 1

            # 学完课程 u 后，它的后续课程少一个先修依赖
            for v in edges[u]:
                in_degree[v] -= 1
                if in_degree[v] == 0:
                    q.append(v)

        # 如果所有课程都被处理，说明不存在环，可以完成所有课程
        return count == numCourses
```

```cpp
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        // edges[x] 存放以 x 为先修课的所有后续课程
        vector<vector<int>> edges(numCourses);

        // inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
        vector<int> inDegree(numCourses, 0);

        // 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
        for (auto& p : prerequisites) {
            int a = p[0];
            int b = p[1];
            edges[b].push_back(a);
            inDegree[a]++;
        }

        // 所有入度为 0 的课程，都可以最开始学习
        vector<int> q;
        for (int i = 0; i < numCourses; i++) {
            if (inDegree[i] == 0) {
                q.push_back(i);
            }
        }

        // count 记录已经可以完成学习的课程数量
        int count = 0;

        // 拓扑排序，不断消除入度为 0 的课程
        while (!q.empty()) {
            int u = q.back();
            q.pop_back();
            count++;

            // 学完课程 u 后，它的后续课程少一个先修依赖
            for (int v : edges[u]) {
                inDegree[v]--;
                if (inDegree[v] == 0) {
                    q.push_back(v);
                }
            }
        }

        // 如果所有课程都被处理，说明不存在环，可以完成所有课程
        return count == numCourses;
    }
};
```

```java
class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        // edges[x] 存放以 x 为先修课的所有后续课程
        List<List<Integer>> edges = new ArrayList<>();
        for (int i = 0; i < numCourses; i++) {
            edges.add(new ArrayList<>());
        }

        // inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
        int[] inDegree = new int[numCourses];

        // 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
        for (int[] p : prerequisites) {
            int a = p[0];
            int b = p[1];
            edges.get(b).add(a);
            inDegree[a]++;
        }

        // 所有入度为 0 的课程，都可以最开始学习
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < numCourses; i++) {
            if (inDegree[i] == 0) {
                queue.offer(i);
            }
        }

        // count 记录已经可以完成学习的课程数量
        int count = 0;

        // 拓扑排序，不断消除入度为 0 的课程
        while (!queue.isEmpty()) {
            int u = queue.poll();
            count++;

            // 学完课程 u 后，它的后续课程少一个先修依赖
            for (int v : edges.get(u)) {
                inDegree[v]--;
                if (inDegree[v] == 0) {
                    queue.offer(v);
                }
            }
        }

        // 如果所有课程都被处理，说明不存在环，可以完成所有课程
        return count == numCourses;
    }
}
```

```go
func canFinish(numCourses int, prerequisites [][]int) bool {
	// edges[x] 存放以 x 为先修课的所有后续课程
	edges := make([][]int, numCourses)

	// inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
	inDegree := make([]int, numCourses)

	// 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
	for _, p := range prerequisites {
		a, b := p[0], p[1]
		edges[b] = append(edges[b], a)
		inDegree[a]++
	}

	// 所有入度为 0 的课程，都可以最开始学习
	q := make([]int, 0)
	for i := 0; i < numCourses; i++ {
		if inDegree[i] == 0 {
			q = append(q, i)
		}
	}

	// count 记录已经可以完成学习的课程数量
	count := 0
	head := 0

	// 拓扑排序，不断消除入度为 0 的课程
	for head < len(q) {
		u := q[head]
		head++
		count++

		// 学完课程 u 后，它的后续课程少一个先修依赖
		for _, v := range edges[u] {
			inDegree[v]--
			if inDegree[v] == 0 {
				q = append(q, v)
			}
		}
	}

	// 如果所有课程都被处理，说明不存在环，可以完成所有课程
	return count == numCourses
}
```

```c
bool canFinish(int numCourses, int** prerequisites, int prerequisitesSize, int* prerequisitesColSize) {
    // head[x] 表示以 x 为先修课的第一条边
    int* head = (int*)malloc((size_t)numCourses * sizeof(int));

    // to[i] 表示第 i 条边指向的后续课程，next[i] 表示同一起点的下一条边
    int* to = (int*)malloc((size_t)prerequisitesSize * sizeof(int));
    int* next = (int*)malloc((size_t)prerequisitesSize * sizeof(int));

    // inDegree[i] 表示学习课程 i 之前，还需要完成多少门先修课
    int* inDegree = (int*)malloc((size_t)numCourses * sizeof(int));

    // 队列中存放所有当前入度为 0 的课程
    int* q = (int*)malloc((size_t)numCourses * sizeof(int));

    for (int i = 0; i < numCourses; i++) {
        head[i] = -1;
        inDegree[i] = 0;
    }

    // 关系 [a, b] 表示学 a 之前要先学 b，所以建边 b -> a
    for (int i = 0; i < prerequisitesSize; i++) {
        int a = prerequisites[i][0];
        int b = prerequisites[i][1];

        to[i] = a;
        next[i] = head[b];
        head[b] = i;
        inDegree[a]++;
    }

    // 所有入度为 0 的课程，都可以最开始学习
    int front = 0;
    int rear = 0;
    for (int i = 0; i < numCourses; i++) {
        if (inDegree[i] == 0) {
            q[rear++] = i;
        }
    }

    // count 记录已经可以完成学习的课程数量
    int count = 0;

    // 拓扑排序，不断消除入度为 0 的课程
    while (front < rear) {
        int u = q[front++];
        count++;

        // 学完课程 u 后，它的后续课程少一个先修依赖
        for (int e = head[u]; e != -1; e = next[e]) {
            int v = to[e];
            inDegree[v]--;

            if (inDegree[v] == 0) {
                q[rear++] = v;
            }
        }
    }

    bool ans = count == numCourses;

    free(head);
    free(to);
    free(next);
    free(inDegree);
    free(q);

    // 如果所有课程都被处理，说明不存在环，可以完成所有课程
    return ans;
}
```
