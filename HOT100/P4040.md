**思路:转置 + 水平翻转**

考虑一个坐标`(i , j)` , 旋转90度之后变成`(j,n-1-i)`

因为旋转后：第一行会到最后一列，第二行会到倒数第二列，以此类推。

所以旋转后的列号是`n-1 - 原来的行号`

旋转后：第一列会到第一行，第二列会到第二行，以此类推。

所以旋转后的行号是`原先的列号`

现在我们需要通过一些原地操作，让`(i,j)` 变成 `(j , n - 1 -  i)`

分成两步：**转置**（行号和列号互换） :`(i,j) -> (j,i)`

接着，想让列号`i` 变成行号`n-1-i` ，本质上就是每个元素跑到它那一行对称的位置，而行号不变。

所以**水平翻转**(行号不变，列号对称翻转):`(j,n-1-i)`

```python id="2czvzu"
class Solution:
    def rotate(self, matrix: List[List[int]]) -> None:
        n = len(matrix)

        # 第一步：转置矩阵
        # 将 matrix[i][j] 和 matrix[j][i] 交换
        for i in range(n):
            for j in range(i + 1, n):
                matrix[i][j], matrix[j][i] = matrix[j][i], matrix[i][j]

        # 第二步：水平翻转每一行
        # 转置后再水平翻转，就等价于顺时针旋转 90 度
        for i in range(n):
            matrix[i].reverse()
```

```cpp id="ct3xek"
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int n = matrix.size();

        // 第一步：转置矩阵
        // 将 matrix[i][j] 和 matrix[j][i] 交换
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                swap(matrix[i][j], matrix[j][i]);
            }
        }

        // 第二步：水平翻转每一行
        // 转置后再水平翻转，就等价于顺时针旋转 90 度
        for (int i = 0; i < n; i++) {
            reverse(matrix[i].begin(), matrix[i].end());
        }
    }
};
```

```java id="l7lk0w"
class Solution {
    public void rotate(int[][] matrix) {
        int n = matrix.length;

        // 第一步：转置矩阵
        // 将 matrix[i][j] 和 matrix[j][i] 交换
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int temp = matrix[i][j];
                matrix[i][j] = matrix[j][i];
                matrix[j][i] = temp;
            }
        }

        // 第二步：水平翻转每一行
        // 转置后再水平翻转，就等价于顺时针旋转 90 度
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n / 2; j++) {
                int temp = matrix[i][j];
                matrix[i][j] = matrix[i][n - 1 - j];
                matrix[i][n - 1 - j] = temp;
            }
        }
    }
}
```

```go id="7ja05z"
func rotate(matrix [][]int) {
    n := len(matrix)

    // 第一步：转置矩阵
    // 将 matrix[i][j] 和 matrix[j][i] 交换
    for i := 0; i < n; i++ {
        for j := i + 1; j < n; j++ {
            matrix[i][j], matrix[j][i] = matrix[j][i], matrix[i][j]
        }
    }

    // 第二步：水平翻转每一行
    // 转置后再水平翻转，就等价于顺时针旋转 90 度
    for i := 0; i < n; i++ {
        left, right := 0, n - 1
        for left < right {
            matrix[i][left], matrix[i][right] = matrix[i][right], matrix[i][left]
            left++
            right--
        }
    }
}
```

```c 
void rotate(int** matrix, int matrixSize, int* matrixColSize) {
    int n = matrixSize;

    // 第一步：转置矩阵
    // 将 matrix[i][j] 和 matrix[j][i] 交换
    for (int i = 0; i < n; i++) {

            int temp = matrix[i][j];
            matrix[i][j] = matrix[j][i];
            matrix[j][i] = temp;
        }
    }

    // 第二步：水平翻转每一行
    // 转置后再水平翻转，就等价于顺时针旋转 90 度
    for (int i = 0; i < n; i++) {
        int left = 0;
        int right = n - 1;

        while (left < right) {
            int temp = matrix[i][left];
            matrix[i][left] = matrix[i][right];
            matrix[i][right] = temp;

            left++;
            right--;
        }
    }
}
```

### **面试问答**

**1.这道题的整体思路是什么？**

顺时针旋转90度，等价于先对矩阵转置再水平翻转。时间复杂度为$O(nm)$ , 因为转置和水平翻转都是对矩阵每个元素操作一次，共操作$2nm$次。 空间复杂度为$O(1)$  ,没有使用额外的数组。
