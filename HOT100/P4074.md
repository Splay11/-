**思路**

回文串的判定：判断`str` 和 `reversed(str)` 是否全等。

本题所有可能情况是$2^{n-1}$ 。 **考虑分割的本质是：在两个相邻的字符中间插入一个隔板 。**   

一个长度为$n$的字符串有$n-1$个相邻位置可以插入或者不插入隔板。

例如:`aab`  可以分割成 `aab` , `aa|b` ,`a|ab` ,`a|a|b`  共$2^{3-1} = 2^2 = 4$ 种情况。

所以不难发现，这个题的递归，每一层枚举的是某个隔板。`index` 指向的是隔板的位置，而不是某个字符的位置。

而由于我们要判断某一段子字符串是不是回文串，**而子串由当前的隔板 + 上一个隔板唯一确定。**

所以我们递归的时候除了`index` ， 还需要记录上一个隔板的位置`last_index` ， 每次决定分割的时候，判断子串是否为回文串，如果是回文串则放入`curr`里。

```python
class Solution:
    def partition(self, s: str) -> List[List[str]]:
        # result 用来存储所有合法的分割方案
        result = []
        # curr 表示当前已经切出来的回文子串
        curr = []
        n = len(s)

        # 判断一个字符串是否是回文串
        def is_palindrome(sub):
            return sub == sub[::-1]

        def dfs(index, last_index):
            # index 走到 n，说明所有隔板都已经考虑完，得到一个完整方案
            if index == n:
                result.append(curr[:])
                return

            # 当前子串由上一个隔板 last_index 和当前隔板 index 唯一确定
            sub = s[last_index + 1:index + 1]

            # 如果当前子串是回文串，才可以在这里放隔板
            if is_palindrome(sub):
                curr.append(sub)
                dfs(index + 1, index)
                curr.pop()

            # 如果不是最后一个位置，可以选择当前不放隔板，继续往后看
            if index != n - 1:
                dfs(index + 1, last_index)

        dfs(0, -1)
        return result
```

```cpp
class Solution {
public:
    vector<vector<string>> result;
    vector<string> curr;
    string str;
    int n;

    bool isPalindrome(string sub) {
        int left = 0;
        int right = sub.size() - 1;

        // 判断 sub 和它反转后是否一致，本质就是左右指针比较
        while (left < right) {
            if (sub[left] != sub[right]) {
                return false;
            }
            left++;
            right--;
        }

        return true;
    }

    void dfs(int index, int lastIndex) {
        // index 走到 n，说明所有隔板都已经考虑完，得到一个完整方案
        if (index == n) {
            result.push_back(curr);
            return;
        }

        // 当前子串由上一个隔板 lastIndex 和当前隔板 index 唯一确定
        string sub = str.substr(lastIndex + 1, index - lastIndex);

        // 如果当前子串是回文串，才可以在这里放隔板
        if (isPalindrome(sub)) {
            curr.push_back(sub);
            dfs(index + 1, index);
            curr.pop_back();
        }

        // 如果不是最后一个位置，可以选择当前不放隔板，继续往后看
        if (index != n - 1) {
            dfs(index + 1, lastIndex);
        }
    }

    vector<vector<string>> partition(string s) {
        result.clear();
        curr.clear();
        str = s;
        n = s.size();

        dfs(0, -1);
        return result;
    }
};
```

```java
    // result 用来存储所有合法的分割方案
    private List<List<String>> result = new ArrayList<>();
    // curr 表示当前已经切出来的回文子串
    private List<String> curr = new ArrayList<>();
    private String str;
    private int n;

    public List<List<String>> partition(String s) {
        str = s;
        n = s.length();

        dfs(0, -1);
        return result;
    }

    private boolean isPalindrome(String sub) {
        int left = 0;
        int right = sub.length() - 1;

        // 判断 sub 和它反转后是否一致，本质就是左右指针比较
        while (left < right) {
            if (sub.charAt(left) != sub.charAt(right)) {
                return false;
            }
            left++;
            right--;
        }

        return true;
    }

    private void dfs(int index, int lastIndex) {
        // index 走到 n，说明所有隔板都已经考虑完，得到一个完整方案
        if (index == n) {
            result.add(new ArrayList<>(curr));
            return;
        }

        // 当前子串由上一个隔板 lastIndex 和当前隔板 index 唯一确定

        if (isPalindrome(sub)) {
            curr.add(sub);
            dfs(index + 1, index);
            curr.remove(curr.size() - 1);
```

```go
func partition(s string) [][]string {
	// result 用来存储所有合法的分割方案
	result := [][]string{}
	// curr 表示当前已经切出来的回文子串
	curr := []string{}
	n := len(s)

	// 判断一个字符串是否是回文串
	isPalindrome := func(sub string) bool {
		left := 0
		right := len(sub) - 1

		// 判断 sub 和它反转后是否一致，本质就是左右指针比较
		for left < right {
			if sub[left] != sub[right] {
				return false
			}
			left++
			right--
		}

		return true
	}

	var dfs func(int, int)
	dfs = func(index int, lastIndex int) {
		// index 走到 n，说明所有隔板都已经考虑完，得到一个完整方案
		if index == n {
			tmp := make([]string, len(curr))
			copy(tmp, curr)
			result = append(result, tmp)
			return
		}

		// 当前子串由上一个隔板 lastIndex 和当前隔板 index 唯一确定
		sub := s[lastIndex+1 : index+1]

		// 如果当前子串是回文串，才可以在这里放隔板
		if isPalindrome(sub) {
			curr = append(curr, sub)
			dfs(index+1, index)
			curr = curr[:len(curr)-1]
		}

		// 如果不是最后一个位置，可以选择当前不放隔板，继续往后看
		if index != n-1 {
			dfs(index+1, lastIndex)
		}
	}

	dfs(0, -1)
	return result
}
```

```javascript
                return false;

            left++;
            right--;
        }

        return true;

            return;

        const sub = s.slice(lastIndex + 1, index + 1);
```

```c
bool isPalindrome(char* s, int left, int right) {
    // 判断子串和它反转后是否一致，本质就是左右指针比较
    while (left < right) {
        if (s[left] != s[right]) {
            return false;
        }
        left++;
        right--;
    }

    return true;
}

void dfs(char* s, int n, int index, int lastIndex, char** curr, int currSize, char**** result, int* returnSize, int** returnColumnSizes) {
    // index 走到 n，说明所有隔板都已经考虑完，得到一个完整方案
    if (index == n) {
        (*result)[*returnSize] = (char**)malloc(sizeof(char*) * currSize);
        (*returnColumnSizes)[*returnSize] = currSize;

        for (int i = 0; i < currSize; i++) {
            int len = strlen(curr[i]);
            (*result)[*returnSize][i] = (char*)malloc(sizeof(char) * (len + 1));
            strcpy((*result)[*returnSize][i], curr[i]);
        }

        (*returnSize)++;
        return;
    }

    // 当前子串由上一个隔板 lastIndex 和当前隔板 index 唯一确定
    int left = lastIndex + 1;
    int len = index - lastIndex;

    // 如果当前子串是回文串，才可以在这里放隔板
    if (isPalindrome(s, left, index)) {
        curr[currSize] = (char*)malloc(sizeof(char) * (len + 1));

        for (int i = 0; i < len; i++) {
            curr[currSize][i] = s[left + i];
        }
        curr[currSize][len] = '\0';

        dfs(s, n, index + 1, index, curr, currSize + 1, result, returnSize, returnColumnSizes);

        free(curr[currSize]);
    }

    // 如果不是最后一个位置，可以选择当前不放隔板，继续往后看
    if (index != n - 1) {
        dfs(s, n, index + 1, lastIndex, curr, currSize, result, returnSize, returnColumnSizes);
    }
}

char*** partition(char* s, int* returnSize, int** returnColumnSizes) {
    int n = strlen(s);

    // result 用来存储所有合法的分割方案
    char*** result = (char***)malloc(sizeof(char**) * (1 << (n - 1)));
    *returnColumnSizes = (int*)malloc(sizeof(int) * (1 << (n - 1)));
    *returnSize = 0;

    // curr 表示当前已经切出来的回文子串
    char** curr = (char**)malloc(sizeof(char*) * n);

    dfs(s, n, 0, -1, curr, 0, &result, returnSize, returnColumnSizes);

    free(curr);
    return result;
}
```

注意：在代码层面，我们实际插入了$n$个隔板（第$n$个隔板放在整个字符串最后），但最后一个隔板一定会放，这样保证了不遗漏最后一段子串

### 面试问答

**1. 本题的核心思路是什么？**

回溯枚举所有分割方案。枚举 `n-1` 个可以切割的位置，每个位置考虑切或者不切来得到所有可能的分割方案。递归时用 `index` 表示当前考虑到哪个隔板位置，用 `last_index` 记录上一个隔板所在的位置，每当决定在当前位置分割时，就取出这段子串，若是回文串则把它加入当前方案并继续向下递归，最终找出所有由回文子串组成的分割方案。

时间复杂度:总共可能有$2^n$种分割方案，而递归时每次分割之前都需要判断子串是否为回文串，这部分开销为$O(n)$ ， 分步相乘得到$O(n * 2^n)$。

空间复杂度:$O(n)$。其中`curr`数组开销为$O(n)$，递归栈的空间开销为$O(n)$。
