**思路：栈**

我们先来观察一下，这种字符串到底是怎么被解码的。

比如：

```
3[a]
```

它表示把 `a` 重复 `3` 次，所以结果就是：

```
aaa
```

这个还比较直接。

但是如果变成：

```
3[a2[c]]
```

就没有这么简单了。

**我们用一张图来为大家拆解一下这个字符串的解码过程：**

![](/file/2/54PY7cUTx1q_XnBWc_4Mo.png)


从图出我们可以看出，这题有一个非常明显的特点：

**最里面的括号要先处理，外面的括号后处理**

换句话说：**后遇到的一层括号，反而要先处理**这个特点恰好和栈的**后进先出**非常吻合，所以这题我们可以考虑用**栈**来解决。

那接下来问题就来了：

**既然我们准备用栈来做，那栈里到底放什么？**

一个很直接的想法就是：

我们从左到右遍历字符串，先把沿途看到的内容都放进栈里。

**那什么时候把栈里面的内容取出来进行解码呢？**

当我们遍历字符串，**遇到 `]` 的时候**，就说明当前这一对括号里的内容已经完整出现了，这时就可以把栈里的内容取出来进行解码。

**我们以`3[a]2[b2[cd]]`这个为例，来看一段动画：**

@[video](https://codefun2000.com/p/4068/file/%E5%AD%97%E7%AC%A6%E4%B8%B2%E8%A7%A3%E7%A0%81.mp4)

从动画我们可以总结出规律：

遇到非`]`的字符直接放入栈中，遇到`]`时，此时栈顶一定是`xxx[yyy`的结构，其中`xxx`是数字段，`yyy`是字符段，我们可以把他们依次拿出栈中处理，解码之后再放入栈中，这样就完成了一对括号中的字符串的解码，重复这个过程最后栈里的字符串就是答案

```python
class Solution:
    def decodeString(self, s: str) -> str:
        # 使用栈模拟解码过程
        stack = []

        # 遍历字符串中的每个字符
        for c in s:
            # 遇到非右括号字符，直接入栈
            if c != "]":
                stack.append(c)
            else:
                # 1. 取出括号中的字符串
                now_str = ""
                while stack[-1] != "[":
                    now_str += stack[-1]
                    stack.pop()

                # 弹出左括号
                stack.pop()

                # 因为是从栈顶取出的，所以需要反转
                now_str = now_str[::-1]

                # 2. 取出左括号前面的数字
                num_str = ""
                while stack and stack[-1].isdigit():
                    num_str += stack[-1]
                    stack.pop()

                # 数字也是倒着取出的，所以需要反转
                num_str = num_str[::-1]
                num = int(num_str)

                # 3. 将解码后的字符串重新压入栈中
                for _ in range(num):
                    for ch in now_str:
                        stack.append(ch)

        # 栈中剩余字符拼接成最终结果
        return "".join(stack)
```

```cpp 
class Solution {
public:
    string decodeString(string s) {
        // 使用栈模拟解码过程
        vector<char> stack;

        // 遍历字符串中的每个字符
        for (char c : s) {
            // 遇到非右括号字符，直接入栈
            if (c != ']') {
                stack.push_back(c);
            } else {
                // 1. 取出括号中的字符串
                string now_str = "";
                while (stack.back() != '[') {
                    now_str += stack.back();
                    stack.pop_back();
                }

                // 弹出左括号
                stack.pop_back();

                // 因为是从栈顶取出的，所以需要反转
                reverse(now_str.begin(), now_str.end());

                // 2. 取出左括号前面的数字
                string num_str = "";
                while (!stack.empty() && isdigit(stack.back())) {
                    num_str += stack.back();
                    stack.pop_back();
                }

                // 数字也是倒着取出的，所以需要反转
                reverse(num_str.begin(), num_str.end());
                int num = stoi(num_str);

                // 3. 将解码后的字符串重新压入栈中
                for (int i = 0; i < num; i++) {
                    for (char ch : now_str) {
                        stack.push_back(ch);
                    }
                }
            }
        }

        // 栈中剩余字符拼接成最终结果
        string result = "";
        for (char c : stack) {
            result += c;
        }

        return result;
    }
};
```

```java
class Solution {
    public String decodeString(String s) {
        // 使用栈模拟解码过程
        List<Character> stack = new ArrayList<>();

        // 遍历字符串中的每个字符
        for (char c : s.toCharArray()) {
            // 遇到非右括号字符，直接入栈
            if (c != ']') {
                stack.add(c);
            } else {
                // 1. 取出括号中的字符串
                String nowStr = "";
                while (stack.get(stack.size() - 1) != '[') {
                    nowStr += stack.get(stack.size() - 1);
                    stack.remove(stack.size() - 1);
                }

                // 弹出左括号
                stack.remove(stack.size() - 1);

                // 因为是从栈顶取出的，所以需要反转
                nowStr = new StringBuilder(nowStr).reverse().toString();

                // 2. 取出左括号前面的数字
                String numStr = "";
                while (!stack.isEmpty() && Character.isDigit(stack.get(stack.size() - 1))) {
                    numStr += stack.get(stack.size() - 1);
                    stack.remove(stack.size() - 1);
                }

                // 数字也是倒着取出的，所以需要反转
                numStr = new StringBuilder(numStr).reverse().toString();
                int num = Integer.parseInt(numStr);

                // 3. 将解码后的字符串重新压入栈中
                for (int i = 0; i < num; i++) {
                    for (char ch : nowStr.toCharArray()) {
                        stack.add(ch);
                    }
                }
            }
        }

        // 栈中剩余字符拼接成最终结果
        StringBuilder result = new StringBuilder();
        for (char c : stack) {
            result.append(c);
        }

        return result.toString();
    }
}
```

```javascript
var decodeString = function(s) {
    // 使用栈模拟解码过程
    const stack = [];

    // 遍历字符串中的每个字符
    for (const c of s) {
        // 遇到非右括号字符，直接入栈
        if (c !== "]") {
            stack.push(c);
        } else {
            // 1. 取出括号中的字符串
            let nowStr = "";
            while (stack[stack.length - 1] !== "[") {
                nowStr += stack.pop();
            }

            // 弹出左括号
            stack.pop();

            // 因为是从栈顶取出的，所以需要反转
            nowStr = nowStr.split("").reverse().join("");

            // 2. 取出左括号前面的数字
            let numStr = "";
            while (stack.length > 0 && stack[stack.length - 1] >= "0" && stack[stack.length - 1] <= "9") {
                numStr += stack.pop();
            }

            // 数字也是倒着取出的，所以需要反转
            numStr = numStr.split("").reverse().join("");
            const num = Number(numStr);

            // 3. 将解码后的字符串重新压入栈中
            for (let i = 0; i < num; i++) {
                for (const ch of nowStr) {
                    stack.push(ch);
                }
            }
        }
    }

    // 栈中剩余字符拼接成最终结果
    return stack.join("");
};
```

```go 
func decodeString(s string) string {
    // 使用栈模拟解码过程
    stack := []byte{}

    // 遍历字符串中的每个字符
    for i := 0; i < len(s); i++ {
        c := s[i]

        // 遇到非右括号字符，直接入栈
        if c != ']' {
            stack = append(stack, c)
        } else {
            // 1. 取出括号中的字符串
            nowStr := []byte{}
            for stack[len(stack)-1] != '[' {
                nowStr = append(nowStr, stack[len(stack)-1])
                stack = stack[:len(stack)-1]
            }

            // 弹出左括号
            stack = stack[:len(stack)-1]

            // 因为是从栈顶取出的，所以需要反转
            for l, r := 0, len(nowStr)-1; l < r; l, r = l+1, r-1 {
                nowStr[l], nowStr[r] = nowStr[r], nowStr[l]
            }

            // 2. 取出左括号前面的数字
            numStr := []byte{}
            for len(stack) > 0 && stack[len(stack)-1] >= '0' && stack[len(stack)-1] <= '9' {
                numStr = append(numStr, stack[len(stack)-1])
                stack = stack[:len(stack)-1]
            }

            // 数字也是倒着取出的，所以需要反转并转成整数
            num := 0
            for j := len(numStr) - 1; j >= 0; j-- {
                num = num*10 + int(numStr[j]-'0')
            }

            // 3. 将解码后的字符串重新压入栈中
            for k := 0; k < num; k++ {
                for _, ch := range nowStr {
                    stack = append(stack, ch)
                }
            }
        }
    }

    // 栈中剩余字符拼接成最终结果
    return string(stack)
}
```

```c 
static void ensureCapacity(char** arr, int* capacity, int need) {
    // 如果当前容量不够，就不断扩容
    while (*capacity <= need) {
        *capacity *= 2;
        *arr = (char*)realloc(*arr, sizeof(char) * (*capacity));
    }
}

char* decodeString(char* s) {
    int n = strlen(s);

    // 使用字符数组模拟栈
    int capacity = n + 10;
    char* stack = (char*)malloc(sizeof(char) * capacity);
    int top = 0;

    // 遍历字符串中的每个字符
    for (int i = 0; i < n; i++) {
        char c = s[i];

        // 遇到非右括号字符，直接入栈
        if (c != ']') {
            ensureCapacity(&stack, &capacity, top + 1);
            stack[top++] = c;
        } else {
            // 1. 取出括号中的字符串
            int strCap = 16;
            char* nowStr = (char*)malloc(sizeof(char) * strCap);
            int strLen = 0;

            while (top > 0 && stack[top - 1] != '[') {
                if (strLen + 1 >= strCap) {
                    strCap *= 2;
                    nowStr = (char*)realloc(nowStr, sizeof(char) * strCap);
                }
                nowStr[strLen++] = stack[--top];
            }

            // 弹出左括号
            top--;

            // 因为是从栈顶取出的，所以需要反转
            for (int l = 0, r = strLen - 1; l < r; l++, r--) {
                char temp = nowStr[l];
                nowStr[l] = nowStr[r];
                nowStr[r] = temp;
            }

            // 2. 取出左括号前面的数字
            int num = 0;
            int base = 1;
            while (top > 0 && isdigit(stack[top - 1])) {
                num += (stack[--top] - '0') * base;
                base *= 10;
            }

            // 3. 将解码后的字符串重新压入栈中
            int need = top + num * strLen + 1;
            ensureCapacity(&stack, &capacity, need);

            for (int k = 0; k < num; k++) {
                for (int j = 0; j < strLen; j++) {
                    stack[top++] = nowStr[j];
                }
            }

            free(nowStr);
        }
    }

    // 栈中剩余字符拼接成最终结果
    ensureCapacity(&stack, &capacity, top + 1);
    stack[top] = '\0';

    return stack;
}
```

### 面试问答

**1. 本题的核心思路是什么？**

字符串的解码过程具有明显的“嵌套”结构，最里面的括号一定先被解码，外层括号后解码，这正好符合栈的后进先出特点。因此我们可以用栈来模拟这个过程：每次遇到右括号则解码完整字符串再放入。

**时间复杂度是 O(m)**，其中 `m` 是最终字符串长度。因为我们从左到右遍历字符串，每个字符都会被处理一次，数字解析也是在线性遍历过程中完成的，所以整体时间复杂度是 `O(n)`。

**空间复杂度是 O(m)**
