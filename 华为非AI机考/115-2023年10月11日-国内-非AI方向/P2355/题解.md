## 题解思路

地址空间是 **32G = 2^35 字节**，但题目保证**实际存储总量不超过 16MB**，所以不能真的开 32G 数组，应该用 **哈希表** 只存被写过的字节。

设：

* `mem[address] = byte`
* 没写过的地址默认是 `00`

### 三种操作

#### 1. Write Address Length Data

把 `Data` 按字节拆开，依次写到：

* `Address`
* `Address + 1`
* ...
* `Address + Length - 1`

规则：

* 如果 `Data` 不足 `Length` 字节，末尾补 `00`
* 如果 `Data` 超过 `Length` 字节，末尾截断

#### 2. Read Address Length

从 `Address` 开始连续读取 `Length` 个字节：

* 写过就输出对应字节
* 没写过就输出 `00`

#### 3. Clear

清空哈希表即可。

## 合法性判断

内存范围是 `[0, 2^35 - 1]`，所以读写区间必须满足：

* `Address < 2^35`
* `Address + Length <= 2^35`

否则该指令无效。

* `Write` 非法：忽略
* `Read` 非法：输出空行
* `Clear` 无参数，始终合法


## 复杂度

若本次操作长度为 `L`：

* `Write`: `O(L)`
* `Read`: `O(L)`
* `Clear`: `O(K)`，`K` 为当前已存字节数

总数据不超过 16MB，可以通过。

## 代码实现

### Python 代码

```python
import sys

MEM_SIZE = 1 << 35

def valid(addr, length):
    return 0 <= addr < MEM_SIZE and 0 <= length <= MEM_SIZE - addr

def parse_hex_byte(s):
    return int(s, 16)

def solve():
    input = sys.stdin.readline
    n = int(input().strip())
    mem = {}

    for _ in range(n):
        parts = input().split()
        cmd = parts[0]

        if cmd == "Clear":
            mem.clear()

        elif cmd == "Write":
            addr = int(parts[1], 16)
            length = int(parts[2])
            data = parts[3]

            if not valid(addr, length):
                continue

            data_bytes = len(data) // 2
            for i in range(length):
                if i < data_bytes:
                    b = parse_hex_byte(data[2 * i: 2 * i + 2])
                else:
                    b = 0
                mem[addr + i] = b

        elif cmd == "Read":
            addr = int(parts[1], 16)
            length = int(parts[2])

            if not valid(addr, length):
                print()
                continue

            ans = []
            for i in range(length):
                ans.append(f"{mem.get(addr + i, 0):02X}")
            print("".join(ans))

if __name__ == "__main__":
    solve()
```


### C++ 代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using ull = unsigned long long;
const ull MEM_SIZE = 1ULL << 35;

bool valid(ull addr, ull len) {
    return addr < MEM_SIZE && len <= MEM_SIZE - addr;
}

unsigned char hexByte(const string &s, int pos) {
    auto val = [](char c) -> int {
        if ('0' <= c && c <= '9') return c - '0';
        return c - 'A' + 10;
    };
    return (unsigned char)(val(s[pos]) * 16 + val(s[pos + 1]));
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    unordered_map<ull, unsigned char> mem;

    while (n--) {
        string cmd;
        cin >> cmd;

        if (cmd == "Clear") {
            mem.clear();
        } else if (cmd == "Write") {
            string addrStr, data;
            ull len;
            cin >> addrStr >> len >> data;
            ull addr = stoull(addrStr, nullptr, 16);

            if (!valid(addr, len)) continue;

            ull dataBytes = data.size() / 2;
            for (ull i = 0; i < len; i++) {
                unsigned char b = (i < dataBytes ? hexByte(data, 2 * i) : 0);
                mem[addr + i] = b;
            }
        } else if (cmd == "Read") {
            string addrStr;
            ull len;
            cin >> addrStr >> len;
            ull addr = stoull(addrStr, nullptr, 16);

            if (!valid(addr, len)) {
                cout << '\n';
                continue;
            }

            string ans;
            ans.reserve(len * 2);
            const char *HEX = "0123456789ABCDEF";

            for (ull i = 0; i < len; i++) {
                unsigned char b = mem.count(addr + i) ? mem[addr + i] : 0;
                ans.push_back(HEX[b >> 4]);
                ans.push_back(HEX[b & 15]);
            }
            cout << ans << '\n';
        }
    }
    return 0;
}
```

### Java 代码

```java
import java.io.*;
import java.util.*;

public class Main {
    static final long MEM_SIZE = 1L << 35;

    static boolean valid(long addr, long len) {
        return addr >= 0 && addr < MEM_SIZE && len >= 0 && len <= MEM_SIZE - addr;
    }

    static int hexByte(String s, int pos) {
        return Integer.parseInt(s.substring(pos, pos + 2), 16);
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        Map<Long, Integer> mem = new HashMap<>();
        StringBuilder out = new StringBuilder();

        for (int t = 0; t < n; t++) {
            String[] parts = br.readLine().trim().split("\\s+");
            String cmd = parts[0];

            if (cmd.equals("Clear")) {
                mem.clear();
            } else if (cmd.equals("Write")) {
                long addr = Long.parseUnsignedLong(parts[1].substring(2), 16);
                long len = Long.parseLong(parts[2]);
                String data = parts[3];

                if (!valid(addr, len)) continue;

                long dataBytes = data.length() / 2;
                for (long i = 0; i < len; i++) {
                    int b = (i < dataBytes) ? hexByte(data, (int)(2 * i)) : 0;
                    mem.put(addr + i, b);
                }
            } else if (cmd.equals("Read")) {
                long addr = Long.parseUnsignedLong(parts[1].substring(2), 16);
                long len = Long.parseLong(parts[2]);

                if (!valid(addr, len)) {
                    out.append('\n');
                    continue;
                }

                for (long i = 0; i < len; i++) {
                    int b = mem.getOrDefault(addr + i, 0);
                    out.append(String.format("%02X", b));
                }
                out.append('\n');
            }
        }

        System.out.print(out.toString());
    }
}
```
