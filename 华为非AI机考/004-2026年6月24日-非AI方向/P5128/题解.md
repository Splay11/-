## 解题思路

根据前序遍历和中序遍历还原二叉树。由于节点值互不相同，可以用哈希表记录每个节点值在中序遍历中的位置。

还原过程中不一定要真的建树，可以递归处理前序和中序对应区间：

$1$：前序区间第一个值就是当前根节点。

$2$：通过中序位置划分左子树和右子树。

$3$：递归时维护从根到当前节点的路径和，也就是当前节点的镜像完整大小。

$4$：如果当前路径和小于等于 $0$，说明当前节点异常，整棵子树剪枝，后续子节点不再统计。

$5$：否则记录当前路径和，并继续递归左右子树。

$6$：最后将所有合法镜像大小从小到大排序，输出最大的 $K$ 个；如果没有合法节点，输出 $null$。

用到的核心算法是：二叉树遍历还原、递归 $DFS$、剪枝、排序。

## 复杂度分析

设节点数量为 $n$。

时间复杂度为 $O(n \log n)$。
其中递归遍历每个节点最多一次，复杂度为 $O(n)$；最后对合法镜像大小排序，复杂度为 $O(n \log n)$。

空间复杂度为 $O(n)$。
哈希表、递归栈和结果数组都需要额外空间。

## 代码实现

### Python

```python
import sys

def get_top_images(preorder, inorder, k):
    # 建立中序遍历中，节点值到下标的映射
    pos = {}
    for i, v in enumerate(inorder):
        pos[v] = i

    ans = []

    def dfs(pre_l, pre_r, in_l, in_r, parent_sum):
        # 区间为空，直接返回
        if pre_l > pre_r:
            return

        root = preorder[pre_l]
        cur_sum = parent_sum + root

        # 当前节点异常，整棵子树剪枝
        if cur_sum <= 0:
            return

        ans.append(cur_sum)

        mid = pos[root]
        left_size = mid - in_l

        # 递归处理左子树
        dfs(pre_l + 1, pre_l + left_size, in_l, mid - 1, cur_sum)
        # 递归处理右子树
        dfs(pre_l + left_size + 1, pre_r, mid + 1, in_r, cur_sum)

    n = len(preorder)
    dfs(0, n - 1, 0, n - 1, 0)

    if not ans:
        return []

    # 从小到大排序后，截取最大的 k 个
    ans.sort()
    if len(ans) <= k:
        return ans
    return ans[-k:]


def main():
    sys.setrecursionlimit(10000)

    preorder = list(map(int, sys.stdin.readline().split()))
    inorder = list(map(int, sys.stdin.readline().split()))
    k = int(sys.stdin.readline())

    res = get_top_images(preorder, inorder, k)

    if not res:
        print("null")
    else:
        print(" ".join(map(str, res)))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;

public class Main {
    static int[] preorder;
    static int[] inorder;
    static HashMap<Integer, Integer> pos;
    static ArrayList<Integer> ans;

    // 递归处理当前子树，并统计合法镜像完整大小
    static void dfs(int preL, int preR, int inL, int inR, int parentSum) {
        if (preL > preR) {
            return;
        }

        int root = preorder[preL];
        int curSum = parentSum + root;

        // 当前节点异常，整棵子树剪枝
        if (curSum <= 0) {
            return;
        }

        ans.add(curSum);

        int mid = pos.get(root);
        int leftSize = mid - inL;

        // 递归处理左子树
        dfs(preL + 1, preL + leftSize, inL, mid - 1, curSum);
        // 递归处理右子树
        dfs(preL + leftSize + 1, preR, mid + 1, inR, curSum);
    }

    // 返回从小到大的最大 k 个镜像大小
    static ArrayList<Integer> getTopImages(int[] pre, int[] in, int k) {
        preorder = pre;
        inorder = in;
        pos = new HashMap<Integer, Integer>();
        ans = new ArrayList<Integer>();

        for (int i = 0; i < inorder.length; i++) {
            pos.put(inorder[i], i);
        }

        dfs(0, preorder.length - 1, 0, inorder.length - 1, 0);

        Collections.sort(ans);

        ArrayList<Integer> res = new ArrayList<Integer>();
        int start = Math.max(0, ans.size() - k);
        for (int i = start; i < ans.size(); i++) {
            res.add(ans.get(i));
        }
        return res;
    }

    // 将一行数字转为数组
    static int[] parseArray(String line) {
        String[] parts = line.trim().split("\\s+");
        int[] arr = new int[parts.length];
        for (int i = 0; i < parts.length; i++) {
            arr[i] = Integer.parseInt(parts[i]);
        }
        return arr;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int[] pre = parseArray(br.readLine());
        int[] in = parseArray(br.readLine());
        int k = Integer.parseInt(br.readLine().trim());

        ArrayList<Integer> res = getTopImages(pre, in, k);

        if (res.size() == 0) {
            System.out.println("null");
        } else {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < res.size(); i++) {
                if (i > 0) {
                    sb.append(" ");
                }
                sb.append(res.get(i));
            }
            System.out.println(sb.toString());
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <sstream>
#include <unordered_map>
#include <algorithm>
using namespace std;

vector<int> preorder, inorder;
unordered_map<int, int> pos;
vector<int> ans;

// 递归处理当前子树，并统计合法镜像完整大小
void dfs(int preL, int preR, int inL, int inR, int parentSum) {
    if (preL > preR) {
        return;
    }

    int root = preorder[preL];
    int curSum = parentSum + root;

    // 当前节点异常，整棵子树剪枝
    if (curSum <= 0) {
        return;
    }

    ans.push_back(curSum);

    int mid = pos[root];
    int leftSize = mid - inL;

    // 递归处理左子树
    dfs(preL + 1, preL + leftSize, inL, mid - 1, curSum);
    // 递归处理右子树
    dfs(preL + leftSize + 1, preR, mid + 1, inR, curSum);
}

// 返回从小到大的最大 k 个镜像大小
vector<int> getTopImages(vector<int>& pre, vector<int>& in, int k) {
    preorder = pre;
    inorder = in;
    pos.clear();
    ans.clear();

    for (int i = 0; i < (int)inorder.size(); i++) {
        pos[inorder[i]] = i;
    }

    dfs(0, (int)preorder.size() - 1, 0, (int)inorder.size() - 1, 0);

    sort(ans.begin(), ans.end());

    vector<int> res;
    int start = max(0, (int)ans.size() - k);
    for (int i = start; i < (int)ans.size(); i++) {
        res.push_back(ans[i]);
    }

    return res;
}

// 将一行数字转为数组
vector<int> parseArray(string line) {
    vector<int> arr;
    stringstream ss(line);
    int x;
    while (ss >> x) {
        arr.push_back(x);
    }
    return arr;
}

int main() {
    string line1, line2;
    getline(cin, line1);
    getline(cin, line2);

    int k;
    cin >> k;

    vector<int> pre = parseArray(line1);
    vector<int> in = parseArray(line2);

    vector<int> res = getTopImages(pre, in, k);

    if (res.empty()) {
        cout << "null" << endl;
    } else {
        for (int i = 0; i < (int)res.size(); i++) {
            if (i > 0) {
                cout << " ";
            }
            cout << res[i];
        }
        cout << endl;
    }

    return 0;
}
```
