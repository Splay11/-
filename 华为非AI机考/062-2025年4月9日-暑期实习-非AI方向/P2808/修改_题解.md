# 二重循环统计逆序朴素解法

这道题的正解是 $树状数组$，但是我们用朴素解法 $二重循环统计逆序$ 也能在考试时拿到一定的分数。

题意：对 $N\times N$ 矩阵做顺时针螺旋遍历，得到序列 $spiral$。允许交换螺旋顺序上相邻的两个元素，最少次数等于把 $spiral$ 排成 $[1,2,\ldots,N^2]$ 的逆序对数。答案对 $1000000007$ 取模。

朴素做法：

1. 按螺旋顺序把矩阵展开成一维数组 $spiral$。
2. 二重循环统计逆序：对所有 $i<j$，若 $spiral[i]>spiral[j]$，答案加 $1$。
3. 最后对 $MOD=1000000007$ 取模。

小数据可以过；当 $N$ 较大时序列长度 $N^2$ 很大，$O((N^2)^2)$ 会超时，需要后面的树状数组做法。

* 时间复杂度：$O((N^2)^2)$，枚举所有下标对。
* 空间复杂度：$O(N^2)$，存放螺旋序列。

```python
def build_spiral(matrix):
    N = len(matrix)
    spiral = []
    top, bottom = 0, N - 1
    left, right = 0, N - 1

            spiral.append(matrix[i][right])

            spiral.append(matrix[i][left])

    return spiral

    inversions = 0
```

# 正确解法

## 题目分析

目标序列是 $[1,2,\ldots,N^2]$。相邻交换的最少次数等于螺旋顺序上的逆序数。

用树状数组在 $O(N^2\log(N^2))$ 内统计逆序：把矩阵按行优先展平成 $flat$，再按螺旋顺序依次处理每个值 $x$：

1. 树状数组 $tree$ 维护已出现数值的个数；$cnt$ 为已处理个数。
2. 查询已出现且 $\le x$ 的个数 $s$，则前面大于 $x$ 的个数为 $cnt-s$，累加到 $inversions$。
3. 再把 $x$ 更新进树状数组，$cnt$ 加 $1$。
4. 答案对 $MOD=1000000007$ 取模。

## 复杂度分析

* 时间复杂度：$O(N^2\log(N^2))$。
* 空间复杂度：$O(N^2)$。

## 代码实现

### Python

```python
MOD = 1000000007


def main():
    import sys
    # N 可达 10^3，二进制整包读入并展平，避免超时
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    flat = data[1:]  # 行优先展平的矩阵
    m = N * N
    tree = [0] * (m + 1)  # 树状数组
    inversions = 0
    cnt = 0  # 已处理的元素个数

    def add_value(x):
        nonlocal inversions, cnt
        # 查询已出现且 <= x 的个数
        s = 0
        idx = x
        while idx:
            s += tree[idx]
            idx -= idx & -idx
        # 前面大于 x 的个数即逆序贡献
        inversions += cnt - s
        idx = x
        while idx <= m:
            tree[idx] += 1
            idx += idx & -idx
        cnt += 1

    # 螺旋遍历，边走边用树状数组统计逆序
    top = left = 0
    bottom = right = N - 1
    while top <= bottom and left <= right:
        base = top * N
        for j in range(left, right + 1):
            add_value(flat[base + j])
        top += 1
        if top > bottom:
            break
        for i in range(top, bottom + 1):
            add_value(flat[i * N + right])
        right -= 1
        if left > right:
            break
        base = bottom * N
        for j in range(right, left - 1, -1):
            add_value(flat[base + j])
        bottom -= 1
        if top > bottom:
            break
        for i in range(bottom, top - 1, -1):
            add_value(flat[i * N + left])
        left += 1

    print(inversions % MOD)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {

    static final int MOD = 1000000007;

    static class BIT {
        int n;
        int[] tree;

        BIT(int n) {
            this.n = n;
            tree = new int[n + 1];
        }

        void update(int idx, int val) {
            // 在位置 idx 加上 val
            while (idx <= n) {
                tree[idx] += val;
                idx += idx & -idx;
            }
        }

        int query(int idx) {
            // 前缀和 [1..idx]
            int sum = 0;
            while (idx > 0) {
                sum += tree[idx];
                idx -= idx & -idx;
            }
            return sum;
        }
    }

    static List<Integer> buildSpiral(int[][] matrix) {
        int N = matrix.length;
        List<Integer> spiral = new ArrayList<>();
        int top = 0, bottom = N - 1;
        int left = 0, right = N - 1;
        while (top <= bottom && left <= right) {
            for (int j = left; j <= right; j++) {
                spiral.add(matrix[top][j]);
            }
            top++;
            if (top > bottom) {
                break;
            }
            for (int i = top; i <= bottom; i++) {
                spiral.add(matrix[i][right]);
            }
            right--;
            if (left > right) {
                break;
            }
            for (int j = right; j >= left; j--) {
                spiral.add(matrix[bottom][j]);
            }
            bottom--;
            if (top > bottom) {
                break;
            }
            for (int i = bottom; i >= top; i--) {
                spiral.add(matrix[i][left]);
            }
            left++;
        }
        return spiral;
    }

    static long countInversions(List<Integer> spiral, int N) {
        BIT bit = new BIT(N * N);
        long inversions = 0;
        for (int i = 0; i < spiral.size(); i++) {
            int x = spiral.get(i);
            // 前面已插入 i 个数，大于 x 的个数为 i - query(x)
            inversions = (inversions + (i - bit.query(x))) % MOD;
            bit.update(x, 1);
        }
        return inversions % MOD;
    }

    public static void main(String[] args) throws Exception {
        // N 可达 10^3，用 BufferedReader 加快读入
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int N = Integer.parseInt(br.readLine().trim());
        int[][] matrix = new int[N][N];
        for (int i = 0; i < N; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            for (int j = 0; j < N; j++) {
                matrix[i][j] = Integer.parseInt(st.nextToken());
            }
        }
        System.out.println(countInversions(buildSpiral(matrix), N));
    }
}
```

```cpp
#include <iostream>

const int MOD = 1000000007;

struct BIT {
    int n;
    vector<int> tree;
    BIT(int n) : n(n), tree(n + 1, 0) {}

    void update(int idx, int val) {
        // 在位置 idx 加上 val
        for (; idx <= n; idx += idx & -idx) {
            tree[idx] += val;
        }
    }

    int query(int idx) {
        // 前缀和 [1..idx]
        int sum = 0;
        for (; idx > 0; idx -= idx & -idx) {
            sum += tree[idx];
        }
        return sum;
    }
};
```
