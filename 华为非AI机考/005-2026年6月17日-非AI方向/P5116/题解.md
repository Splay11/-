## 解题思路

本题最多只能换乘一次，所以不需要做完整的最短路，可以直接枚举“直达”和“一次换乘”两种情况。

核心是注意公交线路是单向的，因此在同一条线路上，只能从前面的站点走到后面的站点。

对每条线路预处理两个信息：

1. `fromA[i][x]`：在第 `i` 条线路上，从起点站 `A` 到站点 `x` 的最少经过站点数。
2. `toB[i][x]`：在第 `i` 条线路上，从站点 `x` 到终点站 `B` 的最少经过站点数。

如果某个站点无法按线路方向到达，则不记录。

然后分两种情况计算答案：

1. 直达
   如果某条线路中存在 `A -> B` 的方向关系，那么答案可以是：

$$
fromA[i][B]
$$

2. 换乘一次
   假设先坐第 `i` 条线路，到换乘站 `x`，再坐第 `j` 条线路到 `B`。
   如果 `x` 同时满足：

$$
x \in fromA[i]
$$

且

$$
x \in toB[j]
$$

则可以换乘，经过站点数为：

$$
fromA[i][x] + toB[j][x] - 1
$$

减去 $1$ 是因为换乘站 `x` 在两段路程中都被计算了一次，但题目要求换乘站只算一次。

最后取所有可行方案中的最小值即可。如果没有任何可行方案，输出 `-1`。

## 复杂度分析

设公交线路数量为 $N$，每条线路最多有 $L$ 个站点。

预处理每条线路时，需要枚举站点区间，因此时间复杂度为：

$$
O(NL^2)
$$

枚举两条线路换乘时，最多枚举 $N^2$ 对线路，每条线路最多 $L$ 个可换乘站，因此时间复杂度为：

$$
O(N^2L)
$$

总时间复杂度为：

$$
O(NL^2 + N^2L)
$$

由于 $N \le 20$，$L \le 100$，该复杂度完全可以接受。

空间复杂度主要来自保存每条线路的可达站点信息：

$$
O(NL)
$$

## 代码实现

### Python

```python
import sys


def min_stops(lines, a, b):
    n = len(lines)
    inf = 10 ** 9

    from_a = []
    to_b = []

    for line in lines:
        fa = {}

        # 预处理：从起点 A 出发，在当前线路上能到达哪些站点
        for i in range(len(line)):
            if line[i] == a:
                for j in range(i, len(line)):
                    st = line[j]
                    cnt = j - i + 1
                    if st not in fa or cnt < fa[st]:
                        fa[st] = cnt

        tb = {}

        # 预处理：当前线路上哪些站点能够到达终点 B
        for i in range(len(line)):
            for j in range(i, len(line)):
                if line[j] == b:
                    st = line[i]
                    cnt = j - i + 1
                    if st not in tb or cnt < tb[st]:
                        tb[st] = cnt

        from_a.append(fa)
        to_b.append(tb)

    ans = inf

    # 情况一：同一条线路直达
    for i in range(n):
        if b in from_a[i]:
            ans = min(ans, from_a[i][b])

    # 情况二：最多换乘一次
    for i in range(n):
        for j in range(n):
            if i == j:
                continue

            # 枚举第一条线路上从 A 能到达的站点作为换乘站
            for st in from_a[i]:
                if st in to_b[j]:
                    cnt = from_a[i][st] + to_b[j][st] - 1
                    ans = min(ans, cnt)

    return -1 if ans == inf else ans


def main():
    data = list(map(int, sys.stdin.read().split()))
    idx = 0

    n = data[idx]
    idx += 1

    lines = []
    for _ in range(n):
        m = data[idx]
        idx += 1
        line = data[idx:idx + m]
        idx += m
        lines.append(line)

    a = data[idx]
    b = data[idx + 1]

    print(min_stops(lines, a, b))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static final int INF = 1000000000;

    static int minStops(List<int[]> lines, int a, int b) {
        int n = lines.size();

        List<Map<Integer, Integer>> fromA = new ArrayList<>();
        List<Map<Integer, Integer>> toB = new ArrayList<>();

        for (int[] line : lines) {
            Map<Integer, Integer> fa = new HashMap<>();

            // 预处理：从起点 A 出发，在当前线路上能到达哪些站点
            for (int i = 0; i < line.length; i++) {
                if (line[i] == a) {
                    for (int j = i; j < line.length; j++) {
                        int st = line[j];
                        int cnt = j - i + 1;
                        fa.put(st, Math.min(fa.getOrDefault(st, INF), cnt));
                    }
                }
            }

            Map<Integer, Integer> tb = new HashMap<>();

            // 预处理：当前线路上哪些站点能够到达终点 B
            for (int i = 0; i < line.length; i++) {
                for (int j = i; j < line.length; j++) {
                    if (line[j] == b) {
                        int st = line[i];
                        int cnt = j - i + 1;
                        tb.put(st, Math.min(tb.getOrDefault(st, INF), cnt));
                    }
                }
            }

            fromA.add(fa);
            toB.add(tb);
        }

        int ans = INF;

        // 情况一：同一条线路直达
        for (int i = 0; i < n; i++) {
            if (fromA.get(i).containsKey(b)) {
                ans = Math.min(ans, fromA.get(i).get(b));
            }
        }

        // 情况二：最多换乘一次
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }

                // 枚举第一条线路上从 A 能到达的站点作为换乘站
                for (int st : fromA.get(i).keySet()) {
                    if (toB.get(j).containsKey(st)) {
                        int cnt = fromA.get(i).get(st) + toB.get(j).get(st) - 1;
                        ans = Math.min(ans, cnt);
                    }
                }
            }
        }

        return ans == INF ? -1 : ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        List<int[]> lines = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            int m = sc.nextInt();
            int[] line = new int[m];

            for (int j = 0; j < m; j++) {
                line[j] = sc.nextInt();
            }

            lines.add(line);
        }

        int a = sc.nextInt();
        int b = sc.nextInt();

        System.out.println(minStops(lines, a, b));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

const int INF = 1000000000;

int minStops(vector<vector<int>>& lines, int a, int b) {
    int n = lines.size();

    vector<unordered_map<int, int>> fromA(n);
    vector<unordered_map<int, int>> toB(n);

    for (int k = 0; k < n; k++) {
        vector<int>& line = lines[k];

        // 预处理：从起点 A 出发，在当前线路上能到达哪些站点
        for (int i = 0; i < (int)line.size(); i++) {
            if (line[i] == a) {
                for (int j = i; j < (int)line.size(); j++) {
                    int st = line[j];
                    int cnt = j - i + 1;

                    if (!fromA[k].count(st) || cnt < fromA[k][st]) {
                        fromA[k][st] = cnt;
                    }
                }
            }
        }

        // 预处理：当前线路上哪些站点能够到达终点 B
        for (int i = 0; i < (int)line.size(); i++) {
            for (int j = i; j < (int)line.size(); j++) {
                if (line[j] == b) {
                    int st = line[i];
                    int cnt = j - i + 1;

                    if (!toB[k].count(st) || cnt < toB[k][st]) {
                        toB[k][st] = cnt;
                    }
                }
            }
        }
    }

    int ans = INF;

    // 情况一：同一条线路直达
    for (int i = 0; i < n; i++) {
        if (fromA[i].count(b)) {
            ans = min(ans, fromA[i][b]);
        }
    }

    // 情况二：最多换乘一次
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) {
                continue;
            }

            // 枚举第一条线路上从 A 能到达的站点作为换乘站
            for (auto& p : fromA[i]) {
                int st = p.first;

                if (toB[j].count(st)) {
                    int cnt = fromA[i][st] + toB[j][st] - 1;
                    ans = min(ans, cnt);
                }
            }
        }
    }

    return ans == INF ? -1 : ans;
}

int main() {
    int n;
    cin >> n;

    vector<vector<int>> lines(n);

    for (int i = 0; i < n; i++) {
        int m;
        cin >> m;

        lines[i].resize(m);
        for (int j = 0; j < m; j++) {
            cin >> lines[i][j];
        }
    }

    int a, b;
    cin >> a >> b;

    cout << minStops(lines, a, b) << endl;

    return 0;
}
```
