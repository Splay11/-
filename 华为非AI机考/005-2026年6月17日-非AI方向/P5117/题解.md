## 解题思路

本题需要模拟一个支持撤销和重做的文本编辑器。核心在于：只需要记录每一次真正改变文本的操作，并在撤销或重做时执行对应的反向操作或原操作。

维护三个结构：

1. `text`：当前文本内容，可以用字符数组或字符串模拟栈，方便在末尾插入和删除。
2. `undo`：已经执行过、可以被撤销的操作记录。
3. `redo`：已经被撤销、可以被重做的操作记录。

对于每种操作：

* `APPEND x`：

  * 将字符串 $x$ 加到文本末尾。
  * 将操作 `APPEND x` 记录到 `undo`。
  * 因为执行了新的编辑操作，所以清空 `redo`。

* `POP`：

  * 如果文本为空，则无效果，不记录历史。
  * 如果文本非空，删除最后一个字符 $c$。
  * 将操作 `POP c` 记录到 `undo`，其中 $c$ 是被删除的字符。
  * 因为执行了新的编辑操作，所以清空 `redo`。

* `UNDO`：

  * 如果上一次操作是 `APPEND x`，说明之前添加了 $x$，撤销时删除末尾长度为 $|x|$ 的字符串。
  * 如果上一次操作是 `POP c`，说明之前删除了字符 $c$，撤销时把 $c$ 加回文本末尾。
  * 被撤销的操作需要放入 `redo`，方便之后重做。

* `REDO`：

  * 如果重做的是 `APPEND x`，则再次把 $x$ 加到文本末尾。
  * 如果重做的是 `POP c`，则再次删除文本末尾一个字符。
  * 重做后的操作重新放回 `undo`。

注意：当在撤销之后执行新的 `APPEND` 或有效的 `POP` 操作时，之前的 `redo` 记录需要被清空，因为操作历史已经产生了新的分支。

## 复杂度分析

设所有 `APPEND` 操作插入的字符总数为 $S$，操作数为 $N$。

每个字符最多在插入、删除、撤销、重做过程中被处理常数次，因此总时间复杂度为：

$$
O(N + S)
$$

其中题目保证 $S \le 200000$，所以复杂度可以满足要求。

需要保存当前文本以及操作历史，空间复杂度为：

$$
O(N + S)
$$

## 代码实现

### Python

```python
import sys


def edit(ops):
    # text 保存当前文本内容，使用列表方便在末尾增删字符
    text = []

    # undo 保存已经执行、可以撤销的操作
    undo = []

    # redo 保存已经撤销、可以重做的操作
    redo = []

    for op in ops:
        if op[0] == "APPEND":
            x = op[1]

            # 执行追加操作
            text.extend(x)

            # 记录该操作，之后可以撤销
            undo.append(("APPEND", x))

            # 执行新的编辑操作后，之前可重做的记录失效
            redo.clear()

        elif op[0] == "POP":
            if text:
                # 记录被删除的字符，撤销时需要恢复它
                c = text.pop()

                # 记录删除操作
                undo.append(("POP", c))

                # 执行新的编辑操作后，之前可重做的记录失效
                redo.clear()

        elif op[0] == "UNDO":
            if undo:
                typ, val = undo.pop()

                if typ == "APPEND":
                    # 撤销追加操作：删除末尾 len(val) 个字符
                    for _ in range(len(val)):
                        text.pop()
                else:
                    # 撤销删除操作：把被删字符加回末尾
                    text.append(val)

                # 被撤销的操作可以被重做
                redo.append((typ, val))

        else:
            if redo:
                typ, val = redo.pop()

                if typ == "APPEND":
                    # 重做追加操作
                    text.extend(val)
                else:
                    # 重做删除操作
                    text.pop()

                # 重做后，该操作再次变为可撤销
                undo.append((typ, val))

    return "".join(text)


def main():
    n = int(sys.stdin.readline())
    ops = []

    for _ in range(n):
        line = sys.stdin.readline().strip()

        if line.startswith("APPEND"):
            # APPEND 后面的字符串不含空格，直接取出即可
            ops.append(("APPEND", line[7:]))
        else:
            ops.append((line, ""))

    ans = edit(ops)

    if ans:
        print(ans)
    else:
        print("nothing")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Stack;

public class Main {
    static class Op {
        String type;
        String val;

        Op(String type, String val) {
            this.type = type;
            this.val = val;
        }
    }

    public static String edit(ArrayList<Op> ops) {
        // text 保存当前文本内容，StringBuilder 支持高效末尾增删
        StringBuilder text = new StringBuilder();

        // undo 保存已经执行、可以撤销的操作
        Stack<Op> undo = new Stack<>();

        // redo 保存已经撤销、可以重做的操作
        Stack<Op> redo = new Stack<>();

        for (Op op : ops) {
            if (op.type.equals("APPEND")) {
                String x = op.val;

                // 执行追加操作
                text.append(x);

                // 记录该操作，之后可以撤销
                undo.push(new Op("APPEND", x));

                // 执行新的编辑操作后，之前可重做的记录失效
                redo.clear();
            } else if (op.type.equals("POP")) {
                if (text.length() > 0) {
                    // 记录被删除的字符，撤销时需要恢复它
                    char c = text.charAt(text.length() - 1);
                    text.deleteCharAt(text.length() - 1);

                    // 记录删除操作
                    undo.push(new Op("POP", String.valueOf(c)));

                    // 执行新的编辑操作后，之前可重做的记录失效
                    redo.clear();
                }
            } else if (op.type.equals("UNDO")) {
                if (!undo.empty()) {
                    Op last = undo.pop();

                    if (last.type.equals("APPEND")) {
                        // 撤销追加操作：删除末尾 last.val.length() 个字符
                        int len = last.val.length();
                        text.delete(text.length() - len, text.length());
                    } else {
                        // 撤销删除操作：把被删字符加回末尾
                        text.append(last.val);
                    }

                    // 被撤销的操作可以被重做
                    redo.push(last);
                }
            } else {
                if (!redo.empty()) {
                    Op last = redo.pop();

                    if (last.type.equals("APPEND")) {
                        // 重做追加操作
                        text.append(last.val);
                    } else {
                        // 重做删除操作
                        text.deleteCharAt(text.length() - 1);
                    }

                    // 重做后，该操作再次变为可撤销
                    undo.push(last);
                }
            }
        }

        return text.toString();
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        ArrayList<Op> ops = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            String line = br.readLine();

            if (line.startsWith("APPEND")) {
                // APPEND 后面的字符串不含空格，直接截取
                ops.add(new Op("APPEND", line.substring(7)));
            } else {
                ops.add(new Op(line, ""));
            }
        }

        String ans = edit(ops);

        if (ans.length() == 0) {
            System.out.println("nothing");
        } else {
            System.out.println(ans);
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Op {
    string type;
    string val;
};

string edit(vector<Op>& ops) {
    // text 保存当前文本内容，string 支持在末尾高效增删
    string text;

    // undo 保存已经执行、可以撤销的操作
    vector<Op> undo;

    // redo 保存已经撤销、可以重做的操作
    vector<Op> redo;

    for (Op op : ops) {
        if (op.type == "APPEND") {
            string x = op.val;

            // 执行追加操作
            text += x;

            // 记录该操作，之后可以撤销
            undo.push_back({"APPEND", x});

            // 执行新的编辑操作后，之前可重做的记录失效
            redo.clear();
        } else if (op.type == "POP") {
            if (!text.empty()) {
                // 记录被删除的字符，撤销时需要恢复它
                char c = text.back();
                text.pop_back();

                // 记录删除操作
                undo.push_back({"POP", string(1, c)});

                // 执行新的编辑操作后，之前可重做的记录失效
                redo.clear();
            }
        } else if (op.type == "UNDO") {
            if (!undo.empty()) {
                Op last = undo.back();
                undo.pop_back();

                if (last.type == "APPEND") {
                    // 撤销追加操作：删除末尾 last.val.size() 个字符
                    int len = (int)last.val.size();
                    text.erase(text.size() - len, len);
                } else {
                    // 撤销删除操作：把被删字符加回末尾
                    text.push_back(last.val[0]);
                }

                // 被撤销的操作可以被重做
                redo.push_back(last);
            }
        } else {
            if (!redo.empty()) {
                Op last = redo.back();
                redo.pop_back();

                if (last.type == "APPEND") {
                    // 重做追加操作
                    text += last.val;
                } else {
                    // 重做删除操作
                    text.pop_back();
                }

                // 重做后，该操作再次变为可撤销
                undo.push_back(last);
            }
        }
    }

    return text;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<Op> ops;

    for (int i = 0; i < n; i++) {
        string type;
        cin >> type;

        if (type == "APPEND") {
            string x;
            cin >> x;
            ops.push_back({"APPEND", x});
        } else {
            ops.push_back({type, ""});
        }
    }

    string ans = edit(ops);

    if (ans.empty()) {
        cout << "nothing\n";
    } else {
        cout << ans << '\n';
    }

    return 0;
}
```
