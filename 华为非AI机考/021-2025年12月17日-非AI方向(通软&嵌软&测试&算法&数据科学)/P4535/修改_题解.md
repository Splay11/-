# 二重循环朴素解法

这道题的正解是 $树状数组$，但是我们用朴素解法 $枚举所有子数组$ 也能在考试时拿到一定的分数。

题意：给定长度为 $n$ 的数组 $papers$，以及闭区间 $[left, right]$，统计有多少个连续子数组，其元素之和落在 $[left, right]$ 内。

朴素做法：

1. 先算前缀和 $pre$：$pre[0]=0$，$pre[i]=papers[0]+\cdots+papers[i-1]$。
2. 子数组 $papers[l..r]$ 的和等于 $pre[r+1]-pre[l]$。
3. 枚举所有 $0\le l\le r<n$，若 $left\le pre[r+1]-pre[l]\le right$，答案加 $1$。

小数据可以过；当 $n$ 接近 $10000$ 时，$O(n^2)$ 可能超时，需要后面的树状数组做法。
## 二重循环解法

```python
def count_subarrays(papers, left, right):
    n = len(papers)
    # pre[i]：前 i 个元素之和
    pre = [0] * (n + 1)
    for i in range(n):
        pre[i + 1] = pre[i] + papers[i]

    ans = 0
    # 枚举所有子数组 [l..r]
    for l in range(n):
        for r in range(l, n):
            s = pre[r + 1] - pre[l]
            if left <= s <= right:
                ans += 1
    return ans


def main():

    papers = list(map(int, input().split()))
    left, right = map(int, input().split())
    print(count_subarrays(papers, left, right))


if __name__ == "__main__":
    main()
```

# 正确解法

## 题目分析

把数组转为前缀和：$pre[0]=0$，$pre[i]=papers[0]+\cdots+papers[i-1]$。任意连续子段 $[l..r]$ 的和为 $pre[r+1]-pre[l]$。

要求 $left\le pre[r+1]-pre[l]\le right$，对每个右端点对应的下标 $j=r+1$，等价于统计已出现的前缀和里有多少个 $pre[i]$（$i<j$）满足：

$$
pre[i]\in [pre[j]-right,\ pre[j]-left]
$$

做法：

1. 对所有 $pre$ 做坐标压缩，得到有序数组 $vals$。
2. 用树状数组维护已出现前缀和的出现次数。
3. 先把 $pre[0]=0$ 加入树状数组。
4. 按 $j=1..n$ 枚举：查询区间 $[pre[j]-right,\ pre[j]-left]$ 内的个数，再把 $pre[j]$ 加入树状数组。

区间计数：用二分在 $vals$ 上找到 $\le hi$ 与 $<lo$ 的位置，再用树状数组前缀和相减。

## 复杂度分析

* 时间复杂度：$O(n\log n)$（排序压缩 $O(n\log n)$，每个前缀若干次二分与树状数组操作 $O(\log n)$）。
* 空间复杂度：$O(n)$（前缀和、压缩数组、树状数组）。

## 代码实现

### Python

```python
from bisect import bisect_left, bisect_right


class BIT:
    def __init__(self, n):
        self.n = n
        self.bit = [0] * (n + 1)

    def add(self, i, v):
        # 单点加
        while i <= self.n:
            self.bit[i] += v
            i += i & -i

    def sum(self, i):
        # 前缀和 [1..i]
        s = 0
        while i > 0:
            s += self.bit[i]
            i -= i & -i
        return s


def count_subarrays(papers, left, right):
    n = len(papers)
    pre = [0] * (n + 1)
    for i in range(n):
        pre[i + 1] = pre[i] + papers[i]

    # 坐标压缩
    vals = sorted(set(pre))
    bit = BIT(len(vals))

    ans = 0
    # 先加入 pre[0]=0
    bit.add(bisect_left(vals, 0) + 1, 1)

    for j in range(1, n + 1):
        pj = pre[j]
        lo = pj - right
        hi = pj - left
        # idx_hi：<=hi 的个数；idx_lo：<lo 的个数
        idx_hi = bisect_right(vals, hi)
        idx_lo = bisect_left(vals, lo)
        ans += bit.sum(idx_hi) - bit.sum(idx_lo)
        # 加入当前前缀和
        bit.add(bisect_left(vals, pj) + 1, 1)

    return ans


def main():
    n = int(input())
    papers = list(map(int, input().split()))
    left, right = map(int, input().split())
    print(count_subarrays(papers, left, right))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static class BIT {
        int n;
        long[] bit;

        BIT(int n) {
            this.n = n;
            this.bit = new long[n + 1];
        }

        void add(int i, long v) {
            // 单点加
            while (i <= n) {
                bit[i] += v;
                i += i & -i;
            }
        }

        long sum(int i) {
            // 前缀和 [1..i]
            long s = 0;
            while (i > 0) {
                s += bit[i];
                i -= i & -i;
            }
            return s;
        }
    }

    // 第一个 >= x 的位置
    static int lowerBound(long[] a, long x) {
        int l = 0, r = a.length;
        while (l < r) {
            int m = (l + r) >>> 1;
            if (a[m] >= x) {
                r = m;
            } else {
                l = m + 1;
            }
        }
        return l;
    }

    // 第一个 > x 的位置
    static int upperBound(long[] a, long x) {
        int l = 0, r = a.length;
        while (l < r) {
            int m = (l + r) >>> 1;
            if (a[m] > x) {
                r = m;
            } else {
                l = m + 1;
            }
        }
        return l;
    }

    static long countSubarrays(int[] papers, long left, long right) {
        int n = papers.length;
        long[] pre = new long[n + 1];
        for (int i = 0; i < n; i++) {
            pre[i + 1] = pre[i] + papers[i];
        }

        // 坐标压缩：排序去重
        long[] tmp = Arrays.copyOf(pre, pre.length);
        Arrays.sort(tmp);
        int m = 0;
        for (int i = 0; i < tmp.length; i++) {
            if (i == 0 || tmp[i] != tmp[i - 1]) {
                tmp[m++] = tmp[i];
            }
        }
        long[] vals = Arrays.copyOf(tmp, m);

        BIT bit = new BIT(m);
        long ans = 0;
        bit.add(lowerBound(vals, 0) + 1, 1);

        for (int j = 1; j <= n; j++) {
            long pj = pre[j];
            long lo = pj - right;
            long hi = pj - left;
            int idxHi = upperBound(vals, hi);
            int idxLo = lowerBound(vals, lo);
            ans += bit.sum(idxHi) - bit.sum(idxLo);
            bit.add(lowerBound(vals, pj) + 1, 1);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] papers = new int[n];
        for (int i = 0; i < n; i++) {
            papers[i] = sc.nextInt();
        }
        long left = sc.nextLong();
        long right = sc.nextLong();
        System.out.println(countSubarrays(papers, left, right));
        sc.close();
    }
}
```

### C++

```cpp
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

struct BIT {
    int n;
    vector<long long> bit;
    BIT(int n) : n(n), bit(n + 1, 0) {}

    void add(int i, long long v) {
        // 单点加
        for (; i <= n; i += i & -i) {
            bit[i] += v;
        }
    }

    long long sum(int i) {
        // 前缀和 [1..i]
        long long s = 0;
        for (; i > 0; i -= i & -i) {
            s += bit[i];
        }
        return s;
    }
};

long long count_subarrays(const vector<int>& papers, long long left, long long right) {
    int n = (int)papers.size();
    vector<long long> pre(n + 1, 0);
    for (int i = 0; i < n; i++) {
        pre[i + 1] = pre[i] + papers[i];
    }

    // 坐标压缩
    vector<long long> vals = pre;
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());

    BIT bit((int)vals.size());
    long long ans = 0;
    int pos0 = (int)(lower_bound(vals.begin(), vals.end(), 0LL) - vals.begin()) + 1;
    bit.add(pos0, 1);

    for (int j = 1; j <= n; j++) {
        long long pj = pre[j];
        long long lo = pj - right;
        long long hi = pj - left;
        int idxHi = (int)(upper_bound(vals.begin(), vals.end(), hi) - vals.begin());
        int idxLo = (int)(lower_bound(vals.begin(), vals.end(), lo) - vals.begin());
        ans += bit.sum(idxHi) - bit.sum(idxLo);
        int pos = (int)(lower_bound(vals.begin(), vals.end(), pj) - vals.begin()) + 1;
        bit.add(pos, 1);
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    vector<int> papers(n);
    for (int i = 0; i < n; i++) {
        cin >> papers[i];
    }
    long long left, right;
    cin >> left >> right;
    cout << count_subarrays(papers, left, right) << "\n";
    return 0;
}
```
