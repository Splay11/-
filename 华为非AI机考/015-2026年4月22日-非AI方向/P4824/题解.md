# 暴力DFS解法(机考中可以通过大部分数据)
这题考试的时候直接枚举所有可能的路径就可以骗很多分。枚举所有只向右、向下的路径，只要走到右下角，就统计一次转弯次数，最后取最小值。

## 暴力做法

```python
import sys

ans = float("inf")
pcb = []
m = n = 0

# dir:
# -1 表示还没有方向
# 0 表示上一步是向下
# 1 表示上一步是向右
def dfs(x, y, direction, turn):
    global ans, pcb, m, n

    # 越界
    if x < 0 or x >= m or y < 0 or y >= n:
        return

    # 不能通过
    if pcb[x][y] != 0:
        return

    # 到达终点
    if x == m - 1 and y == n - 1:
        ans = min(ans, turn)
        return

    # 往下走
    new_turn = turn
    if direction != -1 and direction != 0:
        new_turn += 1
    dfs(x + 1, y, 0, new_turn)

    # 往右走
    new_turn = turn
    if direction != -1 and direction != 1:
        new_turn += 1
    dfs(x, y + 1, 1, new_turn)


def main():
    global ans, pcb, m, n

    data = sys.stdin.read().strip().split()
    if not data:
        return

    idx = 0
    m = int(data[idx])
    idx += 1
    n = int(data[idx])
    idx += 1

    # 行列非法
    if m <= 0 or n <= 0 or m > 100 or n > 100:
        print(-1)
        return

    pcb = [[0] * n for _ in range(m)]

    for i in range(m):
        for j in range(n):
            if idx >= len(data):
                print(-1)
                return

            pcb[i][j] = int(data[idx])
            idx += 1

            # 矩阵元素非法
            if pcb[i][j] < 0 or pcb[i][j] > 4:
                print(-1)
                return

    # 起点或终点不能走
    if pcb[0][0] != 0 or pcb[m - 1][n - 1] != 0:
        print(-1)
        return

    dfs(0, 0, -1, 0)

    if ans == float("inf"):
        print(-1)
    else:
        print(ans)


if __name__ == "__main__":
    main()
```

# 满分解法

## 题目分析

这题可以用$BFS$做，但要用$0$-$1$ $BFS$，因为优化目标是最少转弯次数，而不是最少步数。

把每个状态设计成$(x,y,d)$：

* $(x,y)$表示当前位置
* $d$表示到达当前位置时的方向
* $d=0$表示向下
* $d=1$表示向右

由于只能向下或向右移动，所以每个状态只会扩展这两种方向。

若下一步方向和当前方向相同，则这一步新增代价为$0$；
若方向不同，则新增代价为$1$。

这样问题就转化成边权只可能为$0$或$1$的最短路问题，直接用$0$-$1$ $BFS$即可：

* 边权为$0$，新状态放到队首
* 边权为$1$，新状态放到队尾

定义$dist[x][y][d]$表示到达状态$(x,y,d)$的最少转弯次数，最终答案就是终点两个方向状态的最小值。

实现时要特别注意：

1. 若$n,m$不在题目范围内，直接返回$-1$
2. 若起点或终点不是$0$，直接返回$-1$
3. 只有值为$0$的位置可以经过
4. 起点本身没有方向，因此初始化时从起点尝试向下、向右两种出发方式

## 复杂度分析

状态总数为$n \times m \times 2$，每个状态最多扩展$2$条边。

* 时间复杂度：$O(nm)$
* 空间复杂度：$O(nm)$

## 代码实现

### Python

```python
import sys
from collections import deque

INF = 10 ** 9

# 计算最少转弯次数
def min_turns(grid, n, m):
    # 先判断输入参数是否合法
    if n <= 0 or m <= 0 or n > 100 or m > 100:
        return -1

    # 起点或终点不能走
    if grid[0][0] != 0 or grid[n - 1][m - 1] != 0:
        return -1

    # 特判：只有一个点
    if n == 1 and m == 1:
        return 0

    # dist[i][j][0]：到达(i,j)，最后方向为向下的最少转弯次数
    # dist[i][j][1]：到达(i,j)，最后方向为向右的最少转弯次数
    dist = [[[INF] * 2 for _ in range(m)] for _ in range(n)]
    q = deque()

    # 起点向下出发
    if n > 1 and grid[1][0] == 0:
        dist[1][0][0] = 0
        q.append((1, 0, 0))

    # 起点向右出发
    if m > 1 and grid[0][1] == 0:
        dist[0][1][1] = 0
        q.append((0, 1, 1))

    # 方向数组：0表示向下，1表示向右
    dx = [1, 0]
    dy = [0, 1]

    # 0-1 BFS
    while q:
        x, y, d = q.popleft()

        for nd in range(2):
            nx = x + dx[nd]
            ny = y + dy[nd]

            # 越界
            if nx < 0 or nx >= n or ny < 0 or ny >= m:
                continue

            # 只能走值为0的位置
            if grid[nx][ny] != 0:
                continue

            # 同方向代价为0，转弯代价为1
            cost = 0 if nd == d else 1

            if dist[nx][ny][nd] > dist[x][y][d] + cost:
                dist[nx][ny][nd] = dist[x][y][d] + cost
                if cost == 0:
                    q.appendleft((nx, ny, nd))
                else:
                    q.append((nx, ny, nd))

    ans = min(dist[n - 1][m - 1][0], dist[n - 1][m - 1][1])
    return -1 if ans == INF else ans


def main():
    data = sys.stdin.read().strip().split()
    if not data:
        print(-1)
        return

    n = int(data[0])
    m = int(data[1])

    # 参数不合法时，直接输出-1，不再继续读矩阵
    if n <= 0 or m <= 0 or n > 100 or m > 100:
        print(-1)
        return

    # 数据合法时再读取矩阵
    grid = []
    idx = 2
    for _ in range(n):
        row = []
        for _ in range(m):
            row.append(int(data[idx]))
            idx += 1
        grid.append(row)

    print(min_turns(grid, n, m))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    static final int INF = 1000000000;

    // 状态类
    static class Node {
        int x;
        int y;
        int d;

        Node(int x, int y, int d) {
            this.x = x;
            this.y = y;
            this.d = d;
        }
    }

    // 计算最少转弯次数
    public static int minTurns(int[][] grid, int n, int m) {
        // 先判断输入参数是否合法
        if (n <= 0 || m <= 0 || n > 100 || m > 100) {
            return -1;
        }

        // 起点或终点不能走
        if (grid[0][0] != 0 || grid[n - 1][m - 1] != 0) {
            return -1;
        }

        // 特判：只有一个点
        if (n == 1 && m == 1) {
            return 0;
        }

        // dist[i][j][0]：到达(i,j)，最后方向为向下
        // dist[i][j][1]：到达(i,j)，最后方向为向右
        int[][][] dist = new int[n][m][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                Arrays.fill(dist[i][j], INF);
            }
        }

        Deque<Node> deque = new LinkedList<>();

        // 起点向下出发
        if (n > 1 && grid[1][0] == 0) {
            dist[1][0][0] = 0;
            deque.offerLast(new Node(1, 0, 0));
        }

        // 起点向右出发
        if (m > 1 && grid[0][1] == 0) {
            dist[0][1][1] = 0;
            deque.offerLast(new Node(0, 1, 1));
        }

        // 方向数组：0表示向下，1表示向右
        int[] dx = {1, 0};
        int[] dy = {0, 1};

        // 0-1 BFS
        while (!deque.isEmpty()) {
            Node cur = deque.pollFirst();
            int x = cur.x;
            int y = cur.y;
            int d = cur.d;

            for (int nd = 0; nd < 2; nd++) {
                int nx = x + dx[nd];
                int ny = y + dy[nd];

                // 越界
                if (nx < 0 || nx >= n || ny < 0 || ny >= m) {
                    continue;
                }

                // 只能走值为0的位置
                if (grid[nx][ny] != 0) {
                    continue;
                }

                // 同方向代价为0，转弯代价为1
                int cost = (nd == d) ? 0 : 1;

                if (dist[nx][ny][nd] > dist[x][y][d] + cost) {
                    dist[nx][ny][nd] = dist[x][y][d] + cost;
                    if (cost == 0) {
                        deque.offerFirst(new Node(nx, ny, nd));
                    } else {
                        deque.offerLast(new Node(nx, ny, nd));
                    }
                }
            }
        }

        int ans = Math.min(dist[n - 1][m - 1][0], dist[n - 1][m - 1][1]);
        return ans == INF ? -1 : ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        if (!sc.hasNextInt()) {
            System.out.println(-1);
            return;
        }

        int n = sc.nextInt();
        int m = sc.nextInt();

        // 参数不合法时，直接输出-1
        if (n <= 0 || m <= 0 || n > 100 || m > 100) {
            System.out.println(-1);
            return;
        }

        int[][] grid = new int[n][m];

        // 按空格分隔的矩阵读入
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = sc.nextInt();
            }
        }

        System.out.println(minTurns(grid, n, m));
    }
}
```

```cpp
// 状态结构体
struct Node {
    int x;
    int y;
    int d;
};

// 计算最少转弯次数
int minTurns(vector<vector<int> >& grid, int n, int m) {
    // 先判断输入参数是否合法
    if (n <= 0 || m <= 0 || n > 100 || m > 100) {
        return -1;
    }

    // 起点或终点不能走
    if (grid[0][0] != 0 || grid[n - 1][m - 1] != 0) {
        return -1;
    }

    // 特判：只有一个点
    if (n == 1 && m == 1) {
        return 0;
    }

    // dist[i][j][0]：到达(i,j)，最后方向为向下
    // dist[i][j][1]：到达(i,j)，最后方向为向右
    vector<vector<vector<int> > > dist(
        n, vector<vector<int> >(m, vector<int>(2, INF))
    );

    deque<Node> q;

    // 起点向下出发
    if (n > 1 && grid[1][0] == 0) {
        dist[1][0][0] = 0;
        q.push_back({1, 0, 0});
    }

    // 起点向右出发
    if (m > 1 && grid[0][1] == 0) {
        dist[0][1][1] = 0;
        q.push_back({0, 1, 1});
    }

    // 方向数组：0表示向下，1表示向右
    int dx[2] = {1, 0};
    int dy[2] = {0, 1};

    // 0-1 BFS
    while (!q.empty()) {
        Node cur = q.front();
        q.pop_front();

        int x = cur.x;
        int y = cur.y;
        int d = cur.d;

        for (int nd = 0; nd < 2; nd++) {
            int nx = x + dx[nd];
            int ny = y + dy[nd];

            // 越界
            if (nx < 0 || nx >= n || ny < 0 || ny >= m) {
                continue;
            }

            // 只能走值为0的位置
            if (grid[nx][ny] != 0) {
                continue;
            }

            // 同方向代价为0，转弯代价为1

            if (dist[nx][ny][nd] > dist[x][y][d] + cost) {
                dist[nx][ny][nd] = dist[x][y][d] + cost;
                if (cost == 0) {
                    q.push_front({nx, ny, nd});
                } else {
                    q.push_back({nx, ny, nd});
                }
            }
        }
    }

    int ans = min(dist[n - 1][m - 1][0], dist[n - 1][m - 1][1]);
    return ans == INF ? -1 : ans;
}
```
