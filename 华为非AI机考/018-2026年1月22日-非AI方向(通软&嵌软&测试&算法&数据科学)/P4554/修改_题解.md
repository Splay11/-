# DFS暴力解法

这道题的正解是 $拉格朗日松弛$（二分惩罚参数 $+$ 能量背包），但是我们用朴素解法 $按门槛排序后 DFS/回溯$ 也能在考试时拿到一定的分数。

题意要点：

* 初始经验 $W$、能量 $E$，最多打 $K$ 只怪。
* 怪 $i$：门槛 $X[i]$（不消耗经验）、收益 $Y[i]$、耗能 $Z[i]$。
* 经验只增不减；一组选中的怪总能按门槛从小到大去打。

朴素做法：

1. 先按门槛 $X$ 从小到大排序。
2. 按顺序对每只怪做「打 / 不打」的 DFS：若当前经验 $\ge X[i]$、剩余能量 $\ge Z[i]$、且已打数量 $< K$，则可以打。
3. 搜索过程中维护当前经验、剩余能量、已打数量，用最大经验更新答案；经验相同时取打怪数更大的方案。

样例和小 $N$ 可以算对；但 $N$ 最大 $1000$，枚举子集是指数级，大数据会超时，因此后面再给出正解。

## DFS解法

```python
def solve_naive(n, w, E, K, X, Y, Z):
    mons = sorted(zip(X, Y, Z), key=lambda t: t[0])  # 按门槛排序
    best_y = 0
    best_c = 0

    def dfs(i, exp, energy, cnt):
        nonlocal best_y, best_c
        # 用当前状态更新答案
        if exp > best_y or (exp == best_y and cnt > best_c):
            best_y, best_c = exp, cnt
        if i == n or cnt == K:
            return

        xi, yi, zi = mons[i]
        # 不打当前怪
        dfs(i + 1, exp, energy, cnt)
        # 打当前怪（经验、能量、数量都够）
        if exp >= xi and energy >= zi and cnt < K:
            dfs(i + 1, exp + yi, energy - zi, cnt + 1)

    dfs(0, w, E, 0)
    return best_y, best_c

    X = list(map(int, input().split()))
    Y = list(map(int, input().split()))
    Z = list(map(int, input().split()))
    ans_w, ans_k = solve_naive(n, w, E, K, X, Y, Z)
```

```cpp
#include <iostream>

#include <algorithm>
using namespace std;

struct Monster {
    int x, y, z;
};

int bestY, bestC, nGlobal, KGlobal;
vector<Monster> mons;

void dfs(int i, int exp, int energy, int cnt) {
    if (exp > bestY || (exp == bestY && cnt > bestC)) {
        bestY = exp;
        bestC = cnt;
    }
    if (i == nGlobal || cnt == KGlobal) {
        return;
    }

    // 不打
    dfs(i + 1, exp, energy, cnt);
    // 打
    if (exp >= mons[i].x && energy >= mons[i].z && cnt < KGlobal) {
        dfs(i + 1, exp + mons[i].y, energy - mons[i].z, cnt + 1);
    }
}

pair<int, int> solve_naive(int n, int w, int E, int K, vector<int>& X, vector<int>& Y, vector<int>& Z) {
    nGlobal = n;
    KGlobal = K;
    mons.resize(n);
    for (int i = 0; i < n; i++) {
        mons[i] = {X[i], Y[i], Z[i]};
    }
    sort(mons.begin(), mons.end(), [](const Monster& a, const Monster& b) {
        return a.x < b.x;
    });
    bestY = 0;
    bestC = 0;
    dfs(0, w, E, 0);
    return {bestY, bestC};
}
```

# 正确解法

## 题目分析

直接对「能量 $+$ 打怪数」做二维 $0/1$ 背包，能量维最大 $80000$，会偏紧或不好处理；本题用**拉格朗日松弛**：把数量约束 $K$ 变成惩罚参数 $\lambda$，再对固定 $\lambda$ 做一维能量背包。

关键观察：

1. 经验只增不减，选中的怪按门槛 $X$ 从小到大打不会变差，因此先排序。
2. 固定 $\lambda$ 后，把打一只怪的收益从 $Y$ 改成 $Y-\lambda$，用能量 $0/1$ 背包最大化调整收益 $\mathrm{sum}(Y)-\lambda\cdot cnt$。
3. $\lambda$ 越大，最优解越倾向少打怪，最优打怪数随 $\lambda$ 单调不增，因此可以二分 $\lambda$。

固定 $\lambda$ 的 DP（按门槛排序后逐个怪做 $0/1$ 背包）：

* $dpAdj[e]$：耗能 $e$ 时最大调整收益 $\mathrm{sum}(Y)-\lambda\cdot cnt$
* $dpY[e]$：真实经验收益 $\mathrm{sum}(Y)$
* $dpC[e]$：打怪数 $cnt$
* 仅当 $W+dpY[e]\ge X[i]$ 时，才能转移到 $e+Z[i]$

比较状态优劣的优先级：先看 $dpAdj$，相同再看 $dpY$，再相同取 $dpC$ 更大（保证二分单调）。

二分与答案恢复：

* 先跑 $\lambda=0$：若最优打怪数已 $\le K$，说明 $K$ 不紧，直接在 $cnt\le K$ 的状态里取最大 $dpY$。
* 否则二分，找最大的 $\lambda$ 使最优打怪数 $\ge K$，记为 $lam$。
* 此时原问题最优真实收益满足 $OPT=bestAdj(lam)+lam\cdot K$。
* 再扫描所有 $e$，找 $dpC[e]\le K$ 且 $dpAdj[e]+lam\cdot K=OPT$ 的状态，取 $dpY$ 最大者。

最终输出：最大经验 $W+bestY$，实际打怪数 $bestC$。

## 复杂度分析

设能量上限为 $E$（$\le 80000$），妖怪数 $N$（$\le 1000$），$\lambda$ 二分约 $\log 1000$ 次：

* 单次 DP：$O(N\cdot E)$
* 总时间：$O(N\cdot E\cdot \log 1000)$
* 空间：$O(E)$

## 代码实现

### Python

```python
def solve(n, w, E, K, X, Y, Z):
    mons = list(zip(X, Y, Z))
    mons.sort(key=lambda t: t[0])  # 按门槛排序
    NEG = -10 ** 18

    def run(lam):
        # dpAdj[e]：耗能 e 时最大 (sumY - lam*cnt)
        dpAdj = [NEG] * (E + 1)
        dpY = [0] * (E + 1)
        dpC = [0] * (E + 1)
        dpAdj[0] = 0

        for xi, yi, zi in mons:
            for e in range(E - zi, -1, -1):  # 0/1 背包倒序
                if dpAdj[e] == NEG:
                    continue
                if w + dpY[e] < xi:  # 经验不够不能打
                    continue
                ne = e + zi
                nadj = dpAdj[e] + yi - lam
                ny = dpY[e] + yi
                nc = dpC[e] + 1
                # adj 优先，其次真实 y，再次 count 更大
                if (nadj > dpAdj[ne] or
                        (nadj == dpAdj[ne] and (ny > dpY[ne] or
                         (ny == dpY[ne] and nc > dpC[ne])))):
                    dpAdj[ne] = nadj
                    dpY[ne] = ny
                    dpC[ne] = nc

        bestAdj, bestY, bestC = NEG, 0, 0
        for e in range(E + 1):
            if dpAdj[e] == NEG:
                continue
            a, yy, cc = dpAdj[e], dpY[e], dpC[e]
            if (a > bestAdj or
                    (a == bestAdj and (yy > bestY or
                     (yy == bestY and cc > bestC)))):
                bestAdj, bestY, bestC = a, yy, cc
        return dpAdj, dpY, dpC, bestAdj, bestY, bestC

    dpAdj0, dpY0, dpC0, _, _, bestC0 = run(0)
    if bestC0 <= K:  # K 不紧，直接取最大真实收益
        bestY, bestC = 0, 0
        for e in range(E + 1):
            if dpAdj0[e] == NEG:
                continue
            if dpC0[e] <= K:
                if dpY0[e] > bestY or (dpY0[e] == bestY and dpC0[e] > bestC):
                    bestY, bestC = dpY0[e], dpC0[e]
        return w + bestY, bestC

    # 二分最大 lam，使得最优 count >= K
    lo, hi = 0, 1001
    while lo < hi:
        mid = (lo + hi + 1) // 2
        _, _, _, _, _, cnt = run(mid)
        if cnt >= K:
            lo = mid
        else:
            hi = mid - 1
    lam = lo

    dpAdj, dpY, dpC, bestAdj, _, _ = run(lam)
    OPT = bestAdj + lam * K

    bestY, bestC = 0, 0
    for e in range(E + 1):
        if dpAdj[e] == NEG:
            continue
        if dpC[e] <= K and dpAdj[e] + lam * K == OPT:
            if dpY[e] > bestY or (dpY[e] == bestY and dpC[e] > bestC):
                bestY, bestC = dpY[e], dpC[e]
    return w + bestY, bestC


def main():
    n, w, E, K = map(int, input().split())
    X = list(map(int, input().split()))
    Y = list(map(int, input().split()))
    Z = list(map(int, input().split()))
    ans_w, ans_k = solve(n, w, E, K, X, Y, Z)
    print(ans_w, ans_k)


if __name__ == "__main__":
    main()
```

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

public class Main {

    static class Monster {
        int x, y, z;
        Monster(int x, int y, int z) {
            this.x = x;
            this.y = y;
            this.z = z;
        }
    }

    static class DPRes {
        int[] adj, y, c;
        int bestAdj, bestY, bestC;
        DPRes(int[] adj, int[] y, int[] c, int bestAdj, int bestY, int bestC) {
            this.adj = adj;
            this.y = y;
            this.c = c;
            this.bestAdj = bestAdj;
            this.bestY = bestY;
            this.bestC = bestC;
        }
    }

    static boolean better(int adj1, int y1, int c1, int adj2, int y2, int c2) {
        if (adj1 != adj2) {
            return adj1 > adj2;
        }
        if (y1 != y2) {
            return y1 > y2;
        }
        return c1 > c2;
    }

    static DPRes runDP(List<Monster> mons, int w, int E, int lam) {
        final int NEG = -1_000_000_000;
        int[] dpAdj = new int[E + 1];
        int[] dpY = new int[E + 1];
        int[] dpC = new int[E + 1];
        Arrays.fill(dpAdj, NEG);
        dpAdj[0] = 0;

        for (Monster m : mons) {
            for (int e = E - m.z; e >= 0; e--) {
                if (dpAdj[e] == NEG) {
                    continue;
                }
                if (w + dpY[e] < m.x) {
                    continue;
                }
                int ne = e + m.z;
                int nadj = dpAdj[e] + m.y - lam;
                int ny = dpY[e] + m.y;
                int nc = dpC[e] + 1;
                if (better(nadj, ny, nc, dpAdj[ne], dpY[ne], dpC[ne])) {
                    dpAdj[ne] = nadj;
                    dpY[ne] = ny;
                    dpC[ne] = nc;
                }
            }
        }

        int bestAdj = NEG, bestY = 0, bestC = 0;
        for (int e = 0; e <= E; e++) {
            if (dpAdj[e] == NEG) {
                continue;
            }
            if (better(dpAdj[e], dpY[e], dpC[e], bestAdj, bestY, bestC)) {
                bestAdj = dpAdj[e];
                bestY = dpY[e];
                bestC = dpC[e];
            }
        }
        return new DPRes(dpAdj, dpY, dpC, bestAdj, bestY, bestC);
    }

    static int[] solve(int n, int w, int E, int K, int[] X, int[] Y, int[] Z) {
        List<Monster> mons = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            mons.add(new Monster(X[i], Y[i], Z[i]));
        }
        mons.sort(Comparator.comparingInt(a -> a.x));

        DPRes r0 = runDP(mons, w, E, 0);
        if (r0.bestC <= K) {
            int bestY = 0, bestC = 0;
            for (int e = 0; e <= E; e++) {
                if (r0.adj[e] == -1_000_000_000) {
                    continue;
                }
                if (r0.c[e] <= K) {
                    if (r0.y[e] > bestY || (r0.y[e] == bestY && r0.c[e] > bestC)) {
                        bestY = r0.y[e];
                        bestC = r0.c[e];
                    }
                }
            }
            return new int[]{w + bestY, bestC};
        }

        int lo = 0, hi = 1001;
        while (lo < hi) {
            int mid = (lo + hi + 1) >>> 1;
            DPRes rm = runDP(mons, w, E, mid);
            if (rm.bestC >= K) {
                lo = mid;
            } else {
                hi = mid - 1;
            }
        }

        int lam = lo;
        DPRes r = runDP(mons, w, E, lam);
        int OPT = r.bestAdj + lam * K;

        int bestY = 0, bestC = 0;
        for (int e = 0; e <= E; e++) {
            if (r.adj[e] == -1_000_000_000) {
                continue;
            }
            if (r.c[e] <= K && r.adj[e] + lam * K == OPT) {
                if (r.y[e] > bestY || (r.y[e] == bestY && r.c[e] > bestC)) {
                    bestY = r.y[e];
                    bestC = r.c[e];
                }
            }
        }
        return new int[]{w + bestY, bestC};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), w = sc.nextInt(), E = sc.nextInt(), K = sc.nextInt();
        int[] X = new int[n], Y = new int[n], Z = new int[n];
        for (int i = 0; i < n; i++) {
            X[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            Y[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            Z[i] = sc.nextInt();
        }
        int[] ans = solve(n, w, E, K, X, Y, Z);
        System.out.println(ans[0] + " " + ans[1]);
        sc.close();
    }
}
```
