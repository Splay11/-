## 解题思路

这道题本质上是一个带权无向图的多次最短路查询问题。

已知：

* 计算单元之间有些存在双向通道，所以图是无向图
* 每条通道有一个时延，路径总时延等于经过边权之和
* 需要查询 $K$ 对点之间的最小时延

因为题目会给出多组点对查询，如果每次查询都单独跑一次最短路，总代价可能较高。
而数据范围是 $N \le 100$，所以可以直接使用 Floyd 算法预处理任意两点之间的最短路。

Floyd 算法的核心思想是：

设 $dist[i][j]$ 表示节点 $i$ 到节点 $j$ 的当前最小时延。
依次枚举中转点 $k$，尝试用路径 $i \rightarrow k \rightarrow j$ 来更新 $dist[i][j]$：

$dist[i][j] = \min(dist[i][j],\ dist[i][k] + dist[k][j])$

这样处理完成后，$dist[i][j]$ 就是任意两点之间的最小时延。

实现方法如下：

1. 建立邻接矩阵 $dist$
2. 初始化：

   * 自己到自己时延为 $0$
   * 其他点先设为一个很大值，表示不可达
3. 读入每条双向边，更新：

   * $dist[a][b] = dist[b][a] = \min(dist[a][b], c)$
4. 运行 Floyd 算法
5. 对每个查询：

   * 若不可达，输出 $0$
   * 否则输出最短时延

说明：题面中查询点编号写的是 $0 \le i,j \le N$，为兼容这种写法，代码中数组统一开到 $N+1$。

## 复杂度分析

Floyd 算法共有三重循环，时间复杂度为：

$O(N^3)$

邻接矩阵需要存储任意两点之间的距离，空间复杂度为：

$O(N^2)$

在 $N \le 100$ 的范围内，这个复杂度完全可以接受。

## 代码实现

### Python

```python
def floyd(n, dist):
    # Floyd算法，求任意两点最短路
    for k in range(n + 1):
        for i in range(n + 1):
            for j in range(n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]


def main():
    import sys

    data = sys.stdin.read().split()
    if not data:
        return

    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    INF = 10**18

    # 为兼容题面下标，数组开到 n+1
    dist = [[INF] * (n + 1) for _ in range(n + 1)]

    # 自己到自己距离为0
    for i in range(n + 1):
        dist[i][i] = 0

    # 读入边，注意是双向边，且可能有重边
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        if c < dist[a][b]:
            dist[a][b] = c
            dist[b][a] = c

    # 预处理最短路
    floyd(n, dist)

    k = int(next(it))
    ans = []

    # 输出每组查询结果
    for _ in range(k):
        i = int(next(it))
        j = int(next(it))
        if dist[i][j] == INF:
            ans.append("0")
        else:
            ans.append(str(dist[i][j]))

    sys.stdout.write("\n".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    // Floyd算法，求任意两点最短路
    public static void floyd(int n, long[][] dist) {
        for (int k = 0; k <= n; k++) {
            for (int i = 0; i <= n; i++) {
                for (int j = 0; j <= n; j++) {
                    if (dist[i][k] + dist[k][j] < dist[i][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                    }
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        if (!sc.hasNextInt()) {
            sc.close();
            return;
        }

        int n = sc.nextInt();
        int m = sc.nextInt();

        long INF = (long) 1e18;

        // 为兼容题面下标，数组开到 n+1
        long[][] dist = new long[n + 1][n + 1];

        // 初始化距离数组
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                if (i == j) {
                    dist[i][j] = 0;
                } else {
                    dist[i][j] = INF;
                }
            }
        }

        // 读入边，双向边，可能有重边
        for (int t = 0; t < m; t++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            if (c < dist[a][b]) {
                dist[a][b] = c;
                dist[b][a] = c;
            }
        }

        // 预处理最短路
        floyd(n, dist);

        int k = sc.nextInt();
        StringBuilder sb = new StringBuilder();

        // 处理查询
        for (int t = 0; t < k; t++) {
            int i = sc.nextInt();
            int j = sc.nextInt();
            if (dist[i][j] == INF) {
                sb.append(0);
            } else {
                sb.append(dist[i][j]);
            }
            if (t != k - 1) {
                sb.append('\n');
            }
        }

        System.out.print(sb.toString());
        sc.close();
    }
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

// Floyd算法，求任意两点最短路
void floyd(int n, vector<vector<long long>>& dist) {
    for (int k = 0; k <= n; k++) {
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                if (dist[i][k] + dist[k][j] < dist[i][j]) {
                    dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }
}

int main() {
    int n, m;
    cin >> n >> m;

    const long long INF = (long long)1e18;

    // 为兼容题面下标，数组开到 n+1
    vector<vector<long long>> dist(n + 1, vector<long long>(n + 1, INF));

    // 自己到自己距离为0
    for (int i = 0; i <= n; i++) {
        dist[i][i] = 0;
    }

    // 读入边，双向边，可能有重边
    for (int t = 0; t < m; t++) {
        int a, b, c;
        cin >> a >> b >> c;
        if (c < dist[a][b]) {
            dist[a][b] = c;
            dist[b][a] = c;
        }
    }

    // 预处理最短路
    floyd(n, dist);

    int k;
    cin >> k;

    // 处理查询
    for (int t = 0; t < k; t++) {
        int i, j;
        cin >> i >> j;
        if (dist[i][j] == INF) {
            cout << 0;
        } else {
            cout << dist[i][j];
        }
        if (t != k - 1) {
            cout << '\n';
        }
    }

    return 0;
}
```
