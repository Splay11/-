## 解题思路

题目中一次优化只会影响一个模块，并且总耗时等于所有模块耗时之和，所以可以把问题看成：

* 初始总和为 $\sum a_i$
* 每次操作选择一个模块，让总和尽量减少
* 做 $m$ 次后，总和最小

对于某个模块，若当前耗时为 $t$，下限为 $b_i$，那么一次优化后变成：

$
t'=\max\left(\left\lceil \dfrac{t}{2}\right\rceil,b_i\right)
$

这次操作带来的收益是：

$
\Delta=t-t'
$

显然，每一天都应该优先选择“当前能让总耗时下降最多”的那个模块。
因为总和是否最小，只取决于每一步减少了多少，而每次操作之间彼此独立，所以这是一个典型的 贪心 + 优先队列 问题。

具体做法：

1. 先计算初始总耗时 $sum$
2. 对每个模块计算“当前做一次优化的收益” $\Delta$
3. 把所有 $\Delta>0$ 的模块放入大根堆
4. 重复最多 $m$ 次：

   * 取出收益最大的模块
   * 用这次收益去减少 $sum$
   * 更新该模块的新耗时
   * 重新计算这个模块下一次优化的收益
   * 若还能继续优化，就再放回堆中
5. 堆空了说明所有模块都已到达下限，提前结束

这里用到的相关算法是：

* 贪心：每次都选当前收益最大的模块
* 优先队列（堆）：高效维护当前最大收益

核心实现方法就是用堆动态维护每个模块“下一次优化能减少多少耗时”。

## 复杂度分析

设实际执行了 $k$ 次优化，显然 $k \le m$。

* 建堆复杂度为 $O(n)$
* 每次取堆顶、更新并重新入堆的复杂度为 $O(\log n)$
* 总时间复杂度为 $O(n + k \log n)$，最坏为 $O(n + m \log n)$
* 空间复杂度为 $O(n)$

在数据范围 $n,m \le 10^3$ 下，这个复杂度完全合适。

## 代码实现

### Python

```python
import heapq
import sys


# 计算某个模块优化一次后的新耗时
def next_value(t, b):
    # 向上取整除以2，再和下限取最大值
    return max((t + 1) // 2, b)


# 计算m天后的最小总耗时
def solve(n, m, a, b):
    total = sum(a)

    # Python默认是小根堆，这里存负数模拟大根堆
    heap = []

    for i in range(n):
        nt = next_value(a[i], b[i])
        gain = a[i] - nt  # 本次优化收益
        if gain > 0:
            heapq.heappush(heap, (-gain, i))

    # 最多进行m次优化
    for _ in range(m):
        if not heap:
            break

        gain, idx = heapq.heappop(heap)
        gain = -gain

        # 减少总耗时
        total -= gain

        # 更新该模块当前耗时
        a[idx] = next_value(a[idx], b[idx])

        # 重新计算下一次优化收益
        nt = next_value(a[idx], b[idx])
        new_gain = a[idx] - nt
        if new_gain > 0:
            heapq.heappush(heap, (-new_gain, idx))

    return total


def main():
    data = list(map(int, sys.stdin.read().split()))
    n, m = data[0], data[1]
    a = data[2:2 + n]
    b = data[2 + n:2 + 2 * n]

    print(solve(n, m, a, b))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

// ACM风格，类名必须为Main
public class Main {

    // 结点：记录当前一次优化的收益和模块下标
    static class Node {
        int gain;
        int idx;

        Node(int gain, int idx) {
            this.gain = gain;
            this.idx = idx;
        }
    }

    // 计算某个模块优化一次后的新耗时
    static int nextValue(int t, int b) {
        // 向上取整除以2，再和下限取最大值
        return Math.max((t + 1) / 2, b);
    }

    // 计算m天后的最小总耗时
    static long solve(int n, int m, int[] a, int[] b) {
        long total = 0;
        for (int x : a) {
            total += x;
        }

        // 大根堆：收益大的优先
        PriorityQueue<Node> pq = new PriorityQueue<>((x, y) -> y.gain - x.gain);

        for (int i = 0; i < n; i++) {
            int nt = nextValue(a[i], b[i]);
            int gain = a[i] - nt; // 本次优化收益
            if (gain > 0) {
                pq.offer(new Node(gain, i));
            }
        }

        // 最多进行m次优化
        for (int day = 0; day < m; day++) {
            if (pq.isEmpty()) {
                break;
            }

            Node cur = pq.poll();
            int idx = cur.idx;
            int gain = cur.gain;

            // 减少总耗时
            total -= gain;

            // 更新该模块当前耗时
            a[idx] = nextValue(a[idx], b[idx]);

            // 重新计算下一次优化收益
            int nt = nextValue(a[idx], b[idx]);
            int newGain = a[idx] - nt;
            if (newGain > 0) {
                pq.offer(new Node(newGain, idx));
            }
        }

        return total;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] a = new int[n];
        int[] b = new int[n];

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }

        System.out.println(solve(n, m, a, b));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

// 结点：记录当前一次优化的收益和模块下标
struct Node {
    int gain;
    int idx;

    // 大根堆，收益大的优先
    bool operator<(const Node& other) const {
        return gain < other.gain;
    }
};

// 计算某个模块优化一次后的新耗时
int nextValue(int t, int b) {
    // 向上取整除以2，再和下限取最大值
    return max((t + 1) / 2, b);
}

// 计算m天后的最小总耗时
long long solve(int n, int m, vector<int>& a, vector<int>& b) {
    long long total = 0;
    for (int x : a) {
        total += x;
    }

    priority_queue<Node> pq;

    for (int i = 0; i < n; i++) {
        int nt = nextValue(a[i], b[i]);
        int gain = a[i] - nt; // 本次优化收益
        if (gain > 0) {
            pq.push({gain, i});
        }
    }

    // 最多进行m次优化
    for (int day = 0; day < m; day++) {
        if (pq.empty()) {
            break;
        }

        Node cur = pq.top();
        pq.pop();

        int idx = cur.idx;
        int gain = cur.gain;

        // 减少总耗时
        total -= gain;

        // 更新该模块当前耗时
        a[idx] = nextValue(a[idx], b[idx]);

        // 重新计算下一次优化收益
        int nt = nextValue(a[idx], b[idx]);
        int newGain = a[idx] - nt;
        if (newGain > 0) {
            pq.push({newGain, idx});
        }
    }

    return total;
}

int main() {
    int n, m;
    cin >> n >> m;

    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i];
    }

    cout << solve(n, m, a, b) << '\n';
    return 0;
}
```
