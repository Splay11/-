## 解题思路

题目中 $N \le 15$，数据范围很小，但每个任务有两种资源约束：$CPU$ 和内存。
这类题最适合用 状态压缩动态规划。

先把每个任务看成一个二进制位，用一个状态 $mask$ 表示“选了哪些任务”。

### 1. 预处理单台服务器能否运行一个任务集合

对于每个状态 $mask$：

* 统计其中所有任务的 $CPU$ 总和
* 统计其中所有任务的内存总和
* 统计其中所有任务的价值总和

若满足：

* $CPU$ 总和 $\le C$
* 内存总和 $\le M$

那么这个集合就可以放在一台服务器上运行，记为可行。

### 2. 状态定义

设 $dp[mask]$ 表示：运行任务集合 $mask$ 至少需要多少台服务器。

初始值：

* $dp[0] = 0$
* 其余初始化为很大值

### 3. 状态转移

如果某个子集 $sub \subseteq mask$ 可以由一台服务器运行，
那么可以把这台服务器分配给 $sub$，剩下的任务是 $mask \setminus sub$。

于是有转移：

$$
dp[mask] = \min(dp[mask],\ dp[mask \setminus sub] + 1)
$$

其中 $sub$ 需要满足“单台服务器可运行”。

这样枚举 $mask$ 的所有子集即可。

### 4. 如何得到每个 $K$ 的答案

对于每个任务集合 $mask$，如果 $dp[mask] \le K$，说明用 $K$ 台服务器可以运行它，
那么它的总价值就可以参与答案更新。

于是：

* 先求出每个 $mask$ 的总价值 $value[mask]$
* 再统计所有满足 $dp[mask] \le K$ 的集合的最大价值

由于如果某个集合能用 $K$ 台跑，也一定能用 $K+1$ 台跑，
所以答案天然单调不减。

### 5. 相关算法

这里使用的是：

* 状态压缩
* 子集枚举
* 动态规划

这是本题最简洁且正确的做法。

## 复杂度分析

设总状态数为 $2^N$。

### 时间复杂度

* 预处理每个状态的信息：$O(N \cdot 2^N)$
* 动态规划时，每个状态要枚举所有子集，复杂度为 $O(3^N)$

总时间复杂度：

$$
O(N \cdot 2^N + 3^N)
$$

当 $N \le 15$ 时完全可行。

### 空间复杂度

需要存储：

* 每个状态的总价值
* 每个状态是否可单机运行
* 每个状态的最少服务器数

空间复杂度为：

$$
O(2^N)
$$

## 代码实现

### Python

```python
# 分布式任务调度
# 状态压缩动态规划

def solve(n, C, M, tasks):
    size = 1 << n

    # value[mask]：状态 mask 的总价值
    value = [0] * size
    # ok[mask]：状态 mask 是否可以放在一台服务器上
    ok = [False] * size

    # 预处理每个状态的 CPU、内存、价值
    for mask in range(size):
        cpu_sum = 0
        mem_sum = 0
        val_sum = 0
        for i in range(n):
            if (mask >> i) & 1:
                cpu_sum += tasks[i][0]
                mem_sum += tasks[i][1]
                val_sum += tasks[i][2]
        value[mask] = val_sum
        if cpu_sum <= C and mem_sum <= M:
            ok[mask] = True

    INF = n + 1
    # dp[mask]：运行 mask 中任务最少需要多少台服务器
    dp = [INF] * size
    dp[0] = 0

    # 枚举所有状态
    for mask in range(1, size):
        sub = mask
        # 枚举 mask 的所有子集
        while sub:
            # 如果 sub 可以放在一台服务器上
            if ok[sub]:
                rest = mask ^ sub
                if dp[rest] + 1 < dp[mask]:
                    dp[mask] = dp[rest] + 1
            sub = (sub - 1) & mask

    # ans[k]：恰好输出 k 台服务器时的最大总价值，下标从 1 到 n
    ans = [0] * (n + 1)

    # 统计每个 k 的答案
    for mask in range(size):
        need = dp[mask]
        if need <= n:
            if value[mask] > ans[need]:
                ans[need] = value[mask]

    # 前缀最大值：因为 k 台能跑的，k+1 台也能跑
    for k in range(1, n + 1):
        if ans[k - 1] > ans[k]:
            ans[k] = ans[k - 1]

    return ans[1:]


def main():
    n, C, M = map(int, input().split())
    tasks = []
    for _ in range(n):
        c, m, v = map(int, input().split())
        tasks.append((c, m, v))

    res = solve(n, C, M, tasks)
    for x in res:
        print(x)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    // 求每个 K 的最大总价值
    public static int[] solve(int n, int C, int M, int[][] tasks) {
        int size = 1 << n;

        // value[mask]：状态 mask 的总价值
        int[] value = new int[size];
        // ok[mask]：状态 mask 是否可以放在一台服务器上
        boolean[] ok = new boolean[size];

        // 预处理每个状态的 CPU、内存、价值
        for (int mask = 0; mask < size; mask++) {
            int cpuSum = 0;
            int memSum = 0;
            int valSum = 0;
            for (int i = 0; i < n; i++) {
                if (((mask >> i) & 1) == 1) {
                    cpuSum += tasks[i][0];
                    memSum += tasks[i][1];
                    valSum += tasks[i][2];
                }
            }
            value[mask] = valSum;
            if (cpuSum <= C && memSum <= M) {
                ok[mask] = true;
            }
        }

        int INF = n + 1;
        // dp[mask]：运行 mask 中任务最少需要多少台服务器
        int[] dp = new int[size];
        for (int i = 0; i < size; i++) {
            dp[i] = INF;
        }
        dp[0] = 0;

        // 状态转移
        for (int mask = 1; mask < size; mask++) {
            int sub = mask;
            // 枚举 mask 的所有子集
            while (sub > 0) {
                // 如果 sub 可以放在一台服务器上
                if (ok[sub]) {
                    int rest = mask ^ sub;
                    dp[mask] = Math.min(dp[mask], dp[rest] + 1);
                }
                sub = (sub - 1) & mask;
            }
        }

        // ans[k]：使用不超过 k 台服务器时的最大总价值
        int[] ans = new int[n + 1];

        for (int mask = 0; mask < size; mask++) {
            int need = dp[mask];
            if (need <= n) {
                ans[need] = Math.max(ans[need], value[mask]);
            }
        }

        // 前缀最大值
        for (int k = 1; k <= n; k++) {
            ans[k] = Math.max(ans[k], ans[k - 1]);
        }

        int[] res = new int[n];
        for (int k = 1; k <= n; k++) {
            res[k - 1] = ans[k];
        }
        return res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int C = sc.nextInt();
        int M = sc.nextInt();

        int[][] tasks = new int[n][3];
        for (int i = 0; i < n; i++) {
            tasks[i][0] = sc.nextInt(); // CPU 需求
            tasks[i][1] = sc.nextInt(); // 内存需求
            tasks[i][2] = sc.nextInt(); // 任务价值
        }

        int[] ans = solve(n, C, M, tasks);
        for (int x : ans) {
            System.out.println(x);
        }

        sc.close();
    }
}
```

```cpp
// 求每个 K 的最大总价值
vector<int> solve(int n, int C, int M, const vector<vector<int>>& tasks) {
    int size = 1 << n;

    // value[mask]：状态 mask 的总价值
    vector<int> value(size, 0);
    // ok[mask]：状态 mask 是否可以放在一台服务器上
    vector<bool> ok(size, false);

    // 预处理每个状态的 CPU、内存、价值
    for (int mask = 0; mask < size; mask++) {
        int cpuSum = 0;
        int memSum = 0;
        int valSum = 0;
        for (int i = 0; i < n; i++) {
            if ((mask >> i) & 1) {
                cpuSum += tasks[i][0];
                memSum += tasks[i][1];
                valSum += tasks[i][2];
            }
        }
        value[mask] = valSum;
        if (cpuSum <= C && memSum <= M) {
            ok[mask] = true;
        }
    }

    int INF = n + 1;
    // dp[mask]：运行 mask 中任务最少需要多少台服务器
    vector<int> dp(size, INF);
    dp[0] = 0;

    // 状态转移
    for (int mask = 1; mask < size; mask++) {
        int sub = mask;
        // 枚举 mask 的所有子集
        while (sub) {
            // 如果 sub 可以放在一台服务器上
            if (ok[sub]) {
                int rest = mask ^ sub;
                dp[mask] = min(dp[mask], dp[rest] + 1);
            }
            sub = (sub - 1) & mask;
        }
    }

    // ans[k]：使用不超过 k 台服务器时的最大总价值
    vector<int> ans(n + 1, 0);

    for (int mask = 0; mask < size; mask++) {
        int need = dp[mask];
        if (need <= n) {
            ans[need] = max(ans[need], value[mask]);
        }
    }

    // 前缀最大值
    for (int k = 1; k <= n; k++) {
        ans[k] = max(ans[k], ans[k - 1]);
    }

    vector<int> res;
    for (int k = 1; k <= n; k++) {
        res.push_back(ans[k]);
    }
    return res;
}

int main() {
    int n, C, M;
    cin >> n >> C >> M;

    vector<vector<int>> tasks(n, vector<int>(3));
    for (int i = 0; i < n; i++) {
        cin >> tasks[i][0] >> tasks[i][1] >> tasks[i][2];
    }

    vector<int> ans = solve(n, C, M, tasks);
```
