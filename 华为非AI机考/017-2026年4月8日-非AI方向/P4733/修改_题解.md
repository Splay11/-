# DFS暴力解法

这道题的正解是 $状压DP$，但是我们用朴素解法 $DFS/回溯分配任务$ 也能在考试时拿到一定的分数。

有 $N$ 个任务，每台服务器容量为 $(C,M)$。对服务器台数 $K=1\sim N$，分别求能跑下的任务最大总价值。

朴素思路：

1. 先枚举「要跑哪些任务」的子集（或在搜索里决定跑/不跑）。
2. 再用 $DFS$ 把已选任务依次装进服务器：尝试放进已有某台服务器（$CPU$、内存都够），或新开一台服务器；若使用的服务器数不超过当前 $K$，则用总价值更新答案。
3. 对每个 $K$ 都求一遍最大值。也可以一次搜索里按「最多用了几台服务器」分别更新。

任务少时能算出正确结果；当 $N$ 接近 $15$ 时，子集与装箱搜索都会很慢，需要后面的状压正解。

## DFS解法

```python
def solve(n, C, M, tasks):
    ans = [0] * (n + 1)
    # rem_cpu / rem_mem：每台服务器剩余容量
    rem_cpu = [0] * n
    rem_mem = [0] * n

    def can_pack(chosen, k):
        # 把 chosen 里的任务装进至多 k 台服务器
        m = len(chosen)
        used = [0]

        def dfs(idx):
            if idx == m:
                return True
            ti = chosen[idx]
            need_c, need_m, _ = tasks[ti]
            # 尝试放入已有服务器
            for s in range(used[0]):
                if rem_cpu[s] >= need_c and rem_mem[s] >= need_m:
                    rem_cpu[s] -= need_c
                    rem_mem[s] -= need_m
                    if dfs(idx + 1):
                        rem_cpu[s] += need_c
                        rem_mem[s] += need_m
                        return True
                    rem_cpu[s] += need_c
                    rem_mem[s] += need_m
            # 新开一台
            if used[0] < k:
                s = used[0]
                rem_cpu[s] = C - need_c
                rem_mem[s] = M - need_m
                used[0] += 1
                if dfs(idx + 1):
                    used[0] -= 1
                    return True
                used[0] -= 1
            return False

        return dfs(0)

    def dfs_choose(i, chosen, value):
        # 枚举每个任务选或不选
        if i == n:
            for k in range(1, n + 1):
                if value > ans[k] and can_pack(chosen, k):
                    ans[k] = value
            return
        # 不选
        dfs_choose(i + 1, chosen, value)
        # 选
        chosen.append(i)
        dfs_choose(i + 1, chosen, value + tasks[i][2])
        chosen.pop()

    dfs_choose(0, [], 0)
    return ans[1:]


def main():
    n, C, M = map(int, input().split())
    tasks = []
    for _ in range(n):
        c, m, v = map(int, input().split())
        tasks.append((c, m, v))
    for x in solve(n, C, M, tasks):
        print(x)


if __name__ == "__main__":
    main()
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

int n, C, M;
vector<vector<int>> tasks;
vector<int> ans;
vector<int> rem_cpu, rem_mem;

bool dfs_pack(int idx, const vector<int>& chosen, int k, int& used) {
    int m = (int)chosen.size();
    if (idx == m) {
        return true;
    }
    int ti = chosen[idx];
    int need_c = tasks[ti][0];
    int need_m = tasks[ti][1];
    // 放入已有服务器
    for (int s = 0; s < used; s++) {
        if (rem_cpu[s] >= need_c && rem_mem[s] >= need_m) {
            rem_cpu[s] -= need_c;
            rem_mem[s] -= need_m;
            if (dfs_pack(idx + 1, chosen, k, used)) {
                rem_cpu[s] += need_c;
                rem_mem[s] += need_m;
                return true;
            }
            rem_cpu[s] += need_c;
            rem_mem[s] += need_m;
        }
    }
    // 新开一台
    if (used < k) {
        int s = used;
        rem_cpu[s] = C - need_c;
        rem_mem[s] = M - need_m;
        used++;
        if (dfs_pack(idx + 1, chosen, k, used)) {
            used--;
            return true;
        }
        used--;
    }
    return false;
}

bool can_pack(const vector<int>& chosen, int k) {
    int used = 0;
    return dfs_pack(0, chosen, k, used);
}

// 枚举每个任务选或不选
void dfs_choose(int i, vector<int>& chosen, int value) {
    if (i == n) {
        for (int k = 1; k <= n; k++) {
            if (value > ans[k] && can_pack(chosen, k)) {
                ans[k] = value;
            }
        }
        return;
    }
    dfs_choose(i + 1, chosen, value);
    chosen.push_back(i);
    dfs_choose(i + 1, chosen, value + tasks[i][2]);
    chosen.pop_back();
}

vector<int> solve() {
    ans.assign(n + 1, 0);
    rem_cpu.assign(n, 0);
    rem_mem.assign(n, 0);
    vector<int> chosen;
    dfs_choose(0, chosen, 0);
    vector<int> res;
    for (int k = 1; k <= n; k++) {
        res.push_back(ans[k]);
    }
    return res;
}

int main() {
    cin >> n >> C >> M;
    tasks.assign(n, vector<int>(3));
    for (int i = 0; i < n; i++) {
        cin >> tasks[i][0] >> tasks[i][1] >> tasks[i][2];
```

```java
                    remMem[s] += needM;
                    return true;

                remCpu[s] += needC;
                remMem[s] += needM;

            remCpu[s] = C - needC;
            remMem[s] = M - needM;

            if (dfsPack(idx + 1, m, chosen, k, used)) {
                used[0]--;
                return true;

            used[0]--;

        return false;
```

# 正确解法

## 题目分析

$N\le 15$，用 $状压DP$ 更合适。

1. 用 $mask$ 表示任务集合。预处理每个 $mask$ 的总价值 $value[mask]$，以及它能否放进一台服务器（$CPU$、内存都不超），记为 $ok[mask]$。
2. 定义 $dp[mask]$：跑完集合 $mask$ 至少需要几台服务器。
3. 枚举 $mask$ 的子集 $sub$：若 $ok[sub]$，则 $dp[mask]=\min(dp[mask], dp[mask\oplus sub]+1)$。
4. 对每个 $mask$，若 $dp[mask]\le K$，可用 $value[mask]$ 更新「$K$ 台服务器」的答案；再对 $K$ 做前缀最大值，因为台数更多不会更差。

## 复杂度分析

* 时间复杂度：$O(N\cdot 2^N + 3^N)$。
* 空间复杂度：$O(2^N)$。

## 代码实现

### Python

```python
def solve(n, C, M, tasks):
    size = 1 << n
    value = [0] * size
    ok = [False] * size

    # 预处理每个子集的价值与是否单机可跑
    for mask in range(size):
        cpu_sum = mem_sum = val_sum = 0
        for i in range(n):
            if (mask >> i) & 1:
                cpu_sum += tasks[i][0]
                mem_sum += tasks[i][1]
                val_sum += tasks[i][2]
        value[mask] = val_sum
        if cpu_sum <= C and mem_sum <= M:
            ok[mask] = True

    INF = n + 1
    # dp[mask]：跑完 mask 最少服务器数
    dp = [INF] * size
    dp[0] = 0

    for mask in range(1, size):
        sub = mask
        while sub:
            if ok[sub]:
                rest = mask ^ sub
                if dp[rest] + 1 < dp[mask]:
                    dp[mask] = dp[rest] + 1
            sub = (sub - 1) & mask

    ans = [0] * (n + 1)
    for mask in range(size):
        need = dp[mask]
        if need <= n and value[mask] > ans[need]:
            ans[need] = value[mask]

    # 台数越多，答案只会不降
    for k in range(1, n + 1):
        if ans[k - 1] > ans[k]:
            ans[k] = ans[k - 1]

    return ans[1:]


def main():
    n, C, M = map(int, input().split())
    tasks = []
    for _ in range(n):
        c, m, v = map(int, input().split())
        tasks.append((c, m, v))
    for x in solve(n, C, M, tasks):
        print(x)


if __name__ == "__main__":
    main()
```

```java
import java.util.Scanner;

public class Main {

    public static int[] solve(int n, int C, int M, int[][] tasks) {
        int size = 1 << n;
        int[] value = new int[size];
        boolean[] ok = new boolean[size];

        // 预处理每个子集的价值与是否单机可跑
        for (int mask = 0; mask < size; mask++) {
            int cpuSum = 0, memSum = 0, valSum = 0;
            for (int i = 0; i < n; i++) {
                if (((mask >> i) & 1) == 1) {
                    cpuSum += tasks[i][0];
                    memSum += tasks[i][1];
                    valSum += tasks[i][2];
                }
            }
            value[mask] = valSum;
            if (cpuSum <= C && memSum <= M) {
                ok[mask] = true;
            }
        }

        int INF = n + 1;
        // dp[mask]：跑完 mask 最少服务器数
        int[] dp = new int[size];
        for (int i = 0; i < size; i++) {
            dp[i] = INF;
        }
        dp[0] = 0;

        for (int mask = 1; mask < size; mask++) {
            int sub = mask;
            while (sub > 0) {
                if (ok[sub]) {
                    int rest = mask ^ sub;
                    dp[mask] = Math.min(dp[mask], dp[rest] + 1);
                }
                sub = (sub - 1) & mask;
            }
        }

        int[] ans = new int[n + 1];
        for (int mask = 0; mask < size; mask++) {
            int need = dp[mask];
            if (need <= n) {
                ans[need] = Math.max(ans[need], value[mask]);
            }
        }

        // 台数越多，答案只会不降
        for (int k = 1; k <= n; k++) {
            ans[k] = Math.max(ans[k], ans[k - 1]);
        }

        int[] res = new int[n];
        for (int k = 1; k <= n; k++) {
            res[k - 1] = ans[k];
        }
        return res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int C = sc.nextInt();
        int M = sc.nextInt();
        int[][] tasks = new int[n][3];
        for (int i = 0; i < n; i++) {
            tasks[i][0] = sc.nextInt();
            tasks[i][1] = sc.nextInt();
            tasks[i][2] = sc.nextInt();
        }
        for (int x : solve(n, C, M, tasks)) {
            System.out.println(x);
        }
        sc.close();
    }
}
```
