## 解题思路

使用双指针算法。

对于挡板 $i$ 与挡板 $i+1$ 之间的槽位，其水面高度为：

$$
\min(\text{左侧最高挡板},\text{右侧最高挡板})
$$

设置左右指针 $left$、$right$，并维护：

* $leftMax$：从第 $0$ 块挡板到第 $left$ 块挡板的最大高度。
* $rightMax$：从第 $right$ 块挡板到第 $n-1$ 块挡板的最大高度。

当 $leftMax \leq rightMax$ 时，当前左侧槽位的右边一定存在高度不小于 $leftMax$ 的挡板，因此该槽位的水面高度就是 $leftMax$，随后右移 $left$。

当 $leftMax > rightMax$ 时，当前右侧槽位的左边一定存在高度不小于 $rightMax$ 的挡板，因此该槽位的水面高度就是 $rightMax$，随后左移 $right$。

每次处理一个槽位，直到全部 $n-1$ 个槽位处理完成。

## 复杂度分析

时间复杂度为 $O(n)$，每个槽位只会被处理一次。

空间复杂度为 $O(1)$，只使用常数个额外变量。

## 代码实现

### Python

```python
def total_water(h):
    n = len(h)
    if n < 2:
        return 0

    left = 0
    right = n - 1
    left_max = h[left]
    right_max = h[right]
    answer = 0

    while left < right:
        if left_max <= right_max:
            # 当前左侧槽位的水面高度由 left_max 决定
            answer += left_max
            left += 1
            left_max = max(left_max, h[left])
        else:
            # 当前右侧槽位的水面高度由 right_max 决定
            answer += right_max
            right -= 1
            right_max = max(right_max, h[right])

    return answer


def main():
    n = int(input())
    h = list(map(int, input().split()))

    print(total_water(h))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    public static long totalWater(int[] h) {
        int n = h.length;
        if (n < 2) {
            return 0;
        }

        int left = 0;
        int right = n - 1;
        int leftMax = h[left];
        int rightMax = h[right];
        long answer = 0;

        while (left < right) {
            if (leftMax <= rightMax) {
                // 当前左侧槽位的水面高度由 leftMax 决定
                answer += leftMax;
                left++;

                if (h[left] > leftMax) {
                    leftMax = h[left];
                }
            } else {
                // 当前右侧槽位的水面高度由 rightMax 决定
                answer += rightMax;
                right--;

                if (h[right] > rightMax) {
                    rightMax = h[right];
                }
            }
        }

        return answer;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] h = new int[n];

        for (int i = 0; i < n; i++) {
            h[i] = scanner.nextInt();
        }

        System.out.println(totalWater(h));
        scanner.close();
    }
}
```

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

long long totalWater(const vector<int>& h) {
    int n = h.size();
    if (n < 2) {
        return 0;
    }

    int left = 0;
    int right = n - 1;
    int leftMax = h[left];
    int rightMax = h[right];
    long long answer = 0;

    while (left < right) {
        if (leftMax <= rightMax) {
            // 当前左侧槽位的水面高度由 leftMax 决定
            answer += leftMax;
            left++;
            leftMax = max(leftMax, h[left]);
        } else {
            // 当前右侧槽位的水面高度由 rightMax 决定
            answer += rightMax;
            right--;
            rightMax = max(rightMax, h[right]);
        }
    }

    return answer;
}

int main() {
    int n;
    cin >> n;

    vector<int> h(n);
    for (int i = 0; i < n; i++) {
        cin >> h[i];
    }

    cout << totalWater(h) << '\n';

    return 0;
}
