# DFS暴力解法

这道题的正解是 $状压DP$，但是我们用朴素解法 $DFS$ 选或不选每个节点配置也能在考试时拿到一定的分数。

每个节点可选「配置优先转发」或「不配置」，最多选 $K$ 个。配置节点 $i$ 会让区间 $[i,\min(N-1,i+win\_size[i])]$ 上的时延各减去 $opt\_delay[i]$（减到 $0$ 为止，可叠加）。求总时延最小值。

朴素做法：

1. 用 $DFS$ 从左到右决定每个节点选或不选；已选个数不能超过 $K$。
2. 全部决定完后，按题意扫描每个节点，累加实际时延：对节点 $j$，减去所有覆盖到它的已选节点的 $opt\_delay$。
3. 用得到的总时延更新答案。

$N$ 很小或 $K$ 很小时能过；$N\le 50$ 且 $K$ 较大时搜索量太大，需要后面的正解。

## DFS解法

```python
INF = 10 ** 30


def calc_delay(n, delay, win_size, opt_delay, choose):
    # 根据选择方案计算总时延
    total = 0
    for j in range(n):
        reduce_sum = 0
        for i in range(n):
            if choose[i] and i <= j <= i + win_size[i]:
                reduce_sum += opt_delay[i]
        cur = delay[j] - reduce_sum
        if cur < 0:
            cur = 0
        total += cur
    return total


def optimize_network(n, k, delay, win_size, opt_delay):
    ans = [INF]
    choose = [0] * n

    def dfs(i, selected):
        # i：当前节点；selected：已选个数
        if selected > k:
            return
        if i == n:
            ans[0] = min(ans[0], calc_delay(n, delay, win_size, opt_delay, choose))
            return
        # 不选
        choose[i] = 0
        dfs(i + 1, selected)
        # 选
        choose[i] = 1
        dfs(i + 1, selected + 1)
        choose[i] = 0

    dfs(0, 0)
    return ans[0]

    delay = list(map(int, input().split()))
    win_size = list(map(int, input().split()))
    opt_delay = list(map(int, input().split()))
    print(optimize_network(n, k, delay, win_size, opt_delay))

    main()
```

# 正确解法

## 题目分析

注意到 $win\_size[i]\le 3$，一个节点的时延最多只受「自己 + 前面最多 $3$ 个已选节点」影响。因此用 $状压DP$ 只保留最近 $3$ 个节点的选择状态即可。

定义 $dp[c][mask]$：已经处理完当前节点之前的部分，选了 $c$ 个优先转发节点，最近 $3$ 个节点的选择状态为 $mask$ 时的最小总时延。

处理节点 $i$ 时枚举选或不选：

* 用 $mask$ 里前驱节点是否覆盖 $i$，加上当前是否自选，得到减免量，算出节点 $i$ 的实际时延。
* 更新新的 $mask'=((mask<<1)\&7)|choose$，以及已选个数 $c+choose$。

扫完所有节点后，在全部 $dp[c][mask]$ 中取最小值。

## 复杂度分析

* 时间复杂度：$O(N\cdot K\cdot 2^3)$。
* 空间复杂度：$O(K\cdot 2^3)$。

## 代码实现

### Python

```python
INF = 10 ** 30


def optimize_network(n, k, delay, win_size, opt_delay):
    # dp[c][mask]：已选 c 个，最近 3 个选择状态为 mask 的最小总时延
    dp = [[INF] * 8 for _ in range(k + 1)]
    dp[0][0] = 0

    for i in range(n):
        new_dp = [[INF] * 8 for _ in range(k + 1)]
        for c in range(k + 1):
            for mask in range(8):
                if dp[c][mask] == INF:
                    continue
                for choose in range(2):
                    if c + choose > k:
                        continue
                    reduce_sum = 0
                    # 检查前面最多 3 个节点是否影响 i
                    for d in range(3):
                        if (mask >> d) & 1:
                            pre = i - 1 - d
                            if pre >= 0 and pre + win_size[pre] >= i:
                                reduce_sum += opt_delay[pre]
                    if choose == 1:
                        reduce_sum += opt_delay[i]
                    cur_delay = delay[i] - reduce_sum
                    if cur_delay < 0:
                        cur_delay = 0
                    new_mask = ((mask << 1) & 7) | choose
                    val = dp[c][mask] + cur_delay
                    if val < new_dp[c + choose][new_mask]:
                        new_dp[c + choose][new_mask] = val
        dp = new_dp

    ans = INF
    for c in range(k + 1):
        for mask in range(8):
            ans = min(ans, dp[c][mask])
    return ans


def main():
    n, k = map(int, input().split())
    delay = list(map(int, input().split()))
    win_size = list(map(int, input().split()))
    opt_delay = list(map(int, input().split()))
    print(optimize_network(n, k, delay, win_size, opt_delay))


if __name__ == "__main__":
    main()
```

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static long optimizeNetwork(int n, int k, int[] delay, int[] winSize, int[] optDelay) {
        long INF = Long.MAX_VALUE / 4;
        // dp[c][mask]：已选 c 个，最近 3 个选择状态为 mask 的最小总时延
        long[][] dp = new long[k + 1][8];
        for (int i = 0; i <= k; i++) {
            Arrays.fill(dp[i], INF);
        }
        dp[0][0] = 0;

        for (int i = 0; i < n; i++) {
            long[][] newDp = new long[k + 1][8];
            for (int c = 0; c <= k; c++) {
                Arrays.fill(newDp[c], INF);
            }
            for (int c = 0; c <= k; c++) {
                for (int mask = 0; mask < 8; mask++) {
                    if (dp[c][mask] == INF) {
                        continue;
                    }
                    for (int choose = 0; choose <= 1; choose++) {
                        if (c + choose > k) {
                            continue;
                        }
                        long reduceSum = 0;
                        // 检查前面最多 3 个节点是否影响 i
                        for (int d = 0; d < 3; d++) {
                            if (((mask >> d) & 1) == 1) {
                                int pre = i - 1 - d;
                                if (pre >= 0 && pre + winSize[pre] >= i) {
                                    reduceSum += optDelay[pre];
                                }
                            }
                        }
                        if (choose == 1) {
                            reduceSum += optDelay[i];
                        }
                        long curDelay = delay[i] - reduceSum;
                        if (curDelay < 0) {
                            curDelay = 0;
                        }
                        int newMask = ((mask << 1) & 7) | choose;
                        long val = dp[c][mask] + curDelay;
                        if (val < newDp[c + choose][newMask]) {
                            newDp[c + choose][newMask] = val;
                        }
                    }
                }
            }
            dp = newDp;
        }

        long ans = INF;
        for (int c = 0; c <= k; c++) {
            for (int mask = 0; mask < 8; mask++) {
                ans = Math.min(ans, dp[c][mask]);
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] delay = new int[n];
        int[] winSize = new int[n];
        int[] optDelay = new int[n];
        for (int i = 0; i < n; i++) {
            delay[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            winSize[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            optDelay[i] = sc.nextInt();
        }
        System.out.println(optimizeNetwork(n, k, delay, winSize, optDelay));
        sc.close();
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

long long optimize_network(int n, int k, vector<int>& delay,
                           vector<int>& win_size, vector<int>& opt_delay) {
    const long long INF = (1LL << 60);
    // dp[c][mask]：已选 c 个，最近 3 个选择状态为 mask 的最小总时延
    vector<vector<long long>> dp(k + 1, vector<long long>(8, INF));
    dp[0][0] = 0;

    for (int i = 0; i < n; i++) {
        vector<vector<long long>> new_dp(k + 1, vector<long long>(8, INF));
        for (int c = 0; c <= k; c++) {
            for (int mask = 0; mask < 8; mask++) {
                if (dp[c][mask] == INF) {
                    continue;
                }
                for (int choose = 0; choose <= 1; choose++) {
                    if (c + choose > k) {
                        continue;
                    }
                    long long reduce_sum = 0;
                    // 检查前面最多 3 个节点是否影响 i
                    for (int d = 0; d < 3; d++) {
                        if ((mask >> d) & 1) {
                            int pre = i - 1 - d;
                            if (pre >= 0 && pre + win_size[pre] >= i) {
                                reduce_sum += opt_delay[pre];
                            }
                        }
                    }
                    if (choose == 1) {
                        reduce_sum += opt_delay[i];
                    }
                    long long cur_delay = delay[i] - reduce_sum;
                    if (cur_delay < 0) {
                        cur_delay = 0;
                    }
                    int new_mask = ((mask << 1) & 7) | choose;
                    new_dp[c + choose][new_mask] =
                        min(new_dp[c + choose][new_mask], dp[c][mask] + cur_delay);
                }
            }
        }
        dp.swap(new_dp);
    }

    long long ans = INF;
    for (int c = 0; c <= k; c++) {
        for (int mask = 0; mask < 8; mask++) {
            ans = min(ans, dp[c][mask]);
        }
    }
    return ans;
}

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> delay(n), win_size(n), opt_delay(n);
    for (int i = 0; i < n; i++) {
        cin >> delay[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> win_size[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> opt_delay[i];
    }
    cout << optimize_network(n, k, delay, win_size, opt_delay) << endl;
    return 0;
}
```
