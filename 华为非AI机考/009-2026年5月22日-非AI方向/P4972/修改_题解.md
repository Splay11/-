# 扫描朴素解法

这道题的正解是 $单调栈$，但是我们用朴素解法 $向右扫描找第一个更高建筑$ 也能在考试时拿到一定的分数。

题意：对每个建筑 $i$，向右看的「安全视野距离」等于：从 $i+1$ 起连续能看到的建筑个数。能看到的规则是——一直看到（并包含）右侧第一个严格更高的建筑；若右侧没有更高建筑，则能看到右边全部剩余建筑。相同高度不遮挡。

等价写法：若右侧存在第一个严格大于 $h_i$ 的位置 $j$，答案为 $j-i$；否则答案为 $n-1-i$。

朴素做法：对每个位置 $i$，从 $i+1$ 向右扫，直到找到更高建筑或扫到末尾。时间 $O(n^2)$。

$n$ 较小时可通过；当 $n$ 达到 $10^5$～$10^6$ 时会超时，需要后面的单调栈做法。

## 扫描解法

```python
def calc_distances(heights):
    n = len(heights)
    ans = [0] * n
    for i in range(n):
        # 向右找第一个严格更高的建筑
        j = i + 1
        while j < n and heights[j] <= heights[i]:
            j += 1
        if j < n:
            ans[i] = j - i
        else:
            ans[i] = n - 1 - i
    return ans


def main():

    heights = list(map(int, input().split()))
    ans = calc_distances(heights)
    print(" ".join(str(x) for x in ans))


if __name__ == "__main__":
    main()
```

# 正确解法

## 题目分析

仍然是「找右侧第一个严格更高的建筑」。用单调栈从左到右扫一遍即可。

维护一个下标栈，栈内高度单调不增（严格递减也行，本题相等不遮挡，用「高度严格递增被弹出」即可）：

1. 遇到当前高度 $h_i$ 时，不断弹出栈顶：若 $h_i$ 严格大于栈顶建筑高度，说明 $i$ 就是栈顶建筑右侧第一个更高建筑，答案为 $i - idx$。
2. 把 $i$ 入栈。
3. 扫完后，栈里剩余建筑右侧没有更高者，答案为 $n-1-idx$。

每个下标最多入栈、出栈一次，整体 $O(n)$。

## 复杂度分析

* 时间复杂度：$O(n)$。
* 空间复杂度：$O(n)$。

可满足 $n \le 5\times 10^6$。

## 代码实现

### Python

```python
def calc_distances(heights):
    n = len(heights)
    ans = [0] * n
    stack = []  # 存下标，对应高度单调不增

    for i, h in enumerate(heights):
        # 当前建筑是栈顶建筑右侧第一个更高者
        while stack and h > heights[stack[-1]]:
            idx = stack.pop()
            ans[idx] = i - idx
        stack.append(i)

    # 右侧没有更高建筑
    for idx in stack:
        ans[idx] = n - 1 - idx
    return ans


def main():
    n = int(input())
    heights = list(map(int, input().split()))
    ans = calc_distances(heights)
    print(" ".join(str(x) for x in ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.StringTokenizer;

public class Main {

    static int[] calcDistances(int[] heights) {
        int n = heights.length;
        int[] ans = new int[n];
        int[] stack = new int[n];  // 数组模拟栈，避免装箱
        int top = -1;

        for (int i = 0; i < n; i++) {
            // 当前建筑是栈顶建筑右侧第一个更高者
            while (top >= 0 && heights[i] > heights[stack[top]]) {
                int idx = stack[top--];
                ans[idx] = i - idx;
            }
            stack[++top] = i;
        }

        // 右侧没有更高建筑
        while (top >= 0) {
            int idx = stack[top--];
            ans[idx] = n - 1 - idx;
        }
        return ans;
    }

    public static void main(String[] args) throws Exception {
        // n 最大可达 5e6，用 BufferedReader 加快读写
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        int n = Integer.parseInt(br.readLine().trim());
        int[] heights = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            heights[i] = Integer.parseInt(st.nextToken());
        }
        int[] ans = calcDistances(heights);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(' ');
            sb.append(ans[i]);
            if (sb.length() > 1 << 20) {
                bw.write(sb.toString());
                sb.setLength(0);
            }
        }
        bw.write(sb.toString());
        bw.newLine();
        bw.flush();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

vector<int> calcDistances(const vector<int>& heights) {
    int n = (int)heights.size();
    vector<int> ans(n);
    vector<int> st;  // 存下标

    for (int i = 0; i < n; i++) {
        // 当前建筑是栈顶建筑右侧第一个更高者
        while (!st.empty() && heights[i] > heights[st.back()]) {
            int idx = st.back();
            st.pop_back();
            ans[idx] = i - idx;
        }
        st.push_back(i);
    }

    // 右侧没有更高建筑
    while (!st.empty()) {
        int idx = st.back();
        st.pop_back();
        ans[idx] = n - 1 - idx;
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);  // n 很大，加快读写
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> heights(n);
    for (int i = 0; i < n; i++) {
        cin >> heights[i];
    }
    vector<int> ans = calcDistances(heights);
    for (int i = 0; i < n; i++) {
        if (i) cout << ' ';
        cout << ans[i];
    }
    cout << '\n';
    return 0;
}
```
