## 解题思路

使用单调栈算法。

对于第 $i$ 个建筑，高度为 $h_i$，它向右看的安全视野距离规则是：

如果右侧存在第一个高度严格大于 $h_i$ 的建筑，则距离为这两个建筑的下标差；
如果右侧不存在更高的建筑，则距离为右侧剩余建筑数量，即 $n - 1 - i$。

注意：高度相同的建筑不会遮挡。

从左到右遍历建筑，用栈保存还没有找到更高建筑的下标。
当当前高度 $h_i$ 大于栈顶建筑高度时，说明当前建筑就是栈顶建筑右侧第一个更高建筑，弹出栈顶并计算答案。
每个建筑最多入栈一次、出栈一次。

## 复杂度分析

时间复杂度：$O(n)$，每个建筑最多入栈和出栈一次。
空间复杂度：$O(n)$，需要数组保存答案和单调栈。

## 代码实现

### Python

```python
import sys
from array import array


# 快速读取整数
def fast_ints():
    data = sys.stdin.buffer.read()
    num = 0
    in_num = False
    for c in data:
        if 48 <= c <= 57:
            num = num * 10 + c - 48
            in_num = True
        elif in_num:
            yield num
            num = 0
            in_num = False
    if in_num:
        yield num


# 计算每个建筑的安全视野距离
def calc_distances(heights):
    n = len(heights)
    ans = array('i', [0]) * n
    stack = array('i')  # 保存下标

    for i, h in enumerate(heights):
        # 当前建筑比栈顶建筑高，当前建筑就是栈顶建筑的第一个遮挡物
        while stack and h > heights[stack[-1]]:
            idx = stack.pop()
            ans[idx] = i - idx
        stack.append(i)

    # 没有遇到更高建筑的，能看到最右边
    for idx in stack:
        ans[idx] = n - 1 - idx

    return ans


def main():
    nums = fast_ints()
    n = next(nums)

    heights = array('H')
    for _ in range(n):
        heights.append(next(nums))

    ans = calc_distances(heights)

    out = sys.stdout.write
    size = 100000

    for i in range(0, n, size):
        if i > 0:
            out(' ')
        out(' '.join(str(ans[j]) for j in range(i, min(i + size, n))))
    out('\n')


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    // 快速输入，适合大数据
    static class FastScanner {
        private final InputStream in = System.in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        int nextInt() throws IOException {
            int c;
            do {
                c = read();
            } while (c <= ' ');

            int num = 0;
            while (c > ' ') {
                num = num * 10 + c - '0';
                c = read();
            }
            return num;
        }
    }

    // 计算每个建筑的安全视野距离
    static int[] calcDistances(int[] heights) {
        int n = heights.length;
        int[] ans = new int[n];
        int[] stack = new int[n];
        int top = -1;

        for (int i = 0; i < n; i++) {
            // 当前建筑比栈顶建筑高，当前建筑就是栈顶建筑的第一个遮挡物
            while (top >= 0 && heights[i] > heights[stack[top]]) {
                int idx = stack[top--];
                ans[idx] = i - idx;
            }
            stack[++top] = i;
        }

        // 没有遇到更高建筑的，能看到最右边
        while (top >= 0) {
            int idx = stack[top--];
            ans[idx] = n - 1 - idx;
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();

        int n = fs.nextInt();
        int[] heights = new int[n];

        for (int i = 0; i < n; i++) {
            heights[i] = fs.nextInt();
        }

        int[] ans = calcDistances(heights);

        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(' ');
            sb.append(ans[i]);

            // 分块输出，避免字符串过大
            if (sb.length() > 1 << 20) {
                bw.write(sb.toString());
                sb.setLength(0);
            }
        }

        bw.write(sb.toString());
        bw.newLine();
        bw.flush();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 计算每个建筑的安全视野距离
vector<int> calcDistances(const vector<int>& heights) {
    int n = heights.size();
    vector<int> ans(n);
    vector<int> st;
    st.reserve(n);

    for (int i = 0; i < n; i++) {
        // 当前建筑比栈顶建筑高，当前建筑就是栈顶建筑的第一个遮挡物
        while (!st.empty() && heights[i] > heights[st.back()]) {
            int idx = st.back();
            st.pop_back();
            ans[idx] = i - idx;
        }
        st.push_back(i);
    }

    // 没有遇到更高建筑的，能看到最右边
    while (!st.empty()) {
        int idx = st.back();
        st.pop_back();
        ans[idx] = n - 1 - idx;
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<int> heights(n);
    for (int i = 0; i < n; i++) {
        cin >> heights[i];
    }

    vector<int> ans = calcDistances(heights);

    for (int i = 0; i < n; i++) {
        if (i > 0) cout << ' ';
        cout << ans[i];
    }
    cout << '\n';

    return 0;
}
```
