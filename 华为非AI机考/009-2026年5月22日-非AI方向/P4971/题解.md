## 解题思路

本题的关键算法是字符串的游程编码，也就是把连续相同字符压缩成一个字符和出现次数。

一次操作只能删除两个相邻相同字符中的一个，因此它只会让某一段连续相同字符的长度减少 $1$，不会改变各段字符的顺序，也不能把某一段完全删掉。

所以字符串 $B$ 能变成字符串 $A$，等价于：

$B$ 和 $A$ 的游程字符序列完全相同，并且 $B$ 中每一段的长度都大于等于 $A$ 中对应段的长度。

先将 $A$ 压缩成：

字符序列 $ca$，长度序列 $la$。

再将 $S$ 压缩成：

字符序列 $cs$，长度序列 $ls$。

设 $A$ 压缩后有 $k$ 段。

当 $k=1$ 时，答案只来自 $S$ 中同字符的一段，长度为 $L$，若 $L \ge la_0$，贡献为：

$1+2+\cdots+(L-la_0+1)$

当 $k>1$ 时，枚举 $S$ 中连续 $k$ 段作为子串对应的游程段：

第 $1$ 段和第 $k$ 段可以只取一部分，中间段必须完整包含。

若字符序列相同，且每一段长度满足要求，则贡献为：

$(ls_i-la_0+1)\times(ls_{i+k-1}-la_{k-1}+1)$

## 复杂度分析

设 $A$ 压缩后有 $k$ 段，$S$ 压缩后有 $t$ 段。

枚举 $S$ 中的连续 $k$ 段，并检查是否匹配，时间复杂度为 $O(tk)$。

由于 $n,m \le 5000$，所以最坏约为 $25000000$ 次检查，可以通过。

空间复杂度为 $O(n+m)$，用于保存压缩后的字符串段信息。

## 代码实现

### Python

```python
def compress(s):
    chars = []
    lens = []
    for ch in s:
        if chars and chars[-1] == ch:
            lens[-1] += 1
        else:
            chars.append(ch)
            lens.append(1)
    return chars, lens


def count_good_substrings(a, s):
    ca, la = compress(a)
    cs, ls = compress(s)

    k = len(ca)
    t = len(cs)
    ans = 0

    # 特殊情况：A 只有一段连续相同字符
    if k == 1:
        need = la[0]
        for i in range(t):
            if cs[i] == ca[0] and ls[i] >= need:
                x = ls[i] - need + 1
                ans += x * (x + 1) // 2
        return ans

    # 枚举 S 中连续 k 段
    for i in range(t - k + 1):
        ok = True

        # 检查字符序列和中间段长度
        for j in range(k):
            if cs[i + j] != ca[j]:
                ok = False
                break
            if 0 < j < k - 1 and ls[i + j] < la[j]:
                ok = False
                break

        if not ok:
            continue

        # 检查首尾段长度，并计算可选起点和终点数量
        left = ls[i] - la[0] + 1
        right = ls[i + k - 1] - la[k - 1] + 1

        if left > 0 and right > 0:
            ans += left * right

    return ans


def main():
    n, m = map(int, input().split())
    a = input().strip()
    s = input().strip()

    print(count_good_substrings(a, s))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static class Pair {
        ArrayList<Character> chars;
        ArrayList<Integer> lens;

        Pair(ArrayList<Character> chars, ArrayList<Integer> lens) {
            this.chars = chars;
            this.lens = lens;
        }
    }

    // 对字符串进行游程编码
    static Pair compress(String s) {
        ArrayList<Character> chars = new ArrayList<>();
        ArrayList<Integer> lens = new ArrayList<>();

        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            if (!chars.isEmpty() && chars.get(chars.size() - 1) == ch) {
                int last = lens.size() - 1;
                lens.set(last, lens.get(last) + 1);
            } else {
                chars.add(ch);
                lens.add(1);
            }
        }

        return new Pair(chars, lens);
    }

    static long countGoodSubstrings(String a, String s) {
        Pair pa = compress(a);
        Pair ps = compress(s);

        ArrayList<Character> ca = pa.chars;
        ArrayList<Integer> la = pa.lens;
        ArrayList<Character> cs = ps.chars;
        ArrayList<Integer> ls = ps.lens;

        int k = ca.size();
        int t = cs.size();
        long ans = 0;

        // 特殊情况：A 只有一段连续相同字符
        if (k == 1) {
            int need = la.get(0);

            for (int i = 0; i < t; i++) {
                if (cs.get(i) == ca.get(0) && ls.get(i) >= need) {
                    long x = ls.get(i) - need + 1L;
                    ans += x * (x + 1) / 2;
                }
            }

            return ans;
        }

        // 枚举 S 中连续 k 段
        for (int i = 0; i + k <= t; i++) {
            boolean ok = true;

            // 检查字符序列和中间段长度
            for (int j = 0; j < k; j++) {
                if (cs.get(i + j) != ca.get(j)) {
                    ok = false;
                    break;
                }

                if (j > 0 && j < k - 1 && ls.get(i + j) < la.get(j)) {
                    ok = false;
                    break;
                }
            }

            if (!ok) {
                continue;
            }

            // 检查首尾段长度，并计算可选起点和终点数量
            long left = ls.get(i) - la.get(0) + 1L;
            long right = ls.get(i + k - 1) - la.get(k - 1) + 1L;

            if (left > 0 && right > 0) {
                ans += left * right;
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();
        String a = sc.next();
        String s = sc.next();

        System.out.println(countGoodSubstrings(a, s));
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

// 对字符串进行游程编码
pair<vector<char>, vector<int>> compressString(const string& s) {
    vector<char> chars;
    vector<int> lens;

    for (char ch : s) {
        if (!chars.empty() && chars.back() == ch) {
            lens.back()++;
        } else {
            chars.push_back(ch);
            lens.push_back(1);
        }
    }

    return {chars, lens};
}

long long countGoodSubstrings(const string& a, const string& s) {
    auto pa = compressString(a);
    auto ps = compressString(s);

    vector<char> ca = pa.first;
    vector<int> la = pa.second;
    vector<char> cs = ps.first;
    vector<int> ls = ps.second;

    int k = ca.size();
    int t = cs.size();
    long long ans = 0;

    // 特殊情况：A 只有一段连续相同字符

        int need = la[0];

        for (int i = 0; i < t; i++) {
            if (cs[i] == ca[0] && ls[i] >= need) {
                long long x = ls[i] - need + 1LL;
                ans += x * (x + 1) / 2;
            }
        }

        return ans;
    }

    // 枚举 S 中连续 k 段
    for (int i = 0; i + k <= t; i++) {
        bool ok = true;

        // 检查字符序列和中间段长度
        for (int j = 0; j < k; j++) {
            if (cs[i + j] != ca[j]) {
                ok = false;
                break;

        }
    }

    return ans;
}
```
