# 枚举朴素解法

这道题的正解是 $单调栈$ 贪心求字典序最大子序列，但是我们用朴素解法 $暴力枚举所有长度为 k 的子序列取最大$ 也能在考试时拿到一定的分数。

题意：在保持相对顺序的前提下，从码流中选出恰好 $k$ 个子包 ID，使得到的长度为 $k$ 的序列字典序最大。输入若含非十进制数字字符则输出 `error`。

朴素做法：用 DFS / 回溯枚举「选或不选」当前元素，最终恰好选满 $k$ 个，在所有候选中取字典序最大的那一个。时间约为 $O(C(n,k)\cdot k)$。

样例和很小的 $n$ 可以算对；当 $n$ 接近 $2000$、且 $k$ 不太小也不太大时组合数爆炸，会超时，需要后面的单调栈做法。

## 枚举解法

```python
def max_subsequence(nums, k):
    n = len(nums)
    best = None

    def dfs(i, chosen):
        nonlocal best
        remain = n - i
        need = k - len(chosen)
        # 剩下的不够选满，或已经选超
        if need < 0 or need > remain:
            return
        if i == n:
            if len(chosen) == k:
                if best is None or chosen > best:
                    best = chosen[:]
            return
        # 选当前位置
        chosen.append(nums[i])
        dfs(i + 1, chosen)
        chosen.pop()
        # 不选当前位置
        dfs(i + 1, chosen)

    dfs(0, [])
    return best


def is_valid_line(s):
    for ch in s:
        if not (ch.isdigit() or ch.isspace()):
            return False
    return True


def main():
    try:
        line1 = input().strip()
    except EOFError:
        print("error")
        return
    if not line1 or not is_valid_line(line1):
        print("error")
        return

    try:
        line2 = input().strip()
    except EOFError:
        print("error")
        return
    if not line2 or not is_valid_line(line2):
        print("error")
        return

    nums = list(map(int, line1.split()))
    k = int(line2)
    ans = max_subsequence(nums, k)
    print(" ".join(str(x) for x in ans))


if __name__ == "__main__":
    main()
```

```cpp
#include <iostream>

#include <string>
#include <vector>
#include <cctype>
using namespace std;

vector<int> best;
bool hasBest;

void dfs(int i, const vector<int>& nums, int k, vector<int>& chosen) {
    int n = (int)nums.size();
    int remain = n - i;
    int need = k - (int)chosen.size();
    if (need < 0 || need > remain) return;
    if (i == n) {
        if ((int)chosen.size() == k) {
            if (!hasBest || chosen > best) {
                best = chosen;
                hasBest = true;
            }
        }
        return;
    }
    // 选当前位置
    chosen.push_back(nums[i]);
    dfs(i + 1, nums, k, chosen);
    chosen.pop_back();
    // 不选当前位置
    dfs(i + 1, nums, k, chosen);
}

vector<int> maxSubsequence(const vector<int>& nums, int k) {
    best.clear();
    hasBest = false;
    vector<int> chosen;
    dfs(0, nums, k, chosen);
    return best;
}

bool isValidLine(const string& s) {
    for (char ch : s) {
        if (!isdigit((unsigned char)ch) && !isspace((unsigned char)ch)) {
            return false;
        }
    }
    return true;
}
```

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {

    static List<Integer> best;

    static void dfs(int i, int[] nums, int k, List<Integer> chosen) {
        int n = nums.length;
        int remain = n - i;
        int need = k - chosen.size();
        if (need < 0 || need > remain) return;
        if (i == n) {
            if (chosen.size() == k) {
                if (best == null || compareList(chosen, best) > 0) {
                    best = new ArrayList<Integer>(chosen);
                }
            }
            return;
        }
        // 选当前位置
        chosen.add(nums[i]);
        dfs(i + 1, nums, k, chosen);
        chosen.remove(chosen.size() - 1);
        // 不选当前位置
        dfs(i + 1, nums, k, chosen);
    }

    static int compareList(List<Integer> a, List<Integer> b) {
        for (int i = 0; i < a.size(); i++) {
            if (!a.get(i).equals(b.get(i))) {
                return a.get(i) - b.get(i);
            }
        }
        return 0;
    }

    static int[] maxSubsequence(int[] nums, int k) {
        best = null;
        dfs(0, nums, k, new ArrayList<Integer>());
        int[] res = new int[k];
        for (int i = 0; i < k; i++) {
            res[i] = best.get(i);
        }
        return res;
    }

    static boolean isValidLine(String s) {
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            if (!Character.isDigit(ch) && !Character.isWhitespace(ch)) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextLine()) {
            System.out.println("error");
            return;
        }
        String line1 = sc.nextLine().trim();
        if (line1.isEmpty() || !isValidLine(line1)) {
            System.out.println("error");
            return;
        }
        if (!sc.hasNextLine()) {
            System.out.println("error");
            return;
        }
        String line2 = sc.nextLine().trim();
        if (line2.isEmpty() || !isValidLine(line2)) {
            System.out.println("error");
            return;
        }

        String[] parts = line1.split("\\s+");
        int[] nums = new int[parts.length];
        for (int i = 0; i < parts.length; i++) {
            nums[i] = Integer.parseInt(parts[i]);
        }
        int k = Integer.parseInt(line2);
        int[] ans = maxSubsequence(nums, k);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < ans.length; i++) {
            if (i > 0) sb.append(' ');
            sb.append(ans[i]);
        }
        System.out.println(sb.toString());
    }
}
```

# 正确解法

## 题目分析

要从长度为 $n$ 的序列中选出长度为 $k$ 的字典序最大子序列，需要删掉 $drop=n-k$ 个元素。

用单调栈贪心：从左到右扫描每个数 $x$，当还能删除，且栈顶小于 $x$ 时，弹出栈顶（删掉更小的前缀候选），再把 $x$ 入栈。扫完后若栈长大于 $k$，只保留前 $k$ 个。

输入校验：任一行出现非数字、非空白字符，输出 `error`。

## 复杂度分析

* 时间复杂度：$O(n)$，每个元素最多入栈、出栈一次。
* 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
def max_subsequence(nums, k):
    n = len(nums)
    drop = n - k  # 还能删除的个数
    stack = []
    for x in nums:
        # 当前更大且还能删，就弹掉更小的栈顶
        while drop > 0 and stack and stack[-1] < x:
            stack.pop()
            drop -= 1
        stack.append(x)
    return stack[:k]


def is_valid_line(s):
    for ch in s:
        if not (ch.isdigit() or ch.isspace()):
            return False
    return True


def main():

        line1 = input().strip()
    except EOFError:
        print("error")
        return
    if not line1 or not is_valid_line(line1):
        print("error")
        return

    try:
        line2 = input().strip()
    except EOFError:

        return
    if not line2 or not is_valid_line(line2):
        print("error")
        return

    ans = max_subsequence(nums, k)

if __name__ == "__main__":
```

```java
import java.util.Scanner;

public class Main {

    static int[] maxSubsequence(int[] nums, int k) {
        int n = nums.length;
        int drop = n - k;
        int[] stack = new int[n];
        int top = 0;

        for (int x : nums) {
            // 当前更大且还能删，就弹掉更小的栈顶
            while (drop > 0 && top > 0 && stack[top - 1] < x) {
                top--;
                drop--;
            }
            stack[top++] = x;
        }

        int[] res = new int[k];
        System.arraycopy(stack, 0, res, 0, k);
        return res;
    }

    static boolean isValidLine(String s) {
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            if (!Character.isDigit(ch) && !Character.isWhitespace(ch)) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextLine()) {
            System.out.println("error");
            return;
        }

            System.out.println("error");
```

```cpp
#include <iostream>
#include <sstream>
#include <string>
#include <vector>
#include <cctype>
using namespace std;

vector<int> maxSubsequence(const vector<int>& nums, int k) {
    int n = (int)nums.size();
    int drop = n - k;
    vector<int> st;
    for (int x : nums) {
        // 当前更大且还能删，就弹掉更小的栈顶
        while (drop > 0 && !st.empty() && st.back() < x) {
            st.pop_back();
            --drop;
        }
        st.push_back(x);
    }
    if ((int)st.size() > k) {
        st.resize(k);
    }
    return st;
}

bool isValidLine(const string& s) {
    for (char ch : s) {
        if (!isdigit((unsigned char)ch) && !isspace((unsigned char)ch)) {
            return false;
        }
    }
    return true;
}
```
