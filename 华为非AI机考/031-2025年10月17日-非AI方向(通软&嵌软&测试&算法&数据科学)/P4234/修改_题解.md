## 解题思路

本题等价于**有限次硬币找零（组合数）**：给出若干奖品价值（可有重复），价值相同记为同一奖品；若某价值出现 $cnt$ 次，则该价值最多可选 $cnt$ 个。统计**不计顺序**、总价值恰好为 $num$ 的组合个数。

做法：

1. 先将 $values$ 压缩为「价值 $\to$ 频次」：例如 $4$ 出现 $2$ 次，表示面值 $4$ 最多用 $2$ 个。
2. 使用**二维计数 DP（组合型）**。设共有 $M$ 种不同价值，第 $i$ 种价值为 $v$，可用次数为 $cnt$。定义：

$$
dp[i][s]
$$

表示只考虑前 $i$ 种价值时，凑出总和为 $s$ 的组合数。

边界：$dp[0][0]=1$（什么都不选），其余 $dp[0][s]=0$。

转移：对第 $i$ 种价值，枚举选用个数 $k$（$0\le k\le \min(cnt,\lfloor s/v\rfloor)$）：

$$
dp[i][s]=\sum_{k=0}^{\min(cnt,\lfloor s/v\rfloor)} dp[i-1][s-k\cdot v]
$$

按价值种类依次更新，每种价值只考虑一次，因此不会把排列重复计入。

3. 答案为 $dp[M][num]$。由于 $len,num\le 100$，复杂度完全可接受。

## 复杂度分析

* 时间复杂度：$O(M\times num\times K)$，其中 $M$ 为不同价值种数，$K$ 为每种价值可选件数上界；本题范围内约为 $10^6$ 量级。
* 空间复杂度：$O(M\times num)$。

## 代码实现

### Python

```python
from collections import Counter


def count_combinations(values, num):
    # 压缩为 价值 -> 频次
    freq = list(Counter(values).items())
    m = len(freq)
    # dp[i][s]：考虑前 i 种价值，凑出和 s 的组合数
    dp = [[0] * (num + 1) for _ in range(m + 1)]
    dp[0][0] = 1

    for i in range(1, m + 1):
        v, cnt = freq[i - 1]
        for s in range(0, num + 1):
            total = 0
            max_k = min(cnt, s // v)
            # 枚举选 k 个当前价值
            for k in range(0, max_k + 1):
                total += dp[i - 1][s - k * v]
            dp[i][s] = total
    return dp[m][num]


def main():
    n = int(input())
    values = list(map(int, input().split()))
    num = int(input())
    print(count_combinations(values, num))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

    static long countCombinations(int[] values, int num) {
        Map<Integer, Integer> freqMap = new HashMap<>();
        for (int v : values) {
            freqMap.put(v, freqMap.getOrDefault(v, 0) + 1);
        }
        int m = freqMap.size();
        int[] vs = new int[m];
        int[] cnts = new int[m];
        int idx = 0;
        for (Map.Entry<Integer, Integer> e : freqMap.entrySet()) {
            vs[idx] = e.getKey();
            cnts[idx] = e.getValue();
            idx++;
        }

        // dp[i][s]：考虑前 i 种价值，凑出和 s 的组合数
        long[][] dp = new long[m + 1][num + 1];
        dp[0][0] = 1;

        for (int i = 1; i <= m; i++) {
            int v = vs[i - 1];
            int cnt = cnts[i - 1];
            for (int s = 0; s <= num; s++) {
                long total = 0;
                int maxK = Math.min(cnt, s / v);
                for (int k = 0; k <= maxK; k++) {
                    total += dp[i - 1][s - k * v];
                }
                dp[i][s] = total;
            }
        }
        return dp[m][num];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] values = new int[n];
        for (int i = 0; i < n; i++) {
            values[i] = sc.nextInt();
        }
        int num = sc.nextInt();
        System.out.println(countCombinations(values, num));
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <map>
using namespace std;

long long countCombinations(const vector<int>& values, int num) {
    map<int, int> freq;
    for (int v : values) {
        freq[v]++;
    }
    vector<pair<int, int>> items(freq.begin(), freq.end());
    int m = (int)items.size();

    // dp[i][s]：考虑前 i 种价值，凑出和 s 的组合数
    vector<vector<long long>> dp(m + 1, vector<long long>(num + 1, 0));
    dp[0][0] = 1;

    for (int i = 1; i <= m; i++) {
        int v = items[i - 1].first;
        int cnt = items[i - 1].second;
        for (int s = 0; s <= num; s++) {
            long long total = 0;
            int maxK = min(cnt, s / v);
            for (int k = 0; k <= maxK; k++) {
                total += dp[i - 1][s - k * v];
            }
            dp[i][s] = total;
        }
    }
    return dp[m][num];
}

int main() {
    int n;
    cin >> n;
    vector<int> values(n);
    for (int i = 0; i < n; i++) {
        cin >> values[i];
    }
    int num;
    cin >> num;
    cout << countCombinations(values, num) << "\n";
    return 0;
}
```
