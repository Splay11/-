# 枚举窗口最小值朴素解法

这道题的正解是 $单调队列$，但是我们用朴素解法 $每段暴力求窗口最小值$ 也能在考试时拿到一定的分数。

题意：第 $i$ 号路段（路口 $i$ 与 $i+1$ 之间）可连接的基站编号区间为

$$L=\max(0,\,i+1-k),\quad R=\min(N-1,\,i+k)$$

在该区间内选接入人数最少的基站编号；人数相同时取靠右的编号。非法输入输出 $-1$。

朴素做法：对每个路段 $i$，直接枚举 $[L,R]$ 找最小值下标。时间 $O(Nk)$。

$N$ 较小时可通过；当 $N$ 接近 $10^5$ 时会超时，需要后面的单调队列做法。
## 枚举窗口最小值解法

```python
        ret.append(best)
    return ret
```

```cpp
vector<int> best_station(const vector<int>& crossroads, int k) {
    int n = (int)crossroads.size();
    if (n < 2 || k < 1 || k > n) return {};
    vector<int> ret(n - 1);
    for (int i = 0; i < n - 1; i++) {
        int L = max(0, i + 1 - k);
        int R = min(n - 1, i + k);
        // 暴力扫窗口，相等时取更靠右的下标
        int best = L;
        for (int j = L + 1; j <= R; j++) {
            if (crossroads[j] <= crossroads[best]) {
                best = j;
            }
        }
        ret[i] = best;
    }
    return ret;
}
```

# 正确解法

## 题目分析

每个路段对应一个随 $i$ 滑动的区间 $[L_i,R_i]$，需要快速查区间最小值下标。用双端队列维护下标，使队内对应的接入人数从队首到队尾**单调不降**：

1. 右端扩展到 $R$ 时，新下标入队前弹出队尾所有「人数 $\ge$ 新值」的下标（相等也弹，保证同人数时保留更靠右的）。
2. 左端收缩到 $L$ 时，若队首下标 $<L$，弹出。
3. 此时队首就是当前窗口最小值下标。

每个下标最多入队、出队一次，总时间 $O(N)$。

## 复杂度分析

- 时间复杂度：$O(N)$，每个下标进出队各至多一次。
- 空间复杂度：$O(N)$，双端队列与答案数组。

## 代码实现

### Python

```python
from collections import deque


def best_station(crossroads, k):
    n = len(crossroads)
    if n < 2 or k < 1 or k > n:
        return None
    dq = deque()  # 存下标，crossroads 值单调不降
    ret = []
    R = -1
    for i in range(n - 1):
        L = max(0, i + 1 - k)
        newR = min(n - 1, i + k)
        # 扩展右端
        while R < newR:
            R += 1
            while dq and crossroads[dq[-1]] >= crossroads[R]:
                dq.pop()
            dq.append(R)
        # 收缩左端
        while dq and dq[0] < L:
            dq.popleft()
        ret.append(dq[0])
    return ret

if not parts:

            print(" ".join(map(str, ans)))
```

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;
import java.util.Scanner;

public class Main {
    static List<Integer> bestStation(int[] crossroads, int k) {
        int n = crossroads.length;
        if (n < 2 || k < 1 || k > n) return null;
        Deque<Integer> dq = new ArrayDeque<>(); // 存下标，值单调不降
        List<Integer> ret = new ArrayList<>();
        int R = -1;
        for (int i = 0; i < n - 1; i++) {
            int L = Math.max(0, i + 1 - k);
            int newR = Math.min(n - 1, i + k);
            // 扩展右端
            while (R < newR) {
                R++;
                while (!dq.isEmpty() && crossroads[dq.peekLast()] >= crossroads[R]) {
                    dq.pollLast();
                }
                dq.offerLast(R);
            }
            // 收缩左端
            while (!dq.isEmpty() && dq.peekFirst() < L) {
                dq.pollFirst();
            }
            ret.add(dq.peekFirst());
        }
        return ret;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextLine()) {
            System.out.println(-1);
            return;
        }
        try {
            String[] parts = sc.nextLine().trim().split("\\s+");

            }
            if (!sc.hasNextInt()) {
```

```cpp
#include <deque>
#include <iostream>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

vector<int> best_station(const vector<int>& crossroads, int k) {
    int n = (int)crossroads.size();
    if (n < 2 || k < 1 || k > n) return {};
    deque<int> dq;  // 存下标，值单调不降
    vector<int> ret;
    int R = -1;
    for (int i = 0; i < n - 1; i++) {
        int L = max(0, i + 1 - k);
        int newR = min(n - 1, i + k);
        // 扩展右端
        while (R < newR) {
            R++;
            while (!dq.empty() && crossroads[dq.back()] >= crossroads[R]) {
                dq.pop_back();
            }
            dq.push_back(R);
        }
        // 收缩左端
        while (!dq.empty() && dq.front() < L) {
            dq.pop_front();
        }
        ret.push_back(dq.front());
    }
    return ret;
}
```
