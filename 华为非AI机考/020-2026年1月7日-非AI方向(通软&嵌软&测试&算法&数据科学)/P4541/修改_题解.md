# 枚举朴素解法

这道题的正解是 $分治求最近点对$，但是我们用朴素解法 $枚举所有点对$ 也能在考试时拿到一定的分数。

题意是：给定平面上 $n$ 个点，求任意两点欧氏距离平方 $(x_1-x_2)^2+(y_1-y_2)^2$ 的最小值。

朴素做法：

1. 用两重循环枚举所有点对 $(i,j)$（$0\le i<j<n$）。
2. 计算距离平方 $d=(x_i-x_j)^2+(y_i-y_j)^2$，用长整型避免溢出。
3. 维护全局最小值即可。

样例和小数据可以算对；但 $n$ 最大可达 $10^5$，枚举点对是 $O(n^2)$，大数据会超时，因此后面再给出分治正解。

## 枚举解法

```python
def closest_pair_naive(points):

    best = 10 ** 30
    # 枚举所有点对，求最小距离平方

        x1, y1 = points[i]

            x2, y2 = points[j]

            dy = y1 - y2
            d = dx * dx + dy * dy

    return best
```

# 正确解法

## 题目分析

用经典的**分治求最近点对**，在 $O(n\log n)$ 内求出最小距离平方。

1. 先把所有点按 $x$ 坐标排序。
2. 递归把当前区间分成左右两半，分别求出左右内部的最小距离平方，记为 $d$。
3. 合并阶段：把当前区间按 $y$ 归并排序，使返回后区间内点按 $y$ 有序。
4. 只检查「中线附近」宽度约为 $\sqrt{d}$ 的条带 $strip$：满足 $(x-midx)^2 < d$ 的点。
5. 条带内点已按 $y$ 有序，对每个点只需向后检查 $y$ 差平方小于 $d$ 的少量点，更新最小值。

区间点数 $\le 3$ 时直接暴力。最终得到全局最小距离平方。

## 复杂度分析

* 时间复杂度：$O(n\log n)$（每层分治做一次线性归并与条带检查，共 $\log n$ 层）。
* 空间复杂度：$O(n)$（归并辅助数组）。

## 代码实现

### Python

```python
def closest_pair(points):
    n = len(points)
    tmp = [None] * n  # 归并临时数组

    def solve(l, r):
        # 计算 points[l:r] 的最近距离平方，并保证该区间按 y 有序
        if r - l <= 3:
            best = 10 ** 30
            for i in range(l, r):
                x1, y1 = points[i]
                for j in range(i + 1, r):
                    x2, y2 = points[j]
                    dx = x1 - x2
                    dy = y1 - y2
                    d = dx * dx + dy * dy
                    if d < best:
                        best = d
            points[l:r] = sorted(points[l:r], key=lambda p: p[1])
            return best

        m = (l + r) // 2
        midx = points[m][0]
        d = solve(l, m)
        dr = solve(m, r)
        if dr < d:
            d = dr

        # 两段已按 y 有序，归并成整段按 y 有序
        i, j, k = l, m, l
        while i < m and j < r:
            if points[i][1] <= points[j][1]:
                tmp[k] = points[i]
                i += 1
            else:
                tmp[k] = points[j]
                j += 1
            k += 1
        while i < m:
            tmp[k] = points[i]
            i += 1
            k += 1
        while j < r:
            tmp[k] = points[j]
            j += 1
            k += 1
        points[l:r] = tmp[l:r]

        # 收集中线附近的条带点
        strip = []
        for t in range(l, r):
            dx = points[t][0] - midx
            if dx * dx < d:
                strip.append(points[t])

        # 条带按 y 有序，只检查 y 足够近的后续点
        s = len(strip)
        for a in range(s):
            x1, y1 = strip[a]
            b = a + 1
            while b < s:
                dy = strip[b][1] - y1
                if dy * dy >= d:
                    break
                dx = strip[b][0] - x1
                dist = dx * dx + dy * dy
                if dist < d:
                    d = dist
                b += 1
        return d

    points.sort()  # 先按 x 排序
    return solve(0, n)


def main():
    n = int(input())
    points = []
    for _ in range(n):
        x, y = map(int, input().split())
        points.append((x, y))
    print(closest_pair(points))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.StringTokenizer;

public class Main {

    static class Point {
        int x, y;
        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static Point[] pts;
    static Point[] tmp;

    static long closestPair(Point[] points) {
        pts = points;
        Arrays.sort(pts, (a, b) -> a.x != b.x ? Integer.compare(a.x, b.x) : Integer.compare(a.y, b.y));
        tmp = new Point[pts.length];
        return solve(0, pts.length);
    }

    // 计算 pts[l:r) 的最近距离平方，并保证该区间按 y 有序
    static long solve(int l, int r) {
        if (r - l <= 3) {
            long best = Long.MAX_VALUE;
            for (int i = l; i < r; i++) {
                for (int j = i + 1; j < r; j++) {
                    long dx = (long) pts[i].x - pts[j].x;
                    long dy = (long) pts[i].y - pts[j].y;
                    long d = dx * dx + dy * dy;
                    if (d < best) {
                        best = d;
                    }
                }
            }
            Arrays.sort(pts, l, r, Comparator.comparingInt(o -> o.y));
            return best;
        }

        int m = (l + r) >>> 1;
        int midx = pts[m].x;
        long d = solve(l, m);
        long dr = solve(m, r);
        if (dr < d) {
            d = dr;
        }

        // 归并：两段按 y 有序合并
        int i = l, j = m, k = l;
        while (i < m && j < r) {
            if (pts[i].y <= pts[j].y) {
                tmp[k++] = pts[i++];
            } else {
                tmp[k++] = pts[j++];
            }
        }
        while (i < m) {
            tmp[k++] = pts[i++];
        }
        while (j < r) {
            tmp[k++] = pts[j++];
        }
        for (int t = l; t < r; t++) {
            pts[t] = tmp[t];
        }

        // 收集条带点
        Point[] strip = new Point[r - l];
        int s = 0;
        for (int t = l; t < r; t++) {
            long dx = (long) pts[t].x - midx;
            if (dx * dx < d) {
                strip[s++] = pts[t];
            }
        }

        // 检查 y 足够近的后续点
        for (int a = 0; a < s; a++) {
            for (int b = a + 1; b < s; b++) {
                long dy = (long) strip[b].y - strip[a].y;
                if (dy * dy >= d) {
                    break;
                }
                long dx = (long) strip[b].x - strip[a].x;
                long dist = dx * dx + dy * dy;
                if (dist < d) {
                    d = dist;
                }
            }
        }
        return d;
    }

    public static void main(String[] args) throws Exception {
        // n 可达 1e5，用 BufferedReader 加速读入
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            points[i] = new Point(x, y);
        }
        System.out.println(closestPair(points));
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Point {
    int x, y;
};

static vector<Point> pts;
static vector<Point> tmp;

// 计算 pts[l:r) 的最近距离平方，并保证该区间按 y 有序
long long solve(int l, int r) {
    if (r - l <= 3) {
        long long best = (1LL << 62);
        for (int i = l; i < r; i++) {
            for (int j = i + 1; j < r; j++) {
                long long dx = 1LL * pts[i].x - pts[j].x;
                long long dy = 1LL * pts[i].y - pts[j].y;
                long long d = dx * dx + dy * dy;
                if (d < best) {
                    best = d;
                }
            }
        }
        sort(pts.begin() + l, pts.begin() + r, [](const Point& a, const Point& b) {
            return a.y < b.y;
        });
        return best;
    }

    int m = (l + r) / 2;
    int midx = pts[m].x;
    long long d = solve(l, m);
    long long dr = solve(m, r);
    if (dr < d) {
        d = dr;
    }

    // 归并按 y
    int i = l, j = m, k = l;
    while (i < m && j < r) {
        if (pts[i].y <= pts[j].y) {
            tmp[k++] = pts[i++];
        } else {
            tmp[k++] = pts[j++];
        }
    }
    while (i < m) {
        tmp[k++] = pts[i++];
    }
    while (j < r) {
        tmp[k++] = pts[j++];
    }
    for (int t = l; t < r; t++) {
        pts[t] = tmp[t];
    }

    // 条带点（按 y 已有序）
    vector<Point> strip;
    strip.reserve(r - l);
    for (int t = l; t < r; t++) {
        long long dx = 1LL * pts[t].x - midx;
        if (dx * dx < d) {
            strip.push_back(pts[t]);
        }
    }

    for (int a = 0; a < (int)strip.size(); a++) {
        for (int b = a + 1; b < (int)strip.size(); b++) {
            long long dy = 1LL * strip[b].y - strip[a].y;
            if (dy * dy >= d) {
                break;
            }
            long long dx = 1LL * strip[b].x - strip[a].x;
            long long dist = dx * dx + dy * dy;
            if (dist < d) {
                d = dist;
            }
        }
    }
    return d;
}

long long closest_pair(vector<Point> points) {
    pts = points;
    sort(pts.begin(), pts.end(), [](const Point& a, const Point& b) {
        if (a.x != b.x) {
            return a.x < b.x;
        }
        return a.y < b.y;
    });
    tmp.assign(pts.size(), Point{0, 0});
    return solve(0, (int)pts.size());
}

int main() {
    ios::sync_with_stdio(false);  // n 可达 1e5，关掉同步加速读入
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<Point> points(n);
    for (int i = 0; i < n; i++) {
        cin >> points[i].x >> points[i].y;
    }
    cout << closest_pair(points) << "\n";
    return 0;
}
```
