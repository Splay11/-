# 扫描朴素解法

这道题的正解是 $单调栈$，但是我们用朴素解法 $对每个位置向右暴力扫描$ 也能在考试时拿到一定的分数。

题意：先把每个数 $arr[i]$ 按十六进制各位数字之和算出「权重」，再对每个位置求右侧第一个权重严格更大的元素下标；若不存在则输出 $-1$。

权重计算：反复取最低四位 $n\&15$ 累加，再右移四位，直到数为 $0$。

朴素做法：对每个下标 $i$，从 $i+1$ 向右扫描，找到第一个权重更大的位置。时间 $O(n^2)$。

$n$ 较小时可通过；当 $n$ 接近 $2\times 10^5$ 时会超时，需要后面的单调栈做法。

```python
def compute_weight(n):
    # 十六进制各位数字之和
    s = 0
    while True:
        s += n & 15
        n >>= 4
        if n == 0:
            break
    return s


def next_greater_weight(arr):
    n = len(arr)
    weights = [compute_weight(x) for x in arr]
    ans = [-1] * n
    for i in range(n):
        # 向右找第一个权重更大的下标
        for j in range(i + 1, n):
            if weights[j] > weights[i]:
                ans[i] = j
                break
    return ans

    arr = list(map(int, input().split()))
    ans = next_greater_weight(arr)
    print(" ".join(str(x) for x in ans))

    main()
```

# 正确解法

## 题目分析

先对每个数算十六进制位权和，问题变成经典的「下一个更大元素」，用单调栈 $O(n)$ 解决。

从右向左遍历：

1. 栈中存下标，保持对应权重严格递减。
2. 弹出所有权重 $\le$ 当前权重的下标。
3. 若栈非空，栈顶就是右侧第一个更大权重下标；否则为 $-1$。
4. 把当前下标入栈。

## 复杂度分析

* 时间复杂度：$O(n)$，每个下标最多进栈、出栈一次（权重计算每位常数，可并入 $O(n)$）。
* 空间复杂度：$O(n)$。

可满足 $n \le 2\times 10^5$。

```python
def compute_weight(n):
    # 十六进制各位数字之和
    s = 0
    while True:
        s += n & 15
        n >>= 4
        if n == 0:
            break
    return s


def next_greater_weight(arr):
    n = len(arr)
    weights = [compute_weight(x) for x in arr]
    ans = [-1] * n
    stack = []  # 存下标，权重单调递减

    for i in range(n - 1, -1, -1):
        while stack and weights[stack[-1]] <= weights[i]:
            stack.pop()
        if stack:
            ans[i] = stack[-1]
        stack.append(i)
    return ans


def main():
    N = int(input())
    arr = list(map(int, input().split()))
    ans = next_greater_weight(arr)
    print(" ".join(str(x) for x in ans))


if __name__ == "__main__":
    main()
```

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Main {

    static int computeWeight(long n) {
        // 十六进制各位数字之和
        int s = 0;
        while (true) {
            s += (int) (n & 15L);
            n >>>= 4;
            if (n == 0) break;
        }
        return s;
    }

    static int[] nextGreaterWeight(long[] arr) {
        int n = arr.length;
        int[] weights = new int[n];
        for (int i = 0; i < n; i++) {
            weights[i] = computeWeight(arr[i]);
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            ans[i] = -1;
        }
        Deque<Integer> stack = new ArrayDeque<>();

        for (int i = n - 1; i >= 0; i--) {
            while (!stack.isEmpty() && weights[stack.peek()] <= weights[i]) {
                stack.pop();
            }
            if (!stack.isEmpty()) {
                ans[i] = stack.peek();
            }
            stack.push(i);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[] arr = new long[N];
        for (int i = 0; i < N; i++) {
            arr[i] = Long.parseUnsignedLong(sc.next());
        }
        int[] ans = nextGreaterWeight(arr);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < N; i++) {
            if (i > 0) sb.append(' ');
            sb.append(ans[i]);
        }
        System.out.println(sb.toString());
    }
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

int computeWeight(unsigned int n) {
    // 十六进制各位数字之和
    int s = 0;
    while (true) {
        s += (n & 15u);
        n >>= 4;
        if (n == 0) break;
    }
    return s;
}

vector<int> nextGreaterWeight(const vector<unsigned int>& arr) {
    int n = (int)arr.size();
    vector<int> weights(n);
    for (int i = 0; i < n; i++) {
        weights[i] = computeWeight(arr[i]);
    }
    vector<int> ans(n, -1);
    vector<int> st;  // 存下标

    for (int i = n - 1; i >= 0; i--) {
        while (!st.empty() && weights[st.back()] <= weights[i]) {
            st.pop_back();
        }
        if (!st.empty()) {
            ans[i] = st.back();
        }
        st.push_back(i);
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;
    vector<unsigned int> arr(N);
    for (int i = 0; i < N; i++) {
        cin >> arr[i];
    }
    vector<int> ans = nextGreaterWeight(arr);
    for (int i = 0; i < N; i++) {
        if (i) cout << ' ';
        cout << ans[i];
    }
    cout << '\n';
    return 0;
}
```
