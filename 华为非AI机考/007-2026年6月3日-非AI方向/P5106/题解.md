## 解题思路

任务数量较小，可以使用回溯枚举所有满足依赖关系的任务执行顺序。

核心算法是拓扑顺序枚举 + 最早可行时间调度：

每个任务用一个二进制状态表示所需的$NPU$卡集合。

在回溯过程中维护：

$done$：已经安排过的任务集合。

$finish[i]$：任务$i$的完成时间。

$free[j]$：第$j$张$NPU$卡最早可用时间。

每次选择一个未安排且依赖任务已经完成的任务$i$，计算它的最早开始时间：

$start=max(依赖任务完成时间, 所有所需NPU卡的可用时间)$

然后得到：

$end=start+duration[i]$

更新相关$NPU$卡的可用时间，继续搜索。用当前已知答案做剪枝，如果当前时间已经不可能更优，就停止搜索。

## 复杂度分析

设任务数量为$m$，$NPU$卡数量为$n$。

最坏情况下需要枚举所有任务顺序，时间复杂度为$O(m! \times m \times n)$。

递归深度为$m$，需要维护$NPU$可用时间和任务完成时间，空间复杂度为$O(m+n)$。

由于题目中$m$较小，该复杂度可以通过。

## 代码实现

### Python

```python
def solve(card_cnt, task_cnt, need, duration, pre):
    ans = [10 ** 18]
    free = [0] * card_cnt
    finish = [0] * task_cnt

    def dfs(done, cur_time):
        # 如果所有任务都已安排，更新答案
        if done == (1 << task_cnt) - 1:
            ans[0] = min(ans[0], cur_time)
            return

        # 当前时间已经不优，直接剪枝
        if cur_time >= ans[0]:
            return

        for i in range(task_cnt):
            if done & (1 << i):
                continue

            # 依赖任务未完成，不能执行
            if pre[i] != -1 and not (done & (1 << pre[i])):
                continue

            start = 0
            if pre[i] != -1:
                start = finish[pre[i]]

            # 任务需要的所有NPU卡都必须空闲
            for c in range(card_cnt):
                if need[i] & (1 << c):
                    start = max(start, free[c])

            end = start + duration[i]
            new_time = max(cur_time, end)

            if new_time >= ans[0]:
                continue

            old_free = free[:]
            old_finish = finish[i]

            # 占用对应NPU卡
            for c in range(card_cnt):
                if need[i] & (1 << c):
                    free[c] = end
            finish[i] = end

            dfs(done | (1 << i), new_time)

            # 回溯恢复现场
            free[:] = old_free
            finish[i] = old_finish

    dfs(0, 0)
    return ans[0]


def main():
    card_cnt, task_cnt = map(int, input().split())

    need = []
    duration = []
    pre = []

    for _ in range(task_cnt):
        nums = list(map(int, input().split()))
        mask = 0
        for x in nums:
            mask |= 1 << x
        need.append(mask)

        duration.append(int(input()))
        pre.append(int(input()))

    print(solve(card_cnt, task_cnt, need, duration, pre))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static int cardCnt, taskCnt;
    static int[] need, duration, pre;
    static int[] free, finish;
    static int ans = Integer.MAX_VALUE;

    static int solve() {
        free = new int[cardCnt];
        finish = new int[taskCnt];
        dfs(0, 0);
        return ans;
    }

    static void dfs(int done, int curTime) {
        // 所有任务都已安排，更新答案
        if (done == (1 << taskCnt) - 1) {
            ans = Math.min(ans, curTime);
            return;
        }

        // 当前结果已经不优，剪枝
        if (curTime >= ans) {
            return;
        }

        for (int i = 0; i < taskCnt; i++) {
            if ((done & (1 << i)) != 0) {
                continue;
            }

            // 依赖任务未完成，不能执行
            if (pre[i] != -1 && (done & (1 << pre[i])) == 0) {
                continue;
            }

            int start = 0;
            if (pre[i] != -1) {
                start = finish[pre[i]];
            }

            // 找到所有所需NPU卡都空闲的最早时间
            for (int c = 0; c < cardCnt; c++) {
                if ((need[i] & (1 << c)) != 0) {
                    start = Math.max(start, free[c]);
                }
            }

            int end = start + duration[i];
            int newTime = Math.max(curTime, end);

            if (newTime >= ans) {
                continue;
            }

            int[] oldFree = free.clone();
            int oldFinish = finish[i];

            // 占用对应NPU卡
            for (int c = 0; c < cardCnt; c++) {
                if ((need[i] & (1 << c)) != 0) {
                    free[c] = end;
                }
            }
            finish[i] = end;

            dfs(done | (1 << i), newTime);

            // 回溯恢复现场
            free = oldFree;
            finish[i] = oldFinish;
        }
    }

    static String nextNonEmptyLine(Scanner sc) {
        String line = sc.nextLine();
        while (line.length() == 0 && sc.hasNextLine()) {
            line = sc.nextLine();
        }
        return line;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        cardCnt = sc.nextInt();
        taskCnt = sc.nextInt();
        sc.nextLine();

        need = new int[taskCnt];
        duration = new int[taskCnt];
        pre = new int[taskCnt];

        for (int i = 0; i < taskCnt; i++) {
            String line = nextNonEmptyLine(sc);
            String[] parts = line.trim().split("\\s+");

            int mask = 0;
            for (String s : parts) {
                int x = Integer.parseInt(s);
                mask |= 1 << x;
            }
            need[i] = mask;

            duration[i] = Integer.parseInt(nextNonEmptyLine(sc).trim());
            pre[i] = Integer.parseInt(nextNonEmptyLine(sc).trim());
        }

        System.out.println(solve());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int cardCnt, taskCnt;
vector<int> needMask, durationTime, preTask;
vector<int> freeTime, finishTime;
int ans = INT_MAX;

void dfs(int done, int curTime) {
    // 所有任务都已安排，更新答案
    if (done == (1 << taskCnt) - 1) {
        ans = min(ans, curTime);
        return;
    }

    // 当前时间已经不优，直接剪枝
    if (curTime >= ans) {
        return;
    }

    for (int i = 0; i < taskCnt; i++) {
        if (done & (1 << i)) {
            continue;
        }

        // 依赖任务未完成，不能执行
        if (preTask[i] != -1 && !(done & (1 << preTask[i]))) {
            continue;
        }

        int start = 0;
        if (preTask[i] != -1) {
            start = finishTime[preTask[i]];
        }

        // 找到所有所需NPU卡都空闲的最早时间
        for (int c = 0; c < cardCnt; c++) {
            if (needMask[i] & (1 << c)) {
                start = max(start, freeTime[c]);
            }
        }

        int endTime = start + durationTime[i];
        int newTime = max(curTime, endTime);

        if (newTime >= ans) {
            continue;
        }

        vector<int> oldFree = freeTime;
        int oldFinish = finishTime[i];

        // 占用对应NPU卡
        for (int c = 0; c < cardCnt; c++) {
            if (needMask[i] & (1 << c)) {
                freeTime[c] = endTime;
            }
        }
        finishTime[i] = endTime;

        dfs(done | (1 << i), newTime);

        // 回溯恢复现场
        freeTime = oldFree;
        finishTime[i] = oldFinish;
    }
}

int solve() {
    freeTime.assign(cardCnt, 0);
    finishTime.assign(taskCnt, 0);
    dfs(0, 0);
    return ans;
}

int main() {
    cin >> cardCnt >> taskCnt;

    needMask.resize(taskCnt);
    durationTime.resize(taskCnt);
    preTask.resize(taskCnt);

    string line;
    getline(cin, line);

    for (int i = 0; i < taskCnt; i++) {
        getline(cin, line);
        stringstream ss(line);

        int x;
        int mask = 0;
        while (ss >> x) {
            mask |= 1 << x;
        }
        needMask[i] = mask;

        cin >> durationTime[i];
        cin >> preTask[i];
        getline(cin, line);
    }

    cout << solve() << endl;
    return 0;
}
```
