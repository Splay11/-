## 解题思路

使用字符串模拟和最长连续段扫描。

将输入按冒号切成 $8$ 个字符串组。压缩时先执行规则 $2$，再执行规则 $1$。

可被 `**` 压缩的候选段需要满足：

该段由若干个连续相同的组组成，并且这个组本身 $4$ 位字符完全相同，例如 $0000$、$1111$、$aaaa$。

从左到右扫描所有组，找到最长的候选段。如果长度相同，只保留最左边的一段。候选段长度可以为 $1$，只要该组本身符合条件即可。

之后重新构造答案：

遇到最长候选段的起点时，加入 `**`，并跳过整段。

其他普通组去掉前导 $0$，如果去完为空，则变成 $0$。

最后用冒号拼接即可。

## 复杂度分析

因为题目固定为 $8$ 个组，每组长度为 $4$，所以时间复杂度为 $O(1)$，空间复杂度为 $O(1)$。

如果按通用形式分析，设有 $n$ 个组，每组长度为 $m$，时间复杂度为 $O(nm)$，空间复杂度为 $O(nm)$。

## 代码实现

### Python

```python
import sys


def is_same_digit_group(s):
    # 判断一个组的4位字符是否完全相同
    return s[0] == s[1] == s[2] == s[3]


def remove_leading_zero(s):
    # 去除前导0，如果全部是0，则返回"0"
    i = 0
    while i < len(s) and s[i] == '0':
        i += 1
    if i == len(s):
        return "0"
    return s[i:]


def compress_identifier(s):
    groups = s.split(':')

    best_start = -1
    best_len = 0
    i = 0

    # 先寻找可以用**压缩的最长连续相同数字组
    while i < 8:
        if is_same_digit_group(groups[i]):
            j = i + 1
            while j < 8 and groups[j] == groups[i]:
                j += 1

            length = j - i
            # 只在更长时更新，长度相同自动保留左边
            if length > best_len:
                best_len = length
                best_start = i

            i = j
        else:
            i += 1

    ans = []
    i = 0

    # 构造最终结果
    while i < 8:
        if i == best_start:
            ans.append("**")
            i += best_len
        else:
            ans.append(remove_leading_zero(groups[i]))
            i += 1

    return ":".join(ans)


def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    s = data[0]
    print(compress_identifier(s))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    // 判断一个组的4位字符是否完全相同
    public static boolean isSameDigitGroup(String s) {
        return s.charAt(0) == s.charAt(1)
            && s.charAt(1) == s.charAt(2)
            && s.charAt(2) == s.charAt(3);
    }

    // 去除前导0，如果全部是0，则返回"0"
    public static String removeLeadingZero(String s) {
        int i = 0;
        while (i < s.length() && s.charAt(i) == '0') {
            i++;
        }

        if (i == s.length()) {
            return "0";
        }

        return s.substring(i);
    }

    public static String compressIdentifier(String s) {
        String[] groups = s.split(":");

        int bestStart = -1;
        int bestLen = 0;
        int i = 0;

        // 先寻找可以用**压缩的最长连续相同数字组
        while (i < 8) {
            if (isSameDigitGroup(groups[i])) {
                int j = i + 1;

                while (j < 8 && groups[j].equals(groups[i])) {
                    j++;
                }

                int len = j - i;

                // 只在更长时更新，长度相同自动保留左边
                if (len > bestLen) {
                    bestLen = len;
                    bestStart = i;
                }

                i = j;
            } else {
                i++;
            }
        }

        StringBuilder ans = new StringBuilder();
        i = 0;

        // 构造最终结果
        while (i < 8) {
            if (i == bestStart) {
                ans.append("**");
                i += bestLen;
            } else {
                ans.append(removeLeadingZero(groups[i]));
                i++;
            }

            if (i < 8) {
                ans.append(":");
            }
        }

        return ans.toString();
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        if (scanner.hasNext()) {
            String s = scanner.next();
            System.out.println(compressIdentifier(s));
        }

        scanner.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// 判断一个组的4位字符是否完全相同
bool isSameDigitGroup(const string& s) {
    return s[0] == s[1] && s[1] == s[2] && s[2] == s[3];
}

// 去除前导0，如果全部是0，则返回"0"
string removeLeadingZero(const string& s) {
    int i = 0;
    while (i < (int)s.size() && s[i] == '0') {
        i++;
    }

    if (i == (int)s.size()) {
        return "0";
    }

    return s.substr(i);
}

// 按冒号分割字符串
vector<string> split(const string& s) {
    vector<string> groups;
    string cur = "";

    for (char c : s) {
        if (c == ':') {
            groups.push_back(cur);
            cur = "";
        } else {
            cur += c;
        }
    }

    groups.push_back(cur);
    return groups;
}

string compressIdentifier(const string& s) {
    vector<string> groups = split(s);

    int bestStart = -1;
    int bestLen = 0;
    int i = 0;

    // 先寻找可以用**压缩的最长连续相同数字组
    while (i < 8) {
        if (isSameDigitGroup(groups[i])) {
            int j = i + 1;

            while (j < 8 && groups[j] == groups[i]) {
                j++;
            }

            int len = j - i;

            // 只在更长时更新，长度相同自动保留左边
            if (len > bestLen) {
                bestLen = len;
                bestStart = i;
            }

            i = j;
        } else {
            i++;
        }
    }

    vector<string> ans;
    i = 0;

    // 构造最终结果
    while (i < 8) {
        if (i == bestStart) {
            ans.push_back("**");
            i += bestLen;
        } else {
            ans.push_back(removeLeadingZero(groups[i]));
            i++;
        }
    }

    string res = "";
    for (int k = 0; k < (int)ans.size(); k++) {
        if (k > 0) {
            res += ":";
        }
        res += ans[k];
    }

    return res;
}

int main() {
    string s;

    if (cin >> s) {
        cout << compressIdentifier(s) << endl;
    }

    return 0;
}
```
