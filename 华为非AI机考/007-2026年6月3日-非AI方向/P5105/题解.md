## 解题思路

本题是一棵带边权的树，有 $N$ 个节点，每个节点有收益 $g_i$，炸弹影响范围为 $K$。

如果炸弹放在节点 $x$，那么所有到 $x$ 的路径距离不超过 $K$ 的节点都会被摧毁，收益为这些节点收益之和。

由于树上任意两个节点之间路径唯一，可以枚举每个节点作为炸弹放置点，然后从该节点开始做一次 $DFS$，统计距离不超过 $K$ 的所有节点收益。

核心步骤：

1. 建图，保存每条边的终点和距离。
2. 枚举每个节点 $i$ 作为炸弹放置点。
3. 从 $i$ 开始 $DFS$，累计当前距离 $dist$。
4. 如果 $dist \le K$，加入该节点收益；否则停止继续搜索。
5. 取所有放置点收益的最大值。

使用的算法是树上深度优先搜索 $DFS$。

## 复杂度分析

设节点数为 $N$。

每个节点都作为起点做一次 $DFS$，一次 $DFS$ 最多访问 $N$ 个节点。

时间复杂度为 $O(N^2)$。

邻接表存树，空间复杂度为 $O(N)$。

由于 $N \le 100$，该复杂度完全可以通过。

## 代码实现

### Python

```python
import sys

def dfs(u, parent, dist, k, values, graph):
    # 超出影响范围，不再继续搜索
    if dist > k:
        return 0

    # 当前节点被摧毁，加入收益
    total = values[u]

    # 继续搜索相邻节点
    for v, w in graph[u]:
        if v != parent:
            total += dfs(v, u, dist + w, k, values, graph)

    return total

def solve(n, k, values, graph):
    ans = 0

    # 枚举每个节点作为炸弹放置点
    for i in range(n):
        cur = dfs(i, -1, 0, k, values, graph)
        ans = max(ans, cur)

    return ans

def main():
    data = list(map(int, sys.stdin.read().split()))
    idx = 0

    n = data[idx]
    k = data[idx + 1]
    idx += 2

    values = data[idx:idx + n]
    idx += n

    graph = [[] for _ in range(n)]

    for _ in range(n - 1):
        u = data[idx]
        v = data[idx + 1]
        w = data[idx + 2]
        idx += 3

        # 无向树，双向建边
        graph[u].append((v, w))
        graph[v].append((u, w))

    print(solve(n, k, values, graph))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static class Edge {
        int to;
        long w;

        Edge(int to, long w) {
            this.to = to;
            this.w = w;
        }
    }

    static long dfs(int u, int parent, long dist, long k, long[] values, List<Edge>[] graph) {
        // 超出影响范围，不再继续搜索
        if (dist > k) {
            return 0;
        }

        // 当前节点被摧毁，加入收益
        long total = values[u];

        // 继续搜索相邻节点
        for (Edge e : graph[u]) {
            if (e.to != parent) {
                total += dfs(e.to, u, dist + e.w, k, values, graph);
            }
        }

        return total;
    }

    static long solve(int n, long k, long[] values, List<Edge>[] graph) {
        long ans = 0;

        // 枚举每个节点作为炸弹放置点
        for (int i = 0; i < n; i++) {
            long cur = dfs(i, -1, 0, k, values, graph);
            ans = Math.max(ans, cur);
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        long k = sc.nextLong();

        long[] values = new long[n];
        for (int i = 0; i < n; i++) {
            values[i] = sc.nextLong();
        }

        List<Edge>[] graph = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<Edge>();
        }

        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            long w = sc.nextLong();

            // 无向树，双向建边
            graph[u].add(new Edge(v, w));
            graph[v].add(new Edge(u, w));
        }

        System.out.println(solve(n, k, values, graph));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

struct Edge {
    int to;
    long long w;
};

long long dfs(int u, int parent, long long dist, long long k,
              vector<long long>& values, vector<vector<Edge>>& graph) {
    // 超出影响范围，不再继续搜索
    if (dist > k) {
        return 0;
    }

    // 当前节点被摧毁，加入收益
    long long total = values[u];

    // 继续搜索相邻节点
    for (Edge e : graph[u]) {
        if (e.to != parent) {
            total += dfs(e.to, u, dist + e.w, k, values, graph);
        }
    }

    return total;
}

long long solve(int n, long long k, vector<long long>& values,
                vector<vector<Edge>>& graph) {
    long long ans = 0;

    // 枚举每个节点作为炸弹放置点
    for (int i = 0; i < n; i++) {
        long long cur = dfs(i, -1, 0, k, values, graph);
        if (cur > ans) {
            ans = cur;
        }
    }

    return ans;
}

int main() {
    int n;
    long long k;
    cin >> n >> k;

    vector<long long> values(n);
    for (int i = 0; i < n; i++) {
        cin >> values[i];
    }

    vector<vector<Edge>> graph(n);

    for (int i = 0; i < n - 1; i++) {
        int u, v;
        long long w;
        cin >> u >> v >> w;

        // 无向树，双向建边
        graph[u].push_back({v, w});
        graph[v].push_back({u, w});
    }

    cout << solve(n, k, values, graph) << endl;

    return 0;
}
```
