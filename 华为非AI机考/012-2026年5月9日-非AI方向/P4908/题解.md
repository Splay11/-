## 解题思路

题目要求找最长连续区间 $[L,R]$，使区间内任意两个状态值的差的绝对值都不超过 $d$。

对于一个区间来说：

$任意两数差值 \le d$

等价于：

$区间最大值 - 区间最小值 \le d$

因此问题可以转化为：

维护一个滑动窗口 $[l,r]$，保证窗口内的最大值和最小值之差不超过 $d$。

使用两个单调队列：

* 一个单调递减队列维护当前窗口最大值
* 一个单调递增队列维护当前窗口最小值

当 $max - min > d$ 时，不断移动左端点 $l$，直到窗口重新合法。

每次窗口合法后，用当前长度更新答案。只有当当前长度更长时才更新答案，长度相同不更新，这样可以保证起始编号最小。

如果最终最长长度为 $1$，输出 $1\ 1$。

## 复杂度分析

每个元素最多进队和出队各 $1$ 次。

时间复杂度为 $O(n)$。

空间复杂度为 $O(n)$。

复杂度可以满足 $n \le 100000$ 的要求。

## 代码实现

### Python

```python
from collections import deque
import sys


def find_longest_segment(a, d):
    n = len(a)

    # max_q 维护单调递减队列，队首是当前窗口最大值下标
    max_q = deque()

    # min_q 维护单调递增队列，队首是当前窗口最小值下标
    min_q = deque()

    l = 0
    best_l = 0
    best_r = 0

    for r in range(n):
        # 插入当前元素，维护最大值队列
        while max_q and a[max_q[-1]] <= a[r]:
            max_q.pop()
        max_q.append(r)

        # 插入当前元素，维护最小值队列
        while min_q and a[min_q[-1]] >= a[r]:
            min_q.pop()
        min_q.append(r)

        # 如果窗口不合法，移动左端点
        while a[max_q[0]] - a[min_q[0]] > d:
            if max_q[0] == l:
                max_q.popleft()
            if min_q[0] == l:
                min_q.popleft()
            l += 1

        # 只在长度更长时更新，保证起点最小
        if r - l > best_r - best_l:
            best_l = l
            best_r = r

    return best_l + 1, best_r + 1


def main():
    data = sys.stdin.read().strip().split()
    n = int(data[0])
    d = int(data[1])

    a = list(map(int, data[2:2 + n]))

    l, r = find_longest_segment(a, d)
    print(l, r)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static int[] findLongestSegment(int[] a, int d) {
        int n = a.length;

        // maxDeque 维护单调递减队列，队首是当前窗口最大值下标
        Deque<Integer> maxDeque = new ArrayDeque<>();

        // minDeque 维护单调递增队列，队首是当前窗口最小值下标
        Deque<Integer> minDeque = new ArrayDeque<>();

        int l = 0;
        int bestL = 0;
        int bestR = 0;

        for (int r = 0; r < n; r++) {
            // 插入当前元素，维护最大值队列
            while (!maxDeque.isEmpty() && a[maxDeque.peekLast()] <= a[r]) {
                maxDeque.pollLast();
            }
            maxDeque.offerLast(r);

            // 插入当前元素，维护最小值队列
            while (!minDeque.isEmpty() && a[minDeque.peekLast()] >= a[r]) {
                minDeque.pollLast();
            }
            minDeque.offerLast(r);

            // 如果窗口不合法，移动左端点
            while (a[maxDeque.peekFirst()] - a[minDeque.peekFirst()] > d) {
                if (maxDeque.peekFirst() == l) {
                    maxDeque.pollFirst();
                }
                if (minDeque.peekFirst() == l) {
                    minDeque.pollFirst();
                }
                l++;
            }

            // 只在长度更长时更新，保证起点最小
            if (r - l > bestR - bestL) {
                bestL = l;
                bestR = r;
            }
        }

        return new int[]{bestL + 1, bestR + 1};
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();

        int n = fs.nextInt();
        int d = fs.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = fs.nextInt();
        }

        int[] ans = findLongestSegment(a, d);
        System.out.println(ans[0] + " " + ans[1]);
    }

    // 使用 BufferedReader 读取，适合本题数据范围
    static class FastScanner {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;

        String next() throws Exception {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(br.readLine());
            }
            return st.nextToken();
        }

        int nextInt() throws Exception {
            return Integer.parseInt(next());
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

pair<int, int> findLongestSegment(vector<int>& a, int d) {
    int n = a.size();

    // maxDeque 维护单调递减队列，队首是当前窗口最大值下标
    deque<int> maxDeque;

    // minDeque 维护单调递增队列，队首是当前窗口最小值下标
    deque<int> minDeque;

    int l = 0;
    int bestL = 0;
    int bestR = 0;

    for (int r = 0; r < n; r++) {
        // 插入当前元素，维护最大值队列
        while (!maxDeque.empty() && a[maxDeque.back()] <= a[r]) {
            maxDeque.pop_back();
        }
        maxDeque.push_back(r);

        // 插入当前元素，维护最小值队列
        while (!minDeque.empty() && a[minDeque.back()] >= a[r]) {
            minDeque.pop_back();
        }
        minDeque.push_back(r);

        // 如果窗口不合法，移动左端点
        while (a[maxDeque.front()] - a[minDeque.front()] > d) {
            if (maxDeque.front() == l) {
                maxDeque.pop_front();
            }
            if (minDeque.front() == l) {
                minDeque.pop_front();
            }
            l++;
        }

        // 只在长度更长时更新，保证起点最小
        if (r - l > bestR - bestL) {
            bestL = l;
            bestR = r;
        }
    }

    return {bestL + 1, bestR + 1};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, d;
    cin >> n >> d;

    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    pair<int, int> ans = findLongestSegment(a, d);
    cout << ans.first << " " << ans.second << '\n';

    return 0;
}
```
