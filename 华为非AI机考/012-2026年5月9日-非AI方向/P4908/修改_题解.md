# 双指针朴素解法

这道题的正解是 $单调队列$ 维护滑动窗口最值，但是我们用朴素解法 $双指针 + 每次重算窗口 min/max$ 也能在考试时拿到一定的分数。

题意：找最长连续区间 $[L,R]$，使区间内任意两数差的绝对值都不超过 $d$。这等价于「区间最大值 $-$ 区间最小值 $\le d$」。多解时取起始编号最小的；若最长长度只有 $1$，输出 $1\ 1$。

朴素做法：右端点 $r$ 从左到右扩大窗口；每次用线性扫描得到当前 $[l,r]$ 的最大值与最小值，若 $max-min>d$，就不断右移左端点 $l$ 并重新扫描最值，直到窗口合法。再按长度更新答案（只在更长时更新，保证起点最小）。

每次重算最值使整体约 $O(n^2)$。小数据可以过；$n$ 到 $2\times 10^5$ 时会超时，需要后面的单调队列做法。

## 双指针解法

```python
def find_longest_segment(a, d):
    n = len(a)
    l = 0
    best_l, best_r = 0, 0

    for r in range(n):
        # 窗口不合法时右移左端点，每次重算 min/max
        while l <= r:
            mx = a[l]
            mn = a[l]
            for i in range(l, r + 1):
                if a[i] > mx:
                    mx = a[i]
                if a[i] < mn:
                    mn = a[i]
            if mx - mn <= d:
                break
            l += 1

        # 只在长度更长时更新，保证起点最小
        if r - l > best_r - best_l:
            best_l, best_r = l, r

    return best_l + 1, best_r + 1


def main():
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    L, R = find_longest_segment(a, d)
    print(L, R)


if __name__ == "__main__":
    main()
```

```cpp
    int n = (int)a.size();
    int l = 0;

            bestL = l;
            bestR = r;
        }
    }
    return {bestL + 1, bestR + 1};
}
```

# 正确解法

## 题目分析

仍用双指针维护合法窗口 $[l,r]$，但用两个单调队列在 $O(1)$ 摊还时间内取窗口最大值与最小值：

* 单调递减队列维护最大值下标
* 单调递增队列维护最小值下标

右端点 $r$ 加入时更新两队；若 $max-min>d$，弹出队首中等于 $l$ 的下标并令 $l++$，直到窗口合法。只在长度严格变长时更新答案，保证起始编号最小。

## 复杂度分析

* 时间复杂度：$O(n)$，每个下标进出队列各一次。
* 空间复杂度：$O(n)$。

可满足 $n\le 2\times 10^5$。

## 代码实现

### Python

```python
from collections import deque


def find_longest_segment(a, d):
    n = len(a)
    max_q = deque()  # 队首为窗口最大值下标
    min_q = deque()  # 队首为窗口最小值下标
    l = 0
    best_l, best_r = 0, 0

    for r in range(n):
        while max_q and a[max_q[-1]] <= a[r]:
            max_q.pop()
        max_q.append(r)

        while min_q and a[min_q[-1]] >= a[r]:
            min_q.pop()
        min_q.append(r)

        while a[max_q[0]] - a[min_q[0]] > d:
            if max_q[0] == l:
                max_q.popleft()
            if min_q[0] == l:
                min_q.popleft()
            l += 1

        if r - l > best_r - best_l:
            best_l, best_r = l, r

    return best_l + 1, best_r + 1


def main():
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    L, R = find_longest_segment(a, d)
    print(L, R)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Main {

    static int[] findLongestSegment(int[] a, int d) {
        int n = a.length;
        Deque<Integer> maxDeque = new ArrayDeque<>();
        Deque<Integer> minDeque = new ArrayDeque<>();
        int l = 0;
        int bestL = 0, bestR = 0;

        for (int r = 0; r < n; r++) {
            while (!maxDeque.isEmpty() && a[maxDeque.peekLast()] <= a[r]) {
                maxDeque.pollLast();
            }
            maxDeque.offerLast(r);

            while (!minDeque.isEmpty() && a[minDeque.peekLast()] >= a[r]) {
                minDeque.pollLast();
            }
            minDeque.offerLast(r);

            while (a[maxDeque.peekFirst()] - a[minDeque.peekFirst()] > d) {
                if (maxDeque.peekFirst() == l) {
                    maxDeque.pollFirst();
                }
                if (minDeque.peekFirst() == l) {
                    minDeque.pollFirst();
                }
                l++;
            }

            if (r - l > bestR - bestL) {
                bestL = l;
                bestR = r;
            }
        }
        return new int[]{bestL + 1, bestR + 1};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] ans = findLongestSegment(a, d);
        System.out.println(ans[0] + " " + ans[1]);
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <deque>
using namespace std;

pair<int, int> findLongestSegment(const vector<int>& a, int d) {
    int n = (int)a.size();
    deque<int> maxDeque;  // 队首为窗口最大值下标
    deque<int> minDeque;  // 队首为窗口最小值下标
    int l = 0;
    int bestL = 0, bestR = 0;

    for (int r = 0; r < n; r++) {
        while (!maxDeque.empty() && a[maxDeque.back()] <= a[r]) {
            maxDeque.pop_back();
        }
        maxDeque.push_back(r);

        while (!minDeque.empty() && a[minDeque.back()] >= a[r]) {
            minDeque.pop_back();
        }
        minDeque.push_back(r);

        while (a[maxDeque.front()] - a[minDeque.front()] > d) {
            if (maxDeque.front() == l) {
                maxDeque.pop_front();
            }
            if (minDeque.front() == l) {
                minDeque.pop_front();
            }
            l++;
        }

        if (r - l > bestR - bestL) {
            bestL = l;
            bestR = r;
        }
    }
    return {bestL + 1, bestR + 1};
}
```
