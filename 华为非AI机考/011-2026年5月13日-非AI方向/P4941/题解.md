## 解题思路

先用 $BFS$ 按层序序列建出两棵二叉树，注意只有非空节点才继续读取它的左右孩子。

合并时使用 $DFS$ 判断冲突：

当两个位置都有节点时，节点值必须相同，否则冲突，不能合并；

当某个位置为空时，可以直接接上另一棵树对应位置的子树；

优先尝试把树 $2$ 合并到树 $1$ 中，也就是在树 $1$ 中找到与树 $2$ 根节点值相同的节点，从这里开始整体合并；

如果失败，再尝试把树 $1$ 合并到树 $2$ 中；

如果两种方式都失败，输出 $0$。

层序输出时保留中间的 $null$，最后删除末尾多余的 $null$。

## 复杂度分析

设两棵树节点数分别为 $n$ 和 $m$。

查找节点最多遍历整棵树，合并检查和真正合并也最多遍历两棵树相关部分，因此时间复杂度为 $O(n+m)$。

建树、递归和层序输出需要存储节点与队列，因此空间复杂度为 $O(n+m)$。

## 代码实现

### Python

```python
import sys
from collections import deque


class Node:
    def __init__(self, val):
        self.val = val
        self.left = None
        self.right = None


def build_tree(line):
    vals = line.strip().split()
    if not vals or vals[0] == "null":
        return None

    root = Node(vals[0])
    q = deque([root])
    i = 1

    # 按层序建树，只有非空节点才会继续读取左右孩子
    while q and i < len(vals):
        cur = q.popleft()

        if i < len(vals):
            if vals[i] != "null":
                cur.left = Node(vals[i])
                q.append(cur.left)
            i += 1

        if i < len(vals):
            if vals[i] != "null":
                cur.right = Node(vals[i])
                q.append(cur.right)
            i += 1

    return root


def find_node(root, val):
    if root is None:
        return None
    if root.val == val:
        return root

    left = find_node(root.left, val)
    if left is not None:
        return left

    return find_node(root.right, val)


def can_merge(a, b):
    # 有一边为空，不会冲突
    if a is None or b is None:
        return True

    # 两边都有节点时，值必须相同
    if a.val != b.val:
        return False

    return can_merge(a.left, b.left) and can_merge(a.right, b.right)


def merge_tree(a, b):
    # 空位置直接接上另一棵树
    if a is None:
        return b
    if b is None:
        return a

    # 此时两个节点值相同，继续合并左右子树
    a.left = merge_tree(a.left, b.left)
    a.right = merge_tree(a.right, b.right)
    return a


def level_order(root):
    if root is None:
        return "null"

    res = []
    q = deque([root])

    # 层序输出，并保留中间的 null
    while q:
        cur = q.popleft()
        if cur is None:
            res.append("null")
        else:
            res.append(cur.val)
            q.append(cur.left)
            q.append(cur.right)

    # 删除末尾多余的 null
    while res and res[-1] == "null":
        res.pop()

    return " ".join(res)


def solve():
    lines = [line.strip() for line in sys.stdin.read().splitlines() if line.strip()]
    if len(lines) < 2:
        return

    tree1 = build_tree(lines[0])
    tree2 = build_tree(lines[1])

    if tree1 is None:
        print(level_order(tree2))
        return
    if tree2 is None:
        print(level_order(tree1))
        return

    # 优先尝试树 2 合并到树 1
    pos = find_node(tree1, tree2.val)
    if pos is not None and can_merge(pos, tree2):
        merge_tree(pos, tree2)
        print(level_order(tree1))
        return

    # 再尝试树 1 合并到树 2
    pos = find_node(tree2, tree1.val)
    if pos is not None and can_merge(pos, tree1):
        merge_tree(pos, tree1)
        print(level_order(tree2))
        return

    print(0)


if __name__ == "__main__":
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static class Node {
        String val;
        Node left;
        Node right;

        Node(String val) {
            this.val = val;
        }
    }

    static Node buildTree(String line) {
        String[] vals = line.trim().split("\\s+");
        if (vals.length == 0 || vals[0].equals("null")) {
            return null;
        }

        Node root = new Node(vals[0]);
        Queue<Node> q = new ArrayDeque<>();
        q.offer(root);
        int i = 1;

        // 按层序建树，只有非空节点才会继续读取左右孩子
        while (!q.isEmpty() && i < vals.length) {
            Node cur = q.poll();

            if (i < vals.length) {
                if (!vals[i].equals("null")) {
                    cur.left = new Node(vals[i]);
                    q.offer(cur.left);
                }
                i++;
            }

            if (i < vals.length) {
                if (!vals[i].equals("null")) {
                    cur.right = new Node(vals[i]);
                    q.offer(cur.right);
                }
                i++;
            }
        }

        return root;
    }

    static Node findNode(Node root, String val) {
        if (root == null) {
            return null;
        }
        if (root.val.equals(val)) {
            return root;
        }

        Node left = findNode(root.left, val);
        if (left != null) {
            return left;
        }

        return findNode(root.right, val);
    }

    static boolean canMerge(Node a, Node b) {
        // 有一边为空，不会冲突
        if (a == null || b == null) {
            return true;
        }

        // 两边都有节点时，值必须相同
        if (!a.val.equals(b.val)) {
            return false;
        }

        return canMerge(a.left, b.left) && canMerge(a.right, b.right);
    }

    static Node mergeTree(Node a, Node b) {
        // 空位置直接接上另一棵树
        if (a == null) {
            return b;
        }
        if (b == null) {
            return a;
        }

        // 此时两个节点值相同，继续合并左右子树
        a.left = mergeTree(a.left, b.left);
        a.right = mergeTree(a.right, b.right);
        return a;
    }

    static String levelOrder(Node root) {
        if (root == null) {
            return "null";
        }

        List<String> res = new ArrayList<>();
        Queue<Node> q = new LinkedList<>();
        q.offer(root);

        // 层序输出，并保留中间的 null
        while (!q.isEmpty()) {
            Node cur = q.poll();
            if (cur == null) {
                res.add("null");
            } else {
                res.add(cur.val);
                q.offer(cur.left);
                q.offer(cur.right);
            }
        }

        // 删除末尾多余的 null
        int end = res.size() - 1;
        while (end >= 0 && res.get(end).equals("null")) {
            end--;
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i <= end; i++) {
            if (i > 0) {
                sb.append(" ");
            }
            sb.append(res.get(i));
        }

        return sb.toString();
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        List<String> lines = new ArrayList<>();
        String line;

        while ((line = br.readLine()) != null) {
            line = line.trim();
            if (!line.isEmpty()) {
                lines.add(line);
            }
        }

        if (lines.size() < 2) {
            return;
        }

        Node tree1 = buildTree(lines.get(0));
        Node tree2 = buildTree(lines.get(1));

        if (tree1 == null) {
            System.out.println(levelOrder(tree2));
            return;
        }
        if (tree2 == null) {
            System.out.println(levelOrder(tree1));
            return;
        }

        // 优先尝试树 2 合并到树 1
        Node pos = findNode(tree1, tree2.val);
        if (pos != null && canMerge(pos, tree2)) {
            mergeTree(pos, tree2);
            System.out.println(levelOrder(tree1));
            return;
        }

        // 再尝试树 1 合并到树 2
        pos = findNode(tree2, tree1.val);
        if (pos != null && canMerge(pos, tree1)) {
            mergeTree(pos, tree1);
            System.out.println(levelOrder(tree2));
            return;
        }

        System.out.println(0);
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Node {
    string val;
    Node* left;
    Node* right;

    Node(string v) {
        val = v;
        left = nullptr;
        right = nullptr;
    }
};

Node* buildTree(string line) {
    stringstream ss(line);
    vector<string> vals;
    string x;

    while (ss >> x) {
        vals.push_back(x);
    }

    if (vals.empty() || vals[0] == "null") {
        return nullptr;
    }

    Node* root = new Node(vals[0]);
    queue<Node*> q;
    q.push(root);
    int i = 1;

    // 按层序建树，只有非空节点才会继续读取左右孩子
    while (!q.empty() && i < (int)vals.size()) {
        Node* cur = q.front();
        q.pop();

        if (i < (int)vals.size()) {
            if (vals[i] != "null") {
                cur->left = new Node(vals[i]);
                q.push(cur->left);
            }
            i++;
        }

        if (i < (int)vals.size()) {
            if (vals[i] != "null") {
                cur->right = new Node(vals[i]);
                q.push(cur->right);
            }
            i++;
        }
    }

    return root;
}

Node* findNode(Node* root, string val) {
    if (root == nullptr) {
        return nullptr;
    }
    if (root->val == val) {
        return root;
    }

    Node* left = findNode(root->left, val);
    if (left != nullptr) {
        return left;
    }

    return findNode(root->right, val);
}

bool canMerge(Node* a, Node* b) {
    // 有一边为空，不会冲突
    if (a == nullptr || b == nullptr) {
        return true;
    }

    // 两边都有节点时，值必须相同
    if (a->val != b->val) {
        return false;
    }

    return canMerge(a->left, b->left) && canMerge(a->right, b->right);
}

Node* mergeTree(Node* a, Node* b) {
    // 空位置直接接上另一棵树
    if (a == nullptr) {
        return b;
    }
    if (b == nullptr) {
        return a;
    }

    // 此时两个节点值相同，继续合并左右子树
    a->left = mergeTree(a->left, b->left);
    a->right = mergeTree(a->right, b->right);
    return a;
}

string levelOrder(Node* root) {
    if (root == nullptr) {
        return "null";
    }

    vector<string> res;
    queue<Node*> q;
    q.push(root);

    // 层序输出，并保留中间的 null
    while (!q.empty()) {
        Node* cur = q.front();
        q.pop();

        if (cur == nullptr) {
            res.push_back("null");
        } else {
            res.push_back(cur->val);
            q.push(cur->left);
            q.push(cur->right);
        }
    }

    // 删除末尾多余的 null
    while (!res.empty() && res.back() == "null") {
        res.pop_back();
    }

    string ans;
    for (int i = 0; i < (int)res.size(); i++) {
        if (i > 0) {
            ans += " ";
        }
        ans += res[i];
    }

    return ans;
}

int main() {
    vector<string> lines;
    string line;

    while (getline(cin, line)) {
        stringstream ss(line);
        string tmp;
        if (ss >> tmp) {
            lines.push_back(line);
        }
    }

    if (lines.size() < 2) {
        return 0;
    }

    Node* tree1 = buildTree(lines[0]);
    Node* tree2 = buildTree(lines[1]);

    if (tree1 == nullptr) {
        cout << levelOrder(tree2) << endl;
        return 0;
    }
    if (tree2 == nullptr) {
        cout << levelOrder(tree1) << endl;
        return 0;
    }

    // 优先尝试树 2 合并到树 1
    Node* pos = findNode(tree1, tree2->val);
    if (pos != nullptr && canMerge(pos, tree2)) {
        mergeTree(pos, tree2);
        cout << levelOrder(tree1) << endl;
        return 0;
    }

    // 再尝试树 1 合并到树 2
    pos = findNode(tree2, tree1->val);
    if (pos != nullptr && canMerge(pos, tree1)) {
        mergeTree(pos, tree1);
        cout << levelOrder(tree2) << endl;
        return 0;
    }

    cout << 0 << endl;
    return 0;
}
```
