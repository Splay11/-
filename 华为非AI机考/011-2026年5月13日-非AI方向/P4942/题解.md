## 解题思路

本题需要在满足电量限制的前提下，求从起点 $S$ 到终点 $T$ 的最短总耗时。

可以使用扩展状态图上的 $Dijkstra$ 算法。

将状态定义为：

$(u, e)$

其中：

* $u$ 表示当前所在路口
* $e$ 表示当前剩余电量

### 状态转移

对于当前状态 $(u, e)$：

1. 经过道路

   若存在道路：

   $(u \rightarrow v)$

   通行耗时为 $time$，耗电为 $cost$。

   当：

   $e \ge cost$

   时，可以转移到：

   $(v, e - cost)$

   新增耗时：

   $time$

2. 在充电站充电

   如果当前路口 $u$ 是充电站，并且：

   $e < maxE$

   则可以花费 $1$ 个单位时间，将电量恢复为满电：

   $(u, e) \rightarrow (u, maxE)$

### 答案

终点 $T$ 可以以任意剩余电量到达，因此答案为：

$\min dist[T][e]$

其中：

$0 \le e \le maxE$

若所有状态都不可达，则输出：

$-1$

---

## 复杂度分析

设最大电量为 $maxE$。

状态数为：

$N \times (maxE + 1)$

每条道路会在不同电量状态下被访问，因此总转移数约为：

$M \times maxE$

使用 $Dijkstra$ 算法：

* 时间复杂度：

  $O((N \times maxE + M \times maxE)\log(N \times maxE))$

* 空间复杂度：

  $O(N \times maxE + M)$

由于：

$N \le 1000$

$M \le 1000$

$maxE \le 100$

因此该复杂度可以通过。

---

## 代码实现

### Python

```python
import sys
import heapq


def min_time(n, graph, charge, s, t, start_e, max_e):
    INF = 10**30

    # dist[u][e] 表示到达路口 u，剩余电量为 e 的最短耗时
    dist = [[INF] * (max_e + 1) for _ in range(n)]
    dist[s][start_e] = 0

    heap = []
    heapq.heappush(heap, (0, s, start_e))

    while heap:
        cur_time, u, energy = heapq.heappop(heap)

        # 舍弃过期状态
        if cur_time != dist[u][energy]:
            continue

        # 在充电站选择充满电
        if charge[u] and energy < max_e:
            new_time = cur_time + 1
            if new_time < dist[u][max_e]:
                dist[u][max_e] = new_time
                heapq.heappush(heap, (new_time, u, max_e))

        # 尝试经过道路
        for v, road_time, cost in graph[u]:
            if energy >= cost:
                new_energy = energy - cost
                new_time = cur_time + road_time

                if new_time < dist[v][new_energy]:
                    dist[v][new_energy] = new_time
                    heapq.heappush(heap, (new_time, v, new_energy))

    ans = min(dist[t])
    return -1 if ans == INF else ans


def main():
    data = list(map(int, sys.stdin.read().split()))
    idx = 0

    n = data[idx]
    idx += 1
    k = data[idx]
    idx += 1
    m = data[idx]
    idx += 1
    s = data[idx]
    idx += 1
    t = data[idx]
    idx += 1
    start_e = data[idx]
    idx += 1
    max_e = data[idx]
    idx += 1

    graph = [[] for _ in range(n)]

    for _ in range(m):
        u = data[idx]
        v = data[idx + 1]
        road_time = data[idx + 2]
        cost = data[idx + 3]
        idx += 4
        graph[u].append((v, road_time, cost))

    charge = [False] * n
    for _ in range(k):
        cp = data[idx]
        idx += 1
        charge[cp] = True

    print(min_time(n, graph, charge, s, t, start_e, max_e))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    static class Edge {
        int to;
        int time;
        int cost;

        Edge(int to, int time, int cost) {
            this.to = to;
            this.time = time;
            this.cost = cost;
        }
    }

    static long minTime(int n, List<Edge>[] graph, boolean[] charge,
                        int s, int t, int startE, int maxE) {
        long INF = Long.MAX_VALUE / 4;

        // dist[u][e] 表示到达路口 u，剩余电量为 e 的最短耗时
        long[][] dist = new long[n][maxE + 1];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], INF);
        }

        PriorityQueue<long[]> pq = new PriorityQueue<>(
                (a, b) -> Long.compare(a[0], b[0])
        );

        dist[s][startE] = 0;
        pq.offer(new long[]{0, s, startE});

        while (!pq.isEmpty()) {
            long[] cur = pq.poll();

            long curTime = cur[0];
            int u = (int) cur[1];
            int energy = (int) cur[2];

            // 舍弃过期状态
            if (curTime != dist[u][energy]) {
                continue;
            }

            // 在充电站选择充满电
            if (charge[u] && energy < maxE) {
                long newTime = curTime + 1;

                if (newTime < dist[u][maxE]) {
                    dist[u][maxE] = newTime;
                    pq.offer(new long[]{newTime, u, maxE});
                }
            }

            // 尝试经过道路
            for (Edge edge : graph[u]) {
                if (energy >= edge.cost) {
                    int newEnergy = energy - edge.cost;
                    long newTime = curTime + edge.time;

                    if (newTime < dist[edge.to][newEnergy]) {
                        dist[edge.to][newEnergy] = newTime;
                        pq.offer(new long[]{newTime, edge.to, newEnergy});
                    }
                }
            }
        }

        long ans = INF;
        for (int e = 0; e <= maxE; e++) {
            ans = Math.min(ans, dist[t][e]);
        }

        return ans == INF ? -1 : ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int k = sc.nextInt();
        int m = sc.nextInt();
        int s = sc.nextInt();
        int t = sc.nextInt();
        int startE = sc.nextInt();
        int maxE = sc.nextInt();

        List<Edge>[] graph = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int from = sc.nextInt();
            int to = sc.nextInt();
            int time = sc.nextInt();
            int cost = sc.nextInt();

            graph[from].add(new Edge(to, time, cost));
        }

        boolean[] charge = new boolean[n];
        for (int i = 0; i < k; i++) {
            int cp = sc.nextInt();
            charge[cp] = true;
        }

        System.out.println(minTime(n, graph, charge, s, t, startE, maxE));
    }
}
```

```cpp
        return time > other.time;
    }
};

        pq.pop();

        long long curTime = cur.time;
        int u = cur.node;
        int energy = cur.energy;

        // 舍弃过期状态
        if (curTime != dist[u][energy]) {
            continue;
        }

        // 在充电站选择充满电
        if (charge[u] && energy < maxE) {
            long long newTime = curTime + 1;

            if (newTime < dist[u][maxE]) {
                dist[u][maxE] = newTime;
                pq.push({newTime, u, maxE});
            }
        }

        // 尝试经过道路
        for (Edge edge : graph[u]) {
            if (energy >= edge.cost) {
                int newEnergy = energy - edge.cost;
                long long newTime = curTime + edge.time;

                if (newTime < dist[edge.to][newEnergy]) {
                    dist[edge.to][newEnergy] = newTime;
                    pq.push({newTime, edge.to, newEnergy});
                }
            }
        }
    }

    long long ans = INF;
    for (int e = 0; e <= maxE; e++) {
        ans = min(ans, dist[t][e]);
    }
```
