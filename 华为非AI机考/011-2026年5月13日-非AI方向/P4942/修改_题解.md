# DFS暴力解法

这道题的正解是 $扩展状态Dijkstra$，但是我们用朴素解法 $DFS$ 枚举路径也能在考试时拿到一定的分数。

（以下为朴素解法）

状态可以看成 $(u,e)$：$u$ 是当前路口，$e$ 是剩余电量。从 $(S,E)$ 出发，目标是到达路口 $T$ 的最小耗时（到达时电量任意）。

朴素做法：

1. 用 $DFS$ 从当前状态继续走：若在充电站且未满电，可花 $1$ 时间充满到 $maxE$；也可沿出边走，前提是 $e\ge cost$。
2. 用 $best[u][e]$ 记录到达该状态的已见最小耗时，更差的状态直接剪掉，避免在环上死循环。
3. 每次到达终点 $T$ 就用当前耗时更新答案。

图较小、分叉不多时能算出正确答案；$N、M$ 较大时搜索很慢，需要后面的正解。

## DFS解法

```python
INF = 10 ** 30


def min_time(n, graph, charge, s, t, start_e, max_e):
    # best[u][e]：到达 (u,e) 的最小耗时
    best = [[INF] * (max_e + 1) for _ in range(n)]
    ans = [INF]

    def dfs(u, energy, cur_time):
        if cur_time >= ans[0]:
            return
        if cur_time >= best[u][energy]:
            return
        best[u][energy] = cur_time

        if u == t:
            ans[0] = min(ans[0], cur_time)
            return

        # 充电站可选择充满电
        if charge[u] and energy < max_e:
            dfs(u, max_e, cur_time + 1)

        # 尝试走每条出边
        for v, road_time, cost in graph[u]:
            if energy >= cost:
                dfs(v, energy - cost, cur_time + road_time)

    dfs(s, start_e, 0)
    return -1 if ans[0] == INF else ans[0]


def main():
    n, k, m, s, t, start_e, max_e = map(int, input().split())
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, road_time, cost = map(int, input().split())
        graph[u].append((v, road_time, cost))
    charge = [False] * n
    if k > 0:
        for cp in map(int, input().split()):
            charge[cp] = True
    print(min_time(n, graph, charge, s, t, start_e, max_e))


if __name__ == "__main__":
    main()
```

# 正确解法

## 题目分析

在满足电量约束下求 $S$ 到 $T$ 的最短总耗时，把状态扩成 $(u,e)$，在状态图上跑 $Dijkstra$。

* $u$：当前路口
* $e$：当前剩余电量

转移有两类：

1. 走路：存在边 $u\to v$，耗时 $time$、耗电 $cost$，当 $e\ge cost$ 时可到 $(v,e-cost)$，耗时加 $time$。
2. 充电：若 $u$ 是充电站且 $e < maxE$，可花 $1$ 时间到 $(u,maxE)$。

用 $dist[u][e]$ 表示到达该状态的最短耗时。答案取所有 $dist[T][e]$（$e=0..maxE$）中的最小值；若都不可达则输出 $-1$。

## 复杂度分析

状态数 $N\times(maxE+1)$，边转移约 $M\times maxE$。

* 时间复杂度：$O((N\times maxE+M\times maxE)\log(N\times maxE))$。
* 空间复杂度：$O(N\times maxE+M)$。

$N,M\le 1000$，$maxE\le 100$，可以通过。

## 代码实现

### Python

```python
import heapq

INF = 10 ** 30


def min_time(n, graph, charge, s, t, start_e, max_e):
    # dist[u][e]：到达路口 u、剩余电量 e 的最短耗时
    dist = [[INF] * (max_e + 1) for _ in range(n)]
    dist[s][start_e] = 0
    heap = [(0, s, start_e)]

    while heap:
        cur_time, u, energy = heapq.heappop(heap)
        if cur_time != dist[u][energy]:
            continue

        # 充电站充满电
        if charge[u] and energy < max_e:
            new_time = cur_time + 1
            if new_time < dist[u][max_e]:
                dist[u][max_e] = new_time
                heapq.heappush(heap, (new_time, u, max_e))

        for v, road_time, cost in graph[u]:
            if energy >= cost:
                new_energy = energy - cost
                new_time = cur_time + road_time
                if new_time < dist[v][new_energy]:
                    dist[v][new_energy] = new_time
                    heapq.heappush(heap, (new_time, v, new_energy))

    ans = min(dist[t])
    return -1 if ans == INF else ans


def main():
    n, k, m, s, t, start_e, max_e = map(int, input().split())
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, road_time, cost = map(int, input().split())
        graph[u].append((v, road_time, cost))
    charge = [False] * n
    if k > 0:
        for cp in map(int, input().split()):
            charge[cp] = True
    print(min_time(n, graph, charge, s, t, start_e, max_e))


if __name__ == "__main__":
```

```java
import java.util.*;

public class Main {

    static class Edge {
        int to, time, cost;
        Edge(int to, int time, int cost) {
            this.to = to;
            this.time = time;
            this.cost = cost;
        }
    }

    static long minTime(int n, List<Edge>[] graph, boolean[] charge,
                        int s, int t, int startE, int maxE) {
        long INF = Long.MAX_VALUE / 4;
        // dist[u][e]：到达路口 u、剩余电量 e 的最短耗时
        long[][] dist = new long[n][maxE + 1];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], INF);
        }

        PriorityQueue<long[]> pq = new PriorityQueue<>(
                (a, b) -> Long.compare(a[0], b[0])
        );
        dist[s][startE] = 0;
        pq.offer(new long[]{0, s, startE});

        while (!pq.isEmpty()) {
            long[] cur = pq.poll();
            long curTime = cur[0];
            int u = (int) cur[1];
            int energy = (int) cur[2];
            if (curTime != dist[u][energy]) {
                continue;
            }

            // 充电站充满电
            if (charge[u] && energy < maxE) {
                long newTime = curTime + 1;
                if (newTime < dist[u][maxE]) {
                    dist[u][maxE] = newTime;
                    pq.offer(new long[]{newTime, u, maxE});
                }
            }

            for (Edge edge : graph[u]) {
                if (energy >= edge.cost) {
                    int newEnergy = energy - edge.cost;
                    long newTime = curTime + edge.time;
                    if (newTime < dist[edge.to][newEnergy]) {
                        dist[edge.to][newEnergy] = newTime;
                        pq.offer(new long[]{newTime, edge.to, newEnergy});
                    }
                }
            }
        }

        long ans = INF;
        for (int e = 0; e <= maxE; e++) {
            ans = Math.min(ans, dist[t][e]);
        }
        return ans == INF ? -1 : ans;
    }

        int maxE = sc.nextInt();

        List<Edge>[] graph = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }

            int from = sc.nextInt();
```

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <climits>
using namespace std;

struct Edge {
    int to, time, cost;
};

struct State {
    long long time;
    int node, energy;
    bool operator<(const State& other) const {
        return time > other.time;
    }
};

long long minTime(int n, vector<vector<Edge>>& graph, vector<bool>& charge,
                  int s, int t, int startE, int maxE) {
    const long long INF = LLONG_MAX / 4;
    // dist[u][e]：到达路口 u、剩余电量 e 的最短耗时
    vector<vector<long long>> dist(n, vector<long long>(maxE + 1, INF));
    priority_queue<State> pq;

    dist[s][startE] = 0;
    pq.push({0, s, startE});

    while (!pq.empty()) {
        State cur = pq.top();
        pq.pop();
        long long curTime = cur.time;
        int u = cur.node;
        int energy = cur.energy;
        if (curTime != dist[u][energy]) {
            continue;
        }

        // 充电站充满电
        if (charge[u] && energy < maxE) {
            long long newTime = curTime + 1;
            if (newTime < dist[u][maxE]) {
                dist[u][maxE] = newTime;
                pq.push({newTime, u, maxE});
            }
        }

        for (const Edge& edge : graph[u]) {
            if (energy >= edge.cost) {
                int newEnergy = energy - edge.cost;
                long long newTime = curTime + edge.time;
                if (newTime < dist[edge.to][newEnergy]) {
                    dist[edge.to][newEnergy] = newTime;
                    pq.push({newTime, edge.to, newEnergy});
                }
            }
        }
    }

    long long ans = INF;
    for (int e = 0; e <= maxE; e++) {
        ans = min(ans, dist[t][e]);
    }
    return ans == INF ? -1 : ans;
}
```
