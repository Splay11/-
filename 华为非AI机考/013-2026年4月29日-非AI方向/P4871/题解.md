# 思路

对于每个用户的购物记录：

1. 先对该用户购买的商品 $ID$ 去重。
2. 对去重后的商品列表两两枚举，生成商品对 $(a,b)$。
3. 为了避免 $(1,2)$ 和 $(2,1)$ 被当成不同商品对，统一令 $a < b$。
4. 用哈希表统计每个商品对的共现次数。
5. 对每个查询 $T$，遍历所有商品对，统计共现次数 $cnt \ge T$ 的商品对数量。

设某个用户去重后有 $m$ 个商品，则该用户最多产生：

$$
\frac{m(m-1)}{2}
$$

个商品对。

因为 $k \le 100$，$N \le 1000$，所以暴力枚举商品对是可以接受的。

时间复杂度大约为：

$$
O\left(N \times k^2 + Q \times P\right)
$$

其中 $P$ 是所有不同商品对的数量。

## 代码实现

### Python

```python
from collections import defaultdict

# 读取用户数量 N 和查询数量 Q
N, Q = map(int, input().split())

# 统计每个商品对的共现次数
# key 是二元组 (a, b)，value 是共现次数
pair_count = defaultdict(int)

# 读取每个用户的购物记录
for _ in range(N):
    data = list(map(int, input().split()))

    # data[0] 是该用户购买的商品数量 k
    k = data[0]

    # data[1:] 是该用户购买的商品 ID 列表
    goods = data[1:]

    # 对当前用户购买的商品 ID 去重并排序
    # 例如 [1, 1, 2, 2, 3] 会变成 [1, 2, 3]
    goods = sorted(set(goods))

    m = len(goods)

    # 枚举当前用户购买过的所有不同商品对
    for i in range(m):
        for j in range(i + 1, m):
            a = goods[i]
            b = goods[j]

            # 因为 goods 已经排序，且 j > i，所以一定满足 a < b
            # 将商品对 (a, b) 的共现次数加 1
            pair_count[(a, b)] += 1

# 处理 Q 次查询
for _ in range(Q):
    T = int(input())

    ans = 0

    # 遍历所有商品对的共现次数
    # 统计共现次数 >= T 的商品对数量
    for cnt in pair_count.values():
        if cnt >= T:
            ans += 1

    print(ans)

```

### c++

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, Q;
    cin >> N >> Q;

    // 统计每个商品对出现的次数
    // key 是 pair<int, int>，表示商品对 (a, b)
    // value 是该商品对的共现次数
    map<pair<int, int>, int> pairCount;

    // 读取 N 个用户的购物记录
    for (int i = 0; i < N; i++) {
        int k;
        cin >> k;

        // 使用 set 对当前用户购买的商品 ID 去重
        // 例如 1 1 2 2 3 会变成 1 2 3
        set<int> goods;

        for (int j = 0; j < k; j++) {
            int id;
            cin >> id;
            goods.insert(id);
        }

        // 将 set 转成 vector，方便用下标两两枚举
        vector<int> arr(goods.begin(), goods.end());

        int m = arr.size();

        // 枚举当前用户购买过的所有不同商品对
        for (int x = 0; x < m; x++) {
            for (int y = x + 1; y < m; y++) {
                int a = arr[x];
                int b = arr[y];

                // 因为 set 已经有序，且 y > x，所以一定满足 a < b
                // 将商品对 (a, b) 的共现次数加 1
                pairCount[{a, b}]++;
            }
        }
    }

    // 处理 Q 次查询
    while (Q--) {
        int T;
        cin >> T;

        int ans = 0;

        // 遍历所有商品对，统计共现次数 >= T 的商品对数量
        for (auto &item : pairCount) {
            if (item.second >= T) {
                ans++;
            }
        }

        cout << ans << '\n';
    }

    return 0;
}

```


### Java 

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 读取用户数量 N 和查询数量 Q
        int N = sc.nextInt();
        int Q = sc.nextInt();

        // 统计每个商品对的共现次数
        // key 用字符串 "a#b" 表示商品对 (a, b)
        // value 是该商品对的共现次数
        Map<String, Integer> pairCount = new HashMap<>();

        // 读取 N 个用户的购物记录
        for (int i = 0; i < N; i++) {
            int k = sc.nextInt();

            // 使用 TreeSet 对当前用户购买的商品 ID 去重并排序
            // 例如 1 1 2 2 3 会变成 1 2 3
            Set<Integer> set = new TreeSet<>();

            for (int j = 0; j < k; j++) {
                int id = sc.nextInt();
                set.add(id);
            }

            // 将 Set 转成 List，方便用下标两两枚举
            List<Integer> goods = new ArrayList<>(set);

            int m = goods.size();

            // 枚举当前用户购买过的所有不同商品对
            for (int x = 0; x < m; x++) {
                for (int y = x + 1; y < m; y++) {
                    int a = goods.get(x);
                    int b = goods.get(y);

                    // 因为 TreeSet 已经排序，且 y > x，所以一定满足 a < b
                    // 用 "a#b" 作为商品对的唯一标识
                    String key = a + "#" + b;

                    // 将该商品对的共现次数加 1
                    pairCount.put(key, pairCount.getOrDefault(key, 0) + 1);
                }
            }
        }

        // 处理 Q 次查询
        for (int i = 0; i < Q; i++) {
            int T = sc.nextInt();

            int ans = 0;

            // 遍历所有商品对的共现次数
            // 统计共现次数 >= T 的商品对数量
            for (int cnt : pairCount.values()) {
                if (cnt >= T) {
                    ans++;
                }
            }

            System.out.println(ans);
        }

        sc.close();
    }
}
```
