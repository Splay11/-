## 解题思路

将 $m \times n$ 的矩形网格按行优先编号，编号 $x$ 对应的位置为：

$行 = (x - 1) / n$，$列 = (x - 1) % n$

对于每一个待验证序列，依次检查相邻两个数字 $a$ 和 $b$：

1. 判断 $a$ 和 $b$ 在网格中是否相邻
   两个格子的行差和列差都不超过 $1$，且不能是同一个格子，即满足 $8$ 个方向相邻。

2. 判断路径是否重复
   路径 $(a,b)$ 和 $(b,a)$ 视为同一条边，因此用无向边记录。
   可以把边统一表示为较小编号在前、较大编号在后，例如 $(min(a,b), max(a,b))$。
   如果这条边之前出现过，则该序列无效。

使用集合 $set$ 记录已经走过的边即可。

相关算法：模拟、哈希集合判重。

## 复杂度分析

设一个序列长度为 $L$。

时间复杂度为 $O(L)$，每相邻两个数字只检查一次。

空间复杂度为 $O(L)$，用于存储已经经过的边。

对于 $k$ 个序列，总时间复杂度为 $O(\sum L)$，可以满足题目要求。

## 代码实现

### Python

```python
def is_valid_path(m, n, seq):
    used = set()

    for i in range(len(seq) - 1):
        a = seq[i]
        b = seq[i + 1]

        # 将编号转换为行列坐标
        r1 = (a - 1) // n
        c1 = (a - 1) % n
        r2 = (b - 1) // n
        c2 = (b - 1) % n

        # 判断是否为 8 个方向相邻，且不能是同一个格子
        if a == b or abs(r1 - r2) > 1 or abs(c1 - c2) > 1:
            return False

        # 无向边判重
        edge = (min(a, b), max(a, b))
        if edge in used:
            return False
        used.add(edge)

    return True


def main():
    m = int(input().strip())
    n = int(input().strip())
    k = int(input().strip())

    ans = []

    for _ in range(k):
        line = input().strip()
        seq = list(map(int, line.split(',')))

        # 判断当前序列是否合法
        if is_valid_path(m, n, seq):
            ans.append(line)

    if len(ans) == 0:
        print("false")
    else:
        for s in ans:
            print(s)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    static boolean isValidPath(int m, int n, int[] seq) {
        HashSet<String> used = new HashSet<>();

        for (int i = 0; i < seq.length - 1; i++) {
            int a = seq[i];
            int b = seq[i + 1];

            // 将编号转换为行列坐标
            int r1 = (a - 1) / n;
            int c1 = (a - 1) % n;
            int r2 = (b - 1) / n;
            int c2 = (b - 1) % n;

            // 判断是否为 8 个方向相邻，且不能是同一个格子
            if (a == b || Math.abs(r1 - r2) > 1 || Math.abs(c1 - c2) > 1) {
                return false;
            }

            // 无向边判重
            int x = Math.min(a, b);
            int y = Math.max(a, b);
            String edge = x + "#" + y;

            if (used.contains(edge)) {
                return false;
            }
            used.add(edge);
        }

        return true;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int m = Integer.parseInt(sc.nextLine().trim());
        int n = Integer.parseInt(sc.nextLine().trim());
        int k = Integer.parseInt(sc.nextLine().trim());

        ArrayList<String> ans = new ArrayList<>();

        for (int i = 0; i < k; i++) {
            String line = sc.nextLine().trim();
            String[] parts = line.split(",");
            int[] seq = new int[parts.length];

            // 解析逗号分隔的整数序列
            for (int j = 0; j < parts.length; j++) {
                seq[j] = Integer.parseInt(parts[j]);
            }

            // 判断当前序列是否合法
            if (isValidPath(m, n, seq)) {
                ans.add(line);
            }
        }

        if (ans.size() == 0) {
            System.out.println("false");
        } else {
            for (String s : ans) {
                System.out.println(s);
            }
        }

        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <set>
#include <sstream>
#include <string>
#include <cmath>
using namespace std;

bool isValidPath(int m, int n, vector<int>& seq) {
    set<pair<int, int> > used;

    for (int i = 0; i + 1 < (int)seq.size(); i++) {
        int a = seq[i];
        int b = seq[i + 1];

        // 将编号转换为行列坐标
        int r1 = (a - 1) / n;
        int c1 = (a - 1) % n;
        int r2 = (b - 1) / n;
        int c2 = (b - 1) % n;

        // 判断是否为 8 个方向相邻，且不能是同一个格子
        if (a == b || abs(r1 - r2) > 1 || abs(c1 - c2) > 1) {
            return false;
        }

        // 无向边判重
        int x = min(a, b);
        int y = max(a, b);
        pair<int, int> edge = make_pair(x, y);

        if (used.count(edge)) {
            return false;
        }
        used.insert(edge);
    }

    return true;
}

vector<int> parseSequence(string line) {
    vector<int> seq;
    string item;
    stringstream ss(line);

    // 解析逗号分隔的整数序列
    while (getline(ss, item, ',')) {
        seq.push_back(stoi(item));
    }

    return seq;
}

int main() {
    int m, n, k;
    cin >> m;
    cin >> n;
    cin >> k;

    string line;
    getline(cin, line); // 读取换行符

    vector<string> ans;

    for (int i = 0; i < k; i++) {
        getline(cin, line);
        vector<int> seq = parseSequence(line);

        // 判断当前序列是否合法
        if (isValidPath(m, n, seq)) {
            ans.push_back(line);
        }
    }

    if (ans.empty()) {
        cout << "false" << endl;
    } else {
        for (int i = 0; i < (int)ans.size(); i++) {
            cout << ans[i] << endl;
        }
    }

    return 0;
}
```
