## 解题思路

* 有 $M$ 台服务器（编号 $1\sim M$）、$N$ 个任务（编号 $1\sim N$）。每个任务给出启动时间 $s$ 与执行时间 $t$，启动时间互不相同。
* 任务按启动时间先后处理：到来时，若某服务器的空闲时刻 $\le$ 当前启动时间，则可分配；每次占用**编号最小**的空闲服务器，执行结束后立刻释放。
* 求编号为 $K$ 的任务占用的服务器编号。

做法：

1. 把任务按启动时间升序排序，同时保留原编号。
2. 用数组 $free\_until[1..M]$ 记录每台服务器「下一次空闲的时刻」，初始全为 $0$。
3. 依次处理每个任务 $(s,t)$：从 $1$ 扫到 $M$，找到第一个满足 $free\_until[id]\le s$ 的服务器 $id$，分配给它，并令 $free\_until[id]=s+t$。
4. 记录每个原编号任务分到的服务器，最后输出第 $K$ 个。

$N,M\le 10^4$，每次线性扫服务器，$O(NM)$ 可以通过。

## 复杂度分析

* 时间复杂度：$O(N\log N+NM)$（排序 + 逐任务扫服务器）。
* 空间复杂度：$O(M+N)$。

## 代码实现

### Python

```python
def server_of_k(M, tasks, K):
    # tasks: [(启动时间, 执行时间, 原编号), ...]
    tasks = sorted(tasks)  # 按启动时间排序
    # free_until[id]：服务器 id 下一次空闲时刻
    free_until = [0] * (M + 1)
    ans = [0] * (len(tasks) + 1)

    for s, t, idx in tasks:
        sid = -1
        # 找编号最小且已空闲的服务器
        for sid_cand in range(1, M + 1):
            if free_until[sid_cand] <= s:
                sid = sid_cand
                break
        ans[idx] = sid
        free_until[sid] = s + t

    return ans[K]


def main():
    M, N, K = map(int, input().split())
    tasks = []
    for i in range(1, N + 1):
        s, t = map(int, input().split())
        tasks.append((s, t, i))
    print(server_of_k(M, tasks, K))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static int serverOfK(int M, int[][] tasks, int K) {
        // tasks[i] = {启动时间, 执行时间, 原编号}
        Arrays.sort(tasks, (a, b) -> Integer.compare(a[0], b[0]));

        // freeUntil[id]：服务器 id 下一次空闲时刻
        int[] freeUntil = new int[M + 1];
        int[] ans = new int[tasks.length + 1];

        for (int[] x : tasks) {
            int s = x[0], t = x[1], idx = x[2];
            int sid = -1;
            // 找编号最小且已空闲的服务器
            for (int id = 1; id <= M; id++) {
                if (freeUntil[id] <= s) {
                    sid = id;
                    break;
                }
            }
            ans[idx] = sid;
            freeUntil[sid] = s + t;
        }
        return ans[K];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int M = sc.nextInt();
        int N = sc.nextInt();
        int K = sc.nextInt();

        int[][] tasks = new int[N][3];
        for (int i = 0; i < N; i++) {
            tasks[i][0] = sc.nextInt();
            tasks[i][1] = sc.nextInt();
            tasks[i][2] = i + 1;
        }
        System.out.println(serverOfK(M, tasks, K));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int server_of_k(int M, vector<tuple<int, int, int>>& tasks, int K) {
    // 按启动时间排序
    sort(tasks.begin(), tasks.end());

    // free_until[id]：服务器 id 下一次空闲时刻
    vector<int> free_until(M + 1, 0);
    vector<int> ans((int)tasks.size() + 1, 0);

    for (auto& tp : tasks) {
        int s = get<0>(tp), t = get<1>(tp), idx = get<2>(tp);
        int sid = -1;
        // 找编号最小且已空闲的服务器
        for (int id = 1; id <= M; id++) {
            if (free_until[id] <= s) {
                sid = id;
                break;
            }
        }
        ans[idx] = sid;
        free_until[sid] = s + t;
    }
    return ans[K];
}

int main() {
    int M, N, K;
    cin >> M >> N >> K;
    vector<tuple<int, int, int>> tasks;
    for (int i = 1; i <= N; i++) {
        int s, t;
        cin >> s >> t;
        tasks.emplace_back(s, t, i);
    }
    cout << server_of_k(M, tasks, K) << "\n";
    return 0;
}
```
