# DFS暴力解法

这道题的正解是 $树形背包$，但是我们用朴素解法 $DFS$ 选或不选每个技能也能在考试时拿到一定的分数。

每个技能最多 $1$ 个直接前置，依赖无环，因此技能构成一片森林。加入虚拟根 $0$，把所有无前置的技能连到 $0$ 上。学某个技能时，必须已经学过它的前置（整条祖先链会自然被保证）。

朴素做法：

1. 先按「父在前、子在后」的顺序得到技能序列 $order$（从虚拟根做一次树遍历即可）。
2. 用 $DFS$ 按 $order$ 依次决定每个技能学或不学。
3. 学技能 $u$ 当且仅当：前置 $parent[u]$ 已学，且剩余时间够付 $cost[u]$；学了就累加 $power[u]$。
4. 全部决定完后，用当前总战斗力更新答案。

$N$ 较小时 $2^N$ 能过；题目 $N\le 100$ 时搜索会很慢，需要后面的正解。

```python
def max_power(n, total_time, cost, power, parent):
    children = [[] for _ in range(n + 1)]
    for i in range(1, n + 1):
        children[parent[i]].append(i)

    # 父在前、子在后，保证学子技能时能检查前置
    order = []

    def build(u):
        for v in children[u]:
            order.append(v)
            build(v)

    build(0)

    learned = [False] * (n + 1)
    learned[0] = True  # 虚拟根视为已具备
    ans = [0]

    def dfs(pos, used, got):
        if pos == len(order):
            ans[0] = max(ans[0], got)
            return
        u = order[pos]
        # 不学技能 u（其后子孙也会因前置未学而无法学）
        dfs(pos + 1, used, got)
        # 学技能 u
        if learned[parent[u]] and used + cost[u] <= total_time:
            learned[u] = True
            dfs(pos + 1, used + cost[u], got + power[u])
            learned[u] = False

    dfs(0, 0, 0)
    return ans[0]


def main():
    n, total_time = map(int, input().split())
    cost = [0] * (n + 1)
    power = [0] * (n + 1)
    parent = [0] * (n + 1)
    for _ in range(n):
        parts = list(map(int, input().split()))
        skill_id, t, p, k = parts[0], parts[1], parts[2], parts[3]
        cost[skill_id] = t
        power[skill_id] = p
        if k == 1:
            parent[skill_id] = parts[4]
    print(max_power(n, total_time, cost, power, parent))


if __name__ == "__main__":
    main()
```

```cpp
int totalTime;
vector<int> costValue, powerValue, parentValue;

vector<int> order;

        build(v, children);

void dfs(int pos, int used, int got) {
    if (pos == (int)order.size()) {
        ans = max(ans, got);
        return;
    }
    int u = order[pos];
    // 不学
    dfs(pos + 1, used, got);
    // 学
    if (learned[parentValue[u]] && used + costValue[u] <= totalTime) {
        learned[u] = true;
        dfs(pos + 1, used + costValue[u], got + powerValue[u]);
        learned[u] = false;
    }
}

int maxPower(int n, int t, vector<int>& cost, vector<int>& power, vector<int>& parent) {
    totalTime = t;
    costValue = cost;
    powerValue = power;
    parentValue = parent;

    vector<vector<int>> children(n + 1);
    for (int i = 1; i <= n; i++) {
        children[parent[i]].push_back(i);
    }

    order.clear();
    build(0, children);

    learned.assign(n + 1, false);
    learned[0] = true;
    ans = 0;
    dfs(0, 0, 0);
    return ans;
}
```

# 正确解法

## 题目分析

把技能建成以虚拟根 $0$ 为根的树后，用树形背包求时间上限 $T$ 内的最大战斗力。

定义 $dp[u][t]$：在技能 $u$ 的子树中，已经学习技能 $u$，再花费 $t$ 小时，能获得的最大战斗力。虚拟根 $cost[0]=power[0]=0$。

转移：

1. 先学 $u$ 本身：令 $dp[u][cost[u]] = power[u]$（时间不够则该点不可行）。
2. 依次合并每个子技能 $v$：对当前已有方案耗时 $i$、在 $v$ 子树中耗时 $j$，可转移到 $i+j$，战斗力相加。不学 $v$ 的方案保留在原数组中即可。

最终答案是 $dp[0][0..T]$ 的最大值。

## 复杂度分析

设技能数为 $N$，总时间为 $T$。

* 时间复杂度：$O(N \times T^2)$。
* 空间复杂度：$O(N \times T)$。

在 $N\le 100$，$T\le 1000$ 范围内可以通过。

### Python

```python
NEG = -10 ** 9


def max_power(n, total_time, cost, power, parent):
    children = [[] for _ in range(n + 1)]
    for i in range(1, n + 1):
        children[parent[i]].append(i)

    # dp[u][t]：已学 u，子树内再花 t 小时的最大战斗力
    dp = [[NEG] * (total_time + 1) for _ in range(n + 1)]

    def dfs(u):
        if cost[u] <= total_time:
            dp[u][cost[u]] = power[u]

        for v in children[u]:
            dfs(v)
            new_dp = dp[u][:]
            for i in range(total_time + 1):
                if dp[u][i] == NEG:
                    continue
                for j in range(total_time - i + 1):
                    if dp[v][j] == NEG:
                        continue
                    # 在已学 u 的方案上合并子技能 v
                    new_dp[i + j] = max(new_dp[i + j], dp[u][i] + dp[v][j])
            dp[u] = new_dp

    dfs(0)
    return max(dp[0])


def main():
    n, total_time = map(int, input().split())
    cost = [0] * (n + 1)
    power = [0] * (n + 1)
    parent = [0] * (n + 1)
    for _ in range(n):
        parts = list(map(int, input().split()))
        skill_id, t, p, k = parts[0], parts[1], parts[2], parts[3]
        cost[skill_id] = t
```

```java
import java.util.*;

public class Main {
    static final int NEG = -1000000000;

    static int totalTime;
    static int[] cost;
    static int[] power;
    static ArrayList<Integer>[] children;
    static int[][] dp;

    static void dfs(int u) {
        if (cost[u] <= totalTime) {
            dp[u][cost[u]] = power[u];
        }

        for (int v : children[u]) {
            dfs(v);
            int[] newDp = dp[u].clone();
            for (int i = 0; i <= totalTime; i++) {
                if (dp[u][i] == NEG) {
                    continue;
                }
                for (int j = 0; i + j <= totalTime; j++) {
                    if (dp[v][j] == NEG) {
                        continue;
                    }
                    // 合并子技能 v
                    newDp[i + j] = Math.max(newDp[i + j], dp[u][i] + dp[v][j]);
                }
            }
            dp[u] = newDp;
        }
    }

    static int maxPower(int n, int t, int[] inputCost, int[] inputPower, int[] parent) {
        totalTime = t;
        cost = inputCost;
        power = inputPower;

        children = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            children[i] = new ArrayList<>();
        }
        for (int i = 1; i <= n; i++) {
            children[parent[i]].add(i);
        }

        dp = new int[n + 1][totalTime + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(dp[i], NEG);
        }

        dfs(0);

        int ans = 0;
        for (int i = 0; i <= totalTime; i++) {
            ans = Math.max(ans, dp[0][i]);
        }
        return ans;
    }

            }
```

```cpp
void dfs(int u) {
    if (costValue[u] <= totalTime) {
        dp[u][costValue[u]] = powerValue[u];
    }

    for (int v : children[u]) {
        dfs(v);
        vector<int> newDp = dp[u];
        for (int i = 0; i <= totalTime; i++) {
            if (dp[u][i] == NEG) {
                continue;
```
