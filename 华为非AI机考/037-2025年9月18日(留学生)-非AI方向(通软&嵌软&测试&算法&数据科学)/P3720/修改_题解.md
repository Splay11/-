# 二重循环朴素解法

这道题的正解是 $归并排序$ 过程中统计对数，但是我们用朴素解法 $二重循环$ 也能在考试时拿到一定的分数。

题意：给定数组 $A$ 与整数 $K$，统计满足 $i<j$ 且 $A[i]-A[j]\ge K$ 的下标对个数。

朴素做法：两重循环枚举所有 $0\le i<j<n$，若 $A[i]-A[j]\ge K$，答案加 $1$。

小数据可以过；当 $n$ 达到 $10^5$ 时，$O(n^2)$ 会超时，需要后面的归并统计做法。
## 二重循环解法

```python
def count_k_pairs(arr, k):
    n = len(arr)
    ans = 0
    # 枚举所有下标对 (i, j)
    for i in range(n):
        for j in range(i + 1, n):
            if arr[i] - arr[j] >= k:
                ans += 1
    return ans
```

# 正确解法

## 题目分析

条件 $A[i]-A[j]\ge K$ 且 $i<j$，可以在归并排序合并左右有序段时一起统计。

设左半段、右半段都已各自排好序。合并前用双指针扫一遍：

* 若 $A[i]-A[j]\ge K$（$i$ 在左、$j$ 在右），则左半段中下标 $\ge i$ 的元素与当前 $A[j]$ 都满足条件，一次加上 $mid-i+1$，再让 $j$ 右移；
* 否则让 $i$ 右移。

统计完后再按普通归并把两段合成有序。递归处理左右半段，总对数即为答案。

注意答案可能很大，计数器用 $64$ 位整数。

## 复杂度分析

* 时间复杂度：$O(n\log n)$。
* 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
def count_k_pairs(arr, k):
    def merge_sort_count(l, r):
        if l >= r:
            return 0
        mid = (l + r) // 2
        count = merge_sort_count(l, mid) + merge_sort_count(mid + 1, r)

        # 在左右有序段上统计跨段统治对
        i, j = l, mid + 1
        while i <= mid and j <= r:
            if arr[i] - arr[j] >= k:
                count += mid - i + 1
                j += 1
            else:
                i += 1

        # 归并成有序
        temp = []
        i, j = l, mid + 1
        while i <= mid and j <= r:
            if arr[i] <= arr[j]:
                temp.append(arr[i])
                i += 1
            else:
                temp.append(arr[j])
                j += 1
        temp.extend(arr[i:mid + 1])
        temp.extend(arr[j:r + 1])
        arr[l:r + 1] = temp
        return count

    return merge_sort_count(0, len(arr) - 1)


def main():
    n = int(input())
    arr = list(map(int, input().split()))
    k = int(input())
    print(count_k_pairs(arr, k))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    static long countKPairs(int[] arr, int k) {
        return mergeSortCount(arr, 0, arr.length - 1, k);
    }

    static long mergeSortCount(int[] arr, int left, int right, int k) {
        if (left >= right) {
            return 0;
        }
        int mid = left + (right - left) / 2;
        long count = mergeSortCount(arr, left, mid, k)
                + mergeSortCount(arr, mid + 1, right, k);

        // 在左右有序段上统计跨段统治对
        int i = left, j = mid + 1;
        while (i <= mid && j <= right) {
            if ((long) arr[i] - arr[j] >= k) {
                count += mid - i + 1;
                j++;
            } else {
                i++;
            }
        }

        // 归并成有序
        int[] temp = new int[right - left + 1];
        i = left;
        j = mid + 1;
        int t = 0;
        while (i <= mid && j <= right) {
            if (arr[i] <= arr[j]) {
                temp[t++] = arr[i++];
            } else {
                temp[t++] = arr[j++];
            }
        }
        while (i <= mid) {
            temp[t++] = arr[i++];
        }
        while (j <= right) {
            temp[t++] = arr[j++];
        }
        System.arraycopy(temp, 0, arr, left, temp.length);
        return count;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int k = sc.nextInt();
        System.out.println(countKPairs(arr, k));
    }
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

long long merge_sort_count(vector<int>& arr, int left, int right, int k) {
    if (left >= right) {
        return 0;
    }
    int mid = left + (right - left) / 2;
    long long count = merge_sort_count(arr, left, mid, k)
                    + merge_sort_count(arr, mid + 1, right, k);

    // 在左右有序段上统计跨段统治对
    int i = left, j = mid + 1;
    while (i <= mid && j <= right) {
        if ((long long)arr[i] - arr[j] >= k) {
            count += mid - i + 1;
            j++;
        } else {
            i++;
        }
    }

    // 归并成有序
    vector<int> temp;
    i = left;
    j = mid + 1;
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            temp.push_back(arr[i++]);
        } else {
            temp.push_back(arr[j++]);
        }
    }
    while (i <= mid) {
        temp.push_back(arr[i++]);
    }
    while (j <= right) {
        temp.push_back(arr[j++]);
    }
    copy(temp.begin(), temp.end(), arr.begin() + left);
    return count;
}

long long count_k_pairs(vector<int>& arr, int k) {
    return merge_sort_count(arr, 0, (int)arr.size() - 1, k);
}

int main() {
    int n;
    cin >> n;
    vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    int k;
    cin >> k;
    cout << count_k_pairs(arr, k) << "\n";
    return 0;
}
```
