## 解题思路

把每位法师第 $k$ 次吸收水晶看成一个“吸收槽位”。

对于初始能量为 $a[i]$ 的法师：

第 $1$ 个槽位代价是 $1 \times a[i]$

第 $2$ 个槽位代价是 $2 \times a[i]$

第 $k$ 个槽位代价是 $k \times a[i]$

如果把基础能量为 $b[j]$ 的水晶放入代价为 $cost$ 的槽位，收益就是：

$b[j] - cost$

如果收益 $\le 0$，就没有必要计入答案。

因此问题变成：

从所有法师产生的槽位代价中，取出最小的若干个；从所有水晶中，取出最大的若干个，使总收益最大。

核心做法：

使用最小堆生成前 $m$ 个最小槽位代价。

堆中初始放入每位法师的第 $1$ 个槽位，即 $a[i]$。

每次弹出当前最小代价 $cost$，然后把同一位法师的下一个槽位代价 $cost + a[i]$ 放回堆中。

这样可以得到全局最小的前 $m$ 个槽位代价。

然后将水晶能量 $b$ 从大到小排序，将槽位代价从小到大匹配。

如果当前收益 $b[i] - cost[i] > 0$，加入答案；否则后面的水晶更小、槽位代价更大，不可能再产生正收益，可以直接停止。

使用的算法是排序 + 最小堆。

## 复杂度分析

设法师数量为 $n$，水晶数量为 $m$。

生成前 $m$ 个最小槽位代价需要 $O((n + m)\log n)$。

水晶排序需要 $O(m\log m)$。

时间复杂度为：

$O((n + m)\log n + m\log m)$

空间复杂度为：

$O(n + m)$

## 代码实现

### Python

```python
import sys
import heapq


def max_energy(n, m, a, b):
    # 水晶能量从大到小排序
    b.sort(reverse=True)

    heap = []
    for x in a:
        # 堆元素：(当前槽位代价, 法师初始能量)
        heapq.heappush(heap, (x, x))

    ans = 0

    for i in range(m):
        # 取当前最小的槽位代价
        cost, base = heapq.heappop(heap)

        gain = b[i] - cost
        if gain <= 0:
            # 后续水晶更小，槽位代价更大，不会再有正收益
            break

        ans += gain

        # 生成该法师的下一个槽位
        heapq.heappush(heap, (cost + base, base))

    return ans


def main():
    data = list(map(int, sys.stdin.read().split()))
    n, m = data[0], data[1]
    a = data[2:2 + n]
    b = data[2 + n:2 + n + m]

    print(max_energy(n, m, a, b))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {

    static class Node {
        long cost;
        int base;

        Node(long cost, int base) {
            this.cost = cost;
            this.base = base;
        }
    }

    static long maxEnergy(int n, int m, int[] a, int[] b) {
        // 水晶能量从小到大排序，之后从后往前取最大值
        Arrays.sort(b);

        PriorityQueue<Node> heap = new PriorityQueue<>(new Comparator<Node>() {
            public int compare(Node x, Node y) {
                return Long.compare(x.cost, y.cost);
            }
        });

        // 每位法师先放入第 1 个槽位
        for (int x : a) {
            heap.offer(new Node(x, x));
        }

        long ans = 0;

        for (int i = m - 1; i >= 0; i--) {
            // 取当前最小的槽位代价
            Node cur = heap.poll();

            long gain = (long)b[i] - cur.cost;
            if (gain <= 0) {
                // 后续水晶更小，槽位代价更大，不会再有正收益
                break;
            }

            ans += gain;

            // 生成该法师的下一个槽位
            heap.offer(new Node(cur.cost + cur.base, cur.base));
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        int[] a = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }

        int[] b = new int[m];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < m; i++) {
            b[i] = Integer.parseInt(st.nextToken());
        }

        System.out.println(maxEnergy(n, m, a, b));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Node {
    long long cost;
    int base;

    bool operator>(const Node& other) const {
        return cost > other.cost;
    }
};

long long maxEnergy(int n, int m, vector<int>& a, vector<int>& b) {
    // 水晶能量从大到小排序
    sort(b.begin(), b.end(), greater<int>());

    priority_queue<Node, vector<Node>, greater<Node>> heap;

    // 每位法师先放入第 1 个槽位
    for (int x : a) {
        heap.push({x, x});
    }

    long long ans = 0;

    for (int i = 0; i < m; i++) {
        // 取当前最小的槽位代价
        Node cur = heap.top();
        heap.pop();

        long long gain = (long long)b[i] - cur.cost;
        if (gain <= 0) {
            // 后续水晶更小，槽位代价更大，不会再有正收益
            break;
        }

        ans += gain;

        // 生成该法师的下一个槽位
        heap.push({cur.cost + cur.base, cur.base});
    }

    return ans;
}

int main() {
    int n, m;
    cin >> n >> m;

    vector<int> a(n), b(m);

    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    for (int i = 0; i < m; i++) {
        cin >> b[i];
    }

    cout << maxEnergy(n, m, a, b) << endl;

    return 0;
}
```
