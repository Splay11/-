## 解题思路

先将所有建筑位置排序。

使用贪心算法：

每次从最左边还没有被爆破的建筑开始，设它的位置为 $left$。

为了让当前这颗炸弹覆盖尽量多的右侧建筑，炸弹必须部署在某个建筑位置上，并且要能覆盖 $left$，所以炸弹部署点最远可以选到 $left + M$。

因此：

1. 找到最右边且位置小于等于 $left + M$ 的建筑，将炸弹部署在这里。
2. 设炸弹部署位置为 $center$，它能爆破所有位置小于等于 $center + M$ 的建筑。
3. 跳过这些已经被爆破的建筑，继续处理下一个未爆破建筑。

这样每一颗炸弹都在保证覆盖当前最左未爆破建筑的前提下，尽量向右覆盖更多建筑，因此炸弹数量最少。

## 复杂度分析

排序需要 $O(N \log N)$。

遍历建筑时，每个建筑最多被访问一次，因此遍历复杂度为 $O(N)$。

总时间复杂度为 $O(N \log N)$。

需要存储所有建筑位置，空间复杂度为 $O(N)$。

## 代码实现

### Python

```python
import sys

def min_bombs(pos, m):
    # 先排序，方便从左到右贪心处理
    pos.sort()
    n = len(pos)
    i = 0
    ans = 0

    while i < n:
        ans += 1

        # 当前最左边还没有被爆破的建筑
        left = pos[i]

        # 找到最右边且能覆盖 left 的炸弹部署位置
        while i < n and pos[i] <= left + m:
            i += 1

        # 炸弹部署在该位置
        center = pos[i - 1]

        # 跳过所有能被这颗炸弹爆破的建筑
        while i < n and pos[i] <= center + m:
            i += 1

    return ans

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]
    pos = data[2:2 + n]
    print(min_bombs(pos, m))

if __name__ == "__main__":
    main()
```

```java
            long left = pos[i];

            while (i < n && pos[i] <= left + m) {
                i++;
            }

            // 炸弹部署在该位置
            long center = pos[i - 1];

            // 跳过所有能被这颗炸弹爆破的建筑

                i++;
            }
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int minBombs(vector<int>& pos, int m) {
    // 先排序，方便从左到右贪心处理
    sort(pos.begin(), pos.end());

    int n = pos.size();
    int i = 0;
    int ans = 0;

    while (i < n) {
        ans++;

        // 当前最左边还没有被爆破的建筑
        long long left = pos[i];

        // 找到最右边且能覆盖 left 的炸弹部署位置
        while (i < n && pos[i] <= left + m) {
            i++;
        }

        // 炸弹部署在该位置
        long long center = pos[i - 1];

        // 跳过所有能被这颗炸弹爆破的建筑
        while (i < n && pos[i] <= center + m) {
            i++;
        }
    }

    return ans;
}

int main() {
    int n, m;
    cin >> n >> m;

    vector<int> pos(n);
    for (int i = 0; i < n; i++) {
        cin >> pos[i];
    }

    cout << minBombs(pos, m) << endl;

    return 0;
}
```
