## 解题思路

本题使用哈希表实现去重和费用统计。

核心规则如下：

* 对于每条计费日志，唯一性由时间戳、客户标识、计费因子共同决定。
* 如果同一客户、同一计费因子、相同时间戳出现多次，只保留第一次上报的日志。
* 如果计费时长不在$0$到$100$范围内，则按$0$处理。
* 如果计费因子查不到单价，则单价按$0$处理。
* 最后按照客户标识字典序升序输出。

实现方法：

* 使用集合$set$记录已经出现过的三元组$(timestamp, client, factor)$，用于去重。
* 使用哈希表记录每个客户的总费用。
* 遍历日志时，若该日志未重复，则计算费用：计费时长$\times$计费因子单价。
* 最后对客户名排序并输出。

## 复杂度分析

设计费日志条数为$n$，计费因子数量为$m$。

时间复杂度为$O(n+m+n\log n)$。

其中，读取单价表需要$O(m)$，遍历日志需要$O(n)$，最后对客户排序最坏需要$O(n\log n)$。

空间复杂度为$O(n+m)$。

其中，去重集合最多保存$n$条日志标识，单价表最多保存$m$个计费因子。

## 代码实现

### Python

```python
def calculate_bills(logs, prices):
    seen = set()          # 记录已经计费过的日志唯一标识
    total = {}            # 记录每个客户的总费用

    for timestamp, client, factor, duration_str in logs:
        key = (timestamp, client, factor)

        # 相同时间戳、客户、计费因子的日志只计费第一次
        if key in seen:
            continue
        seen.add(key)

        duration = int(duration_str)

        # 计费时长非法时按0处理
        if duration < 0 or duration > 100:
            duration = 0

        # 查不到计费因子时，单价按0处理
        price = prices.get(factor, 0)

        # 累加客户费用
        total[client] = total.get(client, 0) + duration * price

    return total


def main():
    n = int(input())
    logs = []

    # 读取计费日志
    for _ in range(n):
        line = input().strip()
        logs.append(line.split(","))

    m = int(input())
    prices = {}

    # 读取计费因子单价
    for _ in range(m):
        factor, price = input().strip().split(",")
        prices[factor] = int(price)

    total = calculate_bills(logs, prices)

    # 按客户标识字典序输出
    for client in sorted(total.keys()):
        print(client + "," + str(total[client]))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    // 计算每个客户的话单总费用
    public static Map<String, Integer> calculateBills(List<String[]> logs, Map<String, Integer> prices) {
        Set<String> seen = new HashSet<>();          // 记录已经计费过的日志唯一标识
        Map<String, Integer> total = new HashMap<>(); // 记录每个客户的总费用

        for (String[] log : logs) {
            String timestamp = log[0];
            String client = log[1];
            String factor = log[2];
            int duration = Integer.parseInt(log[3]);

            String key = timestamp + "#" + client + "#" + factor;

            // 相同时间戳、客户、计费因子的日志只计费第一次
            if (seen.contains(key)) {
                continue;
            }
            seen.add(key);

            // 计费时长非法时按0处理
            if (duration < 0 || duration > 100) {
                duration = 0;
            }

            // 查不到计费因子时，单价按0处理
            int price = prices.getOrDefault(factor, 0);

            // 累加客户费用
            total.put(client, total.getOrDefault(client, 0) + duration * price);
        }

        return total;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = Integer.parseInt(sc.nextLine());
        List<String[]> logs = new ArrayList<>();

        // 读取计费日志
        for (int i = 0; i < n; i++) {
            logs.add(sc.nextLine().split(","));
        }

        int m = Integer.parseInt(sc.nextLine());
        Map<String, Integer> prices = new HashMap<>();

        // 读取计费因子单价
        for (int i = 0; i < m; i++) {
            String[] arr = sc.nextLine().split(",");
            prices.put(arr[0], Integer.parseInt(arr[1]));
        }

        Map<String, Integer> total = calculateBills(logs, prices);

        // 按客户标识字典序输出
        List<String> clients = new ArrayList<>(total.keySet());
        Collections.sort(clients);

        for (String client : clients) {
            System.out.println(client + "," + total.get(client));
        }

        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <sstream>
using namespace std;

// 按逗号分割字符串
vector<string> split(const string& s) {
    vector<string> res;
    string item;
    stringstream ss(s);

    while (getline(ss, item, ',')) {
        res.push_back(item);
    }

    return res;
}

// 计算每个客户的话单总费用
map<string, int> calculateBills(vector<vector<string>>& logs, unordered_map<string, int>& prices) {
    unordered_set<string> seen; // 记录已经计费过的日志唯一标识
    map<string, int> total;     // map会按客户标识字典序排序

    for (auto& log : logs) {
        string timestamp = log[0];
        string client = log[1];
        string factor = log[2];
        int duration = stoi(log[3]);

        string key = timestamp + "#" + client + "#" + factor;

        // 相同时间戳、客户、计费因子的日志只计费第一次
        if (seen.count(key)) {
            continue;
        }
        seen.insert(key);

        // 计费时长非法时按0处理
        if (duration < 0 || duration > 100) {
            duration = 0;
        }

        // 查不到计费因子时，单价按0处理
        int price = 0;
        if (prices.count(factor)) {
            price = prices[factor];
        }

        // 累加客户费用
        total[client] += duration * price;
    }

    return total;
}

int main() {
    int n;
    cin >> n;

    vector<vector<string>> logs;
    string line;

    // 读取换行符
    getline(cin, line);

    // 读取计费日志
    for (int i = 0; i < n; i++) {
        getline(cin, line);
        logs.push_back(split(line));
    }

    int m;
    cin >> m;
    getline(cin, line);

    unordered_map<string, int> prices;

    // 读取计费因子单价
    for (int i = 0; i < m; i++) {
        getline(cin, line);
        vector<string> arr = split(line);
        prices[arr[0]] = stoi(arr[1]);
    }

    map<string, int> total = calculateBills(logs, prices);

    // 按客户标识字典序输出
    for (auto& item : total) {
        cout << item.first << "," << item.second << endl;
    }

    return 0;
}
```
