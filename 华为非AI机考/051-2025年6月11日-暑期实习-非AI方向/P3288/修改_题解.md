# 枚举扩展朴素解法

这道题的正解是 $单调栈$ 求无交换最大矩形，再用 $稀疏表RMQ$ 处理至多一次交换，但是我们用朴素解法 $O(n^2)$ 向左右扩展（并可枚举一次交换）也能在考试时拿到一定的分数。

题意：直方图有 $n$ 根宽为 $1$ 的柱子，高度为 $h_i$。允许至多交换任意两根柱子的高度一次，求能勾勒出的最大矩形面积。

朴素做法：

1. **无交换**：把每个下标 $i$ 当作矩形高度，向左、向右扩展到高度 $<h_i$ 为止，面积为 $h_i\times(R-L+1)$，对所有 $i$ 取最大。时间 $O(n^2)$。
2. **一次交换**：再枚举所有交换对 $(p,q)$，交换后同样用上面的 $O(n^2)$ 扩展求面积，最后与无交换答案取最大。总时间约 $O(n^4)$。

小数据可以算出正确答案；当 $n$ 接近 $10^5$ 时会超时，需要后面的正解。
## 枚举扩展解法

```python
def max_rect_no_swap(h):
    n = len(h)
    ans = 0
    for i in range(n):
        # 以 h[i] 为高度向左右扩展
        L = i
        while L > 0 and h[L - 1] >= h[i]:
            L -= 1
        R = i
        while R + 1 < n and h[R + 1] >= h[i]:
            R += 1
        ans = max(ans, h[i] * (R - L + 1))
    return ans


def max_rect_with_one_swap(h):
    n = len(h)
    ans = max_rect_no_swap(h)
    # 枚举一次交换
    for p in range(n):
        for q in range(p + 1, n):
            h[p], h[q] = h[q], h[p]
            ans = max(ans, max_rect_no_swap(h))
            h[p], h[q] = h[q], h[p]
    return ans


n = int(input())
h = list(map(int, input().split()))
print(max_rect_with_one_swap(h))
```

```cpp
long long max_rect_no_swap(const vector<int>& h) {
    int n = (int)h.size();

            swap(h[p], h[q]);
            ans = max(ans, max_rect_no_swap(h));
            swap(h[p], h[q]);
        }
    }
    return ans;
```

# 正确解法

## 题目分析

先求出**不交换**时的柱状图最大矩形面积，再用单调栈在 $O(n)$ 内完成。

考虑一次交换带来的增益：对每根柱子 $i$（高度 $H=h_i$），希望以高度 $H$ 画出尽量宽的矩形。

1. 用稀疏表做区间最小值查询（RMQ）。二分找出包含 $i$ 的最长连续段 $[L_0,R_0]$，使得段内高度均 $\ge H$。
2. 统计全局高度 $\ge H$ 的柱子个数 $count\_good$。
3. 在 $[L_0,R_0]$ 外侧，还可再「借」一根高度 $\ge H$ 的柱子（靠一次交换搬进来）。分别在 $L_0$ 左侧、$R_0$ 右侧找最长的「整段高度均 $\ge H$」的连续段，长度记为 $w_{left}$、$w_{right}$。交换后可达宽度上界为

$$w=(R_0-L_0+1)+1+\max(w_{left},w_{right})$$

再与 $count\_good$ 取 $\min$，面积为 $H\times\min(w,count\_good)$。

对所有 $i$ 取最大，再与无交换基准答案取最大即可。

## 复杂度分析

- 构建稀疏表：$O(n\log n)$。
- 无交换单调栈：$O(n)$。
- 枚举每根柱子，配合 RMQ 二分：$O(n\log n)$。
- 总时间 $O(n\log n)$，空间 $O(n\log n)$，可应对 $n\le 10^5$。

```python
import bisect


def max_area_no_swap(h):
    # 单调栈求柱状图最大矩形
    st = [-1]
    ans = 0
    for i in range(len(h) + 1):
        cur = 0 if i == len(h) else h[i]
        while st[-1] != -1 and h[st[-1]] >= cur:
            height = h[st.pop()]
            width = i - st[-1] - 1
            ans = max(ans, height * width)
        st.append(i)
    return ans


n = int(input())
h = list(map(int, input().split()))

base = max_area_no_swap(h)

# 稀疏表 RMQ：区间最小值
LOG = n.bit_length()
st = [h[:]]
for k in range(1, LOG):
    prev = st[k - 1]
    half = 1 << (k - 1)
    curr = [min(prev[i], prev[i + half]) for i in range(n - (1 << k) + 1)]
    st.append(curr)
log2 = [0] * (n + 1)
for i in range(2, n + 1):
    log2[i] = log2[i // 2] + 1


def rmq(l, r):
    k = log2[r - l + 1]
    return min(st[k][l], st[k][r - (1 << k) + 1])


def find_leftmost(L, R, H):
    ans = -1
    l, r = L, R
    while l <= r:
        m = (l + r) // 2
        if rmq(m, R) >= H:
            ans = m
            r = m - 1
        else:
            l = m + 1
    return ans


def find_rightmost(L, R, H):
    ans = -1
    l, r = L, R
    while l <= r:
        m = (l + r) // 2
        if rmq(L, m) >= H:
            ans = m
            l = m + 1
        else:
            r = m - 1
    return ans


hs = sorted(h)
ans = base
for i, H in enumerate(h):
    l0 = find_leftmost(0, i, H)
    r0 = find_rightmost(i, n - 1, H)
    if l0 == -1 or r0 == -1:
        continue
    w0 = r0 - l0 + 1

    w_left = 0
    if l0 >= 2:
        end = l0 - 2
        stl = find_leftmost(0, end, H)
        if stl != -1:
            w_left = end - stl + 1

    w_right = 0
    if r0 + 2 <= n - 1:
        start = r0 + 2
        stp = find_rightmost(start, n - 1, H)
        if stp != -1:
            w_right = stp - start + 1

    count_good = n - bisect.bisect_left(hs, H)
    w = min(w0 + 1 + max(w_left, w_right), count_good)
    ans = max(ans, w * H)

print(ans)
```

### Java

```java
import java.util.Arrays;
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Main {
    static int n;
    static int[] h;
    static int[][] st;
    static int[] log2;

    static long maxAreaNoSwap() {
        // 单调栈求柱状图最大矩形
        Deque<Integer> stack = new ArrayDeque<>();
        stack.push(-1);
        long res = 0;
        for (int i = 0; i <= n; i++) {
            int cur = (i == n ? 0 : h[i]);
            while (stack.peek() != -1 && h[stack.peek()] >= cur) {
                int height = h[stack.pop()];
                int width = i - stack.peek() - 1;
                res = Math.max(res, 1L * height * width);
            }
            stack.push(i);
        }
        return res;
    }

    static void buildSparseTable() {
        log2 = new int[n + 1];
        for (int i = 2; i <= n; i++) log2[i] = log2[i / 2] + 1;
        int K = log2[n] + 1;
        st = new int[K][n];
        System.arraycopy(h, 0, st[0], 0, n);
        for (int k = 1; k < K; k++) {
            for (int i = 0; i + (1 << k) <= n; i++) {
                st[k][i] = Math.min(st[k - 1][i], st[k - 1][i + (1 << (k - 1))]);
            }
        }
    }

    static int rmq(int l, int r) {
        int k = log2[r - l + 1];
        return Math.min(st[k][l], st[k][r - (1 << k) + 1]);
    }

    static int findLeftmost(int L, int R, int H) {
        int ans = -1, l = L, r = R;
        while (l <= r) {
            int m = (l + r) >> 1;
            if (rmq(m, R) >= H) {
                ans = m;
                r = m - 1;
            } else {
                l = m + 1;
            }
        }
        return ans;
    }

    static int findRightmost(int L, int R, int H) {
        int ans = -1, l = L, r = R;
        while (l <= r) {
            int m = (l + r) >> 1;
            if (rmq(L, m) >= H) {
                ans = m;
                l = m + 1;
            } else {
                r = m - 1;
            }
        }
        return ans;
    }

    // 第一个 >= x 的下标
    static int lowerBound(int[] arr, int x) {
        int l = 0, r = arr.length;
        while (l < r) {
            int m = (l + r) >> 1;
            if (arr[m] < x) l = m + 1;
            else r = m;
        }
        return l;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        h = new int[n];
        for (int i = 0; i < n; i++) h[i] = sc.nextInt();

        long ans = maxAreaNoSwap();
        buildSparseTable();
        int[] hs = Arrays.copyOf(h, n);
        Arrays.sort(hs);

        for (int i = 0; i < n; i++) {
            int H = h[i];
            int l0 = findLeftmost(0, i, H);
            int r0 = findRightmost(i, n - 1, H);
            if (l0 < 0 || r0 < 0) continue;
            int w0 = r0 - l0 + 1;

            int wLeft = 0;
            if (l0 >= 2) {
                int end = l0 - 2;
                int stl = findLeftmost(0, end, H);
                if (stl >= 0) wLeft = end - stl + 1;
            }

            int wRight = 0;
            if (r0 + 2 <= n - 1) {
                int start = r0 + 2;
                int str = findRightmost(start, n - 1, H);
                if (str >= 0) wRight = str - start + 1;
            }

            int countGood = n - lowerBound(hs, H);
            int w = Math.min(w0 + 1 + Math.max(wLeft, wRight), countGood);
            ans = Math.max(ans, 1L * H * w);
        }
        System.out.println(ans);
    }
}
```

```cpp
#include <algorithm>
#include <climits>
#include <iostream>
#include <stack>
#include <vector>
using namespace std;
using ll = long long;

struct SparseTable {
    int n, LOG;
    vector<int> log2v;
    vector<vector<int>> st;

    SparseTable(const vector<int>& a) {
        n = (int)a.size();
        log2v.resize(n + 1);
        log2v[1] = 0;
        for (int i = 2; i <= n; i++) log2v[i] = log2v[i / 2] + 1;
        LOG = log2v[n];
        st.assign(LOG + 1, vector<int>(n));
        st[0] = a;
        for (int k = 1; k <= LOG; k++) {
            int half = 1 << (k - 1);
            for (int i = 0; i + (1 << k) <= n; i++) {
                st[k][i] = min(st[k - 1][i], st[k - 1][i + half]);
            }
        }
    }

    int query(int l, int r) const {
        if (l > r) return INT_MAX;
        int k = log2v[r - l + 1];
        return min(st[k][l], st[k][r - (1 << k) + 1]);
    }
};

ll max_area_no_swap(const vector<int>& h) {
    // 单调栈求柱状图最大矩形
    int n = (int)h.size();
    stack<int> st;
    st.push(-1);
    ll ans = 0;
    for (int i = 0; i <= n; i++) {
        int cur = (i == n ? 0 : h[i]);
        while (st.top() != -1 && h[st.top()] >= cur) {
            int height = h[st.top()];
            st.pop();
            int width = i - st.top() - 1;
            ans = max(ans, (ll)height * width);
        }
        st.push(i);
    }
    return ans;
}

int find_leftmost(int l, int r, const SparseTable& st, int H, int rightEnd) {
    int ans = -1;
    while (l <= r) {
        int m = (l + r) >> 1;
        if (st.query(m, rightEnd) >= H) {
            ans = m;
            r = m - 1;
        } else {
            l = m + 1;
        }
    }
    return ans;
}

int find_rightmost(int l, int r, const SparseTable& st, int H, int leftStart) {
    int ans = -1;
    while (l <= r) {
        int m = (l + r) >> 1;
        if (st.query(leftStart, m) >= H) {
            ans = m;
            l = m + 1;
        } else {
            r = m - 1;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];

    ll max_area = max_area_no_swap(h);
    SparseTable st(h);
    vector<int> hs = h;
    sort(hs.begin(), hs.end());

    for (int i = 0; i < n; i++) {
        int H = h[i];
        int count_good = n - int(lower_bound(hs.begin(), hs.end(), H) - hs.begin());

        int l0 = find_leftmost(0, i, st, H, i);
        int r0 = find_rightmost(i, n - 1, st, H, i);
        if (l0 == -1 || r0 == -1) continue;
        int w0 = r0 - l0 + 1;

        int w_left = 0;
        if (l0 >= 2) {
            int end = l0 - 2;
            int start = find_leftmost(0, end, st, H, end);
            if (start != -1) w_left = end - start + 1;
        }

        int w_right = 0;
        if (r0 + 2 <= n - 1) {
            int start = r0 + 2;
            int end = find_rightmost(start, n - 1, st, H, start);
            if (end != -1) w_right = end - start + 1;
        }

        int w_swap = w0 + 1 + max(w_left, w_right);
        int achievable = min(w_swap, count_good);
        max_area = max(max_area, (ll)H * achievable);
    }

    cout << max_area << "\n";
    return 0;
}
```
