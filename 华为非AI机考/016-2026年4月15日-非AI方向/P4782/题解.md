## 解题思路

题目中的一次操作是：选择节点 $x$，使得 $x$ 子树中与 $x$ 距离为偶数的节点翻转，与 $x$ 距离为奇数的节点不变。

设节点 $i$ 当前值与目标值是否不同为：

$$
diff_i = init_i \oplus goal_i
$$

那么问题就变成：最少选择多少个节点操作，使得每个节点最终被翻转的次数模 $2$ 后恰好等于 $diff_i$。

### 影响规律分析

对于某个节点 $u$，只有它的祖先节点中的某些操作会影响它。

若祖先节点为 $x$，那么当且仅当：

$$
depth[u] - depth[x]
$$

为偶数时，节点 $u$ 才会被节点 $x$ 的操作影响。

这等价于：

* 若 $depth[u]$ 是偶数，则只会受到路径上**偶数层**被操作节点的影响
* 若 $depth[u]$ 是奇数，则只会受到路径上**奇数层**被操作节点的影响

因此，沿着从根到当前节点的路径，只需要分别记录：

* $flip0$：偶数层节点操作次数的异或和
* $flip1$：奇数层节点操作次数的异或和

### 贪心 + DFS

从根节点开始 DFS。

访问节点 $u$ 时：

* 若 $u$ 在偶数层，则当前实际值为

$$
init_u \oplus flip0
$$

* 若 $u$ 在奇数层，则当前实际值为

$$
init_u \oplus flip1
$$

如果当前值已经等于目标值，那么不需要操作。

如果当前值不等于目标值，那么**必须在当前节点操作一次**。因为之后只能操作它的后代节点，而后代节点的操作不会再影响祖先节点 $u$，所以这是唯一选择，也是最优选择。

操作当前节点后：

* 答案加 $1$
* 如果当前节点在偶数层，则令 $flip0 \oplus= 1$
* 如果当前节点在奇数层，则令 $flip1 \oplus= 1$

然后继续递归处理子节点即可。

### 相关算法

这里使用的算法是：

* 邻接表建树
* 深度优先搜索 $DFS$
* 贪心决策
* 异或状态维护

整个过程只需要遍历整棵树一次。

## 复杂度分析

设节点数为 $n$。

建图需要 $O(n)$。

DFS 遍历每个节点一次，每条边访问两次，复杂度也是 $O(n)$。

因此总时间复杂度为：

$$
O(n)
$$

邻接表、递归栈等额外空间复杂度为：

$$
O(n)
$$

对于题目给出的数据范围，复杂度是完全可行的。

## 代码实现

### Python

```python
import sys

# 题面功能：返回最少操作次数
def solve(n, edges, init, goal):
    # 建立邻接表
    graph = [[] for _ in range(n + 1)]
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    ans = 0
    sys.setrecursionlimit(300000)

    # 深度优先搜索
    # u: 当前节点
    # parent: 父节点
    # depth: 当前深度
    # flip0: 当前路径上偶数层操作的异或值
    # flip1: 当前路径上奇数层操作的异或值
    def dfs(u, parent, depth, flip0, flip1):
        nonlocal ans

        # 计算当前节点在祖先操作影响后的实际值
        if depth % 2 == 0:
            cur = init[u] ^ flip0
        else:
            cur = init[u] ^ flip1

        # 如果当前值不等于目标值，则必须在当前节点操作
        if cur != goal[u]:
            ans += 1
            if depth % 2 == 0:
                flip0 ^= 1
            else:
                flip1 ^= 1

        # 递归处理子节点
        for v in graph[u]:
            if v != parent:
                dfs(v, u, depth + 1, flip0, flip1)

    dfs(1, 0, 0, 0, 0)
    return ans


def main():
    input = sys.stdin.readline

    n = int(input().strip())
    edges = []

    # 输入边
    for _ in range(n - 1):
        u, v = map(int, input().split())
        edges.append((u, v))

    # 输入初始状态
    init_list = list(map(int, input().split()))
    init = [0] + init_list

    # 输入目标状态
    goal_list = list(map(int, input().split()))
    goal = [0] + goal_list

    print(solve(n, edges, init, goal))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;
import java.util.StringTokenizer;

public class Main {
    // 邻接表
    static ArrayList<Integer>[] graph;
    static int[] init;
    static int[] goal;
    static int ans = 0;

    // 题面功能：返回最少操作次数
    static int solve(int n) {
        dfs(1, 0, 0, 0, 0);
        return ans;
    }

    // 深度优先搜索
    // u: 当前节点
    // parent: 父节点
    // depth: 当前深度
    // flip0: 当前路径上偶数层操作的异或值
    // flip1: 当前路径上奇数层操作的异或值
    static void dfs(int u, int parent, int depth, int flip0, int flip1) {
        int cur;

        // 计算当前节点在祖先操作影响后的实际值
        if ((depth & 1) == 0) {
            cur = init[u] ^ flip0;
        } else {
            cur = init[u] ^ flip1;
        }

        // 如果当前值不等于目标值，则必须在当前节点操作
        if (cur != goal[u]) {
            ans++;
            if ((depth & 1) == 0) {
                flip0 ^= 1;
            } else {
                flip1 ^= 1;
            }
        }

        // 递归处理子节点
        for (int v : graph[u]) {
            if (v != parent) {
                dfs(v, u, depth + 1, flip0, flip1);
            }
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());

        graph = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        // 输入边并建图
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            graph[u].add(v);
            graph[v].add(u);
        }

        init = new int[n + 1];
        goal = new int[n + 1];

        // 输入初始状态
        StringTokenizer st1 = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            init[i] = Integer.parseInt(st1.nextToken());
        }

        // 输入目标状态
        StringTokenizer st2 = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            goal[i] = Integer.parseInt(st2.nextToken());
        }

        System.out.println(solve(n));
    }
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

vector<vector<int> > graph;
vector<int> initVal, goalVal;
int ans = 0;

// 深度优先搜索
// u: 当前节点
// parent: 父节点
// depth: 当前深度
// flip0: 当前路径上偶数层操作的异或值
// flip1: 当前路径上奇数层操作的异或值
void dfs(int u, int parent, int depth, int flip0, int flip1) {
    int cur;

    // 计算当前节点在祖先操作影响后的实际值
    if (depth % 2 == 0) {
        cur = initVal[u] ^ flip0;
    } else {
        cur = initVal[u] ^ flip1;
    }

    // 如果当前值不等于目标值，则必须在当前节点操作
    if (cur != goalVal[u]) {
        ans++;
        if (depth % 2 == 0) {
            flip0 ^= 1;
        } else {
            flip1 ^= 1;
        }
    }

    // 递归处理子节点
    for (int v : graph[u]) {
        if (v != parent) {
            dfs(v, u, depth + 1, flip0, flip1);
        }
    }
}

// 题面功能：返回最少操作次数
int solve(int n) {
    dfs(1, 0, 0, 0, 0);
    return ans;
}

int main() {
    int n;
    cin >> n;

    graph.assign(n + 1, vector<int>());
    initVal.assign(n + 1, 0);
    goalVal.assign(n + 1, 0);

        cin >> u >> v;
        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    // 输入初始状态
    for (int i = 1; i <= n; i++) {
        cin >> initVal[i];
    }

    // 输入目标状态
    for (int i = 1; i <= n; i++) {
        cin >> goalVal[i];
    }

    cout << solve(n) << '\n';
```
