## 解题思路

根据前序遍历和中序遍历可以唯一还原二叉树。

前序遍历的第一个节点一定是当前子树根节点，利用中序遍历中根节点的位置，可以划分出左子树和右子树。

本题不需要真正建树，可以在递归还原子树的过程中，直接计算每个节点的完整镜像大小。

设当前节点的父节点完整镜像大小为 $parentSum$，当前节点镜像层大小为 $val$，则当前节点完整镜像大小为：

$curSum = parentSum + val$

如果 $curSum <= 0$，说明当前节点异常，需要剪枝，整棵子树都不参与统计。

否则，将 $curSum$ 加入总和，并继续递归处理左右子树。

使用的算法是：二叉树递归还原 + 深度优先搜索。

## 复杂度分析

设节点数量为 $n$。

时间复杂度为 $O(n)$，每个节点最多被访问一次，中序下标通过哈希表查询。

空间复杂度为 $O(n)$，主要来自哈希表和递归调用栈。

## 代码实现

### Python

```python
import sys

def calc_average(preorder, inorder):
    n = len(preorder)
    pos = {}

    # 记录中序遍历中每个节点值的位置
    for i, v in enumerate(inorder):
        pos[v] = i

    sys.setrecursionlimit(3000)

    def dfs(pl, pr, il, ir, parent_sum):
        # 空子树
        if pl > pr:
            return 0, 0

        root = preorder[pl]
        idx = pos[root]
        cur_sum = parent_sum + root

        # 当前节点异常，整棵子树剪枝
        if cur_sum <= 0:
            return 0, 0

        left_size = idx - il

        # 递归统计左子树
        left_sum, left_cnt = dfs(
            pl + 1,
            pl + left_size,
            il,
            idx - 1,
            cur_sum
        )

        # 递归统计右子树
        right_sum, right_cnt = dfs(
            pl + left_size + 1,
            pr,
            idx + 1,
            ir,
            cur_sum
        )

        return cur_sum + left_sum + right_sum, 1 + left_cnt + right_cnt

    total, cnt = dfs(0, n - 1, 0, n - 1, 0)

    if cnt == 0:
        return 0
    return total // cnt


if __name__ == "__main__":
    preorder = list(map(int, input().split()))
    inorder = list(map(int, input().split()))

    print(calc_average(preorder, inorder))
```

### Java

```java
import java.util.*;

public class Main {
    static int[] preorder;
    static int[] inorder;
    static Map<Integer, Integer> pos = new HashMap<>();

    static class Result {
        long sum;
        int count;

        Result(long sum, int count) {
            this.sum = sum;
            this.count = count;
        }
    }

    static Result dfs(int pl, int pr, int il, int ir, long parentSum) {
        // 空子树
        if (pl > pr) {
            return new Result(0, 0);
        }

        int root = preorder[pl];
        int idx = pos.get(root);
        long curSum = parentSum + root;

        // 当前节点异常，整棵子树剪枝
        if (curSum <= 0) {
            return new Result(0, 0);
        }

        int leftSize = idx - il;

        // 递归统计左子树
        Result left = dfs(
            pl + 1,
            pl + leftSize,
            il,
            idx - 1,
            curSum
        );

        // 递归统计右子树
        Result right = dfs(
            pl + leftSize + 1,
            pr,
            idx + 1,
            ir,
            curSum
        );

        return new Result(
            curSum + left.sum + right.sum,
            1 + left.count + right.count
        );
    }

    static long calcAverage(int[] pre, int[] in) {
        preorder = pre;
        inorder = in;
        pos.clear();

        // 记录中序遍历中每个节点值的位置
        for (int i = 0; i < inorder.length; i++) {
            pos.put(inorder[i], i);
        }

        Result ans = dfs(0, preorder.length - 1, 0, inorder.length - 1, 0);

        if (ans.count == 0) {
            return 0;
        }
        return ans.sum / ans.count;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String preLine = sc.nextLine();
        String inLine = sc.nextLine();

        String[] preStr = preLine.split(" ");
        String[] inStr = inLine.split(" ");

        int[] pre = new int[preStr.length];
        int[] in = new int[inStr.length];

        for (int i = 0; i < preStr.length; i++) {
            pre[i] = Integer.parseInt(preStr[i]);
        }

        for (int i = 0; i < inStr.length; i++) {
            in[i] = Integer.parseInt(inStr[i]);
        }

        System.out.println(calcAverage(pre, in));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <unordered_map>
#include <sstream>
using namespace std;

vector<int> preorder, inorder;
unordered_map<int, int> pos;

struct Result {
    long long sum;
    int count;
};

Result dfs(int pl, int pr, int il, int ir, long long parentSum) {
    // 空子树
    if (pl > pr) {
        return {0, 0};
    }

    int root = preorder[pl];
    int idx = pos[root];
    long long curSum = parentSum + root;

    // 当前节点异常，整棵子树剪枝
    if (curSum <= 0) {
        return {0, 0};
    }

    int leftSize = idx - il;

    // 递归统计左子树
    Result left = dfs(
        pl + 1,
        pl + leftSize,
        il,
        idx - 1,
        curSum
    );

    // 递归统计右子树
    Result right = dfs(
        pl + leftSize + 1,
        pr,
        idx + 1,
        ir,
        curSum
    );

    return {
        curSum + left.sum + right.sum,
        1 + left.count + right.count
    };
}

long long calcAverage(vector<int>& pre, vector<int>& in) {
    preorder = pre;
    inorder = in;
    pos.clear();

    // 记录中序遍历中每个节点值的位置
    for (int i = 0; i < (int)inorder.size(); i++) {
        pos[inorder[i]] = i;
    }

    Result ans = dfs(0, preorder.size() - 1, 0, inorder.size() - 1, 0);

    if (ans.count == 0) {
        return 0;
    }
    return ans.sum / ans.count;
}

vector<int> parseLine(string line) {
    vector<int> nums;
    stringstream ss(line);
    int x;

    // 读取一行中的所有整数
    while (ss >> x) {
        nums.push_back(x);
    }

    return nums;
}

int main() {
    string preLine, inLine;

    getline(cin, preLine);
    getline(cin, inLine);

    vector<int> pre = parseLine(preLine);
    vector<int> in = parseLine(inLine);

    cout << calcAverage(pre, in) << endl;

    return 0;
}
```
