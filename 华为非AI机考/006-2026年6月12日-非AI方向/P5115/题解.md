## 解题思路

将每个可通行格子看成图中的一个节点，节点状态为 $(floor,x,y)$，其中 $floor$ 表示楼层，$x$ 和 $y$ 表示坐标。

由于玩家每次移动一步的代价都是 $1$，包括通过楼梯跨楼层也算 $1$ 步，所以可以使用 $BFS$ 求最短路。

具体做法：

1. 读入 $2$ 层地图，找到起点 $2$、终点 $3$，以及两层楼的楼梯 $4$。
2. 从起点开始做 $BFS$。
3. 每次可以向上下左右 $4$ 个方向移动，不能越界，不能走到墙壁 $1$。
4. 如果当前位置是当前楼层的楼梯，可以用 $1$ 步移动到另一层的楼梯位置。
5. 第一次到达终点时，当前步数就是最短路径长度。
6. 如果 $BFS$ 结束仍未到达终点，返回 $-1$。

## 复杂度分析

地图共有 $2 \times m \times n$ 个格子，每个格子最多访问一次。

时间复杂度为 $O(mn)$。

空间复杂度为 $O(mn)$，用于保存距离数组和队列。

## 代码实现

### Python

```python
from collections import deque


def shortest_path(grid, m, n):
    start = None
    end = None
    stairs = [None, None]

    # 找到起点、终点和两层楼的楼梯
    for f in range(2):
        for i in range(m):
            for j in range(n):
                if grid[f][i][j] == 2:
                    start = (f, i, j)
                elif grid[f][i][j] == 3:
                    end = (f, i, j)
                elif grid[f][i][j] == 4:
                    stairs[f] = (i, j)

    dist = [[[-1] * n for _ in range(m)] for _ in range(2)]
    q = deque()

    sf, sx, sy = start
    dist[sf][sx][sy] = 0
    q.append(start)

    dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

    while q:
        f, x, y = q.popleft()

        # 第一次到达终点就是最短距离
        if (f, x, y) == end:
            return dist[f][x][y]

        # 同一层上下左右移动
        for dx, dy in dirs:
            nx, ny = x + dx, y + dy
            if 0 <= nx < m and 0 <= ny < n:
                if grid[f][nx][ny] != 1 and dist[f][nx][ny] == -1:
                    dist[f][nx][ny] = dist[f][x][y] + 1
                    q.append((f, nx, ny))

        # 如果在楼梯上，可以移动到另一层楼梯
        if (x, y) == stairs[f]:
            nf = 1 - f
            nx, ny = stairs[nf]
            if dist[nf][nx][ny] == -1:
                dist[nf][nx][ny] = dist[f][x][y] + 1
                q.append((nf, nx, ny))

    return -1


def main():
    m, n = map(int, input().split())

    grid = []
    for _ in range(2):
        floor = []
        for _ in range(m):
            floor.append(list(map(int, input().split())))
        grid.append(floor)

    print(shortest_path(grid, m, n))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int shortestPath(int[][][] grid, int m, int n) {
        int[] start = new int[3];
        int[] end = new int[3];
        int[][] stairs = new int[2][2];

        // 找到起点、终点和两层楼的楼梯
        for (int f = 0; f < 2; f++) {
            for (int i = 0; i < m; i++) {
                for (int j = 0; j < n; j++) {
                    if (grid[f][i][j] == 2) {
                        start[0] = f;
                        start[1] = i;
                        start[2] = j;
                    } else if (grid[f][i][j] == 3) {
                        end[0] = f;
                        end[1] = i;
                        end[2] = j;
                    } else if (grid[f][i][j] == 4) {
                        stairs[f][0] = i;
                        stairs[f][1] = j;
                    }
                }
            }
        }

        int[][][] dist = new int[2][m][n];
        for (int f = 0; f < 2; f++) {
            for (int i = 0; i < m; i++) {
                for (int j = 0; j < n; j++) {
                    dist[f][i][j] = -1;
                }
            }
        }

        Queue<int[]> queue = new LinkedList<>();
        dist[start[0]][start[1]][start[2]] = 0;
        queue.offer(start);

        int[] dx = {1, -1, 0, 0};
        int[] dy = {0, 0, 1, -1};

        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            int f = cur[0];
            int x = cur[1];
            int y = cur[2];

            // 第一次到达终点就是最短距离
            if (f == end[0] && x == end[1] && y == end[2]) {
                return dist[f][x][y];
            }

            // 同一层上下左右移动
            for (int k = 0; k < 4; k++) {
                int nx = x + dx[k];
                int ny = y + dy[k];

                if (nx >= 0 && nx < m && ny >= 0 && ny < n) {
                    if (grid[f][nx][ny] != 1 && dist[f][nx][ny] == -1) {
                        dist[f][nx][ny] = dist[f][x][y] + 1;
                        queue.offer(new int[]{f, nx, ny});
                    }
                }
            }

            // 如果在楼梯上，可以移动到另一层楼梯
            if (x == stairs[f][0] && y == stairs[f][1]) {
                int nf = 1 - f;
                int nx = stairs[nf][0];
                int ny = stairs[nf][1];

                if (dist[nf][nx][ny] == -1) {
                    dist[nf][nx][ny] = dist[f][x][y] + 1;
                    queue.offer(new int[]{nf, nx, ny});
                }
            }
        }

        return -1;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int m = sc.nextInt();
        int n = sc.nextInt();

        int[][][] grid = new int[2][m][n];

        for (int f = 0; f < 2; f++) {
            for (int i = 0; i < m; i++) {
                for (int j = 0; j < n; j++) {
                    grid[f][i][j] = sc.nextInt();
                }
            }
        }

        System.out.println(shortestPath(grid, m, n));
        sc.close();
    }
}
```

```cpp
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

struct Node {
    int f;
    int x;
    int y;
};

int shortestPath(vector<vector<vector<int>>>& grid, int m, int n) {
    Node start, target;
    pair<int, int> stairs[2];

    // 找到起点、终点和两层楼的楼梯
    for (int f = 0; f < 2; f++) {
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[f][i][j] == 2) {
                    start = {f, i, j};
                } else if (grid[f][i][j] == 3) {
                    target = {f, i, j};
                } else if (grid[f][i][j] == 4) {
                    stairs[f] = {i, j};
                }
            }
        }
    }

    vector<vector<vector<int>>> dist(2, vector<vector<int>>(m, vector<int>(n, -1)));
    queue<Node> q;

    dist[start.f][start.x][start.y] = 0;
    q.push(start);

    int dx[4] = {1, -1, 0, 0};
    int dy[4] = {0, 0, 1, -1};

    while (!q.empty()) {
        Node cur = q.front();
        q.pop();

        int f = cur.f;
        int x = cur.x;
        int y = cur.y;

        // 第一次到达终点就是最短距离
        if (f == target.f && x == target.x && y == target.y) {
            return dist[f][x][y];
        }

        // 同一层上下左右移动
        for (int k = 0; k < 4; k++) {
            int nx = x + dx[k];
            int ny = y + dy[k];

            if (nx >= 0 && nx < m && ny >= 0 && ny < n) {
                if (grid[f][nx][ny] != 1 && dist[f][nx][ny] == -1) {
                    dist[f][nx][ny] = dist[f][x][y] + 1;
                    q.push({f, nx, ny});
                }
            }
        }

        // 如果在楼梯上，可以移动到另一层楼梯
        if (x == stairs[f].first && y == stairs[f].second) {
            int nf = 1 - f;
            int nx = stairs[nf].first;
            int ny = stairs[nf].second;

            if (dist[nf][nx][ny] == -1) {
                dist[nf][nx][ny] = dist[f][x][y] + 1;
                q.push({nf, nx, ny});
            }
        }
    }

    return -1;
}

int main() {
    int m, n;
    cin >> m >> n;

    vector<vector<vector<int>>> grid(2, vector<vector<int>>(m, vector<int>(n)));

    for (int f = 0; f < 2; f++) {
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                cin >> grid[f][i][j];
            }
        }
    }

    cout << shortestPath(grid, m, n) << endl;

    return 0;
}
```
