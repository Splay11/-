## 解题思路

本题使用的是$循环异或$算法。

由于每个十六进制字符都对应$4$位二进制，所以可以直接按十六进制字符进行处理。

对于$data$中的第$i$个字符：

* 将$data[i]$转换成对应的十六进制数值
* 将$key[i \bmod key.length]$转换成对应的十六进制数值
* 两个数值进行异或运算
* 将结果再转换成大写十六进制字符

这样就等价于题目中要求的$4$位二进制异或，同时也实现了$key$循环使用。

## 复杂度分析

设$data$长度为$n$，$key$长度为$m$。

时间复杂度为$O(n)$，每个$data$字符只处理一次。

空间复杂度为$O(n)$，用于保存输出结果。

## 代码实现

### Python

```python
# 将十六进制字符转换为对应数值
def hex_to_val(c):
    if '0' <= c <= '9':
        return ord(c) - ord('0')
    return ord(c) - ord('A') + 10


# 将数值转换为大写十六进制字符
def val_to_hex(x):
    if x < 10:
        return chr(ord('0') + x)
    return chr(ord('A') + x - 10)


# 实现循环异或
def cyclic_xor(key, data):
    res = []
    m = len(key)

    for i in range(len(data)):
        # key 按照下标循环使用
        a = hex_to_val(data[i])
        b = hex_to_val(key[i % m])
        res.append(val_to_hex(a ^ b))

    return ''.join(res)


def main():
    t = int(input())
    for _ in range(t):
        key, data = input().split()
        print(cyclic_xor(key, data))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    // 将十六进制字符转换为对应数值
    public static int hexToVal(char c) {
        if (c >= '0' && c <= '9') {
            return c - '0';
        }
        return c - 'A' + 10;
    }

    // 将数值转换为大写十六进制字符
    public static char valToHex(int x) {
        if (x < 10) {
            return (char)('0' + x);
        }
        return (char)('A' + x - 10);
    }

    // 实现循环异或
    public static String cyclicXor(String key, String data) {
        StringBuilder sb = new StringBuilder();
        int m = key.length();

        for (int i = 0; i < data.length(); i++) {
            // key 按照下标循环使用
            int a = hexToVal(data.charAt(i));
            int b = hexToVal(key.charAt(i % m));
            sb.append(valToHex(a ^ b));
        }

        return sb.toString();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        StringBuilder ans = new StringBuilder();

        for (int i = 0; i < t; i++) {
            String key = sc.next();
            String data = sc.next();

            ans.append(cyclicXor(key, data)).append('\n');
        }

        System.out.print(ans.toString());
        sc.close();
    }
}
```

#include <iostream>
#include <string>
using namespace std;

// 将十六进制字符转换为对应数值
int hexToVal(char c) {
    if (c >= '0' && c <= '9') {
        return c - '0';
    }
    return c - 'A' + 10;
}

// 将数值转换为大写十六进制字符
char valToHex(int x) {
    if (x < 10) {
        return char('0' + x);
    }
    return char('A' + x - 10);
}

// 实现循环异或
string cyclicXor(const string& key, const string& data) {
    string res;
    int m = key.size();

    for (int i = 0; i < (int)data.size(); i++) {
        // key 按照下标循环使用
        int a = hexToVal(data[i]);
        int b = hexToVal(key[i % m]);
        res.push_back(valToHex(a ^ b));
    }

    return res;
}

int main() {
    int t;
    cin >> t;

    while (t--) {
        string key, data;
        cin >> key >> data;

        cout << cyclicXor(key, data) << '\n';
    }

    return 0;
}
