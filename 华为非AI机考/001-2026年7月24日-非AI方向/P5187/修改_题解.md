## 解题思路

一共有 $n$ 件商品（$n\le 20$），每件只能买或不买，用深度优先搜索（$DFS$）枚举全部 $2^n$ 种方案。

搜索时维护：

* $P$：已选商品原价之和
* $S$：已选商品满意度之和
* $cat[c]$：类别 $c$ 买了几件（$c=1\sim 10$）

搜到第 $n$ 件之后结算实付：

* 若有至少 $3$ 个类别买过，则 $cost=P-\lfloor P/200\rfloor\times 30$
* 否则 $cost=P-\lfloor P/200\rfloor\times 20$

两种优惠不能叠加；类别数 $\ge 3$ 时减 $30$ 一定更便宜。若 $cost\le budget$，用 $S$ 更新答案。

买第 $i$ 件时：$cat[类别]\mathrel{+}=1$，递归后再减回去。商品编号只读入，不参与计算。

## 复杂度分析

* 时间复杂度：$O(2^n)$。
* 空间复杂度：$O(n)$。

$n\le 20$，可以通过。

## 代码实现

### Python

```python
def max_satisfaction(budget, items):
    n = len(items)
    ans = 0
    cat = [0] * 11  # cat[c]：类别 c 的购买件数

    def dfs(i, P, S):
        nonlocal ans
        if i == n:
            kinds = sum(1 for c in range(1, 11) if cat[c] > 0)
            cost = P - P // 200 * (30 if kinds >= 3 else 20)
            if cost <= budget:
                ans = max(ans, S)
            return

        # 不买
        dfs(i + 1, P, S)

        # 买
        category, price, score = items[i]
        cat[category] += 1
        dfs(i + 1, P + price, S + score)
        cat[category] -= 1

    dfs(0, 0, 0)
    return ans


def main():
    budget = int(input())
    n = int(input())
    items = []
    for _ in range(n):
        _id, category, price, score = map(int, input().split())
        items.append((category, price, score))
    print(max_satisfaction(budget, items))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    static int ans;
    static int[] cat = new int[11];

    static int kinds() {
        int cnt = 0;
        for (int c = 1; c <= 10; c++) {
            if (cat[c] > 0) {
                cnt++;
            }
        }
        return cnt;
    }

    // i：当前考虑到第 i 件；P：原价和；S：满意度和
    static void dfs(int i, int P, int S, int budget, int[][] items) {
        if (i == items.length) {
            int cost = P - P / 200 * (kinds() >= 3 ? 30 : 20);
            if (cost <= budget) {
                ans = Math.max(ans, S);
            }
            return;
        }

        dfs(i + 1, P, S, budget, items);

        int category = items[i][0];
        int price = items[i][1];
        int score = items[i][2];
        cat[category]++;
        dfs(i + 1, P + price, S + score, budget, items);
        cat[category]--;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int budget = sc.nextInt();
        int n = sc.nextInt();
        int[][] items = new int[n][3];
        for (int i = 0; i < n; i++) {
            sc.nextInt();
            items[i][0] = sc.nextInt();
            items[i][1] = sc.nextInt();
            items[i][2] = sc.nextInt();
        }
        ans = 0;
        dfs(0, 0, 0, budget, items);
        System.out.println(ans);
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

struct Item {
    int category, price, score;
};

int ans;
int cat[11];

int kinds() {
    int cnt = 0;
    for (int c = 1; c <= 10; c++) {
        if (cat[c] > 0) {
            cnt++;
        }
    }
    return cnt;
}

void dfs(int i, int P, int S, int budget, const vector<Item>& items) {
    if (i == (int)items.size()) {
        int cost = P - P / 200 * (kinds() >= 3 ? 30 : 20);
        if (cost <= budget) {
            ans = max(ans, S);
        }
        return;
    }

    dfs(i + 1, P, S, budget, items);

    int c = items[i].category;
    cat[c]++;
    dfs(i + 1, P + items[i].price, S + items[i].score, budget, items);
    cat[c]--;
}

int main() {
    int budget, n;
    cin >> budget >> n;
    vector<Item> items(n);
    for (int i = 0; i < n; i++) {
        int id;
        cin >> id >> items[i].category >> items[i].price >> items[i].score;
    }
    ans = 0;
    dfs(0, 0, 0, budget, items);
    cout << ans << '\n';
    return 0;
}
```
