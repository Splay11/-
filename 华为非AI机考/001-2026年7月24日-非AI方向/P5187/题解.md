## 解题思路

商品数量最多为 $20$，可以使用状态压缩与深度优先搜索（$DFS$），枚举所有 $2^n$ 种商品组合。

搜索过程中维护：

* 商品原价总和 $P$。
* 满意度总和 $S$。
* 商品类别的二进制状态。第 $c$ 类商品对应二进制的第 $c-1$ 位。

对于每一种组合：

1. 若商品类别数量不少于 $3$，使用每满 $200$ 减 $30$：
   $$
   cost=P-\left\lfloor\frac{P}{200}\right\rfloor\times30
   $$
2. 否则使用每满 $200$ 减 $20$：
   $$
   cost=P-\left\lfloor\frac{P}{200}\right\rfloor\times20
   $$
3. 两种优惠不能叠加。满足 $3$ 个类别时，每满 $200$ 减 $30$ 一定优于减 $20$，因此直接使用减 $30$。
4. 若实际支付金额不超过预算，则用当前满意度 $S$ 更新答案。

商品编号仅用于标识，不参与计算。

## 复杂度分析

共有 $2^n$ 种商品组合，每个搜索状态只进行常数次计算。

* 时间复杂度：$O(2^n)$。
* 空间复杂度：$O(n)$，主要为递归栈空间。

由于 $n\leq20$，该复杂度可以通过。

## 代码实现

### Python

```python
def max_satisfaction(budget, items):
    n = len(items)
    answer = 0

    # 预处理所有类别状态中包含的类别数量
    category_count = [0] * (1 << 10)
    for mask in range(1, 1 << 10):
        category_count[mask] = category_count[mask >> 1] + (mask & 1)

    def dfs(index, total_price, total_score, category_mask):
        nonlocal answer

        # 已决定所有商品是否购买
        if index == n:
            if category_count[category_mask] >= 3:
                cost = total_price - total_price // 200 * 30
            else:
                cost = total_price - total_price // 200 * 20

            if cost <= budget:
                answer = max(answer, total_score)
            return

        # 不购买当前商品
        dfs(index + 1, total_price, total_score, category_mask)

        # 购买当前商品
        category, price, score = items[index]
        new_mask = category_mask | (1 << (category - 1))
        dfs(index + 1, total_price + price, total_score + score, new_mask)

    dfs(0, 0, 0, 0)
    return answer


def main():
    budget = int(input())
    n = int(input())

    items = []
    for _ in range(n):
        number, category, price, score = map(int, input().split())
        items.append((category, price, score))

    print(max_satisfaction(budget, items))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    // 枚举每件商品购买或不购买
    public static void dfs(
            int index,
            int totalPrice,
            int totalScore,
            int categoryMask,
            int budget,
            int[][] items,
            int[] answer) {

        // 已决定所有商品是否购买
        if (index == items.length) {
            int discount;

            if (Integer.bitCount(categoryMask) >= 3) {
                discount = 30;
            } else {
                discount = 20;
            }

            int cost = totalPrice - totalPrice / 200 * discount;

            if (cost <= budget) {
                answer[0] = Math.max(answer[0], totalScore);
            }
            return;
        }

        // 不购买当前商品
        dfs(index + 1, totalPrice, totalScore,
                categoryMask, budget, items, answer);

        // 购买当前商品
        int category = items[index][0];
        int price = items[index][1];
        int score = items[index][2];

        int newMask = categoryMask | (1 << (category - 1));

        dfs(index + 1, totalPrice + price, totalScore + score,
                newMask, budget, items, answer);
    }

    public static int maxSatisfaction(int budget, int[][] items) {
        int[] answer = new int[1];
        dfs(0, 0, 0, 0, budget, items, answer);
        return answer[0];
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int budget = scanner.nextInt();
        int n = scanner.nextInt();

        int[][] items = new int[n][3];

        for (int i = 0; i < n; i++) {
            scanner.nextInt();        // 商品编号
            items[i][0] = scanner.nextInt(); // 类别
            items[i][1] = scanner.nextInt(); // 价格
            items[i][2] = scanner.nextInt(); // 满意度
        }

        System.out.println(maxSatisfaction(budget, items));
        scanner.close();
    }
}
```

### c++

```cpp
#include <iostream>
#include <vector>
using namespace std;

struct Item {
    int category;
    int price;
    int score;
};

// 枚举每件商品购买或不购买
void dfs(int index,
         int totalPrice,
         int totalScore,
         int categoryMask,
         int budget,
         const vector<Item>& items,
         int& answer) {

    // 已决定所有商品是否购买
    if (index == (int)items.size()) {
        int discount;

        if (__builtin_popcount(categoryMask) >= 3) {
            discount = 30;
        } else {
            discount = 20;
        }

        int cost = totalPrice - totalPrice / 200 * discount;

        if (cost <= budget) {
            answer = max(answer, totalScore);
        }
        return;
    }

    // 不购买当前商品
    dfs(index + 1, totalPrice, totalScore,
        categoryMask, budget, items, answer);

    // 购买当前商品
    int newMask = categoryMask |
                  (1 << (items[index].category - 1));

    dfs(index + 1,
        totalPrice + items[index].price,
        totalScore + items[index].score,
        newMask,
        budget,
        items,
        answer);
}

int maxSatisfaction(int budget, const vector<Item>& items) {
    int answer = 0;
    dfs(0, 0, 0, 0, budget, items, answer);
    return answer;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int budget, n;
    cin >> budget >> n;

    vector<Item> items(n);

    for (int i = 0; i < n; i++) {
        int number;
        cin >> number
            >> items[i].category
            >> items[i].price
            >> items[i].score;
    }

    cout << maxSatisfaction(budget, items) << '\n';

    return 0;
}
```
