## 解题思路

地砖排成一条直线，编号 $1$ 到 $n$，人一开始在地宫外的 $0$ 号位置。每一回合只能向前跳 $1$、$2$ 或 $3$ 块，落到哪块就拿走那块的财宝。必须在 $m$ 回合内到达第 $n$ 块，否则输出 $-1$。

* 每回合最多走 $3$ 块，若 $n>3m$，一定走不到终点，直接输出 $-1$。
* 财宝价值都不小于 $0$，多用回合不会变差，实际回合数取 $k=\min(n,m)$。

用动态规划。设 $dp[t][i]$ 表示恰好用 $t$ 个回合走到第 $i$ 块地砖时，能拿到的最大财宝。起点：$dp[0][0]=0$。

到达 $i$ 只能从 $i-1$、$i-2$ 或 $i-3$ 跳过来：

$$
dp[t][i]=\max(dp[t-1][i-1],dp[t-1][i-2],dp[t-1][i-3])+value[i]
$$

第 $t$ 回合时，位置 $i$ 还要保证「前面 $t$ 步走得到，后面 $k-t$ 步还能赶到终点」，因此只更新：

$$
\max(t,\,n-3(k-t))\le i\le \min(3t,\,n-(k-t))
$$

转移只依赖上一回合：代码里用一维数组 $dp$ 存上一回合（对应 $dp[t-1]$），用 $new\_dp$ 存本回合（对应 $dp[t]$）。最后答案是走完 $k$ 回合后 $dp[n]$ 的值。

## 复杂度分析

设 $k=\min(n,m)$。每一回合只更新一段有效位置。

* 时间复杂度：$O\left(k\cdot(\min(n-k,3k-n)+1)\right)$。
* 空间复杂度：$O(n)$。

## 代码实现

### Python

```python
def get_max_treasure(n, m, value):
    if n > 3 * m:
        return -1

    k = min(n, m)
    neg = -10 ** 9
    # dp[i]：上一回合走到位置 i 的最大财宝（对应公式里的 dp[t-1][i]）
    dp = [neg] * (n + 1)
    dp[0] = 0
    prev_left, prev_right = 0, 0

    for t in range(1, k + 1):
        left = max(t, n - 3 * (k - t))
        right = min(3 * t, n - (k - t))
        # new_dp[i]：本回合走到位置 i（对应公式里的 dp[t][i]）
        new_dp = [neg] * (n + 1)
        for i in range(left, right + 1):
            best = neg
            for step in (1, 2, 3):
                j = i - step
                if prev_left <= j <= prev_right:
                    best = max(best, dp[j])
            if best != neg:
                new_dp[i] = best + value[i - 1]
        dp = new_dp
        prev_left, prev_right = left, right

    return dp[n]


def main():
    n, m = map(int, input().split())
    value = list(map(int, input().split()))
    print(get_max_treasure(n, m, value))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    public static int getMaxTreasure(int n, int m, int[] value) {
        if (n > 3 * m) {
            return -1;
        }

        int k = Math.min(n, m);
        int neg = -1000000000;
        // dp[i]：上一回合走到位置 i（对应 dp[t-1][i]）
        int[] dp = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            dp[i] = neg;
        }
        dp[0] = 0;
        int prevLeft = 0, prevRight = 0;

        for (int t = 1; t <= k; t++) {
            int left = Math.max(t, n - 3 * (k - t));
            int right = Math.min(3 * t, n - (k - t));
            // newDp[i]：本回合走到位置 i（对应 dp[t][i]）
            int[] newDp = new int[n + 1];
            for (int i = 0; i <= n; i++) {
                newDp[i] = neg;
            }
            for (int i = left; i <= right; i++) {
                int best = neg;
                for (int step = 1; step <= 3; step++) {
                    int j = i - step;
                    if (j >= prevLeft && j <= prevRight) {
                        best = Math.max(best, dp[j]);
                    }
                }
                if (best != neg) {
                    newDp[i] = best + value[i - 1];
                }
            }
            dp = newDp;
            prevLeft = left;
            prevRight = right;
        }
        return dp[n];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] value = new int[n];
        for (int i = 0; i < n; i++) {
            value[i] = sc.nextInt();
        }
        System.out.println(getMaxTreasure(n, m, value));
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int getMaxTreasure(int n, int m, const vector<int>& value) {
    if (n > 3 * m) {
        return -1;
    }

    int k = min(n, m);
    const int neg = -1000000000;
    // dp[i]：上一回合走到位置 i（对应 dp[t-1][i]）
    vector<int> dp(n + 1, neg);
    dp[0] = 0;
    int prevLeft = 0, prevRight = 0;

    for (int t = 1; t <= k; t++) {
        int left = max(t, n - 3 * (k - t));
        int right = min(3 * t, n - (k - t));
        // new_dp[i]：本回合走到位置 i（对应 dp[t][i]）
        vector<int> new_dp(n + 1, neg);
        for (int i = left; i <= right; i++) {
            int best = neg;
            for (int step = 1; step <= 3; step++) {
                int j = i - step;
                if (j >= prevLeft && j <= prevRight) {
                    best = max(best, dp[j]);
                }
            }
            if (best != neg) {
                new_dp[i] = best + value[i - 1];
            }
        }
        dp.swap(new_dp);
        prevLeft = left;
        prevRight = right;
    }
    return dp[n];
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> value(n);
    for (int i = 0; i < n; i++) {
        cin >> value[i];
    }
    cout << getMaxTreasure(n, m, value) << '\n';
    return 0;
}
```
