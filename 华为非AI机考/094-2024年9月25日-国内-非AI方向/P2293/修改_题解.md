# DFS暴力解法

这道题的正解是 $折半搜索$（Meet in the Middle），但是我们用朴素解法 $DFS$ 枚举每个需求选或不选也能在考试时拿到一定的分数。

题意：有 $n$ 个需求，第 $i$ 个需要 $t_i$ 人天，预算为 $T$。每个需求要么全做要么不做，求不超过 $T$ 的最大工作量之和。

朴素做法：对每个需求做或不做，$DFS$ 枚举全部 $2^n$ 种方案，在不超过 $T$ 的方案里取最大和。

$n$ 较小时可过；题面 $n\le 40$，直接 $2^{40}$ 会超时，需要折半。

* 时间复杂度：$O(2^n)$。
* 空间复杂度：$O(n)$。

## DFS解法
#code-switcher
```python
def max_workload(t_list, T):
    n = len(t_list)
    ans = 0

    def dfs(i, cur):
        nonlocal ans
        if cur > T:
            return
        if i == n:
            if cur > ans:
                ans = cur
            return
        # 不做
        dfs(i + 1, cur)
        # 做
        dfs(i + 1, cur + t_list[i])

    dfs(0, 0)
    return ans


def main():
    n, T = map(int, input().split())
    t_list = list(map(int, input().split()))
    print(max_workload(t_list, T))


if __name__ == "__main__":
    main()
```

# 正确解法

## 题目分析

$n\le 40$、$T$ 与 $t_i$ 可达 $10^9$，不能开一维背包。把需求拆成左右两半（每半约 $20$ 个）：

1. 用 $DFS$ 分别枚举左半、右半所有子集和，得到数组 $v1$、$v2$。
2. 将 $v2$ 排序。
3. 枚举左半每个和 $x$（$x\le T$），在 $v2$ 里二分找不超过 $T-x$ 的最大 $y$，用 $x+y$ 更新答案。

复杂度降到约 $O(2^{n/2}\log 2^{n/2})$。

## 复杂度分析

* 时间复杂度：$O(2^{n/2}\cdot n)$（生成子集并排序、二分）。
* 空间复杂度：$O(2^{n/2})$。

## 代码实现

### Python

```python
def gen_sums(arr):
    res = []

    def dfs(i, cur):
        if i == len(arr):
            res.append(cur)
            return
        dfs(i + 1, cur)
        dfs(i + 1, cur + arr[i])

    dfs(0, 0)
    return res


def max_workload(t_list, T):
    n = len(t_list)
    mid = n // 2
    left = t_list[:mid]
    right = t_list[mid:]
    v1 = gen_sums(left)
    v2 = gen_sums(right)
    v2.sort()

    ans = 0
    for x in v1:
        if x > T:
            continue
        # 在 v2 中找 <= T-x 的最大值
        target = T - x
        lo, hi = 0, len(v2) - 1
        best = 0
        while lo <= hi:
            mid_idx = (lo + hi) // 2
            if v2[mid_idx] <= target:
                best = v2[mid_idx]
                lo = mid_idx + 1
            else:
                hi = mid_idx - 1
        if x + best > ans:
            ans = x + best
    return ans


def main():
    n, T = map(int, input().split())
    t_list = list(map(int, input().split()))
    print(max_workload(t_list, T))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {

    static void genSums(long[] arr, int i, long cur, List<Long> res) {
        if (i == arr.length) {
            res.add(cur);
            return;
        }
        genSums(arr, i + 1, cur, res);
        genSums(arr, i + 1, cur + arr[i], res);
    }

    static long maxWorkload(long[] tList, long T) {
        int n = tList.length;
        int mid = n / 2;
        long[] left = new long[mid];
        long[] right = new long[n - mid];
        System.arraycopy(tList, 0, left, 0, mid);
        System.arraycopy(tList, mid, right, 0, n - mid);

        List<Long> v1 = new ArrayList<>();
        List<Long> v2 = new ArrayList<>();
        genSums(left, 0, 0, v1);
        genSums(right, 0, 0, v2);
        Collections.sort(v2);

        long ans = 0;
        for (long x : v1) {
            if (x > T) {
                continue;
            }
            long target = T - x;
            // 二分找 <= target 的最大值
            int lo = 0, hi = v2.size() - 1;
            long best = 0;
            while (lo <= hi) {
                int midIdx = (lo + hi) / 2;
                if (v2.get(midIdx) <= target) {
                    best = v2.get(midIdx);
                    lo = midIdx + 1;
                } else {
                    hi = midIdx - 1;
                }
            }
            ans = Math.max(ans, x + best);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long T = sc.nextLong();
        long[] tList = new long[n];
        for (int i = 0; i < n; i++) {
            tList[i] = sc.nextLong();
        }
        System.out.println(maxWorkload(tList, T));
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

void gen_sums(const vector<long long>& arr, int i, long long cur, vector<long long>& res) {
    if (i == (int)arr.size()) {
        res.push_back(cur);
        return;
    }
    gen_sums(arr, i + 1, cur, res);
    gen_sums(arr, i + 1, cur + arr[i], res);
}

long long max_workload(const vector<long long>& t_list, long long T) {
    int n = (int)t_list.size();
    int mid = n / 2;
    vector<long long> left(t_list.begin(), t_list.begin() + mid);
    vector<long long> right(t_list.begin() + mid, t_list.end());
    vector<long long> v1, v2;
    gen_sums(left, 0, 0, v1);
    gen_sums(right, 0, 0, v2);
    sort(v2.begin(), v2.end());

    long long ans = 0;
    for (long long x : v1) {
        if (x > T) {
            continue;
        }
        long long target = T - x;
        // upper_bound 找第一个 > target，再退一格
        auto it = upper_bound(v2.begin(), v2.end(), target);
        if (it != v2.begin()) {
            --it;
            ans = max(ans, x + *it);
        } else {
            ans = max(ans, x);
        }
    }
    return ans;
}

int main() {
    int n;
    long long T;
    cin >> n >> T;
    vector<long long> t_list(n);
    for (int i = 0; i < n; i++) {
        cin >> t_list[i];
    }
    cout << max_workload(t_list, T) << "\n";
    return 0;
}
```
