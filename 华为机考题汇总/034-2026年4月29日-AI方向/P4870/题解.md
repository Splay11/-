## 解题思路

本题按顺序模拟五个清洗步骤即可。先把每篇文档中的连续空白合并成一个空格，并去掉首尾空格，然后判断长度是否合法；接着扫描规范化后的字符串，把连续的字母数字字符提取成小写单词；之后用哈希集合判断是否精确命中黑名单；再用哈希表统计每个连续三个单词组成的 3-gram，若某个 3-gram 出现次数超过 M，则丢弃；最后把整篇文档的单词序列作为语义指纹，放入哈希集合中去重，只保留第一次出现的文档。最终输出的是规范化后的字符串。

## 复杂度分析

设所有文档总长度为 S，提取出的单词总数为 W。

每篇文档只需要进行常数次线性扫描，3-gram 统计和去重也都是哈希操作，因此总时间复杂度为 O(S + W)，可视为 O(S)。

空间复杂度主要来自黑名单集合、3-gram 计数表和语义去重集合，总空间复杂度为 O(K + W)。

## 代码实现

### Python

```python
import sys


def normalize(text):
    # 合并连续空白，并去掉首尾空白
    return " ".join(text.split())


def extract_words(text):
    words = []
    cur = []

    # 提取由字母数字组成的单词，并转为小写
    for ch in text:
        if ch.isascii() and ch.isalnum():
            cur.append(ch.lower())
        else:
            if cur:
                words.append("".join(cur))
                cur = []

    if cur:
        words.append("".join(cur))

    return words


def check_ngram(words, m):
    cnt = {}

    # 统计连续三个单词组成的 3-gram
    for i in range(len(words) - 2):
        gram = (words[i], words[i + 1], words[i + 2])
        cnt[gram] = cnt.get(gram, 0) + 1
        if cnt[gram] > m:
            return False

    return True


def clean_documents(documents, l, r, m, blacklist):
    ans = []
    seen = set()

    for doc in documents:
        # 规则一：空格规范化和长度检查
        s = normalize(doc)
        if len(s) < l or len(s) > r:
            continue

        # 规则二：提取小写单词
        words = extract_words(s)

        # 规则三：黑名单独立单词匹配
        if any(word in blacklist for word in words):
            continue

        # 规则四：3-gram 复读检查
        if not check_ngram(words, m):
            continue

        # 规则五：按单词序列语义去重
        key = tuple(words)
        if key in seen:
            continue
        seen.add(key)

        ans.append(s)

    return ans


def main():
    n, l, r, m, k = map(int, sys.stdin.readline().split())

    blacklist = set()
    if k > 0:
        blacklist = set(sys.stdin.readline().split())

    documents = []
    for _ in range(n):
        documents.append(sys.stdin.readline().rstrip("\n"))

    ans = clean_documents(documents, l, r, m, blacklist)

    for s in ans:
        print(s)


if __name__ == "__main__":
    main()
```

```java
import java.util.*;

public class Main {

    static String normalize(String text) {
        text = text.trim();
        StringBuilder sb = new StringBuilder();
        boolean inSpace = false;

        // 合并连续空白字符
        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);
            if (Character.isWhitespace(ch)) {
                if (!inSpace) {
                    sb.append(' ');
                    inSpace = true;
                }
            } else {
                sb.append(ch);
                inSpace = false;
            }
        }

        return sb.toString();
    }

    static ArrayList<String> extractWords(String text) {
        ArrayList<String> words = new ArrayList<>();
        StringBuilder cur = new StringBuilder();

        // 按非字母数字字符切分单词
        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);
            if ((ch >= 'a' && ch <= 'z') ||
                (ch >= 'A' && ch <= 'Z') ||
                (ch >= '0' && ch <= '9')) {
                cur.append(Character.toLowerCase(ch));
            } else {
                if (cur.length() > 0) {
                    words.add(cur.toString());
                    cur.setLength(0);
                }
            }
        }

        if (cur.length() > 0) {
            words.add(cur.toString());
        }

        return words;
    }

    static boolean checkNgram(ArrayList<String> words, int m) {
        HashMap<String, Integer> cnt = new HashMap<>();

        // 统计所有 3-gram 出现次数
        for (int i = 0; i + 2 < words.size(); i++) {
            String gram = words.get(i) + "\u0001" + words.get(i + 1) + "\u0001" + words.get(i + 2);
            int times = cnt.getOrDefault(gram, 0) + 1;
            cnt.put(gram, times);

            if (times > m) {
                return false;
            }
        }

        return true;
    }

    static ArrayList<String> cleanDocuments(
            ArrayList<String> documents,
            int l,
            int r,
            int m,
            HashSet<String> blacklist
    ) {
        ArrayList<String> ans = new ArrayList<>();
        HashSet<String> seen = new HashSet<>();

        for (String doc : documents) {
            // 规则一：规范化并检查长度
            String s = normalize(doc);
            if (s.length() < l || s.length() > r) {
                continue;
            }

            // 规则二：提取单词
            ArrayList<String> words = extractWords(s);

            // 规则三：黑名单检查
            boolean blocked = false;
            for (String word : words) {
                if (blacklist.contains(word)) {
                    blocked = true;
                    break;
                }
            }
            if (blocked) {
                continue;
            }

            // 规则四：3-gram 检查
            if (!checkNgram(words, m)) {
                continue;
            }

            // 规则五：语义去重
            StringBuilder keyBuilder = new StringBuilder();
            for (String word : words) {
                keyBuilder.append(word).append('\u0002');
            }
            String key = keyBuilder.toString();

            if (seen.contains(key)) {
                continue;
            }
            seen.add(key);

            ans.add(s);
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int l = sc.nextInt();
        int r = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();

        HashSet<String> blacklist = new HashSet<>();

        // 读取黑名单
        for (int i = 0; i < k; i++) {
            blacklist.add(sc.next().toLowerCase());
        }

        if (sc.hasNextLine()) {
            sc.nextLine();
        }

        ArrayList<String> documents = new ArrayList<>();

        // 每行是一篇完整文档
        for (int i = 0; i < n; i++) {
            documents.add(sc.hasNextLine() ? sc.nextLine() : "");
        }

        ArrayList<String> ans = cleanDocuments(documents, l, r, m, blacklist);

        for (String s : ans) {
            System.out.println(s);
        }

        sc.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

string normalize(const string& text) {
    string res;
    bool inSpace = true;

    // 合并连续空白，并去掉首尾空白
    for (char ch : text) {
        if (isspace((unsigned char)ch)) {
            if (!inSpace) {
                res.push_back(' ');
                inSpace = true;
            }
        } else {
            res.push_back(ch);
            inSpace = false;
        }
    }

    if (!res.empty() && res.back() == ' ') {
        res.pop_back();
    }

    return res;
}

vector<string> extractWords(const string& text) {
    vector<string> words;
    string cur;

    // 提取字母数字组成的单词，并转小写
    for (char ch : text) {
        if ((ch >= 'a' && ch <= 'z') ||
            (ch >= 'A' && ch <= 'Z') ||
            (ch >= '0' && ch <= '9')) {
            cur.push_back((char)tolower((unsigned char)ch));
        } else {
            if (!cur.empty()) {
                words.push_back(cur);
                cur.clear();
            }
        }
    }

    if (!cur.empty()) {
        words.push_back(cur);
    }

    return words;
}

bool checkNgram(const vector<string>& words, int m) {
    unordered_map<string, int> cnt;

    // 统计所有连续三个单词组成的 3-gram
    for (int i = 0; i + 2 < (int)words.size(); i++) {
        string gram = words[i] + '\1' + words[i + 1] + '\1' + words[i + 2];
        cnt[gram]++;

        if (cnt[gram] > m) {
            return false;
        }
    }

    return true;
}

vector<string> cleanDocuments(
    const vector<string>& documents,
    int l,
    int r,
    int m,
    const unordered_set<string>& blacklist
) {
    vector<string> ans;
    unordered_set<string> seen;

    for (const string& doc : documents) {
        // 规则一：规范化并检查长度
        string s = normalize(doc);
        if ((int)s.size() < l || (int)s.size() > r) {
            continue;
        }

        // 规则二：提取单词
        vector<string> words = extractWords(s);

        // 规则三：黑名单检查
        bool blocked = false;
        for (const string& word : words) {
            if (blacklist.count(word)) {
                blocked = true;
                break;
            }
        }
        if (blocked) {
            continue;
        }

        // 规则四：3-gram 检查
        if (!checkNgram(words, m)) {
            continue;
        }

        // 规则五：语义去重
        string key;
        for (const string& word : words) {
            key += word;
            key += '\2';
        }

        if (seen.count(key)) {
            continue;
        }
        seen.insert(key);

        ans.push_back(s);
    }

    return ans;
}

int main() {
    int n, l, r, m, k;
    cin >> n >> l >> r >> m >> k;

    unordered_set<string> blacklist;

    // 读取黑名单
    for (int i = 0; i < k; i++) {
        string word;
        cin >> word;
        blacklist.insert(word);
    }

    string line;
    getline(cin, line);

    vector<string> documents;

    // 每行读取一篇文档
    for (int i = 0; i < n; i++) {
        getline(cin, line);
        documents.push_back(line);
    }

    vector<string> ans = cleanDocuments(documents, l, r, m, blacklist);

    for (const string& s : ans) {
        cout << s << '\n';
    }

    return 0;
}
```
