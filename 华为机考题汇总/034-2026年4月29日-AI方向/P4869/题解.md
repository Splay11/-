## 解题思路

本题就是模拟 LSTM 的前向传播过程。先读取长度为 $T$ 的输入序列，每个时间步包含 $B$ 个样本，每个样本有 $D=5$ 个特征。然后读取输入门、遗忘门、输出门、候选细胞状态四组参数。

初始时隐藏状态和细胞状态都为 0。对于每个时间步，依次计算输入门、遗忘门、输出门和候选细胞状态，再用它们更新当前细胞状态和隐藏状态。每一步的隐藏状态都需要保存，最后输出所有时间步的隐藏状态以及最终细胞状态。

关键算法是 LSTM 前向传播模拟，按时间顺序递推即可。

## 复杂度分析

每个时间步需要对 $B$ 个样本计算 4 个门，每个门包含输入权重和隐藏权重的矩阵运算。

时间复杂度为：

$$
O(T \times B \times (D \times H + H^2))
$$

需要保存所有隐藏状态，因此空间复杂度为：

$$
O(T \times B \times H)
$$

## 代码实现

### Python

```python
import sys
import numpy as np


def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))


def read_gate(data, idx, D, H):
    wx = np.array(data[idx:idx + D * H]).reshape(D, H)
    idx += D * H

    wh = np.array(data[idx:idx + H * H]).reshape(H, H)
    idx += H * H

    b = np.array(data[idx:idx + H])
    idx += H

    return (wx, wh, b), idx


def lstm_forward(x, params, T, B, D, H):
    # 参数顺序：输入门、遗忘门、输出门、候选细胞状态
    wx_i, wh_i, b_i = params[0]
    wx_f, wh_f, b_f = params[1]
    wx_o, wh_o, b_o = params[2]
    wx_c, wh_c, b_c = params[3]

    h = np.zeros((B, H))
    c = np.zeros((B, H))
    all_h = []

    for t in range(T):
        xt = x[t]

        # 分别计算四个门
        i = sigmoid(xt @ wx_i + h @ wh_i + b_i)
        f = sigmoid(xt @ wx_f + h @ wh_f + b_f)
        o = sigmoid(xt @ wx_o + h @ wh_o + b_o)
        g = np.tanh(xt @ wx_c + h @ wh_c + b_c)

        # 更新细胞状态和隐藏状态
        c = f * c + i * g
        h = o * np.tanh(c)

        all_h.append(h.copy())

    return np.array(all_h), c


def main():
    data = list(map(float, sys.stdin.read().split()))
    idx = 0

    T, B, D, H = map(int, data[idx:idx + 4])
    idx += 4

    x = np.array(data[idx:idx + T * B * D]).reshape(T, B, D)
    idx += T * B * D

    params = []
    for _ in range(4):
        gate, idx = read_gate(data, idx, D, H)
        params.append(gate)

    hidden, cell = lstm_forward(x, params, T, B, D, H)

    print(" ".join(f"{v:.4f}" for v in hidden.reshape(-1)))
    print(" ".join(f"{v:.4f}" for v in cell.reshape(-1)))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static class Gate {
        double[][] wx, wh;
        double[] b;

        Gate(int D, int H, Scanner sc) {
            wx = new double[D][H];
            wh = new double[H][H];
            b = new double[H];

            // 读取输入权重
            for (int i = 0; i < D; i++) {
                for (int j = 0; j < H; j++) {
                    wx[i][j] = sc.nextDouble();
                }
            }

            // 读取隐藏权重
            for (int i = 0; i < H; i++) {
                for (int j = 0; j < H; j++) {
                    wh[i][j] = sc.nextDouble();
                }
            }

            // 读取偏置
            for (int i = 0; i < H; i++) {
                b[i] = sc.nextDouble();
            }
        }
    }

    static double sigmoid(double x) {
        return 1.0 / (1.0 + Math.exp(-x));
    }

    static double[][] calc(double[][] x, double[][] h, Gate gate,
                           int B, int D, int H, boolean useTanh) {
        double[][] res = new double[B][H];

        for (int b = 0; b < B; b++) {
            for (int j = 0; j < H; j++) {
                double sum = gate.b[j];

                // 当前输入部分
                for (int d = 0; d < D; d++) {
                    sum += x[b][d] * gate.wx[d][j];
                }

                // 上一隐藏状态部分
                for (int k = 0; k < H; k++) {
                    sum += h[b][k] * gate.wh[k][j];
                }

                res[b][j] = useTanh ? Math.tanh(sum) : sigmoid(sum);
            }
        }

        return res;
    }

    static double[][][] ansHidden;
    static double[][] ansCell;

    static void lstmForward(double[][][] x, Gate[] gates,
                            int T, int B, int D, int H) {
        double[][] h = new double[B][H];
        double[][] c = new double[B][H];
        ansHidden = new double[T][B][H];

        for (int t = 0; t < T; t++) {
            // 依次计算输入门、遗忘门、输出门、候选细胞状态
            double[][] iGate = calc(x[t], h, gates[0], B, D, H, false);
            double[][] fGate = calc(x[t], h, gates[1], B, D, H, false);
            double[][] oGate = calc(x[t], h, gates[2], B, D, H, false);
            double[][] gGate = calc(x[t], h, gates[3], B, D, H, true);

            double[][] newH = new double[B][H];
            double[][] newC = new double[B][H];

            for (int b = 0; b < B; b++) {
                for (int j = 0; j < H; j++) {
                    // 更新状态
                    newC[b][j] = fGate[b][j] * c[b][j] + iGate[b][j] * gGate[b][j];
                    newH[b][j] = oGate[b][j] * Math.tanh(newC[b][j]);
                    ansHidden[t][b][j] = newH[b][j];
                }
            }

            h = newH;
            c = newC;
        }

        ansCell = c;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();
        int B = sc.nextInt();
        int D = sc.nextInt();
        int H = sc.nextInt();

        double[][][] x = new double[T][B][D];

        // 读取输入序列
        for (int t = 0; t < T; t++) {
            for (int b = 0; b < B; b++) {
                for (int d = 0; d < D; d++) {
                    x[t][b][d] = sc.nextDouble();
                }
            }
        }

        Gate[] gates = new Gate[4];

        // 顺序：输入门、遗忘门、输出门、候选细胞状态
        for (int i = 0; i < 4; i++) {
            gates[i] = new Gate(D, H, sc);
        }

        lstmForward(x, gates, T, B, D, H);

        StringBuilder sb = new StringBuilder();

        boolean first = true;
        for (int t = 0; t < T; t++) {
            for (int b = 0; b < B; b++) {
                for (int j = 0; j < H; j++) {
                    if (!first) sb.append(" ");
                    sb.append(String.format("%.4f", ansHidden[t][b][j]));
                    first = false;
                }
            }
        }

        sb.append("\n");
        first = true;

        for (int b = 0; b < B; b++) {
            for (int j = 0; j < H; j++) {
                if (!first) sb.append(" ");
                sb.append(String.format("%.4f", ansCell[b][j]));
                first = false;
            }
        }

        System.out.print(sb);
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Gate {
    vector<vector<double>> wx, wh;
    vector<double> b;
};

double sigmoid(double x) {
    return 1.0 / (1.0 + exp(-x));
}

Gate readGate(int D, int H) {
    Gate g;
    g.wx.assign(D, vector<double>(H));
    g.wh.assign(H, vector<double>(H));
    g.b.assign(H, 0);

    // 读取输入权重
    for (int i = 0; i < D; i++) {
        for (int j = 0; j < H; j++) {
            cin >> g.wx[i][j];
        }
    }

    // 读取隐藏权重
    for (int i = 0; i < H; i++) {
        for (int j = 0; j < H; j++) {
            cin >> g.wh[i][j];
        }
    }

    // 读取偏置
    for (int i = 0; i < H; i++) {
        cin >> g.b[i];
    }

    return g;
}

vector<vector<double>> calc(vector<vector<double>>& x,
                            vector<vector<double>>& h,
                            Gate& gate,
                            int B, int D, int H,
                            bool useTanh) {
    vector<vector<double>> res(B, vector<double>(H));

    for (int b = 0; b < B; b++) {
        for (int j = 0; j < H; j++) {
            double sum = gate.b[j];

            // 当前输入部分
            for (int d = 0; d < D; d++) {
                sum += x[b][d] * gate.wx[d][j];
            }

            // 上一隐藏状态部分
            for (int k = 0; k < H; k++) {
                sum += h[b][k] * gate.wh[k][j];
            }

            res[b][j] = useTanh ? tanh(sum) : sigmoid(sum);
        }
    }

    return res;
}

int main() {
    int T, B, D, H;
    cin >> T >> B >> D >> H;

    vector<vector<vector<double>>> x(T, vector<vector<double>>(B, vector<double>(D)));

    // 读取输入序列
    for (int t = 0; t < T; t++) {
        for (int b = 0; b < B; b++) {
            for (int d = 0; d < D; d++) {
                cin >> x[t][b][d];
            }
        }
    }

    vector<Gate> gates(4);

    // 顺序：输入门、遗忘门、输出门、候选细胞状态
    for (int i = 0; i < 4; i++) {
        gates[i] = readGate(D, H);
    }

    vector<vector<double>> h(B, vector<double>(H, 0));
    vector<vector<double>> c(B, vector<double>(H, 0));
    vector<vector<vector<double>>> allH(T, vector<vector<double>>(B, vector<double>(H)));

    for (int t = 0; t < T; t++) {
        // 依次计算输入门、遗忘门、输出门、候选细胞状态
        auto iGate = calc(x[t], h, gates[0], B, D, H, false);
        auto fGate = calc(x[t], h, gates[1], B, D, H, false);
        auto oGate = calc(x[t], h, gates[2], B, D, H, false);
        auto gGate = calc(x[t], h, gates[3], B, D, H, true);

        vector<vector<double>> newH(B, vector<double>(H));
        vector<vector<double>> newC(B, vector<double>(H));

        for (int b = 0; b < B; b++) {
            for (int j = 0; j < H; j++) {
                // 更新状态
                newC[b][j] = fGate[b][j] * c[b][j] + iGate[b][j] * gGate[b][j];
                newH[b][j] = oGate[b][j] * tanh(newC[b][j]);
                allH[t][b][j] = newH[b][j];
            }
        }

        h = newH;
        c = newC;
    }

    cout << fixed << setprecision(4);

    bool first = true;

    for (int t = 0; t < T; t++) {
        for (int b = 0; b < B; b++) {
            for (int j = 0; j < H; j++) {
                if (!first) cout << " ";
                cout << allH[t][b][j];
                first = false;
            }
        }
    }

    cout << "\n";
    first = true;

    for (int b = 0; b < B; b++) {
        for (int j = 0; j < H; j++) {
            if (!first) cout << " ";
            cout << c[b][j];
            first = false;
        }
    }

    return 0;
}
```
