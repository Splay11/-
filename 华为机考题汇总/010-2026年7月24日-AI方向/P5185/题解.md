## 解题思路

核心思路：

设前缀和 $pre[i]$ 表示前 $i$ 层的计算时间之和，则层 $k+1$ 到层 $i$ 组成一个阶段时，该阶段的计算时间为：

$$
pre[i]-pre[k]
$$

定义 $dp_s[i]$ 表示将前 $i$ 层恰好划分为 $s$ 个非空阶段时，所需的最小通信开销。

假设最后一个阶段包含第 $k+1$ 层到第 $i$ 层，那么前 $k$ 层需要被划分为 $s-1$ 个阶段，并且需要在第 $k$ 层后进行一次阶段划分，产生通信开销 $comm[k]$。

状态转移为：

$$
dp_s[i]=
\min\left(dp_{s-1}[k]+comm[k]\right)
$$

其中 $k$ 需要满足：

$$
s-1\le k<i
$$

以及最后一个阶段的计算时间不超过 $T$：

$$
pre[i]-pre[k]\le T
$$

对于固定的 $i$，由于每层计算时间均为正数，满足 $pre[i]-pre[k]\le T$ 的 $k$ 构成一个连续区间。设其中最小的 $k$ 为 $left[i]$，则合法转移范围为：

$$
\max(s-1,left[i])\le k\le i-1
$$

在固定阶段数 $s$ 时，候选值：

$$
dp_{s-1}[k]+comm[k]
$$

只与 $k$ 有关。随着 $i$ 增大，合法的 $k$ 区间左右端点均单调右移，因此可以使用单调队列维护滑动区间内的最小值，将每一层状态转移优化为均摊 $O(1)$。

实现方法：

1. 计算计算时间数组的前缀和。
2. 使用双指针计算每个 $left[i]$，表示最后一个阶段以第 $i$ 层结尾时，允许的最小前置层数。
3. 初始化一个阶段的情况：
   $$
   dp_1[i]=
   \begin{cases}
   0,&pre[i]\le T\
   +\infty,&pre[i]>T
   \end{cases}
   $$
4. 从 $2$ 到 $p$ 枚举阶段数量，使用单调队列维护合法切分点对应的最小通信开销。
5. 最终若 $dp_p[n]$ 不可达，则输出 $-1$；否则输出 $dp_p[n]$。

## 复杂度分析

双指针计算所有 $left[i]$ 的时间复杂度为 $O(n)$。

动态规划共有 $p$ 轮，每轮中每个切分点最多进入和离开单调队列一次，因此每轮时间复杂度为 $O(n)$。

总时间复杂度为：

$$
O(pn)
$$

动态规划使用滚动数组，前缀和、合法左端点和单调队列均需要 $O(n)$ 空间，因此空间复杂度为：

$$
O(n)
$$

## 代码实现

### Python

```python
import sys
from collections import deque


def min_comm(n, p, limit, times, comm):
    inf = 10 ** 18

    # 计算前缀和，用于快速计算任意连续阶段的总计算时间
    pre = [0] * (n + 1)
    for i in range(1, n + 1):
        pre[i] = pre[i - 1] + times[i]

    # left[i] 表示满足 pre[i] - pre[k] <= limit 的最小 k
    # 由于所有计算时间均为正数，k 只会单调右移
    left = [0] * (n + 1)
    k = 0
    for i in range(1, n + 1):
        while k < i and pre[i] - pre[k] > limit:
            k += 1
        left[i] = k

    # dp[i] 表示将前 i 层恰好划分为当前阶段数时的最小通信开销
    # 一个阶段时不能产生阶段间通信开销
    dp = [inf] * (n + 1)
    for i in range(1, n + 1):
        if pre[i] <= limit:
            dp[i] = 0

    if p == 1:
        return -1 if dp[n] == inf else dp[n]

    # 依次计算划分为 2 到 p 个阶段的结果
    for s in range(2, p + 1):
        ndp = [inf] * (n + 1)

        # 队列中保存二元组：
        # 切分位置 k，以及 dp[k] + comm[k]
        q = deque()

        # 至少需要 s 层才能划分为 s 个非空阶段
        for i in range(s, n + 1):
            # 当最后一个阶段以 i 结尾时，新增的最大切分位置为 i - 1
            cut = i - 1

            # 只有前 cut 层能够被划分为 s - 1 个阶段时才可转移
            if dp[cut] < inf:
                value = dp[cut] + comm[cut]

                # 保持队列中的候选值单调递增
                # 相同值保留位置更靠后的切分点，使其更晚失效
                while q and q[-1][1] >= value:
                    q.pop()
                q.append((cut, value))

            # 最后一个阶段需要满足计算时间不超过 limit
            low = max(s - 1, left[i])

            # 删除已经不在合法切分范围内的位置
            while q and q[0][0] < low:
                q.popleft()

            # 队首即当前合法范围内的最小通信开销
            if q:
                ndp[i] = q[0][1]

        dp = ndp

    return -1 if dp[n] == inf else dp[n]


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, p, limit = data[0], data[1], data[2]

    times = [0] * (n + 1)
    pos = 3
    for i in range(1, n + 1):
        times[i] = data[pos]
        pos += 1

    comm = [0] * n
    for i in range(1, n):
        comm[i] = data[pos]
        pos += 1

    print(min_comm(n, p, limit, times, comm))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static long minComm(int n, int p, int limit, int[] time, int[] comm) {
        long inf = 1L << 60;

        // 计算前缀和，用于快速计算任意连续阶段的总计算时间
        long[] pre = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            pre[i] = pre[i - 1] + time[i];
        }

        // left[i] 表示满足 pre[i] - pre[k] <= limit 的最小 k
        // 因为所有计算时间均为正数，指针 k 只会向右移动
        int[] left = new int[n + 1];
        int k = 0;
        for (int i = 1; i <= n; i++) {
            while (k < i && pre[i] - pre[k] > limit) {
                k++;
            }
            left[i] = k;
        }

        // dp[i] 表示将前 i 层恰好划分为当前阶段数时的最小通信开销
        // 只有一个阶段时，不会产生阶段间通信开销
        long[] dp = new long[n + 1];
        Arrays.fill(dp, inf);

        for (int i = 1; i <= n; i++) {
            if (pre[i] <= limit) {
                dp[i] = 0;
            }
        }

        if (p == 1) {
            return dp[n] == inf ? -1 : dp[n];
        }

        // 依次计算划分为 2 到 p 个阶段的结果
        for (int s = 2; s <= p; s++) {
            long[] ndp = new long[n + 1];
            Arrays.fill(ndp, inf);

            // 使用数组模拟单调队列，队列中保存切分位置
            int[] queue = new int[n + 1];
            int head = 0;
            int tail = 0;

            // 至少需要 s 层才能划分为 s 个非空阶段
            for (int i = s; i <= n; i++) {
                // 新增的最大切分位置为 i - 1
                int cut = i - 1;

                // 只有前 cut 层可以划分为 s - 1 个阶段时才能加入队列
                if (dp[cut] < inf) {
                    long value = dp[cut] + comm[cut];

                    // 按 dp[k] + comm[k] 单调递增维护队列
                    // 相同值保留位置更靠后的切分点
                    while (head < tail) {
                        int last = queue[tail - 1];
                        long lastValue = dp[last] + comm[last];

                        if (lastValue < value) {
                            break;
                        }
                        tail--;
                    }

                    queue[tail++] = cut;
                }

                // 合法切分位置的左边界
                int low = Math.max(s - 1, left[i]);

                // 删除已经离开合法范围的切分位置
                while (head < tail && queue[head] < low) {
                    head++;
                }

                // 队首对应当前合法范围内的最小通信开销
                if (head < tail) {
                    int best = queue[head];
                    ndp[i] = dp[best] + comm[best];
                }
            }

            dp = ndp;
        }

        return dp[n] == inf ? -1 : dp[n];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int p = sc.nextInt();
        int limit = sc.nextInt();

        int[] time = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            time[i] = sc.nextInt();
        }

        int[] comm = new int[n];
        for (int i = 1; i < n; i++) {
            comm[i] = sc.nextInt();
        }

        System.out.println(minComm(n, p, limit, time, comm));
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <deque>
#include <algorithm>
using namespace std;

long long minComm(int n, int p, int limit,
                  const vector<int>& time,
                  const vector<int>& comm) {
    const long long INF = 1LL << 60;

    // 计算前缀和，用于快速计算任意连续阶段的总计算时间
    vector<long long> pre(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        pre[i] = pre[i - 1] + time[i];
    }

    // left[i] 表示满足 pre[i] - pre[k] <= limit 的最小 k
    // 因为所有计算时间均为正数，指针 k 只会向右移动
    vector<int> left(n + 1, 0);
    int k = 0;

    for (int i = 1; i <= n; i++) {
        while (k < i && pre[i] - pre[k] > limit) {
            k++;
        }
        left[i] = k;
    }

    // dp[i] 表示将前 i 层恰好划分为当前阶段数时的最小通信开销
    // 一个阶段时不会产生阶段间通信开销
    vector<long long> dp(n + 1, INF);

    for (int i = 1; i <= n; i++) {
        if (pre[i] <= limit) {
            dp[i] = 0;
        }
    }

    if (p == 1) {
        return dp[n] == INF ? -1 : dp[n];
    }

    // 依次计算划分为 2 到 p 个阶段的结果
    for (int s = 2; s <= p; s++) {
        vector<long long> ndp(n + 1, INF);

        // 单调队列中保存候选切分位置
        deque<int> q;

        // 至少需要 s 层才能划分为 s 个非空阶段
        for (int i = s; i <= n; i++) {
            // 新增的最大切分位置为 i - 1
            int cut = i - 1;

            // 只有前 cut 层能够划分为 s - 1 个阶段时才可转移
            if (dp[cut] < INF) {
                long long value = dp[cut] + comm[cut];

                // 按 dp[k] + comm[k] 单调递增维护队列
                // 相同值保留位置更靠后的切分点
                while (!q.empty()) {
                    int last = q.back();
                    long long lastValue = dp[last] + comm[last];

                    if (lastValue < value) {
                        break;
                    }
                    q.pop_back();
                }

                q.push_back(cut);
            }

            // 合法切分位置的左边界
            int low = max(s - 1, left[i]);

            // 删除已经离开合法范围的切分位置
            while (!q.empty() && q.front() < low) {
                q.pop_front();
            }

            // 队首对应当前合法范围内的最小通信开销
            if (!q.empty()) {
                int best = q.front();
                ndp[i] = dp[best] + comm[best];
            }
        }

        dp.swap(ndp);
    }

    return dp[n] == INF ? -1 : dp[n];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, p, limit;
    cin >> n >> p >> limit;

    vector<int> time(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> time[i];
    }

    vector<int> comm(n, 0);
    for (int i = 1; i < n; i++) {
        cin >> comm[i];
    }

    cout << minComm(n, p, limit, time, comm) << '\n';

    return 0;
}
```
