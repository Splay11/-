## 解题思路

核心思路：

按照 Token 的输入顺序逐行处理打分矩阵。对于每个 Token，将所有专家按照以下规则排序：

1. 得分更高的专家排在前面。
2. 得分相同时，索引更小的专家排在前面。

排序后的前 $K$ 个专家就是该 Token 的 Top-K 路由结果。

随后依次检查这 $K$ 个专家：

* 如果专家 $j$ 当前的负载满足 $\operatorname{load}[j]<C$，则本次路由生效，将 $\operatorname{load}[j]$ 加一。
* 否则该路由被容量掩码丢弃，负载保持不变。

一个 Token 的 Top-K 结果中不会重复出现同一个专家。每个专家的容量判断只会修改自己的负载，不会影响该 Token 对其他专家的判断，因此可以直接遍历这 $K$ 个专家并更新负载。

不需要显式构造 $M^{(1)}$ 和 $M^{(2)}$ 两个 $N\times E$ 的矩阵。对于每个 Token 找到 Top-K 专家后，立即完成容量判断即可。

所有 Token 处理完成后，根据最终负载计算：

$$
P=\sum_{j=0}^{E-1}\operatorname{load}[j]^2
$$

实现方法：

* Python 和 C++ 直接对专家索引排序，排序关键字为 $(-S_{i,j},j)$。
* Java 使用一个非负整数编码排序关键字：

$$
\operatorname{key}*j=(10000-S*{i,j})(E+1)+j
$$

对编码值升序排序后，得分更高、索引更小的专家会排在前面，并且可以通过对 $E+1$ 取模还原专家索引。

## 复杂度分析

对于每个 Token，需要对 $E$ 个专家排序，时间复杂度为 $O(E\log E)$，随后检查前 $K$ 个专家，时间复杂度为 $O(K)$。

总时间复杂度为：

$$
O\left(N(E\log E+K)\right)=O(NE\log E)
$$

因为 $E\le 100$、$N\le 10^4$，该复杂度可以满足要求。

代码中保存了完整的打分矩阵，空间复杂度为：

$$
O(NE)
$$

排序数组和负载数组使用的额外空间为 $O(E)$。

## 代码实现

### Python

```python
import sys


def route(scores, e, k, c):
    # load[j] 表示专家 j 当前已经接收的 Token 数量
    load = [0] * e

    for row in scores:
        # 按照得分降序排列，得分相同时按照专家索引升序排列
        order = sorted(range(e), key=lambda j: (-row[j], j))

        # 只检查当前 Token 得分排名前 K 的专家
        for t in range(k):
            j = order[t]

            # 专家未达到容量上限时，本次路由生效
            if load[j] < c:
                load[j] += 1

    # 计算所有专家实际负载的平方和
    penalty = 0
    for x in load:
        penalty += x * x

    return penalty, load


def main():
    n, e, k, c = map(int, sys.stdin.buffer.readline().split())

    # 读取 N 行专家打分
    scores = []
    for _ in range(n):
        row = list(map(int, sys.stdin.buffer.readline().split()))
        scores.append(row)

    penalty, load = route(scores, e, k, c)

    print(penalty)
    print(*load)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {

    static long route(int[][] scores, int e, int k, int c, int[] load) {
        int n = scores.length;
        int base = e + 1;
        long[] order = new long[e];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < e; j++) {
                /*
                 * 得分越高，10000 - scores[i][j] 越小。
                 * 得分相同时，专家索引越小，编码值越小。
                 */
                order[j] = (long) (10000 - scores[i][j]) * base + j;
            }

            // 升序排序后，前 K 项就是当前 Token 应选择的专家
            Arrays.sort(order);

            for (int t = 0; t < k; t++) {
                // 通过取模还原专家索引
                int j = (int) (order[t] % base);

                // 专家未达到容量上限时，本次路由生效
                if (load[j] < c) {
                    load[j]++;
                }
            }
        }

        // 计算所有专家实际负载的平方和
        long penalty = 0;
        for (int j = 0; j < e; j++) {
            penalty += (long) load[j] * load[j];
        }

        return penalty;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br =
                new BufferedReader(new InputStreamReader(System.in));

        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int e = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        int c = Integer.parseInt(st.nextToken());

        int[][] scores = new int[n][e];

        // 读取 N 行专家打分
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < e; j++) {
                scores[i][j] = Integer.parseInt(st.nextToken());
            }
        }

        int[] load = new int[e];
        long penalty = route(scores, e, k, c, load);

        StringBuilder ans = new StringBuilder();
        ans.append(penalty).append('\n');

        for (int j = 0; j < e; j++) {
            if (j > 0) {
                ans.append(' ');
            }
            ans.append(load[j]);
        }

        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <utility>
using namespace std;

long long route(const vector<vector<int>>& scores,
                int e, int k, int c, vector<int>& load) {
    int n = scores.size();

    // order 中保存 {-得分, 专家索引}
    vector<pair<int, int>> order(e);

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < e; j++) {
            /*
             * pair 默认按照第一项、第二项依次升序排序。
             * 使用负得分可以实现得分降序；
             * 得分相同时，专家索引较小的排在前面。
             */
            order[j] = {-scores[i][j], j};
        }

        // 排序后的前 K 项就是当前 Token 选择的专家
        sort(order.begin(), order.end());

        for (int t = 0; t < k; t++) {
            int j = order[t].second;

            // 专家未达到容量上限时，本次路由生效
            if (load[j] < c) {
                load[j]++;
            }
        }
    }

    // 计算所有专家实际负载的平方和
    long long penalty = 0;
    for (int j = 0; j < e; j++) {
        penalty += 1LL * load[j] * load[j];
    }

    return penalty;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, e, k, c;
    cin >> n >> e >> k >> c;

    vector<vector<int>> scores(n, vector<int>(e));

    // 读取 N 行专家打分
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < e; j++) {
            cin >> scores[i][j];
        }
    }

    vector<int> load(e, 0);
    long long penalty = route(scores, e, k, c, load);

    cout << penalty << '\n';

    for (int j = 0; j < e; j++) {
        if (j > 0) {
            cout << ' ';
        }
        cout << load[j];
    }
    cout << '\n';

    return 0;
}
```
