## 解题思路

随机森林预测可以拆成两层：

1. 单棵决策树预测
   从根节点 $1$ 开始：

   * 如果当前节点是叶子节点，直接返回该叶子的 $label$。
   * 如果当前节点是内部节点，读取交易的第 $idx$ 个特征值：

     * 若 $feature[idx] \le val$，进入左子节点；
     * 否则进入右子节点。

2. 多棵树多数表决
   对同一笔交易，依次让所有树独立预测，统计输出 $1$ 的票数 `cnt1`。

   因为平票时输出 $0$，所以只有当：

   $$cnt1 \times 2 > T$$

   时，最终结果才输出 $1$，否则输出 $0$。

实现时需要注意：

* 输入中的节点编号从 $1$ 开始，代码中可以继续使用 $1$ 下标存储节点，避免频繁转换。
* 特征索引 $idx$ 也从 $1$ 开始，代码中可以把特征数组开成长度 $M+1$，让特征也按 $1$ 下标访问。
* 每棵树的节点数最多 $500$，树的数量最多 $51$，交易数最多 $2000$，直接逐笔交易、逐棵树遍历即可。
* 虽然题目背景是机器学习，但本题核心是决策树分支遍历，不是矩阵运算，因此直接用数组和循环实现更合适。

## 复杂度分析

设第 $i$ 棵树的节点数为 $K_i$，树高为 $H_i$。

对于一笔交易，遍历第 $i$ 棵树最多访问 $H_i$ 个节点，因此一笔交易的预测复杂度为：

$$O\left(\sum_{i=1}^{T} H_i\right)$$

因为 $H_i \le K_i$，所以最坏情况下为：

$$O\left(\sum_{i=1}^{T} K_i\right)$$

共有 $N$ 笔交易，因此总时间复杂度为：

$$O\left(N \times \sum_{i=1}^{T} K_i\right)$$

在数据范围内，最多约为：

$$2000 \times 51 \times 500$$

可以接受。

空间复杂度主要来自存储所有树节点：

$$O\left(\sum_{i=1}^{T} K_i + M\right)$$

## 代码实现

### Python

```python
import sys


def predict_tree(tree, feat):
    # 从根节点 1 开始向下遍历
    cur = 1

    while True:
        node = tree[cur]

        # type 为 0，说明当前节点是叶子节点，直接返回分类结果
        if node[0] == 0:
            return node[1]

        # type 为 1，说明当前节点是内部判断节点
        _, idx, val, left, right = node

        # 根据第 idx 个特征值与阈值 val 的关系选择子节点
        if feat[idx] <= val:
            cur = left
        else:
            cur = right


def predict_forest(trees, feat):
    cnt1 = 0
    total = len(trees)

    # 依次使用每棵树对当前交易进行预测
    for tree in trees:
        if predict_tree(tree, feat) == 1:
            cnt1 += 1

    # 只有 1 的票数严格超过一半才输出 1，平票时输出 0
    if cnt1 * 2 > total:
        return 1
    return 0


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    p = 0

    T = data[p]
    M = data[p + 1]
    N = data[p + 2]
    p += 3

    trees = []

    # 读取 T 棵决策树
    for _ in range(T):
        K = data[p]
        p += 1

        # 节点编号从 1 到 K，因此多开一个位置
        tree = [None] * (K + 1)

        for i in range(1, K + 1):
            typ = data[p]
            p += 1

            if typ == 1:
                idx = data[p]
                val = data[p + 1]
                left = data[p + 2]
                right = data[p + 3]
                p += 4

                # 内部节点保存判断特征、阈值和左右子节点
                tree[i] = (1, idx, val, left, right)
            else:
                label = data[p]
                p += 1

                # 叶子节点保存分类结果
                tree[i] = (0, label)

        trees.append(tree)

    ans = []

    # 逐笔读取交易并预测
    for _ in range(N):
        feat = [0] * (M + 1)

        # 特征也按照 1 下标存储，方便直接使用题目给出的 idx
        for j in range(1, M + 1):
            feat[j] = data[p]
            p += 1

        ans.append(str(predict_forest(trees, feat)))

    sys.stdout.write("\n".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static class Node {
        int type;
        int idx;
        int val;
        int left;
        int right;
        int label;
    }

    static int predictTree(Node[] tree, int[] feat) {
        // 从根节点 1 开始遍历
        int cur = 1;

        while (true) {
            Node node = tree[cur];

            // 叶子节点直接返回分类结果
            if (node.type == 0) {
                return node.label;
            }

            // 内部节点根据特征值和阈值选择左右子树
            if (feat[node.idx] <= node.val) {
                cur = node.left;
            } else {
                cur = node.right;
            }
        }
    }

    static int predictForest(List<Node[]> trees, int[] feat) {
        int cnt1 = 0;
        int total = trees.size();

        // 每棵树独立预测，并统计类别 1 的票数
        for (Node[] tree : trees) {
            if (predictTree(tree, feat) == 1) {
                cnt1++;
            }
        }

        // 平票时输出 0，因此只有 1 的票数严格过半才输出 1
        if (cnt1 * 2 > total) {
            return 1;
        }
        return 0;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();
        int M = sc.nextInt();
        int N = sc.nextInt();

        List<Node[]> trees = new ArrayList<>();

        // 读取所有决策树
        for (int t = 0; t < T; t++) {
            int K = sc.nextInt();

            // 节点编号从 1 到 K，因此数组长度为 K + 1
            Node[] tree = new Node[K + 1];

            for (int i = 1; i <= K; i++) {
                int type = sc.nextInt();
                Node node = new Node();
                node.type = type;

                if (type == 1) {
                    node.idx = sc.nextInt();
                    node.val = sc.nextInt();
                    node.left = sc.nextInt();
                    node.right = sc.nextInt();
                } else {
                    node.label = sc.nextInt();
                }

                tree[i] = node;
            }

            trees.add(tree);
        }

        StringBuilder sb = new StringBuilder();

        // 逐笔读取交易特征并输出预测结果
        for (int i = 0; i < N; i++) {
            int[] feat = new int[M + 1];

            // 特征索引从 1 开始，和题目中的 idx 保持一致
            for (int j = 1; j <= M; j++) {
                feat[j] = sc.nextInt();
            }

            sb.append(predictForest(trees, feat)).append('\n');
        }

        System.out.print(sb.toString());
        sc.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Node {
    int type;
    int idx;
    int val;
    int left;
    int right;
    int label;
};

int predictTree(const vector<Node>& tree, const vector<int>& feat) {
    // 从根节点 1 开始遍历
    int cur = 1;

    while (true) {
        const Node& node = tree[cur];

        // 叶子节点直接返回分类结果
        if (node.type == 0) {
            return node.label;
        }

        // 内部节点根据特征值和阈值选择左右子节点
        if (feat[node.idx] <= node.val) {
            cur = node.left;
        } else {
            cur = node.right;
        }
    }
}

int predictForest(const vector<vector<Node>>& trees, const vector<int>& feat) {
    int cnt1 = 0;
    int total = (int)trees.size();

    // 每棵树独立预测，并统计类别 1 的票数
    for (const auto& tree : trees) {
        if (predictTree(tree, feat) == 1) {
            cnt1++;
        }
    }

    // 平票时输出 0，因此只有 1 的票数严格过半才输出 1
    if (cnt1 * 2 > total) {
        return 1;
    }
    return 0;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T, M, N;
    cin >> T >> M >> N;

    vector<vector<Node>> trees;

    // 读取所有决策树
    for (int t = 0; t < T; t++) {
        int K;
        cin >> K;

        // 节点编号从 1 到 K，因此数组长度为 K + 1
        vector<Node> tree(K + 1);

        for (int i = 1; i <= K; i++) {
            int type;
            cin >> type;

            tree[i].type = type;

            if (type == 1) {
                cin >> tree[i].idx >> tree[i].val >> tree[i].left >> tree[i].right;
            } else {
                cin >> tree[i].label;
            }
        }

        trees.push_back(tree);
    }

    // 逐笔读取交易特征并输出预测结果
    for (int i = 0; i < N; i++) {
        vector<int> feat(M + 1);

        for (int j = 1; j <= M; j++) {

        cout << predictForest(trees, feat) << '\n';
    }

}
```
