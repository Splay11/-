## 解题思路

设某个节点 $u$ 的缓存中有若干条完整 token 序列，请求序列为 $r$，则该节点与请求的匹配长度为：

$$
LCP(u,r)=\max_{s \in cache_u} commonPrefixLen(s,r)
$$

每次请求到来时，需要选择：

1. $LCP$ 最大的节点；
2. 若多个节点 $LCP$ 相同，选择当前负载最小的节点；
3. 若负载也相同，代码中取编号较小的节点，保证结果确定。

核心做法是使用一棵全局前缀树 Trie。

Trie 中每个节点表示一个 token 前缀，例如前缀 $[10,20,30]$。
对于 Trie 的每个前缀节点，维护：

* 哪些推理节点的 KV Cache 中包含这个前缀；
* 一个小根堆，按当前负载选择该前缀下负载最小的推理节点。

根节点表示空前缀，也就是 $LCP=0$，所有推理节点都属于根节点。

处理一个请求时：

1. 从 Trie 根节点开始，沿着请求 token 依次向下走；
2. 能走到的最深 Trie 节点，就是当前请求能匹配到的最长公共前缀；
3. 在这个 Trie 节点维护的小根堆中，取当前负载最小的推理节点；
4. 将该请求分配给该节点；
5. 将请求完整 token 序列插入 Trie，表示该节点新增了一条 KV Cache；
6. 将请求长度累加到该节点负载上。

由于节点负载会变化，而一个节点可能出现在很多 Trie 前缀节点的小根堆中，如果每次负载变化都去更新所有堆，代价很高。因此使用懒更新：

* 堆中存储的是某一时刻的负载；
* 查询堆顶时，如果发现堆顶负载不是当前真实负载，就弹出并重新压入真实负载；
* 直到堆顶负载是最新的为止。

这样可以避免在负载变化时主动更新所有前缀节点。

## 复杂度分析

设所有初始缓存序列和所有请求序列的 token 总数为 $T$，推理节点数为 $N$。

每条序列插入 Trie 的复杂度与其长度成正比。
每个请求查询时，也只会沿着请求 token 走一遍 Trie。

小根堆每次调整的复杂度为 $O(\log N)$。

因此整体时间复杂度可认为是：

$$
O(T \log N)
$$

空间复杂度主要来自 Trie 节点、前缀到推理节点的关系以及堆：

$$
O(T)
$$

该复杂度适合处理较多请求和较长 token 序列的情况。

## 代码实现

### Python

```python
import sys
import heapq
import ast


class TrieNode:
    def __init__(self):
        # child[x] 表示当前前缀后面接 token x 到达的 Trie 节点
        self.child = {}

        # ids 记录哪些推理节点拥有当前前缀，避免同一节点重复加入
        self.ids = set()

        # 小根堆，元素为 (负载, 节点编号)
        self.heap = []


def parse_cache_seq(s):
    # 缓存序列形如 1,2,3，literal_eval 会解析成 tuple
    obj = ast.literal_eval(s)
    if isinstance(obj, tuple) or isinstance(obj, list):
        return [int(x) for x in obj]
    return [int(obj)]


def add_id(trie_node, idx, load):
    # 同一个推理节点只需要在同一个前缀节点中出现一次
    if idx not in trie_node.ids:
        trie_node.ids.add(idx)
        heapq.heappush(trie_node.heap, (int(load[idx]), idx))


def insert_seq(root, seq, idx, load):
    # 根节点表示空前缀，所有节点都拥有 LCP = 0 的能力
    cur = root
    add_id(cur, idx, load)

    # 将完整序列的所有前缀插入 Trie
    for x in seq:
        if x not in cur.child:
            cur.child[x] = TrieNode()
        cur = cur.child[x]
        add_id(cur, idx, load)


def get_best(trie_node, load):
    # 懒更新堆顶，直到堆顶负载与当前真实负载一致
    while True:
        old_load, idx = trie_node.heap[0]
        cur_load = int(load[idx])

        if old_load == cur_load:
            return idx

        heapq.heapreplace(trie_node.heap, (cur_load, idx))


def solve(n, load, caches, requests):
    root = TrieNode()

    # 根节点加入所有推理节点，用于处理 LCP = 0 的情况
    for i in range(n):
        add_id(root, i, load)

    # 插入初始 KV Cache
    for i in range(n):
        for seq in caches[i]:
            insert_seq(root, seq, i, load)

    ans = []

    for req in requests:
        cur = root
        best_prefix = root

        # 沿着请求 token 在 Trie 中尽量向下走
        for x in req:
            if x not in cur.child:
                break
            cur = cur.child[x]
            best_prefix = cur

        # best_prefix 就是当前请求能匹配到的最长公共前缀
        idx = get_best(best_prefix, load)
        ans.append(idx)

        # 当前请求完成 prefill 后，加入所选节点的 KV Cache
        load[idx] += len(req)
        insert_seq(root, req, idx, load)

    return ans


def main():
    input = sys.stdin.readline

    n, m = map(int, input().split())

    load = list(map(int, input().split()))

    caches = []
    for _ in range(n):
        parts = input().split()
        k = int(parts[0])

        seqs = []
        for i in range(k):
            seqs.append(parse_cache_seq(parts[i + 1]))
        caches.append(seqs)

    requests = []
    for _ in range(m):
        line = input().strip()
        if line:
            requests.append(list(map(int, line.split())))
        else:
            requests.append([])

    ans = solve(n, load, caches, requests)
    print(" ".join(map(str, ans)))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.PriorityQueue;
import java.util.Comparator;

public class Main {
    static class TrieNode {
        // child 记录 token 到下一个 Trie 节点编号的映射
        HashMap<Long, Integer> child = new HashMap<Long, Integer>();

        // ids 记录哪些推理节点拥有当前前缀，避免重复加入
        HashSet<Integer> ids = new HashSet<Integer>();

        // 小根堆，元素为 long[]{负载, 节点编号}
        PriorityQueue<long[]> heap = new PriorityQueue<long[]>(new Comparator<long[]>() {
            public int compare(long[] a, long[] b) {
                if (a[0] != b[0]) {
                    return Long.compare(a[0], b[0]);
                }
                return Long.compare(a[1], b[1]);
            }
        });
    }

    static long[] parseSeq(String s) {
        // 缓存序列用逗号分隔，替换为空格后用 StringTokenizer 解析
        s = s.replace(',', ' ');
        StringTokenizer st = new StringTokenizer(s);

        long[] arr = new long[st.countTokens()];
        for (int i = 0; i < arr.length; i++) {
            arr[i] = Long.parseLong(st.nextToken());
        }
        return arr;
    }

    static void addId(ArrayList<TrieNode> trie, int pos, int idx, long[] load) {
        TrieNode node = trie.get(pos);

        // 同一个推理节点只需要在同一个前缀节点中出现一次
        if (node.ids.add(idx)) {
            node.heap.offer(new long[]{load[idx], idx});
        }
    }

    static void insertSeq(ArrayList<TrieNode> trie, long[] seq, int idx, long[] load) {
        int cur = 0;
        addId(trie, cur, idx, load);

        // 将完整序列的所有前缀插入 Trie
        for (long x : seq) {
            TrieNode node = trie.get(cur);

            if (!node.child.containsKey(x)) {
                node.child.put(x, trie.size());
                trie.add(new TrieNode());
            }

            cur = node.child.get(x);
            addId(trie, cur, idx, load);
        }
    }

    static int getBest(ArrayList<TrieNode> trie, int pos, long[] load) {
        TrieNode node = trie.get(pos);

        // 懒更新堆顶，直到堆顶负载与当前真实负载一致
        while (true) {
            long[] top = node.heap.peek();
            int idx = (int) top[1];

            if (top[0] == load[idx]) {
                return idx;
            }

            node.heap.poll();
            node.heap.offer(new long[]{load[idx], idx});
        }
    }

    static int[] solve(int n, long[] load, ArrayList<long[]>[] caches, long[][] requests) {
        ArrayList<TrieNode> trie = new ArrayList<TrieNode>();
        trie.add(new TrieNode());

        // 根节点表示空前缀，用于处理 LCP = 0 的情况
        for (int i = 0; i < n; i++) {
            addId(trie, 0, i, load);
        }

        // 插入初始 KV Cache
        for (int i = 0; i < n; i++) {
            for (long[] seq : caches[i]) {
                insertSeq(trie, seq, i, load);
            }
        }

        int[] ans = new int[requests.length];

        for (int i = 0; i < requests.length; i++) {
            long[] req = requests[i];

            int cur = 0;
            int bestPrefix = 0;

            // 沿着请求 token 在 Trie 中尽量向下走
            for (long x : req) {
                TrieNode node = trie.get(cur);

                if (!node.child.containsKey(x)) {
                    break;
                }

                cur = node.child.get(x);
                bestPrefix = cur;
            }

            // bestPrefix 就是当前请求能匹配到的最长公共前缀
            int idx = getBest(trie, bestPrefix, load);
            ans[i] = idx;

            // 当前请求完成 prefill 后，加入所选节点的 KV Cache
            load[idx] += req.length;
            insertSeq(trie, req, idx, load);
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        long[] load = new long[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            load[i] = Long.parseLong(st.nextToken());
        }

        ArrayList<long[]>[] caches = new ArrayList[n];

        for (int i = 0; i < n; i++) {
            caches[i] = new ArrayList<long[]>();

            st = new StringTokenizer(br.readLine());
            int k = Integer.parseInt(st.nextToken());

            for (int j = 0; j < k; j++) {
                caches[i].add(parseSeq(st.nextToken()));
            }
        }

        long[][] requests = new long[m][];

        for (int i = 0; i < m; i++) {
            String line = br.readLine().trim();
            requests[i] = parseSeq(line);
        }

        int[] ans = solve(n, load, caches, requests);

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < ans.length; i++) {
            if (i > 0) {
                sb.append(' ');
            }
            sb.append(ans[i]);
        }

        System.out.println(sb.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

struct TrieNode {
    // child 记录 token 到下一个 Trie 节点编号的映射
    unordered_map<long long, int> child;

    // ids 记录哪些推理节点拥有当前前缀，避免重复加入
    unordered_set<int> ids;

    // 小根堆，元素为 (负载, 节点编号)
    priority_queue<
        pair<long long, int>,
        vector<pair<long long, int>>,
        greater<pair<long long, int>>
    > heap;
};

vector<long long> parseSeq(string s) {
    // 缓存序列用逗号分隔，替换为空格后用 stringstream 解析
    for (char &c : s) {
        if (c == ',') {
            c = ' ';
        }
    }

    stringstream ss(s);
    vector<long long> seq;
    long long x;

    while (ss >> x) {
        seq.push_back(x);
    }

    return seq;
}

void addId(vector<TrieNode> &trie, int pos, int idx, vector<long long> &load) {
    TrieNode &node = trie[pos];

    // 同一个推理节点只需要在同一个前缀节点中出现一次
    if (node.ids.insert(idx).second) {
        node.heap.push({load[idx], idx});
    }
}

void insertSeq(vector<TrieNode> &trie, const vector<long long> &seq, int idx, vector<long long> &load) {
    int cur = 0;
    addId(trie, cur, idx, load);

    // 将完整序列的所有前缀插入 Trie
    for (long long x : seq) {
        if (!trie[cur].child.count(x)) {
            trie[cur].child[x] = (int)trie.size();
            trie.push_back(TrieNode());
        }

        cur = trie[cur].child[x];
        addId(trie, cur, idx, load);
    }
}

int getBest(vector<TrieNode> &trie, int pos, vector<long long> &load) {
    TrieNode &node = trie[pos];

    // 懒更新堆顶，直到堆顶负载与当前真实负载一致
    while (true) {
        pair<long long, int> top = node.heap.top();
        int idx = top.second;

        if (top.first == load[idx]) {
            return idx;
        }

        node.heap.pop();
        node.heap.push({load[idx], idx});
    }
}

vector<int> solve(
    int n,
    vector<long long> &load,
    vector<vector<vector<long long>>> &caches,
    vector<vector<long long>> &requests
) {
    vector<TrieNode> trie;
    trie.push_back(TrieNode());

    // 根节点表示空前缀，用于处理 LCP = 0 的情况
    for (int i = 0; i < n; i++) {
        addId(trie, 0, i, load);
    }

    // 插入初始 KV Cache
    for (int i = 0; i < n; i++) {
        for (const vector<long long> &seq : caches[i]) {
            insertSeq(trie, seq, i, load);
        }
    }

    vector<int> ans;

    for (const vector<long long> &req : requests) {
        int cur = 0;
        int bestPrefix = 0;

        // 沿着请求 token 在 Trie 中尽量向下走
        for (long long x : req) {
            if (!trie[cur].child.count(x)) {
                break;
            }

            cur = trie[cur].child[x];
            bestPrefix = cur;
        }

        // bestPrefix 就是当前请求能匹配到的最长公共前缀
        int idx = getBest(trie, bestPrefix, load);
        ans.push_back(idx);

        // 当前请求完成 prefill 后，加入所选节点的 KV Cache
        load[idx] += (long long)req.size();
        insertSeq(trie, req, idx, load);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<long long> load(n);
    for (int i = 0; i < n; i++) {
        cin >> load[i];
    }

    string line;
    getline(cin, line);

    vector<vector<vector<long long>>> caches(n);

    for (int i = 0; i < n; i++) {
        getline(cin, line);

        stringstream ss(line);
        int k;
        ss >> k;

        for (int j = 0; j < k; j++) {
            string part;
            ss >> part;
            caches[i].push_back(parseSeq(part));
        }
    }

    vector<vector<long long>> requests;

    for (int i = 0; i < m; i++) {
        getline(cin, line);
        requests.push_back(parseSeq(line));
    }

    vector<int> ans = solve(n, load, caches, requests);

    for (int i = 0; i < (int)ans.size(); i++) {
        if (i > 0) {
            cout << ' ';
        }
        cout << ans[i];
    }

    cout << '\n';

    return 0;
}
```
