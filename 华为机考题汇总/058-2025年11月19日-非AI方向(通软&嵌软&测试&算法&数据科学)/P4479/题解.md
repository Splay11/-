## 解题思路

设前缀和数组为 $pre$，其中：

$$
pre[0]=0,\quad pre[i]=A_0+A_1+\dots+A_{i-1}
$$

若子数组 $[l,r)$ 能在 $k$ 处分成两个连续子子数组，且两段和都为 $0$，则有：

* $[l,k)$ 和为 $0$，说明 $pre[l]=pre[k]$
* $[k,r)$ 和为 $0$，说明 $pre[k]=pre[r]$

所以本题等价为：

> 在前缀和数组中，找三个下标 $i<j<k$，满足 $pre[i]=pre[j]=pre[k]$，并让 $k-i$ 最小。

对于某个固定前缀和值，它出现的位置为：

$$
p_1<p_2<p_3<\dots<p_m
$$

若要找最短区间，只需看连续三个位置：

$$
(p_t,p_{t+1},p_{t+2})
$$

因为如果不连续选，中间跨度只会更大。

### 实现方法

用哈希表记录每个前缀和值出现过的位置，只保留最近两个位置即可：

1. 从左到右计算前缀和
2. 当前前缀和值若之前已经出现至少两次
3. 则可以和“最近两个位置中的更早那个”构成一个连续三元组
4. 计算长度并更新最短长度和方案数
5. 再更新该前缀和值最近的两个出现位置

## 复杂度分析

* 时间复杂度：$O(N)$
* 空间复杂度：$O(N)$

其中 $N \le 10^6$，可以通过。

## 代码实现

### Python

```python
import sys


def solve(nums):
    # 哈希表：前缀和 -> 最近两个出现位置
    pos = {0: [0]}  # pre[0] = 0，位置是 0
    s = 0
    min_len = float('inf')
    cnt = 0

    # i 是前缀和位置，范围 1..n
    for i, x in enumerate(nums, 1):
        s += x

        if s not in pos:
            pos[s] = [i]
        else:
            arr = pos[s]

            # 若之前已有两个位置，则当前与倒数第二个历史位置组成连续三元组
            if len(arr) >= 2:
                length = i - arr[0]
                if length < min_len:
                    min_len = length
                    cnt = 1
                elif length == min_len:
                    cnt += 1

            # 只保留最近两个位置
            if len(arr) == 2:
                arr.pop(0)
            arr.append(i)

    if min_len == float('inf'):
        return -1, -1
    return min_len, cnt


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n = data[0]
    nums = data[1:1 + n]
    ans1, ans2 = solve(nums)
    print(ans1, ans2)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedInputStream;
import java.util.HashMap;
import java.util.Map;

public class Main {

    // 题面功能写在外部函数里
    static long[] solve(int[] nums) {
        // 哈希表：前缀和 -> 最近两个出现位置
        Map<Long, int[]> map = new HashMap<>();
        map.put(0L, new int[]{0, -1}); // pre[0] = 0，位置 0

        long sum = 0;
        long minLen = Long.MAX_VALUE;
        long count = 0;

        for (int i = 1; i <= nums.length; i++) {
            sum += nums[i - 1];

            int[] arr = map.get(sum);
            if (arr == null) {
                // 第一次出现，只记录一个位置
                map.put(sum, new int[]{i, -1});
            } else {
                int first = arr[0];
                int second = arr[1];

                // 已经有两个历史位置，当前形成连续三元组
                if (second != -1) {
                    long len = i - first;
                    if (len < minLen) {
                        minLen = len;
                        count = 1;
                    } else if (len == minLen) {
                        count++;
                    }
                    // 更新最近两个位置
                    arr[0] = second;
                    arr[1] = i;
                } else {
                    // 之前只有一个位置，现在补成两个
                    arr[1] = i;
                }
            }
        }

        if (minLen == Long.MAX_VALUE) {
            return new long[]{-1, -1};
        }
        return new long[]{minLen, count};
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();
        int n = fs.nextInt();
        int[] nums = new int[n];

        for (int i = 0; i < n; i++) {
            nums[i] = fs.nextInt();
        }

        long[] ans = solve(nums);
        System.out.println(ans[0] + " " + ans[1]);
    }

    // 数据范围较大，使用快读
    static class FastScanner {
        private final BufferedInputStream in = new BufferedInputStream(System.in);
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        private int read() throws Exception {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) {
                    return -1;
                }
            }
            return buffer[ptr++];
        }

        int nextInt() throws Exception {
            int c;
            do {
                c = read();
            } while (c <= ' ');

            int sign = 1;
            if (c == '-') {
                sign = -1;
                c = read();
            }

            int val = 0;
            while (c > ' ') {
                val = val * 10 + (c - '0');
                c = read();
            }
            return val * sign;
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 题面功能写在外部函数里
pair<int, int> solve(const vector<int>& nums) {
    // 哈希表：前缀和 -> 最近两个出现位置
    unordered_map<long long, pair<int, int>> mp;
    mp[0] = {0, -1};  // pre[0] = 0，位置 0

    long long sum = 0;
    int minLen = INT_MAX;
    int cnt = 0;

    for (int i = 1; i <= (int)nums.size(); i++) {
        sum += nums[i - 1];

        auto it = mp.find(sum);
        if (it == mp.end()) {
            // 第一次出现
            mp[sum] = {i, -1};
        } else {
            int first = it->second.first;
            int second = it->second.second;

            // 已经有两个历史位置，当前形成连续三元组
            if (second != -1) {
                int len = i - first;
                if (len < minLen) {
                    minLen = len;
                    cnt = 1;
                } else if (len == minLen) {
                    cnt++;
                }
                // 更新最近两个位置
                it->second = {second, i};
            } else {
                // 之前只有一个位置
                it->second.second = i;
            }
        }
    }

    if (minLen == INT_MAX) {
        return {-1, -1};
    }
    return {minLen, cnt};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> nums(n);

    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    pair<int, int> ans = solve(nums);
    cout << ans.first << " " << ans.second << '\n';

    return 0;
}
```
