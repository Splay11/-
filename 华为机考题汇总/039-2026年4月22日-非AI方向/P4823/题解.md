## 补充：本题同时能拓扑排序解决

基础一般的同学可通过本篇文章学习[《一文搞定拓扑排序》](https://codefun2000.com/codenote/hot100/P0038)

## 解题思路

这道题本质上分成两步：

$1.$ 判断依赖关系图中是否存在循环依赖。
$2.$ 如果不存在循环依赖，则把同一个被依赖包的版本号统一成所有要求中的最大版本号。

### 核心抽象

把每条依赖关系 $[a,b,v]$ 看成一条有向边：

* 包 $a$ 依赖包 $b$
* 边为 $a \rightarrow b$
* 版本号是这条边上对包 $b$ 的版本要求

于是：

* 判断是否有环，就是标准的有向图判环问题
* 版本处理就是统计每个“被依赖包”对应的最大版本号，再回填到所有指向它的边上

### 第一步：有向图判环

这里使用 $DFS$ 染色法判断有向图是否有环。

设每个点有 $3$ 种状态：

* $0$：未访问
* $1$：正在访问
* $2$：访问完成

当我们从某个点 $u$ 继续访问它依赖的点 $v$ 时：

* 如果 $v$ 的状态是 $1$，说明访问路径回到了当前递归栈中的点，存在环
* 如果 $v$ 的状态是 $0$，继续递归
* 如果 $v$ 的状态是 $2$，说明这个点已经处理完，跳过即可

题目还特别提示：自己依赖自己也算循环依赖。
而自环 $u \rightarrow u$ 在 $DFS$ 染色法中也会被自然识别出来。

注意，版本号不参与判环，判环只看包之间的依赖关系。

### 第二步：统一版本号

如果图无环，再处理版本号。

对于每条关系 $[a,b,v]$，我们关心的是“包 $b$ 被依赖时要求的版本号”。
因此只需要统计：

* 对每个被依赖包 $b$
* 记录所有指向它的边中的最大版本号

记为：

$maxVer[b] = \max(\text{所有依赖 } b \text{ 的版本号})$

然后再把每条边 $[a,b,v]$ 输出为：

$[a,b,maxVer[b]]$

这样就完成了题目要求的版本统一。

### 实现方法

对于每组输入：

$1.$ 读入所有依赖关系并保存原始顺序
$2.$ 建图 $a \rightarrow b$
$3.$ 用 $DFS$ 染色法判环
$4.$ 若有环，输出 $false$
$5.$ 若无环，统计每个被依赖包的最大版本号并按原顺序输出更新后的关系

题目固定输入两组数据，所以处理两次即可。

## 复杂度分析

设一组输入中有 $n$ 条依赖关系。

### 时间复杂度

* 建图：$O(n)$
* $DFS$ 判环：$O(n)$
* 统计最大版本号：$O(n)$
* 输出结果：$O(n)$

总时间复杂度为：

$O(n)$

### 空间复杂度

* 存储边：$O(n)$
* 邻接表：$O(n)$
* 访问状态与版本映射：$O(n)$

总空间复杂度为：

$O(n)$

这个复杂度对题目中的数据范围完全足够。

## 代码实现

### Python

```python
# 简易的二进制包依赖关系检查和处理
# ACM 风格：输入输出写在主函数里，核心逻辑写成外部函数

import sys


def has_cycle(graph):
    # 0-未访问，1-正在访问，2-访问完成
    color = {}

    def dfs(u):
        color[u] = 1  # 标记为正在访问
        for v in graph.get(u, []):
            if color.get(v, 0) == 1:
                # 访问到当前递归栈中的点，存在环
                return True
            if color.get(v, 0) == 0:
                if dfs(v):
                    return True
        color[u] = 2  # 当前点处理完成
        return False

    for u in graph:
        if color.get(u, 0) == 0:
            if dfs(u):
                return True
    return False


def solve_case(relations):
    graph = {}
    max_ver = {}
    nodes = set()

    # 建图，并统计每个被依赖包的最大版本号
    for a, b, v in relations:
        if a not in graph:
            graph[a] = []
        graph[a].append(b)

        nodes.add(a)
        nodes.add(b)

        if b not in max_ver or v > max_ver[b]:
            max_ver[b] = v

    # 保证没有出边的点也能参与判环入口遍历
    for x in nodes:
        if x not in graph:
            graph[x] = []

    # 先判环
    if has_cycle(graph):
        return None

    # 无环，按原顺序输出，并把版本号替换为目标包的最大版本号
    ans = []
    for a, b, v in relations:
        ans.append((a, b, max_ver[b]))
    return ans


def main():
    data = sys.stdin.read().strip().splitlines()
    idx = 0
    outputs = []

    # 题目说明固定输入两组数据
    for _ in range(2):
        n = int(data[idx].strip())
        idx += 1

        relations = []
        for _ in range(n):
            a, b, v = map(int, data[idx].strip().split(','))
            idx += 1
            relations.append((a, b, v))

        res = solve_case(relations)
        if res is None:
            outputs.append("false")
        else:
            for a, b, v in res:
                outputs.append(f"{a},{b},{v}")

    print("\n".join(outputs))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

// 简易的二进制包依赖关系检查和处理
// ACM 风格：类名必须为 Main
public class Main {

    // 保存一条依赖关系
    static class Edge {
        int from;
        int to;
        int ver;

        Edge(int from, int to, int ver) {
            this.from = from;
            this.to = to;
            this.ver = ver;
        }
    }

    // 使用 DFS 染色法判断有向图是否有环
    static boolean hasCycle(Map<Integer, List<Integer>> graph, Set<Integer> nodes) {
        Map<Integer, Integer> color = new HashMap<>();
        for (int node : nodes) {
            if (color.getOrDefault(node, 0) == 0) {
                if (dfs(node, graph, color)) {
                    return true;
                }
            }
        }
        return false;
    }

    static boolean dfs(int u, Map<Integer, List<Integer>> graph, Map<Integer, Integer> color) {
        color.put(u, 1); // 正在访问
        List<Integer> nextList = graph.getOrDefault(u, new ArrayList<>());

        for (int v : nextList) {
            int state = color.getOrDefault(v, 0);
            if (state == 1) {
                // 回到当前递归栈中的点，存在环
                return true;
            }
            if (state == 0) {
                if (dfs(v, graph, color)) {
                    return true;
                }
            }
        }

        color.put(u, 2); // 访问完成
        return false;
    }

    // 处理一组数据
    static List<String> solveCase(List<Edge> relations) {
        Map<Integer, List<Integer>> graph = new HashMap<>();
        Map<Integer, Integer> maxVer = new HashMap<>();
        Set<Integer> nodes = new HashSet<>();

        // 建图，并统计每个被依赖包的最大版本号
        for (Edge e : relations) {
            graph.computeIfAbsent(e.from, k -> new ArrayList<>()).add(e.to);
            nodes.add(e.from);
            nodes.add(e.to);

            if (!maxVer.containsKey(e.to) || e.ver > maxVer.get(e.to)) {
                maxVer.put(e.to, e.ver);
            }
        }

        // 判环
        if (hasCycle(graph, nodes)) {
            return null;
        }

        // 无环，按原顺序输出更新后的结果
        List<String> ans = new ArrayList<>();
        for (Edge e : relations) {
            ans.add(e.from + "," + e.to + "," + maxVer.get(e.to));
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        List<String> output = new ArrayList<>();

        // 题目固定输入两组数据
        for (int t = 0; t < 2; t++) {
            int n = Integer.parseInt(sc.nextLine().trim());
            List<Edge> relations = new ArrayList<>();

            for (int i = 0; i < n; i++) {
                String line = sc.nextLine().trim();
                String[] parts = line.split(",");
                int a = Integer.parseInt(parts[0]);
                int b = Integer.parseInt(parts[1]);
                int v = Integer.parseInt(parts[2]);
                relations.add(new Edge(a, b, v));
            }

            List<String> res = solveCase(relations);
            if (res == null) {
                output.add("false");
            } else {
                output.addAll(res);
            }
        }

        for (String s : output) {
            System.out.println(s);
        }

        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <unordered_map>
#include <unordered_set>
#include <string>
using namespace std;

// 保存一条依赖关系
struct Edge {
    int from;
    int to;
    int ver;
};

// DFS 染色法判断有向图是否有环
bool dfs(int u, unordered_map<int, vector<int>>& graph, unordered_map<int, int>& color) {
    color[u] = 1; // 正在访问

    for (int v : graph[u]) {
        if (color[v] == 1) {
            // 回到当前递归栈中的点，存在环
            return true;
        }
        if (color[v] == 0) {
            if (dfs(v, graph, color)) {
                return true;
            }
        }
    }

    color[u] = 2; // 访问完成
    return false;
}

bool hasCycle(unordered_map<int, vector<int>>& graph, unordered_set<int>& nodes) {
    unordered_map<int, int> color; // 0-未访问，1-正在访问，2-访问完成

    for (int u : nodes) {
        if (color[u] == 0) {
            if (dfs(u, graph, color)) {
                return true;
            }
        }
    }
    return false;
}

// 处理一组数据
vector<string> solveCase(vector<Edge>& relations, bool& ok) {
    unordered_map<int, vector<int>> graph;
    unordered_map<int, int> maxVer;
    unordered_set<int> nodes;

    // 建图，并统计每个被依赖包的最大版本号
    for (auto& e : relations) {
        graph[e.from].push_back(e.to);
        nodes.insert(e.from);
        nodes.insert(e.to);

        if (!maxVer.count(e.to) || e.ver > maxVer[e.to]) {
            maxVer[e.to] = e.ver;
        }
    }

    // 保证没有出边的点也在图中
    for (int x : nodes) {
        if (!graph.count(x)) {
            graph[x] = vector<int>();
        }
    }

    // 判环
    if (hasCycle(graph, nodes)) {
        ok = false;
        return {};
    }

    ok = true;
    vector<string> ans;

    // 无环，按原顺序输出更新后的结果
    for (auto& e : relations) {
        ans.push_back(to_string(e.from) + "," + to_string(e.to) + "," + to_string(maxVer[e.to]));
    }

    return ans;
}

int main() {
    vector<string> output;

    // 题目固定输入两组数据
    for (int t = 0; t < 2; t++) {
        int n;
        cin >> n;

        vector<Edge> relations;
        for (int i = 0; i < n; i++) {
            string s;
            cin >> s;

            int p1 = s.find(',');
            int p2 = s.find(',', p1 + 1);

            int a = stoi(s.substr(0, p1));
            int b = stoi(s.substr(p1 + 1, p2 - p1 - 1));
            int v = stoi(s.substr(p2 + 1));

            relations.push_back({a, b, v});
        }

        bool ok;
        vector<string> res = solveCase(relations, ok);

        if (!ok) {
            output.push_back("false");
        } else {
            for (string& s : res) {
                output.push_back(s);
            }
        }
    }

    for (string& s : output) {
        cout << s << '\n';
    }

    return 0;
}
```
