## 解题思路

本题要求模拟 $BPE$ 的迭代合并过程。

初始时，字符串 $S$ 中的每个字符都是一个独立的 $Token$。每一轮操作分为三步：

1. 统计当前序列中所有相邻 $Token$ 对的出现次数。
2. 找到出现频率最高的相邻对。

   * 如果最高频率为 $1$，说明没有任何值得继续合并的重复相邻对，直接停止。
   * 如果多个相邻对频率相同，则选择字典序最小的二元组。
3. 从左到右扫描当前 $Token$ 序列：

   * 如果当前位置和下一个位置正好组成最佳相邻对，则将它们合并成一个新 $Token$。
   * 合并后跳过下一个位置。
   * 否则保留当前 $Token$。

由于 $|S| \le 1000$，$K \le 100$，直接模拟每一轮即可，复杂度完全可以接受。

需要注意的是，字典序比较的是相邻对整体：

$$(x_1, y_1) < (x_2, y_2)$$

当且仅当：

* $x_1 < x_2$，或
* $x_1 = x_2$ 且 $y_1 < y_2$

例如：

$$(a,b) < (b,a)$$

并且：

$$(a,b) < (aa,a)$$

因为字符串 $a$ 按字典序小于 $aa$。

## 复杂度分析

设初始字符串长度为 $n$，最大迭代次数为 $K$。

每一轮中：

* 统计相邻对需要扫描当前 $Token$ 序列一次，时间复杂度为 $O(n)$。
* 选择最佳相邻对需要遍历所有不同的相邻对，最多 $O(n)$ 个。
* 合并操作也需要扫描一次当前序列，时间复杂度为 $O(n)$。

因此总时间复杂度为：

$$O(Kn)$$

由于 $K \le 100$，$n \le 1000$，该复杂度足够通过。

空间上，需要保存当前 $Token$ 序列和相邻对频率表，空间复杂度为：

$$O(n)$$

## 代码实现

### Python

```python
def bpe(k, s):
    # 初始时，每个字符都是一个独立的 Token
    arr = list(s)

    # 最多执行 k 轮合并
    for _ in range(k):
        # 如果当前 Token 数量不足 2，则不存在相邻对
        if len(arr) < 2:
            break

        cnt = {}

        # 统计所有相邻 Token 对的出现次数
        for i in range(len(arr) - 1):
            p = (arr[i], arr[i + 1])
            cnt[p] = cnt.get(p, 0) + 1

        best = None
        best_cnt = 0

        # 选择频率最高的相邻对，频率相同则选择字典序最小的
        for p in cnt:
            if cnt[p] > best_cnt:
                best_cnt = cnt[p]
                best = p
            elif cnt[p] == best_cnt and p < best:
                best = p

        # 如果最高频率为 1，则提前终止
        if best_cnt == 1:
            break

        new_arr = []
        i = 0

        # 从左到右贪心合并所有最佳相邻对
        while i < len(arr):
            if i + 1 < len(arr) and arr[i] == best[0] and arr[i + 1] == best[1]:
                new_arr.append(arr[i] + arr[i + 1])
                i += 2
            else:
                new_arr.append(arr[i])
                i += 1

        arr = new_arr

    return arr


def main():
    k = int(input().strip())
    s = input().strip()

    ans = bpe(k, s)
    print(" ".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static class Pair {
        String a;
        String b;

        Pair(String a, String b) {
            this.a = a;
            this.b = b;
        }

        public boolean equals(Object obj) {
            if (!(obj instanceof Pair)) {
                return false;
            }
            Pair p = (Pair) obj;
            return a.equals(p.a) && b.equals(p.b);
        }

        public int hashCode() {
            return a.hashCode() * 31 + b.hashCode();
        }
    }

    static boolean less(Pair x, Pair y) {
        // 按二元组字典序比较
        int c = x.a.compareTo(y.a);
        if (c != 0) {
            return c < 0;
        }
        return x.b.compareTo(y.b) < 0;
    }

    static List<String> bpe(int k, String s) {
        List<String> arr = new ArrayList<>();

        // 初始时，每个字符都是一个独立的 Token
        for (int i = 0; i < s.length(); i++) {
            arr.add(String.valueOf(s.charAt(i)));
        }

        // 最多执行 k 轮合并
        for (int t = 0; t < k; t++) {
            // 如果当前 Token 数量不足 2，则不存在相邻对
            if (arr.size() < 2) {
                break;
            }

            Map<Pair, Integer> cnt = new HashMap<>();

            // 统计所有相邻 Token 对的出现次数
            for (int i = 0; i + 1 < arr.size(); i++) {
                Pair p = new Pair(arr.get(i), arr.get(i + 1));
                cnt.put(p, cnt.getOrDefault(p, 0) + 1);
            }

            Pair best = null;
            int bestCnt = 0;

            // 选择频率最高的相邻对，频率相同则选择字典序最小的
            for (Pair p : cnt.keySet()) {
                int v = cnt.get(p);

                if (v > bestCnt) {
                    bestCnt = v;
                    best = p;
                } else if (v == bestCnt && less(p, best)) {
                    best = p;
                }
            }

            // 如果最高频率为 1，则提前终止
            if (bestCnt == 1) {
                break;
            }

            List<String> next = new ArrayList<>();
            int i = 0;

            // 从左到右贪心合并所有最佳相邻对
            while (i < arr.size()) {
                if (i + 1 < arr.size()
                        && arr.get(i).equals(best.a)
                        && arr.get(i + 1).equals(best.b)) {
                    next.add(arr.get(i) + arr.get(i + 1));
                    i += 2;
                } else {
                    next.add(arr.get(i));
                    i++;
                }
            }

            arr = next;
        }

        return arr;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int k = sc.nextInt();
        String s = sc.next();

        List<String> ans = bpe(k, s);

        for (int i = 0; i < ans.size(); i++) {
            if (i > 0) {
                System.out.print(" ");
            }
            System.out.print(ans.get(i));
        }
        System.out.println();

        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<string> bpe(int k, string s) {
    vector<string> arr;

    // 初始时，每个字符都是一个独立的 Token
    for (int i = 0; i < (int)s.size(); i++) {
        arr.push_back(string(1, s[i]));
    }

    // 最多执行 k 轮合并
    for (int t = 0; t < k; t++) {
        // 如果当前 Token 数量不足 2，则不存在相邻对
        if ((int)arr.size() < 2) {
            break;
        }

        map<pair<string, string>, int> cnt;

        // 统计所有相邻 Token 对的出现次数
        for (int i = 0; i + 1 < (int)arr.size(); i++) {
            cnt[{arr[i], arr[i + 1]}]++;
        }

        pair<string, string> best;
        int bestCnt = 0;

        // map 会按照 pair 的字典序从小到大遍历
        // 因此频率相同时，不更新即可保留字典序最小的相邻对
        for (auto it : cnt) {
            if (it.second > bestCnt) {
                bestCnt = it.second;
                best = it.first;
            }
        }

        // 如果最高频率为 1，则提前终止
        if (bestCnt == 1) {
            break;
        }

        vector<string> next;
        int i = 0;

        // 从左到右贪心合并所有最佳相邻对
        while (i < (int)arr.size()) {
            if (i + 1 < (int)arr.size()
                && arr[i] == best.first
                && arr[i + 1] == best.second) {
                next.push_back(arr[i] + arr[i + 1]);
                i += 2;
            } else {
                next.push_back(arr[i]);
                i++;
            }
        }

        arr = next;
    }

    return arr;
}

int main() {
    int k;
    string s;

    cin >> k;
    cin >> s;

    vector<string> ans = bpe(k, s);

    for (int i = 0; i < (int)ans.size(); i++) {
        if (i > 0) {
            cout << " ";
        }
        cout << ans[i];
    }
    cout << endl;

    return 0;
}
```
