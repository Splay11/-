## 解题思路

维护三个核心结构：

1. 队列 `window`
   用于保存滑动窗口中的关键词顺序，方便在窗口超过大小 $W$ 时淘汰最早元素。

2. 哈希表 `cnt`
   记录当前窗口中每个关键词的出现次数。

3. 优先队列 `heap`
   按照规则维护候选 Top-K：

   * 频率越高优先级越高；
   * 频率相同，字典序越小优先级越高。

由于关键词频率会不断变化，堆中可能存在旧数据。为了解决这个问题，给每个关键词维护一个版本号 `ver`。每次关键词频率变化时，版本号加一，并把新的状态放入堆中。

查询时，从堆顶不断弹出无效数据，直到找到当前有效的关键词。取出前 $k$ 个有效词后，再放回堆中，保证后续查询仍可使用。

## 复杂度分析

设操作总数为 $Q$，一次查询输出数量为 $k$。

每次 `add` 操作需要更新哈希表和堆，时间复杂度为：

$$
O(\log Q)
$$

每次 `get` 操作最多取出 $k$ 个有效元素，并清理若干过期元素，时间复杂度为：

$$
O(k \log Q)
$$

所有过期元素最多被弹出一次，因此整体复杂度是合适的。

空间复杂度为：

$$
O(Q)
$$

## 代码实现

### Python

```python
import sys
import heapq
from collections import deque, defaultdict

def add_word(word, w, window, cnt, ver, heap):
    # 新关键词进入滑动窗口
    window.append(word)

    # 更新该关键词频率和版本号
    cnt[word] += 1
    ver[word] += 1
    heapq.heappush(heap, (-cnt[word], word, ver[word]))

    # 如果窗口超过大小，需要淘汰最早的关键词
    if len(window) > w:
        old = window.popleft()

        # 被淘汰关键词频率减一
        cnt[old] -= 1
        ver[old] += 1

        # 如果频率为 0，则从统计表中移除
        if cnt[old] == 0:
            del cnt[old]
        else:
            heapq.heappush(heap, (-cnt[old], old, ver[old]))

def get_top(k, cnt, ver, heap):
    ans = []
    temp = []

    # 取出当前有效的前 k 个关键词
    while heap and len(ans) < k:
        neg_f, word, v = heapq.heappop(heap)
        f = -neg_f

        # 判断堆顶数据是否仍然是当前最新状态
        if word in cnt and cnt[word] == f and ver[word] == v:
            ans.append(word)
            temp.append((neg_f, word, v))

    # 将本次查询取出的有效元素放回堆中
    for item in temp:
        heapq.heappush(heap, item)

    return ans

def main():
    input = sys.stdin.readline

    w, q = map(int, input().split())

    window = deque()
    cnt = defaultdict(int)
    ver = defaultdict(int)
    heap = []

    res = []

    for _ in range(q):
        op = input().split()

        if op[0] == "add":
            add_word(op[1], w, window, cnt, ver, heap)
        else:
            k = int(op[1])
            ans = get_top(k, cnt, ver, heap)
            res.append(" ".join(ans))

    print("\n".join(res))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static class Node {
        String word;
        int freq;
        int ver;

        Node(String word, int freq, int ver) {
            this.word = word;
            this.freq = freq;
            this.ver = ver;
        }
    }

    static void addWord(String word, int w, Queue<String> window,
                        Map<String, Integer> cnt,
                        Map<String, Integer> ver,
                        PriorityQueue<Node> heap) {
        // 新关键词进入滑动窗口
        window.offer(word);

        // 更新该关键词频率和版本号
        cnt.put(word, cnt.getOrDefault(word, 0) + 1);
        ver.put(word, ver.getOrDefault(word, 0) + 1);
        heap.offer(new Node(word, cnt.get(word), ver.get(word)));

        // 如果窗口超过大小，需要淘汰最早的关键词
        if (window.size() > w) {
            String old = window.poll();

            // 被淘汰关键词频率减一
            cnt.put(old, cnt.get(old) - 1);
            ver.put(old, ver.getOrDefault(old, 0) + 1);

            // 如果频率为 0，则从统计表中移除
            if (cnt.get(old) == 0) {
                cnt.remove(old);
            } else {
                heap.offer(new Node(old, cnt.get(old), ver.get(old)));
            }
        }
    }

    static List<String> getTop(int k, Map<String, Integer> cnt,
                               Map<String, Integer> ver,
                               PriorityQueue<Node> heap) {
        List<String> ans = new ArrayList<>();
        List<Node> temp = new ArrayList<>();

        // 取出当前有效的前 k 个关键词
        while (!heap.isEmpty() && ans.size() < k) {
            Node cur = heap.poll();

            // 判断堆顶数据是否仍然是当前最新状态
            if (cnt.containsKey(cur.word)
                    && cnt.get(cur.word) == cur.freq
                    && ver.get(cur.word) == cur.ver) {
                ans.add(cur.word);
                temp.add(cur);
            }
        }

        // 将本次查询取出的有效元素放回堆中
        for (Node node : temp) {
            heap.offer(node);
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int w = sc.nextInt();
        int q = sc.nextInt();

        Queue<String> window = new LinkedList<>();
        Map<String, Integer> cnt = new HashMap<>();
        Map<String, Integer> ver = new HashMap<>();

        PriorityQueue<Node> heap = new PriorityQueue<>((a, b) -> {
            // 频率高的优先
            if (a.freq != b.freq) {
                return b.freq - a.freq;
            }
            // 频率相同，字典序小的优先
            return a.word.compareTo(b.word);
        });

        StringBuilder out = new StringBuilder();

        for (int i = 0; i < q; i++) {
            String op = sc.next();

            if (op.equals("add")) {
                String word = sc.next();
                addWord(word, w, window, cnt, ver, heap);
            } else {
                int k = sc.nextInt();
                List<String> ans = getTop(k, cnt, ver, heap);

                for (int j = 0; j < ans.size(); j++) {
                    if (j > 0) {
                        out.append(' ');
                    }
                    out.append(ans.get(j));
                }
                out.append('\n');
            }
        }

        System.out.print(out.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Node {
    string word;
    int freq;
    int ver;
};

struct Cmp {
    bool operator()(const Node& a, const Node& b) const {
        // 频率高的优先
        if (a.freq != b.freq) {
            return a.freq < b.freq;
        }
        // 频率相同，字典序小的优先
        return a.word > b.word;
    }
};

void addWord(string word, int w,
             queue<string>& window,
             unordered_map<string, int>& cnt,
             unordered_map<string, int>& ver,
             priority_queue<Node, vector<Node>, Cmp>& heap) {
    // 新关键词进入滑动窗口
    window.push(word);

    // 更新该关键词频率和版本号
    cnt[word]++;
    ver[word]++;
    heap.push({word, cnt[word], ver[word]});

    // 如果窗口超过大小，需要淘汰最早的关键词
    if ((int)window.size() > w) {
        string old = window.front();
        window.pop();

        // 被淘汰关键词频率减一
        cnt[old]--;
        ver[old]++;

        // 如果频率为 0，则从统计表中移除
        if (cnt[old] == 0) {
            cnt.erase(old);
        } else {
            heap.push({old, cnt[old], ver[old]});
        }
    }
}

vector<string> getTop(int k,
                      unordered_map<string, int>& cnt,
                      unordered_map<string, int>& ver,
                      priority_queue<Node, vector<Node>, Cmp>& heap) {
    vector<string> ans;
    vector<Node> temp;

    // 取出当前有效的前 k 个关键词
    while (!heap.empty() && (int)ans.size() < k) {
        Node cur = heap.top();
        heap.pop();

        // 判断堆顶数据是否仍然是当前最新状态
        if (cnt.count(cur.word)
            && cnt[cur.word] == cur.freq
            && ver[cur.word] == cur.ver) {
            ans.push_back(cur.word);
            temp.push_back(cur);
        }
    }

    // 将本次查询取出的有效元素放回堆中
    for (Node node : temp) {
        heap.push(node);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int w, q;
    cin >> w >> q;

    queue<string> window;
    unordered_map<string, int> cnt;
    unordered_map<string, int> ver;
    priority_queue<Node, vector<Node>, Cmp> heap;

    for (int i = 0; i < q; i++) {
        string op;
        cin >> op;

        if (op == "add") {
            string word;
            cin >> word;
            addWord(word, w, window, cnt, ver, heap);
        } else {
            int k;
            cin >> k;

            vector<string> ans = getTop(k, cnt, ver, heap);

            for (int j = 0; j < (int)ans.size(); j++) {
                if (j > 0) {
                    cout << ' ';
                }
                cout << ans[j];
            }
            cout << '\n';
        }
    }

    return 0;
}
```
