## 解题思路
### 相关算法与实现要点

* **K-Means**

  * 初始中心：取前 $k$ 个点作为初始中心（简单且可重复）。

  * 迭代过程：

    * 分配：每个点归到**欧氏距离**最近的中心；
    * 更新：每个簇的新中心为簇内点坐标的均值；

  * 终止条件：迭代次数上限 100 次，或所有中心移动距离之和 $\le 1e{-6}$。

  > 若出现空簇，保持其中心不变（在本题给定范围下可稳定收敛；也可按需要将最远点“拉”成新簇中心，本文实现采用保持不变的简洁策略）。

* **轮廓系数（Silhouette）**
  对于样本 $p$：

  * $a(p)$：与同簇其它点的平均距离；若该簇仅单个样本，则 $a(p)=1$；
  * $b(p)$：与**最近的其它簇**内全部点的平均距离；
  * 轮廓值：$s(p)=\dfrac{b(p)-a(p)}{\max\{a(p),\,b(p)\}} \in [-1,1]$。
    簇的轮廓系数为簇内样本的 $s(p)$ 平均值。
    实现时按定义直接枚举计算，时间复杂度 $O(n^2)$，对 $n\le 500$ 可接受。

* **输出与舍入**

  * 输出为被选中簇中心的 $(x,y)$，采用**银行家舍入**保留两位：

    * Python：`Decimal(...).quantize(Decimal('0.00'), ROUND_HALF_EVEN)`
    * Java：`BigDecimal.setScale(2, RoundingMode.HALF_EVEN)`
    * C++：自写“半舍六入五留双”的函数，避免默认“四舍五入”。

## 代码实现

### Python

```python
import sys
from decimal import Decimal, ROUND_HALF_EVEN
import numpy as np


def format_half_even(x):
    """按题意使用 HALF_EVEN 保留两位小数"""
    return str(Decimal(str(x)).quantize(Decimal("0.00"), rounding=ROUND_HALF_EVEN))


def kmeans(points, k, max_iter=100, eps=1e-6):
    """
    按题意实现 K-Means：
    1. 前 k 个点作为初始中心
    2. 每轮按欧氏距离分配
    3. 每轮更新为簇内均值
    4. 空簇时中心保持不变
    5. 终止条件：最多 100 次，或所有中心移动距离都 <= 1e-6
    """
    n = points.shape[0]

    # 前 k 个点作为初始中心
    centers = points[:k].astype(np.float64).copy()

    labels = np.zeros(n, dtype=np.int32)

    for _ in range(max_iter):
        # 计算每个点到每个中心的欧氏距离
        # diff shape: (n, k, 2)
        diff = points[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff * diff, axis=2))  # shape: (n, k)

        # 分配到最近中心
        new_labels = np.argmin(dist, axis=1)

        # 重新计算中心
        new_centers = centers.copy()
        for j in range(k):
            cluster_points = points[new_labels == j]
            if len(cluster_points) > 0:
                new_centers[j] = np.mean(cluster_points, axis=0)
            # 若为空簇，保持中心不变

        # 判断是否收敛：所有中心移动距离都 <= eps
        shift = np.sqrt(np.sum((new_centers - centers) ** 2, axis=1))

        centers = new_centers
        labels = new_labels

        if np.all(shift <= eps):
            break

    return labels, centers


def compute_cluster_silhouette(points, labels, k):
    """
    计算每个簇的轮廓系数（簇内样本轮廓系数平均值）
    若某点所在簇仅有 1 个样本，则按题意取 $a(p)=1$，再按轮廓系数定义计算 $s(p)$。

    """
    n = points.shape[0]

    # 预计算两两欧氏距离矩阵，O(n^2)，n<=500 可接受
    diff = points[:, np.newaxis, :] - points[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff * diff, axis=2))  # shape: (n, n)

    cluster_indices = [np.where(labels == c)[0] for c in range(k)]
    point_silhouette = np.zeros(n, dtype=np.float64)

    for i in range(n):
        ci = labels[i]
        same_cluster = cluster_indices[ci]

        # 若所在簇只有自己一个点，按题意 a(i)=1
        if len(same_cluster) <= 1:
            a = 1.0

            # b(i): 到其他每个簇的平均距离中的最小值
            b = float("inf")
            for c in range(k):
                if c == ci or len(cluster_indices[c]) == 0:
                    continue
                avg_dist = np.mean(dist_matrix[i, cluster_indices[c]])
                if avg_dist < b:
                    b = avg_dist

            denom = max(a, b)
            if denom == 0 or b == float("inf"):
                s = 0.0
            else:
                s = (b - a) / denom

            point_silhouette[i] = s
            continue


        # a(i): 到同簇其他点的平均距离
        same_others = same_cluster[same_cluster != i]
        a = np.mean(dist_matrix[i, same_others])

        # b(i): 到其他每个簇的平均距离中的最小值
        b = float("inf")
        for c in range(k):
            if c == ci or len(cluster_indices[c]) == 0:
                continue
            avg_dist = np.mean(dist_matrix[i, cluster_indices[c]])
            if avg_dist < b:
                b = avg_dist

        # 按定义计算
        # 理论上这里 max(a,b) 不会为 0；稳妥起见做保护
        denom = max(a, b)
        if denom == 0:
            s = 0.0
        else:
            s = (b - a) / denom

        point_silhouette[i] = s

    # 计算每个簇的平均轮廓系数
    cluster_silhouette = np.zeros(k, dtype=np.float64)
    for c in range(k):
        idx = cluster_indices[c]
        if len(idx) == 0:
            # 正常来说空簇中心保持不变，但最终可能仍为空
            # 这里给一个极大值或 0 都行；按题意一般不会专门选空簇输出
            # 为避免误选，设为 +inf
            cluster_silhouette[c] = float("inf")
        else:
            cluster_silhouette[c] = np.mean(point_silhouette[idx])

    return cluster_silhouette


def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    n = int(data[0])
    k = int(data[1])

    coords = list(map(int, data[2:]))
    points = np.array(coords, dtype=np.float64).reshape(n, 2)

    labels, centers = kmeans(points, k)
    cluster_silhouette = compute_cluster_silhouette(points, labels, k)

    # 找轮廓系数最低的簇
    # 若有并列，argmin 默认取下标最小的
    worst_cluster = int(np.argmin(cluster_silhouette))
    ans_x, ans_y = centers[worst_cluster]

    print(f"{format_half_even(ans_x)},{format_half_even(ans_y)}")


if __name__ == "__main__":
    solve()
```

### Java


```java
import java.util.*;
import java.math.*;

public class Main {

    static class Res {
        int[] lab; double[][] cen;
        Res(int[] l, double[][] c){ lab=l; cen=c; }
    }

    static Res kmeans(double[][] pts, int k){
        int n = pts.length;
        double[][] cen = new double[k][2];
        for(int c=0;c<k;c++){ cen[c][0]=pts[c][0]; cen[c][1]=pts[c][1]; }
        int[] lab = new int[n];

        for(int it=0; it<100; it++){
            // 分配
            for(int i=0;i<n;i++){
                int bi=0; double bd=dist2(pts[i], cen[0]);
                for(int c=1;c<k;c++){
                    double d = dist2(pts[i], cen[c]);
                    if(d<bd){ bd=d; bi=c; }
                }
                lab[i]=bi;
            }
            // 更新
            double[] sx=new double[k], sy=new double[k]; int[] cnt=new int[k];
            for(int i=0;i<n;i++){ int c=lab[i]; sx[c]+=pts[i][0]; sy[c]+=pts[i][1]; cnt[c]++; }
            double moved=0;
            for(int c=0;c<k;c++){
                double nx=cen[c][0], ny=cen[c][1];
                if(cnt[c]>0){ nx=sx[c]/cnt[c]; ny=sy[c]/cnt[c]; }
                moved += Math.abs(nx-cen[c][0])+Math.abs(ny-cen[c][1]);
                cen[c][0]=nx; cen[c][1]=ny;
            }
            if(moved<=1e-6) break;
        }
        return new Res(lab, cen);
    }

    static double dist2(double[] a, double[] b){
        double dx=a[0]-b[0], dy=a[1]-b[1];
        return dx*dx+dy*dy;
    }

    static double[] silhouette(double[][] pts, int[] lab, int k){
        int n=pts.length;
        ArrayList<Integer>[] g=new ArrayList[k];
        for(int i=0;i<k;i++) g[i]=new ArrayList<>();
        for(int i=0;i<n;i++) g[lab[i]].add(i);

        double[] avg=new double[k];
        for(int c=0;c<k;c++){
            ArrayList<Integer> idx=g[c];
            if(idx.isEmpty()){ avg[c]=0; continue; }
            double sum=0;
            for(int id: idx){
                double a;
                if(idx.size()==1) a=1;
                else{
                    double t=0;
                    for(int j: idx) if(j!=id)
                        t += Math.hypot(pts[id][0]-pts[j][0], pts[id][1]-pts[j][1]);
                    a = t/(idx.size()-1);
                }
                double b=Double.POSITIVE_INFINITY;
                for(int c2=0;c2<k;c2++){
                    if(c2==c || g[c2].isEmpty()) continue;
                    double t=0;
                    for(int j: g[c2])
                        t += Math.hypot(pts[id][0]-pts[j][0], pts[id][1]-pts[j][1]);
                    b = Math.min(b, t/g[c2].size());
                }
                double s;
                if(Double.isInfinite(b)) s=0;
                else{
                    double m=Math.max(a,b);
                    s = (m==0)?0:(b-a)/m;
                }
                sum += s;
            }
            avg[c]=sum/idx.size();
        }
        return avg;
    }

    static String r2(double v){
        BigDecimal bd=new BigDecimal(Double.toString(v)).setScale(2, RoundingMode.HALF_EVEN);
        return String.format(java.util.Locale.US, "%.2f", bd.doubleValue());
    }

    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        if(!sc.hasNext()) return;
        int n=sc.nextInt(), k=sc.nextInt();
        double[][] pts=new double[n][2];
        for(int i=0;i<n;i++){ pts[i][0]=sc.nextDouble(); pts[i][1]=sc.nextDouble(); }
        Res r=kmeans(pts,k);
        double[] sil=silhouette(pts, r.lab, k);
        int bad=0;
        for(int c=1;c<k;c++) if(sil[c]<sil[bad]) bad=c;
        System.out.println(r2(r.cen[bad][0])+","+r2(r.cen[bad][1]));
        sc.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

static inline double dist2(double x1,double y1,double x2,double y2){
    double dx=x1-x2, dy=y1-y2; return dx*dx+dy*dy;
}
static inline double dist(double x1,double y1,double x2,double y2){
    return hypot(x1-x2, y1-y2);
}

// 银行家舍入到两位
string r2(double v){
    long double s = (long double)v * 100.0L;
    long double f = floor(s);
    long double frac = s - f;
    const long double EPS = 1e-12L;
    long long u;
    if(frac > 0.5L + EPS) u = (long long)f + 1;
    else if(frac < 0.5L - EPS) u = (long long)f;
    else u = ((long long)f % 2 == 0) ? (long long)f : (long long)f + 1;
    long long ip = u / 100, fp = llabs(u % 100);
    ostringstream os; os<<ip<<'.'<<setw(2)<<setfill('0')<<fp; return os.str();
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,k; if(!(cin>>n>>k)) return 0;
    vector<double> x(n), y(n);
    for(int i=0;i<n;i++) cin>>x[i]>>y[i];

    // K-Means：前 k 个点为初始中心
    vector<double> cx(k), cy(k);
    for(int c=0;c<k;c++){ cx[c]=x[c]; cy[c]=y[c]; }
    vector<int> lab(n);

    for(int it=0; it<100; ++it){
        // 分配
        for(int i=0;i<n;i++){
            int bi=0; double bd=dist2(x[i],y[i],cx[0],cy[0]);
            for(int c=1;c<k;c++){
                double d=dist2(x[i],y[i],cx[c],cy[c]);
                if(d<bd){ bd=d; bi=c; }
            }
            lab[i]=bi;
        }
        // 更新
        vector<double> sx(k,0), sy(k,0); vector<int> cnt(k,0);
        for(int i=0;i<n;i++){ int c=lab[i]; sx[c]+=x[i]; sy[c]+=y[i]; cnt[c]++; }
        double moved=0;
        for(int c=0;c<k;c++){
            double nx=cx[c], ny=cy[c];
            if(cnt[c]>0){ nx=sx[c]/cnt[c]; ny=sy[c]/cnt[c]; }
            moved += fabs(nx-cx[c])+fabs(ny-cy[c]);
            cx[c]=nx; cy[c]=ny;
        }
        if(moved<=1e-6) break;
    }

    // 轮廓系数
    vector<vector<int>> g(k);
    for(int i=0;i<n;i++) g[lab[i]].push_back(i);
    vector<double> sil(k,0.0);
    for(int c=0;c<k;c++){
        if(g[c].empty()) { sil[c]=0; continue; }
        double sum=0;
        for(int id: g[c]){
            double a=1.0;
            if(g[c].size()>1){
                a=0;
                for(int j: g[c]) if(j!=id) a += dist(x[id],y[id],x[j],y[j]);
                a /= (int)g[c].size()-1;
            }
            double b = numeric_limits<double>::infinity();
            for(int c2=0;c2<k;c2++){
                if(c2==c || g[c2].empty()) continue;
                double t=0;
                for(int j: g[c2]) t += dist(x[id],y[id],x[j],y[j]);
                b = min(b, t/(int)g[c2].size());
            }
            double s = isinf(b)?0:((max(a,b)==0)?0:(b-a)/max(a,b));
            sum += s;
        }
        sil[c] = sum / (int)g[c].size();
    }

    int bad=0;
    for(int c=1;c<k;c++) if(sil[c]<sil[bad]) bad=c;
    cout<<r2(cx[bad])<<","<<r2(cy[bad])<<"\n";
    return 0;
}
```
