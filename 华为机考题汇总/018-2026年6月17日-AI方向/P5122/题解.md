## 解题思路

先对每组长度为 $L$ 的信号做 $DFT$，只考虑非冗余正频率，即 $k=1$ 到 $k=(L-1)/2$。
计算每个频率的幅值，因为比较大小时平方根不影响顺序，所以直接比较 $Re^2+Im^2$。

对每组信号选出幅值最大的 $3$ 个频率下标，幅值相同则频率下标 $k$ 小的优先，得到三维特征向量。

然后使用 $K$-$Means$ 聚类：

初始中心由输入的 $K$ 个原始信号下标确定，即使用这些信号的特征向量作为初始中心。每轮把所有样本分配给距离最近的中心，再用簇内特征向量的平均值更新中心。平均值需要使用银行家舍入，也就是四舍六入五成双。若某个簇为空，则该中心保持不变。最多迭代 $100$ 次，或者所有中心不再变化时停止。

最后将 $K$ 个中心按字典序升序输出。

## 复杂度分析

设共有 $N$ 组信号，每组长度为 $L$，聚类数为 $K$。

$DFT$ 部分需要枚举每组信号、每个频率和每个采样点，时间复杂度为 $O(NL^2)$。

$K$-$Means$ 最多迭代 $100$ 次，每次计算 $N$ 个样本到 $K$ 个中心的距离，时间复杂度为 $O(100NK)$。

总时间复杂度为 $O(NL^2+100NK)$。

需要保存信号、特征和中心，空间复杂度为 $O(NL+N)$。

```python
import sys
import math
from functools import cmp_to_key

EPS = 1e-9


def cmp_freq(a, b):
    # 按幅值平方降序，若相等则按频率下标升序
    if abs(a[1] - b[1]) > EPS:
        return -1 if a[1] > b[1] else 1
    return a[0] - b[0]


def round_half_even(total, count):
    # 银行家舍入：对 total / count 四舍六入五成双
    q, r = divmod(total, count)
    if r * 2 < count:
        return q
    if r * 2 > count:
        return q + 1
    return q if q % 2 == 0 else q + 1


def get_features(data, n, l):
    features = []
    max_k = (l - 1) // 2

    for arr in data:
        vals = []
        for k in range(1, max_k + 1):
            re = 0.0
            im = 0.0

            # 直接计算 DFT 的实部和虚部
            for i in range(l):
                angle = 2.0 * math.pi * k * i / l
                re += arr[i] * math.cos(angle)
                im -= arr[i] * math.sin(angle)

            mag2 = re * re + im * im
            vals.append((k, mag2))

        vals.sort(key=cmp_to_key(cmp_freq))

        # 取幅值最大的 3 个频率下标作为特征向量
        features.append([vals[0][0], vals[1][0], vals[2][0]])

    return features


def solve(data, n, l, k, init_index):
    features = get_features(data, n, l)

    # 初始化聚类中心
    centers = [features[idx][:] for idx in init_index]

    for _ in range(100):
        belong = [0] * n

        # 分配每个样本到最近的聚类中心
        for i in range(n):
            best_id = 0
            best_dist = 10 ** 18

            for j in range(k):
                dist = 0
                for t in range(3):
                    diff = features[i][t] - centers[j][t]
                    dist += diff * diff

                if dist < best_dist:
                    best_dist = dist
                    best_id = j

            belong[i] = best_id

        sums = [[0, 0, 0] for _ in range(k)]
        cnt = [0] * k

        # 统计每个簇的特征和
        for i in range(n):
            cid = belong[i]
            cnt[cid] += 1
            for t in range(3):
                sums[cid][t] += features[i][t]

        new_centers = [c[:] for c in centers]

        # 更新聚类中心，空簇保持不变
        for j in range(k):
            if cnt[j] == 0:
                continue
            for t in range(3):
                new_centers[j][t] = round_half_even(sums[j][t], cnt[j])

        if new_centers == centers:
            break

        centers = new_centers

    centers.sort()
    return centers


def main():
    tokens = sys.stdin.read().strip().split()
    if not tokens:
        return

    p = 0
    n = int(tokens[p])
    p += 1
    l = int(tokens[p])
    p += 1
    k = int(tokens[p])
    p += 1

    data = []
    for _ in range(n):
        arr = []
        for _ in range(l):
            arr.append(float(tokens[p]))
            p += 1
        data.append(arr)

    init_index = []
    for _ in range(k):
        init_index.append(int(tokens[p]))
        p += 1

    ans = solve(data, n, l, k, init_index)

    out = []
    for row in ans:
        out.append("{} {} {}".format(row[0], row[1], row[2]))

    print("\n".join(out))


if __name__ == "__main__":
    main()
```

```java
import java.util.*;

public class Main {
    static final double EPS = 1e-9;

    static int roundHalfEven(int total, int count) {
        // 银行家舍入：对 total / count 四舍六入五成双
        int q = total / count;
        int r = total % count;

        if (r * 2 < count) {
            return q;
        }
        if (r * 2 > count) {
            return q + 1;
        }
        return q % 2 == 0 ? q : q + 1;
    }

    static int[][] getFeatures(double[][] data, int n, int l) {
        int[][] features = new int[n][3];
        int maxK = (l - 1) / 2;

        for (int i = 0; i < n; i++) {
            double[][] vals = new double[maxK][2];

            for (int k = 1; k <= maxK; k++) {
                double re = 0.0;
                double im = 0.0;

                // 直接计算 DFT 的实部和虚部
                for (int t = 0; t < l; t++) {
                    double angle = 2.0 * Math.PI * k * t / l;
                    re += data[i][t] * Math.cos(angle);
                    im -= data[i][t] * Math.sin(angle);
                }

                double mag2 = re * re + im * im;
                vals[k - 1][0] = k;
                vals[k - 1][1] = mag2;
            }

            // 按幅值平方降序，若相等则按频率下标升序
            Arrays.sort(vals, new Comparator<double[]>() {
                public int compare(double[] a, double[] b) {
                    if (Math.abs(a[1] - b[1]) > EPS) {
                        return a[1] > b[1] ? -1 : 1;
                    }
                    return (int)a[0] - (int)b[0];
                }
            });

            // 取幅值最大的 3 个频率下标作为特征向量
            for (int t = 0; t < 3; t++) {
                features[i][t] = (int)vals[t][0];
            }
        }

        return features;
    }

    static int[][] solve(double[][] data, int n, int l, int k, int[] initIndex) {
        int[][] features = getFeatures(data, n, l);
        int[][] centers = new int[k][3];

        // 初始化聚类中心
        for (int i = 0; i < k; i++) {
            for (int t = 0; t < 3; t++) {
                centers[i][t] = features[initIndex[i]][t];
            }
        }

        for (int iter = 0; iter < 100; iter++) {
            int[] belong = new int[n];

            // 分配每个样本到最近的聚类中心
            for (int i = 0; i < n; i++) {
                int bestId = 0;
                int bestDist = Integer.MAX_VALUE;

                for (int j = 0; j < k; j++) {
                    int dist = 0;
                    for (int t = 0; t < 3; t++) {
                        int diff = features[i][t] - centers[j][t];
                        dist += diff * diff;
                    }

                    if (dist < bestDist) {
                        bestDist = dist;
                        bestId = j;
                    }
                }

                belong[i] = bestId;
            }

            int[][] sums = new int[k][3];
            int[] cnt = new int[k];

            // 统计每个簇的特征和
            for (int i = 0; i < n; i++) {
                int cid = belong[i];
                cnt[cid]++;
                for (int t = 0; t < 3; t++) {
                    sums[cid][t] += features[i][t];
                }
            }

            int[][] newCenters = new int[k][3];
            for (int i = 0; i < k; i++) {
                for (int t = 0; t < 3; t++) {
                    newCenters[i][t] = centers[i][t];
                }
            }

            // 更新聚类中心，空簇保持不变
            for (int i = 0; i < k; i++) {
                if (cnt[i] == 0) {
                    continue;
                }
                for (int t = 0; t < 3; t++) {
                    newCenters[i][t] = roundHalfEven(sums[i][t], cnt[i]);
                }
            }

            boolean same = true;
            for (int i = 0; i < k; i++) {
                for (int t = 0; t < 3; t++) {
                    if (newCenters[i][t] != centers[i][t]) {
                        same = false;
                    }
                }
            }

            centers = newCenters;

            if (same) {
                break;
            }
        }

        // 最终按字典序排序
        Arrays.sort(centers, new Comparator<int[]>() {
            public int compare(int[] a, int[] b) {
                for (int i = 0; i < 3; i++) {
                    if (a[i] != b[i]) {
                        return a[i] - b[i];
                    }
                }
                return 0;
            }
        });

        return centers;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        if (!sc.hasNext()) {
            return;
        }

        int n = sc.nextInt();
        int l = sc.nextInt();
        int k = sc.nextInt();

        double[][] data = new double[n][l];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < l; j++) {
                data[i][j] = sc.nextDouble();
            }
        }

        int[] initIndex = new int[k];
        for (int i = 0; i < k; i++) {
            initIndex[i] = sc.nextInt();
        }

        int[][] ans = solve(data, n, l, k, initIndex);

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < k; i++) {
            sb.append(ans[i][0]).append(" ")
              .append(ans[i][1]).append(" ")
              .append(ans[i][2]);
            if (i + 1 < k) {
                sb.append('\n');
            }
        }

        System.out.print(sb.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

const double EPS = 1e-9;

bool cmpFreq(const pair<int, double>& a, const pair<int, double>& b) {
    // 按幅值平方降序，若相等则按频率下标升序
    if (fabs(a.second - b.second) > EPS) {
        return a.second > b.second;
    }
    return a.first < b.first;
}

int roundHalfEven(int total, int count) {
    // 银行家舍入：对 total / count 四舍六入五成双
    int q = total / count;
    int r = total % count;

    if (r * 2 < count) {
        return q;
    }
    if (r * 2 > count) {
        return q + 1;
    }
    return q % 2 == 0 ? q : q + 1;
}

vector<vector<int>> getFeatures(const vector<vector<double>>& data, int n, int l) {
    vector<vector<int>> features(n, vector<int>(3));
    int maxK = (l - 1) / 2;

    for (int i = 0; i < n; i++) {
        vector<pair<int, double>> vals;

        for (int k = 1; k <= maxK; k++) {
            double re = 0.0;
            double im = 0.0;

            // 直接计算 DFT 的实部和虚部
            for (int t = 0; t < l; t++) {
                double angle = 2.0 * acos(-1.0) * k * t / l;
                re += data[i][t] * cos(angle);
                im -= data[i][t] * sin(angle);
            }

            double mag2 = re * re + im * im;
            vals.push_back({k, mag2});
        }

        sort(vals.begin(), vals.end(), cmpFreq);

        // 取幅值最大的 3 个频率下标作为特征向量
        for (int t = 0; t < 3; t++) {
            features[i][t] = vals[t].first;
        }
    }

    return features;
}

vector<vector<int>> solve(const vector<vector<double>>& data, int n, int l, int k, const vector<int>& initIndex) {
    vector<vector<int>> features = getFeatures(data, n, l);
    vector<vector<int>> centers(k, vector<int>(3));

    // 初始化聚类中心
    for (int i = 0; i < k; i++) {
        centers[i] = features[initIndex[i]];
    }

    for (int iter = 0; iter < 100; iter++) {
        vector<int> belong(n);

        // 分配每个样本到最近的聚类中心
        for (int i = 0; i < n; i++) {
            int bestId = 0;
            int bestDist = INT_MAX;

            for (int j = 0; j < k; j++) {
                int dist = 0;
                for (int t = 0; t < 3; t++) {
                    int diff = features[i][t] - centers[j][t];
                    dist += diff * diff;
                }

                if (dist < bestDist) {
                    bestDist = dist;
                    bestId = j;
                }
            }

            belong[i] = bestId;
        }

        vector<vector<int>> sums(k, vector<int>(3, 0));
        vector<int> cnt(k, 0);

        // 统计每个簇的特征和
        for (int i = 0; i < n; i++) {
            int cid = belong[i];
            cnt[cid]++;
            for (int t = 0; t < 3; t++) {
                sums[cid][t] += features[i][t];
            }
        }

        vector<vector<int>> newCenters = centers;

        // 更新聚类中心，空簇保持不变
        for (int i = 0; i < k; i++) {
            if (cnt[i] == 0) {
                continue;
            }
            for (int t = 0; t < 3; t++) {
                newCenters[i][t] = roundHalfEven(sums[i][t], cnt[i]);
            }
        }

        if (newCenters == centers) {
            break;
        }

        centers = newCenters;
    }

    // 最终按字典序排序
    sort(centers.begin(), centers.end());
    return centers;
}
```
