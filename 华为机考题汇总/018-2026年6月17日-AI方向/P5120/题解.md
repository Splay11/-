## 解题思路

设模型总层数为 $n$，单卡最大承载层数为 $c$，$micro\ batch$ 数为 $m$，硬件总卡数为 $h$。

可行的 $PP$ 需要满足：

* $PP \le h$
* $n % PP = 0$
* $\frac{n}{PP} \le c$

由第三个条件可得：$PP \ge \lceil \frac{n}{c} \rceil$。

气泡率公式为：

$$
\frac{PP - 1}{m + PP - 1}
$$

因为 $m$ 为正整数，所以气泡率随 $PP$ 增大而增大。因此只需要找到最小的可行 $PP$，它就是最优答案。

使用约数枚举算法，枚举 $n$ 的所有约数，找出满足范围 $[\lceil \frac{n}{c} \rceil, h]$ 的最小约数即可。

若输入参数个数不是 $4$ 个，或不存在可行 $PP$，输出 $-1\ -1$。

## 复杂度分析

时间复杂度为 $O(\sqrt n)$，因为只需要枚举模型总层数 $n$ 的约数。

空间复杂度为 $O(1)$，只使用常数个变量。

## 代码实现

### Python

```python
import sys

# 计算最优 PP 和气泡率
def get_best_pp(layers, max_layer, micro_batch, card_count):
    low = (layers + max_layer - 1) // max_layer
    best = None

    d = 1
    # 枚举总层数的所有约数
    while d * d <= layers:
        if layers % d == 0:
            # 检查约数 d
            if low <= d <= card_count:
                if best is None or d < best:
                    best = d

            # 检查另一个约数 layers // d
            other = layers // d
            if low <= other <= card_count:
                if best is None or other < best:
                    best = other

        d += 1

    if best is None:
        return -1, -1

    # 根据公式计算气泡率
    bubble = (best - 1) / (micro_batch + best - 1)
    return best, bubble


def main():
    data = sys.stdin.read().split()

    # 参数个数不为 4，直接输出 -1 -1
    if len(data) != 4:
        print("-1 -1")
        return

    try:
        nums = list(map(int, data))
    except:
        print("-1 -1")
        return

    # 参数必须为正整数
    for x in nums:
        if x <= 0:
            print("-1 -1")
            return

    layers, max_layer, micro_batch, card_count = nums
    pp, bubble = get_best_pp(layers, max_layer, micro_batch, card_count)

    if pp == -1:
        print("-1 -1")
    else:
        print(f"{pp} {bubble:.4f}")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static class Result {
        long pp;
        double bubble;

        Result(long pp, double bubble) {
            this.pp = pp;
            this.bubble = bubble;
        }
    }

    // 计算最优 PP 和气泡率
    static Result getBestPP(long layers, long maxLayer, long microBatch, long cardCount) {
        long low = (layers + maxLayer - 1) / maxLayer;
        long best = Long.MAX_VALUE;

        // 枚举总层数的所有约数
        for (long d = 1; d <= layers / d; d++) {
            if (layers % d == 0) {
                // 检查约数 d
                if (d >= low && d <= cardCount && d < best) {
                    best = d;
                }

                // 检查另一个约数 layers / d
                long other = layers / d;
                if (other >= low && other <= cardCount && other < best) {
                    best = other;
                }
            }
        }

        if (best == Long.MAX_VALUE) {
            return new Result(-1, -1);
        }

        // 根据公式计算气泡率
        double bubble = (double)(best - 1) / ((double)microBatch + best - 1);
        return new Result(best, bubble);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        ArrayList<String> data = new ArrayList<>();

        while (sc.hasNext()) {
            data.add(sc.next());
        }

        // 参数个数不为 4，直接输出 -1 -1
        if (data.size() != 4) {
            System.out.println("-1 -1");
            return;
        }

        long[] nums = new long[4];

        try {
            for (int i = 0; i < 4; i++) {
                nums[i] = Long.parseLong(data.get(i));

                // 参数必须为正整数
                if (nums[i] <= 0) {
                    System.out.println("-1 -1");
                    return;
                }
            }
        } catch (Exception e) {
            System.out.println("-1 -1");
            return;
        }

        long layers = nums[0];
        long maxLayer = nums[1];
        long microBatch = nums[2];
        long cardCount = nums[3];

        Result ans = getBestPP(layers, maxLayer, microBatch, cardCount);

        if (ans.pp == -1) {
            System.out.println("-1 -1");
        } else {
            System.out.printf(Locale.US, "%d %.4f\n", ans.pp, ans.bubble);
        }
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

// 判断字符串是否为正整数，并转换为 long long
bool parsePositiveInteger(const string& s, long long& x) {
    if (s.empty()) return false;

    x = 0;
    for (char ch : s) {
        if (!isdigit(ch)) return false;

        int digit = ch - '0';

        // 防止整数溢出
        if (x > (LLONG_MAX - digit) / 10) return false;

        x = x * 10 + digit;
    }

    return x > 0;
}

// 计算最优 PP 和气泡率

                                  long long microBatch, long long cardCount) {

            if (other >= low && other <= cardCount && other < best) {

}
```
