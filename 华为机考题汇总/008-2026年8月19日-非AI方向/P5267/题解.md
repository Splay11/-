## 解题思路


设原数组共有 $n$ 个消息，总权重和为 $S$。

假设其中一个端口分到 $k$ 个消息，这 $k$ 个消息的权重和为 $x$，那么另一个端口有 $n-k$ 个消息，权重和为 $S-x$。

要求两个端口处理消息的权重均值相同，因此有：

$$
\frac{x}{k}=\frac{S-x}{n-k}
$$

交叉相乘：

$$
x(n-k)=k(S-x)
$$

整理得到：

$$
xn=kS
$$

因此，对于某个分组大小 $k$，它的元素和必须恰好为：

$$
x=\frac{kS}{n}
$$

于是问题转化为：

在数组中，是否能够选择 $1\le k<n$ 个元素，使这些元素的和恰好为 $\frac{kS}{n}$。

可以使用二维动态规划：

`dp[k][s]` 表示是否能够从已经处理的元素中选出恰好 $k$ 个，使权重和为 $s$。

初始状态：

$$
dp[0][0]=true
$$

遍历每个权重 $v$ 时，倒序枚举元素个数 $k$ 和权重和 $s$，进行转移：

$$
dp[k][s]=dp[k][s]\lor dp[k-1][s-v]
$$

因为题目中的权重只有 `0、5、15、20`，全部都是 $5$ 的倍数，可以先全部除以 $5$，变成：

`0、1、3、4`

这样最大权重和由 $20n$ 降低为 $4n$，可以明显缩小动态规划状态范围。

DP 完成后枚举 $k=1,2,\dots,n-1$：

1. 如果 $kS$ 不能被 $n$ 整除，则该 $k$ 不可能满足要求。
2. 否则令 $x=\frac{kS}{n}$。
3. 如果 `dp[k][x]` 为真，则存在合法划分。
4. 两组元素和分别为 $x$ 和 $S-x$，输出其中较小的一个即可。

整个过程只进行整数运算，不需要计算浮点平均值，因此不存在精度问题。

需要注意，以上算法严格按照题目中“两个分组的算术平均值相等”这一条件实现。题面给出的样例 2、样例 3 与该数学定义存在明显矛盾：例如样例 3 中 `[15]` 的均值为 $15$，而 `[20,20]` 的均值为 $20$，两者并不相等。若原题确实要求算术平均值相等，则应以公式 $xn=kS$ 为准。

* 实现方法

1. 将所有权重除以 $5$，缩小状态空间。
2. 计算缩放后的总权重和 `sum`。
3. 建立 `dp[n + 1][sum + 1]`。
4. 对每个消息权重进行 0-1 背包更新：

   * 元素数量必须倒序枚举，防止一个消息被重复选取。
   * 权重和同样倒序枚举。
5. 枚举一个端口可能取得的消息数量 `k`：

   * 判断 `k * sum` 是否可以整除 `n`。
   * 得到目标和 `target`。
   * 判断 `dp[k][target]` 是否存在。
6. 若存在，返回两个分组中较小的权重和，并乘回 $5$。
7. 如果所有 `k` 都无法满足，则返回 `0`。

## 复杂度分析

设缩放后的总权重和为 $T$。

由于每个权重除以 $5$ 后最大为 $4$，因此：

$$
T\le 4n<400
$$

动态规划需要依次处理 $n$ 个元素，每次最多枚举 $n$ 个元素数量以及 $T$ 个权重和，因此时间复杂度为：

$$
O(n^2T)
$$

因为 $T\le4n$，最坏情况下也可以写为：

$$
O(n^3)
$$

在 $n<100$ 的数据范围下，最多只有约数百万次状态转移，可以满足时间限制。

二维 DP 数组大小为：

$$
O(nT)
$$

由于 $T<400$，空间消耗很小，可以满足内存限制。

## 代码实现

### Python

```python
def solve(arr):
    n = len(arr)

    # 所有权重都是 5 的倍数，先除以 5 缩小动态规划范围
    a = [x // 5 for x in arr]
    total = sum(a)

    # dp[k][s] 表示能否选出恰好 k 个元素，使元素和为 s
    dp = [[False] * (total + 1) for _ in range(n + 1)]
    dp[0][0] = True

    cnt = 0
    cur = 0

    for v in a:
        cnt += 1
        cur += v

        # 倒序枚举元素数量，保证每个消息只能选择一次
        for k in range(cnt, 0, -1):
            # 倒序枚举当前可能达到的权重和
            for s in range(cur, v - 1, -1):
                if dp[k - 1][s - v]:
                    dp[k][s] = True

    # 枚举其中一个端口处理的消息数量
    for k in range(1, n):
        # target = k * total / n 必须是整数
        if k * total % n != 0:
            continue

        target = k * total // n

        if dp[k][target]:
            # 两组元素和分别为 target 和 total - target
            ans = min(target, total - target) * 5
            return 1, ans

    return 0, 0


if __name__ == "__main__":
    arr = list(map(int, input().split()))

    ok, ans = solve(arr)

    print(ok)
    if ok == 1:
        print(ans)
```

### Java

```java
import java.util.*;

public class Main {

    public static int[] solve(int[] arr) {
        int n = arr.length;
        int[] a = new int[n];
        int total = 0;

        // 所有权重都是 5 的倍数，先除以 5 缩小动态规划范围
        for (int i = 0; i < n; i++) {
            a[i] = arr[i] / 5;
            total += a[i];
        }

        // dp[k][s] 表示能否选出恰好 k 个元素，使元素和为 s
        boolean[][] dp = new boolean[n + 1][total + 1];
        dp[0][0] = true;

        int cnt = 0;
        int cur = 0;

        for (int v : a) {
            cnt++;
            cur += v;

            // 倒序枚举元素数量，避免同一个消息被重复选择
            for (int k = cnt; k >= 1; k--) {
                // 倒序枚举权重和
                for (int s = cur; s >= v; s--) {
                    if (dp[k - 1][s - v]) {
                        dp[k][s] = true;
                    }
                }
            }
        }

        // 枚举其中一个端口分到的消息数量
        for (int k = 1; k < n; k++) {
            // target = k * total / n 必须为整数
            if (k * total % n != 0) {
                continue;
            }

            int target = k * total / n;

            if (dp[k][target]) {
                // 输出两个分组中较小的元素和，并恢复原来的权重
                int ans = Math.min(target, total - target) * 5;
                return new int[]{1, ans};
            }
        }

        return new int[]{0, 0};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        ArrayList<Integer> list = new ArrayList<>();

        while (sc.hasNextInt()) {
            list.add(sc.nextInt());
        }

        int[] arr = new int[list.size()];
        for (int i = 0; i < list.size(); i++) {
            arr[i] = list.get(i);
        }

        int[] res = solve(arr);

        System.out.println(res[0]);
        if (res[0] == 1) {
            System.out.println(res[1]);
        }

        sc.close();
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

pair<bool, int> solve(const vector<int>& arr) {
    int n = arr.size();
    vector<int> a(n);
    int total = 0;

    // 所有权重都是 5 的倍数，先除以 5 缩小动态规划范围
    for (int i = 0; i < n; i++) {
        a[i] = arr[i] / 5;
        total += a[i];
    }

    // dp[k][s] 表示能否选出恰好 k 个元素，使元素和为 s
    vector<vector<bool>> dp(n + 1, vector<bool>(total + 1, false));
    dp[0][0] = true;

    int cnt = 0;
    int cur = 0;

    for (int v : a) {
        cnt++;
        cur += v;

        // 倒序枚举元素数量，避免同一个消息被重复选择
        for (int k = cnt; k >= 1; k--) {
            // 倒序枚举权重和
            for (int s = cur; s >= v; s--) {
                if (dp[k - 1][s - v]) {
                    dp[k][s] = true;
                }
            }
        }
    }

    // 枚举其中一个端口处理的消息数量
    for (int k = 1; k < n; k++) {
        // target = k * total / n 必须为整数
        if (k * total % n != 0) {
            continue;
        }

        int target = k * total / n;

        if (dp[k][target]) {
            // 输出两个分组中较小的元素和，并恢复原来的权重
            int ans = min(target, total - target) * 5;
            return {true, ans};
        }
    }

    return {false, 0};
}

int main() {
    vector<int> arr;
    int x;

    while (cin >> x) {
        arr.push_back(x);
    }

    pair<bool, int> res = solve(arr);

    if (!res.first) {
        cout << 0 << endl;
    } else {
        cout << 1 << endl;
        cout << res.second << endl;
    }

    return 0;
}
```
