## 解题思路

机器人到达同一个位置时，剩余生命值可能不同，而剩余生命值会直接影响后续能否继续经过陷阱。因此，不能只记录机器人当前所在的坐标，还需要把当前生命值一起作为状态。

定义状态：

$$
(x,y,h)
$$

其中：

* $x,y$ 表示机器人当前所在位置；
* $h$ 表示机器人当前剩余生命值，满足 $1\le h\le k$。

因为每次移动的步数都增加 $1$，所有边的代价完全相同，所以可以使用 BFS 求最少步数。

从起点开始，初始状态为：

$$
(s_x,s_y,k)
$$

对于当前状态 $(x,y,h)$，枚举上下左右四个相邻位置 $(nx,ny)$：

1. 如果越界或者目标位置是墙壁 `1`，不能移动。
2. 如果目标位置是普通空地 `0`、起点 `3` 或出口 `4`，生命值不变。
3. 如果目标位置是陷阱 `2`，生命值减少 $1$：
   $$
   nh=h-1
   $$
   如果 $nh=0$，机器人损毁，该状态不能加入 BFS。
4. 如果目标位置是生命药水 `5`，生命值直接恢复为满生命：
   $$
   nh=k
   $$

使用三维数组 `vis[x][y][h]` 记录状态是否已经访问。

需要特别注意：即使两个状态处于同一个格子，只要剩余生命值不同，就不能简单地认为它们是同一个状态。例如，剩余 $1$ 点生命和剩余 $3$ 点生命到达同一个位置，后者可能还能继续经过陷阱，而前者可能无法通过。因此必须分别记录。

BFS 按照步数从小到大扩展状态，所以第一次到达出口时，对应的步数一定是最少步数，可以直接返回。

如果 BFS 结束后仍然没有到达出口，则返回 `-1`。

## 复杂度分析

迷宫共有 $m\times n$ 个位置，每个位置最多对应 $k$ 种有效生命值，因此状态数量最多为：

$$
m\times n\times k
$$

每个状态最多向四个方向扩展一次。

时间复杂度为：

$$
O(mnk)
$$

空间复杂度主要来自访问数组和 BFS 队列：

$$
O(mnk)
$$

由于 $m,n\le 50,\ k\le 20$，最多只有：

$$
50\times50\times20=50000
$$

个状态，可以轻松满足要求。

## 代码实现

### Python

```python
from collections import deque


def escape(a, m, n, k):
    sx = sy = -1

    # 找到唯一的起点
    for i in range(m):
        for j in range(n):
            if a[i][j] == 3:
                sx, sy = i, j

    # vis[x][y][h] 表示到达 (x, y) 且剩余 h 点生命的状态是否访问过
    vis = [[[False] * (k + 1) for _ in range(n)] for _ in range(m)]

    # 队列元素依次表示：行、列、剩余生命值、已经走的步数
    q = deque()
    q.append((sx, sy, k, 0))
    vis[sx][sy][k] = True

    dx = [-1, 1, 0, 0]
    dy = [0, 0, -1, 1]

    while q:
        x, y, hp, step = q.popleft()

        # BFS 第一次到达出口时，当前步数一定最少
        if a[x][y] == 4:
            return step

        for d in range(4):
            nx = x + dx[d]
            ny = y + dy[d]

            # 越界不能移动
            if nx < 0 or nx >= m or ny < 0 or ny >= n:
                continue

            # 墙壁不能通过
            if a[nx][ny] == 1:
                continue

            nh = hp

            # 进入陷阱后损失 1 点生命
            if a[nx][ny] == 2:
                nh -= 1
                if nh == 0:
                    continue

            # 进入生命药水格后恢复满生命
            elif a[nx][ny] == 5:
                nh = k

            # 同一个位置、同一种剩余生命值只需要处理一次
            if not vis[nx][ny][nh]:
                vis[nx][ny][nh] = True
                q.append((nx, ny, nh, step + 1))

    return -1


def main():
    m, n, k = map(int, input().split())

    a = []
    for _ in range(m):
        row = input().split()

        if len(row) == 1 and len(row[0]) == n:
            a.append([int(c) for c in row[0]])
        else:
            a.append(list(map(int, row)))

    ans = escape(a, m, n, k)
    print(ans)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    static int escape(int[][] a, int m, int n, int k) {
        int sx = -1, sy = -1;

        // 找到唯一的起点
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (a[i][j] == 3) {
                    sx = i;
                    sy = j;
                }
            }
        }

        // vis[x][y][h] 表示到达 (x, y) 且剩余 h 点生命的状态是否访问过
        boolean[][][] vis = new boolean[m][n][k + 1];

        // 队列中的数组依次表示：行、列、剩余生命值、已经走的步数
        Queue<int[]> q = new LinkedList<>();
        q.offer(new int[]{sx, sy, k, 0});
        vis[sx][sy][k] = true;

        int[] dx = {-1, 1, 0, 0};
        int[] dy = {0, 0, -1, 1};

        while (!q.isEmpty()) {
            int[] cur = q.poll();

            int x = cur[0];
            int y = cur[1];
            int hp = cur[2];
            int step = cur[3];

            // BFS 第一次到达出口时，当前步数一定最少
            if (a[x][y] == 4) {
                return step;
            }

            for (int d = 0; d < 4; d++) {
                int nx = x + dx[d];
                int ny = y + dy[d];

                // 越界不能移动
                if (nx < 0 || nx >= m || ny < 0 || ny >= n) {
                    continue;
                }

                // 墙壁不能通过
                if (a[nx][ny] == 1) {
                    continue;
                }

                int nh = hp;

                // 进入陷阱后损失 1 点生命
                if (a[nx][ny] == 2) {
                    nh--;
                    if (nh == 0) {
                        continue;
                    }
                }
                // 进入生命药水格后恢复满生命
                else if (a[nx][ny] == 5) {
                    nh = k;
                }

                // 同一个位置、同一种剩余生命值只需要处理一次
                if (!vis[nx][ny][nh]) {
                    vis[nx][ny][nh] = true;
                    q.offer(new int[]{nx, ny, nh, step + 1});
                }
            }
        }

        return -1;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int m = sc.nextInt();
        int n = sc.nextInt();
        int k = sc.nextInt();

        int[][] a = new int[m][n];

        for (int i = 0; i < m; i++) {
            String s = sc.next();

            if (s.length() == n) {
                for (int j = 0; j < n; j++) {
                    a[i][j] = s.charAt(j) - '0';
                }
            } else {
                // 如果网格以空格分隔，则第一个数字已经读取
                a[i][0] = Integer.parseInt(s);
                for (int j = 1; j < n; j++) {
                    a[i][j] = sc.nextInt();
                }
            }
        }

        int ans = escape(a, m, n, k);
        System.out.println(ans);

        sc.close();
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <string>
using namespace std;

// BFS 状态
struct Node {
    int x;
    int y;
    int hp;
    int step;
};

int escapeMaze(vector<vector<int>>& a, int m, int n, int k) {
    int sx = -1, sy = -1;

    // 找到唯一的起点
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (a[i][j] == 3) {
                sx = i;
                sy = j;
            }
        }
    }

    // vis[x][y][h] 表示到达 (x, y) 且剩余 h 点生命的状态是否访问过
    vector<vector<vector<bool>>> vis(
        m, vector<vector<bool>>(n, vector<bool>(k + 1, false))
    );

    queue<Node> q;
    q.push({sx, sy, k, 0});
    vis[sx][sy][k] = true;

    int dx[4] = {-1, 1, 0, 0};
    int dy[4] = {0, 0, -1, 1};

    while (!q.empty()) {
        Node cur = q.front();
        q.pop();

        // BFS 第一次到达出口时，当前步数一定最少
        if (a[cur.x][cur.y] == 4) {
            return cur.step;
        }

        for (int d = 0; d < 4; d++) {
            int nx = cur.x + dx[d];
            int ny = cur.y + dy[d];

            // 越界不能移动
            if (nx < 0 || nx >= m || ny < 0 || ny >= n) {
                continue;
            }

            // 墙壁不能通过
            if (a[nx][ny] == 1) {
                continue;
            }

            int nh = cur.hp;

            // 进入陷阱后损失 1 点生命
            if (a[nx][ny] == 2) {
                nh--;

                if (nh == 0) {
                    continue;
                }
            }
            // 进入生命药水格后恢复满生命
            else if (a[nx][ny] == 5) {
                nh = k;
            }

            // 同一个位置、同一种剩余生命值只需要处理一次
            if (!vis[nx][ny][nh]) {
                vis[nx][ny][nh] = true;
                q.push({nx, ny, nh, cur.step + 1});
            }
        }
    }

    return -1;
}

int main() {
    int m, n, k;
    cin >> m >> n >> k;

    vector<vector<int>> a(m, vector<int>(n));

    for (int i = 0; i < m; i++) {
        string s;
        cin >> s;

        if ((int)s.size() == n) {
            for (int j = 0; j < n; j++) {
                a[i][j] = s[j] - '0';
            }
        } else {
            // 如果网格以空格分隔，则第一个数字已经读取
            a[i][0] = stoi(s);

            for (int j = 1; j < n; j++) {
                cin >> a[i][j];
            }
        }
    }

    int ans = escapeMaze(a, m, n, k);
    cout << ans << '\n';

    return 0;
}
```
