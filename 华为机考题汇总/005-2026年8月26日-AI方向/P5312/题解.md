## 解题思路

本题没有复杂的模型训练过程，只需要严格按照题目给出的公式完成一次前向计算，整体采用模拟算法和矩阵乘法。

首先处理 $S$ 层双向循环网络。对于每一层，正向状态按照 $1\rightarrow N$ 的顺序递推，反向状态按照 $N\rightarrow1$ 的顺序递推。每个位置分别计算输入向量与输入权重的乘积、上一状态与循环权重的乘积，再加上偏置并对 $MOD$ 取模，最后将正向和反向状态拼接，作为下一层的输入。

完成 $S$ 层编码后得到每个位置的 $2R$ 维向量 $e_t$。

接着计算 $A$ 个注意力头。对于每个头，先由 $e_t$ 分别计算查询向量、键向量和值向量，再枚举位置 $t$ 和 $s$，通过点积得到注意力分数，并使用该分数对所有值向量进行加权求和。题目明确不需要使用缩放、$softmax$ 或 $mask$，因此直接按照公式计算并取模即可。

将 $A$ 个注意力头的结果拼接后，通过投影矩阵映射到 $2R$ 维，再与编码结果 $e_t$ 相加完成残差连接，最后通过输出层得到答案。

实现时不需要保存完整的 $N\times N$ 注意力分数矩阵，可以在计算每个 $(t,s)$ 的注意力分数后立即累加到对应输出中，从而减少空间使用。

## 复杂度分析

双向循环网络中，每层每个位置需要进行输入矩阵乘法和状态矩阵乘法，复杂度可以写为：

$O(NSR(\max(F,2R)+R))$

多头注意力中，查询、键和值的计算复杂度为：

$O(ANRQ)$

计算所有位置之间的注意力分数以及加权求和的复杂度为：

$O(AN^2Q)$

投影层和输出层的复杂度分别为：

$O(NAQR)$ 和 $O(NRC)$

因此总时间复杂度为：

$O(NSR(\max(F,2R)+R)+ANQ(N+R)+NRC)$

由于题目中各维度都很小，该复杂度可以轻松通过。

不保存完整注意力矩阵时，主要保存循环网络状态、注意力向量以及拼接结果，因此额外空间复杂度为：

$O(N(R+AQ+C))$

def solve(N, F, R, C, S, A, Q, MOD, x, layers, heads, Wp, cp, Wy, cy):
    # 行向量乘矩阵
    def mul(v, mat):
        cols = len(mat[0])
        res = [0] * cols

        for j in range(cols):
            s = 0
            for i in range(len(v)):
                s += v[i] * mat[i][j]
            res[j] = s % MOD

        return res

    cur = x

    # 依次处理每一层双向 RNN
    for Uf, Vf, bf, Ub, Vb, bb, p0, rn in layers:
        f = [[0] * R for _ in range(N)]
        g = [[0] * R for _ in range(N)]

        # 正向递推
        prev = p0[:]
        for t in range(N):
            a = mul(cur[t], Uf)
            b = mul(prev, Vf)

            for j in range(R):
                f[t][j] = (a[j] + b[j] + bf[j]) % MOD

            prev = f[t]

        # 反向递推
        nxt = rn[:]
        for t in range(N - 1, -1, -1):
            a = mul(cur[t], Ub)
            b = mul(nxt, Vb)

            for j in range(R):
                g[t][j] = (a[j] + b[j] + bb[j]) % MOD

            nxt = g[t]

        # 拼接正向和反向状态
        cur = [f[t] + g[t] for t in range(N)]

    e = cur

    # 保存所有注意力头拼接后的结果
    att = [[0] * (A * Q) for _ in range(N)]

    for h in range(A):
        Qm, Km, Vm = heads[h]

        q = [mul(e[t], Qm) for t in range(N)]
        k = [mul(e[t], Km) for t in range(N)]
        v = [mul(e[t], Vm) for t in range(N)]

        # 直接计算注意力分数并累加，不保存完整注意力矩阵
        for t in range(N):
            for s in range(N):
                alpha = 0
                for j in range(Q):
                    alpha += q[t][j] * k[s][j]
                alpha %= MOD

                for d in range(Q):
                    pos = h * Q + d
                    att[t][pos] = (att[t][pos] + alpha * v[s][d]) % MOD

    ans = []

    for t in range(N):
        # 投影并加入残差
        p = mul(att[t], Wp)
        z = [0] * (2 * R)

        for j in range(2 * R):
            z[j] = (p[j] + cp[j] + e[t][j]) % MOD

        # 输出层
        y = mul(z, Wy)

        for j in range(C):
            y[j] = (y[j] + cy[j]) % MOD

        ans.append(y)

    return ans


def main():
    import sys

    data = list(map(int, sys.stdin.read().split()))
    idx = 0

    N, F, R, C, S, A, Q, MOD = data[:8]
    idx = 8

    # 在主函数中完成所有输入解析
    def read_matrix(rows, cols):
        nonlocal idx
        mat = []

        for _ in range(rows):
            row = data[idx:idx + cols]
            idx += cols
            mat.append(row)

        return mat

    x = read_matrix(N, F)

    layers = []

    for l in range(S):
        d = F if l == 0 else 2 * R

        Uf = read_matrix(d, R)
        Vf = read_matrix(R, R)
        bf = read_matrix(1, R)[0]

        Ub = read_matrix(d, R)
        Vb = read_matrix(R, R)
        bb = read_matrix(1, R)[0]

        p0 = read_matrix(1, R)[0]
        rn = read_matrix(1, R)[0]

        layers.append((Uf, Vf, bf, Ub, Vb, bb, p0, rn))

    heads = []

    for _ in range(A):
        Qm = read_matrix(2 * R, Q)
        Km = read_matrix(2 * R, Q)
        Vm = read_matrix(2 * R, Q)

        heads.append((Qm, Km, Vm))

    Wp = read_matrix(A * Q, 2 * R)
    cp = read_matrix(1, 2 * R)[0]
    Wy = read_matrix(2 * R, C)
    cy = read_matrix(1, C)[0]

    ans = solve(N, F, R, C, S, A, Q, MOD,
                x, layers, heads, Wp, cp, Wy, cy)

    for row in ans:
        print(*row)


if __name__ == "__main__":
    main()

import java.util.Scanner;

public class Main {

    static class Layer {
        int[][] Uf, Vf, Ub, Vb;
        int[] bf, bb, p0, rn;
    }

    static class Head {
        int[][] Qm, Km, Vm;
    }

    // 行向量乘矩阵
    static int[] mul(int[] v, int[][] mat, int MOD) {
        int cols = mat[0].length;
        int[] res = new int[cols];

        for (int j = 0; j < cols; j++) {
            int sum = 0;

            for (int i = 0; i < v.length; i++) {
                sum += v[i] * mat[i][j];
            }

            res[j] = sum % MOD;
        }

        return res;
    }

    static int[][] solve(int N, int F, int R, int C,
                         int S, int A, int Q, int MOD,
                         int[][] x, Layer[] layers,
                         Head[] heads, int[][] Wp,
                         int[] cp, int[][] Wy, int[] cy) {

        int[][] cur = x;

        // 依次计算每一层双向 RNN
        for (int l = 0; l < S; l++) {
            Layer layer = layers[l];

            int[][] f = new int[N][R];
            int[][] g = new int[N][R];

            // 正向递推
            int[] prev = layer.p0.clone();

            for (int t = 0; t < N; t++) {
                int[] a = mul(cur[t], layer.Uf, MOD);
                int[] b = mul(prev, layer.Vf, MOD);

                for (int j = 0; j < R; j++) {
                    f[t][j] = (a[j] + b[j] + layer.bf[j]) % MOD;
                }

                prev = f[t];
            }

            // 反向递推
            int[] next = layer.rn.clone();

            for (int t = N - 1; t >= 0; t--) {
                int[] a = mul(cur[t], layer.Ub, MOD);
                int[] b = mul(next, layer.Vb, MOD);

                for (int j = 0; j < R; j++) {
                    g[t][j] = (a[j] + b[j] + layer.bb[j]) % MOD;
                }

                next = g[t];
            }

            // 拼接两个方向的结果
            int[][] nxt = new int[N][2 * R];

            for (int t = 0; t < N; t++) {
                for (int j = 0; j < R; j++) {
                    nxt[t][j] = f[t][j];
                    nxt[t][R + j] = g[t][j];
                }
            }

            cur = nxt;
        }

        int[][] e = cur;
        int[][] att = new int[N][A * Q];

        // 计算每个注意力头
        for (int h = 0; h < A; h++) {
            Head head = heads[h];

            int[][] q = new int[N][Q];
            int[][] k = new int[N][Q];
            int[][] v = new int[N][Q];

            for (int t = 0; t < N; t++) {
                q[t] = mul(e[t], head.Qm, MOD);
                k[t] = mul(e[t], head.Km, MOD);
                v[t] = mul(e[t], head.Vm, MOD);
            }

            // 直接计算注意力并累加
            for (int t = 0; t < N; t++) {
                for (int s = 0; s < N; s++) {
                    int alpha = 0;

                    for (int j = 0; j < Q; j++) {
                        alpha += q[t][j] * k[s][j];
                    }

                    alpha %= MOD;

                    for (int d = 0; d < Q; d++) {
                        int pos = h * Q + d;
                        att[t][pos] =
                                (att[t][pos] + alpha * v[s][d]) % MOD;
                    }
                }
            }
        }

        int[][] ans = new int[N][C];

        for (int t = 0; t < N; t++) {
            // 投影并完成残差连接
            int[] p = mul(att[t], Wp, MOD);
            int[] z = new int[2 * R];

            for (int j = 0; j < 2 * R; j++) {
                z[j] = (p[j] + cp[j] + e[t][j]) % MOD;
            }

            // 输出层
            int[] y = mul(z, Wy, MOD);

            for (int j = 0; j < C; j++) {
                ans[t][j] = (y[j] + cy[j]) % MOD;
            }
        }

        return ans;
    }

struct Layer {
    vector<vector<int>> Uf, Vf, Ub, Vb;
    vector<int> bf, bb, p0, rn;
};

struct Head {
    vector<vector<int>> Qm, Km, Vm;
};

// 行向量乘矩阵
vector<int> mul(const vector<int>& v,
                const vector<vector<int>>& mat,
                int MOD) {
    int cols = mat[0].size();
    vector<int> res(cols, 0);

    for (int j = 0; j < cols; j++) {
        int sum = 0;

        for (int i = 0; i < (int)v.size(); i++) {
            sum += v[i] * mat[i][j];
        }

        res[j] = sum % MOD;
    }

    return res;
}

vector<vector<int>> solve(
    int N, int F, int R, int C,
    int S, int A, int Q, int MOD,
    vector<vector<int>> x,
    const vector<Layer>& layers,
    const vector<Head>& heads,
    const vector<vector<int>>& Wp,
    const vector<int>& cp,
    const vector<vector<int>>& Wy,
    const vector<int>& cy) {

    vector<vector<int>> cur = x;

    // 依次处理每一层双向 RNN
    for (int l = 0; l < S; l++) {
        const Layer& layer = layers[l];

        vector<vector<int>> f(N, vector<int>(R));
        vector<vector<int>> g(N, vector<int>(R));

        // 正向递推
        vector<int> prev = layer.p0;

        for (int t = 0; t < N; t++) {
            vector<int> a = mul(cur[t], layer.Uf, MOD);
            vector<int> b = mul(prev, layer.Vf, MOD);

            for (int j = 0; j < R; j++) {
                f[t][j] = (a[j] + b[j] + layer.bf[j]) % MOD;
            }

            prev = f[t];
        }

        // 反向递推
        vector<int> nxtState = layer.rn;

        for (int t = N - 1; t >= 0; t--) {
            vector<int> a = mul(cur[t], layer.Ub, MOD);
            vector<int> b = mul(nxtState, layer.Vb, MOD);

            for (int j = 0; j < R; j++) {
                g[t][j] = (a[j] + b[j] + layer.bb[j]) % MOD;
            }

            nxtState = g[t];
        }

        // 拼接两个方向的状态
        vector<vector<int>> nxt(N, vector<int>(2 * R));

        for (int t = 0; t < N; t++) {
            for (int j = 0; j < R; j++) {
                nxt[t][j] = f[t][j];
                nxt[t][R + j] = g[t][j];
            }
        }

        cur = nxt;
    }

    vector<vector<int>> e = cur;
    vector<vector<int>> att(N, vector<int>(A * Q, 0));

    // 逐个计算注意力头
    for (int h = 0; h < A; h++) {
        vector<vector<int>> q(N), k(N), v(N);

        for (int t = 0; t < N; t++) {
            q[t] = mul(e[t], heads[h].Qm, MOD);
            k[t] = mul(e[t], heads[h].Km, MOD);
            v[t] = mul(e[t], heads[h].Vm, MOD);
        }

        // 不保存完整注意力矩阵，直接累加结果
        for (int t = 0; t < N; t++) {
            for (int s = 0; s < N; s++) {
                int alpha = 0;

                for (int j = 0; j < Q; j++) {
                    alpha += q[t][j] * k[s][j];
                }

                alpha %= MOD;

                for (int d = 0; d < Q; d++) {
                    int pos = h * Q + d;
                    att[t][pos] =
                        (att[t][pos] + alpha * v[s][d]) % MOD;
                }
            }
        }
    }

    vector<vector<int>> ans(N, vector<int>(C));

    for (int t = 0; t < N; t++) {
        // 投影并加入残差
        vector<int> p = mul(att[t], Wp, MOD);
        vector<int> z(2 * R);

        for (int j = 0; j < 2 * R; j++) {
            z[j] = (p[j] + cp[j] + e[t][j]) % MOD;
        }

        // 输出层
        vector<int> y = mul(z, Wy, MOD);

        for (int j = 0; j < C; j++) {
            ans[t][j] = (y[j] + cy[j]) % MOD;
        }
    }

    return ans;
}
