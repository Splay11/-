## 解题思路

本题是典型的二维 $0/1$ 背包问题。

每个任务只能选择一次，同时受到内存和算力两个资源的限制。

定义：

$dp[j][k]$ 表示恰好使用 $j$ 单位内存、$k$ 单位算力时，能够获得的最大收益。

初始时：

* $dp[0][0]=0$
* 其他状态设为 $-1$，表示当前状态无法达到。

对于每个任务，设它的内存消耗为 $m$，算力消耗为 $g$，收益为 $v$。

由于每个任务只能选择一次，需要让内存和算力都从大到小遍历：

```text
dp[j][k] = max(dp[j][k], dp[j-m][k-g] + v)
```

只有当 $dp[j-m][k-g]$ 可以达到时，才进行状态转移。

所有任务处理完后，遍历所有不超过资源限制的状态：

* 首先选择收益最大的状态；
* 如果收益相同，选择内存消耗更小的状态；
* 如果内存也相同，选择算力消耗更小的状态。

这样即可得到题目要求的答案。

## 复杂度分析

设任务数量为 $n$，内存上限为 $A$，算力上限为 $B$。

每个任务都需要遍历最多 $A \times B$ 个状态，因此：

* 时间复杂度：$O(nAB)$
* 空间复杂度：$O(AB)$

题目中 $n \le 100$，$A,B \le 500$，该复杂度可以满足要求。

## 代码实现

### Python

```python
def solve(n, A, B, tasks):
    # dp[j][k] 表示恰好使用 j 内存、k 算力时的最大收益
    dp = [[-1] * (B + 1) for _ in range(A + 1)]
    dp[0][0] = 0

    for m, g, v in tasks:
        # 0/1 背包需要倒序遍历，保证每个任务只使用一次
        for j in range(A, m - 1, -1):
            for k in range(B, g - 1, -1):
                if dp[j - m][k - g] != -1:
                    dp[j][k] = max(
                        dp[j][k],
                        dp[j - m][k - g] + v
                    )

    best_value = 0
    best_memory = 0
    best_power = 0

    # 找最大收益，并处理内存、算力的优先级
    for j in range(A + 1):
        for k in range(B + 1):
            if dp[j][k] > best_value:
                best_value = dp[j][k]
                best_memory = j
                best_power = k
            elif dp[j][k] == best_value:
                if j < best_memory or (j == best_memory and k < best_power):
                    best_memory = j
                    best_power = k

    return best_value, best_memory, best_power


def main():
    n, A, B = map(int, input().split())

    tasks = []
    for _ in range(n):
        m, g, v = map(int, input().split())
        tasks.append((m, g, v))

    ans = solve(n, A, B, tasks)
    print(ans[0], ans[1], ans[2])


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    // 返回：最大收益、总内存消耗、总算力消耗
    public static int[] solve(int n, int A, int B, int[][] tasks) {
        // dp[j][k] 表示恰好使用 j 内存、k 算力时的最大收益
        int[][] dp = new int[A + 1][B + 1];

        // -1 表示该状态无法达到
        for (int j = 0; j <= A; j++) {
            for (int k = 0; k <= B; k++) {
                dp[j][k] = -1;
            }
        }

        dp[0][0] = 0;

        for (int i = 0; i < n; i++) {
            int m = tasks[i][0];
            int g = tasks[i][1];
            int v = tasks[i][2];

            // 0/1 背包倒序遍历，避免一个任务被重复选择
            for (int j = A; j >= m; j--) {
                for (int k = B; k >= g; k--) {
                    if (dp[j - m][k - g] != -1) {
                        dp[j][k] = Math.max(
                            dp[j][k],
                            dp[j - m][k - g] + v
                        );
                    }
                }
            }
        }

        int bestValue = 0;
        int bestMemory = 0;
        int bestPower = 0;

        // 找最大收益，并处理内存、算力的优先级
        for (int j = 0; j <= A; j++) {
            for (int k = 0; k <= B; k++) {
                if (dp[j][k] > bestValue) {
                    bestValue = dp[j][k];
                    bestMemory = j;
                    bestPower = k;
                } else if (dp[j][k] == bestValue) {
                    if (j < bestMemory ||
                        (j == bestMemory && k < bestPower)) {
                        bestMemory = j;
                        bestPower = k;
                    }
                }
            }
        }

        return new int[]{bestValue, bestMemory, bestPower};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int A = sc.nextInt();
        int B = sc.nextInt();

        int[][] tasks = new int[n][3];

        for (int i = 0; i < n; i++) {
            tasks[i][0] = sc.nextInt();
            tasks[i][1] = sc.nextInt();
            tasks[i][2] = sc.nextInt();
        }

        int[] ans = solve(n, A, B, tasks);

        System.out.println(ans[0] + " " + ans[1] + " " + ans[2]);

        sc.close();
    }
}
```

```cpp
vector<int> solve(int n, int A, int B, const vector<vector<int>>& tasks) {
    // dp[j][k] 表示恰好使用 j 内存、k 算力时的最大收益
    vector<vector<int>> dp(A + 1, vector<int>(B + 1, -1));
    dp[0][0] = 0;

    for (int i = 0; i < n; i++) {
        int m = tasks[i][0];
        int g = tasks[i][1];
        int v = tasks[i][2];

        // 0/1 背包倒序遍历，保证每个任务只选择一次
        for (int j = A; j >= m; j--) {
            for (int k = B; k >= g; k--) {
                if (dp[j - m][k - g] != -1) {
                    dp[j][k] = max(
                        dp[j][k],
                        dp[j - m][k - g] + v
                    );
                }
            }
        }
    }

    int bestValue = 0;
    int bestMemory = 0;
    int bestPower = 0;

    // 找最大收益，并处理内存、算力的优先级
    for (int j = 0; j <= A; j++) {
        for (int k = 0; k <= B; k++) {
            if (dp[j][k] > bestValue) {
                bestValue = dp[j][k];
                bestMemory = j;

}

int main() {
    int n, A, B;

    for (int i = 0; i < n; i++) {
        cin >> tasks[i][0] >> tasks[i][1] >> tasks[i][2];
```
