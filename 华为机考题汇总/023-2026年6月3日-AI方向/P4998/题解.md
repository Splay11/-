## 解题思路

本题要求完成一次三层 $MLP$ 的前向传播、反向传播和梯度下降更新。网络结构为：

$$
d_{in}\rightarrow d_{hidden}\rightarrow d_{out}
$$

隐藏层使用 $ReLU$ 激活函数，输出层为线性输出。

前向传播时，先计算隐藏层的预激活值：

$$
z_{1,i}=\sum_{j=0}^{d_{in}-1}W_{1,i,j}x_j+b_{1,i}
$$

然后经过 $ReLU$：

$$
a_{1,i}=\max(0,z_{1,i})
$$

最后计算输出层预测值：

$$
y_{pred,i}=\sum_{j=0}^{d_{hidden}-1}W_{2,i,j}a_{1,j}+b_{2,i}
$$

损失函数为平方误差的一半：

$$
L=\frac12\sum_{i=0}^{d_{out}-1}(y_{pred,i}-y_{true,i})^2
$$

由于输出层是线性层，因此输出层误差项为：

$$
\delta_{2,i}=y_{pred,i}-y_{true,i}
$$

根据链式法则，$W_2$ 和 $b_2$ 的梯度为：

$$
\nabla W_{2,i,j}=\delta_{2,i}a_{1,j}
$$

$$
\nabla b_{2,i}=\delta_{2,i}
$$

隐藏层误差需要由输出层误差反传得到：

$$
\delta_{1,i}=\left(\sum_{j=0}^{d_{out}-1}W_{2,j,i}\delta_{2,j}\right)\cdot ReLU'(z_{1,i})
$$

其中：

$$
ReLU'(z)=
\begin{cases}
1, & z>0\
0, & z\le 0
\end{cases}
$$

特别注意题目规定 $z=0$ 时导数为 $0$。

然后隐藏层参数梯度为：

$$
\nabla W_{1,i,j}=\delta_{1,i}x_j
$$

$$
\nabla b_{1,i}=\delta_{1,i}
$$

最后对所有参数执行梯度下降更新：

$$
W_{new}=W-\eta\nabla W
$$

$$
b_{new}=b-\eta\nabla b
$$

实现时需要注意，所有梯度都应该基于更新前的参数计算。因此不能先更新 $W_2$ 后再用新的 $W_2$ 计算隐藏层误差，必须先完成全部前向传播和反向传播，再统一更新参数。

## 复杂度分析

设输入维度为 $d_{in}$，隐藏层维度为 $d_{hidden}$，输出层维度为 $d_{out}$。

前向传播中，计算 $W_1x$ 的时间复杂度为 $O(d_{hidden}d_{in})$，计算 $W_2a_1$ 的时间复杂度为 $O(d_{out}d_{hidden})$。

反向传播中，计算 $\nabla W_2$ 的时间复杂度为 $O(d_{out}d_{hidden})$，计算隐藏层误差 $\delta_1$ 的时间复杂度为 $O(d_{out}d_{hidden})$，计算 $\nabla W_1$ 的时间复杂度为 $O(d_{hidden}d_{in})$。

因此总时间复杂度为：

$$
O(d_{hidden}d_{in}+d_{out}d_{hidden})
$$

需要额外保存 $z_1$、$a_1$、$\delta_2$、$\delta_1$ 以及梯度矩阵，因此空间复杂度为：

$$
O(d_{hidden}d_{in}+d_{out}d_{hidden}+d_{hidden}+d_{out})
$$

由于题目中三个维度均不超过 $10$，该复杂度完全可以满足要求。

```python
import sys
import numpy as np


def train(W1, b1, W2, b2, x, yt, eta):
    # 计算隐藏层预激活值 z1 = W1 * x + b1
    z1 = W1.dot(x) + b1

    # 计算隐藏层激活值 a1 = ReLU(z1)
    a1 = np.maximum(z1, 0.0)

    # 计算输出层预测值 y_pred = W2 * a1 + b2
    yp = W2.dot(a1) + b2

    # 输出层误差项，平方误差损失对预测值的导数
    d2 = yp - yt

    # W2 的梯度为 delta2 和 a1 的外积
    gW2 = np.outer(d2, a1)

    # b2 的梯度就是输出层误差项
    gb2 = d2

    # ReLU 的导数，题目规定 z = 0 时导数为 0
    relu_grad = (z1 > 0).astype(float)

    # 将输出层误差通过原始 W2 反传到隐藏层
    d1 = W2.T.dot(d2) * relu_grad

    # W1 的梯度为 delta1 和 x 的外积
    gW1 = np.outer(d1, x)

    # b1 的梯度就是隐藏层误差项
    gb1 = d1

    # 使用梯度下降更新所有参数
    W1n = W1 - eta * gW1
    b1n = b1 - eta * gb1
    W2n = W2 - eta * gW2
    b2n = b2 - eta * gb2

    return W1n, b1n, W2n, b2n, yp


def main():
    data = sys.stdin.read().strip().split()
    idx = 0

    din = int(data[idx])
    idx += 1
    dh = int(data[idx])
    idx += 1
    dout = int(data[idx])
    idx += 1

    W1 = []
    for i in range(dh):
        row = []
        for j in range(din):
            row.append(float(data[idx]))
            idx += 1
        W1.append(row)

    b1 = []
    for i in range(dh):
        b1.append(float(data[idx]))
        idx += 1

    W2 = []
    for i in range(dout):
        row = []
        for j in range(dh):
            row.append(float(data[idx]))
            idx += 1
        W2.append(row)

    b2 = []
    for i in range(dout):
        b2.append(float(data[idx]))
        idx += 1

    x = []
    for i in range(din):
        x.append(float(data[idx]))
        idx += 1

    yt = []
    for i in range(dout):
        yt.append(float(data[idx]))
        idx += 1

    eta = float(data[idx])

    W1 = np.array(W1, dtype=float)
    b1 = np.array(b1, dtype=float)
    W2 = np.array(W2, dtype=float)
    b2 = np.array(b2, dtype=float)
    x = np.array(x, dtype=float)
    yt = np.array(yt, dtype=float)

    W1n, b1n, W2n, b2n, yp = train(W1, b1, W2, b2, x, yt, eta)

    for i in range(dh):
        print(" ".join(f"{W1n[i][j]:.6f}" for j in range(din)))

    print(" ".join(f"{b1n[i]:.6f}" for i in range(dh)))

    for i in range(dout):
        print(" ".join(f"{W2n[i][j]:.6f}" for j in range(dh)))

    print(" ".join(f"{b2n[i]:.6f}" for i in range(dout)))
    print(" ".join(f"{yp[i]:.6f}" for i in range(dout)))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    static class Result {
        double[][] W1n;
        double[] b1n;
        double[][] W2n;
        double[] b2n;
        double[] yp;

        Result(double[][] W1n, double[] b1n, double[][] W2n, double[] b2n, double[] yp) {
            this.W1n = W1n;
            this.b1n = b1n;
            this.W2n = W2n;
            this.b2n = b2n;
            this.yp = yp;
        }
    }

    static Result train(int din, int dh, int dout,
                        double[][] W1, double[] b1,
                        double[][] W2, double[] b2,
                        double[] x, double[] yt,
                        double eta) {
        double[] z1 = new double[dh];
        double[] a1 = new double[dh];

        // 计算隐藏层预激活值 z1 = W1 * x + b1
        for (int i = 0; i < dh; i++) {
            z1[i] = b1[i];
            for (int j = 0; j < din; j++) {
                z1[i] += W1[i][j] * x[j];
            }
        }

        // 计算隐藏层激活值 a1 = ReLU(z1)
        for (int i = 0; i < dh; i++) {
            if (z1[i] > 0) {
                a1[i] = z1[i];
            } else {
                a1[i] = 0.0;
            }
        }

        double[] yp = new double[dout];

        // 计算输出层预测值 y_pred = W2 * a1 + b2
        for (int i = 0; i < dout; i++) {
            yp[i] = b2[i];
            for (int j = 0; j < dh; j++) {
                yp[i] += W2[i][j] * a1[j];
            }
        }

        double[] d2 = new double[dout];

        // 输出层误差项，平方误差损失对预测值的导数
        for (int i = 0; i < dout; i++) {
            d2[i] = yp[i] - yt[i];
        }

        double[][] gW2 = new double[dout][dh];
        double[] gb2 = new double[dout];

        // W2 的梯度为 delta2 和 a1 的外积，b2 的梯度为 delta2
        for (int i = 0; i < dout; i++) {
            gb2[i] = d2[i];
            for (int j = 0; j < dh; j++) {
                gW2[i][j] = d2[i] * a1[j];
            }
        }

        double[] d1 = new double[dh];

        // 使用更新前的 W2 将误差反传到隐藏层，并乘以 ReLU 的导数
        for (int i = 0; i < dh; i++) {
            double sum = 0.0;
            for (int j = 0; j < dout; j++) {
                sum += W2[j][i] * d2[j];
            }

            if (z1[i] > 0) {
                d1[i] = sum;
            } else {
                d1[i] = 0.0;
            }
        }

        double[][] gW1 = new double[dh][din];
        double[] gb1 = new double[dh];

        // W1 的梯度为 delta1 和 x 的外积，b1 的梯度为 delta1
        for (int i = 0; i < dh; i++) {
            gb1[i] = d1[i];
            for (int j = 0; j < din; j++) {
                gW1[i][j] = d1[i] * x[j];
            }
        }

        double[][] W1n = new double[dh][din];
        double[] b1n = new double[dh];
        double[][] W2n = new double[dout][dh];
        double[] b2n = new double[dout];

        // 使用梯度下降更新 W1 和 b1
        for (int i = 0; i < dh; i++) {
            b1n[i] = b1[i] - eta * gb1[i];
            for (int j = 0; j < din; j++) {
                W1n[i][j] = W1[i][j] - eta * gW1[i][j];
            }
        }

        // 使用梯度下降更新 W2 和 b2
        for (int i = 0; i < dout; i++) {
            b2n[i] = b2[i] - eta * gb2[i];
            for (int j = 0; j < dh; j++) {
                W2n[i][j] = W2[i][j] - eta * gW2[i][j];
            }
        }

        return new Result(W1n, b1n, W2n, b2n, yp);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int din = sc.nextInt();
        int dh = sc.nextInt();
        int dout = sc.nextInt();

        double[][] W1 = new double[dh][din];
        for (int i = 0; i < dh; i++) {
            for (int j = 0; j < din; j++) {
                W1[i][j] = sc.nextDouble();
            }
        }

        double[] b1 = new double[dh];
        for (int i = 0; i < dh; i++) {
            b1[i] = sc.nextDouble();
        }

        double[][] W2 = new double[dout][dh];
        for (int i = 0; i < dout; i++) {
            for (int j = 0; j < dh; j++) {
                W2[i][j] = sc.nextDouble();
            }
        }

        double[] b2 = new double[dout];
        for (int i = 0; i < dout; i++) {
            b2[i] = sc.nextDouble();
        }

        double[] x = new double[din];
        for (int i = 0; i < din; i++) {
            x[i] = sc.nextDouble();
        }

        double[] yt = new double[dout];
        for (int i = 0; i < dout; i++) {
            yt[i] = sc.nextDouble();
        }

        double eta = sc.nextDouble();

        Result res = train(din, dh, dout, W1, b1, W2, b2, x, yt, eta);

        for (int i = 0; i < dh; i++) {
            for (int j = 0; j < din; j++) {
                if (j > 0) {
                    System.out.print(" ");
                }
                System.out.printf("%.6f", res.W1n[i][j]);
            }
            System.out.println();
        }

        for (int i = 0; i < dh; i++) {
            if (i > 0) {
                System.out.print(" ");
            }
            System.out.printf("%.6f", res.b1n[i]);
        }
        System.out.println();

        for (int i = 0; i < dout; i++) {
            for (int j = 0; j < dh; j++) {
                if (j > 0) {
                    System.out.print(" ");
                }
                System.out.printf("%.6f", res.W2n[i][j]);
            }
            System.out.println();
        }

        for (int i = 0; i < dout; i++) {
            if (i > 0) {
                System.out.print(" ");
            }
            System.out.printf("%.6f", res.b2n[i]);
        }
        System.out.println();

        for (int i = 0; i < dout; i++) {

                System.out.print(" ");
            }
            System.out.printf("%.6f", res.yp[i]);
        }
        System.out.println();

        sc.close();
    }
}
```
