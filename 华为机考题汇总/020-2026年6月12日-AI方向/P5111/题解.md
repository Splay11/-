## 解题思路

本题要求实现 $K$-Means 聚类算法。

初始时，直接选择输入数据的前 $K$ 个点作为 $K$ 个初始质心，编号分别为 $0$ 到 $K-1$。

每轮迭代执行两步：

$1.$ 对于每个点，计算它到所有质心的 $3D$ 欧氏距离，将其分配到距离最近的簇中。

由于开平方不会影响距离大小比较，因此代码中直接比较平方距离：

$$
d^2=(x_1-x_2)^2+(y_1-y_2)^2+(z_1-z_2)^2
$$

如果一个点到多个质心的距离相同，则选择编号最小的簇。实现时按编号从小到大遍历质心，只有当距离严格更小时才更新答案。

$2.$ 根据每个簇内所有点的坐标平均值更新质心。

如果某个簇没有点，保持该簇质心不变。

当所有质心的 $x,y,z$ 坐标变化量都小于 $10^{-6}$，或迭代次数达到 $100$ 次时停止。最后输出每个输入点所属的簇编号。

## 复杂度分析

设点数为 $N$，簇数为 $K$，最大迭代次数为 $100$。

每轮需要计算每个点到每个质心的距离，时间复杂度为 $O(NK)$。

总时间复杂度为：

$$
O(100NK)
$$

因为 $100$ 是常数，所以也可以记为 $O(NK)$。

需要存储点、质心、簇编号等信息，空间复杂度为：

$$
O(N+K)
$$

## 代码实现

### Python

```python
import sys

# 计算平方距离，避免开方
def dist2(p, c):
    return (p[0] - c[0]) ** 2 + (p[1] - c[1]) ** 2 + (p[2] - c[2]) ** 2

def kmeans(points, n, k):
    # 前 k 个点作为初始质心
    centers = [points[i][:] for i in range(k)]
    belong = [0] * n

    for _ in range(100):
        # 分配每个点到最近的质心
        for i in range(n):
            best = 0
            best_dist = dist2(points[i], centers[0])
            for j in range(1, k):
                d = dist2(points[i], centers[j])
                # 只有距离严格更小时才更新，保证距离相等时选编号小的簇
                if d < best_dist:
                    best_dist = d
                    best = j
            belong[i] = best

        sums = [[0.0, 0.0, 0.0] for _ in range(k)]
        cnt = [0] * k

        # 统计每个簇的坐标和
        for i in range(n):
            b = belong[i]
            sums[b][0] += points[i][0]
            sums[b][1] += points[i][1]
            sums[b][2] += points[i][2]
            cnt[b] += 1

        new_centers = [centers[i][:] for i in range(k)]

        # 更新质心
        for i in range(k):
            if cnt[i] > 0:
                new_centers[i][0] = sums[i][0] / cnt[i]
                new_centers[i][1] = sums[i][1] / cnt[i]
                new_centers[i][2] = sums[i][2] / cnt[i]

        ok = True

        # 判断是否收敛
        for i in range(k):
            for j in range(3):
                if abs(new_centers[i][j] - centers[i][j]) >= 1e-6:
                    ok = False

        centers = new_centers

        if ok:
            break

    return belong

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    n = int(data[0])
    k = int(data[1])

    points = []
    idx = 2
    for _ in range(n):
        x = float(data[idx])
        y = float(data[idx + 1])
        z = float(data[idx + 2])
        points.append([x, y, z])
        idx += 3

    ans = kmeans(points, n, k)

    for x in ans:
        print(x)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    // 计算平方距离，避免开方
    static double dist2(double[] p, double[] c) {
        double dx = p[0] - c[0];
        double dy = p[1] - c[1];
        double dz = p[2] - c[2];
        return dx * dx + dy * dy + dz * dz;
    }

    static int[] kmeans(double[][] points, int n, int k) {
        double[][] centers = new double[k][3];

        // 前 k 个点作为初始质心
        for (int i = 0; i < k; i++) {
            centers[i][0] = points[i][0];
            centers[i][1] = points[i][1];
            centers[i][2] = points[i][2];
        }

        int[] belong = new int[n];

        for (int t = 0; t < 100; t++) {
            // 分配每个点到最近的质心
            for (int i = 0; i < n; i++) {
                int best = 0;
                double bestDist = dist2(points[i], centers[0]);

                for (int j = 1; j < k; j++) {
                    double d = dist2(points[i], centers[j]);
                    // 只有距离严格更小时才更新，保证距离相等时选编号小的簇
                    if (d < bestDist) {
                        bestDist = d;
                        best = j;
                    }
                }

                belong[i] = best;
            }

            double[][] sums = new double[k][3];
            int[] cnt = new int[k];

            // 统计每个簇的坐标和
            for (int i = 0; i < n; i++) {
                int b = belong[i];
                sums[b][0] += points[i][0];
                sums[b][1] += points[i][1];
                sums[b][2] += points[i][2];
                cnt[b]++;
            }

            double[][] newCenters = new double[k][3];

            // 更新质心，空簇保持原质心不变
            for (int i = 0; i < k; i++) {
                if (cnt[i] > 0) {
                    newCenters[i][0] = sums[i][0] / cnt[i];
                    newCenters[i][1] = sums[i][1] / cnt[i];
                    newCenters[i][2] = sums[i][2] / cnt[i];
                } else {
                    newCenters[i][0] = centers[i][0];
                    newCenters[i][1] = centers[i][1];
                    newCenters[i][2] = centers[i][2];
                }
            }

            boolean ok = true;

            // 判断是否收敛
            for (int i = 0; i < k; i++) {
                for (int j = 0; j < 3; j++) {
                    if (Math.abs(newCenters[i][j] - centers[i][j]) >= 1e-6) {
                        ok = false;
                    }
                }
            }

            centers = newCenters;

            if (ok) {
                break;
            }
        }

        return belong;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int k = sc.nextInt();

        double[][] points = new double[n][3];

        for (int i = 0; i < n; i++) {
            points[i][0] = sc.nextDouble();
            points[i][1] = sc.nextDouble();
            points[i][2] = sc.nextDouble();
        }

        int[] ans = kmeans(points, n, k);

        for (int i = 0; i < n; i++) {
            System.out.println(ans[i]);
        }

        sc.close();
    }
}
```
