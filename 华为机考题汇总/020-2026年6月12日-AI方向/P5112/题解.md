## 解题思路

本题使用的算法是矩阵乘法、数值稳定 $softmax$ 和排序选 $Top-S$。

先对每个 $token$ 计算它到每个专家的门控分数：

$score[i][e]=X_i \times W_{g,e}$

然后对同一个 $token$ 的 $E$ 个专家分数做数值稳定的 $softmax$，得到归一化门控权重 $gate[i][e]$。

接着每个专家 $e$ 独立按照 $gate[i][e]$ 从大到小选择 $Top-S$ 个 $token$，如果分数相同，选择编号更小的 $token$。

对于被专家 $e$ 选中的 $token$，计算专家变换：

$X_i \times W_e$

再乘以对应门控权重 $gate[i][e]$ 累加到输出 $Y_i$ 中。

如果某个 $token$ 没有被任何专家选中，则它的输出保持为 $0$ 向量。

当 $N,d,E,S$ 不满足正整数要求，或者 $S>N$ 时，直接输出 $0$。

## 复杂度分析

设 $N$ 为 $token$ 数量，$d$ 为特征维度，$E$ 为专家数量，$S$ 为每个专家选择的 $token$ 数。

计算门控分数需要 $O(N \times d \times E)$。

每个专家排序选择 $Top-S$，总复杂度为 $O(E \times N \log N)$。

专家变换最多处理 $E \times S$ 个 $token$，每次矩阵乘法需要 $O(d^2)$，所以为 $O(E \times S \times d^2)$。

总时间复杂度为：

$O(NdE+EN\log N+ESd^2)$

空间复杂度为：

$O(NE+Nd+Ed^2)$

复杂度适合本题要求。

### Python

```python
import sys
import math


def expert_choice_routing(N, d, E, S, X, Wg, experts):
    # gate[i][e] 表示第 i 个 token 分配给专家 e 的归一化权重
    gate = [[0.0] * E for _ in range(N)]

    # 计算门控分数，并对每个 token 做数值稳定 softmax
    for i in range(N):
        scores = [0.0] * E
        for e in range(E):
            s = 0.0
            for k in range(d):
                s += X[i][k] * Wg[k][e]
            scores[e] = s

        max_score = max(scores)
        total = 0.0
        for e in range(E):
            gate[i][e] = math.exp(scores[e] - max_score)
            total += gate[i][e]

        for e in range(E):
            gate[i][e] /= total

    # 输出矩阵，初始为 0，未被选中的 token 保持 0 向量
    Y = [[0.0] * d for _ in range(N)]

    # 每个专家独立选择 Top-S 个 token
    for e in range(E):
        order = list(range(N))
        order.sort(key=lambda i: (-gate[i][e], i))

        for idx in range(S):
            i = order[idx]

            # 计算 X[i] * W_e，并乘以门控权重累加
            for j in range(d):
                val = 0.0
                for k in range(d):
                    val += X[i][k] * experts[e][k][j]
                Y[i][j] += gate[i][e] * val

    return Y


def main():
    data = sys.stdin.read().strip().split()
    if len(data) < 4:
        print(0)
        return

    N = int(data[0])
    d = int(data[1])
    E = int(data[2])
    S = int(data[3])

    # 检查基本约束
    if N <= 0 or d <= 0 or E <= 0 or S <= 0 or S > N:
        print(0)
        return

    pos = 4

    X = []
    for _ in range(N):
        row = []
        for _ in range(d):
            row.append(float(data[pos]))
            pos += 1
        X.append(row)

    Wg = []
    for _ in range(d):
        row = []
        for _ in range(E):
            row.append(float(data[pos]))
            pos += 1
        Wg.append(row)

    experts = []
    for _ in range(E):
        mat = []
        for _ in range(d):
            row = []
            for _ in range(d):
                row.append(float(data[pos]))
                pos += 1
            mat.append(row)
        experts.append(mat)

    Y = expert_choice_routing(N, d, E, S, X, Wg, experts)

    ans = []
    for i in range(N):
        row = []
        for j in range(d):
            v = Y[i][j]
            if abs(v) < 1e-9:
                v = 0.0
            row.append(f"{v:.2f}")
        ans.append(" ".join(row))

    print("\n".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static double[][] expertChoiceRouting(int N, int d, int E, int S,
                                          double[][] X,
                                          double[][] Wg,
                                          double[][][] experts) {
        // gate[i][e] 表示第 i 个 token 分配给专家 e 的归一化权重
        double[][] gate = new double[N][E];

        // 计算门控分数，并对每个 token 做数值稳定 softmax
        for (int i = 0; i < N; i++) {
            double[] scores = new double[E];
            double maxScore = -Double.MAX_VALUE;

            for (int e = 0; e < E; e++) {
                double s = 0.0;
                for (int k = 0; k < d; k++) {
                    s += X[i][k] * Wg[k][e];
                }
                scores[e] = s;
                if (s > maxScore) {
                    maxScore = s;
                }
            }

            double total = 0.0;
            for (int e = 0; e < E; e++) {
                gate[i][e] = Math.exp(scores[e] - maxScore);
                total += gate[i][e];
            }

            for (int e = 0; e < E; e++) {
                gate[i][e] /= total;
            }
        }

        // 输出矩阵，初始为 0，未被选中的 token 保持 0 向量
        double[][] Y = new double[N][d];

        // 每个专家独立选择 Top-S 个 token
        for (int e = 0; e < E; e++) {
            Integer[] order = new Integer[N];
            for (int i = 0; i < N; i++) {
                order[i] = i;
            }

            final int expertId = e;
            Arrays.sort(order, new Comparator<Integer>() {
                public int compare(Integer a, Integer b) {
                    double diff = gate[b][expertId] - gate[a][expertId];
                    if (Math.abs(diff) > 1e-12) {
                        return diff > 0 ? 1 : -1;
                    }
                    return a - b;
                }
            });

            for (int idx = 0; idx < S; idx++) {
                int i = order[idx];

                // 计算 X[i] * W_e，并乘以门控权重累加
                for (int j = 0; j < d; j++) {
                    double val = 0.0;
                    for (int k = 0; k < d; k++) {
                        val += X[i][k] * experts[e][k][j];
                    }
                    Y[i][j] += gate[i][e] * val;
                }
            }
        }

        return Y;
    }

    public static void main(String[] args) throws Exception {
        Locale.setDefault(Locale.US);

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        ArrayList<String> list = new ArrayList<String>();

        String line;
        while ((line = br.readLine()) != null) {
            StringTokenizer st = new StringTokenizer(line);
            while (st.hasMoreTokens()) {
                list.add(st.nextToken());
            }
        }

        if (list.size() < 4) {
            System.out.print(0);
            return;
        }

        int pos = 0;
        int N = Integer.parseInt(list.get(pos++));
        int d = Integer.parseInt(list.get(pos++));
        int E = Integer.parseInt(list.get(pos++));
        int S = Integer.parseInt(list.get(pos++));

        // 检查基本约束
        if (N <= 0 || d <= 0 || E <= 0 || S <= 0 || S > N) {
            System.out.print(0);
            return;
        }

        double[][] X = new double[N][d];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < d; j++) {
                X[i][j] = Double.parseDouble(list.get(pos++));
            }
        }

        double[][] Wg = new double[d][E];
        for (int i = 0; i < d; i++) {
            for (int j = 0; j < E; j++) {
                Wg[i][j] = Double.parseDouble(list.get(pos++));
            }
        }

        double[][][] experts = new double[E][d][d];
        for (int e = 0; e < E; e++) {
            for (int i = 0; i < d; i++) {
                for (int j = 0; j < d; j++) {
                    experts[e][i][j] = Double.parseDouble(list.get(pos++));
                }
            }
        }

        double[][] Y = expertChoiceRouting(N, d, E, S, X, Wg, experts);

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < d; j++) {
                double v = Y[i][j];
                if (Math.abs(v) < 1e-9) {
                    v = 0.0;
                }
                if (j > 0) {
                    sb.append(" ");
                }
                sb.append(String.format("%.2f", v));
            }
            if (i + 1 < N) {
                sb.append("\n");
            }
        }

        System.out.print(sb.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<vector<double> > expertChoiceRouting(
    int N, int d, int E, int S,
    vector<vector<double> >& X,
    vector<vector<double> >& Wg,
    vector<vector<vector<double> > >& experts
) {
    // gate[i][e] 表示第 i 个 token 分配给专家 e 的归一化权重
    vector<vector<double> > gate(N, vector<double>(E, 0.0));

    // 计算门控分数，并对每个 token 做数值稳定 softmax
    for (int i = 0; i < N; i++) {
        vector<double> scores(E, 0.0);
        double maxScore = -1e100;

        for (int e = 0; e < E; e++) {
            double s = 0.0;
            for (int k = 0; k < d; k++) {
                s += X[i][k] * Wg[k][e];
            }
            scores[e] = s;
            maxScore = max(maxScore, s);
        }

        double total = 0.0;
        for (int e = 0; e < E; e++) {
            gate[i][e] = exp(scores[e] - maxScore);
            total += gate[i][e];
        }

        for (int e = 0; e < E; e++) {
            gate[i][e] /= total;
        }
    }

    // 输出矩阵，初始为 0，未被选中的 token 保持 0 向量
    vector<vector<double> > Y(N, vector<double>(d, 0.0));

    // 每个专家独立选择 Top-S 个 token
    for (int e = 0; e < E; e++) {
        vector<int> order(N);
        for (int i = 0; i < N; i++) {
            order[i] = i;
        }

        sort(order.begin(), order.end(), [&](int a, int b) {
            if (fabs(gate[a][e] - gate[b][e]) > 1e-12) {
                return gate[a][e] > gate[b][e];
            }
            return a < b;
        });

        for (int idx = 0; idx < S; idx++) {
            int i = order[idx];

            // 计算 X[i] * W_e，并乘以门控权重累加
            for (int j = 0; j < d; j++) {
                double val = 0.0;
                for (int k = 0; k < d; k++) {
                    val += X[i][k] * experts[e][k][j];
                }
                Y[i][j] += gate[i][e] * val;
            }
        }
    }

    return Y;
}

int main() {
    int N, d, E, S;
    if (!(cin >> N >> d >> E >> S)) {
        return 0;
    }

    // 检查基本约束
    if (N <= 0 || d <= 0 || E <= 0 || S <= 0 || S > N) {
        cout << 0 << endl;
        return 0;
    }

    vector<vector<double> > X(N, vector<double>(d));
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < d; j++) {
            cin >> X[i][j];
        }
    }

    vector<vector<double> > Wg(d, vector<double>(E));
    for (int i = 0; i < d; i++) {
        for (int j = 0; j < E; j++) {
            cin >> Wg[i][j];
        }
    }

    vector<vector<vector<double> > > experts(E, vector<vector<double> >(d, vector<double>(d)));
    for (int e = 0; e < E; e++) {
        for (int i = 0; i < d; i++) {
            for (int j = 0; j < d; j++) {
                cin >> experts[e][i][j];
            }
        }
    }

    vector<vector<double> > Y = expertChoiceRouting(N, d, E, S, X, Wg, experts);

    cout << fixed << setprecision(2);
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < d; j++) {
            double v = Y[i][j];
            if (fabs(v) < 1e-9) {
                v = 0.0;
            }
            if (j > 0) {
                cout << " ";
            }
            cout << v;
        }
        if (i + 1 < N) {
            cout << "\n";
        }
    }

    return 0;
}
```
