## 解题思路

本题可以看成经典的完全背包 / 最少硬币数问题。

有 $4$ 种任务，每种任务的资源消耗分别为 $A,B,C,D$。对于每台机器的内存大小 $X$，需要用若干个任务消耗值相加得到 $X$，并且任务数量尽量少。

设 $dp[i]$ 表示组成资源消耗 $i$ 所需的最少任务数量。

状态转移为：

$dp[i]=\min(dp[i-c]+1)$

其中 $c$ 是 $A,B,C,D$ 中不超过 $i$ 的任务消耗值。

为了满足输出要求：

先将 $A,B,C,D$ 升序排序。
得到最少任务数后，从小到大尝试选择任务消耗值。
如果选择某个值 $c$ 后，仍然能用 $dp[remain-c]=cnt-1$ 完成最优方案，就选择它。
这样可以保证输出的序列升序，并且在多种最优方案中按字典序最小。

## 复杂度分析

设最大机器内存为 $M$，题目中 $M \le 11000$。

预处理 $dp$ 的时间复杂度为 $O(4M)$。

设所有输出方案的任务总数为 $S$，重构答案的时间复杂度为 $O(4S)$。

总时间复杂度为 $O(4M+4S)$，可以满足数据范围。

空间复杂度为 $O(M)$。

## 代码实现

### Python

```python
import sys


def build_plans(coins, machines):
    # 将任务资源消耗升序排列，方便构造字典序最小方案
    coins.sort()

    max_x = max(machines)
    INF = 10 ** 9

    # dp[i] 表示组成资源消耗 i 的最少任务数量
    dp = [INF] * (max_x + 1)
    dp[0] = 0

    # 完全背包，计算每个资源值的最少任务数
    for i in range(1, max_x + 1):
        for c in coins:
            if c > i:
                break
            dp[i] = min(dp[i], dp[i - c] + 1)

    ans = []

    for x in machines:
        remain = x
        cnt = dp[x]
        plan = []

        # 从小到大尝试任务消耗，保证字典序最小
        while remain > 0:
            for c in coins:
                if c <= remain and dp[remain - c] == cnt - 1:
                    plan.append(str(c))
                    remain -= c
                    cnt -= 1
                    break

        ans.append("+".join(plan))

    return ans


def main():
    data = list(map(int, sys.stdin.read().split()))

    coins = data[:4]
    n = data[4]
    machines = data[5:5 + n]

    ans = build_plans(coins, machines)

    sys.stdout.write("\n".join(ans))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    public static List<String> buildPlans(int[] coins, int[] machines) {
        // 将任务资源消耗升序排列，方便构造字典序最小方案
        Arrays.sort(coins);

        int maxX = 0;
        for (int x : machines) {
            if (x > maxX) {
                maxX = x;
            }
        }

        int INF = 1000000000;

        // dp[i] 表示组成资源消耗 i 的最少任务数量
        int[] dp = new int[maxX + 1];
        Arrays.fill(dp, INF);
        dp[0] = 0;

        // 完全背包，计算每个资源值的最少任务数
        for (int i = 1; i <= maxX; i++) {
            for (int c : coins) {
                if (c > i) {
                    break;
                }
                if (dp[i - c] + 1 < dp[i]) {
                    dp[i] = dp[i - c] + 1;
                }
            }
        }

        List<String> ans = new ArrayList<>();

        for (int x : machines) {
            int remain = x;
            int cnt = dp[x];
            StringBuilder sb = new StringBuilder();
            boolean first = true;

            // 从小到大尝试任务消耗，保证字典序最小
            while (remain > 0) {
                for (int c : coins) {
                    if (c <= remain && dp[remain - c] == cnt - 1) {
                        if (!first) {
                            sb.append("+");
                        }
                        sb.append(c);
                        first = false;

                        remain -= c;
                        cnt--;
                        break;
                    }
                }
            }

            ans.add(sb.toString());
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int[] coins = new int[4];
        for (int i = 0; i < 4; i++) {
            coins[i] = sc.nextInt();
        }

        int n = sc.nextInt();
        int[] machines = new int[n];

        for (int i = 0; i < n; i++) {
            machines[i] = sc.nextInt();
        }

        List<String> ans = buildPlans(coins, machines);

        StringBuilder out = new StringBuilder();
        for (int i = 0; i < ans.size(); i++) {
            if (i > 0) {
                out.append("\n");
            }
            out.append(ans.get(i));
        }

        System.out.print(out.toString());
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<string> buildPlans(vector<int> coins, vector<int> machines) {
    // 将任务资源消耗升序排列，方便构造字典序最小方案
    sort(coins.begin(), coins.end());

    int maxX = 0;
    for (int x : machines) {
        maxX = max(maxX, x);
    }

    const int INF = 1000000000;

    // dp[i] 表示组成资源消耗 i 的最少任务数量
    vector<int> dp(maxX + 1, INF);
    dp[0] = 0;

    // 完全背包，计算每个资源值的最少任务数
    for (int i = 1; i <= maxX; i++) {
        for (int c : coins) {
            if (c > i) {
                break;
            }
            dp[i] = min(dp[i], dp[i - c] + 1);
        }
    }

    vector<string> ans;

    for (int x : machines) {
        int remain = x;
        int cnt = dp[x];
        string line = "";

        // 从小到大尝试任务消耗，保证字典序最小
        while (remain > 0) {
            for (int c : coins) {
                if (c <= remain && dp[remain - c] == cnt - 1) {
                    if (!line.empty()) {
                        line += "+";
                    }
                    line += to_string(c);

                    remain -= c;
                    cnt--;
                    break;
                }
            }
        }

        ans.push_back(line);
    }

    return ans;
}

int main() {
    vector<int> coins(4);

    for (int i = 0; i < 4; i++) {
        cin >> coins[i];
    }

    int n;
    cin >> n;

    vector<int> machines(n);
    for (int i = 0; i < n; i++) {
        cin >> machines[i];
    }

    vector<string> ans = buildPlans(coins, machines);

    for (int i = 0; i < n; i++) {
        if (i > 0) {
            cout << '\n';
        }
        cout << ans[i];
    }

    return 0;
}
```
