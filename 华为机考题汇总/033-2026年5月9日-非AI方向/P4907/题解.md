## 解题思路

使用位图算法，也就是 $bitmap$。

手机号范围为 $0$ 到 $99999999$，一共有 $10^8$ 个可能值。若直接使用集合存储，最多 $10^7$ 个手机号会占用较大内存；使用位图时，每个手机号只占 $1$ 位，总空间约为 $12.5\text{MB}$。

具体做法如下：

1. 读取 $A$ 商品清单中的 $N$ 个手机号，将对应位置在位图中标记为 $1$。
2. 读取 $B$ 商品清单中的 $M$ 个手机号。
3. 如果当前手机号在位图中为 $1$，说明该用户同时购买了两个商品，答案加 $1$。
4. 统计后将该手机号对应位置清零，避免 $B$ 商品清单中重复手机号被重复统计。
5. 最后输出答案。

这样可以保证同一个用户即使多次购买，也只统计一次。

## 复杂度分析

时间复杂度为 $O(N+M)$，每个手机号只处理一次。

空间复杂度为 $O(U)$ 位，其中 $U=10^8$，约为 $12.5\text{MB}$，满足题目内存限制。

## 代码实现

### Python

```python
import sys

MAX_NUM = 100000000

# 将手机号加入位图
def add_number(bitmap, x):
    idx = x >> 3
    mask = 1 << (x & 7)
    bitmap[idx] |= mask

# 判断手机号是否出现过，若出现则清除，避免重复统计
def check_and_remove(bitmap, x):
    idx = x >> 3
    mask = 1 << (x & 7)

    if bitmap[idx] & mask:
        bitmap[idx] &= 0xFF ^ mask
        return True

    return False

def main():
    input = sys.stdin.buffer.readline

    bitmap = bytearray((MAX_NUM + 7) // 8)

    n = int(input())
    for _ in range(n):
        x = int(input())
        add_number(bitmap, x)

    m = int(input())
    ans = 0

    for _ in range(m):
        x = int(input())
        if check_and_remove(bitmap, x):
            ans += 1

    print(ans)

if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.BitSet;

public class Main {
    static final int MAX_NUM = 100000000;

    // 判断手机号是否出现过，若出现则清除，避免重复统计
    static boolean checkAndClear(BitSet bitmap, int x) {
        if (bitmap.get(x)) {
            bitmap.clear(x);
            return true;
        }
        return false;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        BitSet bitmap = new BitSet(MAX_NUM);

        int n = Integer.parseInt(br.readLine());
        for (int i = 0; i < n; i++) {
            int x = Integer.parseInt(br.readLine());
            bitmap.set(x);
        }

        int m = Integer.parseInt(br.readLine());
        int ans = 0;

        for (int i = 0; i < m; i++) {
            int x = Integer.parseInt(br.readLine());

            // 如果在A清单中出现过，则统计一次
            if (checkAndClear(bitmap, x)) {
                ans++;
            }
        }

        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

const int MAX_NUM = 100000000;

// 将手机号加入位图
void setBit(vector<unsigned long long>& bitmap, int x) {
    bitmap[x >> 6] |= 1ULL << (x & 63);
}

// 判断手机号是否出现过，若出现则清除，避免重复统计
bool checkAndClear(vector<unsigned long long>& bitmap, int x) {
    int idx = x >> 6;
    unsigned long long mask = 1ULL << (x & 63);

    if (bitmap[idx] & mask) {
        bitmap[idx] &= ~mask;
        return true;
    }

    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    vector<unsigned long long> bitmap((MAX_NUM + 63) >> 6);

    int n;
    cin >> n;

    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        setBit(bitmap, x);
    }

    int m;
    cin >> m;

    int ans = 0;

    for (int i = 0; i < m; i++) {
        int x;
        cin >> x;

        // 如果在A清单中出现过，则统计一次
        if (checkAndClear(bitmap, x)) {
            ans++;
        }
    }

    cout << ans << '\n';
    return 0;
}
```
