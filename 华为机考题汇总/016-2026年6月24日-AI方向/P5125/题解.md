## 解题思路

序列已经按非降序排列。对于一个区间 $[l,r]$，把所有数替换成该区间的中位数时，绝对误差和最小。

设中位数位置为：

$mid=(l+r)/2$

用前缀和 $sum$ 可以在 $O(1)$ 时间计算区间代价：

$cost(l,r)=a[mid]\times(mid-l+1)-(sum[mid]-sum[l-1])+(sum[r]-sum[mid])-a[mid]\times(r-mid)$

设 $dp_t[i]$ 表示前 $i$ 个数恰好分成 $t$ 段的最小代价，则：

$dp_t[i]=\min(dp_{t-1}[p]+cost(p+1,i))$

其中 $t-1\le p<i$。

由于该区间代价满足 $Monge$ 性质，最优决策点具有单调性，可以使用动态规划分治优化，把每一层 $dp$ 的计算优化到 $O(N\log N)$。

因为继续拆分区间不会让总代价变大，所以“最多 $K$ 段”的最优答案可以用“恰好 $K$ 段”的 $dp_K[N]$ 表示。

## 复杂度分析

计算每个区间代价为 $O(1)$。

动态规划共有 $K$ 层，每层使用分治优化计算，时间复杂度为：

$O(KN\log N)$

只保留上一层和当前层 $dp$，空间复杂度为：

$O(N)$

该复杂度适合 $N\le 5000$，$K\le 50$ 的数据范围。

## 代码实现

### Python

```python
import sys

def min_total_cost(n, k, nums):
    a = [0] + nums

    # 前缀和，方便 O(1) 计算区间代价
    pre = [0] * (n + 1)
    for i in range(1, n + 1):
        pre[i] = pre[i - 1] + a[i]

    def cost(l, r):
        # 区间 [l, r] 的中位数位置
        mid = (l + r) // 2

        # 左侧到中位数的代价
        left = a[mid] * (mid - l + 1) - (pre[mid] - pre[l - 1])

        # 右侧到中位数的代价
        right = (pre[r] - pre[mid]) - a[mid] * (r - mid)

        return left + right

    INF = 10 ** 30
    prev = [INF] * (n + 1)
    prev[0] = 0

    def compute(cur, prev, l, r, opt_l, opt_r, t):
        if l > r:
            return

        mid = (l + r) // 2
        best_val = INF
        best_pos = opt_l

        # 枚举上一段结束位置 p
        start = max(opt_l, t - 1)
        end = min(opt_r, mid - 1)

        for p in range(start, end + 1):
            val = prev[p] + cost(p + 1, mid)
            if val < best_val:
                best_val = val
                best_pos = p

        cur[mid] = best_val

        # 利用最优决策点单调性进行分治
        compute(cur, prev, l, mid - 1, opt_l, best_pos, t)
        compute(cur, prev, mid + 1, r, best_pos, opt_r, t)

    k = min(k, n)

    for t in range(1, k + 1):
        cur = [INF] * (n + 1)
        compute(cur, prev, t, n, t - 1, n - 1, t)
        prev = cur

    return prev[n]

def main():
    data = list(map(int, sys.stdin.read().split()))
    n, k = data[0], data[1]
    nums = data[2:2 + n]
    print(min_total_cost(n, k, nums))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {
    static int n, k;
    static long[] a, pre, prev, cur;
    static final long INF = (1L << 60);

    static long cost(int l, int r) {
        // 区间 [l, r] 的中位数位置
        int mid = (l + r) / 2;

        // 左侧到中位数的代价
        long left = a[mid] * (mid - l + 1) - (pre[mid] - pre[l - 1]);

        // 右侧到中位数的代价
        long right = (pre[r] - pre[mid]) - a[mid] * (r - mid);

        return left + right;
    }

    static void compute(int l, int r, int optL, int optR, int t) {
        if (l > r) {
            return;
        }

        int mid = (l + r) / 2;
        long bestVal = INF;
        int bestPos = optL;

        // 枚举上一段结束位置 p
        int start = Math.max(optL, t - 1);
        int end = Math.min(optR, mid - 1);

        for (int p = start; p <= end; p++) {
            long val = prev[p] + cost(p + 1, mid);
            if (val < bestVal) {
                bestVal = val;
                bestPos = p;
            }
        }

        cur[mid] = bestVal;

        // 利用最优决策点单调性进行分治
        compute(l, mid - 1, optL, bestPos, t);
        compute(mid + 1, r, bestPos, optR, t);
    }

    static long minTotalCost() {
        prev = new long[n + 1];

        for (int i = 0; i <= n; i++) {
            prev[i] = INF;
        }
        prev[0] = 0;

        k = Math.min(k, n);

        for (int t = 1; t <= k; t++) {
            cur = new long[n + 1];
            for (int i = 0; i <= n; i++) {
                cur[i] = INF;
            }

            compute(t, n, t - 1, n - 1, t);
            prev = cur;
        }

        return prev[n];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        n = sc.nextInt();
        k = sc.nextInt();

        a = new long[n + 1];
        pre = new long[n + 1];

        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextLong();
            pre[i] = pre[i - 1] + a[i];
        }

        System.out.println(minTotalCost());
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, k;
vector<long long> a, pre, prev_dp, cur_dp;
const long long INF = (1LL << 60);

long long cost(int l, int r) {
    // 区间 [l, r] 的中位数位置
    int mid = (l + r) / 2;

    // 左侧到中位数的代价
    long long left = a[mid] * (mid - l + 1) - (pre[mid] - pre[l - 1]);

    // 右侧到中位数的代价
    long long right = (pre[r] - pre[mid]) - a[mid] * (r - mid);

    return left + right;
}

void compute(int l, int r, int optL, int optR, int t) {
    if (l > r) {
        return;
    }

    int mid = (l + r) / 2;
    long long bestVal = INF;
    int bestPos = optL;

    // 枚举上一段结束位置 p
    int start = max(optL, t - 1);
    int end = min(optR, mid - 1);

    for (int p = start; p <= end; p++) {
        long long val = prev_dp[p] + cost(p + 1, mid);
        if (val < bestVal) {
            bestVal = val;
            bestPos = p;
        }
    }

    cur_dp[mid] = bestVal;

    // 利用最优决策点单调性进行分治
    compute(l, mid - 1, optL, bestPos, t);
    compute(mid + 1, r, bestPos, optR, t);
}

long long minTotalCost() {
    prev_dp.assign(n + 1, INF);
    prev_dp[0] = 0;

    k = min(k, n);

    for (int t = 1; t <= k; t++) {
        cur_dp.assign(n + 1, INF);
        compute(t, n, t - 1, n - 1, t);
        prev_dp = cur_dp;
    }

    return prev_dp[n];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> k;

    a.assign(n + 1, 0);
    pre.assign(n + 1, 0);

    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        pre[i] = pre[i - 1] + a[i];
    }

    cout << minTotalCost() << '\n';

    return 0;
}
```
