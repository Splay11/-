## 解题思路

本题使用动态 $NTK$ 缩放算法调整 $RoPE$ 的基频，再对向量中每组相邻维度进行二维旋转。

首先根据当前序列长度 $T$ 和原始预训练长度 $L$ 计算缩放因子：

$$
s=\max\left(1,\frac{T}{L}\right)
$$

然后计算新的基频：

$$
b'=b\cdot s^{\frac{D}{D-2}}
$$

向量按照相邻两个元素分组。对于第 $i$ 组，其中 $0\le i<\frac{D}{2}$，对应的维度下标为：

$$
d=2i
$$

该组使用的旋转角度为：

$$
\theta_{m,i}=m\cdot (b')^{-\frac{2i}{D}}
$$

对 $(x_d,x_{d+1})$ 应用二维旋转：

$$
x'*d=x_d\cos(\theta*{m,i})-x_{d+1}\sin(\theta_{m,i})
$$

$$
x'*{d+1}=x_d\sin(\theta*{m,i})+x_{d+1}\cos(\theta_{m,i})
$$

依次处理全部 $\frac{D}{2}$ 组元素即可。计算过程中使用 $float64$ 精度，最后将每个结果输出为四位小数。

## 复杂度分析

需要遍历长度为 $D$ 的向量一次，因此时间复杂度为：

$$
O(D)
$$

使用一个长度为 $D$ 的数组保存结果，因此空间复杂度为：

$$
O(D)
$$

除去返回结果所需空间，额外空间复杂度为 $O(1)$。

## 代码实现

### Python

```python
import sys
import math


def dynamic_rope(T, L, D, base, position, x):
    # 计算动态缩放因子
    scale = max(1.0, T / L)

    # 根据动态 NTK 缩放公式更新基频
    new_base = base * math.pow(scale, D / (D - 2))

    result = [0.0] * D

    # 每两个相邻维度进行一次二维旋转
    for i in range(D // 2):
        d = 2 * i
        angle = position * math.pow(new_base, -2.0 * i / D)

        cos_value = math.cos(angle)
        sin_value = math.sin(angle)

        result[d] = x[d] * cos_value - x[d + 1] * sin_value
        result[d + 1] = x[d] * sin_value + x[d + 1] * cos_value

    return result


def main():
    data = sys.stdin.read().split()

    T = int(data[0])
    L = int(data[1])
    D = int(data[2])
    base = float(data[3])
    position = int(data[4])

    x = []
    for i in range(D):
        x.append(float(data[5 + i]))

    answer = dynamic_rope(T, L, D, base, position, x)

    print(" ".join(f"{value:.4f}" for value in answer))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Locale;
import java.util.Scanner;

public class Main {

    public static double[] dynamicRoPE(
            int T, int L, int D, double base, int position, double[] x) {

        // 计算动态缩放因子
        double scale = Math.max(1.0, (double) T / L);

        // 根据动态 NTK 缩放公式更新基频
        double newBase = base * Math.pow(scale, (double) D / (D - 2));

        double[] result = new double[D];

        // 每两个相邻维度进行一次二维旋转
        for (int i = 0; i < D / 2; i++) {
            int d = 2 * i;
            double angle = position
                    * Math.pow(newBase, -2.0 * i / D);

            double cosValue = Math.cos(angle);
            double sinValue = Math.sin(angle);

            result[d] = x[d] * cosValue - x[d + 1] * sinValue;
            result[d + 1] = x[d] * sinValue + x[d + 1] * cosValue;
        }

        return result;
    }

    public static void main(String[] args) {
        Locale.setDefault(Locale.US);
        Scanner scanner = new Scanner(System.in);

        int T = scanner.nextInt();
        int L = scanner.nextInt();
        int D = scanner.nextInt();
        double base = scanner.nextDouble();
        int position = scanner.nextInt();

        double[] x = new double[D];
        for (int i = 0; i < D; i++) {
            x[i] = scanner.nextDouble();
        }

        double[] answer = dynamicRoPE(T, L, D, base, position, x);

        for (int i = 0; i < D; i++) {
            if (i > 0) {
                System.out.print(" ");
            }
            System.out.printf("%.4f", answer[i]);
        }
        System.out.println();

        scanner.close();
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <iomanip>
using namespace std;

vector<double> dynamicRoPE(
    int T, int L, int D, double base,
    int position, const vector<double>& x
) {
    // 计算动态缩放因子
    double scale = max(1.0, (double)T / L);

    // 根据动态 NTK 缩放公式更新基频
    double newBase = base * pow(scale, (double)D / (D - 2));

    vector<double> result(D);

    // 每两个相邻维度进行一次二维旋转
    for (int i = 0; i < D / 2; i++) {
        int d = 2 * i;
        double angle = position * pow(newBase, -2.0 * i / D);

        double cosValue = cos(angle);
        double sinValue = sin(angle);

        result[d] = x[d] * cosValue - x[d + 1] * sinValue;
        result[d + 1] = x[d] * sinValue + x[d + 1] * cosValue;
    }

    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T, L, D, position;
    double base;

    cin >> T >> L >> D >> base >> position;

    vector<double> x(D);
    for (int i = 0; i < D; i++) {
        cin >> x[i];
    }

    vector<double> answer =
        dynamicRoPE(T, L, D, base, position, x);

    cout << fixed << setprecision(4);
    for (int i = 0; i < D; i++) {
        if (i > 0) {
            cout << " ";
        }
        cout << answer[i];
    }
    cout << "\n";

    return 0;
}
```
