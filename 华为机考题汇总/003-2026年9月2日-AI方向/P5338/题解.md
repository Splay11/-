## 解题思路

矩阵 $A$ 已经完成了按行量化，每一行都给出了：

* 整数量化值 $A_q$
* 缩放因子 $s_A=\frac{a_A}{2^{b_A}}$

因此只需要对矩阵 $B$ 做同样的 8 bit 量化，然后直接计算整数点积。

题目中 $B$ 的数据实际上按照 $B^T$ 输入。也就是说，每一行对应 $B$ 的一列。样例 2 中 $B$ 是 $2\times1$，输入为一行 `1.0 3.0`，也验证了这一点。

对于 $B$ 的某一列 $r$，首先计算真实缩放因子：

$$
s=\frac{\max |r_i|}{127}
$$

然后量化：

$$
q_i=\operatorname{round}\left(\frac{r_i}{s}\right)
$$

并限制到 $[-127,127]$。

接下来还需要把浮点缩放因子 $s$ 表示成：

$$
s\approx\frac{a}{2^b}
$$

其中 $0\le a\le127$，$0\le b\le30$。

为了尽量提高缩放因子的表示精度，可以从 $b=30$ 开始向下寻找，第一个满足

$$
a=\operatorname{round}(s\cdot2^b)\le127
$$

的位置，就是所需的 $a,b$。

例如：

$$
s=\frac{3}{127}
$$

当 $b=12$ 时：

$$
a=\operatorname{round}\left(\frac{3}{127}\times4096\right)=97
$$

所以得到：

$$
s\approx\frac{97}{2^{12}}
$$

与样例一致。

对于结果 $C_{ij}$，$A$ 的第 $i$ 行和 $B$ 的第 $j$ 列分别表示为：

$$
A_i=\frac{a_{A_i}}{2^{b_{A_i}}}A_{q_i}
$$

$$
B_j=\frac{a_{B_j}}{2^{b_{B_j}}}B_{q_j}
$$

因此：

$$
C_{ij}
=

\frac{a_{A_i}a_{B_j}}
{2^{b_{A_i}+b_{B_j}}}
\left(A_{q_i}\cdot B_{q_j}\right)
$$

所以输出的三个整数直接为：

$$
q_C=A_{q_i}\cdot B_{q_j}
$$

$$
a_C=a_{A_i}a_{B_j}
$$

$$
b_C=b_{A_i}+b_{B_j}
$$

题目明确说明结果无需再次压缩到 int8，也无需对缩放因子进行约分。

实现时，Python 使用 `numpy`：

1. 对输入的 $B^T$ 按行计算最大绝对值；
2. 得到每一列的量化整数和整数缩放因子；
3. 用 `A @ Bq.T` 一次计算所有整数点积；
4. 组合对应的 $a$ 和 $b$ 输出结果。

如果某列全部为 $0$，直接令其量化结果全为 $0$，缩放因子记为 $0/2^0$。

## 复杂度分析

设矩阵 $A$ 大小为 $m\times k$，矩阵 $B$ 大小为 $k\times n$。

量化矩阵 $B$ 需要遍历全部 $kn$ 个元素，时间复杂度为：

$$
O(kn)
$$

矩阵乘法需要计算 $mn$ 个长度为 $k$ 的点积，时间复杂度为：

$$
O(mkn)
$$

因此总时间复杂度为：

$$
O(mkn)
$$

需要保存 $A$、$B$ 的量化结果以及结果矩阵，空间复杂度为：

$$
O(mk+nk+mn)
$$

## 代码实现

### Python

```python
import sys
import numpy as np


# 按题目规则进行四舍五入，避免不同语言的 round 行为不同
def rnd(x):
    return np.sign(x) * np.floor(np.abs(x) + 0.5)


# 把浮点缩放因子表示为 a / 2^b
def get_ab(s):
    if s == 0:
        return 0, 0

    for b in range(30, -1, -1):
        a = int(s * (1 << b) + 0.5)
        if a <= 127:
            return a, b

    return 0, 0


# 对 B 的转置按行量化，每一行就是 B 的一列
def quant_b(bt):
    n, k = bt.shape

    q = np.zeros((n, k), dtype=np.int64)
    aa = np.zeros(n, dtype=np.int64)
    bb = np.zeros(n, dtype=np.int64)

    for i in range(n):
        mx = np.max(np.abs(bt[i]))

        if mx == 0:
            continue

        s = mx / 127.0

        # 先按照真实 scale 完成 int8 量化
        x = rnd(bt[i] / s)
        x = np.clip(x, -127, 127)
        q[i] = x.astype(np.int64)

        # 再把 scale 本身量化为 a / 2^b
        aa[i], bb[i] = get_ab(s)

    return q, aa, bb


# 完成全整型矩阵乘法
def solve(aq, aa, ab, bt):
    bq, ba, bb = quant_b(bt)

    # aq: m*k，bq.T: k*n
    val = aq @ bq.T

    sa = aa[:, None] * ba[None, :]
    sb = ab[:, None] + bb[None, :]

    return val, sa, sb


def main():
    input = sys.stdin.readline

    m, k, n = map(int, input().split())

    aq = np.zeros((m, k), dtype=np.int64)
    aa = np.zeros(m, dtype=np.int64)
    ab = np.zeros(m, dtype=np.int64)

    for i in range(m):
        x = list(map(int, input().split()))
        aq[i] = x[:k]
        aa[i] = x[k]
        ab[i] = x[k + 1]

    # 题目实际按 B^T 输入，因此共有 n 行，每行 k 个数
    bt = np.zeros((n, k), dtype=np.float64)

    for i in range(n):
        bt[i] = list(map(float, input().split()))

    val, sa, sb = solve(aq, aa, ab, bt)

    for i in range(m):
        out = []
        for j in range(n):
            out.extend([
                str(val[i, j]),
                str(sa[i, j]),
                str(sb[i, j])
            ])
        print(" ".join(out))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    // 按题目要求完成普通四舍五入
    static long rnd(double x) {
        if (x >= 0) {
            return (long)(x + 0.5);
        } else {
            return (long)(x - 0.5);
        }
    }

    // 把浮点缩放因子表示为 a / 2^b
    static int[] getAB(double s) {
        if (s == 0) {
            return new int[]{0, 0};
        }

        for (int b = 30; b >= 0; b--) {
            double x = s * (1L << b);
            long a = rnd(x);

            if (a <= 127) {
                return new int[]{(int)a, b};
            }
        }

        return new int[]{0, 0};
    }

    // 对 B^T 按行量化
    static long[][] quantB(double[][] bt, int[] ba, int[] bb) {
        int n = bt.length;
        int k = bt[0].length;

        long[][] q = new long[n][k];

        for (int i = 0; i < n; i++) {
            double mx = 0;

            for (int j = 0; j < k; j++) {
                mx = Math.max(mx, Math.abs(bt[i][j]));
            }

            if (mx == 0) {
                ba[i] = 0;
                bb[i] = 0;
                continue;
            }

            double s = mx / 127.0;

            // 使用真实 scale 计算 int8 量化值
            for (int j = 0; j < k; j++) {
                long v = rnd(bt[i][j] / s);

                if (v > 127) v = 127;
                if (v < -127) v = -127;

                q[i][j] = v;
            }

            // 把 scale 表示为两个整数
            int[] t = getAB(s);
            ba[i] = t[0];
            bb[i] = t[1];
        }

        return q;
    }

    // 计算整数矩阵点积
    static long[][] mul(long[][] a, long[][] bt) {
        int m = a.length;
        int k = a[0].length;
        int n = bt.length;

        long[][] c = new long[m][n];

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                long sum = 0;

                for (int p = 0; p < k; p++) {
                    sum += a[i][p] * bt[j][p];
                }

                c[i][j] = sum;
            }
        }

        return c;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int m = sc.nextInt();
        int k = sc.nextInt();
        int n = sc.nextInt();

        long[][] a = new long[m][k];
        int[] aa = new int[m];
        int[] ab = new int[m];

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < k; j++) {
                a[i][j] = sc.nextInt();
            }

            aa[i] = sc.nextInt();
            ab[i] = sc.nextInt();
        }

        // 输入的是 B^T，共 n 行，每行 k 个浮点数
        double[][] bt = new double[n][k];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                bt[i][j] = sc.nextDouble();
            }
        }

        int[] ba = new int[n];
        int[] bb = new int[n];

        long[][] bq = quantB(bt, ba, bb);
        long[][] val = mul(a, bq);

        for (int i = 0; i < m; i++) {
            StringBuilder out = new StringBuilder();

            for (int j = 0; j < n; j++) {
                // 两个缩放因子相乘，分子相乘、指数相加
                long sa = (long)aa[i] * ba[j];
                int sb = ab[i] + bb[j];

                if (j > 0) {
                    out.append(' ');
                }

                out.append(val[i][j]).append(' ')
                   .append(sa).append(' ')
                   .append(sb);
            }

            System.out.println(out);
        }
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
using namespace std;


// 按题目要求进行普通四舍五入
long long rnd(double x) {
    if (x >= 0) {
        return (long long)(x + 0.5);
    } else {
        return (long long)(x - 0.5);
    }
}


// 把浮点缩放因子表示为 a / 2^b
pair<int, int> get_ab(double s) {
    if (s == 0) {
        return {0, 0};
    }

    for (int b = 30; b >= 0; b--) {
        long long a = rnd(s * (1LL << b));

        if (a <= 127) {
            return {(int)a, b};
        }
    }

    return {0, 0};
}


// 对 B^T 按行进行 8 bit 量化
vector<vector<long long>> quant_b(
    const vector<vector<double>>& bt,
    vector<int>& ba,
    vector<int>& bb
) {
    int n = bt.size();
    int k = bt[0].size();

    vector<vector<long long>> q(n, vector<long long>(k));

    for (int i = 0; i < n; i++) {
        double mx = 0;

        for (int j = 0; j < k; j++) {
            mx = max(mx, abs(bt[i][j]));
        }

        if (mx == 0) {
            ba[i] = 0;
            bb[i] = 0;
            continue;
        }

        double s = mx / 127.0;

        // 先按照真实 scale 计算 int8 量化值
        for (int j = 0; j < k; j++) {
            long long v = rnd(bt[i][j] / s);

            v = max(-127LL, min(127LL, v));
            q[i][j] = v;
        }

        // 再把 scale 自身量化为 a / 2^b
        auto t = get_ab(s);
        ba[i] = t.first;
        bb[i] = t.second;
    }

    return q;
}


// 计算两个量化整数矩阵之间的点积
vector<vector<long long>> mul(
    const vector<vector<long long>>& a,
    const vector<vector<long long>>& bt
) {
    int m = a.size();
    int k = a[0].size();
    int n = bt.size();

    vector<vector<long long>> c(m, vector<long long>(n));

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            long long sum = 0;

            for (int p = 0; p < k; p++) {
                sum += a[i][p] * bt[j][p];
            }

            c[i][j] = sum;
        }
    }

    return c;
}


int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int m, k, n;
    cin >> m >> k >> n;

    vector<vector<long long>> a(m, vector<long long>(k));
    vector<int> aa(m), ab(m);

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < k; j++) {
            cin >> a[i][j];
        }

        cin >> aa[i] >> ab[i];
    }

    // 输入数据实际按照 B^T 给出，共 n 行，每行 k 个元素
    vector<vector<double>> bt(n, vector<double>(k));

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < k; j++) {
            cin >> bt[i][j];
        }
    }

    vector<int> ba(n), bb(n);

    vector<vector<long long>> bq = quant_b(bt, ba, bb);
    vector<vector<long long>> val = mul(a, bq);

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            // 缩放因子直接相乘，不进行约分
            long long sa = 1LL * aa[i] * ba[j];
            int sb = ab[i] + bb[j];

            if (j > 0) {
                cout << ' ';
            }

            cout << val[i][j] << ' '
                 << sa << ' '
                 << sb;
        }

        cout << '\n';
    }

    return 0;
}
```
