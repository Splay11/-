## 解题思路

对于第 $i$ 个数据源，选择清洗等级 $l$ 后，会产生三类影响：

清洗等级的花费倍率为：

$$
cost = [0, 1, 3, 6]
$$

算力倍率为：

$$
power = [0, 1, 2, 3]
$$

清洗后的脏数据保留比例为：

$$
rate = [1, 0.6, 0.2, 0]
$$

因此第 $i$ 个数据源选择等级 $l$ 时：

$$
费用 = cost_l \times c_i
$$

$$
算力 = power_l \times t_i
$$

$$
污染贡献 = w_i \times r_i \times rate_l
$$

直接最小化污染率不太方便，可以换一个等价角度：先假设所有数据源都不清洗，则初始污染率为：

$$
base = \sum_{i=1}^{N} w_i r_i
$$

如果第 $i$ 个数据源选择等级 $l$，可以减少的污染率为：

$$
gain_{i,l} = w_i r_i (1 - rate_l)
$$

于是问题转化为：

在预算不超过 $B$、算力不超过 $T$ 的条件下，每个数据源选择一个清洗等级，使得总减少污染率最大。

这就是一个多重选择二维背包问题。

定义状态：

$$
dp[b][t]
$$

表示在当前已经处理过的数据源中，花费恰好为 $b$、算力恰好为 $t$ 时，可以获得的最大污染率减少量。

初始状态：

$$
dp[0][0] = 0
$$

其他状态为不可达。

对于每个数据源，都用一个新的数组 `ndp` 从旧的 `dp` 转移，保证同一个数据源只会选择一个清洗等级。

最终答案为：

$$
ans = base - \max dp[b][t]
$$

其中：

$$
0 \le b \le B,\quad 0 \le t \le T
$$

输出保留 $6$ 位小数即可。

## 复杂度分析

设清洗等级数量为 $4$。

时间复杂度为：

$$
O(N \times B \times T \times 4)
$$

由于清洗等级数量是常数，所以可以写成：

$$
O(NBT)
$$

空间复杂度为：

$$
O(BT)
$$

预算 $B$ 和算力 $T$ 都是整数上限，因此使用二维背包可以避免枚举所有 $4^N$ 种方案，复杂度是合适的。

## 代码实现

### Python

```python
import sys


def solve(n, B, T, items):
    # 四种清洗等级对应的花费倍率
    cost_mul = [0, 1, 3, 6]

    # 四种清洗等级对应的算力倍率
    power_mul = [0, 1, 2, 3]

    # 四种清洗等级对应的脏数据保留比例
    rate = [1.0, 0.6, 0.2, 0.0]

    # base 表示所有数据源都不清洗时的初始污染率
    base = 0.0

    # dp[b][t] 表示花费 b、算力 t 时可以获得的最大污染率减少量
    neg = -1e100
    dp = [[neg] * (T + 1) for _ in range(B + 1)]
    dp[0][0] = 0.0

    for w, r, c, p in items:
        origin = w * r
        base += origin

        # ndp 从旧 dp 转移，避免同一个数据源被重复选择多个等级
        ndp = [row[:] for row in dp]

        for lv in range(1, 4):
            cb = cost_mul[lv] * c
            ct = power_mul[lv] * p

            if cb <= B and ct <= T:
                gain = origin * (1.0 - rate[lv])

                # 从旧状态 dp 转移到新状态 ndp
                for b in range(B + 1 - cb):
                    old_row = dp[b]
                    new_row = ndp[b + cb]
                    for t in range(T + 1 - ct):
                        cand = old_row[t] + gain
                        if cand > new_row[t + ct]:
                            new_row[t + ct] = cand

        dp = ndp

    # 在所有不超过预算和算力上限的状态中取最大收益
    best = max(max(row) for row in dp)
    ans = base - best

    # 处理浮点误差导致的 -0.000000
    if -1e-9 < ans < 0:
        ans = 0.0

    return ans


def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    n = int(data[0])
    B = int(data[1])
    T = int(data[2])

    items = []
    idx = 3
    for _ in range(n):
        w = float(data[idx])
        r = float(data[idx + 1])
        c = int(data[idx + 2])
        p = int(data[idx + 3])
        items.append((w, r, c, p))
        idx += 4

    ans = solve(n, B, T, items)
    print(f"{ans:.6f}")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static double solve(int n, int B, int T, double[] w, double[] r, int[] c, int[] p) {
        // 四种清洗等级对应的花费倍率
        int[] costMul = {0, 1, 3, 6};

        // 四种清洗等级对应的算力倍率
        int[] powerMul = {0, 1, 2, 3};

        // 四种清洗等级对应的脏数据保留比例
        double[] rate = {1.0, 0.6, 0.2, 0.0};

        // base 表示所有数据源都不清洗时的初始污染率
        double base = 0.0;

        // dp[b][t] 表示花费 b、算力 t 时可以获得的最大污染率减少量
        double neg = -1e100;
        double[][] dp = new double[B + 1][T + 1];

        for (int i = 0; i <= B; i++) {
            Arrays.fill(dp[i], neg);
        }
        dp[0][0] = 0.0;

        for (int i = 0; i < n; i++) {
            double origin = w[i] * r[i];
            base += origin;

            double[][] ndp = new double[B + 1][T + 1];

            // 先复制旧状态，表示当前数据源选择 0 级，也就是不清洗
            for (int b = 0; b <= B; b++) {
                System.arraycopy(dp[b], 0, ndp[b], 0, T + 1);
            }

            for (int lv = 1; lv <= 3; lv++) {
                int cb = costMul[lv] * c[i];
                int ct = powerMul[lv] * p[i];

                if (cb > B || ct > T) {
                    continue;
                }

                double gain = origin * (1.0 - rate[lv]);

                // 从旧 dp 转移到 ndp，保证当前数据源只选择一个清洗等级
                for (int b = 0; b + cb <= B; b++) {
                    for (int t = 0; t + ct <= T; t++) {
                        if (dp[b][t] <= neg / 2) {
                            continue;
                        }

                        double val = dp[b][t] + gain;
                        if (val > ndp[b + cb][t + ct]) {
                            ndp[b + cb][t + ct] = val;
                        }
                    }
                }
            }

            dp = ndp;
        }

        // 在所有不超过预算和算力上限的状态中取最大收益
        double best = 0.0;
        for (int b = 0; b <= B; b++) {
            for (int t = 0; t <= T; t++) {
                if (dp[b][t] > best) {
                    best = dp[b][t];
                }
            }
        }

        double ans = base - best;

        // 处理浮点误差导致的 -0.000000
        if (ans < 0 && ans > -1e-9) {
            ans = 0.0;
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int B = sc.nextInt();
        int T = sc.nextInt();

        double[] w = new double[n];
        double[] r = new double[n];
        int[] c = new int[n];
        int[] p = new int[n];

        for (int i = 0; i < n; i++) {
            w[i] = sc.nextDouble();
            r[i] = sc.nextDouble();
            c[i] = sc.nextInt();
            p[i] = sc.nextInt();
        }

        double ans = solve(n, B, T, w, r, c, p);
        System.out.printf("%.6f\n", ans);

        sc.close();
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Item {
    double w;
    double r;
    int c;
    int p;
};

double solve(int n, int B, int T, vector<Item>& items) {
    // 四种清洗等级对应的花费倍率
    int costMul[4] = {0, 1, 3, 6};

    // 四种清洗等级对应的算力倍率
    int powerMul[4] = {0, 1, 2, 3};

    // 四种清洗等级对应的脏数据保留比例
    double rate[4] = {1.0, 0.6, 0.2, 0.0};

    // base 表示所有数据源都不清洗时的初始污染率
    double base = 0.0;

    // dp[b][t] 表示花费 b、算力 t 时可以获得的最大污染率减少量
    double neg = -1e100;
    vector<vector<double>> dp(B + 1, vector<double>(T + 1, neg));
    dp[0][0] = 0.0;

    for (int i = 0; i < n; i++) {
        double origin = items[i].w * items[i].r;
        base += origin;

        // 先复制旧状态，表示当前数据源选择 0 级，也就是不清洗
        vector<vector<double>> ndp = dp;

        for (int lv = 1; lv <= 3; lv++) {
            int cb = costMul[lv] * items[i].c;
            int ct = powerMul[lv] * items[i].p;

            if (cb > B || ct > T) {
                continue;
            }

            double gain = origin * (1.0 - rate[lv]);

            // 从旧 dp 转移到 ndp，保证当前数据源只选择一个清洗等级
            for (int b = 0; b + cb <= B; b++) {
                for (int t = 0; t + ct <= T; t++) {
                    if (dp[b][t] <= neg / 2) {
                        continue;
                    }

                    double val = dp[b][t] + gain;
                    if (val > ndp[b + cb][t + ct]) {
                        ndp[b + cb][t + ct] = val;
                    }
                }
            }
        }

        dp = ndp;
    }

    // 在所有不超过预算和算力上限的状态中取最大收益
    double best = 0.0;
    for (int b = 0; b <= B; b++) {
        for (int t = 0; t <= T; t++) {
            best = max(best, dp[b][t]);
        }
    }

    double ans = base - best;

    // 处理浮点误差导致的 -0.000000
    if (ans < 0 && ans > -1e-9) {
        ans = 0.0;
    }

    return ans;
}

int main() {
    int n, B, T;
    cin >> n >> B >> T;

    vector<Item> items(n);

    for (int i = 0; i < n; i++) {
        cin >> items[i].w >> items[i].r >> items[i].c >> items[i].p;
    }

    double ans = solve(n, B, T, items);

    cout << fixed << setprecision(6) << ans << '\n';

    return 0;
}
```
