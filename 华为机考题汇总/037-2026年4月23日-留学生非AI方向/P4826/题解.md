## 解题思路

先把每个版本号拆成两部分：

* 主版本号：按 $.$ 分割得到若干组整数
* $beta$ 版本号：如果存在，则格式一定是 $beta数字$

比较两个版本号时，按照题意分两步：

### 1. 比较主版本号

这是一个典型的“自定义排序”问题，核心算法就是**排序 + 自定义比较规则**。

主版本号比较方法：

1. 先将主版本号按 $.$ 分割成整数数组
2. 从左到右逐组比较
3. 某一组不同，则该组数字较小的版本更小
4. 如果前面的组都相同，但组数不同，则**组数少的更小**
   例如：$6.1 < 6.1.0.0$

这部分本质上就是**按数组字典序比较**，如果前缀完全相同，再比较长度。

### 2. 主版本号相同，再比较 $beta$

当主版本号完全相同时：

1. 有 $beta$ 的版本小于正式版
   即：$6.0\ beta1 < 6.0$
2. 如果两个都是 $beta$ 版本，则比较后面的数字
   即：$beta1 < beta2$
3. 如果两个都是正式版，则相等

### 3. 实现方法

每个版本号可以解析成一个统一的排序键：

* 主版本号整数数组
* 是否为正式版
* 若是 $beta$，对应的 $beta$ 编号

为了方便排序，可以直接写比较函数：

* 先比较主版本号数组
* 再比较是否为正式版
* 最后比较 $beta$ 编号

然后对所有字符串排序后输出即可。

## 复杂度分析

设版本号个数为 $n$，每个版本号最多有 $4$ 组主版本号，因此单次比较的代价是 $O(1)$。

排序总复杂度为：

* 时间复杂度：$O(n \log n)$
* 空间复杂度：$O(n)$

在 $n \le 100$ 的范围内，复杂度完全合适。

## 代码实现

### Python

```python
import sys
from functools import cmp_to_key


# 解析版本号，返回：主版本数组、是否为正式版、beta编号
def parse_version(s):
    parts = s.strip().split()
    main_nums = list(map(int, parts[0].split('.')))
    
    # 默认是正式版
    is_release = True
    beta_num = 0
    
    # 有 beta 版本
    if len(parts) > 1:
        is_release = False
        beta_num = int(parts[1][4:])  # 去掉前缀 beta
    
    return main_nums, is_release, beta_num


# 比较两个版本号
def compare(a, b):
    main_a, release_a, beta_a = parse_version(a)
    main_b, release_b, beta_b = parse_version(b)
    
    # 先比较主版本号
    len_a = len(main_a)
    len_b = len(main_b)
    min_len = min(len_a, len_b)
    
    for i in range(min_len):
        if main_a[i] != main_b[i]:
            return main_a[i] - main_b[i]
    
    # 前面都相同，组数少的更小
    if len_a != len_b:
        return len_a - len_b
    
    # 主版本号相同，再比较 beta
    # beta 版本小于正式版
    if release_a != release_b:
        return 1 if release_a else -1
    
    # 都是 beta，比较 beta 编号
    if not release_a:
        return beta_a - beta_b
    
    return 0


def solve(versions):
    versions.sort(key=cmp_to_key(compare))
    return versions


def main():
    input = sys.stdin.readline
    n = int(input().strip())
    versions = [input().strip() for _ in range(n)]
    
    ans = solve(versions)
    for v in ans:
        print(v)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

public class Main {

    // 版本号信息类
    static class Version {
        List<Integer> mainNums; // 主版本号数组
        boolean isRelease;      // 是否为正式版
        int betaNum;            // beta编号
        String raw;             // 原始字符串

        Version(String s) {
            raw = s;
            mainNums = new ArrayList<>();
            isRelease = true;
            betaNum = 0;

            String[] parts = s.split(" ");
            String[] nums = parts[0].split("\\.");
            for (String num : nums) {
                mainNums.add(Integer.parseInt(num));
            }

            // 如果有 beta 版本
            if (parts.length > 1) {
                isRelease = false;
                betaNum = Integer.parseInt(parts[1].substring(4)); // 去掉 beta
            }
        }
    }

    // 比较两个版本号
    static int compareVersion(Version a, Version b) {
        // 先比较主版本号
        int lenA = a.mainNums.size();
        int lenB = b.mainNums.size();
        int minLen = Math.min(lenA, lenB);

        for (int i = 0; i < minLen; i++) {
            int x = a.mainNums.get(i);
            int y = b.mainNums.get(i);
            if (x != y) {
                return x - y;
            }
        }

        // 前缀都相同，组数少的更小
        if (lenA != lenB) {
            return lenA - lenB;
        }

        // 主版本号相同，比较 beta
        // beta 版本小于正式版
        if (a.isRelease != b.isRelease) {
            return a.isRelease ? 1 : -1;
        }

        // 都是 beta，比较 beta 编号
        if (!a.isRelease) {
            return a.betaNum - b.betaNum;
        }

        return 0;
    }

    // 排序函数
    static List<String> solve(List<String> versions) {
        List<Version> list = new ArrayList<>();
        for (String s : versions) {
            list.add(new Version(s));
        }

        // 按题意排序
        Collections.sort(list, new Comparator<Version>() {
            @Override
            public int compare(Version a, Version b) {
                return compareVersion(a, b);
            }
        });

        List<String> ans = new ArrayList<>();
        for (Version v : list) {
            ans.add(v.raw);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        int n = Integer.parseInt(sc.nextLine());
        List<String> versions = new ArrayList<>();
        
        for (int i = 0; i < n; i++) {
            versions.add(sc.nextLine());
        }

        List<String> ans = solve(versions);
        for (String s : ans) {
            System.out.println(s);
        }
        
        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <sstream>
#include <algorithm>

using namespace std;

// 版本号结构体
struct Version {
    vector<int> mainNums; // 主版本号数组
    bool isRelease;       // 是否为正式版
    int betaNum;          // beta编号
    string raw;           // 原始字符串
};

// 解析版本号
Version parseVersion(const string& s) {
    Version v;
    v.raw = s;
    v.isRelease = true;
    v.betaNum = 0;

    size_t pos = s.find(' ');
    string mainPart, betaPart;

    // 判断是否存在 beta
    if (pos == string::npos) {
        mainPart = s;
    } else {
        mainPart = s.substr(0, pos);
        betaPart = s.substr(pos + 1);
        v.isRelease = false;
        v.betaNum = stoi(betaPart.substr(4)); // 去掉 beta
    }

    // 按 '.' 分割主版本号
    stringstream ss(mainPart);
    string num;
    while (getline(ss, num, '.')) {
        v.mainNums.push_back(stoi(num));
    }

    return v;
}

// 比较函数
bool cmp(const Version& a, const Version& b) {
    // 先比较主版本号
    int lenA = a.mainNums.size();
    int lenB = b.mainNums.size();
    int minLen = min(lenA, lenB);

    for (int i = 0; i < minLen; i++) {
        if (a.mainNums[i] != b.mainNums[i]) {
            return a.mainNums[i] < b.mainNums[i];
        }
    }

    // 前缀相同，组数少的更小
    if (lenA != lenB) {
        return lenA < lenB;
    }

    // 主版本号相同，beta 版本小于正式版
    if (a.isRelease != b.isRelease) {
        return !a.isRelease;
    }

    // 都是 beta，比较 beta 编号
    if (!a.isRelease) {
        return a.betaNum < b.betaNum;
    }

    return false;
}

// 排序函数
vector<string> solve(const vector<string>& versions) {
    vector<Version> list;
    for (const string& s : versions) {
        list.push_back(parseVersion(s));
    }

    // 自定义排序
    sort(list.begin(), list.end(), cmp);

    vector<string> ans;
    for (const auto& v : list) {
        ans.push_back(v.raw);
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    cin.ignore(); // 忽略换行

    vector<string> versions(n);
    for (int i = 0; i < n; i++) {
        getline(cin, versions[i]);
    }

    vector<string> ans = solve(versions);
    for (const string& s : ans) {
        cout << s << '\n';
    }

    return 0;
}
```
