## 解题思路

本题要求统计连续浏览片段中，不同商品类别数不超过 $k$ 的片段数量。

使用滑动窗口算法。

维护一个窗口 $[left, right]$，表示当前连续片段。用哈希表记录窗口内每种类别出现的次数，同时维护不同类别数量。

每次将 $nums[right]$ 加入窗口：

如果窗口内不同类别数超过 $k$，就不断移动 $left$，缩小窗口，直到不同类别数不超过 $k$。

此时，以 $right$ 结尾的合法片段数量为：

$right - left + 1$

因为这些片段分别是：

$[right, right]$、$[right - 1, right]$、...、$[left, right]$

把每个位置作为右端点的合法片段数累加，即为答案。

## 复杂度分析

时间复杂度：$O(n)$

每个元素最多进入窗口一次、离开窗口一次。

空间复杂度：$O(n)$

哈希表最多存储 $n$ 种不同类别。

## 代码实现

### Python

```python
def count_segments(nums, k):
    cnt = {}
    left = 0
    ans = 0

    for right in range(len(nums)):
        # 将右端点商品加入窗口
        x = nums[right]
        cnt[x] = cnt.get(x, 0) + 1

        # 若不同类别数超过 k，移动左端点
        while len(cnt) > k:
            y = nums[left]
            cnt[y] -= 1
            if cnt[y] == 0:
                del cnt[y]
            left += 1

        # 以 right 结尾的合法片段数量
        ans += right - left + 1

    return ans


def main():
    n, k = map(int, input().split())
    nums = list(map(int, input().split()))

    print(count_segments(nums, k))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 统计不同类别数不超过 k 的连续片段数量
    public static long countSegments(int[] nums, int k) {
        Map<Integer, Integer> cnt = new HashMap<>();
        int left = 0;
        long ans = 0;

        for (int right = 0; right < nums.length; right++) {
            // 将右端点商品加入窗口
            int x = nums[right];
            cnt.put(x, cnt.getOrDefault(x, 0) + 1);

            // 若不同类别数超过 k，移动左端点
            while (cnt.size() > k) {
                int y = nums[left];
                cnt.put(y, cnt.get(y) - 1);
                if (cnt.get(y) == 0) {
                    cnt.remove(y);
                }
                left++;
            }

            // 以 right 结尾的合法片段数量
            ans += right - left + 1;
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());

        int[] nums = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            nums[i] = Integer.parseInt(st.nextToken());
        }

        System.out.println(countSegments(nums, k));
    }
}
```

### C++

```cpp
#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;

// 统计不同类别数不超过 k 的连续片段数量
long long countSegments(vector<int>& nums, int k) {
    unordered_map<int, int> cnt;
    int left = 0;
    long long ans = 0;

    for (int right = 0; right < (int)nums.size(); right++) {
        // 将右端点商品加入窗口
        int x = nums[right];
        cnt[x]++;

        // 若不同类别数超过 k，移动左端点
        while ((int)cnt.size() > k) {
            int y = nums[left];
            cnt[y]--;
            if (cnt[y] == 0) {
                cnt.erase(y);
            }
            left++;
        }

        // 以 right 结尾的合法片段数量
        ans += right - left + 1;
    }

    return ans;
}

int main() {
    int n, k;
    cin >> n >> k;

    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    cout << countSegments(nums, k) << endl;

    return 0;
}
```
