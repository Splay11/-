## 解题思路

这题本质上是一个模拟题，可以用两个栈配合一个当前页面来实现浏览器地址栏功能。

相关算法：$栈$模拟。

核心思路如下：

* 用一个栈 `back_stack` 表示当前页面之前的历史页面；
* 用一个栈 `forward_stack` 表示可以前进到的页面；
* 用变量 `current` 表示当前页面，初始为 `Blank`。

然后按操作要求模拟：

1. 对于 $visit\ url$

   * 先把当前页面加入 `back_stack`
   * 再把当前页面更新为新访问的 $url$
   * 清空 `forward_stack`
   * 由于最多只能保存 `max_history` 个历史记录，而总记录数等于 `back_stack` 中页面数加上当前页面这 $1$ 个，所以当
     `len(back_stack) + 1 > max_history`
     时，需要删除最早的历史记录，也就是删除 `back_stack` 的栈底元素

2. 对于 $back$

   * 只有当 `back_stack` 不为空时才能后退
   * 先把当前页面压入 `forward_stack`
   * 再弹出 `back_stack` 栈顶作为新的当前页面

3. 对于 $forward$

   * 只有当 `forward_stack` 不为空时才能前进
   * 先把当前页面压入 `back_stack`
   * 再弹出 `forward_stack` 栈顶作为新的当前页面
   * 前进后同样要检查历史总数是否超过 `max_history`，如果超过则删除最早记录

4. 对于 $print$

   * 直接输出当前页面即可

实现时，删除最早历史记录可以直接删除数组下标为 $0$ 的元素。由于数据范围很小，$n \le 200$，这样做完全可以通过。

## 复杂度分析

设操作数为 $n$。

* 时间复杂度：$O(n \cdot max_history)$
  因为普通操作基本是 $O(1)$，但删除最早记录时可能需要数组整体移动，单次最坏为 `O(max_history)`。由于数据范围很小，这个复杂度完全合适。
* 空间复杂度：`O(max_history)`
  历史记录和前进记录总量受到操作规模限制，实际保存的历史页面数量不会超过要求范围。

## 代码实现

### Python

```python
def solve_browser(n, max_history, ops):
    # 返回所有 print 操作的输出结果
    back_stack = []      # 历史栈，存当前页面之前的页面
    forward_stack = []   # 前进栈
    current = "Blank"    # 当前页面

    ans = []

    for op in ops:
        parts = op.split()

        if parts[0] == "visit":
            url = parts[1]

            # 当前页面进入历史栈
            back_stack.append(current)
            # 更新当前页面
            current = url
            # 访问新页面后清空前进栈
            forward_stack.clear()

            # 若总历史数量超过上限，删除最早记录
            if len(back_stack) + 1 > max_history:
                back_stack.pop(0)

        elif parts[0] == "back":
            # 有历史页面才能后退
            if back_stack:
                forward_stack.append(current)
                current = back_stack.pop()

        elif parts[0] == "forward":
            # 有前进页面才能前进
            if forward_stack:
                back_stack.append(current)
                current = forward_stack.pop()

                # 前进后也要检查历史容量
                if len(back_stack) + 1 > max_history:
                    back_stack.pop(0)

        else:  # print
            ans.append(current)

    return ans


def main():
    # 输入操作
    n = int(input().strip())
    max_history = int(input().strip())
    ops = [input().strip() for _ in range(n)]

    # 调用外部函数处理
    res = solve_browser(n, max_history, ops)

    # 输出结果
    for x in res:
        print(x)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {

    // 处理浏览器操作，返回所有 print 的结果
    public static List<String> solveBrowser(int n, int maxHistory, List<String> ops) {
        List<String> backStack = new ArrayList<>();     // 历史栈
        List<String> forwardStack = new ArrayList<>();  // 前进栈
        String current = "Blank";                       // 当前页面

        List<String> ans = new ArrayList<>();

        for (String op : ops) {
            String[] parts = op.split(" ");

            if (parts[0].equals("visit")) {
                String url = parts[1];

                // 当前页面加入历史栈
                backStack.add(current);
                // 更新当前页面
                current = url;
                // 清空前进栈
                forwardStack.clear();

                // 超过容量则删除最早记录
                if (backStack.size() + 1 > maxHistory) {
                    backStack.remove(0);
                }

            } else if (parts[0].equals("back")) {
                // 有历史页面才能后退
                if (!backStack.isEmpty()) {
                    forwardStack.add(current);
                    current = backStack.remove(backStack.size() - 1);
                }

            } else if (parts[0].equals("forward")) {
                // 有前进页面才能前进
                if (!forwardStack.isEmpty()) {
                    backStack.add(current);
                    current = forwardStack.remove(forwardStack.size() - 1);

                    // 前进后也要检查容量
                    if (backStack.size() + 1 > maxHistory) {
                        backStack.remove(0);
                    }
                }

            } else { // print
                ans.add(current);
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        // 输入操作
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine());
        int maxHistory = Integer.parseInt(sc.nextLine());

        List<String> ops = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            ops.add(sc.nextLine());
        }

        // 调用外部函数处理
        List<String> res = solveBrowser(n, maxHistory, ops);

        // 输出结果
        for (String s : res) {
            System.out.println(s);
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// 处理浏览器操作，返回所有 print 的结果
vector<string> solveBrowser(int n, int max_history, const vector<string>& ops) {
    vector<string> back_stack;      // 历史栈
    vector<string> forward_stack;   // 前进栈
    string current = "Blank";       // 当前页面

    vector<string> ans;

    for (string op : ops) {
        if (op.substr(0, 5) == "visit") {
            // 提取访问的 url
            string url = op.substr(6);

            // 当前页面加入历史栈
            back_stack.push_back(current);
            // 更新当前页面
            current = url;
            // 清空前进栈
            forward_stack.clear();

            // 超过容量则删除最早记录
            if ((int)back_stack.size() + 1 > max_history) {
                back_stack.erase(back_stack.begin());
            }

        } else if (op == "back") {
            // 有历史页面才能后退
            if (!back_stack.empty()) {
                forward_stack.push_back(current);
                current = back_stack.back();
                back_stack.pop_back();
            }

        } else if (op == "forward") {
            // 有前进页面才能前进
            if (!forward_stack.empty()) {
                back_stack.push_back(current);
                current = forward_stack.back();
                forward_stack.pop_back();

                // 前进后也要检查容量
                if ((int)back_stack.size() + 1 > max_history) {
                    back_stack.erase(back_stack.begin());
                }
            }

        } else { // print
            ans.push_back(current);
        }
    }

    return ans;
}

int main() {
    // 输入操作
    int n, max_history;
    cin >> n >> max_history;
    cin.ignore();

    vector<string> ops(n);
    for (int i = 0; i < n; i++) {
        getline(cin, ops[i]);
    }

    // 调用外部函数处理
    vector<string> res = solveBrowser(n, max_history, ops);

    // 输出结果
    for (string s : res) {
        cout << s << '\n';
    }

    return 0;
}
```
