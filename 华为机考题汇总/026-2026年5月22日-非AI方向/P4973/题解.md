## 解题思路

由于 $win_size[i] \le 3$，配置节点 $i$ 的影响范围只会覆盖区间 $[i,\min(N-1,i+win_size[i])]$，也就是说一个节点 $i$ 的延迟最多只会被它前面 $3$ 个节点以及它自己影响。

因此可以使用状态压缩动态规划。

设 $dp[c][mask]$ 表示当前处理到某个节点前，已经选择了 $c$ 个优先转发节点，最近 $3$ 个节点的选择状态为 $mask$ 时的最小总延迟。

处理节点 $i$ 时，有两种选择：

1. 不配置节点 $i$
2. 配置节点 $i$，前提是已选数量不超过 $K$

计算节点 $i$ 的实际延迟时，只需要检查：

* 最近 $3$ 个已处理节点是否被选中
* 它们的影响范围是否覆盖节点 $i$
* 当前节点 $i$ 是否被选中

设总优化值为 $sum$，则节点 $i$ 的实际延迟为：

$max(0, delay[i]-sum)$

不断转移即可得到答案。

## 复杂度分析

状态数量为 $K \times 2^3$，每个节点枚举选或不选两种情况。

时间复杂度为 $O(N \times K \times 2^3)$。

空间复杂度为 $O(K \times 2^3)$。

该复杂度对于 $N \le 50$、$K \le 10$ 完全合适。

## 代码实现

### Python

```python
import sys

def optimize_network(n, k, delay, win_size, opt_delay):
    INF = 10 ** 30

    # dp[c][mask]：已选 c 个节点，最近 3 个节点选择状态为 mask 的最小总延迟
    dp = [[INF] * 8 for _ in range(k + 1)]
    dp[0][0] = 0

    for i in range(n):
        ndp = [[INF] * 8 for _ in range(k + 1)]

        for c in range(k + 1):
            for mask in range(8):
                if dp[c][mask] == INF:
                    continue

                # choose 表示当前节点 i 是否配置优先转发模式
                for choose in range(2):
                    if c + choose > k:
                        continue

                    reduce_sum = 0

                    # 检查前面最多 3 个节点是否影响当前节点 i
                    for d in range(3):
                        if (mask >> d) & 1:
                            pre = i - 1 - d
                            if pre >= 0 and pre + win_size[pre] >= i:
                                reduce_sum += opt_delay[pre]

                    # 如果当前节点被选择，它一定影响自己
                    if choose == 1:
                        reduce_sum += opt_delay[i]

                    # 当前节点的实际处理延迟
                    cur_delay = delay[i] - reduce_sum
                    if cur_delay < 0:
                        cur_delay = 0

                    # 更新最近 3 个节点的选择状态
                    new_mask = ((mask << 1) & 7) | choose

                    if dp[c][mask] + cur_delay < ndp[c + choose][new_mask]:
                        ndp[c + choose][new_mask] = dp[c][mask] + cur_delay

        dp = ndp

    ans = INF
    for c in range(k + 1):
        for mask in range(8):
            ans = min(ans, dp[c][mask])

    return ans


def main():
    data = list(map(int, sys.stdin.read().split()))
    n, k = data[0], data[1]
    delay = data[2:2 + n]
    win_size = data[2 + n:2 + 2 * n]
    opt_delay = data[2 + 2 * n:2 + 3 * n]

    print(optimize_network(n, k, delay, win_size, opt_delay))


if __name__ == "__main__":
    main()
```

```java
import java.util.*;

public class Main {
    static long optimizeNetwork(int n, int k, int[] delay, int[] winSize, int[] optDelay) {
        long INF = Long.MAX_VALUE / 4;

        // dp[c][mask]：已选 c 个节点，最近 3 个节点选择状态为 mask 的最小总延迟
        long[][] dp = new long[k + 1][8];
        for (int i = 0; i <= k; i++) {
            Arrays.fill(dp[i], INF);
        }
        dp[0][0] = 0;

        for (int i = 0; i < n; i++) {
            long[][] ndp = new long[k + 1][8];
            for (int c = 0; c <= k; c++) {
                Arrays.fill(ndp[c], INF);
            }

            for (int c = 0; c <= k; c++) {
                for (int mask = 0; mask < 8; mask++) {
                    if (dp[c][mask] == INF) {
                        continue;
                    }

                    // choose 表示当前节点 i 是否配置优先转发模式
                    for (int choose = 0; choose <= 1; choose++) {
                        if (c + choose > k) {
                            continue;
                        }

                        long reduceSum = 0;

                        // 检查前面最多 3 个节点是否影响当前节点 i
                        for (int d = 0; d < 3; d++) {
                            if (((mask >> d) & 1) == 1) {
                                int pre = i - 1 - d;
                                if (pre >= 0 && pre + winSize[pre] >= i) {
                                    reduceSum += optDelay[pre];
                                }
                            }

                        // 如果当前节点被选择，它一定影响自己
                        if (choose == 1) {
                            reduceSum += optDelay[i];
                        }

                        // 当前节点的实际处理延迟
                        long curDelay = delay[i] - reduceSum;
                        if (curDelay < 0) {
                            curDelay = 0;

                        int newMask = ((mask << 1) & 7) | choose;

                        long newValue = dp[c][mask] + curDelay;
                        if (newValue < ndp[c + choose][newMask]) {
                            ndp[c + choose][newMask] = newValue;
                        }

                }

            for (int mask = 0; mask < 8; mask++) {
                ans = Math.min(ans, dp[c][mask]);
            }
        }
```
