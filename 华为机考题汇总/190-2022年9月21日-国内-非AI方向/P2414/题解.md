```cpp
#include <iostream>
#include <algorithm>
using namespace std;

const int N = 100000 + 10;
int arr[N];
int n;
int tot;

bool isNum(int mid)
{
    long long sum = 0;
    for(int i = 0;i < n;i ++)
    {
        sum += min(mid, arr[i]);
        if(sum > tot) return false;
    }
    if(sum <= tot) return true;
    else return false;
}

int main()
{
    
    cin >> n >> tot;
    long long sum = 0;
    int maxv = 0;
    for(int i = 0;i < n;i ++)
    {
        cin >> arr[i];
        sum += arr[i];
        maxv = max(maxv, arr[i]);
    }
    if(sum <= tot)
    {
        cout << "-1";
        return 0;
    }
    int l = tot / n;
    int r = maxv;
    while(l < r)
    {
        int mid = l + (r-l+1)/2;
        if(isNum(mid)) l = mid;
        else r = mid - 1;
    }
    cout << l << endl;

    return 0;
}
```
