## 解题思路

本题可以直接使用队列模拟。

把大厅中的等待顾客按顺序放入队列，窗口用数组记录当前正在服务的顾客编号。

核心规则如下：

1. 开始时，将队列前面的顾客依次分配给空窗口，优先分配编号小的窗口。
2. 每经过 $1$ 秒，每个非空窗口中的顾客买到 $1$ 笼包子。
3. 如果顾客还没买够，就回到队尾。
4. 如果顾客已经买够，就直接离开。
5. 同一秒内，先处理所有窗口服务完成的顾客，再重新给空窗口分配大厅队首顾客。
6. 当编号为 $k$ 的顾客买完最后一笼包子时，当前时间就是答案。

由于最多只有 $101$ 位顾客，每位最多买 $100$ 笼，直接模拟总服务次数即可，算法简单且不会超时。

## 复杂度分析

设顾客总共需要购买的包子笼数为 $S$。

每秒最多处理 $m$ 个窗口，但每次处理都会让某位顾客的需求减少 $1$，因此总处理次数为 $S$。

时间复杂度为 $O(S)$。

队列和窗口数组最多存储 $n+m$ 个元素。

空间复杂度为 $O(n+m)$。

## 代码实现

### Python

```python
from collections import deque

def solve(baskets, k, m):
    n = len(baskets)
    rest = baskets[:]              # 记录每位顾客还需要买几笼
    q = deque(range(n))            # 大厅等待队列
    windows = [-1] * m             # 每个窗口当前服务的顾客编号，-1 表示空

    # 初始分配顾客到窗口
    for i in range(m):
        if q:
            windows[i] = q.popleft()

    time = 0

    while True:
        time += 1

        # 当前这一秒内，所有非空窗口各服务一位顾客一次
        for i in range(m):
            if windows[i] != -1:
                cid = windows[i]
                rest[cid] -= 1

        # 处理服务结束后的顾客
        for i in range(m):
            if windows[i] != -1:
                cid = windows[i]

                # 目标顾客买完，直接返回答案
                if cid == k and rest[cid] == 0:
                    return time

                # 没买够的顾客回到队尾
                if rest[cid] > 0:
                    q.append(cid)

                # 当前窗口变为空
                windows[i] = -1

        # 给空窗口重新分配顾客，优先编号小的窗口
        for i in range(m):
            if windows[i] == -1 and q:
                windows[i] = q.popleft()


if __name__ == "__main__":
    baskets = list(map(int, input().split()))
    k = int(input())
    m = int(input())
    print(solve(baskets, k, m))
```

### Java

```java
import java.util.*;

public class Main {
    public static int solve(int[] baskets, int k, int m) {
        int n = baskets.length;
        int[] rest = baskets.clone();       // 记录每位顾客还需要买几笼
        Queue<Integer> queue = new LinkedList<>();

        // 大厅等待队列
        for (int i = 0; i < n; i++) {
            queue.offer(i);
        }

        int[] windows = new int[m];         // 每个窗口当前服务的顾客编号
        Arrays.fill(windows, -1);           // -1 表示窗口为空

        // 初始分配顾客到窗口
        for (int i = 0; i < m; i++) {
            if (!queue.isEmpty()) {
                windows[i] = queue.poll();
            }
        }

        int time = 0;

        while (true) {
            time++;

            // 当前这一秒内，所有非空窗口各服务一位顾客一次
            for (int i = 0; i < m; i++) {
                if (windows[i] != -1) {
                    int cid = windows[i];
                    rest[cid]--;
                }
            }

            // 处理服务结束后的顾客
            for (int i = 0; i < m; i++) {
                if (windows[i] != -1) {
                    int cid = windows[i];

                    // 目标顾客买完，直接返回答案
                    if (cid == k && rest[cid] == 0) {
                        return time;
                    }

                    // 没买够的顾客回到队尾
                    if (rest[cid] > 0) {
                        queue.offer(cid);
                    }

                    // 当前窗口变为空
                    windows[i] = -1;
                }
            }

            // 给空窗口重新分配顾客，优先编号小的窗口
            for (int i = 0; i < m; i++) {
                if (windows[i] == -1 && !queue.isEmpty()) {
                    windows[i] = queue.poll();
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 第一行是长度不固定的数组，所以先读取整行
        String line = sc.nextLine();
        String[] parts = line.trim().split("\\s+");

        int[] baskets = new int[parts.length];
        for (int i = 0; i < parts.length; i++) {
            baskets[i] = Integer.parseInt(parts[i]);
        }

        int k = sc.nextInt();
        int m = sc.nextInt();

        System.out.println(solve(baskets, k, m));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

int solve(vector<int>& baskets, int k, int m) {
    int n = baskets.size();
    vector<int> rest = baskets;        // 记录每位顾客还需要买几笼
    queue<int> q;                      // 大厅等待队列

    for (int i = 0; i < n; i++) {
        q.push(i);
    }

    vector<int> windows(m, -1);        // 每个窗口当前服务的顾客编号，-1 表示空

    // 初始分配顾客到窗口
    for (int i = 0; i < m; i++) {
        if (!q.empty()) {
            windows[i] = q.front();
            q.pop();
        }
    }

    int time = 0;

    while (true) {
        time++;

        // 当前这一秒内，所有非空窗口各服务一位顾客一次
        for (int i = 0; i < m; i++) {
            if (windows[i] != -1) {
                int cid = windows[i];
                rest[cid]--;
            }
        }

        // 处理服务结束后的顾客
        for (int i = 0; i < m; i++) {
            if (windows[i] != -1) {
                int cid = windows[i];

                // 目标顾客买完，直接返回答案
                if (cid == k && rest[cid] == 0) {
                    return time;
                }

                // 没买够的顾客回到队尾
                if (rest[cid] > 0) {
                    q.push(cid);
                }

                // 当前窗口变为空
                windows[i] = -1;
            }
        }

        // 给空窗口重新分配顾客，优先编号小的窗口
        for (int i = 0; i < m; i++) {
            if (windows[i] == -1 && !q.empty()) {
                windows[i] = q.front();
                q.pop();
            }
        }
    }
}

int main() {
    vector<int> baskets;
    string line;
    getline(cin, line);

    stringstream ss(line);
    int x;
    while (ss >> x) {
        baskets.push_back(x);
    }

    int k, m;
    cin >> k >> m;

    cout << solve(baskets, k, m) << endl;

}
```
