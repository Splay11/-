## 解题思路

本题给出的是一棵按完全二叉树层序遍历存储的目录树。

设总节点数为：

$N = 2^n - 1$

数组中每个位置表示一个目录节点：

* $-1$ 表示该节点为空，不参与统计；
* $0$ 表示该节点没有文件大小，但可能有子目录；
* 正整数表示该目录自身文件大小。

对每个非空目录，需要统计它的总大小：

$当前目录总大小 = 当前目录自身大小 + 左子目录总大小 + 右子目录总大小$

由于父节点依赖子节点的统计结果，所以可以使用自底向上的动态规划思想，从数组最后一个节点往前遍历。

对于下标为 $i$ 的节点：

* 左孩子下标为 $2i + 1$
* 右孩子下标为 $2i + 2$

如果当前节点不是 $-1$，就把左右孩子中非空节点的统计值加到当前节点上。

最后输出统计后的层序数组，并删除末尾连续的 $-1$，因为题目要求最后一层末尾的空节点可以省略。

相关算法：完全二叉树数组存储、自底向上动态规划。

## 复杂度分析

设节点总数为 $N = 2^n - 1$。

时间复杂度为 $O(N)$，每个节点只处理一次。

空间复杂度为 $O(N)$，需要存储整棵树的数组。

由于 $n \le 10$，所以 $N$ 最大为 $1023$，复杂度完全满足要求。

## 代码实现

### Python

```python
def calc_sizes(n, arr):
    # 完全二叉树总节点数
    total = (1 << n) - 1

    # 输入可能省略了末尾空节点，补成完整数组
    while len(arr) < total:
        arr.append(-1)

    # 自底向上统计每个目录的总大小
    for i in range(total - 1, -1, -1):
        if arr[i] == -1:
            continue

        left = 2 * i + 1
        right = 2 * i + 2

        # 加上左子目录大小
        if left < total and arr[left] != -1:
            arr[i] += arr[left]

        # 加上右子目录大小
        if right < total and arr[right] != -1:
            arr[i] += arr[right]

    # 删除末尾多余的空节点
    while arr and arr[-1] == -1:
        arr.pop()

    return arr


def main():
    n = int(input().strip())

    # 第二行是层序数组，可能省略末尾空节点
    arr = list(map(int, input().split()))

    ans = calc_sizes(n, arr)

    print(" ".join(map(str, ans)))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    public static List<Integer> calcSizes(int n, List<Integer> arr) {
        // 完全二叉树总节点数
        int total = (1 << n) - 1;

        // 输入可能省略了末尾空节点，补成完整数组
        while (arr.size() < total) {
            arr.add(-1);
        }

        // 自底向上统计每个目录的总大小
        for (int i = total - 1; i >= 0; i--) {
            if (arr.get(i) == -1) {
                continue;
            }

            int left = 2 * i + 1;
            int right = 2 * i + 2;
            int sum = arr.get(i);

            // 加上左子目录大小
            if (left < total && arr.get(left) != -1) {
                sum += arr.get(left);
            }

            // 加上右子目录大小
            if (right < total && arr.get(right) != -1) {
                sum += arr.get(right);
            }

            arr.set(i, sum);
        }

        // 删除末尾多余的空节点
        while (!arr.isEmpty() && arr.get(arr.size() - 1) == -1) {
            arr.remove(arr.size() - 1);
        }

        return arr;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        List<Integer> arr = new ArrayList<>();

        // 读取剩余所有整数
        while (sc.hasNextInt()) {
            arr.add(sc.nextInt());
        }

        List<Integer> ans = calcSizes(n, arr);

        for (int i = 0; i < ans.size(); i++) {
            if (i > 0) {
                System.out.print(" ");
            }
            System.out.print(ans.get(i));
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> calcSizes(int n, vector<int> arr) {
    // 完全二叉树总节点数
    int total = (1 << n) - 1;

    // 输入可能省略了末尾空节点，补成完整数组
    while ((int)arr.size() < total) {
        arr.push_back(-1);
    }

    // 自底向上统计每个目录的总大小
    for (int i = total - 1; i >= 0; i--) {
        if (arr[i] == -1) {
            continue;
        }

        int left = 2 * i + 1;
        int right = 2 * i + 2;

        // 加上左子目录大小
        if (left < total && arr[left] != -1) {
            arr[i] += arr[left];
        }

        // 加上右子目录大小
        if (right < total && arr[right] != -1) {
            arr[i] += arr[right];
        }
    }

    // 删除末尾多余的空节点
    while (!arr.empty() && arr.back() == -1) {
        arr.pop_back();
    }

    return arr;
}

int main() {
    int n;
    cin >> n;

    vector<int> arr;
    int x;

    // 读取剩余所有整数
    while (cin >> x) {
        arr.push_back(x);
    }

    vector<int> ans = calcSizes(n, arr);

    for (int i = 0; i < (int)ans.size(); i++) {
        if (i > 0) {
            cout << " ";
        }
        cout << ans[i];
    }

    return 0;
}
```
