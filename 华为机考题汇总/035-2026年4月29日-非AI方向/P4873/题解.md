# 正确解法

## 解题思路

本题要求在 $n$ 架无人机的坐标中，找到任意两架无人机之间的最小欧式距离。

两点 $(x_1,y_1)$ 和 $(x_2,y_2)$ 之间的欧式距离为：

$distance=\sqrt{(x_1-x_2)^2+(y_1-y_2)^2}$

如果直接枚举所有点对，需要 $O(n^2)$ 的时间复杂度。由于 $n \le 50000$，这种做法会超时。

因此使用最近点对分治算法。

核心做法如下：

1. 先将所有点按照 $x$ 坐标从小到大排序。
2. 将点集分成左右两部分，分别递归求出左右两边内部的最近距离平方。
3. 设当前最小距离平方为 $d$。
4. 跨越左右两边的最近点对，只可能出现在分界线附近，因此只保留满足 $(x-midX)^2<d$ 的点。
5. 将这些候选点按照 $y$ 坐标排序。
6. 对于每个候选点，只继续检查后面 $y$ 坐标差满足条件的点。
7. 全程使用距离平方进行比较，避免浮点误差。
8. 最后对最小距离平方开方，并向下取整输出。

如果 $n=1$，不存在两架无人机之间的距离，直接输出 $-1$。

## 复杂度分析

排序需要 $O(n \log n)$。

分治过程中，每一层会处理当前区间内的点，并对候选点按照 $y$ 坐标排序，因此总时间复杂度为：

$O(n \log^2 n)$

由于 $n \le 50000$，该复杂度可以通过。

需要存储所有点和递归过程中的候选点，因此空间复杂度为：

$O(n)$

### Python

```python
import sys
import math

# 计算两个点之间的距离平方
def dist2(a, b):
    dx = a[0] - b[0]
    dy = a[1] - b[1]
    return dx * dx + dy * dy

# 分治求最近点对的最小距离平方
def closest_pair(points, left, right):
    # 点数较少时，直接暴力计算
    if right - left <= 3:
        ans = 10**30
        for i in range(left, right):
            for j in range(i + 1, right):
                ans = min(ans, dist2(points[i], points[j]))
        return ans

    mid = (left + right) // 2
    mid_x = points[mid][0]

    # 分别递归求左右两部分的最小距离平方
    d = min(closest_pair(points, left, mid), closest_pair(points, mid, right))

    temp = []

    # 只保留分界线附近可能产生更优答案的点
    for i in range(left, right):
        dx = points[i][0] - mid_x
        if dx * dx < d:
            temp.append(points[i])

    # 按 y 坐标排序，方便检查跨区间点对
    temp.sort(key=lambda p: p[1])

    m = len(temp)

    # 检查候选点中的跨区间最近距离
    for i in range(m):
        j = i + 1
        while j < m:
            dy = temp[j][1] - temp[i][1]

            # y 坐标差已经不可能更优，直接停止
            if dy * dy >= d:
                break

            d = min(d, dist2(temp[i], temp[j]))
            j += 1

    return d

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    n = int(data[0])
    points = []

    idx = 1
    for _ in range(n):
        x = int(data[idx])
        y = int(data[idx + 1])
        points.append((x, y))
        idx += 2

    # 只有一架无人机时，不存在两点距离
    if n == 1:
        print(-1)
        return

    # 按 x 坐标排序
    points.sort()

    ans2 = closest_pair(points, 0, n)

    # 低版本 Python 不支持 math.isqrt，使用 int(math.sqrt()) 向下取整
    print(int(math.sqrt(ans2)))

if __name__ == "__main__":
    main()
```

```java
import java.util.*;

public class Main {
    static class Point {
        long x;
        long y;

        Point(long x, long y) {
            this.x = x;
            this.y = y;
        }
    }

    // 计算两个点之间的距离平方
    static long dist2(Point a, Point b) {
        long dx = a.x - b.x;
        long dy = a.y - b.y;
        return dx * dx + dy * dy;
    }

    // 分治求最近点对的最小距离平方
    static long closestPair(Point[] points, int left, int right) {
        // 点数较少时，直接暴力计算
        if (right - left <= 3) {
            long ans = Long.MAX_VALUE / 4;

            for (int i = left; i < right; i++) {
                for (int j = i + 1; j < right; j++) {
                    ans = Math.min(ans, dist2(points[i], points[j]));
                }
            }

            return ans;
        }

        int mid = (left + right) / 2;
        long midX = points[mid].x;

        // 分别递归求左右两部分的最小距离平方
        long d = Math.min(
            closestPair(points, left, mid),
            closestPair(points, mid, right)
        );

        ArrayList<Point> temp = new ArrayList<Point>();

        // 只保留分界线附近可能产生更优答案的点
        for (int i = left; i < right; i++) {
            long dx = points[i].x - midX;
            if (dx * dx < d) {
                temp.add(points[i]);
            }
        }

        // 按 y 坐标排序，方便检查跨区间点对
        Collections.sort(temp, new Comparator<Point>() {
            public int compare(Point a, Point b) {
                return Long.compare(a.y, b.y);
            }
        });

        int m = temp.size();

        // 检查候选点中的跨区间最近距离
        for (int i = 0; i < m; i++) {
            for (int j = i + 1; j < m; j++) {
                long dy = temp.get(j).y - temp.get(i).y;

                // y 坐标差已经不可能更优，直接停止
                if (dy * dy >= d) {
                    break;
                }

                d = Math.min(d, dist2(temp.get(i), temp.get(j)));
            }
        }

        return d;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        Point[] points = new Point[n];

        for (int i = 0; i < n; i++) {
            long x = sc.nextLong();
            long y = sc.nextLong();
            points[i] = new Point(x, y);
        }

        // 只有一架无人机时，不存在两点距离
        if (n == 1) {
            System.out.println(-1);
            return;
        }

        // 按 x 坐标排序
        Arrays.sort(points, new Comparator<Point>() {
            public int compare(Point a, Point b) {
                if (a.x != b.x) {
                    return Long.compare(a.x, b.x);
                }
                return Long.compare(a.y, b.y);
            }
        });

        long ans2 = closestPair(points, 0, n);

        // 题目要求只保留整数部分，直接向下取整
        System.out.println((long)Math.sqrt(ans2));
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Point {
    long long x;
    long long y;
};

// 计算两个点之间的距离平方
long long dist2(const Point& a, const Point& b) {
    long long dx = a.x - b.x;
    long long dy = a.y - b.y;
    return dx * dx + dy * dy;
}

// 分治求最近点对的最小距离平方
long long closestPair(vector<Point>& points, int left, int right) {
    // 点数较少时，直接暴力计算
    if (right - left <= 3) {
        long long ans = LLONG_MAX / 4;

        for (int i = left; i < right; i++) {
            for (int j = i + 1; j < right; j++) {
                ans = min(ans, dist2(points[i], points[j]));
            }
        }

        return ans;
    }

    int mid = (left + right) / 2;
    long long midX = points[mid].x;

    // 分别递归求左右两部分的最小距离平方
    long long d = min(
        closestPair(points, left, mid),
        closestPair(points, mid, right)
    );

    vector<Point> temp;

    // 只保留分界线附近可能产生更优答案的点
    for (int i = left; i < right; i++) {
        long long dx = points[i].x - midX;
        if (dx * dx < d) {
            temp.push_back(points[i]);
        }
    }

    // 按 y 坐标排序，方便检查跨区间点对
    sort(temp.begin(), temp.end(), [](const Point& a, const Point& b) {
        return a.y < b.y;
    });

    int m = temp.size();

    // 检查候选点中的跨区间最近距离
    for (int i = 0; i < m; i++) {
        for (int j = i + 1; j < m; j++) {
            long long dy = temp[j].y - temp[i].y;

            // y 坐标差已经不可能更优，直接停止
            if (dy * dy >= d) {
                break;
            }

            d = min(d, dist2(temp[i], temp[j]));
        }
    }

    return d;
}

int main() {
    int n;
    cin >> n;

    vector<Point> points(n);

    for (int i = 0; i < n; i++) {
        cin >> points[i].x >> points[i].y;
    }

    // 只有一架无人机时，不存在两点距离
    if (n == 1) {
        cout << -1 << endl;
        return 0;
    }

    // 按 x 坐标排序
    sort(points.begin(), points.end(), [](const Point& a, const Point& b) {
        if (a.x != b.x) {
            return a.x < b.x;
        }
        return a.y < b.y;
    });

    long long ans2 = closestPair(points, 0, n);

    // 题目要求只保留整数部分，直接向下取整
    cout << (long long)sqrt((long double)ans2) << endl;

    return 0;
}
```
