## 解题思路

1. 对每个 Batch 和每个 Head 独立处理一个 $S \times S$ 的得分矩阵，输出顺序与输入顺序一致。
2. 对于第 $b$ 个 Batch，设有效长度为 $L_b$。若行索引 $i \ge L_b$，说明该查询位置本身是 Padding，整行输出全 $0$。
3. 否则该行的有效列必须同时满足因果掩码和 Padding 掩码，即 $0 \le j \le i$ 且 $j < L_b$。
4. 对有效位置计算位置惩罚后的值：
   $$
   v_j = Score_{i,j} - (i-j)\times(s+1)
   $$
   其中 $s$ 是 Head 下标，从 $0$ 开始。
5. 对这些有效位置做稳定 Softmax：先减去最大值再求指数，掩码位置直接输出 $0$。这样可以避免指数溢出，也能自然得到题目要求的 Safe Softmax。

## 复杂度分析

- 时间复杂度：$O(BHS^2)$，每个矩阵元素最多参与一次读入和一次处理。
- 空间复杂度：$O(S^2)$，逐个矩阵处理时只需要保存当前矩阵和当前矩阵的输出。

## 代码实现

### Python

```python
import sys
import numpy as np


def process_matrix(matrix, length, head_index, size):
    result = np.zeros((size, size), dtype=float)
    if length <= 0:
        return result

    slope = head_index + 1
    row = np.arange(size).reshape(size, 1)
    col = np.arange(size).reshape(1, size)

    mask = (row < length) & (col < length) & (col <= row)
    adjusted = matrix - (row - col) * slope
    masked = np.where(mask, adjusted, -np.inf)

    valid_rows = np.arange(size) < length
    part = masked[valid_rows]

    max_value = np.max(part, axis=1, keepdims=True)
    exp_values = np.exp(part - max_value)
    exp_values = np.where(mask[valid_rows], exp_values, 0.0)

    result[valid_rows] = exp_values / np.sum(exp_values, axis=1, keepdims=True)
    return result


def solve(batch_count, head_count, size, lengths, matrices):
    answer = []
    index = 0
    for b in range(batch_count):
        for h in range(head_count):
            answer.extend(process_matrix(matrices[index], lengths[b], h, size))
            index += 1
    return answer


def main():
    data = sys.stdin.buffer.read().split()
    idx = 0

    batch_count = int(data[idx])
    head_count = int(data[idx + 1])
    size = int(data[idx + 2])
    idx += 3

    lengths = [int(data[idx + i]) for i in range(batch_count)]
    idx += batch_count

    matrices = []
    for _ in range(batch_count * head_count):
        values = [float(data[idx + i]) for i in range(size * size)]
        idx += size * size
        matrices.append(np.array(values, dtype=float).reshape(size, size))

    answer = solve(batch_count, head_count, size, lengths, matrices)
    print("\n".join(" ".join(f"{x:.4f}" for x in row) for row in answer))


if __name__ == "__main__":
    main()
```

```java
import java.io.BufferedInputStream;
import java.io.IOException;
import java.util.Locale;

public class Main {
    static class FastScanner {
        private final BufferedInputStream in = new BufferedInputStream(System.in);
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0;
        private int len = 0;

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) {
                    return -1;
                }
            }
            return buffer[ptr++];
        }

        int nextInt() throws IOException {
            return (int) nextDouble();
        }

        double nextDouble() throws IOException {
            int c;
            do {
                c = read();
            } while (c <= ' ' && c != -1);
            int sign = 1;
            if (c == '-') {
                sign = -1;
                c = read();
            }
            double value = 0.0;
            while (c > ' ' && c != '.') {
                value = value * 10.0 + (c - '0');
                c = read();
            }
            if (c == '.') {
                double base = 0.1;
                c = read();
                while (c > ' ') {
                    value += (c - '0') * base;
                    base *= 0.1;
                    c = read();
                }
            }
            return value * sign;
        }
    }

    private static double[][] processMatrix(double[][] matrix, int length, int headIndex, int size) {
        int slope = headIndex + 1;
        double[][] result = new double[size][size];
        for (int i = 0; i < size; i++) {
            // 查询位置为 padding，整行保持 0
            if (i >= length) {
                continue;
            }
            int limit = Math.min(i, length - 1);
            double[] values = new double[limit + 1];
            for (int j = 0; j <= limit; j++) {
                // 有效列同时满足 j <= i 和 j < length
                values[j] = matrix[i][j] - (double) (i - j) * slope;
            }
            double maxValue = values[0];
            for (int j = 0; j < values.length; j++) {
                maxValue = Math.max(maxValue, values[j]);
            }
            double[] weights = new double[values.length];
            double total = 0.0;
            for (int j = 0; j < values.length; j++) {
                weights[j] = Math.exp(values[j] - maxValue);
                total += weights[j];
            }
            for (int j = 0; j < weights.length; j++) {
                result[i][j] = weights[j] / total;
            }
        }
        return result;
    }

    private static String format4(double value) {
        long scaled = Math.round(value * 10000.0);
        long integer = scaled / 10000;
        long fraction = scaled % 10000;
        String frac = Long.toString(fraction);
        while (frac.length() < 4) {
            frac = "0" + frac;
        }
        return integer + "." + frac;
    }

    public static void main(String[] args) throws Exception {
        Locale.setDefault(Locale.US);
        FastScanner scanner = new FastScanner();
        int batchCount = scanner.nextInt();
        int headCount = scanner.nextInt();
        int size = scanner.nextInt();
        int[] lengths = new int[batchCount];
        for (int i = 0; i < batchCount; i++) {
            lengths[i] = scanner.nextInt();
        }

        StringBuilder output = new StringBuilder();
        for (int b = 0; b < batchCount; b++) {
            for (int h = 0; h < headCount; h++) {
                double[][] matrix = new double[size][size];
                for (int i = 0; i < size; i++) {
                    for (int j = 0; j < size; j++) {
                        matrix[i][j] = scanner.nextDouble();
                    }
                }
                double[][] answer = processMatrix(matrix, lengths[b], h, size);
                for (int i = 0; i < size; i++) {
                    for (int j = 0; j < size; j++) {
                        if (j > 0) {
                            output.append(' ');
                        }
                        output.append(format4(answer[i][j]));
                    }
                    output.append('\n');
                }
            }
        }
        System.out.print(output.toString());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<vector<double>> processMatrix(const vector<vector<double>>& matrix, int length, int headIndex, int size) {
    int slope = headIndex + 1;
    vector<vector<double>> result(size, vector<double>(size, 0.0));
    for (int i = 0; i < size; ++i) {
        // padding 查询行直接输出全 0
        if (i >= length) {
            continue;
        }
        int limit = min(i, length - 1);
        vector<double> values;
        values.reserve(limit + 1);
        for (int j = 0; j <= limit; ++j) {
            // 有效位置注入 head 相关的位置惩罚
            values.push_back(matrix[i][j] - 1.0 * (i - j) * slope);
        }
        double maxValue = *max_element(values.begin(), values.end());
        double total = 0.0;
        vector<double> weights(values.size());
        for (int j = 0; j < (int)values.size(); ++j) {
            weights[j] = exp(values[j] - maxValue);
            total += weights[j];
        }
        for (int j = 0; j < (int)weights.size(); ++j) {
            result[i][j] = weights[j] / total;
        }
    }
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int batchCount, headCount, size;
    cin >> batchCount >> headCount >> size;
    vector<int> lengths(batchCount);
    for (int i = 0; i < batchCount; ++i) {
        cin >> lengths[i];
    }

    cout << fixed << setprecision(4);
    for (int b = 0; b < batchCount; ++b) {
        for (int h = 0; h < headCount; ++h) {
            vector<vector<double>> matrix(size, vector<double>(size));
            for (int i = 0; i < size; ++i) {
                for (int j = 0; j < size; ++j) {
                    cin >> matrix[i][j];
                }
            }
            auto answer = processMatrix(matrix, lengths[b], h, size);
            for (const auto& row : answer) {
                for (int j = 0; j < size; ++j) {
                    if (j) {
                        cout << ' ';
                    }
                    cout << row[j];
                }
                cout << '\n';
            }
        }
    }
    return 0;
}
```
