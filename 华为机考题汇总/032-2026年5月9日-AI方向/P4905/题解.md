## 解题思路

1. 题目已经给出了逻辑回归的模型、初始化方式、梯度公式和参数更新公式，因此需要按公式模拟批量梯度下降。
2. 每一轮迭代都要先基于当前的 $w_1,w_2,b$ 计算所有样本的预测值 $\hat y$，累加得到 $dw_1,dw_2,db$，再统一除以 $N$。
3. 注意这是 Batch Gradient Descent，同一轮中不能边遍历样本边立即更新参数，否则会变成随机梯度下降，结果会不同。
4. 最后使用迭代 $E$ 次后的参数计算新样本的 $z$ 和 sigmoid 值，并按题目要求保留 $4$ 位小数。
5. 当 $z$ 的绝对值很大时，直接计算 $\exp(-z)$ 可能溢出。可以对 sigmoid 分情况计算：若 $z \ge 0$，使用 $\frac{1}{1+\exp(-z)}$；否则使用 $\frac{\exp(z)}{1+\exp(z)}$。

## 复杂度分析

- 时间复杂度：$O(NE)$，其中 $N$ 为训练样本数，$E$ 为迭代次数。
- 空间复杂度：$O(N)$，用于保存训练样本；除输入外只需要常数个参数和梯度变量。

## 代码实现

### Python

```python
import numpy as np


def sigmoid(z):
    z = np.asarray(z, dtype=np.float64)
    # 分正负计算，避免 z 绝对值很大时 exp 溢出
    return np.where(
        z >= 0.0,
        1.0 / (1.0 + np.exp(-z)),
        np.exp(z) / (1.0 + np.exp(z))
    )


def solve(samples, epochs, alpha, query):
    data = np.array(samples, dtype=np.float64)
    x = data[:, :2]
    y = data[:, 2]
    n = x.shape[0]
    w = np.zeros(2, dtype=np.float64)
    b = 0.0
    for _ in range(epochs):
        # 使用矩阵运算一次性计算全体样本的预测误差
        pred = sigmoid(x @ w + b)
        diff = pred - y
        # 全量梯度下降：先对所有样本求平均梯度，再统一更新
        dw = (x.T @ diff) / n
        db = np.mean(diff)
        w -= alpha * dw
        b -= alpha * db
    query = np.array(query, dtype=np.float64)
    return float(sigmoid(query @ w + b))


def main():
    n, epochs, alpha = input().split()
    n = int(n)
    epochs = int(epochs)
    alpha = float(alpha)
    samples = []
    for _ in range(n):
        x1, x2, y = input().split()
        samples.append((float(x1), float(x2), int(y)))
    query = tuple(map(float, input().split()))
    print(f"{solve(samples, epochs, alpha, query):.4f}")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Locale;
import java.util.Scanner;

public class Main {
    private static double sigmoid(double z) {
        if (z >= 0.0) {
            return 1.0 / (1.0 + Math.exp(-z));
        }
        double ez = Math.exp(z);
        return ez / (1.0 + ez);
    }

    private static double solve(double[] x1, double[] x2, int[] y, int epochs, double alpha, double q1, double q2) {
        int n = x1.length;
        double w1 = 0.0;
        double w2 = 0.0;
        double b = 0.0;
        for (int epoch = 0; epoch < epochs; epoch++) {
            double dw1 = 0.0;
            double dw2 = 0.0;
            double db = 0.0;
            for (int i = 0; i < n; i++) {
                // 使用当前参数计算预测误差，先累加梯度
                double pred = sigmoid(w1 * x1[i] + w2 * x2[i] + b);
                double diff = pred - y[i];
                dw1 += diff * x1[i];
                dw2 += diff * x2[i];
                db += diff;
            }
            // 全量梯度下降：遍历完所有样本后再统一更新
            dw1 /= n;
            dw2 /= n;
            db /= n;
            w1 -= alpha * dw1;
            w2 -= alpha * dw2;
            b -= alpha * db;
        }
        return sigmoid(w1 * q1 + w2 * q2 + b);
    }

    public static void main(String[] args) {
        Locale.setDefault(Locale.US);
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int epochs = sc.nextInt();
        double alpha = sc.nextDouble();
        double[] x1 = new double[n];
        double[] x2 = new double[n];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            x1[i] = sc.nextDouble();
            x2[i] = sc.nextDouble();
            y[i] = sc.nextInt();
        }
        double q1 = sc.nextDouble();
        double q2 = sc.nextDouble();
        System.out.printf("%.4f%n", solve(x1, x2, y, epochs, alpha, q1, q2));
        sc.close();
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Sample {
    double x1;
    double x2;
    int y;
};

double sigmoid(double z) {
    if (z >= 0.0) {
        return 1.0 / (1.0 + exp(-z));
    }
    double ez = exp(z);
    return ez / (1.0 + ez);
}

double solve(const vector<Sample>& samples, int epochs, double alpha, double q1, double q2) {
    int n = (int)samples.size();
    double w1 = 0.0, w2 = 0.0, b = 0.0;
    for (int epoch = 0; epoch < epochs; ++epoch) {
        double dw1 = 0.0, dw2 = 0.0, db = 0.0;
        for (const auto& s : samples) {
            // 使用当前参数计算预测误差，先累加梯度
            double pred = sigmoid(w1 * s.x1 + w2 * s.x2 + b);
            double diff = pred - s.y;
            dw1 += diff * s.x1;
            dw2 += diff * s.x2;
            db += diff;
        }
        // 全量梯度下降：遍历完所有样本后再统一更新
        dw1 /= n;
        dw2 /= n;
        db /= n;
        w1 -= alpha * dw1;
        w2 -= alpha * dw2;
        b -= alpha * db;
    }
    return sigmoid(w1 * q1 + w2 * q2 + b);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, epochs;
    double alpha;
    cin >> n >> epochs >> alpha;
    vector<Sample> samples(n);
    for (int i = 0; i < n; ++i) {
        cin >> samples[i].x1 >> samples[i].x2 >> samples[i].y;
    }
    double q1, q2;
    cin >> q1 >> q2;

    cout << fixed << setprecision(4) << solve(samples, epochs, alpha, q1, q2) << '\n';
    return 0;
}
```
