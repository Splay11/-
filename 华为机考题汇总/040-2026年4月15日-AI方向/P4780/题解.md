## 解题思路

这道题本质上是一个经典的“相邻约束下的最小分配”问题，和常见的“分糖果”问题一致。

题目要求：

* 每个有效任务（优先级大于 $0$）至少分配 $1$ 千个 $token$
* 对于相邻的两个有效任务：

  * 若左边优先级更高，则左边分配的 $token$ 必须更多
  * 若右边优先级更高，则右边分配的 $token$ 必须更多
* 若某个任务优先级小于等于 $0$，则该任务直接放弃，不分配 $token$
* 被放弃的任务会切断相邻关系，两边任务不再跨过它进行比较

### 核心思路

先只考虑优先级大于 $0$ 的任务。对于这些任务，要满足：

* 从左往右看：如果当前任务优先级比左边高，那么当前任务的分配数必须比左边大 $1$
* 从右往左看：如果当前任务优先级比右边高，那么当前任务的分配数也必须比右边大 $1$

因此可以使用“两次遍历”解决：

1. 先把所有有效任务的初始分配设为 $1$
2. 从左到右遍历：

   * 若当前任务和左边任务都有效，且当前优先级更高，则
     $$
     tokens[i] = tokens[i-1] + 1
     $$
3. 从右到左遍历：

   * 若当前任务和右边任务都有效，且当前优先级更高，则
     $$
     tokens[i] = \max(tokens[i], tokens[i+1] + 1)
     $$
4. 最后把所有分配数求和即可

这里取 $\max$ 的原因是：第一次遍历已经保证了“左边约束”，第二次遍历要在不破坏原有约束的前提下，再满足“右边约束”。

## 复杂度分析

设输入数组长度为 $N$。

* 两次线性遍历数组，因此时间复杂度为：
  $$
  O(N)
  $$

* 额外使用一个长度为 $N$ 的分配数组，因此空间复杂度为：
  $$
  O(N)
  $$

## 代码实现

### Python

```python
import sys
from ast import literal_eval

def min_tokens(priority):
    n = len(priority)
    
    # tokens[i] 表示第 i 个任务最终分配的 token 数
    tokens = [0] * n

    # 所有有效任务先至少分配 1
    for i in range(n):
        if priority[i] > 0:
            tokens[i] = 1

    # 从左到右处理：
    # 如果当前任务和左边任务都有效，且当前优先级更高，
    # 那么当前任务的 token 数必须比左边多 1
    for i in range(1, n):
        if priority[i] > 0 and priority[i - 1] > 0 and priority[i] > priority[i - 1]:
            tokens[i] = tokens[i - 1] + 1

    # 从右到左处理：
    # 如果当前任务和右边任务都有效，且当前优先级更高，
    # 那么当前任务的 token 数必须至少比右边多 1
    for i in range(n - 2, -1, -1):
        if priority[i] > 0 and priority[i + 1] > 0 and priority[i] > priority[i + 1]:
            tokens[i] = max(tokens[i], tokens[i + 1] + 1)

    return sum(tokens)

def main():
    s = sys.stdin.read().strip()

    # 优先尝试按列表格式解析，例如 [1,-1,2]
    # 如果不是标准列表格式，例如 1,2,3，则补上中括号再解析
    try:
        arr = literal_eval(s)
        if isinstance(arr, int):
            arr = [arr]
    except:
        arr = literal_eval("[" + s + "]")

    print(min_tokens(arr))

if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    public static int minTokens(int[] priority) {
        int n = priority.length;

        // tokens[i] 表示第 i 个任务最终分配的 token 数
        int[] tokens = new int[n];

        // 所有有效任务先至少分配 1
        for (int i = 0; i < n; i++) {
            if (priority[i] > 0) {
                tokens[i] = 1;
            }
        }

        // 从左到右处理：
        // 如果当前任务和左边任务都有效，且当前优先级更高，
        // 那么当前任务的 token 数必须比左边多 1
        for (int i = 1; i < n; i++) {
            if (priority[i] > 0 && priority[i - 1] > 0 && priority[i] > priority[i - 1]) {
                tokens[i] = tokens[i - 1] + 1;
            }
        }

        // 从右到左处理：
        // 如果当前任务和右边任务都有效，且当前优先级更高，
        // 那么当前任务的 token 数必须至少比右边多 1
        for (int i = n - 2; i >= 0; i--) {
            if (priority[i] > 0 && priority[i + 1] > 0 && priority[i] > priority[i + 1]) {
                tokens[i] = Math.max(tokens[i], tokens[i + 1] + 1);
            }
        }

        // 统计总 token 数
        int sum = 0;
        for (int x : tokens) {
            sum += x;
        }
        return sum;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine().trim();

        // 去掉可能出现的中括号和空格，便于按逗号分割
        s = s.replace("[", "").replace("]", "").replace(" ", "");

        if (s.length() == 0) {
            System.out.println(0);
            return;
        }

        String[] parts = s.split(",");
        int[] priority = new int[parts.length];
        for (int i = 0; i < parts.length; i++) {
            priority[i] = Integer.parseInt(parts[i]);
        }

        System.out.println(minTokens(priority));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <sstream>
#include <algorithm>
using namespace std;

int minTokens(const vector<int>& priority) {
    int n = priority.size();

    // tokens[i] 表示第 i 个任务最终分配的 token 数
    vector<int> tokens(n, 0);

    // 所有有效任务先至少分配 1
    for (int i = 0; i < n; i++) {
        if (priority[i] > 0) {
            tokens[i] = 1;
        }
    }

    // 从左到右处理：
    // 如果当前任务和左边任务都有效，且当前优先级更高，
    // 那么当前任务的 token 数必须比左边多 1
    for (int i = 1; i < n; i++) {
        if (priority[i] > 0 && priority[i - 1] > 0 && priority[i] > priority[i - 1]) {
            tokens[i] = tokens[i - 1] + 1;
        }
    }

    // 从右到左处理：
    // 如果当前任务和右边任务都有效，且当前优先级更高，
    // 那么当前任务的 token 数必须至少比右边多 1
    for (int i = n - 2; i >= 0; i--) {
        if (priority[i] > 0 && priority[i + 1] > 0 && priority[i] > priority[i + 1]) {
            tokens[i] = max(tokens[i], tokens[i + 1] + 1);
        }
    }

    // 统计总 token 数
    int sum = 0;
    for (int x : tokens) {
        sum += x;
    }
    return sum;
}

int main() {
    string s;
    getline(cin, s);

    // 去掉可能出现的中括号和空格，便于按逗号分割
    string t;
    for (char c : s) {
        if (c != '[' && c != ']' && c != ' ') {
            t += c;
        }
    }

    if (t.empty()) {
        cout << 0 << endl;
        return 0;
    }

    vector<int> priority;
    stringstream ss(t);
    string part;

    while (getline(ss, part, ',')) {
        priority.push_back(stoi(part));
    }

    cout << minTokens(priority) << endl;
    return 0;
}
```
