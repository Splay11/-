## 解题思路

本题要求把长度为 $L$ 的序列 $w$ 划成恰好 $k$ 段连续非空子段，使各段元素和的最大值（峰值负载 $peak$）最小。这是经典的 **二分答案 + 贪心验证**（与 LeetCode 410「分割数组的最大值」同型）。

1. **单调性**：若峰值上限 $limit$ 可行，则任意更大的 $limit$ 也可行；不可行则更小的 $limit$ 也不可行。在 $[\max(w),\ \sum w]$ 上二分。
2. **检验 `can(limit)`**：从左到右贪心累加当前段和，超过 $limit$ 就新开一段；统计最少段数 $cnt$。若 $cnt \le k$，说明可在不超过 $limit$ 的前提下拆成至多 $k$ 段；因 $k \le L$ 且每段非空，还可继续切到恰好 $k$ 段而不增大峰值。
3. 二分得到的最小可行 $limit$ 即为最小 $peak$。

## 复杂度分析

设 $S = \sum w_i$。

- 时间复杂度：$O(L \log S)$，每次 `can` 为 $O(L)$。
- 空间复杂度：$O(1)$。

## 代码实现

### Python

```python
def min_peak(w, k):
    """二分最小峰值 limit，每次用贪心检验是否可行。"""

    def can(limit):
        """
        检验：能否把 w 划成不超过 k 段，且每段和 <= limit。
        贪心策略：从左到右累加，一旦超限就另起一段。
        """
        cnt = 1       # 当前已用的段数，至少 1 段
        s = 0         # 当前段正在累加的和
        for x in w:
            # 若把 x 放进当前段会超过 limit，必须先开新段
            if s + x > limit:
                cnt += 1
                s = 0     # 新段从 x 重新开始累加
            s += x
        # 贪心得到的最少段数 <= k，说明 limit 足够大，划分可行
        return cnt <= k

    # 二分答案：lo 一定可行（或恰为答案），hi 初始为不可行的上界
    lo = max(w)       # 下界：单层最大 w_i，任何段至少这么长
    hi = sum(w)       # 上界：全部分给 1 段，峰值 = 总和
    while lo < hi:
        mid = (lo + hi) // 2
        if can(mid):
            hi = mid          # mid 可行，尝试更小的峰值
        else:
            lo = mid + 1      # mid 不可行，峰值必须更大
    return lo


L = int(input())
w = list(map(int, input().split()))
k = int(input())
print(min_peak(w, k))
```

### Java

```java
import java.util.Scanner;

public class Main {
    /**
     * 贪心检验：在峰值上限 limit 下，最少需要几段？
     * 若最少段数 <= k，则 limit 可行。
     */
    static boolean canSplit(int[] w, int k, int limit) {
        int cnt = 1;    // 段数，从 1 开始
        long s = 0;     // 当前段累加和
        for (int x : w) {
            if (s + x > limit) {
                // 放不进当前段，必须新开一段
                cnt++;
                s = 0;
            }
            s += x;
        }
        return cnt <= k;
    }

    static int minPeak(int[] w, int k) {
        int lo = 0, hi = 0;
        for (int x : w) {
            lo = Math.max(lo, x);  // 下界：单元素最大值
            hi += x;               // 上界：全部一段的总和
        }
        // 标准二分：找最小的可行 limit
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (canSplit(w, k, mid)) {
                hi = mid;
            } else {
                lo = mid + 1;
            }
        }
        return lo;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int L = sc.nextInt();
        int[] w = new int[L];
        for (int i = 0; i < L; i++) {
            w[i] = sc.nextInt();
        }
        int k = sc.nextInt();
        System.out.println(minPeak(w, k));
        sc.close();
    }
}
```

### C++

```cpp
#include <algorithm>
#include <iostream>
#include <numeric>
#include <vector>
using namespace std;

int minPeak(const vector<int>& w, int k) {
    // can(limit)：峰值上限 limit 是否可行
    auto can = [&](long long limit) {
        int cnt = 1;
        long long s = 0;
        for (int x : w) {
            if (s + x > limit) {
                cnt++;   // 当前段放不下 x，新开一段
                s = 0;
            }
            s += x;
        }
        return cnt <= k;  // 段数不超过 k 则可行
    };

    // 二分范围：[max(w), sum(w)]
    long long lo = *max_element(w.begin(), w.end());
    long long hi = accumulate(w.begin(), w.end(), 0LL);
    while (lo < hi) {
        long long mid = (lo + hi) / 2;
        if (can(mid)) {
            hi = mid;       // 可行，往左缩
        } else {
            lo = mid + 1;   // 不可行，往右扩
        }
    }
    return (int)lo;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int L;
    cin >> L;
    vector<int> w(L);
    for (int i = 0; i < L; i++) {
        cin >> w[i];
    }
    int k;
    cin >> k;
    cout << minPeak(w, k) << '\n';
    return 0;
}
```
