## 解题思路

将所有频段名称转换为小写，统一处理题目要求的不区分大小写。

对于任意两个载波，如果满足下面任意条件，则它们不能同时选择：

* 两个载波使用相同频段。
* 两个载波使用的频段存在直接冲突关系。

冲突关系不具有传递性，因此只记录输入中明确给出的冲突频段对。

由于载波总数满足 $N\le 25$，最多选择数量满足 $K<8$，可以使用回溯枚举所有长度不超过 $K$ 的载波组合。

具体实现方法如下：

1. 为每个载波建立一个冲突位掩码，记录与其不能同时选择的载波。
2. 回溯时按照编号递增的顺序选择载波，保证组合不会重复，并且所选编号天然有序。
3. 使用一个整数位掩码记录当前已经选择的载波，可以在 $O(1)$ 时间内判断新载波是否与当前组合冲突。
4. 每到达一个合法组合，就计算总带宽：

   * 总带宽更大时更新答案。
   * 总带宽相同时，保留编号序列字典序更小的组合。

## 复杂度分析

设

$S=\sum_{i=0}^{K}\binom{N}{i}$

为最多需要枚举的组合数量。

建立冲突位掩码需要 $O(N^2+C)$ 时间。

回溯过程中，每个组合最多处理 $K$ 个元素，因此时间复杂度为：

$O(N^2+C+KS)$

由于 $N\le25$ 且 $K<8$，该复杂度可以通过。

冲突位掩码、递归路径和答案所需的空间复杂度为：

$O(N+C+K)$

### Python

```python
def select_carriers(n, k, bandwidth, frequencies, conflict_pairs):
    # 频段名称统一转换为小写
    frequencies = [name.lower() for name in frequencies]

    # 保存直接冲突的频段对
    conflicts = set()
    for a, b in conflict_pairs:
        a = a.lower()
        b = b.lower()
        conflicts.add(tuple(sorted((a, b))))

    # ban[i] 表示所有与载波 i 冲突的载波
    ban = [0] * n

    for i in range(n):
        for j in range(i + 1, n):
            pair = tuple(sorted((frequencies[i], frequencies[j])))

            # 同频段或存在直接冲突时，两个载波不能同时选择
            if frequencies[i] == frequencies[j] or pair in conflicts:
                ban[i] |= 1 << j
                ban[j] |= 1 << i

    best_sum = -1
    best_ids = []
    chosen = []

    def dfs(start, total, selected_mask):
        nonlocal best_sum, best_ids

        # 更新最大带宽，并处理字典序最小的要求
        if total > best_sum or (
            total == best_sum and chosen < best_ids
        ):
            best_sum = total
            best_ids = chosen.copy()

        # 已经选择了 k 个载波
        if len(chosen) == k:
            return

        for i in range(start, n):
            # 当前载波与已选载波存在冲突
            if ban[i] & selected_mask:
                continue

            chosen.append(i)
            dfs(i + 1, total + bandwidth[i],
                selected_mask | (1 << i))
            chosen.pop()

    dfs(0, 0, 0)
    return best_sum, best_ids


def main():
    n, k = map(int, input().split())
    bandwidth = list(map(int, input().split()))
    frequencies = input().split()

    c = int(input())
    conflict_pairs = []

    for _ in range(c):
        a, b = input().split()
        conflict_pairs.append((a, b))

    best_sum, best_ids = select_carriers(
        n, k, bandwidth, frequencies, conflict_pairs
    )

    print(best_sum)
    print(*best_ids)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static int n;
    static int k;
    static int[] bandwidth;
    static int[] ban;

    static int bestSum;
    static ArrayList<Integer> chosen;
    static ArrayList<Integer> bestIds;

    // 将频段对转换为统一形式
    static String getKey(String a, String b) {
        if (a.compareTo(b) > 0) {
            String temp = a;
            a = b;
            b = temp;
        }
        return a + "#" + b;
    }

    // 判断当前组合的字典序是否更小
    static boolean lexicographicallySmaller(
            ArrayList<Integer> a,
            ArrayList<Integer> b) {

        int length = Math.min(a.size(), b.size());

        for (int i = 0; i < length; i++) {
            if (!a.get(i).equals(b.get(i))) {
                return a.get(i) < b.get(i);
            }
        }

        return a.size() < b.size();
    }

    // 回溯枚举所有合法组合
    static void dfs(int start, int total, int selectedMask) {
        // 更新最大带宽，并处理字典序最小的要求
        if (total > bestSum
                || (total == bestSum
                && lexicographicallySmaller(chosen, bestIds))) {
            bestSum = total;
            bestIds = new ArrayList<>(chosen);
        }

        // 已经选择了 k 个载波
        if (chosen.size() == k) {
            return;
        }

        for (int i = start; i < n; i++) {
            // 当前载波与已选载波存在冲突
            if ((ban[i] & selectedMask) != 0) {
                continue;
            }

            chosen.add(i);
            dfs(i + 1, total + bandwidth[i],
                    selectedMask | (1 << i));
            chosen.remove(chosen.size() - 1);
        }
    }

    static int[] selectCarriers(
            int carrierCount,
            int maxCount,
            int[] carrierBandwidth,
            String[] frequencies,
            String[][] conflictPairs) {

        n = carrierCount;
        k = maxCount;
        bandwidth = carrierBandwidth;
        ban = new int[n];

        // 频段名称统一转换为小写
        for (int i = 0; i < n; i++) {
            frequencies[i] =
                    frequencies[i].toLowerCase(Locale.ROOT);
        }

        // 保存直接冲突的频段对
        HashSet<String> conflicts = new HashSet<>();

        for (String[] pair : conflictPairs) {
            String a = pair[0].toLowerCase(Locale.ROOT);
            String b = pair[1].toLowerCase(Locale.ROOT);
            conflicts.add(getKey(a, b));
        }

        // 建立每个载波的冲突位掩码
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                boolean sameFrequency =
                        frequencies[i].equals(frequencies[j]);

                boolean hasConflict = conflicts.contains(
                        getKey(frequencies[i], frequencies[j])
                );

                if (sameFrequency || hasConflict) {
                    ban[i] |= 1 << j;
                    ban[j] |= 1 << i;
                }
            }
        }

        bestSum = -1;
        chosen = new ArrayList<>();
        bestIds = new ArrayList<>();

        dfs(0, 0, 0);

        int[] result = new int[bestIds.size() + 1];
        result[0] = bestSum;

        for (int i = 0; i < bestIds.size(); i++) {
            result[i + 1] = bestIds.get(i);
        }

        return result;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] bandwidth = new int[n];
        for (int i = 0; i < n; i++) {
            bandwidth[i] = scanner.nextInt();
        }

        String[] frequencies = new String[n];
        for (int i = 0; i < n; i++) {
            frequencies[i] = scanner.next();
        }

        int c = scanner.nextInt();
        String[][] conflictPairs = new String[c][2];

        for (int i = 0; i < c; i++) {
            conflictPairs[i][0] = scanner.next();
            conflictPairs[i][1] = scanner.next();
        }

        int[] result = selectCarriers(
                n, k, bandwidth, frequencies, conflictPairs
        );

        System.out.println(result[0]);

        for (int i = 1; i < result.length; i++) {
            if (i > 1) {
                System.out.print(" ");
            }
            System.out.print(result[i]);
        }
        System.out.println();

        scanner.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <unordered_set>
#include <algorithm>
#include <cctype>
using namespace std;

int n, k;
vector<int> bandwidth;
vector<int> banMask;

int bestSum;
vector<int> chosen;
vector<int> bestIds;

// 将字符串转换为小写
string toLowerCase(string text) {
    for (char &ch : text) {
        ch = static_cast<char>(
            tolower(static_cast<unsigned char>(ch))
        );
    }
    return text;
}

// 将频段对转换为统一形式
string getKey(string a, string b) {
    if (a > b) {
        swap(a, b);
    }
    return a + "#" + b;
}

// 回溯枚举所有合法组合
void dfs(int start, int total, int selectedMask) {
    // 更新最大带宽，并处理字典序最小的要求
    if (total > bestSum ||
        (total == bestSum && chosen < bestIds)) {
        bestSum = total;
        bestIds = chosen;
    }

    // 已经选择了 k 个载波
    if (static_cast<int>(chosen.size()) == k) {
        return;
    }

    for (int i = start; i < n; i++) {
        // 当前载波与已选载波存在冲突
        if (banMask[i] & selectedMask) {
            continue;
        }

        chosen.push_back(i);
        dfs(i + 1, total + bandwidth[i],
            selectedMask | (1 << i));
        chosen.pop_back();
    }
}

pair<int, vector<int>> selectCarriers(
    int carrierCount,
    int maxCount,
    const vector<int> &carrierBandwidth,
    vector<string> frequencies,
    const vector<pair<string, string>> &conflictPairs) {

    n = carrierCount;
    k = maxCount;
    bandwidth = carrierBandwidth;
    banMask.assign(n, 0);

    // 频段名称统一转换为小写
    for (string &name : frequencies) {
        name = toLowerCase(name);
    }

    // 保存直接冲突的频段对
    unordered_set<string> conflicts;

    for (auto pair : conflictPairs) {
        string a = toLowerCase(pair.first);
        string b = toLowerCase(pair.second);
        conflicts.insert(getKey(a, b));
    }

    // 建立每个载波的冲突位掩码
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            bool sameFrequency =
                frequencies[i] == frequencies[j];

            bool hasConflict = conflicts.count(
                getKey(frequencies[i], frequencies[j])
            );

            if (sameFrequency || hasConflict) {
                banMask[i] |= 1 << j;
                banMask[j] |= 1 << i;
            }
        }
    }

    bestSum = -1;
    chosen.clear();
    bestIds.clear();

    dfs(0, 0, 0);

    return {bestSum, bestIds};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;

    vector<int> bandwidth(n);
    for (int i = 0; i < n; i++) {
        cin >> bandwidth[i];
    }

    vector<string> frequencies(n);
    for (int i = 0; i < n; i++) {
        cin >> frequencies[i];
    }

    int c;
    cin >> c;

    vector<pair<string, string>> conflictPairs(c);
    for (int i = 0; i < c; i++) {
        cin >> conflictPairs[i].first
            >> conflictPairs[i].second;
    }

    pair<int, vector<int>> answer = selectCarriers(
        n, k, bandwidth, frequencies, conflictPairs
    );

    cout << answer.first << '\n';

    for (int i = 0; i < static_cast<int>(answer.second.size()); i++) {
        if (i > 0) {
            cout << ' ';
        }
        cout << answer.second[i];
    }
    cout << '\n';

    return 0;
}
```
