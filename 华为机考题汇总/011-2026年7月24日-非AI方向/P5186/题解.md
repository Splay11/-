## 解题思路

使用排序、前缀和与二分答案。

先将所有河堤高度从小到大排序，并计算前缀和。

假设要让所有河堤的最低高度达到 $H$：

* 通过二分查找找到第一个高度不小于 $H$ 的位置 $k$。
* 只有前 $k$ 段河堤需要加固。
* 所需土壤总量为：

$$
S=H\times k-prefix[k]
$$

设每车最大运量为 $C$，则运输次数为：

$$
\left\lceil\frac{S}{C}\right\rceil
=\frac{S+C-1}{C}
$$

总成本为：

$$
S\times UnitCost+
\left\lceil\frac{S}{C}\right\rceil
\times TransportationCost
$$

如果总成本不超过预算，说明高度 $H$ 可行；否则不可行。

可行性具有单调性：若高度 $H$ 可行，则所有小于 $H$ 的高度也一定可行。因此可以二分查找最大的可行高度。

二分下界为原始最低高度，上界可取：

$$
minHeight+\frac{MaxCosts}{UnitCost}
$$

因为即使不考虑运输费用，预算也不可能购买更多土壤。

## 复杂度分析

设河堤数量为 $n$，二分高度范围为 $R$。

排序和构造前缀和的时间复杂度为：

$$
O(n\log n)
$$

每次判断通过二分查找确定 $k$，时间复杂度为：

$$
O(\log n)
$$

总时间复杂度为：

$$
O(n\log n+\log R\log n)
$$

前缀和数组需要额外空间，空间复杂度为：

$$
O(n)
$$

## 代码实现

### Python

```python
import sys
from bisect import bisect_left


def max_min_height(budget, capacity, transport_cost, unit_cost, heights):
    # 排序后使用前缀和快速计算所需土壤量
    heights.sort()
    n = len(heights)

    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + heights[i]

    def can_reach(target):
        # 找到所有高度小于目标高度的河堤
        k = bisect_left(heights, target)

        # 将前 k 段河堤提升到 target 所需的土壤量
        soil = target * k - prefix[k]

        # 运输次数向上取整
        shipments = (soil + capacity - 1) // capacity
        total_cost = soil * unit_cost + shipments * transport_cost

        return total_cost <= budget

    left = heights[0]
    right = heights[0] + budget // unit_cost
    answer = left

    # 二分查找最大的可行最低高度
    while left <= right:
        mid = (left + right) // 2

        if can_reach(mid):
            answer = mid
            left = mid + 1
        else:
            right = mid - 1

    return answer


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))

    budget = data[0]
    capacity = data[1]
    transport_cost = data[2]
    unit_cost = data[3]
    n = data[4]
    heights = data[5:5 + n]

    print(max_min_height(
        budget,
        capacity,
        transport_cost,
        unit_cost,
        heights
    ))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.Arrays;

public class Main {

    // 找到第一个大于等于 target 的位置
    static int lowerBound(long[] heights, long target) {
        int left = 0;
        int right = heights.length;

        while (left < right) {
            int mid = (left + right) / 2;

            if (heights[mid] < target) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }

        return left;
    }

    // 判断预算内能否让最低高度达到 target
    static boolean canReach(long target, long budget, long capacity,
                            long transportCost, long unitCost,
                            long[] heights, long[] prefix) {
        int k = lowerBound(heights, target);

        // 将前 k 段河堤提升到 target 所需的土壤量
        long soil = target * k - prefix[k];

        // 运输次数向上取整
        long shipments = (soil + capacity - 1) / capacity;
        long totalCost = soil * unitCost + shipments * transportCost;

        return totalCost <= budget;
    }

    static long maxMinHeight(long budget, long capacity,
                             long transportCost, long unitCost,
                             long[] heights) {
        Arrays.sort(heights);

        int n = heights.length;
        long[] prefix = new long[n + 1];

        // 构造前缀和
        for (int i = 0; i < n; i++) {
            prefix[i + 1] = prefix[i] + heights[i];
        }

        long left = heights[0];
        long right = heights[0] + budget / unitCost;
        long answer = left;

        // 二分查找最大的可行最低高度
        while (left <= right) {
            long mid = left + (right - left) / 2;

            if (canReach(mid, budget, capacity, transportCost,
                         unitCost, heights, prefix)) {
                answer = mid;
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }

        return answer;
    }

    public static void main(String[] args) throws Exception {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in))
        );

        input.nextToken();
        long budget = (long) input.nval;

        input.nextToken();
        long capacity = (long) input.nval;

        input.nextToken();
        long transportCost = (long) input.nval;

        input.nextToken();
        long unitCost = (long) input.nval;

        input.nextToken();
        int n = (int) input.nval;

        long[] heights = new long[n];

        for (int i = 0; i < n; i++) {
            input.nextToken();
            heights[i] = (long) input.nval;
        }

        System.out.println(maxMinHeight(
                budget,
                capacity,
                transportCost,
                unitCost,
                heights
        ));
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;


// 判断预算内能否让最低高度达到 target
bool canReach(long long target,
              long long budget,
              long long capacity,
              long long transportCost,
              long long unitCost,
              const vector<long long>& heights,
              const vector<long long>& prefix) {
    // 找到第一个大于等于 target 的位置
    int k = lower_bound(heights.begin(), heights.end(), target)
            - heights.begin();

    // 将前 k 段河堤提升到 target 所需的土壤量
    long long soil = target * k - prefix[k];

    // 运输次数向上取整
    long long shipments = (soil + capacity - 1) / capacity;
    long long totalCost = soil * unitCost
                        + shipments * transportCost;

    return totalCost <= budget;
}


long long maxMinHeight(long long budget,
                       long long capacity,
                       long long transportCost,
                       long long unitCost,
                       vector<long long>& heights) {
    sort(heights.begin(), heights.end());

    int n = heights.size();
    vector<long long> prefix(n + 1, 0);

    // 构造前缀和
    for (int i = 0; i < n; i++) {
        prefix[i + 1] = prefix[i] + heights[i];
    }

    long long left = heights[0];
    long long right = heights[0] + budget / unitCost;
    long long answer = left;

    // 二分查找最大的可行最低高度
    while (left <= right) {
        long long mid = left + (right - left) / 2;

        if (canReach(mid, budget, capacity, transportCost,
                     unitCost, heights, prefix)) {
            answer = mid;
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

    return answer;
}
```
