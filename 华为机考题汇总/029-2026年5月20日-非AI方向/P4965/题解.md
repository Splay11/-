## 解题思路

使用二分答案。

如果同时启用 $k$ 台服务器，那么第 $i$ 个任务需要的时间为：

$ceil(tasks[i] / k)$

也就是：

$(tasks[i] + k - 1) // k$

因为每个小时所有服务器只能处理同一个任务，所以所有任务总耗时为：

$sum(ceil(tasks[i] / k))$

随着 $k$ 增大，总耗时不会增加，具有单调性，因此可以二分最小的 $k$。

特殊情况：每个任务至少需要 $1$ 小时，如果 $h < n$，即使服务器无限多也无法完成，返回 $-1$。

## 复杂度分析

设最大任务量为 $M$。

时间复杂度为 $O(n \log M)$。

空间复杂度为 $O(1)$。

## 代码实现

### Python

```python
def min_servers(tasks, h):
    n = len(tasks)

    # 每个任务至少需要 1 小时，若 h 小于任务数，则无法完成
    if h < n:
        return -1

    left = 1
    right = max(tasks)
    ans = right

    # 二分最少需要的服务器数量
    while left <= right:
        mid = (left + right) // 2

        total = 0
        # 计算使用 mid 台服务器时的总耗时
        for x in tasks:
            total += (x + mid - 1) // mid
            # 超过 h 后提前结束，避免多余计算
            if total > h:
                break

        if total <= h:
            ans = mid
            right = mid - 1
        else:
            left = mid + 1

    return ans


if __name__ == "__main__":
    n = int(input())
    tasks = list(map(int, input().split()))
    h = int(input())

    print(min_servers(tasks, h))
```

### Java

```java
import java.util.Scanner;

public class Main {
    public static int minServers(int[] tasks, int h) {
        int n = tasks.length;

        // 每个任务至少需要 1 小时，若 h 小于任务数，则无法完成
        if (h < n) {
            return -1;
        }

        int maxTask = 0;
        for (int x : tasks) {
            if (x > maxTask) {
                maxTask = x;
            }
        }

        int left = 1;
        int right = maxTask;
        int ans = right;

        // 二分最少需要的服务器数量
        while (left <= right) {
            int mid = (left + right) / 2;

            long total = 0;
            // 计算使用 mid 台服务器时的总耗时
            for (int x : tasks) {
                total += (x + mid - 1) / mid;
                // 超过 h 后提前结束，避免多余计算
                if (total > h) {
                    break;
                }
            }

            if (total <= h) {
                ans = mid;
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] tasks = new int[n];

        for (int i = 0; i < n; i++) {
            tasks[i] = sc.nextInt();
        }

        int h = sc.nextInt();

        System.out.println(minServers(tasks, h));
    }
}
```

```cpp
int minServers(vector<int>& tasks, int h) {
    int n = tasks.size();

    // 每个任务至少需要 1 小时，若 h 小于任务数，则无法完成
    if (h < n) {
        return -1;
    }

    int left = 1;
    int right = *max_element(tasks.begin(), tasks.end());
    int ans = right;

    // 二分最少需要的服务器数量
    while (left <= right) {
        int mid = (left + right) / 2;

        long long total = 0;
        // 计算使用 mid 台服务器时的总耗时
        for (int x : tasks) {
            total += (x + mid - 1) / mid;
            // 超过 h 后提前结束，避免多余计算
            if (total > h) {
                break;
            }
        }

        if (total <= h) {
            ans = mid;
            right = mid - 1;
        } else {
            left = mid + 1;
        }
    }

    return ans;
}

int main() {
    int n;
    cin >> n;

    vector<int> tasks(n);
    for (int i = 0; i < n; i++) {
        cin >> tasks[i];
    }

    int h;
```
