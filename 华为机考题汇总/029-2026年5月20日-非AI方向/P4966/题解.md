## 解题思路

本题需要根据二叉树的前序遍历和中序遍历还原原树，然后按规则简化树，最后输出后序遍历。

核心算法包括：二叉树重建、深度优先搜索、路径和剪枝、链式节点合并。

具体做法：

先用前序遍历和中序遍历重建二叉树。前序遍历的第$1$个值一定是当前子树根节点，在中序遍历中找到根节点位置后，可以确定左子树和右子树范围。

然后进行一次递归处理：

先计算从根到当前节点的路径和，记为当前节点的完整大小。如果完整大小小于等于$0$，则删除当前节点整棵子树，返回空。

如果当前节点不被删除，则递归处理左右子树。

处理完剪枝后，再执行合并规则：如果当前节点只有$1$个子节点，则将当前节点和该子节点不断向下合并，直到遇到空节点或有$2$个子节点的节点。合并后的节点值为路径上所有被合并节点值之和，位置保持为最上方节点的位置。

最后对简化后的二叉树进行后序遍历。如果树为空，输出$null$。

## 复杂度分析

设节点数为$n$。

重建二叉树时，每个节点只处理一次，时间复杂度为$O(n)$。

剪枝和合并过程中，每个节点最多被访问常数次，时间复杂度为$O(n)$。

存储二叉树、哈希表和递归栈，空间复杂度为$O(n)$。

## 代码实现

### Python

```python
import sys

sys.setrecursionlimit(1000000)

class Node:
    def __init__(self, val):
        self.val = val
        self.left = None
        self.right = None


def build(pre, pl, pr, ino, il, ir, pos):
    if pl > pr:
        return None

    root_val = pre[pl]
    root = Node(root_val)

    k = pos[root_val]
    left_size = k - il

    root.left = build(pre, pl + 1, pl + left_size, ino, il, k - 1, pos)
    root.right = build(pre, pl + left_size + 1, pr, ino, k + 1, ir, pos)

    return root


def simplify(root, path_sum):
    if root is None:
        return None

    # 规则一：完整镜像大小 <= 0，整棵子树剪掉
    cur_sum = path_sum + root.val
    if cur_sum <= 0:
        return None

    # 先对子树做剪枝和合并
    root.left = simplify(root.left, cur_sum)
    root.right = simplify(root.right, cur_sum)

    # 规则二：如果当前节点只有一个子节点，则向下合并
    while (root.left is None) ^ (root.right is None):
        child = root.left if root.left is not None else root.right

        root.val += child.val
        root.left = child.left
        root.right = child.right

    return root


def postorder(root, ans):
    if root is None:
        return

    postorder(root.left, ans)
    postorder(root.right, ans)
    ans.append(str(root.val))


def main():
    lines = sys.stdin.read().strip().splitlines()
    if not lines:
        return

    pre = list(map(int, lines[0].split()))
    ino = list(map(int, lines[1].split()))

    n = len(pre)
    pos = {v: i for i, v in enumerate(ino)}

    root = build(pre, 0, n - 1, ino, 0, n - 1, pos)
    root = simplify(root, 0)

    ans = []
    postorder(root, ans)

    print(" ".join(ans) if ans else "null")


if __name__ == "__main__":
    main()

```

### Java

```java
import java.util.*;

public class Main {
    static class Node {
        int val;
        Node left;
        Node right;

        Node(int val) {
            this.val = val;
        }
    }

    static int[] pre;
    static int[] in;
    static Map<Integer, Integer> pos = new HashMap<>();

    static Node build(int pl, int pr, int il, int ir) {
        // 根据前序和中序递归建树
        if (pl > pr) {
            return null;
        }

        int rootVal = pre[pl];
        Node root = new Node(rootVal);

        int k = pos.get(rootVal);
        int leftSize = k - il;

        root.left = build(pl + 1, pl + leftSize, il, k - 1);
        root.right = build(pl + leftSize + 1, pr, k + 1, ir);

        return root;
    }

    static Node simplify(Node root, int pathSum) {
        if (root == null) {
            return null;
        }

        // 先执行规则一：完整大小小于等于 0，整棵子树删除
        int curSum = pathSum + root.val;
        if (curSum <= 0) {
            return null;
        }

        // 递归处理左右子树
        root.left = simplify(root.left, curSum);
        root.right = simplify(root.right, curSum);

        // 再执行规则二：只有一个子节点时向下合并
        while ((root.left == null) ^ (root.right == null)) {
            Node child = root.left != null ? root.left : root.right;

            // 合并当前节点和唯一子节点
            root.val += child.val;
            root.left = child.left;
            root.right = child.right;
        }

        return root;
    }

    static void postorder(Node root, List<String> ans) {
        // 后序遍历输出结果
        if (root == null) {
            return;
        }

        postorder(root.left, ans);
        postorder(root.right, ans);
        ans.add(String.valueOf(root.val));
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        if (!sc.hasNextLine()) {
            return;
        }

        String line1 = sc.nextLine().trim();
        String line2 = sc.nextLine().trim();

        String[] a = line1.split("\\s+");
        String[] b = line2.split("\\s+");

        int n = a.length;
        pre = new int[n];
        in = new int[n];

        for (int i = 0; i < n; i++) {
            pre[i] = Integer.parseInt(a[i]);
            in[i] = Integer.parseInt(b[i]);
            pos.put(in[i], i);
        }

        Node root = build(0, n - 1, 0, n - 1);
        root = simplify(root, 0);

        List<String> ans = new ArrayList<>();
        postorder(root, ans);

        if (ans.isEmpty()) {
            System.out.println("null");
        } else {
            System.out.println(String.join(" ", ans));
        }
    }
}
```

```cpp
vector<int> pre, ino;

    root->left = build(pl + 1, pl + leftSize, il, k - 1);
    root->right = build(pl + leftSize + 1, pr, k + 1, ir);

}

Node* simplify(Node* root, int pathSum) {
    if (root == nullptr) {
        return nullptr;
    }

    // 先执行规则一：完整大小小于等于 0，整棵子树删除
    int curSum = pathSum + root->val;
    if (curSum <= 0) {

    // 再执行规则二：只有一个子节点时向下合并

        // 合并当前节点和唯一子节点
        root->val += child->val;
        root->left = child->left;
        root->right = child->right;
    }

    return root;
}

void postorder(Node* root, vector<int>& ans) {
    // 后序遍历输出结果
    if (root == nullptr) {
        return;
    }

    postorder(root->left, ans);
    postorder(root->right, ans);
    ans.push_back(root->val);
}

int main() {
    string line1, line2;

    if (!getline(cin, line1)) {
        return 0;
    }
    getline(cin, line2);

    stringstream ss1(line1), ss2(line2);
    int x;

    while (ss1 >> x) {
        pre.push_back(x);
    }
    while (ss2 >> x) {
        ino.push_back(x);
    }

    int n = pre.size();

    for (int i = 0; i < n; i++) {
        pos[ino[i]] = i;
    }

    Node* root = build(0, n - 1, 0, n - 1);
    root = simplify(root, 0);

    vector<int> ans;
    postorder(root, ans);

    if (ans.empty()) {
        cout << "null" << endl;
    } else {
        for (int i = 0; i < (int)ans.size(); i++) {
            if (i > 0) {
                cout << " ";
            }
            cout << ans[i];
        }
        cout << endl;
    }
```
