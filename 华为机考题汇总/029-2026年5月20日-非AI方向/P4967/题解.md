## 解题思路

由于每个技能最多只有 $1$ 个直接前置技能，且依赖关系无环，所以所有技能会形成一片森林。

我们加入一个虚拟根节点 $0$，把所有没有前置技能的技能都连到 $0$ 上，然后使用树形背包。

定义 $dp[u][t]$ 表示：

在技能 $u$ 的子树中，花费 $t$ 小时，并且已经学习技能 $u$ 时，能获得的最大战斗力。

转移方法：

先选择技能 $u$ 本身，然后依次合并它的子技能。

对于每个子技能 $v$：

* 可以不学 $v$；
* 如果学 $v$，就必须满足 $v$ 子树内部的前置关系。

最终答案为虚拟根节点 $0$ 在 $0$ 到 $T$ 时间内的最大值。

## 复杂度分析

设技能数量为 $N$，总时间为 $T$。

时间复杂度为 $O(N \times T^2)$。

空间复杂度为 $O(N \times T)$。

在 $N \le 100$，$T \le 1000$ 的范围内可以通过。

## 代码实现

### Python

```python
import sys

NEG = -10 ** 9

def solve(n, total_time, cost, power, parent):
    children = [[] for _ in range(n + 1)]

    # 根据前置技能建树，虚拟根节点为 0
    for i in range(1, n + 1):
        children[parent[i]].append(i)

    dp = [[NEG] * (total_time + 1) for _ in range(n + 1)]

    def dfs(u):
        # 先学习当前技能
        if cost[u] <= total_time:
            dp[u][cost[u]] = power[u]

        # 合并所有子技能
        for v in children[u]:
            dfs(v)
            new_dp = dp[u][:]

            for i in range(total_time + 1):
                if dp[u][i] == NEG:
                    continue
                for j in range(total_time - i + 1):
                    if dp[v][j] == NEG:
                        continue
                    new_dp[i + j] = max(new_dp[i + j], dp[u][i] + dp[v][j])

            dp[u] = new_dp

    dfs(0)

    # 虚拟根节点下的最大战斗力
    return max(dp[0])


def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    idx = 0
    n = int(data[idx])
    idx += 1
    total_time = int(data[idx])
    idx += 1

    cost = [0] * (n + 1)
    power = [0] * (n + 1)
    parent = [0] * (n + 1)

    for _ in range(n):
        skill_id = int(data[idx])
        idx += 1
        cost[skill_id] = int(data[idx])
        idx += 1
        power[skill_id] = int(data[idx])
        idx += 1
        k = int(data[idx])
        idx += 1

        if k == 1:
            parent[skill_id] = int(data[idx])
            idx += 1

    print(solve(n, total_time, cost, power, parent))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static final int NEG = -1000000000;

    static int totalTime;
    static int[] cost;
    static int[] power;
    static ArrayList<Integer>[] children;
    static int[][] dp;

    static void dfs(int u) {
        // 先学习当前技能
        if (cost[u] <= totalTime) {
            dp[u][cost[u]] = power[u];
        }

        // 合并所有子技能
        for (int v : children[u]) {
            dfs(v);

            int[] newDp = dp[u].clone();

            for (int i = 0; i <= totalTime; i++) {
                if (dp[u][i] == NEG) {
                    continue;
                }

                for (int j = 0; i + j <= totalTime; j++) {
                    if (dp[v][j] == NEG) {
                        continue;
                    }

                    newDp[i + j] = Math.max(newDp[i + j], dp[u][i] + dp[v][j]);
                }
            }

            dp[u] = newDp;
        }
    }

    static int solve(int n, int t, int[] inputCost, int[] inputPower, int[] parent) {
        totalTime = t;
        cost = inputCost;
        power = inputPower;

        children = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            children[i] = new ArrayList<>();
        }

        // 根据前置技能建树，虚拟根节点为 0
        for (int i = 1; i <= n; i++) {
            children[parent[i]].add(i);
        }

        dp = new int[n + 1][totalTime + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(dp[i], NEG);
        }

        dfs(0);

        // 虚拟根节点下的最大战斗力
        int ans = 0;
        for (int i = 0; i <= totalTime; i++) {
            ans = Math.max(ans, dp[0][i]);
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int totalTime = sc.nextInt();

        int[] cost = new int[n + 1];
        int[] power = new int[n + 1];
        int[] parent = new int[n + 1];

        for (int i = 0; i < n; i++) {
            int skillId = sc.nextInt();
            cost[skillId] = sc.nextInt();
            power[skillId] = sc.nextInt();

            int k = sc.nextInt();
            if (k == 1) {
                parent[skillId] = sc.nextInt();
            }
        }

        System.out.println(solve(n, totalTime, cost, power, parent));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

const int NEG = -1000000000;

int totalTime;
vector<int> costValue;
vector<int> powerValue;
vector<vector<int>> children;
vector<vector<int>> dp;

void dfs(int u) {
    // 先学习当前技能
    if (costValue[u] <= totalTime) {
        dp[u][costValue[u]] = powerValue[u];
    }

    // 合并所有子技能
    for (int v : children[u]) {
        dfs(v);

        vector<int> newDp = dp[u];

        for (int i = 0; i <= totalTime; i++) {
            if (dp[u][i] == NEG) {
                continue;
            }

            for (int j = 0; i + j <= totalTime; j++) {
                if (dp[v][j] == NEG) {
                    continue;
                }

                newDp[i + j] = max(newDp[i + j], dp[u][i] + dp[v][j]);
            }
        }

        dp[u] = newDp;
    }
}

int solve(int n, int t, vector<int>& cost, vector<int>& power, vector<int>& parent) {
    totalTime = t;
    costValue = cost;
    powerValue = power;

    children.assign(n + 1, vector<int>());

    // 根据前置技能建树，虚拟根节点为 0
    for (int i = 1; i <= n; i++) {
        children[parent[i]].push_back(i);
    }

    dp.assign(n + 1, vector<int>(totalTime + 1, NEG));

    dfs(0);

    // 虚拟根节点下的最大战斗力
    int ans = 0;
    for (int i = 0; i <= totalTime; i++) {
        ans = max(ans, dp[0][i]);
    }

    return ans;
}

int main() {
    int n, t;
    cin >> n >> t;

    vector<int> cost(n + 1, 0);
    vector<int> power(n + 1, 0);
    vector<int> parent(n + 1, 0);

    for (int i = 0; i < n; i++) {
        int skillId, k;
        cin >> skillId;

        cin >> cost[skillId] >> power[skillId] >> k;

        if (k == 1) {
            cin >> parent[skillId];
        }
    }

    cout << solve(n, t, cost, power, parent) << endl;

    return 0;
}
```
