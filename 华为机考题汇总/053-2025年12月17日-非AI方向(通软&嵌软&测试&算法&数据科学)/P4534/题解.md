## 解题思路
1. 先用哈希表统计数组每个值出现次数 `freq[v]`（数组已排序但不影响，用哈希更快）。
2. 对每个询问：

   * 若 `(x=0)：(0^K=0)（(K>=1)）`，所以只可能是 `(a=y)`
     总数=峰值=`freq[y]`
   * 若 `(x=1)：(1^K=1)`，所以只可能是 `(a=y+1)`
     总数=峰值=`freq[y+1]`
   * 若 `(x>=2)：(x^K)` 严格递增，只需枚举 `(K=1,2,3......)`，不断乘 (x) 得到幂 `p`：

     * 若 `p + y > maxA` 则停止（`maxA` 为数组最大值）
     * 累加 `freq[p+y]`，并更新峰值
     * 下一次令 `p *= x`，注意用“乘前判断”避免溢出：若 `p > (maxA - y) / x` 则下一次会超界，直接停

相关算法：哈希表计数、幂次枚举（指数增长）、溢出/上界剪枝。

## 复杂度分析

* 预处理建表：时间 (O(n))，空间 (O(n))（按不同值计数）。
* 每次询问：

  * $(x\in{0,1})$ 为 $(O(1))$
  * $(x\ge2)$ 枚举次数约为 $(\lfloor \log_x(maxA-y)\rfloor + 1)$，最坏在 $(x=2)$ 时约 $34$ 次（因为 ($a[i]\le 10^{10})$）
* 总体时间：$(O(n + m\log maxA))$，本题约 $(O(n + 34m))$，可通过。
* 总体空间：$(O(n))$

## 代码实现

### Python

```python
import sys

# 功能函数：返回每个询问的(总数, 峰值)
def solve(a, queries):
    freq = {}
    for v in a:
        freq[v] = freq.get(v, 0) + 1
    maxA = a[-1]

    ans = []
    for x, y in queries:
        # x=0: 0^K=0(K>=1)，只可能等于y
        if x == 0:
            c = freq.get(y, 0)
            ans.append((c, c))
            continue

        # x=1: 1^K=1
        if x == 1:
            c = freq.get(y + 1, 0)
            ans.append((c, c))
            continue

        # x>=2：枚举幂 p=x^K
        total = 0
        peak = 0
        limit = maxA - y  # 需要 p <= limit
        if limit < 0:
            ans.append((0, 0))
            continue

        p = x  # K=1
        while p <= limit:
            v = p + y
            c = freq.get(v, 0)
            if c:
                total += c
                if c > peak:
                    peak = c

            # 乘前判断，避免溢出/无意义循环
            if p > limit // x:
                break
            p *= x

        ans.append((total, peak))
    return ans

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    queries = [tuple(map(int, input().split())) for _ in range(m)]
    res = solve(a, queries)
    sys.stdout.write("\n".join(f"{t} {p}" for t, p in res))

if __name__ == "__main__":
    main()
```

```java
import java.io.*;
import java.util.*;

public class Main {

    // 功能函数：返回每个询问的(总数, 峰值)
    static List<long[]> solve(long[] a, long[][] queries) {
        Map<Long, Long> freq = new HashMap<>();
        for (long v : a) {
            freq.put(v, freq.getOrDefault(v, 0L) + 1);
        }

        long maxA = a[a.length - 1];
        List<long[]> ans = new ArrayList<>();

        for (long[] q : queries) {
            long x = q[0], y = q[1];

            // x=0: 0^K=0(K>=1)，只可能等于y
            if (x == 0) {
                long c = freq.getOrDefault(y, 0L);
                ans.add(new long[]{c, c});
                continue;
            }

            // x=1: 1^K=1
            if (x == 1) {
                long c = freq.getOrDefault(y + 1, 0L);
                ans.add(new long[]{c, c});
                continue;
            }

            // x>=2：枚举幂 p=x^K
            long total = 0, peak = 0;
            long limit = maxA - y; // 需要 p <= limit
            if (limit < 0) {
                ans.add(new long[]{0, 0});
                continue;
            }

            long p = x; // K=1
            while (p <= limit) {
                long v = p + y;
                long c = freq.getOrDefault(v, 0L);
                if (c > 0) {
                    total += c;
                    if (c > peak) {
                        peak = c;
                    }
                }

                // 乘前判断，避免溢出/无意义循环
                if (p > limit / x) {
                    break;
                }
                p *= x;
            }

            ans.add(new long[]{total, peak});
        }

        return ans;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);

        int n = fs.nextInt();
        int m = fs.nextInt();

        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = fs.nextLong();
        }

        long[][] queries = new long[m][2];
        for (int i = 0; i < m; i++) {
            queries[i][0] = fs.nextLong();
            queries[i][1] = fs.nextLong();
        }

        List<long[]> res = solve(a, queries);

        StringBuilder sb = new StringBuilder();
        for (long[] item : res) {
            sb.append(item[0]).append(' ').append(item[1]).append('\n');
        }
        System.out.print(sb);
    }

    // 快读
    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        FastScanner(InputStream is) {
            in = is;
        }

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        long nextLong() throws IOException {
            long sign = 1, val = 0;
            int c;
            do {
                c = read();
            } while (c <= ' ' && c != -1);

            if (c == '-') {
                sign = -1;
                c = read();
            }

            while (c > ' ') {
                val = val * 10 + (c - '0');
                c = read();
            }
            return val * sign;
        }

        int nextInt() throws IOException {
            return (int) nextLong();
        }
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

// 功能函数：返回每个询问的(总数, 峰值)
vector<pair<long long, long long>> solve(const vector<long long>& a, const vector<pair<long long, long long>>& queries) {
    unordered_map<long long, long long> freq;
    for (long long v : a) {
        freq[v]++;
    }

    long long maxA = a.back();
    vector<pair<long long, long long>> ans;

    for (auto [x, y] : queries) {
        // x=0: 0^K=0(K>=1)，只可能等于y
        if (x == 0) {
            long long c = freq.count(y) ? freq[y] : 0;
            ans.push_back({c, c});
            continue;
        }

        // x=1: 1^K=1
        if (x == 1) {
            long long c = freq.count(y + 1) ? freq[y + 1] : 0;
            ans.push_back({c, c});
            continue;
        }

        // x>=2：枚举幂 p=x^K
        long long total = 0, peak = 0;
        long long limit = maxA - y;  // 需要 p <= limit
        if (limit < 0) {
            ans.push_back({0, 0});
            continue;
        }

        long long p = x;  // K=1
        while (p <= limit) {
            long long v = p + y;
            long long c = freq.count(v) ? freq[v] : 0;
            if (c) {
                total += c;
                peak = max(peak, c);
            }

            // 乘前判断，避免溢出/无意义循环
            if (p > limit / x) break;
            p *= x;
        }

        ans.push_back({total, peak});
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<pair<long long, long long>> queries(m);
    for (int i = 0; i < m; i++) {
        cin >> queries[i].first >> queries[i].second;
    }

    auto res = solve(a, queries);

    for (auto [t, p] : res) {
        cout << t << ' ' << p << '\n';
    }

    return 0;
}
```
