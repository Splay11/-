## 解题思路

### 核心思路

本题要求实现一个带容量上限 $K$ 的 KV Cache 管理器，核心操作是在插入新 token 时，若缓存超出容量则淘汰注意力分数最低的 token（分数相同则淘汰位置编号较小者），并在查询时按位置升序输出所有缓存项。

需要维护两种检索需求：
1. **快速找到分数最小的 token**（用于淘汰）：使用**最小堆**或**平衡二叉搜索树**，以 $(score, pos)$ 为键，每次淘汰时取堆顶（或集合首元素）即可得到当前分数最低的 token。
2. **按位置升序输出**（用于查询）：由于输入的 token 位置严格递增，所有被插入缓存的位置天然有序，查询时只需对字典的键排序即可。

### 实现方法

- **数据结构**：使用哈希表（`dict` / `HashMap` / `unordered_map`）按位置存储每个 token 的完整数据（分数、Key 向量、Value 向量），键为整数位置 $pos$。
- **淘汰机制**：
  - Python 使用 `heapq` 维护最小堆，堆元素为 $(score, pos)$。由于堆不支持直接删除，采用**惰性删除**：淘汰时不断弹出堆顶，直到找到一个仍在哈希表中的 $pos$，将其从哈希表中移除。
  - C++ 使用 `set<pair<double, int>>` 维护分数排序，可直接通过 `begin()` 获取并删除最小分数元素，无需惰性删除。
  - Java 使用 `PriorityQueue<long[]>` 维护最小堆（将 $score$ 转换为 `long` 位表示以保证非负浮点数的比较正确性），同样采用惰性删除。
- **查询**：对哈希表的所有键排序后按顺序输出对应数据。

---

## 复杂度分析

- **时间复杂度**：
  - ADD 操作：堆/集合的插入为 $O(\log K)$，淘汰时弹出（惰性删除可能多次弹出，但每个元素最多被弹出一次）均摊为 $O(\log K)$。总体 $O(N \log K)$。
  - QUERY 操作：需要对最多 $K$ 个键排序，复杂度 $O(K \log K)$。若查询次数为 $Q$，总查询复杂度 $O(Q \cdot K \log K)$，均摊仍为 $O(N \log K)$（每个 token 至多在一次查询中被输出一次排序）。

- **空间复杂度**：哈希表与堆/集合均最多同时存储 $K+1$ 个元素，空间复杂度 $O(K)$。

## 代码实现

### Python

```python
import sys
import heapq


def solve():
    # 读取所有输入行
    data = sys.stdin.read().splitlines()
    it = iter(data)

    K, N = map(int, next(it).split())

    cache = {}          # pos -> (score_str, key, val)，保留原始 score 字符串以保证输出精度
    heap = []           # 最小堆: (score_float, pos)，用于快速找到分数最低的 token

    out_lines = []

    for _ in range(N):
        line = next(it)
        parts = line.split()

        if parts[0] == 'ADD':
            pos = int(parts[1])
            score = float(parts[2])
            score_str = parts[2]  # 保留原始字符串表示

            # 读取 Key 和 Value 向量
            key = list(map(float, next(it).split()))
            val = list(map(float, next(it).split()))

            # 插入缓存
            cache[pos] = (score_str, key, val)
            heapq.heappush(heap, (score, pos))

            # 超过容量上限，淘汰分数最低的 token
            if len(cache) > K:
                while heap:
                    s, p = heapq.heappop(heap)
                    if p in cache:  # 惰性删除：只处理仍在缓存中的元素
                        del cache[p]
                        out_lines.append(f"PRUNED {p}")
                        break
        else:  # QUERY
            # 按位置升序排列
            items = sorted(cache.items())
            out_lines.append(str(len(items)))
            for p, (s_str, k, v) in items:
                out_lines.append(f"{p} {s_str}")
                out_lines.append(' '.join(map(str, k)))
                out_lines.append(' '.join(map(str, v)))

    sys.stdout.write('\n'.join(out_lines))


if __name__ == '__main__':
    solve()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    // 缓存条目：存储分数、Key 向量、Value 向量
    static class Entry {
        double score;
        double[] key, val;

        Entry(double s, double[] k, double[] v) {
            score = s; key = k; val = v;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] first = br.readLine().split(" ");
        int K = Integer.parseInt(first[0]);
        int N = Integer.parseInt(first[1]);

        // 缓存: pos -> Entry
        Map<Integer, Entry> cache = new HashMap<>();
        // 最小堆: 按(score, pos)排序，score 使用 doubleToLongBits 转换以支持非负浮点数比较
        PriorityQueue<long[]> heap = new PriorityQueue<>((a, b) -> {
            if (a[0] != b[0]) return Long.compare(a[0], b[0]);
            return Long.compare(a[1], b[1]);
        });

        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < N; i++) {
            String line = br.readLine();
            String[] parts = line.split(" ");

            if (parts[0].equals("ADD")) {
                int pos = Integer.parseInt(parts[1]);
                double score = Double.parseDouble(parts[2]);

                // 读取 Key 向量 (4 个浮点数)
                double[] key = new double[4];
                String[] kp = br.readLine().split(" ");
                for (int j = 0; j < 4; j++)
                    key[j] = Double.parseDouble(kp[j]);

                // 读取 Value 向量 (4 个浮点数)
                double[] val = new double[4];
                String[] vp = br.readLine().split(" ");
                for (int j = 0; j < 4; j++)
                    val[j] = Double.parseDouble(vp[j]);

                // 插入缓存和堆
                cache.put(pos, new Entry(score, key, val));
                heap.offer(new long[]{Double.doubleToLongBits(score), pos});

                // 超出容量，淘汰分数最低的 token
                if (cache.size() > K) {
                    while (!heap.isEmpty()) {
                        long[] top = heap.poll();
                        int p = (int) top[1];
                        if (cache.containsKey(p)) {  // 惰性删除
                            cache.remove(p);
                            sb.append("PRUNED ").append(p).append("\n");
                            break;
                        }
                    }
                }
            } else {  // QUERY
                // 按位置升序排序后输出
                List<Integer> keys = new ArrayList<>(cache.keySet());
                Collections.sort(keys);
                sb.append(keys.size()).append("\n");
                for (int p : keys) {
                    Entry e = cache.get(p);
                    sb.append(p).append(" ").append(e.score).append("\n");
                    for (int j = 0; j < 4; j++) {
                        if (j > 0) sb.append(" ");
                        sb.append(e.key[j]);
                    }
                    sb.append("\n");
                    for (int j = 0; j < 4; j++) {
                        if (j > 0) sb.append(" ");
                        sb.append(e.val[j]);
                    }
                    sb.append("\n");
                }
            }
        }
        System.out.print(sb);
    }
}
```

### C++


```cpp
#include <bits/stdc++.h>
using namespace std;

// 缓存条目结构体
struct Node {
    double score;
    vector<double> key, val;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int K, N;
    cin >> K >> N;

    map<int, Node> cache;            // pos -> 缓存数据，自动按位置排序
    set<pair<double, int>> minSet;   // 按 (score, pos) 排序，用于快速取最小值淘汰

    string op;
    for (int i = 0; i < N; i++) {
        cin >> op;

        if (op == "ADD") {
            int pos;
            double score;
            cin >> pos >> score;

            // 读取 Key 和 Value 向量
            vector<double> key(4), val(4);
            for (int j = 0; j < 4; j++) cin >> key[j];
            for (int j = 0; j < 4; j++) cin >> val[j];

            // 插入缓存与有序集合
            cache[pos] = {score, key, val};
            minSet.insert({score, pos});

            // 超出容量，淘汰分数最低的 token
            if ((int)cache.size() > K) {
                auto it = minSet.begin();       // 分数最小的元素
                int evict = it->second;
                minSet.erase(it);
                cache.erase(evict);
                cout << "PRUNED " << evict << "\n";
            }
        } else {  // QUERY
            // map 已经按位置升序排列，直接遍历即可
            cout << cache.size() << "\n";
            
            // 设置浮点数输出格式：固定小数位，保留1位
            cout << fixed << setprecision(1);
            
            for (auto &[p, node] : cache) {
                cout << p << " " << node.score << "\n";
                for (int j = 0; j < 4; j++) {
                    cout << (j ? " " : "") << node.key[j];
                }
                cout << "\n";
                for (int j = 0; j < 4; j++) {
                    cout << (j ? " " : "") << node.val[j];
                }
                cout << "\n";
            }
        }
    }
    return 0;
}
```
