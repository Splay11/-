## 解题思路

本题是一个典型的有向图拓扑排序问题。

每个微服务看作一个点。若服务 $A$ 依赖服务 $B$，表示 $B$ 必须先于 $A$ 部署，因此建边：

$B \rightarrow A$

同时，$A$ 的入度加 $1$。

如果图中不存在环，则可以完成部署。部署顺序需要满足：

当一个服务入度为 $0$ 时，说明它依赖的服务都已经部署，可以加入候选集合。

如果同时有多个入度为 $0$ 的服务，需要按照服务名的字典序选择最小的服务先部署，因此使用小根堆维护当前所有入度为 $0$ 的服务。

算法流程：

$1.$ 读入所有服务及其依赖关系。

$2.$ 对每个依赖服务 $up$，建立边 $up \rightarrow service$。

$3.$ 统计每个服务的入度。

$4.$ 将所有入度为 $0$ 的服务加入小根堆。

$5.$ 每次取出字典序最小的服务加入答案，并更新它指向的服务入度。

$6.$ 如果最终答案数量等于 $N$，说明无环，输出部署顺序。

$7.$ 否则说明存在循环依赖，输出 $Can\ not\ deploy$。

## 复杂度分析

设服务数量为 $N$，依赖关系总数为 $E$。

时间复杂度为 $O((N+E)\log N)$。

其中每个点最多入堆出堆一次，每条边最多遍历一次。

空间复杂度为 $O(N+E)$。

用于存储图、入度数组和小根堆。

## 代码实现

### Python

```python
import sys
import heapq


def deploy_order(n, lines):
    graph = {}
    indegree = {}
    services = []

    # 初始化所有服务
    for parts in lines:
        name = parts[0]
        services.append(name)
        graph[name] = []
        indegree[name] = 0

    # 建图：依赖服务必须先部署，所以 upstream -> name
    for parts in lines:
        name = parts[0]
        m = int(parts[1])

        for i in range(m):
            up = parts[2 + i]
            graph[up].append(name)
            indegree[name] += 1

    heap = []

    # 将入度为 0 的服务加入小根堆
    for name in services:
        if indegree[name] == 0:
            heapq.heappush(heap, name)

    ans = []

    # 拓扑排序
    while heap:
        cur = heapq.heappop(heap)
        ans.append(cur)

        for nxt in graph[cur]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                heapq.heappush(heap, nxt)

    # 如果没有部署完，说明存在环
    if len(ans) != n:
        return "Can not deploy"

    return " ".join(ans)


def main():
    data = sys.stdin.read().strip().splitlines()
    if not data:
        return

    n = int(data[0].strip())
    lines = []

    for i in range(1, n + 1):
        lines.append(data[i].strip().split())

    print(deploy_order(n, lines))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {

    static String deployOrder(int n, List<String[]> lines) {
        Map<String, List<String>> graph = new HashMap<>();
        Map<String, Integer> indegree = new HashMap<>();
        List<String> services = new ArrayList<>();

        // 初始化所有服务
        for (String[] parts : lines) {
            String name = parts[0];
            services.add(name);
            graph.put(name, new ArrayList<>());
            indegree.put(name, 0);
        }

        // 建图：依赖服务必须先部署，所以 upstream -> name
        for (String[] parts : lines) {
            String name = parts[0];
            int m = Integer.parseInt(parts[1]);

            for (int i = 0; i < m; i++) {
                String up = parts[2 + i];
                graph.get(up).add(name);
                indegree.put(name, indegree.get(name) + 1);
            }
        }

        // 小根堆保证字典序最小的服务优先部署
        PriorityQueue<String> pq = new PriorityQueue<>();

        for (String name : services) {
            if (indegree.get(name) == 0) {
                pq.offer(name);
            }
        }

        List<String> ans = new ArrayList<>();

        // 拓扑排序
        while (!pq.isEmpty()) {
            String cur = pq.poll();
            ans.add(cur);

            for (String next : graph.get(cur)) {
                indegree.put(next, indegree.get(next) - 1);
                if (indegree.get(next) == 0) {
                    pq.offer(next);
                }
            }
        }

        // 如果没有部署完，说明存在环
        if (ans.size() != n) {
            return "Can not deploy";
        }

        return String.join(" ", ans);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        List<String[]> lines = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            String name = sc.next();
            int m = sc.nextInt();

            String[] parts = new String[m + 2];
            parts[0] = name;
            parts[1] = String.valueOf(m);

            for (int j = 0; j < m; j++) {
                parts[2 + j] = sc.next();
            }

            lines.add(parts);
        }

        System.out.println(deployOrder(n, lines));
    }
}
```

### C++

```cpp
#include <bits/stdc++.h>
using namespace std;

string deployOrder(int n, vector<vector<string>>& lines) {
    unordered_map<string, vector<string>> graph;
    unordered_map<string, int> indegree;
    vector<string> services;

    // 初始化所有服务
    for (auto& parts : lines) {
        string name = parts[0];
        services.push_back(name);
        graph[name] = vector<string>();
        indegree[name] = 0;
    }

    // 建图：依赖服务必须先部署，所以 upstream -> name
    for (auto& parts : lines) {
        string name = parts[0];
        int m = stoi(parts[1]);

        for (int i = 0; i < m; i++) {
            string up = parts[2 + i];
            graph[up].push_back(name);
            indegree[name]++;
        }
    }

    // 小根堆保证字典序最小的服务优先部署
    priority_queue<string, vector<string>, greater<string>> pq;

    for (string& name : services) {
        if (indegree[name] == 0) {
            pq.push(name);
        }
    }

    vector<string> ans;

    // 拓扑排序
    while (!pq.empty()) {
        string cur = pq.top();
        pq.pop();
        ans.push_back(cur);

        for (string& next : graph[cur]) {
            indegree[next]--;
            if (indegree[next] == 0) {
                pq.push(next);
            }
        }
    }

    // 如果没有部署完，说明存在环
    if ((int)ans.size() != n) {
        return "Can not deploy";
    }

    string res;
    for (int i = 0; i < (int)ans.size(); i++) {
        if (i > 0) res += " ";
        res += ans[i];
    }

    return res;
}

int main() {
    int n;
    cin >> n;

    vector<vector<string>> lines;

    for (int i = 0; i < n; i++) {
        string name;
        int m;
        cin >> name >> m;

        vector<string> parts;
        parts.push_back(name);
        parts.push_back(to_string(m));

        for (int j = 0; j < m; j++) {
            string up;
            cin >> up;
            parts.push_back(up);
        }

        lines.push_back(parts);
    }

    cout << deployOrder(n, lines) << endl;

    return 0;
}
```
