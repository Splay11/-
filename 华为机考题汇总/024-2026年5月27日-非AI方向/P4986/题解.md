## 解题思路

本题的关键是：碎片必须按编号升序收集，但相同编号的多个碎片可以选择任意顺序收集。

由于网格最大只有 $20 \times 20$，可以先用 $BFS$ 求出关键点之间的最短距离。

关键点包括：

起点 $2$、所有碎片 $4-9$、宝箱 $3$。

具体做法：

$1$. 先扫描网格，记录起点、宝箱和所有碎片位置。

$2$. 将碎片按照编号 $4$ 到 $9$ 分组。

$3$. 对每个关键点做一次 $BFS$，得到它到其他关键点的最短距离。

$4$. 按碎片编号从小到大处理。

对于同一编号的碎片，最多只有 $3$ 个，因此可以枚举它们的所有收集顺序。

用 $dp$ 表示：当前已经收集完前面所有编号碎片，并停在某个位置的最短耗时。

处理一组相同编号碎片时，从上一组的每个结束位置出发，枚举当前组内所有排列，更新最短耗时。

$5$. 所有碎片收集完后，再从当前位置走到宝箱，取最小值。

这样既保证了碎片编号升序，也能处理重复编号的最优顺序。

## 复杂度分析

设关键点数量为 $k$，网格大小为 $n \times m$。

每个关键点做一次 $BFS$，时间复杂度为：

$O(k \times n \times m)$

每个编号的碎片最多 $3$ 个，因此组内排列最多 $3! = 6$ 种，$DP$ 转移规模很小。

总时间复杂度为：

$O(k \times n \times m)$

空间复杂度为：

$O(k^2 + n \times m)$

其中 $k$ 最大不超过 $20$，可以轻松通过。

## 代码实现

### Python

```python
from collections import deque
from itertools import permutations
import sys


def bfs(grid, start):
    # 从一个关键点出发，求到网格所有位置的最短距离
    n, m = len(grid), len(grid[0])
    dist = [[-1] * m for _ in range(n)]
    q = deque()

    x, y = start
    dist[x][y] = 0
    q.append((x, y))

    dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

    while q:
        x, y = q.popleft()

        for dx, dy in dirs:
            nx, ny = x + dx, y + dy

            if 0 <= nx < n and 0 <= ny < m:
                # 只有障碍物 1 不能通过
                if grid[nx][ny] != 1 and dist[nx][ny] == -1:
                    dist[nx][ny] = dist[x][y] + 1
                    q.append((nx, ny))

    return dist


def shortest_time(grid):
    n, m = len(grid), len(grid[0])

    start = None
    chest = None
    groups_pos = [[] for _ in range(10)]

    # 记录起点、宝箱和碎片
    for i in range(n):
        for j in range(m):
            if grid[i][j] == 2:
                start = (i, j)
            elif grid[i][j] == 3:
                chest = (i, j)
            elif 4 <= grid[i][j] <= 9:
                groups_pos[grid[i][j]].append((i, j))

    points = [start]
    groups = [[] for _ in range(10)]

    # 给每个碎片分配关键点编号
    for num in range(4, 10):
        for pos in groups_pos[num]:
            groups[num].append(len(points))
            points.append(pos)

    chest_idx = len(points)
    points.append(chest)

    k = len(points)

    # 计算所有关键点之间的最短距离
    dist = [[0] * k for _ in range(k)]
    for i in range(k):
        d = bfs(grid, points[i])
        for j in range(k):
            x, y = points[j]
            dist[i][j] = d[x][y]

    # dp[位置编号] = 收集完当前编号前所有碎片后的最短耗时
    dp = {0: 0}

    # 按编号升序处理碎片
    for num in range(4, 10):
        if not groups[num]:
            continue

        new_dp = {}

        # 同编号碎片可以任意顺序收集，枚举所有排列
        for last_idx, cur_cost in dp.items():
            for order in permutations(groups[num]):
                cost = cur_cost
                pre = last_idx

                # 按当前排列依次收集
                for idx in order:
                    cost += dist[pre][idx]
                    pre = idx

                if pre not in new_dp or cost < new_dp[pre]:
                    new_dp[pre] = cost

        dp = new_dp

    # 最后进入宝箱
    ans = min(cost + dist[idx][chest_idx] for idx, cost in dp.items())
    return ans


def main():
    data = sys.stdin.read().strip().split()
    n, m = int(data[0]), int(data[1])

    grid = []
    p = 2
    for _ in range(n):
        row = []
        for _ in range(m):
            row.append(int(data[p]))
            p += 1
        grid.append(row)

    print(shortest_time(grid))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static int n, m;
    static int[][] grid;
    static int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

    static class Point {
        int x, y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    // 从一个关键点出发，求到所有格子的最短距离
    static int[][] bfs(Point start) {
        int[][] dist = new int[n][m];

        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], -1);
        }

        Queue<Point> queue = new LinkedList<>();
        dist[start.x][start.y] = 0;
        queue.offer(start);

        while (!queue.isEmpty()) {
            Point cur = queue.poll();

            for (int[] d : dirs) {
                int nx = cur.x + d[0];
                int ny = cur.y + d[1];

                if (nx >= 0 && nx < n && ny >= 0 && ny < m) {
                    // 只有障碍物 1 不能通过
                    if (grid[nx][ny] != 1 && dist[nx][ny] == -1) {
                        dist[nx][ny] = dist[cur.x][cur.y] + 1;
                        queue.offer(new Point(nx, ny));
                    }
                }
            }
        }

        return dist;
    }

    // 枚举当前组碎片的所有收集顺序
    static void dfsPerm(List<Integer> group, boolean[] used, List<Integer> path, List<List<Integer>> res) {
        if (path.size() == group.size()) {
            res.add(new ArrayList<>(path));
            return;
        }

        for (int i = 0; i < group.size(); i++) {
            if (!used[i]) {
                used[i] = true;
                path.add(group.get(i));

                dfsPerm(group, used, path, res);

                path.remove(path.size() - 1);
                used[i] = false;
            }
        }
    }

    static List<List<Integer>> getPermutations(List<Integer> group) {
        List<List<Integer>> res = new ArrayList<>();
        boolean[] used = new boolean[group.size()];
        dfsPerm(group, used, new ArrayList<>(), res);
        return res;
    }

    static int shortestTime() {
        Point start = null;
        Point chest = null;

        List<Point>[] groupPos = new ArrayList[10];
        for (int i = 0; i < 10; i++) {
            groupPos[i] = new ArrayList<>();
        }

        // 记录起点、宝箱和碎片
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == 2) {
                    start = new Point(i, j);
                } else if (grid[i][j] == 3) {
                    chest = new Point(i, j);
                } else if (grid[i][j] >= 4 && grid[i][j] <= 9) {
                    groupPos[grid[i][j]].add(new Point(i, j));
                }
            }
        }

        List<Point> points = new ArrayList<>();
        points.add(start);

        List<Integer>[] groups = new ArrayList[10];
        for (int i = 0; i < 10; i++) {
            groups[i] = new ArrayList<>();
        }

        // 给每个碎片分配关键点编号
        for (int num = 4; num <= 9; num++) {
            for (Point p : groupPos[num]) {
                groups[num].add(points.size());
                points.add(p);
            }
        }

        int chestIdx = points.size();
        points.add(chest);

        int k = points.size();
        int[][] dist = new int[k][k];

        // 计算所有关键点之间的最短距离
        for (int i = 0; i < k; i++) {
            int[][] d = bfs(points.get(i));

            for (int j = 0; j < k; j++) {
                Point p = points.get(j);
                dist[i][j] = d[p.x][p.y];
            }
        }

        // dp：key 是当前位置编号，value 是当前最短耗时
        Map<Integer, Integer> dp = new HashMap<>();
        dp.put(0, 0);

        // 按碎片编号升序处理
        for (int num = 4; num <= 9; num++) {
            if (groups[num].isEmpty()) {
                continue;
            }

            Map<Integer, Integer> newDp = new HashMap<>();
            List<List<Integer>> perms = getPermutations(groups[num]);

            for (int lastIdx : dp.keySet()) {
                int curCost = dp.get(lastIdx);

                for (List<Integer> order : perms) {
                    int cost = curCost;
                    int pre = lastIdx;

                    // 按当前排列收集同编号碎片
                    for (int idx : order) {
                        cost += dist[pre][idx];
                        pre = idx;
                    }

                    if (!newDp.containsKey(pre) || cost < newDp.get(pre)) {
                        newDp.put(pre, cost);
                    }
                }
            }

            dp = newDp;
        }

        // 最后走到宝箱
        int ans = Integer.MAX_VALUE;
        for (int idx : dp.keySet()) {
            ans = Math.min(ans, dp.get(idx) + dist[idx][chestIdx]);
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        n = sc.nextInt();
        m = sc.nextInt();

        grid = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = sc.nextInt();
            }
        }

        System.out.println(shortestTime());
    }
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, m;
vector<vector<int>> grid;
int dx[4] = {1, -1, 0, 0};
int dy[4] = {0, 0, 1, -1};

// 从一个关键点出发，求到所有格子的最短距离
vector<vector<int>> bfs(pair<int, int> start) {
    vector<vector<int>> dist(n, vector<int>(m, -1));
    queue<pair<int, int>> q;

    dist[start.first][start.second] = 0;
    q.push(start);

    while (!q.empty()) {
        pair<int, int> cur = q.front();
        q.pop();

        int x = cur.first;
        int y = cur.second;

        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];

            if (nx >= 0 && nx < n && ny >= 0 && ny < m) {
                // 只有障碍物 1 不能通过
                if (grid[nx][ny] != 1 && dist[nx][ny] == -1) {
                    dist[nx][ny] = dist[x][y] + 1;
                    q.push({nx, ny});
                }
            }
        }
    }

    return dist;
}

    pair<int, int> start, chest;
    vector<pair<int, int>> groupPos[10];

    // 记录起点、宝箱和碎片
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == 2) {
                start = {i, j};
            } else if (grid[i][j] == 3) {
                chest = {i, j};
            } else if (grid[i][j] >= 4 && grid[i][j] <= 9) {
                groupPos[grid[i][j]].push_back({i, j});
            }
        }
    }

    vector<pair<int, int>> points;
    points.push_back(start);

    vector<int> groups[10];

    // 给每个碎片分配关键点编号
    for (int num = 4; num <= 9; num++) {
        for (auto p : groupPos[num]) {
            groups[num].push_back((int)points.size());
            points.push_back(p);
        }
    }

    int chestIdx = points.size();
    points.push_back(chest);

    int k = points.size();
    vector<vector<int>> dist(k, vector<int>(k, 0));

    // 计算所有关键点之间的最短距离
    for (int i = 0; i < k; i++) {
        vector<vector<int>> d = bfs(points[i]);

        for (int j = 0; j < k; j++) {
            int x = points[j].first;
            int y = points[j].second;
            dist[i][j] = d[x][y];
        }
    }

    // dp[位置编号] = 收集完当前编号前所有碎片后的最短耗时
    map<int, int> dp;
    dp[0] = 0;

    // 按碎片编号升序处理
    for (int num = 4; num <= 9; num++) {
        if (groups[num].empty()) {
            continue;
        }

        map<int, int> newDp;
        vector<int> order = groups[num];

        sort(order.begin(), order.end());

        // 同编号碎片最多 3 个，直接枚举所有排列
        do {
            for (auto item : dp) {
                int lastIdx = item.first;
                int curCost = item.second;

                int cost = curCost;
                int pre = lastIdx;

                // 按当前排列依次收集碎片
                for (int idx : order) {
                    cost += dist[pre][idx];
                    pre = idx;

                    newDp[pre] = cost;
                }

        dp = newDp;

    // 最后进入宝箱
    int ans = INT_MAX;
    for (auto item : dp) {
        int idx = item.first;
        int cost = item.second;

        ans = min(ans, cost + dist[idx][chestIdx]);
    }

    return ans;
}

    grid.assign(n, vector<int>(m));
```
