## DFS暴力解法(机考中通过大部分用例)

由于边只会从时间早指向时间晚，所以不会有环，DFS 不需要 visited 也可以。

核心变化：

1. 先找关键节点。
2. 再建有向图。
3. 从每个关键节点出发 DFS。
4. DFS 过程中累计 Users。
5. 只有路径长度至少为 2，也就是至少走过一条边，才更新答案。

```python
import sys

def manhattan(a, b):
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

def main():
    input = sys.stdin.readline

    N, eps_dist, W_threshold = map(int, input().split())

    nodes = []
    for _ in range(N):
        x, y, t, w, users = map(int, input().split())
        nodes.append((x, y, t, w, users))

    is_key = [False] * N

    # 判断关键节点
    for i in range(N):
        total_w = 0
        for j in range(N):
            if manhattan(nodes[i], nodes[j]) <= eps_dist:
                total_w += nodes[j][3]

        if total_w >= W_threshold:
            is_key[i] = True

    # 建图
    graph = [[] for _ in range(N)]

    for i in range(N):
        if not is_key[i]:
            continue

        for j in range(N):
            if not is_key[j] or i == j:
                continue

            if manhattan(nodes[i], nodes[j]) <= eps_dist:
                if nodes[i][2] < nodes[j][2]:
                    graph[i].append(j)

    ans = 0

    def dfs(u, total_users, length):
        nonlocal ans

        # length >= 2 说明至少有两个节点，即形成了一条链
        if length >= 2:
            ans = max(ans, total_users)

        for v in graph[u]:
            dfs(v, total_users + nodes[v][4], length + 1)

    for i in range(N):
        if is_key[i]:
            dfs(i, nodes[i][4], 1)

    print(ans)

if __name__ == "__main__":
    main()
```

## 正确做法-解题思路

### 第一步：关键节点识别

对于每个基站i，需要计算其邻域负载和：
- 邻域定义：所有与i的曼哈顿距离 ≤ ε_dist 的基站（包括i自身）
- 邻域负载和 = 这些基站的负载w之和
- 如果邻域负载和 ≥ W_threshold，则i是关键节点

```python
# 伪代码示例
for i in range(N):
    total_w = 0
    for j in range(N):
        if |x_i-x_j| + |y_i-y_j| ≤ ε_dist:
            total_w += w_j
    if total_w ≥ W_threshold:
        is_key[i] = True
```

### 第二步：构建有向传播网络

只在关键节点之间建立连接：
- 首先检查两个关键节点的曼哈顿距离 ≤ ε_dist（直接关联）
- 然后检查时间戳：如果t_i < t_j，则建立 i → j 的有向边
- 注意：时间相等时不能建立边（避免环路）

### 第三步：求解最长路径

由于边总是从时间早指向时间晚，图是**有向无环图(DAG)**，可以使用动态规划：

**状态定义**：
- `dp[i]`：以第i个关键节点结尾的传播链的最大用户数

**状态转移**：
- `dp[i] = max(dp[j] + Users[i])`，其中j能到达i
- 初始值：`dp[i] = Users[i]`（只有自己）

**优化技巧**：
- 先按时间排序关键节点，保证转移时所有前驱都已计算
- 在转移时直接判断距离条件，无需显式建图

## 复杂度分析

- **时间复杂度**：O(N² + K²) ≤ O(N²) = 40000，N≤200完全可行
- **空间复杂度**：O(N + K) ≤ O(N)

```python
import sys

def solve(n, e_dist, w_threshold, nodes):
    """
    解决网络监控最长传播链问题
    
    参数:
        n: 基站总数
        e_dist: 空间距离阈值
        w_threshold: 负载阈值
        nodes: 基站信息列表，每个元素为(x, y, t, w, users)
    
    返回:
        最大用户总数
    """
    # 第一步：标记关键节点
    is_key = [False] * n
    
    # 遍历每个基站，判断是否为关键节点
    for i in range(n):
        total_w = 0  # 邻域负载和
        x1, y1, t1, w1, u1 = nodes[i]
        
        # 计算所有与i相邻的基站的负载和
        for j in range(n):
            x2, y2, t2, w2, u2 = nodes[j]
            # 曼哈顿距离判断
            if abs(x1 - x2) + abs(y1 - y2) <= e_dist:
                total_w += w2
        
        # 判断是否达到阈值
        if total_w >= w_threshold:
            is_key[i] = True
    
    # 收集所有关键节点
    keys = []
    for i in range(n):
        if is_key[i]:
            keys.append(nodes[i])
    
    # 如果没有关键节点，无法形成传播链
    if not keys:
        return 0
    
    # 按时间戳排序，确保DP时前驱节点都已处理
    # 时间戳是第3个元素（索引2）
    keys.sort(key=lambda x: x[2])
    
    m = len(keys)  # 关键节点数量
    dp = [0] * m   # dp[i]表示以第i个节点结尾的最大用户数
    ans = 0        # 全局最大用户数
    
    # 第三步：动态规划求解最长路径
    for i in range(m):
        xi, yi, ti, wi, ui = keys[i]
        # 初始化：只包含当前节点
        dp[i] = ui
        has_path = False
        
        # 枚举所有可能的前驱节点（时间更早的节点）
        for j in range(i):
            xj, yj, tj, wj, uj = keys[j]
            
            # 检查是否可以建立有向边 j -> i
            # 条件1：时间顺序（tj < ti）
            # 条件2：空间距离（曼哈顿距离 ≤ e_dist）
            if tj < ti and abs(xi - xj) + abs(yi - yj) <= e_dist:
                # 状态转移：以j结尾的最优路径 + 当前节点
                dp[i] = max(dp[i], dp[j] + ui)
                has_path = True

        # 更新全局最优解
        if has_path: ans = max(ans, dp[i])
    
    return ans

def main():
    # 读取输入
    input = sys.stdin.readline
    n, e_dist, w_threshold = map(int, input().split())
    
    nodes = []
    for _ in range(n):
        x, y, t, w, users = map(int, input().split())
        nodes.append((x, y, t, w, users))
    
    # 输出结果

if __name__ == "__main__":
    main()
```

```java
import java.util.*;

public class Main {
    
    /**
     * 解决网络监控最长传播链问题
     */
    static long solve(int n, long eDist, long wThreshold, long[][] nodes) {
        // 第一步：标记关键节点
        boolean[] isKey = new boolean[n];
        
        for (int i = 0; i < n; i++) {
            long totalW = 0;
            // 计算邻域负载和
            for (int j = 0; j < n; j++) {
                long dist = Math.abs(nodes[i][0] - nodes[j][0]) + 
                           Math.abs(nodes[i][1] - nodes[j][1]);
                if (dist <= eDist) {
                    totalW += nodes[j][3];  // 累加负载w
                }
            }
            // 判断是否为关键节点
            if (totalW >= wThreshold) {
                isKey[i] = true;
            }
        }
        
        // 第二步：收集关键节点
        List<long[]> keys = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (isKey[i]) {
                keys.add(nodes[i]);
            }
        }
        
        // 没有关键节点的情况
        if (keys.isEmpty()) {
            return 0;
        }
        
        // 按时间戳排序（第3个元素，索引2）
        keys.sort((a, b) -> Long.compare(a[2], b[2]));
        
        int m = keys.size();
        long[] dp = new long[m];
        long ans = 0;
        
        // 第三步：动态规划求解最长路径
        for (int i = 0; i < m; i++) {
            long[] cur = keys.get(i);
            dp[i] = cur[4];  // 初始化为当前节点的用户数
            boolean hasPath = false;
            
            // 枚举所有可能的前驱节点
            for (int j = 0; j < i; j++) {
                long[] pre = keys.get(j);
                long dist = Math.abs(cur[0] - pre[0]) + Math.abs(cur[1] - pre[1]);
                
                // 检查是否可以建立有向边 pre -> cur
                if (pre[2] < cur[2] && dist <= eDist) {
                    dp[i] = Math.max(dp[i], dp[j] + cur[4]);
                    hasPath = true;
                }
            }
            
            if(hasPath) ans = Math.max(ans, dp[i]);
        }
        
        return ans;
    }
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long eDist = sc.nextLong();
        long wThreshold = sc.nextLong();
        
        long[][] nodes = new long[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                nodes[i][j] = sc.nextLong();
            }
        }
        
        System.out.println(solve(n, eDist, wThreshold, nodes));
        sc.close();
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

/**
 * 解决网络监控最长传播链问题
 * 
 * @param n 基站总数
 * @param eDist 空间距离阈值
 * @param wThreshold 负载阈值
 * @param nodes 基站信息，每个元素为[x, y, t, w, users]
 * @return 最大用户总数
 */
long long solve(int n, long long eDist, long long wThreshold, 
                vector<vector<long long>>& nodes) {
    // 第一步：识别关键节点
    vector<bool> isKey(n, false);
    
    for (int i = 0; i < n; i++) {
        long long totalW = 0;
        
        // 计算邻域负载和
        for (int j = 0; j < n; j++) {
            long long dist = llabs(nodes[i][0] - nodes[j][0]) + 
                           llabs(nodes[i][1] - nodes[j][1]);
            if (dist <= eDist) {
                totalW += nodes[j][3];  // 累加负载w
            }
        }
        
        // 判断是否为关键节点
        if (totalW >= wThreshold) {
            isKey[i] = true;
        }
    }
    
    // 第二步：收集关键节点
    vector<vector<long long>> keys;
    for (int i = 0; i < n; i++) {
        if (isKey[i]) {
            keys.push_back(nodes[i]);
        }
    }
    
    // 没有关键节点的情况
    if (keys.empty()) {
        return 0;
    }
    
    // 按时间戳排序（第3个元素，索引2）
    sort(keys.begin(), keys.end(), 
         [](const vector<long long>& a, const vector<long long>& b) {
             return a[2] < b[2];
         });
    
    int m = keys.size();
    vector<long long> dp(m, 0);
    long long ans = 0;
    
    // 第三步：动态规划求解最长路径
    for (int i = 0; i < m; i++) {
        // 初始化：只包含当前节点
        dp[i] = keys[i][4];
        bool hasPath = false;
        
        // 枚举所有可能的前驱节点
        for (int j = 0; j < i; j++) {
            // 计算曼哈顿距离
            long long dist = llabs(keys[i][0] - keys[j][0]) + 
                           llabs(keys[i][1] - keys[j][1]);
            
            // 检查是否可以建立有向边 j -> i
            // 条件1：时间顺序
            // 条件2：空间距离满足要求
            if (keys[j][2] < keys[i][2] && dist <= eDist) {
                // 状态转移：以j结尾的最优路径 + 当前节点
                dp[i] = max(dp[i], dp[j] + keys[i][4]);
                hasPath = true;
            }
        }
        
        // 更新全局最优解
        if(hasPath) ans = max(ans, dp[i]);
    }
    
    return ans;
}

    long long eDist, wThreshold;

        for (int j = 0; j < 5; j++) {
            cin >> nodes[i][j];
        }
    }
```
