## 解题思路

这题适合用队列模拟。

用一个队列保存当前还没有领到粮食的副将需求，按顺序处理每一辆粮车，设当前粮车数量为 $K$：

* 若当前队首副将需求等于 $K$，则该副将直接领取并离队。
* 若当前队首副将需求小于 $K$，则继续从队首开始向后累加，找到最多的一段连续副将，使其总需求和 $\le K$，这些副将一起领取并离队。
* 若当前队首副将需求大于 $K$，则该副将本轮放弃，移动到队尾。

为了避免一辆车无限轮转，如果当前这辆车让整个队列轮转一圈后，仍然没有任何副将能够领取，那么这辆车就作废，处理下一辆车。

相关算法就是 队列模拟。

另外，这题还要求处理非法输入，除了输入格式错误外，还要判断是否超出题目给定范围：

* 两行数组都必须存在
* 两个数组长度都要满足 $1 \le n,m \le 1000$
* 每个元素都要满足 $1 \le value \le 100$

若不满足，直接输出 $-1$。

## 复杂度分析

设粮车数量为 $n$，副将数量为 $m$。

每辆车最多让当前队列轮转一圈，每次还要从队首开始尝试累加，因此最坏时间复杂度为 $O(n \times m)$。

由于 $n,m \le 1000$，这个复杂度可以通过。

空间复杂度为 $O(m)$，用于维护副将队列。

## 代码实现

### Python

```python
from collections import deque

# 判断输入是否合法
def check(cars, needs):
    if len(cars) == 0 or len(needs) == 0:
        return False
    if len(cars) > 1000 or len(needs) > 1000:
        return False
    for x in cars:
        if x < 1 or x > 100:
            return False
    for x in needs:
        if x < 1 or x > 100:
            return False
    return True

# 计算不能领到粮食的副将人数
def solve(cars, needs):
    q = deque(needs)

    # 依次处理每一辆车
    for k in cars:
        if not q:
            break

        size = len(q)
        success = False

        # 当前这辆车最多让队列轮转一圈
        for _ in range(size):
            total = 0
            cnt = 0

            # 从当前队首开始，找最多能满足的连续副将
            for x in q:
                if total + x <= k:
                    total += x
                    cnt += 1
                else:
                    break

            # 能满足至少一个副将
            if cnt > 0:
                for _ in range(cnt):
                    q.popleft()
                success = True
                break
            else:
                # 当前队首需求大于k，移到队尾
                q.append(q.popleft())

        # 如果这一辆车一圈下来没人能领，则作废
        if not success:
            continue

    return len(q)

if __name__ == "__main__":
    try:
        cars = list(map(int, input().split()))
        needs = list(map(int, input().split()))

        if not check(cars, needs):
            print(-1)
        else:
            print(solve(cars, needs))
    except:
        print(-1)
```

### Java

```java
import java.util.*;

public class Main {

    // 判断输入是否合法
    public static boolean check(int[] cars, int[] needs) {
        if (cars.length == 0 || needs.length == 0) {
            return false;
        }
        if (cars.length > 1000 || needs.length > 1000) {
            return false;
        }
        for (int x : cars) {
            if (x < 1 || x > 100) {
                return false;
            }
        }
        for (int x : needs) {
            if (x < 1 || x > 100) {
                return false;
            }
        }
        return true;
    }

    // 计算不能领到粮食的副将人数
    public static int solve(int[] cars, int[] needs) {
        Deque<Integer> queue = new LinkedList<>();

        // 初始化副将队列
        for (int x : needs) {
            queue.offerLast(x);
        }

        // 依次处理每一辆车
        for (int k : cars) {
            if (queue.isEmpty()) {
                break;
            }

            int size = queue.size();
            boolean success = false;

            // 当前这辆车最多让队列轮转一圈
            for (int i = 0; i < size; i++) {
                int total = 0;
                int cnt = 0;

                // 从当前队首开始，找最多能满足的连续副将
                for (int x : queue) {
                    if (total + x <= k) {
                        total += x;
                        cnt++;
                    } else {
                        break;
                    }
                }

                // 能满足至少一个副将
                if (cnt > 0) {
                    for (int j = 0; j < cnt; j++) {
                        queue.pollFirst();
                    }
                    success = true;
                    break;
                } else {
                    // 当前队首需求大于k，移到队尾
                    queue.offerLast(queue.pollFirst());
                }
            }
        }

        return queue.size();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        try {
            String line1 = sc.nextLine().trim();
            String line2 = sc.nextLine().trim();

            if (line1.length() == 0 || line2.length() == 0) {
                System.out.println(-1);
                return;
            }

            String[] s1 = line1.split(" ");
            String[] s2 = line2.split(" ");

            int[] cars = new int[s1.length];
            int[] needs = new int[s2.length];

            for (int i = 0; i < s1.length; i++) {
                cars[i] = Integer.parseInt(s1[i]);
            }
            for (int i = 0; i < s2.length; i++) {
                needs[i] = Integer.parseInt(s2[i]);
            }

            if (!check(cars, needs)) {
                System.out.println(-1);
            } else {
                System.out.println(solve(cars, needs));
            }
        } catch (Exception e) {
            System.out.println(-1);
        }
    }
}
```

### C++

```cpp
#include <iostream>
#include <sstream>
#include <deque>
#include <vector>
using namespace std;

// 判断输入是否合法
bool check(vector<int>& cars, vector<int>& needs) {
    if (cars.empty() || needs.empty()) {
        return false;
    }
    if ((int)cars.size() > 1000 || (int)needs.size() > 1000) {
        return false;
    }
    for (int x : cars) {
        if (x < 1 || x > 100) {
            return false;
        }
    }
    for (int x : needs) {
        if (x < 1 || x > 100) {
            return false;
        }
    }
    return true;
}

// 计算不能领到粮食的副将人数
int solve(vector<int>& cars, vector<int>& needs) {
    deque<int> q;

    // 初始化副将队列
    for (int x : needs) {
        q.push_back(x);
    }

    // 依次处理每一辆车
    for (int k : cars) {
        if (q.empty()) {
            break;
        }

        int size = q.size();
        bool success = false;

        // 当前这辆车最多让队列轮转一圈
        for (int i = 0; i < size; i++) {
            int total = 0;
            int cnt = 0;

            // 从当前队首开始，找最多能满足的连续副将
            for (int x : q) {
                if (total + x <= k) {
                    total += x;
                    cnt++;
                } else {
                    break;
                }
            }

            // 能满足至少一个副将
            if (cnt > 0) {
                while (cnt--) {
                    q.pop_front();
                }
                success = true;
                break;
            } else {
                // 当前队首需求大于k，移到队尾
                int x = q.front();
                q.pop_front();
                q.push_back(x);
            }
        }
    }

    return (int)q.size();
}

int main() {
    try {
        string line1, line2;
        getline(cin, line1);
        getline(cin, line2);

        if (line1.empty() || line2.empty()) {
            cout << -1 << endl;
            return 0;
        }

        vector<int> cars, needs;
        stringstream ss1(line1), ss2(line2);
        int x;

        while (ss1 >> x) {
            cars.push_back(x);
        }
        while (ss2 >> x) {
            needs.push_back(x);
        }

        if (!check(cars, needs)) {
            cout << -1 << endl;
        } else {
            cout << solve(cars, needs) << endl;
        }
    } catch (...) {
        cout << -1 << endl;
    }

    return 0;
}
```
