## 解题思路

设当前任务进度为 $s$，最后一次调用的 Agent 为 $i$。

最直接的动态规划状态是：

$$
dp[s][i]
$$

表示总进度达到 $s$，且最后一次调用 Agent $i$ 时的最小总代价。进度超过 $100$ 时统一记为 $100$。

如果下一次调用 Agent $j$，必须满足 $j\ne i$，状态转移为：

$$
dp[\min(100,s+p_j)][j]
=
\min\left(
dp[\min(100,s+p_j)][j],
dp[s][i]+c_j
\right)
$$

直接枚举前一个 Agent 和当前 Agent，时间复杂度为 $O(100N^2)$，没有必要。

对于固定进度 $s$，假设下一次准备调用 Agent $i$，只需要知道：

$$
\min_{j\ne i} dp[s][j]
$$

因此，对于每个进度，只需要保留以下两个状态：

* 代价最小的状态，以及该状态最后调用的 Agent。
* 最后调用的 Agent 与第一个状态不同的、代价第二小的状态。

设这两个最小代价分别为 $best1[s]$ 和 $best2[s]$，对应的 Agent 分别为 $id1[s]$ 和 $id2[s]$。

当下一次调用 Agent $i$ 时：

* 如果 $id1[s]\ne i$，可以从 $best1[s]$ 转移。
* 如果 $id1[s]=i$，为了避免连续调用同一个 Agent，只能从 $best2[s]$ 转移。

即：

$$
base=
\begin{cases}
best1[s], & id1[s]\ne i \
best2[s], & id1[s]=i
\end{cases}
$$

然后转移到：

$$
t=\min(100,s+p_i)
$$

新的代价为：

$$
base+c_i
$$

由于每次调用都会增加进度，因此按照进度从小到大处理即可。

初始时进度为 $0$，总代价为 $0$，并且不存在上一次调用的 Agent，可以将其编号设为 $-1$。最终若 $best1[100]$ 不存在，输出 `-1`；否则输出 $best1[100]$。

## 复杂度分析

需要枚举 $0$ 到 $99$ 的每个进度，并在每个进度下枚举全部 $N$ 个 Agent。

时间复杂度为：

$$
O(100N)
$$

由于 $100$ 是常数，也可以记为 $O(N)$。

保存所有 Agent 的参数需要 $O(N)$ 空间，动态规划数组只需要 $O(100)$ 空间。

空间复杂度为：

$$
O(N+100)=O(N)
$$

## 代码实现

### Python

```python
import sys

INF = 10 ** 30


def update(best1, best2, id1, id2, pos, cost, agent):
    """使用一个以 agent 结尾的新状态，更新 pos 进度下的两个最优状态。"""

    if id1[pos] == agent:
        # 已经存在以该 Agent 结尾的最优状态，只需更新其代价
        if cost < best1[pos]:
            best1[pos] = cost
        return

    if id2[pos] == agent:
        # 更新当前第二优状态，并在必要时与第一优状态交换
        if cost < best2[pos]:
            best2[pos] = cost
            if best2[pos] < best1[pos]:
                best1[pos], best2[pos] = best2[pos], best1[pos]
                id1[pos], id2[pos] = id2[pos], id1[pos]
        return

    if cost < best1[pos]:
        # 新状态成为第一优，原第一优下降为第二优
        best2[pos] = best1[pos]
        id2[pos] = id1[pos]
        best1[pos] = cost
        id1[pos] = agent
    elif cost < best2[pos]:
        # 新状态成为第二优
        best2[pos] = cost
        id2[pos] = agent


def min_cost(p, c):
    n = len(p)

    # best1、best2 保存每个进度下最后 Agent 不同的两个最小代价
    best1 = [INF] * 101
    best2 = [INF] * 101
    id1 = [-2] * 101
    id2 = [-2] * 101

    # 初始进度为 0，-1 表示还没有调用过任何 Agent
    best1[0] = 0
    id1[0] = -1

    for s in range(100):
        if best1[s] == INF:
            continue

        for i in range(n):
            # 不能连续调用同一个 Agent
            if id1[s] != i:
                base = best1[s]
            else:
                base = best2[s]

            if base == INF:
                continue

            # 超过 100 的进度统一压缩为 100
            t = min(100, s + p[i])
            new_cost = base + c[i]

            update(best1, best2, id1, id2, t, new_cost, i)

    if best1[100] == INF:
        return -1
    return best1[100]


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n = data[0]

    p = [0] * n
    c = [0] * n

    pos = 1
    for i in range(n):
        p[i] = data[pos]
        c[i] = data[pos + 1]
        pos += 2

    print(min_cost(p, c))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    static final long INF = Long.MAX_VALUE / 4;

    static void update(long[] best1, long[] best2,
                       int[] id1, int[] id2,
                       int pos, long cost, int agent) {
        if (id1[pos] == agent) {
            // 已经存在以该 Agent 结尾的最优状态，只需更新其代价
            if (cost < best1[pos]) {
                best1[pos] = cost;
            }
            return;
        }

        if (id2[pos] == agent) {
            // 更新当前第二优状态，并在必要时与第一优状态交换
            if (cost < best2[pos]) {
                best2[pos] = cost;

                if (best2[pos] < best1[pos]) {
                    long tempCost = best1[pos];
                    best1[pos] = best2[pos];
                    best2[pos] = tempCost;

                    int tempId = id1[pos];
                    id1[pos] = id2[pos];
                    id2[pos] = tempId;
                }
            }
            return;
        }

        if (cost < best1[pos]) {
            // 新状态成为第一优，原第一优下降为第二优
            best2[pos] = best1[pos];
            id2[pos] = id1[pos];

            best1[pos] = cost;
            id1[pos] = agent;
        } else if (cost < best2[pos]) {
            // 新状态成为第二优
            best2[pos] = cost;
            id2[pos] = agent;
        }
    }

    static long minCost(int[] p, long[] c) {
        int n = p.length;

        // 每个进度只保留最后 Agent 不同的两个最优状态
        long[] best1 = new long[101];
        long[] best2 = new long[101];
        int[] id1 = new int[101];
        int[] id2 = new int[101];

        for (int i = 0; i <= 100; i++) {
            best1[i] = INF;
            best2[i] = INF;
            id1[i] = -2;
            id2[i] = -2;
        }

        // 初始时进度为 0，-1 表示没有上一次调用的 Agent
        best1[0] = 0;
        id1[0] = -1;

        for (int s = 0; s < 100; s++) {
            if (best1[s] == INF) {
                continue;
            }

            for (int i = 0; i < n; i++) {
                long base;

                // 如果最优状态最后调用的就是当前 Agent，只能使用第二优状态
                if (id1[s] != i) {
                    base = best1[s];
                } else {
                    base = best2[s];
                }

                if (base == INF) {
                    continue;
                }

                // 超过 100 的进度统一记为 100
                int next = Math.min(100, s + p[i]);
                long newCost = base + c[i];

                update(best1, best2, id1, id2, next, newCost, i);
            }
        }

        if (best1[100] == INF) {
            return -1;
        }
        return best1[100];
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(
                new InputStreamReader(System.in)
        );

        int n = Integer.parseInt(br.readLine().trim());
        int[] p = new int[n];
        long[] c = new long[n];

        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            p[i] = Integer.parseInt(st.nextToken());
            c[i] = Long.parseLong(st.nextToken());
        }

        System.out.println(minCost(p, c));
    }
}
```

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const long long INF = (1LL << 62);

void updateState(vector<long long>& best1,
                 vector<long long>& best2,
                 vector<int>& id1,
                 vector<int>& id2,
                 int pos,
                 long long cost,
                 int agent) {
    if (id1[pos] == agent) {
        // 已经存在以该 Agent 结尾的最优状态，只需更新其代价
        best1[pos] = min(best1[pos], cost);
        return;
    }

    if (id2[pos] == agent) {
        // 更新当前第二优状态，并在必要时与第一优状态交换
        if (cost < best2[pos]) {
            best2[pos] = cost;

            if (best2[pos] < best1[pos]) {
                swap(best1[pos], best2[pos]);
                swap(id1[pos], id2[pos]);
            }
        }
        return;
    }

    if (cost < best1[pos]) {
        // 新状态成为第一优，原第一优下降为第二优
        best2[pos] = best1[pos];
        id2[pos] = id1[pos];

        best1[pos] = cost;
        id1[pos] = agent;
    } else if (cost < best2[pos]) {
        // 新状态成为第二优
        best2[pos] = cost;
        id2[pos] = agent;
    }
}

long long minCost(const vector<int>& p, const vector<long long>& c) {
    int n = p.size();

    // 每个进度只保留最后 Agent 不同的两个最优状态
    vector<long long> best1(101, INF);
    vector<long long> best2(101, INF);
    vector<int> id1(101, -2);
    vector<int> id2(101, -2);

    // 初始时进度为 0，-1 表示没有上一次调用的 Agent
    best1[0] = 0;
    id1[0] = -1;

    for (int s = 0; s < 100; s++) {
        if (best1[s] == INF) {
            continue;
        }

        for (int i = 0; i < n; i++) {
            long long base;

            // 不能连续调用同一个 Agent
            if (id1[s] != i) {
                base = best1[s];
            } else {
                base = best2[s];
            }

            if (base == INF) {
                continue;
            }

            // 超过 100 的进度统一压缩为 100
            int next = min(100, s + p[i]);
            long long newCost = base + c[i];

            updateState(best1, best2, id1, id2, next, newCost, i);
        }
    }

    if (best1[100] == INF) {
        return -1;
    }
    return best1[100];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<int> p(n);
    vector<long long> c(n);

    for (int i = 0; i < n; i++) {
        cin >> p[i] >> c[i];
    }

    cout << minCost(p, c) << '\n';

    return 0;
}
```
