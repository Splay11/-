## 解题思路

首先根据曼哈顿距离判断两个基站是否互为空间邻域。对于基站 $P_i(x_i,y_i)$ 和 $P_j(x_j,y_j)$，距离为

$$
d_{space}=|x_i-x_j|+|y_i-y_j|
$$

当 $d_{space}\le \epsilon_{dist}$ 时，$P_j$ 属于 $P_i$ 的空间邻域。遍历邻域时需要包含基站自身，因此每个基站的邻域数量至少为 $1$。

算法分为两个阶段：

1. 遍历每个基站，并枚举其他所有基站，统计其空间邻域内的基站数量。若数量大于等于 $MinPts$，则将该基站标记为核心点。
2. 再次遍历所有基站：

   * 若当前基站是核心点，属性标记为 `0`。
   * 若当前基站不是核心点，则枚举所有核心点。只要存在一个核心点与当前基站的曼哈顿距离不超过 $\epsilon_{dist}$，当前基站就是边界点，属性标记为 `1`。
   * 若不存在满足条件的核心点，则当前基站是噪声点，属性标记为 `2`。

本题只需要标注每个基站的空间属性，不需要进行 DBSCAN 的簇扩展和簇编号。

## 复杂度分析

设基站数量为 $N$。

核心点判定需要枚举所有基站对，时间复杂度为 $O(N^2)$。边界点判定最多同样需要枚举所有基站对，时间复杂度为 $O(N^2)$。

因此总时间复杂度为

$$
O(N^2)
$$

使用数组记录核心点状态和最终结果，空间复杂度为

$$
O(N)
$$

在 $N\le 2000$ 的数据范围内，该复杂度可以满足要求。

## 代码实现

### Python

```python
import sys

def solve(p, e, m):
    n = len(p)
    core = [False] * n

    # 统计每个基站的邻域数量，判断是否为核心点
    for i in range(n):
        cnt = 0
        for j in range(n):
            # 计算两个基站之间的曼哈顿距离
            d = abs(p[i][0] - p[j][0]) + abs(p[i][1] - p[j][1])
            if d <= e:
                cnt += 1

        # 邻域数量包含自身
        if cnt >= m:
            core[i] = True

    # 默认所有基站均为噪声点
    ans = [2] * n

    # 根据核心点信息标注每个基站
    for i in range(n):
        if core[i]:
            ans[i] = 0
        else:
            # 非核心点的邻域中存在核心点，则为边界点
            for j in range(n):
                if core[j]:
                    d = abs(p[i][0] - p[j][0]) + abs(p[i][1] - p[j][1])
                    if d <= e:
                        ans[i] = 1
                        break

    return ans

def main():
    # 读取全部输入数据
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, e, m = data[0], data[1], data[2]

    # 读取所有基站坐标
    p = []
    k = 3
    for _ in range(n):
        p.append((data[k], data[k + 1]))
        k += 2

    # 计算并输出每个基站的属性
    ans = solve(p, e, m)
    sys.stdout.write("\n".join(map(str, ans)))

if __name__ == "__main__":
    main()
```

import java.util.Scanner;

public class Main {
    static int[] solve(int[][] p, int e, int m) {
        int n = p.length;
        boolean[] core = new boolean[n];

        // 统计每个基站的邻域数量，判断是否为核心点
        for (int i = 0; i < n; i++) {
            int cnt = 0;

            for (int j = 0; j < n; j++) {
                // 计算两个基站之间的曼哈顿距离
                int d = Math.abs(p[i][0] - p[j][0])
                        + Math.abs(p[i][1] - p[j][1]);

                if (d <= e) {
                    cnt++;
                }
            }

            // 邻域数量包含基站自身
            if (cnt >= m) {
                core[i] = true;
            }
        }

        int[] ans = new int[n];

        // 根据核心点信息标注每个基站
        for (int i = 0; i < n; i++) {
            if (core[i]) {
                // 核心点
                ans[i] = 0;
            } else {
                // 默认标记为噪声点
                ans[i] = 2;

                // 检查邻域中是否存在核心点
                for (int j = 0; j < n; j++) {
                    if (core[j]) {
                        int d = Math.abs(p[i][0] - p[j][0])
                                + Math.abs(p[i][1] - p[j][1]);

                        if (d <= e) {
                            // 非核心点的邻域中存在核心点，为边界点
                            ans[i] = 1;
                            break;
                        }
                    }
                }
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 读取基站数量、距离门限和数量门限
        int n = sc.nextInt();
        int e = sc.nextInt();
        int m = sc.nextInt();

        // 读取所有基站坐标
        int[][] p = new int[n][2];
        for (int i = 0; i < n; i++) {
            p[i][0] = sc.nextInt();
            p[i][1] = sc.nextInt();
        }

        // 计算每个基站的属性
        int[] ans = solve(p, e, m);

        // 拼接并输出结果
        StringBuilder out = new StringBuilder();
        for (int x : ans) {
            out.append(x).append('\n');
        }

        System.out.print(out);
        sc.close();
    }
}

vector<int> solve(const vector<pair<int, int>>& p, int e, int m) {
    int n = p.size();
    vector<bool> core(n, false);

    // 统计每个基站的邻域数量，判断是否为核心点
    for (int i = 0; i < n; i++) {
        int cnt = 0;

        for (int j = 0; j < n; j++) {
            // 计算两个基站之间的曼哈顿距离
            int d = abs(p[i].first - p[j].first)
                    + abs(p[i].second - p[j].second);

            if (d <= e) {
                cnt++;
            }
        }

        // 邻域数量包含基站自身
        if (cnt >= m) {
            core[i] = true;
        }
    }

    // 默认所有基站均为噪声点
    vector<int> ans(n, 2);

    // 根据核心点信息标注每个基站
    for (int i = 0; i < n; i++) {
        if (core[i]) {
            // 核心点
            ans[i] = 0;
        } else {
            // 检查邻域中是否存在核心点
            for (int j = 0; j < n; j++) {
                if (core[j]) {
                    int d = abs(p[i].first - p[j].first)
                            + abs(p[i].second - p[j].second);

                    if (d <= e) {
                        // 非核心点的邻域中存在核心点，为边界点
                        ans[i] = 1;
                        break;
                    }
                }
            }
        }
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    // 读取基站数量、距离门限和数量门限
    int n, e, m;

    // 读取所有基站坐标

    for (int i = 0; i < n; i++) {
        cin >> p[i].first >> p[i].second;
    }

    vector<int> ans = solve(p, e, m);
