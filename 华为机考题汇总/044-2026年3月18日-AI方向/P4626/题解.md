## 解题思路

这道题本质上就是使用 $KNN$ 算法 对一个待分类的 $3$ 维特征向量进行分类。

已知有 $N$ 个带类别的语音特征向量，每个向量有 $3$ 个特征值和一个类别 $label$。对于待分类向量，我们需要：

1. 计算它与所有已知向量之间的欧式距离
2. 按距离从小到大排序
3. 取距离最近的 $K$ 个邻居
4. 统计这 $K$ 个邻居中每个类别出现的次数
5. 出现次数最多的类别就是最终分类结果

欧式距离公式为：

$$
d = \sqrt{(x_1-y_1)^2 + (x_2-y_2)^2 + (x_3-y_3)^2}
$$

由于比较距离大小时，平方根不会影响结果，因此实现时可以直接比较平方距离：

$$
(x_1-y_1)^2 + (x_2-y_2)^2 + (x_3-y_3)^2
$$

这样写更简洁，也能避免不必要的浮点开方运算。

实现方法上，只需要：

* 读入所有训练数据
* 定义函数计算待分类点到每个样本的距离
* 排序后取前 $K$ 个
* 用哈希表或映射统计类别出现次数
* 输出出现次数最多的类别即可

## 复杂度分析

设已知语音特征向量数量为 $N$。

时间复杂度：

* 计算所有距离需要 $O(N)$
* 对所有距离排序需要 $O(N \log N)$
* 统计前 $K$ 个邻居类别需要 $O(K)$

总时间复杂度为：

$$
O(N \log N)
$$

空间复杂度：

* 需要存储所有样本及其距离信息，空间复杂度为：

$$
O(N)
$$

这个复杂度对于本题是完全合适的。

## 代码实现

### Python

```python
import sys
import numpy as np

def solve():
    lines = sys.stdin.read().strip().splitlines()
    if not lines:
        return

    # 读取 N 和 K
    N, K = map(int, lines[0].split())

    # 读取训练数据
    train = []
    for i in range(1, N + 1):
        train.append(list(map(float, lines[i].split())))

    train = np.array(train)
    
    # 前3列是特征，最后1列是标签
    X_train = train[:, :3]
    y_train = train[:, 3].astype(int)

    # 读取待分类向量
    x_test = np.array(list(map(float, lines[N + 1].split())))

    # 计算待分类点到所有训练样本的平方欧式距离
    # 不开平方，因为平方根不影响大小比较
    diff = X_train - x_test
    distances = np.sum(diff * diff, axis=1)

    # 按距离从小到大排序，取最近的 K 个邻居下标
    nearest_indices = np.argsort(distances)[:K]

    # 取出这 K 个邻居的标签
    nearest_labels = y_train[nearest_indices]

    # 统计每个类别出现次数
    label_count = {}
    for label in nearest_labels:
        label_count[label] = label_count.get(label, 0) + 1

    # 找出出现次数最多的类别
    ans = max(label_count, key=label_count.get)

    print(ans)

if __name__ == "__main__":
    solve()

```

### Java

```java
import java.util.*;

// ACM风格，类名必须为Main
public class Main {

    // 样本类，保存距离和类别
    static class Node {
        double dist;
        int label;

        Node(double dist, int label) {
            this.dist = dist;
            this.label = label;
        }
    }

    // KNN分类函数，返回待分类向量的类别
    public static int knnClassify(double[][] data, int[] labels, double[] target, int k) {
        int n = data.length;
        Node[] arr = new Node[n];

        // 计算目标向量到每个已知向量的平方距离
        for (int i = 0; i < n; i++) {
            double dx = data[i][0] - target[0];
            double dy = data[i][1] - target[1];
            double dz = data[i][2] - target[2];
            double dist = dx * dx + dy * dy + dz * dz;
            arr[i] = new Node(dist, labels[i]);
        }

        // 按距离从小到大排序
        Arrays.sort(arr, new Comparator<Node>() {
            @Override
            public int compare(Node a, Node b) {
                return Double.compare(a.dist, b.dist);
            }
        });

        // 统计前k个最近邻的类别出现次数
        HashMap<Integer, Integer> count = new HashMap<>();
        for (int i = 0; i < k; i++) {
            int label = arr[i].label;
            count.put(label, count.getOrDefault(label, 0) + 1);
        }

        // 找出出现次数最多的类别
        int ans = -1;
        int maxCnt = -1;
        for (Map.Entry<Integer, Integer> entry : count.entrySet()) {
            int label = entry.getKey();
            int cnt = entry.getValue();
            if (cnt > maxCnt) {
                maxCnt = cnt;
                ans = label;
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 读入N和K
        int n = sc.nextInt();
        int k = sc.nextInt();

        double[][] data = new double[n][3];
        int[] labels = new int[n];

        // 读入已知语音特征向量和类别
        for (int i = 0; i < n; i++) {
            data[i][0] = sc.nextDouble();
            data[i][1] = sc.nextDouble();
            data[i][2] = sc.nextDouble();
            labels[i] = sc.nextInt();
        }

        // 读入待分类向量
        double[] target = new double[3];
        target[0] = sc.nextDouble();
        target[1] = sc.nextDouble();
        target[2] = sc.nextDouble();

        // 输出分类结果
        System.out.println(knnClassify(data, labels, target, k));
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <map>
using namespace std;

// 保存距离和类别
struct Node {
    double dist;
    int label;
};

// KNN分类函数，返回待分类向量的类别
int knnClassify(const vector<vector<double>>& data, const vector<int>& labels, const vector<double>& target, int k) {
    vector<Node> arr;

    // 计算目标向量到每个已知向量的平方距离
    for (int i = 0; i < data.size(); i++) {
        double dx = data[i][0] - target[0];
        double dy = data[i][1] - target[1];
        double dz = data[i][2] - target[2];
        double dist = dx * dx + dy * dy + dz * dz;
        arr.push_back({dist, labels[i]});
    }

    // 按距离从小到大排序
    sort(arr.begin(), arr.end(), [](const Node& a, const Node& b) {
        return a.dist < b.dist;
    });

    // 统计前k个最近邻的类别出现次数
    map<int, int> cnt;
    for (int i = 0; i < k; i++) {
        cnt[arr[i].label]++;
    }

    // 找出出现次数最多的类别
    int ans = -1;
    int maxCnt = -1;
    for (auto& p : cnt) {
        if (p.second > maxCnt) {
            maxCnt = p.second;
            ans = p.first;
        }
    }

    return ans;
}

int main() {
    int n, k;
    cin >> n >> k;

    vector<vector<double>> data(n, vector<double>(3));
    vector<int> labels(n);

    // 读入已知语音特征向量和类别
    for (int i = 0; i < n; i++) {
        cin >> data[i][0] >> data[i][1] >> data[i][2] >> labels[i];
    }

    // 读入待分类向量
    vector<double> target(3);
    cin >> target[0] >> target[1] >> target[2];

    // 输出分类结果
    cout << knnClassify(data, labels, target, k) << endl;

    return 0;
}
```
