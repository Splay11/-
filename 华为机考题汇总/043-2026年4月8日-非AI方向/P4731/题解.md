## 解题思路

先按题意把整数 $N$ 转成混合进制编码数组，再把数组映射成字符串，最后判断是否为回文；若不是，则求最长回文子串。

核心分为三步：

1. 处理符号位
   若 $N<0$，编码首位为 $1$，后续对 $|N|$ 做混合进制编码。
   若 $N\ge 0$，编码首位为 $0$，后续对 $N$ 做混合进制编码。

2. 混合进制编码
   按输入的 $bases$ 顺序依次做“整除取余”。
   设当前值为 $x$，对每个进制 $b$：

   * 当前编码位为 $x \bmod b$
   * 更新 $x = x \div b$

   这样得到长度为 $m+1$ 的编码数组，其中第 $1$ 位是符号位，后面 $m$ 位是每一层的余数。

3. 字符串处理
   将编码数组中每个数字映射为字母：

   * $0 \to a$
   * $1 \to b$
   * ...
   * $25 \to z$

   得到字符串 $s$ 后：

   * 若 $s$ 本身是回文串，输出 $s + "(palindrome)"$
   * 否则求 $s$ 的最长回文子串
     这里用“中心扩展算法”即可：枚举每个回文中心，分别向两边扩展，处理奇数长度和偶数长度两种情况。
     若最长回文子串有多个，返回字典序最小的那个。

由于字符串长度只有 $m+1$，而 $m \le 10$，所以直接用中心扩展最简洁也完全足够。

## 复杂度分析

设最终字符串长度为 $L$，则有 $L=m+1$，且 $L \le 11$。

* 混合进制编码的时间复杂度为 $O(m)$
* 回文检测时间复杂度为 $O(L)$
* 中心扩展求最长回文子串的时间复杂度为 $O(L^2)$
* 总时间复杂度为 $O(L^2)$

空间方面：

* 编码数组和字符串都只需线性额外空间
* 空间复杂度为 $O(L)$

该复杂度完全满足要求。

## 代码实现

### Python

```python
# 包裹的编码 - Python ACM风格

import sys


def build_string(n, bases):
    # 处理符号位
    code = [1 if n < 0 else 0]
    x = abs(n)

    # 按bases顺序依次取余并更新商
    for b in bases:
        code.append(x % b)
        x //= b

    # 数字转字母，0->a, 1->b, ...
    return ''.join(chr(ord('a') + v) for v in code)


def is_palindrome(s):
    # 判断整个字符串是否为回文
    return s == s[::-1]


def longest_palindrome_substring(s):
    n = len(s)
    best = s[0]  # 至少单个字符一定是回文

    # 中心扩展
    def expand(left, right):
        while left >= 0 and right < n and s[left] == s[right]:
            left -= 1
            right += 1
        # 返回扩展后的回文子串
        return s[left + 1:right]

    for i in range(n):
        # 奇数长度回文
        p1 = expand(i, i)
        if len(p1) > len(best) or (len(p1) == len(best) and p1 < best):
            best = p1

        # 偶数长度回文
        p2 = expand(i, i + 1)
        if len(p2) > len(best) or (len(p2) == len(best) and p2 < best):
            best = p2

    return best


def solve(n, bases):
    s = build_string(n, bases)

    # 若整体是回文串，按题意添加后缀
    if is_palindrome(s):
        return s + "(palindrome)"

    # 否则输出最长回文子串，若有多个取字典序最小
    return longest_palindrome_substring(s)


def main():
    # 输入：第一行整数N，第二行一组bases
    lines = sys.stdin.read().strip().splitlines()
    n = int(lines[0].strip())
    bases = list(map(int, lines[1].split()))

    print(solve(n, bases))


if __name__ == "__main__":
    main()
```

### Java

```java
// 包裹的编码 - Java ACM风格

import java.util.*;

public class Main {

    // 构造编码后的字符串
    public static String buildString(int n, int[] bases) {
        StringBuilder sb = new StringBuilder();

        // 处理符号位：负数为1，非负数为0
        sb.append((char) ('a' + (n < 0 ? 1 : 0)));

        int x = Math.abs(n);

        // 按bases顺序依次取余并更新商
        for (int b : bases) {
            int digit = x % b;
            sb.append((char) ('a' + digit));
            x /= b;
        }

        return sb.toString();
    }

    // 判断字符串是否为回文
    public static boolean isPalindrome(String s) {
        int left = 0, right = s.length() - 1;
        while (left < right) {
            if (s.charAt(left) != s.charAt(right)) {
                return false;
            }
            left++;
            right--;
        }
        return true;
    }

    // 中心扩展，返回回文子串
    public static String expand(String s, int left, int right) {
        int n = s.length();
        while (left >= 0 && right < n && s.charAt(left) == s.charAt(right)) {
            left--;
            right++;
        }
        return s.substring(left + 1, right);
    }

    // 求最长回文子串，若有多个取字典序最小
    public static String longestPalindromeSubstring(String s) {
        String best = s.substring(0, 1);

        for (int i = 0; i < s.length(); i++) {
            // 奇数长度回文
            String p1 = expand(s, i, i);
            if (p1.length() > best.length() || 
               (p1.length() == best.length() && p1.compareTo(best) < 0)) {
                best = p1;
            }

            // 偶数长度回文
            String p2 = expand(s, i, i + 1);
            if (p2.length() > best.length() || 
               (p2.length() == best.length() && p2.compareTo(best) < 0)) {
                best = p2;
            }
        }

        return best;
    }

    // 题面要求的功能写在外部函数里
    public static String solve(int n, int[] bases) {
        String s = buildString(n, bases);

        // 若整体是回文串，按题意添加后缀
        if (isPalindrome(s)) {
            return s + "(palindrome)";
        }

        // 否则输出最长回文子串
        return longestPalindromeSubstring(s);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 读取第一行整数N
        int n = sc.nextInt();

        // 读取第二行所有bases，题目保证输入合法
        ArrayList<Integer> list = new ArrayList<>();
        while (sc.hasNextInt()) {
            list.add(sc.nextInt());
        }

        int[] bases = new int[list.size()];
        for (int i = 0; i < list.size(); i++) {
            bases[i] = list.get(i);
        }

        System.out.println(solve(n, bases));
    }
}
```

```cpp
// 包裹的编码 - C++ ACM风格

#include <iostream>
#include <vector>
#include <string>
#include <cstdlib>
using namespace std;

// 构造编码后的字符串
string buildString(int n, const vector<int>& bases) {
    string s;

    // 处理符号位：负数为1，非负数为0
    s.push_back(char('a' + (n < 0 ? 1 : 0)));

    int x = abs(n);

    // 按bases顺序依次取余并更新商
    for (int b : bases) {
        int digit = x % b;
        s.push_back(char('a' + digit));
        x /= b;
    }

    return s;
}

// 判断字符串是否为回文
bool isPalindrome(const string& s) {
    int left = 0, right = (int)s.size() - 1;
    while (left < right) {
        if (s[left] != s[right]) {
            return false;
        }
        left++;
        right--;
    }
    return true;
}

// 中心扩展，返回回文子串
string expand(const string& s, int left, int right) {
    int n = (int)s.size();
    while (left >= 0 && right < n && s[left] == s[right]) {
        left--;
        right++;
    }
    return s.substr(left + 1, right - left - 1);
}

// 求最长回文子串，若有多个取字典序最小
string longestPalindromeSubstring(const string& s) {
    string best(1, s[0]);  // 至少单个字符是回文

    for (int i = 0; i < (int)s.size(); i++) {
        // 奇数长度回文
        string p1 = expand(s, i, i);
        if (p1.size() > best.size() || (p1.size() == best.size() && p1 < best)) {
            best = p1;
        }

        // 偶数长度回文
        string p2 = expand(s, i, i + 1);
        if (p2.size() > best.size() || (p2.size() == best.size() && p2 < best)) {
            best = p2;
        }
    }

    return best;
}

// 题面要求的功能写在外部函数里
string solve(int n, const vector<int>& bases) {
    string s = buildString(n, bases);

    // 若整体是回文串，按题意添加后缀
    if (isPalindrome(s)) {
        return s + "(palindrome)";
    }

    // 否则输出最长回文子串
    return longestPalindromeSubstring(s);
}

int main() {
    // 输入：第一行整数N，第二行一组bases
    int n;
    cin >> n;

    vector<int> bases;
    int x;
    while (cin >> x) {
        bases.push_back(x);
    }

    cout << solve(n, bases) << endl;
    return 0;
}
```
