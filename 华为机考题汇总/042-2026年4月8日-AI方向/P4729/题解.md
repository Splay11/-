## 解题思路

本题是一个典型的多变量线性回归问题，要求使用**批量梯度下降（BGD）**训练模型参数，并结合**特征归一化与参数还原**。

模型形式为：

$$
y = w_0 + w_1x_1 + w_2x_2 + w_3x_3
$$

### 1. 特征归一化

由于各特征取值范围差异较大，需要先对每一维做 Min-Max 归一化：

$$
x_j^{norm} = \frac{x_j - \min(x_j)}{\max(x_j) - \min(x_j)}
$$

若某一维没有变化（最大值等于最小值），则该维归一化后全部为 0。



### 2. 批量梯度下降（BGD）

在归一化后的数据上训练模型，初始化所有参数为 0，每轮迭代执行：

1. 计算预测值：

$$
y_{pred} = w_0 + w_1x_1^{norm} + w_2x_2^{norm} + w_3x_3^{norm}
$$

2. 计算误差：

$$
err = y_{pred} - y
$$

3. 按如下公式更新参数（同时更新）：

$$
w_j = w_j - \alpha \cdot \frac{1}{m} \sum (err \cdot x_j)
$$

其中偏置项对应 $x_0 = 1$。



### 3. 参数还原

训练得到的是归一化空间下的参数，需要还原：

特征权重：

$$
w_j = \frac{w_j^{norm}}{\max(x_j) - \min(x_j)}
$$

若分母为 0，则该权重为 0。

偏置项：

$$
w_0 = w_0^{norm} - \sum w_j \cdot \min(x_j)
$$



### 4. 实现流程

1. 读入数据并记录每一维的最小值和最大值
2. 对特征进行归一化
3. 使用 BGD 迭代 $N$ 次更新参数
4. 将参数从归一化空间还原
5. 按要求格式输出结果


## 代码实现

### Python

```python
import sys
import numpy as np


# 使用批量梯度下降训练线性回归模型，并返回原始空间下的参数
def train_bgd(data, n_iter, alpha):
    # 将输入数据转成 numpy 数组，便于向量化计算
    arr = np.array(data, dtype=float)

    # 前三列是特征，最后一列是标签
    X = arr[:, :3]
    y = arr[:, 3]

    # 计算每个特征维度的最小值、最大值和范围
    min_vals = X.min(axis=0)
    max_vals = X.max(axis=0)
    ranges = max_vals - min_vals

    # 按题意做 Min-Max 归一化；若某维范围为 0，则该维归一化后全为 0
    X_norm = np.zeros_like(X, dtype=float)
    for j in range(3):
        if ranges[j] != 0:
            X_norm[:, j] = (X[:, j] - min_vals[j]) / ranges[j]

    # 初始化归一化空间下的参数，w[0] 是偏置，w[1:4] 是三个特征权重
    w = np.zeros(4, dtype=float)

    m = len(y)

    # 固定迭代 N 次
    for _ in range(n_iter):
        # 计算所有样本的预测值
        y_pred = w[0] + X_norm @ w[1:]

        # 计算预测误差
        err = y_pred - y

        # 分别计算偏置和各特征的梯度
        grad0 = err.mean()
        grad_rest = (X_norm.T @ err) / m

        # 按照 BGD 规则同步更新参数
        w[0] -= alpha * grad0
        w[1:] -= alpha * grad_rest

    # 将归一化空间下的参数还原到原始特征空间
    real_w = np.zeros(4, dtype=float)

    # 先还原三个特征权重
    for j in range(3):
        if ranges[j] != 0:
            real_w[j + 1] = w[j + 1] / ranges[j]
        else:
            real_w[j + 1] = 0.0

    # 再还原偏置项
    real_w[0] = w[0] - np.sum(real_w[1:] * min_vals)

    return real_w


def main():
    lines = sys.stdin.read().strip().splitlines()
    if not lines:
        return

    # 读取样本数、迭代次数、学习率
    m = int(lines[0].strip())
    n_iter = int(lines[1].strip())
    alpha = float(lines[2].strip())

    # 读取样本数据
    data = []
    for i in range(3, 3 + m):
        nums = list(map(float, lines[i].split()))
        data.append(nums)

    # 调用训练函数
    ans = train_bgd(data, n_iter, alpha)

    # 按题意保留两位小数输出
    print(f"{ans[0]:.2f} {ans[1]:.2f} {ans[2]:.2f} {ans[3]:.2f}")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.Scanner;

public class Main {

    // 使用批量梯度下降训练模型，并返回原始空间下的四个参数
    public static double[] trainBGD(double[][] data, int m, int nIter, double alpha) {
        // X 存储三个特征，y 存储标签
        double[][] X = new double[m][3];
        double[] y = new double[m];

        for (int i = 0; i < m; i++) {
            X[i][0] = data[i][0];
            X[i][1] = data[i][1];
            X[i][2] = data[i][2];
            y[i] = data[i][3];
        }

        // 统计每一维特征的最小值和最大值
        double[] minVals = new double[3];
        double[] maxVals = new double[3];
        for (int j = 0; j < 3; j++) {
            minVals[j] = X[0][j];
            maxVals[j] = X[0][j];
        }

        for (int i = 1; i < m; i++) {
            for (int j = 0; j < 3; j++) {
                if (X[i][j] < minVals[j]) {
                    minVals[j] = X[i][j];
                }
                if (X[i][j] > maxVals[j]) {
                    maxVals[j] = X[i][j];
                }
            }
        }

        // 计算每一维的范围
        double[] ranges = new double[3];
        for (int j = 0; j < 3; j++) {
            ranges[j] = maxVals[j] - minVals[j];
        }

        // 对特征做归一化
        double[][] XNorm = new double[m][3];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < 3; j++) {
                if (ranges[j] != 0) {
                    XNorm[i][j] = (X[i][j] - minVals[j]) / ranges[j];
                } else {
                    XNorm[i][j] = 0.0;
                }
            }
        }

        // 归一化空间下的参数，w[0] 为偏置
        double[] w = new double[4];

        // 固定迭代 N 次
        for (int iter = 0; iter < nIter; iter++) {
            // 梯度数组，grad[0] 对应偏置，grad[1..3] 对应三个权重
            double[] grad = new double[4];

            // 遍历全部样本，累加梯度
            for (int i = 0; i < m; i++) {
                double yPred = w[0] + w[1] * XNorm[i][0] + w[2] * XNorm[i][1] + w[3] * XNorm[i][2];
                double err = yPred - y[i];

                grad[0] += err;
                grad[1] += err * XNorm[i][0];
                grad[2] += err * XNorm[i][1];
                grad[3] += err * XNorm[i][2];
            }

            // 按题意用平均梯度更新参数
            for (int j = 0; j < 4; j++) {
                grad[j] /= m;
                w[j] -= alpha * grad[j];
            }
        }

        // 还原到原始特征空间
        double[] realW = new double[4];

        for (int j = 0; j < 3; j++) {
            if (ranges[j] != 0) {
                realW[j + 1] = w[j + 1] / ranges[j];
            } else {
                realW[j + 1] = 0.0;
            }
        }

        realW[0] = w[0];
        for (int j = 0; j < 3; j++) {
            realW[0] -= realW[j + 1] * minVals[j];
        }

        return realW;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 读取样本数、迭代次数、学习率
        int m = sc.nextInt();
        int nIter = sc.nextInt();
        double alpha = sc.nextDouble();

        // 读取所有样本
        double[][] data = new double[m][4];
        for (int i = 0; i < m; i++) {
            data[i][0] = sc.nextDouble();
            data[i][1] = sc.nextDouble();
            data[i][2] = sc.nextDouble();
            data[i][3] = sc.nextDouble();
        }

        // 调用训练函数
        double[] ans = trainBGD(data, m, nIter, alpha);

        // 按题意保留两位小数输出
        System.out.printf("%.2f %.2f %.2f %.2f%n", ans[0], ans[1], ans[2], ans[3]);

        sc.close();
    }
}
```

```cpp
        }
```
