## 解题思路

本题核心分为两部分：**K-Means 聚类** + **路径距离计算**。

首先使用 **K-Means 算法**对所有包裹进行聚类：

1. 初始化：

   * 按点到原点距离升序排序（相同则按输入顺序）
   * 取前 $M=\min(K,N)$ 个点作为初始聚类中心

2. 迭代优化：

   * 每个点分配到最近的中心
   * 对每个簇重新计算中心（取均值）
   * 若某簇无点，则保持原中心不变

3. 收敛条件：

   * 所有中心移动距离之和 < $1e-4$，或达到最大迭代次数（50次）

得到最终聚类中心后：

* 按中心到原点 $(0,0)$ 的距离升序排序
* 按顺序依次配送，并最终返回原点

总路程为：

$$
(0,0)\rightarrow C_1 \rightarrow C_2 \rightarrow \cdots \rightarrow C_M \rightarrow (0,0)
$$

最后根据速度换算时间：

$$
time = \left\lfloor \frac{total_dist \times 3600}{speed} \right\rfloor
$$



## 复杂度分析

* 时间复杂度：
  K-Means 每轮为 $O(NM)$，最多迭代 50 次，总复杂度为
  $O(50 \cdot N \cdot M)$，其中 $M=\min(K,N)$

* 空间复杂度：
  主要存储点、中心和标签，复杂度为
  $O(N + M)$


## 代码实现

### Python

```python
import sys
import math
import numpy as np


# 计算两个二维点之间的欧式距离
def distance(a, b):
    diff = a - b
    return math.sqrt(diff[0] * diff[0] + diff[1] * diff[1])


# 按题意执行 K-Means，返回最终聚类中心
def kmeans(points, k, max_iters=50, tol=1e-4):
    n = len(points)

    # 实际聚类个数不能超过点的数量
    m = min(k, n)

    # 按到原点距离升序、输入顺序升序进行排序，选前 m 个点作为初始中心
    order = sorted(range(n), key=lambda i: (points[i][0] ** 2 + points[i][1] ** 2, i))
    centers = points[order[:m]].copy()

    # labels[i] 表示第 i 个点当前属于哪个簇
    labels = np.zeros(n, dtype=int)

    for _ in range(max_iters):
        # 记录旧中心，用于判断收敛
        old_centers = centers.copy()

        # 第一步：把每个点分配到最近的聚类中心
        for i in range(n):
            best_idx = 0
            best_dist = float('inf')
            for j in range(m):
                d = distance(points[i], centers[j])
                if d < best_dist:
                    best_dist = d
                    best_idx = j
            labels[i] = best_idx

        # 第二步：根据当前分组重新计算每个聚类中心
        new_centers = centers.copy()
        for j in range(m):
            cluster_points = points[labels == j]

            # 如果当前簇非空，则更新为该簇所有点的均值
            if len(cluster_points) > 0:
                new_centers[j] = cluster_points.mean(axis=0)
            # 如果当前簇为空，则保持原中心不变

        centers = new_centers

        # 第三步：计算所有中心移动距离之和，判断是否收敛
        move_sum = 0.0
        for j in range(m):
            move_sum += distance(old_centers[j], centers[j])

        if move_sum < tol:
            break

    return centers


# 根据最终聚类中心计算配送总时间（秒，向下取整）
def solve(k, n, speed, points):
    centers = kmeans(points, k)

    # 按聚类中心到原点距离升序排序
    centers = sorted(centers, key=lambda c: c[0] * c[0] + c[1] * c[1])

    origin = np.array([0.0, 0.0])

    # 计算路径总长度：原点 -> 各中心依次配送 -> 原点
    total_dist = 0.0
    prev = origin

    for center in centers:
        total_dist += distance(prev, center)
        prev = center

    total_dist += distance(prev, origin)

    # 距离 / 速度 = 小时，再乘 3600 转成秒，最后向下取整
    total_time = math.floor(total_dist * 3600.0 / speed)
    return total_time


def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return

    k = int(data[0])
    n = int(data[1])
    speed = int(data[2])

    vals = list(map(float, data[3:]))
    points = []
    for i in range(0, 2 * n, 2):
        points.append([vals[i], vals[i + 1]])

    points = np.array(points, dtype=float)

    ans = solve(k, n, speed, points)
    print(ans)


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

public class Main {

    // 最大迭代次数
    static final int MAX_ITERS = 50;

    // 收敛阈值
    static final double TOL = 1e-4;

    // 表示二维点
    static class Point {
        double x;
        double y;

        Point(double x, double y) {
            this.x = x;
            this.y = y;
        }
    }

    // 计算两个点之间的欧式距离
    static double distance(Point a, Point b) {
        double dx = a.x - b.x;
        double dy = a.y - b.y;
        return Math.sqrt(dx * dx + dy * dy);
    }

    // 执行 K-Means，返回最终聚类中心
    static List<Point> kmeans(List<Point> points, int k) {
        int n = points.size();

        // 实际聚类个数不能超过点数
        int m = Math.min(k, n);

        // 保存点的下标，用于按题意排序后选择初始中心
        List<Integer> order = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            order.add(i);
        }

        // 按到原点距离升序排序；距离相同按输入顺序升序
        Collections.sort(order, new Comparator<Integer>() {
            @Override
            public int compare(Integer i, Integer j) {
                double di = points.get(i).x * points.get(i).x + points.get(i).y * points.get(i).y;
                double dj = points.get(j).x * points.get(j).x + points.get(j).y * points.get(j).y;
                if (Math.abs(di - dj) > 1e-12) {
                    return di < dj ? -1 : 1;
                }
                return i - j;
            }
        });

        // 取前 m 个点作为初始中心
        List<Point> centers = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            Point p = points.get(order.get(i));
            centers.add(new Point(p.x, p.y));
        }

        // labels[i] 表示第 i 个点归属的簇编号
        int[] labels = new int[n];

        for (int iter = 0; iter < MAX_ITERS; iter++) {
            // 保存旧中心，用于判断是否收敛
            List<Point> oldCenters = new ArrayList<>();
            for (Point c : centers) {
                oldCenters.add(new Point(c.x, c.y));
            }

            // 第一步：把每个点分配到最近的中心
            for (int i = 0; i < n; i++) {
                int bestIdx = 0;
                double bestDist = Double.MAX_VALUE;

                for (int j = 0; j < m; j++) {
                    double d = distance(points.get(i), centers.get(j));
                    if (d < bestDist) {
                        bestDist = d;
                        bestIdx = j;
                    }
                }
                labels[i] = bestIdx;
            }

            // 第二步：重新计算每个簇的中心
            double[] sumX = new double[m];
            double[] sumY = new double[m];
            int[] cnt = new int[m];

            for (int i = 0; i < n; i++) {
                int idx = labels[i];
                sumX[idx] += points.get(i).x;
                sumY[idx] += points.get(i).y;
                cnt[idx]++;
            }

            for (int j = 0; j < m; j++) {
                // 只有当前簇非空时才更新中心
                if (cnt[j] > 0) {
                    centers.get(j).x = sumX[j] / cnt[j];
                    centers.get(j).y = sumY[j] / cnt[j];
                }
                // 如果当前簇为空，则保留原中心不变
            }

            // 第三步：计算所有中心移动距离之和，判断是否收敛
            double moveSum = 0.0;
            for (int j = 0; j < m; j++) {
                moveSum += distance(oldCenters.get(j), centers.get(j));
            }

            if (moveSum < TOL) {
                break;
            }
        }

        return centers;
    }

    // 计算配送总时间
    static long solve(int k, int n, int speed, List<Point> points) {
        List<Point> centers = kmeans(points, k);

        // 按聚类中心到原点距离升序排序
        Collections.sort(centers, new Comparator<Point>() {
            @Override
            public int compare(Point a, Point b) {
                double da = a.x * a.x + a.y * a.y;
                double db = b.x * b.x + b.y * b.y;
                if (Math.abs(da - db) > 1e-12) {
                    return da < db ? -1 : 1;
                }
                return 0;
            }
        });

        Point origin = new Point(0.0, 0.0);
        double totalDist = 0.0;
        Point prev = origin;

        // 依次走过所有中心
        for (Point c : centers) {
            totalDist += distance(prev, c);
            prev = c;
        }

        // 最后回到原点
        totalDist += distance(prev, origin);

        // 换算为秒并向下取整
        return (long) Math.floor(totalDist * 3600.0 / speed);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int k = sc.nextInt();
        int n = sc.nextInt();
        int speed = sc.nextInt();

        List<Point> points = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            double x = sc.nextDouble();
            double y = sc.nextDouble();
            points.add(new Point(x, y));
        }

        long ans = solve(k, n, speed, points);
        System.out.println(ans);
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

// 最大迭代次数
const int MAX_ITERS = 50;

// 收敛阈值
const double TOL = 1e-4;

// 表示二维点
struct Point {
    double x, y;
};

// 计算两个点之间的欧式距离
double distancePoint(const Point& a, const Point& b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    return sqrt(dx * dx + dy * dy);
}

// 执行 K-Means，返回最终聚类中心
vector<Point> kmeans(const vector<Point>& points, int k) {
    int n = (int)points.size();

    // 实际聚类个数不能超过点数
    int m = min(k, n);

    // 按下标排序，便于按题意选择初始中心
    vector<int> order(n);
    for (int i = 0; i < n; i++) {
        order[i] = i;
    }

    // 按到原点距离升序排序；若相同则按输入顺序升序
    sort(order.begin(), order.end(), [&](int i, int j) {
        double di = points[i].x * points[i].x + points[i].y * points[i].y;
        double dj = points[j].x * points[j].x + points[j].y * points[j].y;
        if (fabs(di - dj) > 1e-12) {
            return di < dj;
        }
        return i < j;
    });

    // 取前 m 个点作为初始中心
    vector<Point> centers;
    for (int i = 0; i < m; i++) {
        centers.push_back(points[order[i]]);
    }

    // labels[i] 表示第 i 个点当前属于哪个簇
    vector<int> labels(n, 0);

    for (int iter = 0; iter < MAX_ITERS; iter++) {
        // 保存旧中心，用于判断是否收敛
        vector<Point> oldCenters = centers;

        // 第一步：把每个点分配到最近的中心
        for (int i = 0; i < n; i++) {
            int bestIdx = 0;
            double bestDist = 1e100;

            for (int j = 0; j < m; j++) {
                double d = distancePoint(points[i], centers[j]);
                if (d < bestDist) {
                    bestDist = d;
                    bestIdx = j;
                }
            }
            labels[i] = bestIdx;
        }

        // 第二步：重新计算每个簇的中心
        vector<double> sumX(m, 0.0), sumY(m, 0.0);
        vector<int> cnt(m, 0);

        for (int i = 0; i < n; i++) {
            int idx = labels[i];
            sumX[idx] += points[i].x;
            sumY[idx] += points[i].y;
            cnt[idx]++;
        }

        for (int j = 0; j < m; j++) {
            // 只有当前簇非空时才更新中心
            if (cnt[j] > 0) {
                centers[j].x = sumX[j] / cnt[j];
                centers[j].y = sumY[j] / cnt[j];
            }
            // 如果当前簇为空，则保持原中心不变
        }

        // 第三步：计算所有中心移动距离之和，判断是否收敛
        double moveSum = 0.0;
        for (int j = 0; j < m; j++) {
            moveSum += distancePoint(oldCenters[j], centers[j]);
        }

        if (moveSum < TOL) {
            break;
        }
    }

    return centers;
}

// 计算配送总时间
long long solve(int k, int n, int speed, const vector<Point>& points) {
    vector<Point> centers = kmeans(points, k);

    // 按聚类中心到原点距离升序排序
    sort(centers.begin(), centers.end(), [](const Point& a, const Point& b) {
        double da = a.x * a.x + a.y * a.y;
        double db = b.x * b.x + b.y * b.y;
        if (fabs(da - db) > 1e-12) {
            return da < db;
        }
        return false;
    });

    Point origin = {0.0, 0.0};
    double totalDist = 0.0;
    Point prev = origin;

    // 依次走过所有中心
    for (int i = 0; i < (int)centers.size(); i++) {
        totalDist += distancePoint(prev, centers[i]);
        prev = centers[i];
    }

    // 最后回到原点
    totalDist += distancePoint(prev, origin);

    // 换算为秒并向下取整
    return (long long)floor(totalDist * 3600.0 / speed);
}

int main() {
    int k, n, speed;
    cin >> k >> n >> speed;

    vector<Point> points(n);
    for (int i = 0; i < n; i++) {
        cin >> points[i].x >> points[i].y;
    }

    cout << solve(k, n, speed, points) << '\n';
    return 0;
}
```
