## 解题思路

本题需要根据目标用户向量 $T$、权重向量 $W$ 和每个商品的特征向量，计算商品与目标用户之间的加权欧氏距离平方：

$$
Score = \sum_{j=1}^{D} W_j \times (t_j - i_j)^2
$$

$Score$ 越小，表示商品与目标用户越匹配。

核心做法如下：

1. 依次读取每个商品的 $ID$ 和 $D$ 维特征向量。
2. 对每个商品计算加权距离平方：

   * 对每一维 $j$，计算目标值 $t_j$ 和商品特征值 $i_j$ 的差值。
   * 将差值平方后乘以该维度权重 $w_j$。
   * 将所有维度的结果累加，得到该商品的 $Score$。
3. 将每个商品表示为二元组 $(Score, ID)$。
4. 按照以下规则排序：

   * 先按 $Score$ 从小到大排序；
   * 若 $Score$ 相同，再按商品 $ID$ 从小到大排序。
5. 输出排序后前 $K$ 个商品的 $ID$。

由于题目要求返回距离最近的 $K$ 个商品，直接对全部 $N$ 个商品排序即可，逻辑简单且稳定。

## 复杂度分析

设商品数量为 $N$，特征维度为 $D$。

计算每个商品的 $Score$ 需要遍历 $D$ 个维度，共有 $N$ 个商品，因此计算距离的时间复杂度为：

$$
O(ND)
$$

之后需要对 $N$ 个商品按照 $(Score, ID)$ 排序，时间复杂度为：

$$
O(N \log N)
$$

所以总时间复杂度为：

$$
O(ND + N \log N)
$$

需要存储每个商品的 $Score$ 和 $ID$，空间复杂度为：

$$
O(N)
$$

该复杂度能够完整处理所有商品，并保证排序规则正确。

## 代码实现

### Python

```python
def top_k(items, w, t, k):
    res = []

    # 逐个计算商品的加权欧氏距离平方
    for item in items:
        item_id = item[0]
        vec = item[1]

        score = 0
        for i in range(len(w)):
            # 计算当前维度差值
            diff = t[i] - vec[i]
            # 累加当前维度的加权平方差
            score += w[i] * diff * diff

        # 保存排序所需的分数和商品ID
        res.append((score, item_id))

    # 先按分数升序排序，分数相同按商品ID升序排序
    res.sort()

    ans = []
    for i in range(k):
        ans.append(res[i][1])

    return ans


def main():
    n, d, k = map(int, input().split())
    w = list(map(int, input().split()))
    t = list(map(int, input().split()))

    items = []
    for _ in range(n):
        arr = list(map(int, input().split()))
        item_id = arr[0]
        vec = arr[1:]
        items.append((item_id, vec))

    ans = top_k(items, w, t, k)
    print(" ".join(map(str, ans)))


if __name__ == "__main__":
    main()
```

### Java

```java
import java.util.*;

public class Main {
    static class Node implements Comparable<Node> {
        long score;
        int id;

        Node(long score, int id) {
            this.score = score;
            this.id = id;
        }

        public int compareTo(Node other) {
            // 分数小的商品排在前面
            if (this.score != other.score) {
                return Long.compare(this.score, other.score);
            }
            // 分数相同，商品ID小的排在前面
            return Integer.compare(this.id, other.id);
        }
    }

    public static List<Integer> topK(int[][] items, int[] w, int[] t, int k) {
        List<Node> list = new ArrayList<>();

        // 逐个计算商品的加权欧氏距离平方
        for (int i = 0; i < items.length; i++) {
            int id = items[i][0];
            long score = 0;

            for (int j = 0; j < w.length; j++) {
                // 计算当前维度差值
                long diff = (long)t[j] - items[i][j + 1];
                // 累加当前维度的加权平方差
                score += (long)w[j] * diff * diff;
            }

            // 保存排序所需的分数和商品ID
            list.add(new Node(score, id));
        }

        // 先按分数升序排序，分数相同按商品ID升序排序
        Collections.sort(list);

        List<Integer> ans = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            ans.add(list.get(i).id);
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int d = sc.nextInt();
        int k = sc.nextInt();

        int[] w = new int[d];
        for (int i = 0; i < d; i++) {
            w[i] = sc.nextInt();
        }

        int[] t = new int[d];
        for (int i = 0; i < d; i++) {
            t[i] = sc.nextInt();
        }

        int[][] items = new int[n][d + 1];
        for (int i = 0; i < n; i++) {
            items[i][0] = sc.nextInt();
            for (int j = 1; j <= d; j++) {
                items[i][j] = sc.nextInt();
            }
        }

        List<Integer> ans = topK(items, w, t, k);

        for (int i = 0; i < ans.size(); i++) {
            if (i > 0) {
                System.out.print(" ");
            }
            System.out.print(ans.get(i));
        }
        System.out.println();

        sc.close();
    }
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<int> topK(vector<vector<int>>& items, vector<int>& w, vector<int>& t, int k) {
    vector<pair<long long, int>> res;

    // 逐个计算商品的加权欧氏距离平方
    for (int i = 0; i < (int)items.size(); i++) {
        int id = items[i][0];
        long long score = 0;

        for (int j = 0; j < (int)w.size(); j++) {
            // 计算当前维度差值
            long long diff = (long long)t[j] - items[i][j + 1];
            // 累加当前维度的加权平方差
            score += (long long)w[j] * diff * diff;
        }

        // 保存排序所需的分数和商品ID
        res.push_back({score, id});
    }

    // pair默认先按first升序，再按second升序，正好符合题目排序规则
    sort(res.begin(), res.end());

    vector<int> ans;
    for (int i = 0; i < k; i++) {
        ans.push_back(res[i].second);
    }

    return ans;
}

int main() {
    int n, d, k;
    cin >> n >> d >> k;

    vector<int> w(d);
    for (int i = 0; i < d; i++) {
        cin >> w[i];
    }

    vector<int> t(d);
    for (int i = 0; i < d; i++) {
        cin >> t[i];
    }

    vector<vector<int>> items(n, vector<int>(d + 1));
    for (int i = 0; i < n; i++) {
        cin >> items[i][0];
        for (int j = 1; j <= d; j++) {
            cin >> items[i][j];
        }
    }

    vector<int> ans = topK(items, w, t, k);

    for (int i = 0; i < (int)ans.size(); i++) {
        if (i > 0) {
            cout << " ";
        }
        cout << ans[i];
    }
    cout << endl;

    return 0;
}
```
