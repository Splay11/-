## 解题思路

先对每个 $token$ 根据 $logits$ 计算稳定版 $softmax$。

令：

$$
m_i=\max_j z_{i,j}
$$

则：

$$
p_{i,j}=\frac{e^{z_{i,j}-m_i}}{\sum_t e^{z_{i,t}-m_i}}
$$

交叉熵为：

$$
\ell_i=-\log p_{i,y_i}
$$

熵为：

$$
H_i=-\sum_{j=0}^{V-1}p_{i,j}\log p_{i,j}
$$

题目要求在总影响值不超过 $B$ 的条件下，选择若干个 $token$，使它们的交叉熵之和最大。每个 $token$ 只能选一次，因此使用 $0/1$ 背包。

设 $dp[w]$ 表示影响值不超过 $w$ 时能得到的最大交叉熵之和，$eh[w]$ 表示该最优方案对应的熵之和。

转移时倒序枚举容量：

$$
dp[w]=\max(dp[w],dp[w-c_i]+\ell_i)
$$

若更新了 $dp[w]$，同步更新对应熵和：

$$
eh[w]=eh[w-c_i]+H_i
$$

最终答案为：

$$
loss=\sum_i \ell_i+\lambda \cdot eh[best]
$$

其中 $best$ 为背包得到最大交叉熵的位置。

## 复杂度分析

计算所有 $token$ 的交叉熵和熵需要遍历 $logits$，时间复杂度为：

$$
O(NV)
$$

背包部分时间复杂度为：

$$
O(NB)
$$

总时间复杂度为：

$$
O(NV+NB)
$$

背包数组大小为 $O(B)$，保存输入和中间结果需要 $O(NV)$，因此空间复杂度为：

$$
O(NV+B)
$$

## 代码实现

### Python

```python
import sys
import math


def calc_values(logits, y):
    n = len(logits)
    v = len(logits[0]) if n else 0

    ces = [0.0] * n
    ents = [0.0] * n

    for i in range(n):
        row = logits[i]
        # 每行减去最大 logit，避免 exp 溢出
        m = max(row)
        a = [row[j] - m for j in range(v)]

        # 计算 softmax
        exps = [math.exp(a[j]) for j in range(v)]
        s = sum(exps)
        p = [exps[j] / s for j in range(v)]

        # 计算交叉熵：-log(p_y)
        log_sum = math.log(s)
        ces[i] = log_sum - a[y[i]]

        # 计算熵：-sum(p * log(p)) = log_sum - sum(p * a)
        # 因为 log(p_j) = a_j - log_sum
        ents[i] = log_sum - sum(p[j] * a[j] for j in range(v))

    return ces, ents


def calc_loss(ces, ents, costs, b, lam):
    n = len(ces)
    base = float(sum(ces))

    neg = -1e100
    eps = 1e-12

    dp = [neg] * (b + 1)
    eh = [0.0] * (b + 1)

    dp[0] = 0.0

    for i in range(n):
        c = int(costs[i])

        # 0/1 背包倒序枚举，保证每个 token 只选一次
        for w in range(b, c - 1, -1):
            if dp[w - c] < neg / 2:
                continue

            nv = dp[w - c] + float(ces[i])

            # 若交叉熵更大，则更新价值和对应熵和
            if nv > dp[w] + eps:
                dp[w] = nv
                eh[w] = eh[w - c] + float(ents[i])

    best = 0
    for w in range(1, b + 1):
        if dp[w] > dp[best] + eps:
            best = w

    return base + lam * eh[best]


def main():
    data = sys.stdin.buffer.read().decode()
    if not data.strip():
        return

    arr = list(map(float, data.split()))

    n = int(arr[0])
    v = int(arr[1])
    b = int(arr[2])
    lam = float(arr[3])

    # 每行包含 V 个 logits、1 个标签 y、1 个影响值 c
    logits = []
    y = []
    costs = []
    idx = 4
    for _ in range(n):
        row = arr[idx:idx + v]
        logits.append(row)
        y.append(int(arr[idx + v]))
        costs.append(int(arr[idx + v + 1]))
        idx += v + 2

    ces, ents = calc_values(logits, y)
    ans = calc_loss(ces, ents, costs, b, lam)

    print(f"{ans:.2f}")


if __name__ == "__main__":
    main()
```

### Java

```java
import java.io.*;
import java.util.*;

public class Main {
    static double[] calcToken(double[] z, int y) {
        int v = z.length;

        // 求最大 logit，用于稳定计算 softmax
        double m = z[0];
        for (int j = 1; j < v; j++) {
            m = Math.max(m, z[j]);
        }

        double[] exps = new double[v];
        double sum = 0.0;

        // 计算 exp(z_j - m)
        for (int j = 0; j < v; j++) {
            exps[j] = Math.exp(z[j] - m);
            sum += exps[j];
        }

        double logSum = Math.log(sum);

        // 交叉熵：-log(p_y)
        double ce = logSum - (z[y] - m);

        // 熵：-sum(p * log(p))
        double ent = logSum;
        for (int j = 0; j < v; j++) {
            double p = exps[j] / sum;
            ent -= p * (z[j] - m);
        }

        return new double[]{ce, ent};
    }

    static double calcLoss(double[] ces, double[] ents, int[] costs, int b, double lam) {
        int n = ces.length;

        double base = 0.0;
        for (int i = 0; i < n; i++) {
            base += ces[i];
        }

        double neg = -1e100;
        double eps = 1e-12;

        double[] dp = new double[b + 1];
        double[] eh = new double[b + 1];

        Arrays.fill(dp, neg);
        dp[0] = 0.0;

        for (int i = 0; i < n; i++) {
            int c = costs[i];

            // 0/1 背包倒序枚举，保证每个 token 只选一次
            for (int w = b; w >= c; w--) {
                if (dp[w - c] < neg / 2) {
                    continue;
                }

                double nv = dp[w - c] + ces[i];

                // 若交叉熵更大，则更新价值和对应熵和
                if (nv > dp[w] + eps) {
                    dp[w] = nv;
                    eh[w] = eh[w - c] + ents[i];
                }
            }
        }

        int best = 0;
        for (int w = 1; w <= b; w++) {
            if (dp[w] > dp[best] + eps) {
                best = w;
            }
        }

        return base + lam * eh[best];
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        String first = br.readLine();
        if (first == null || first.length() == 0) {
            return;
        }

        StringTokenizer st = new StringTokenizer(first);

        int n = Integer.parseInt(st.nextToken());
        int v = Integer.parseInt(st.nextToken());
        int b = Integer.parseInt(st.nextToken());
        double lam = Double.parseDouble(st.nextToken());

        double[] ces = new double[n];
        double[] ents = new double[n];
        int[] costs = new int[n];

        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());

            double[] z = new double[v];
            for (int j = 0; j < v; j++) {
                z[j] = Double.parseDouble(st.nextToken());
            }

            int y = Integer.parseInt(st.nextToken());
            int c = Integer.parseInt(st.nextToken());

            double[] res = calcToken(z, y);
            ces[i] = res[0];
            ents[i] = res[1];
            costs[i] = c;
        }

        double ans = calcLoss(ces, ents, costs, b, lam);
        System.out.printf("%.2f%n", ans);
    }
}
```

```cpp
using namespace std;

pair<double, double> calcToken(vector<double>& z, int y) {
    int v = z.size();

    // 求最大 logit，用于稳定计算 softmax
    double m = z[0];
    for (int j = 1; j < v; j++) {
        m = max(m, z[j]);
    }

    vector<double> exps(v);
    double sum = 0.0;

    // 计算 exp(z_j - m)
    for (int j = 0; j < v; j++) {
        exps[j] = exp(z[j] - m);
        sum += exps[j];
    }

    double logSum = log(sum);

    // 交叉熵：-log(p_y)
    double ce = logSum - (z[y] - m);

    // 熵：-sum(p * log(p))
    double ent = logSum;
    for (int j = 0; j < v; j++) {
        double p = exps[j] / sum;
        ent -= p * (z[j] - m);
    }

    return {ce, ent};
}

double calcLoss(vector<double>& ces, vector<double>& ents, vector<int>& costs, int b, double lam) {
    int n = ces.size();

    double base = 0.0;
    for (int i = 0; i < n; i++) {
        base += ces[i];
    }

    double neg = -1e100;
    double eps = 1e-12;

    vector<double> dp(b + 1, neg);
    vector<double> eh(b + 1, 0.0);

    dp[0] = 0.0;

    for (int i = 0; i < n; i++) {
        int c = costs[i];

        // 0/1 背包倒序枚举，保证每个 token 只选一次
        for (int w = b; w >= c; w--) {
            if (dp[w - c] < neg / 2) {
                continue;
            }

            double nv = dp[w - c] + ces[i];

            // 若交叉熵更大，则更新价值和对应熵和
            if (nv > dp[w] + eps) {
                dp[w] = nv;
                eh[w] = eh[w - c] + ents[i];
            }
        }
    }

    int best = 0;
    for (int w = 1; w <= b; w++) {
        if (dp[w] > dp[best] + eps) {
            best = w;
        }
    }

    return base + lam * eh[best];
}

int main() {
    ios::sync_with_stdio(false);

        ces[i] = res.first;
        ents[i] = res.second;
        costs[i] = c;

    double ans = calcLoss(ces, ents, costs, b, lam);
```
