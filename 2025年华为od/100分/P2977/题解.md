## 解题思路

本题是**离散事件模拟 + 队列 + 多执行者调度**，核心在于同一时刻的事件顺序。

1. **状态**：FIFO 任务队列（存执行时长）、每个执行者的下次空闲时刻 `freeAt`、丢弃计数与最后完成时刻。
2. **时间推进**：下一事件时刻为「下一提交时刻」与「最早执行者释放时刻」的较小值。
3. **同一时刻 `t` 的处理顺序**（对应题面特别规则）：
   - 将 `freeAt <= t` 的执行者置为空闲；
   - **第一次分配**：按编号从小到大，让空闲执行者从队首取任务并更新 `freeAt`；
   - **处理提交**：若队列已满则弹出队首并 `discarded++`，再将新任务入队；
   - **第二次分配**：再次尝试分配（提交后队列中可能有新任务）。
4. 当无未处理提交、队列为空且所有执行者空闲时结束。

**常见假解**：

- 同刻先提交再分配 → 队列满时同刻「释放+提交」会多丢弃（样例 2/3 错）；
- 同刻多个空闲执行者不按编号优先 → 完成时刻错误；
- 只模拟提交时刻、忽略执行者释放时刻 → 漏掉中间调度；
- 丢弃时未按「队首最老任务」弹出 → 计数错误。

## 复杂度分析

设 $n$ 为任务数（$n \le 20$），$m$ 为执行者数（$\le 1000$）。

- 时间：每个任务最多入队、出队、被分配各常数次，事件次数 $O(n + \text{完成次数})$，在题面约束下可视为 $O(n \cdot m)$ 上界且实际极小；
- 空间：$O(n + m)$，队列与执行者状态。

## 代码实现

### Python

```python
from collections import deque
from typing import List


class Solution:
    def simulateTaskQueue(
        self,
        submitTimes: List[int],
        execTimes: List[int],
        queueCapacity: int,
        numWorkers: int,
    ) -> List[int]:
        queue = deque()  # FIFO 任务队列，存执行时长
        discarded = 0
        last_finish = 0
        m = numWorkers
        # free_at[i]==0 表示空闲；否则为该执行者最早可接新任务的时刻
        free_at = [0] * (m + 1)

        # 按提交时刻升序处理（题面保证可排序）
        subs = sorted(zip(submitTimes, execTimes))
        si = 0
        n = len(subs)

        def assign(time: int) -> None:
            nonlocal last_finish
            # 编号小的空闲执行者优先取队首（最老）任务
            idle = sorted(i for i in range(1, m + 1) if free_at[i] <= time)
            for wid in idle:
                if not queue:
                    break
                dur = queue.popleft()
                finish = time + dur
                last_finish = max(last_finish, finish)
                free_at[wid] = finish

        while True:
            busy = [free_at[i] for i in range(1, m + 1) if free_at[i] > 0]
            # 无待提交、队列空、无人在忙则结束
            if si >= n and not queue and not busy:
                break

            # 下一事件：提交或最早释放，取较早者
            next_submit = subs[si][0] if si < n else 10**18
            next_free = min(busy) if busy else 10**18
            t = min(next_submit, next_free)

            # 到达时刻 t 时，已完成任务的执行者变为空闲
            for i in range(1, m + 1):
                if free_at[i] > 0 and free_at[i] <= t:
                    free_at[i] = 0

            # 先分配：保证「同刻释放与提交」时先腾出队首给执行者
            assign(t)

            # 再处理所有在 t 提交的任务；满队则丢弃队首
            while si < n and subs[si][0] == t:
                d = subs[si][1]
                if len(queue) >= queueCapacity:
                    queue.popleft()
                    discarded += 1
                queue.append(d)
                si += 1

            # 提交后再分配一次，吃掉刚入队的任务
            assign(t)

        return [last_finish, discarded]
```

### Java

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;

public class Solution {
    public int[] simulateTaskQueue(int[] submitTimes, int[] execTimes, int queueCapacity,
                                   int numWorkers) {
        ArrayDeque<Integer> queue = new ArrayDeque<>(); // FIFO，存执行时长
        int discarded = 0;
        int lastFinish = 0;
        int m = numWorkers;
        // freeAt[i]==0 空闲，否则为最早可接新任务时刻
        int[] freeAt = new int[m + 1];

        int n = submitTimes.length;
        int[][] subs = new int[n][2];
        for (int i = 0; i < n; i++) {
            subs[i][0] = submitTimes[i];
            subs[i][1] = execTimes[i];
        }
        Arrays.sort(subs, (a, b) -> Integer.compare(a[0], b[0])); // 按提交时刻升序
        int si = 0;
        final int INF = Integer.MAX_VALUE / 4;

        while (true) {
            ArrayList<Integer> busy = new ArrayList<>();
            for (int i = 1; i <= m; i++) {
                if (freeAt[i] > 0) busy.add(freeAt[i]);
            }
            // 无待提交、队列空、无人在忙则结束
            if (si >= n && queue.isEmpty() && busy.isEmpty()) break;

            // 下一事件取「下一提交」与「最早释放」的较早时刻
            int nextSubmit = si < n ? subs[si][0] : INF;
            int nextFree = INF;
            for (int x : busy) nextFree = Math.min(nextFree, x);
            int time = Math.min(nextSubmit, nextFree);

            // 完成时刻已到的执行者置为空闲
            for (int i = 1; i <= m; i++) {
                if (freeAt[i] > 0 && freeAt[i] <= time) freeAt[i] = 0;
            }

            // 先分配再提交，对应题面同刻特别规则
            lastFinish = assignWorkers(time, m, freeAt, queue, lastFinish);

            while (si < n && subs[si][0] == time) {
                int d = subs[si][1];
                if (queue.size() >= queueCapacity) {
                    queue.pollFirst(); // 丢弃最老任务
                    discarded++;
                }
                queue.addLast(d);
                si++;
            }

            // 提交后再分配，避免新入队任务空等
            lastFinish = assignWorkers(time, m, freeAt, queue, lastFinish);
        }

        return new int[] {lastFinish, discarded};
    }

    // 编号小的空闲执行者优先从队首取任务
    private int assignWorkers(int time, int m, int[] freeAt, ArrayDeque<Integer> queue,
                              int lastFinish) {
        ArrayList<Integer> idle = new ArrayList<>();
        for (int i = 1; i <= m; i++) {
            if (freeAt[i] <= time) idle.add(i);
        }
        Collections.sort(idle);
        for (int wid : idle) {
            if (queue.isEmpty()) break;
            int dur = queue.pollFirst();
            int finish = time + dur;
            lastFinish = Math.max(lastFinish, finish);
            freeAt[wid] = finish;
        }
        return lastFinish;
    }
}
```

### C++

```cpp
#include <algorithm>
#include <climits>
#include <deque>
#include <vector>

using namespace std;

class Solution {
 public:
  vector<int> simulateTaskQueue(vector<int>& submitTimes, vector<int>& execTimes,
                                int queueCapacity, int numWorkers) {
    deque<int> queue;  // FIFO，存执行时长
    int discarded = 0;
    int lastFinish = 0;
    int m = numWorkers;
    // freeAt[i]==0 空闲，否则为最早可接新任务时刻
    vector<int> freeAt(m + 1, 0);

    int n = (int)submitTimes.size();
    vector<pair<int, int>> subs(n);
    for (int i = 0; i < n; i++) subs[i] = {submitTimes[i], execTimes[i]};
    sort(subs.begin(), subs.end());  // 按提交时刻升序
    int si = 0;

    // 编号小的空闲执行者优先取队首任务
    auto assign = [&](int time) {
      vector<int> idle;
      for (int i = 1; i <= m; i++)
        if (freeAt[i] <= time) idle.push_back(i);
      sort(idle.begin(), idle.end());
      for (int wid : idle) {
        if (queue.empty()) break;
        int dur = queue.front();
        queue.pop_front();
        int finish = time + dur;
        lastFinish = max(lastFinish, finish);
        freeAt[wid] = finish;
      }
    };

    const int INF = INT_MAX / 4;
    while (true) {
      vector<int> busy;
      for (int i = 1; i <= m; i++)
        if (freeAt[i] > 0) busy.push_back(freeAt[i]);
      // 无待提交、队列空、无人在忙则结束
      if (si >= n && queue.empty() && busy.empty()) break;

      // 下一事件：提交或最早释放
      int nextSubmit = si < n ? subs[si].first : INF;
      int nextFree = busy.empty() ? INF : *min_element(busy.begin(), busy.end());
      int t = min(nextSubmit, nextFree);

      // 完成时刻已到的执行者置为空闲
      for (int i = 1; i <= m; i++)
        if (freeAt[i] > 0 && freeAt[i] <= t) freeAt[i] = 0;

      // 先分配再提交，保证同刻「释放+提交」不误丢弃
      assign(t);

      while (si < n && subs[si].first == t) {
        int d = subs[si].second;
        if ((int)queue.size() >= queueCapacity) {
          queue.pop_front();  // 丢弃最老任务
          discarded++;
        }
        queue.push_back(d);
        si++;
      }

      // 提交后再分配一次
      assign(t);
    }

    return {lastFinish, discarded};
  }
};
```
