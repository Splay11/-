## 解题思路

本题分两步：**按词条件反转**，再在结果串上**滑动统计**长度为 $4$ 的相对开音节子串。

1. **反转规则**：以空格拆分单词；若单词非空且每一位都是小写字母 `a`–`z`，则反转该单词；一旦含数字、符号等非字母，整词保持原样。
2. **相对开音节**：连续四字符 `C V C' e`，其中
   - `C`：辅音（小写字母且不在 `aeiou`）；
   - `V`：元音 `aeiou`；
   - `C'`：辅音且 **不是** `r`；
   - 末位必须是 `e`。
3. **计数**：在反转后的整串（含空格）上枚举起点 $i$，检查 $s[i..i+3]$；允许重叠（如 `cakeke` 中的 `cake` 与 `keke`）。

**常见假解**：

- 含非字母仍反转 → 样例 2 错；
- 第三位允许 `r` → 多计；
- 只在单词内计数却漏掉规则允许的跨位置重叠时逻辑不一致；
- 用 `split()` 默认吞掉连续空格，与按单空格拆分语义不一致（本题样例为空格分隔）。

## 复杂度分析

设字符串长度为 $n$（$n < 10000$）。

- 时间：$O(n)$，反转与滑动窗口各一遍；
- 空间：$O(n)$，存放处理后的字符串。

## 代码实现

### Python

```python
class Solution:
    def countOpenSyllables(self, s: str) -> int:
        vowels = set("aeiou")

        def is_consonant(ch: str) -> bool:
            # 小写字母且非元音
            return "a" <= ch <= "z" and ch not in vowels

        words = s.split(" ")
        parts = []
        for w in words:
            # 纯小写字母才反转
            if w and all("a" <= ch <= "z" for ch in w):
                parts.append(w[::-1])
            else:
                parts.append(w)
        text = " ".join(parts)

        ans = 0
        for i in range(len(text) - 3):
            a, b, c, d = text[i], text[i + 1], text[i + 2], text[i + 3]
            # C + 元音 + 非r辅音 + e
            if is_consonant(a) and b in vowels and is_consonant(c) and c != "r" and d == "e":
                ans += 1
        return ans
```

### Java

```java
public class Solution {
    public int countOpenSyllables(String s) {
        StringBuilder text = new StringBuilder();
        // -1 保留末尾空段，与按单空格拆分一致
        String[] words = s.split(" ", -1);
        for (int wi = 0; wi < words.length; wi++) {
            if (wi > 0) text.append(' ');
            String w = words[wi];
            boolean pure = w.length() > 0;
            for (int i = 0; i < w.length(); i++) {
                char ch = w.charAt(i);
                if (ch < 'a' || ch > 'z') {
                    pure = false;
                    break;
                }
            }
            if (pure) {
                text.append(new StringBuilder(w).reverse());
            } else {
                text.append(w);
            }
        }

        int ans = 0;
        String t = text.toString();
        for (int i = 0; i + 3 < t.length(); i++) {
            char a = t.charAt(i), b = t.charAt(i + 1), c = t.charAt(i + 2), d = t.charAt(i + 3);
            if (isConsonant(a) && isVowel(b) && isConsonant(c) && c != 'r' && d == 'e') {
                ans++;
            }
        }
        return ans;
    }

    private boolean isVowel(char ch) {
        return ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u';
    }

    private boolean isConsonant(char ch) {
        return ch >= 'a' && ch <= 'z' && !isVowel(ch);
    }
}
```

### C++

```cpp
#include <string>

using namespace std;

class Solution {
 public:
  int countOpenSyllables(string s) {
    auto isVowel = [](char ch) {
      return ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u';
    };
    auto isConsonant = [&](char ch) {
      return ch >= 'a' && ch <= 'z' && !isVowel(ch);
    };

    string text;
    string word;
    bool first = true;
    auto flush = [&]() {
      if (!first) text.push_back(' ');
      first = false;
      bool pure = !word.empty();
      for (char ch : word) {
        if (ch < 'a' || ch > 'z') {
          pure = false;
          break;
        }
      }
      // 纯小写字母则反转写入
      if (pure) {
        for (int i = (int)word.size() - 1; i >= 0; --i) text.push_back(word[i]);
      } else {
        text += word;
      }
      word.clear();
    };

    for (char ch : s) {
      if (ch == ' ')
        flush();
      else
        word.push_back(ch);
    }
    flush();

    int ans = 0;
    for (int i = 0; i + 3 < (int)text.size(); ++i) {
      char a = text[i], b = text[i + 1], c = text[i + 2], d = text[i + 3];
      if (isConsonant(a) && isVowel(b) && isConsonant(c) && c != 'r' && d == 'e') ans++;
    }
    return ans;
  }
};
```
